package com.lps.cms.embossing.perso.event.repository;

import java.util.UUID;

import jakarta.persistence.EntityManager;
import jakarta.persistence.PersistenceContext;
import org.springframework.stereotype.Repository;

@Repository
public class OutboxEventRepositoryImpl implements OutboxEventRepository {

    private static final String INSERT_IF_ABSENT = """
            INSERT INTO cms_v2.outbox_event (
                outbox_event_id, institution_id, aggregate_type, aggregate_id, event_type,
                event_version, aggregate_sequence, event_payload, correlation_id, idempotency_key,
                publication_status, occurred_at, retry_count)
            VALUES (
                :id, :institutionId, :aggregateType, :aggregateId, :eventType,
                1, :aggregateSequence, CAST(:payload AS jsonb), :correlationId, :idempotencyKey,
                'PENDING', now(), 0)
            ON CONFLICT (institution_id, idempotency_key) DO NOTHING
            """;

    private static final String NEXT_AGGREGATE_SEQUENCE = """
            INSERT INTO cms_v2.outbox_aggregate_sequence (
                institution_id, aggregate_type, aggregate_id, next_sequence
            )
            VALUES (:institutionId, :aggregateType, :aggregateId, 2)
            ON CONFLICT (institution_id, aggregate_type, aggregate_id)
            DO UPDATE SET next_sequence = cms_v2.outbox_aggregate_sequence.next_sequence + 1
            RETURNING next_sequence - 1
            """;

    @PersistenceContext
    private EntityManager entityManager;

    @Override
    public int insertIfAbsent(
            UUID id,
            UUID institutionId,
            String aggregateType,
            String aggregateId,
            String eventType,
            long aggregateSequence,
            String payload,
            UUID correlationId,
            UUID idempotencyKey) {
        return entityManager.createNativeQuery(INSERT_IF_ABSENT)
                .setParameter("id", id)
                .setParameter("institutionId", institutionId)
                .setParameter("aggregateType", aggregateType)
                .setParameter("aggregateId", aggregateId)
                .setParameter("eventType", eventType)
                .setParameter("aggregateSequence", aggregateSequence)
                .setParameter("payload", payload)
                .setParameter("correlationId", correlationId)
                .setParameter("idempotencyKey", idempotencyKey)
                .executeUpdate();
    }

    @Override
    public long nextAggregateSequence(UUID institutionId, String aggregateType, String aggregateId) {
        Number sequence = (Number) entityManager.createNativeQuery(NEXT_AGGREGATE_SEQUENCE)
                .setParameter("institutionId", institutionId)
                .setParameter("aggregateType", aggregateType)
                .setParameter("aggregateId", aggregateId)
                .getSingleResult();
        return sequence.longValue();
    }
}
