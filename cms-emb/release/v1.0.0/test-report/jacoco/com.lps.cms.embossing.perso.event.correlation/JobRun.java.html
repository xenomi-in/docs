package com.lps.cms.embossing.perso.event.correlation;

import java.util.UUID;
import java.util.concurrent.ThreadLocalRandom;

/** Correlation id for one embossing job execution. */
public final class JobRun {

    private static final ThreadLocal<String> CURRENT = new ThreadLocal<>();

    private JobRun() {
    }

    public static String start() {
        String id = uuidV7().toString();
        CURRENT.set(id);
        return id;
    }

    public static String peek() {
        return CURRENT.get();
    }

    public static String current() {
        String id = CURRENT.get();
        return id != null ? id : start();
    }

    public static void clear() {
        CURRENT.remove();
    }

    private static UUID uuidV7() {
        long timestamp = System.currentTimeMillis() & 0xFFFFFFFFFFFFL;
        long randomA = ThreadLocalRandom.current().nextLong() & 0xFFFL;
        long msb = (timestamp << 16) | 0x7000L | randomA;
        long lsb = (ThreadLocalRandom.current().nextLong() & 0x3FFFFFFFFFFFFFFFL) | 0x8000000000000000L;
        return new UUID(msb, lsb);
    }
}
