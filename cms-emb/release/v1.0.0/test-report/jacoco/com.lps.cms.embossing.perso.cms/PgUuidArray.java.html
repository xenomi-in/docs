package com.lps.cms.embossing.perso.cms;

import java.sql.Array;
import java.util.Collection;
import java.util.UUID;

import org.springframework.jdbc.core.JdbcTemplate;

/** Builds a PostgreSQL {@code uuid[]} on the current transactional connection. */
public final class PgUuidArray {

    private PgUuidArray() {
    }

    public static Array of(JdbcTemplate jdbc, Collection<UUID> ids) {
        return jdbc.execute((java.sql.Connection connection) ->
                connection.createArrayOf("uuid", ids.toArray(UUID[]::new)));
    }
}
