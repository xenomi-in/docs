package com.lps.cms.embossing.perso.file.data;

import java.time.format.DateTimeFormatter;

import com.lps.cms.embossing.perso.cms.card.CardFileLine;

/**
 * Test stand-in for the cms-core secure-data export. It builds a PAN from the card's last four
 * digits and fixed CVVs. Enable it with {@code embossing.v2.fixture-card-data=true}.
 */
public class FixtureSensitiveCardDataProvider implements SensitiveCardDataProvider {

    private static final DateTimeFormatter YYMM = DateTimeFormatter.ofPattern("yyMM");

    @Override
    public SensitiveCardData resolve(CardFileLine card) {
        String pan = "411111111111" + lastFour(card.panLastFour());
        String expiry = card.expiryDate() == null ? "0000" : card.expiryDate().format(YYMM);
        return new SensitiveCardData(pan, "123", "456", pan + "=" + expiry + "101");
    }

    private static String lastFour(String panLastFour) {
        String digits = panLastFour == null ? "" : panLastFour.replaceAll("\\D", "");
        if (digits.length() > 4) {
            digits = digits.substring(digits.length() - 4);
        }
        return "0".repeat(4 - digits.length()) + digits;
    }
}
