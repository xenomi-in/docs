package com.lps.cms.embossing.perso.file.data;

import com.lps.cms.embossing.perso.cms.card.CardFileLine;
import com.lps.cms.embossing.perso.exception.EmbossingErrorCodes;
import com.lps.cms.embossing.perso.exception.FileGenerationException;

/** Rejects file generation when no secure-data export provider is configured. */
public class SecureDataExportRequiredProvider implements SensitiveCardDataProvider {

    @Override
    public SensitiveCardData resolve(CardFileLine card) {
        throw new FileGenerationException(EmbossingErrorCodes.SENSITIVE_DATA_PROVIDER_MISSING,
                "No v2 sensitive-card-data provider is configured for card="
                + card.cardId() + ". cms-core must supply a scoped secure-data export; the vendor service "
                + "will not decrypt CMS2 envelopes or load core encryption keys.");
    }
}
