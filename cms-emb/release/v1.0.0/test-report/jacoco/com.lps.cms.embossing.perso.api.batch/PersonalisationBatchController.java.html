package com.lps.cms.embossing.perso.api.batch;

import java.util.UUID;

import com.lps.cms.embossing.perso.api.ApiResponseEnvelope;
import com.lps.cms.embossing.perso.api.ApiResponseHelper;
import com.lps.cms.embossing.perso.api.PageableDefaults;
import com.lps.cms.embossing.perso.api.PagedResponse;
import org.springframework.data.domain.Pageable;
import org.springframework.http.ResponseEntity;
import org.springframework.validation.annotation.Validated;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.bind.annotation.RestController;

@Validated
@RestController
@RequestMapping("/cms/api/v2/institutions/{institutionId}/personalisation-batches")
public class PersonalisationBatchController {

    private final PersonalisationQueryService service;

    public PersonalisationBatchController(PersonalisationQueryService service) {
        this.service = service;
    }

    @GetMapping
    public ResponseEntity<ApiResponseEnvelope<PagedResponse<PersonalisationBatchResponse>>> list(
            @PathVariable UUID institutionId,
            @RequestParam(required = false) String status,
            @RequestParam(required = false) Integer page,
            @RequestParam(required = false) Integer limit) {
        Pageable pageable = PageableDefaults.ofOneBased(page, limit);
        return ApiResponseHelper.ok(service.listBatches(institutionId, status, pageable));
    }

    @GetMapping("/{batchId}")
    public ResponseEntity<ApiResponseEnvelope<PersonalisationBatchResponse>> get(
            @PathVariable UUID institutionId,
            @PathVariable UUID batchId) {
        return ApiResponseHelper.ok(service.getBatch(institutionId, batchId));
    }
}
