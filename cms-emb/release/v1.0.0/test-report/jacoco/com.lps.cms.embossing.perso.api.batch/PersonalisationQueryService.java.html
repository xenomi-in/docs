package com.lps.cms.embossing.perso.api.batch;

import java.util.List;
import java.util.UUID;

import com.lps.cms.embossing.perso.api.PagedResponse;
import com.lps.cms.embossing.perso.api.file.PersonalisationFileResponse;
import com.lps.cms.embossing.perso.cms.batch.PersonalisationReadStore;
import com.lps.cms.embossing.perso.cms.batch.PersonalisationReadStore.BatchRow;
import com.lps.cms.embossing.perso.cms.batch.PersonalisationReadStore.FileRow;
import com.lps.cms.embossing.perso.exception.EmbossingErrorCodes;
import com.lps.cms.embossing.perso.exception.EmbossingRequestException;
import com.lps.cms.embossing.perso.exception.ResourceNotFoundException;
import org.springframework.data.domain.Pageable;
import org.springframework.stereotype.Service;

@Service
public class PersonalisationQueryService {

    private static final String STATUS_TOKEN = "[A-Z0-9_]{1,40}";

    private final PersonalisationReadStore batches;

    public PersonalisationQueryService(PersonalisationReadStore batches) {
        this.batches = batches;
    }

    public PagedResponse<PersonalisationBatchResponse> listBatches(UUID institutionId, String status, Pageable pageable) {
        String resolvedStatus = status(status);
        List<PersonalisationBatchResponse> content = batches.listBatches(
                        institutionId, resolvedStatus, pageable.getPageSize(), pageable.getOffset())
                .stream()
                .map(PersonalisationQueryService::batch)
                .toList();
        return PagedResponse.of(content, pageable.getPageNumber(), pageable.getPageSize(),
                batches.countBatches(institutionId, resolvedStatus));
    }

    public PersonalisationBatchResponse getBatch(UUID institutionId, UUID batchId) {
        return batches.findBatch(institutionId, batchId)
                .map(PersonalisationQueryService::batch)
                .orElseThrow(() -> new ResourceNotFoundException("Personalisation batch was not found"));
    }

    public List<PersonalisationFileResponse> listFiles(UUID institutionId, UUID batchId) {
        getBatch(institutionId, batchId);
        return batches.listFiles(institutionId, batchId).stream()
                .map(PersonalisationQueryService::file)
                .toList();
    }

    public PersonalisationFileResponse getFile(UUID institutionId, UUID fileId) {
        return batches.findFile(institutionId, fileId)
                .map(PersonalisationQueryService::file)
                .orElseThrow(() -> new ResourceNotFoundException("Personalisation file was not found"));
    }

    private static String status(String status) {
        if (status == null || status.isBlank()) {
            return null;
        }
        if (!status.matches(STATUS_TOKEN)) {
            throw new EmbossingRequestException(EmbossingErrorCodes.REQUEST_INVALID, "batch status is not valid");
        }
        return status;
    }

    private static PersonalisationBatchResponse batch(BatchRow row) {
        return new PersonalisationBatchResponse(
                row.batchId(), row.institutionId(), row.vendorId(), row.configurationVersionId(),
                row.batchReference(), row.batchStatus(), row.itemCount(), row.createdAt(), row.updatedAt());
    }

    private static PersonalisationFileResponse file(FileRow row) {
        return new PersonalisationFileResponse(
                row.fileId(), row.institutionId(), row.batchId(), row.fileSequence(), row.fileName(),
                row.schemaVersion(), row.recordCount(), row.fileStatus(), row.generatedAt());
    }
}
