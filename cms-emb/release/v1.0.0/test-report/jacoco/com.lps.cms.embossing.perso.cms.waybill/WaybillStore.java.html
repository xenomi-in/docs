package com.lps.cms.embossing.perso.cms.waybill;

import java.util.List;
import java.util.UUID;

import com.lps.cms.embossing.perso.cms.PgUuidArray;
import com.lps.cms.embossing.perso.exception.EmbossingErrorCodes;
import com.lps.cms.embossing.perso.exception.FulfilmentConflictException;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.stereotype.Repository;

/** Locks available waybills for one institution, then reserves or assigns them. */
@Repository
public class WaybillStore {

    private final JdbcTemplate jdbc;

    public WaybillStore(JdbcTemplate jdbc) {
        this.jdbc = jdbc;
    }

    public List<Waybill> lockAvailable(UUID institutionId, int limit) {
        if (limit <= 0) {
            return List.of();
        }
        return jdbc.query("""
                SELECT courier_waybill_id, courier_provider_id, waybill_number
                FROM cms_v2.courier_waybill
                WHERE institution_id = ? AND waybill_status = 'AVAILABLE'
                ORDER BY created_at
                FOR UPDATE SKIP LOCKED
                LIMIT ?
                """, (rs, row) -> new Waybill(
                rs.getObject("courier_waybill_id", UUID.class),
                rs.getObject("courier_provider_id", UUID.class),
                rs.getString("waybill_number")), institutionId, limit);
    }

    public void reserve(List<UUID> waybillIds) {
        if (waybillIds.isEmpty()) {
            return;
        }
        int updated = jdbc.update("""
                UPDATE cms_v2.courier_waybill
                SET waybill_status = 'RESERVED', reserved_at = now()
                WHERE courier_waybill_id = ANY(?) AND waybill_status = 'AVAILABLE'
                """, PgUuidArray.of(jdbc, waybillIds));
        if (updated != waybillIds.size()) {
            throw new FulfilmentConflictException(EmbossingErrorCodes.WAYBILL_RESERVATION_LOST,
                    "Waybill reservation lost a locked row");
        }
    }

    public void assign(List<UUID> waybillIds) {
        if (waybillIds.isEmpty()) {
            return;
        }
        int updated = jdbc.update("""
                UPDATE cms_v2.courier_waybill
                SET waybill_status = 'ASSIGNED', assigned_at = now()
                WHERE courier_waybill_id = ANY(?) AND waybill_status = 'RESERVED'
                """, PgUuidArray.of(jdbc, waybillIds));
        if (updated != waybillIds.size()) {
            throw new FulfilmentConflictException(EmbossingErrorCodes.WAYBILL_ASSIGNMENT_LOST,
                    "Waybill assignment lost a reserved row");
        }
    }
}
