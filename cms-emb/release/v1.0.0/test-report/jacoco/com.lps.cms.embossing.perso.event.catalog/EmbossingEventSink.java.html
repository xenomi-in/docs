package com.lps.cms.embossing.perso.event.catalog;

import java.time.Instant;
import java.util.List;
import java.util.Map;
import java.util.UUID;

import com.lps.cms.embossing.perso.event.mapper.PayloadValues;
import com.lps.cms.embossing.perso.event.model.EventTypes;
import com.lps.cms.embossing.perso.event.model.IdempotencyKeys;
import com.lps.cms.embossing.perso.event.model.NotificationPriority;
import com.lps.cms.embossing.perso.event.model.OutboxEventCommand;
import com.lps.cms.embossing.perso.event.writer.OutboxEventWriter;
import org.springframework.stereotype.Component;

/**
 * Embossing lifecycle notifications. The envelope {@code customerId} stays null.
 * File events join the batch through {@code payload.batchId}.
 */
@Component
public class EmbossingEventSink {

    private final OutboxEventWriter writer;

    public EmbossingEventSink(OutboxEventWriter writer) {
        this.writer = writer;
    }

    public void batchCreated(UUID institutionId, UUID batchId, String batchReference, String productId,
                             UUID vendorId, int cardCount, List<CardRef> cards) {
        if (institutionId == null || batchId == null) {
            return;
        }
        List<Map<String, Object>> cardPayload = cards == null ? List.of() : cards.stream()
                .map(CardRef::toMap)
                .toList();
        Map<String, Object> payload = batchPayload(batchId, batchReference, productId, vendorId, institutionId, cardCount);
        payload.put("cards", cardPayload);
        enqueue(institutionId, EventTypes.CARD_PRODUCTION_BATCH_CREATED, EventTypes.AGGREGATE_PRODUCTION_BATCH,
                batchId, NotificationPriority.MEDIUM, payload);
    }

    public void batchFailed(UUID institutionId, UUID batchId, String batchReference, String productId,
                            UUID vendorId, int cardCount) {
        if (institutionId == null || batchId == null) {
            return;
        }
        Map<String, Object> payload = batchPayload(batchId, batchReference, productId, vendorId, institutionId, cardCount);
        payload.put("failureStage", "FILE_GENERATION");
        payload.put("occurredAt", PayloadValues.instant(Instant.now()));
        enqueue(institutionId, EventTypes.CARD_PRODUCTION_BATCH_FAILED, EventTypes.AGGREGATE_PRODUCTION_BATCH,
                batchId, NotificationPriority.HIGH, payload);
    }

    public void fileGenerated(UUID institutionId, UUID fileId, UUID batchId, UUID vendorId,
                              int cardCount, String generatedAt) {
        fileEvent(institutionId, fileId, batchId, vendorId, cardCount, EventTypes.EMBOSSING_FILE_GENERATED,
                NotificationPriority.MEDIUM, "generatedAt", generatedAt, null);
    }

    public void fileTransmitted(UUID institutionId, UUID fileId, UUID batchId, UUID vendorId,
                                int cardCount, String transmittedAt) {
        fileEvent(institutionId, fileId, batchId, vendorId, cardCount, EventTypes.EMBOSSING_FILE_TRANSMITTED,
                NotificationPriority.MEDIUM, "transmittedAt", transmittedAt, null);
    }

    public void fileAcknowledged(UUID institutionId, UUID fileId, UUID batchId, UUID vendorId,
                                 int cardCount, String acknowledgedAt) {
        fileEvent(institutionId, fileId, batchId, vendorId, cardCount, EventTypes.EMBOSSING_FILE_ACKNOWLEDGED,
                NotificationPriority.MEDIUM, "acknowledgedAt", acknowledgedAt, null);
    }

    public void transmissionFailed(UUID institutionId, UUID fileId, UUID batchId, UUID vendorId,
                                   int cardCount, String errorCategory) {
        fileEvent(institutionId, fileId, batchId, vendorId, cardCount, EventTypes.EMBOSSING_FILE_TRANSMISSION_FAILED,
                NotificationPriority.HIGH, "occurredAt", PayloadValues.instant(Instant.now()), errorCategory);
    }

    private void fileEvent(UUID institutionId, UUID fileId, UUID batchId, UUID vendorId, int cardCount,
                           String eventType, NotificationPriority priority, String timestampKey, String timestamp,
                           String errorCategory) {
        if (institutionId == null || fileId == null) {
            return;
        }
        Map<String, Object> payload = PayloadValues.entries(
                "fileId", PayloadValues.uuid(fileId),
                "batchId", PayloadValues.uuid(batchId),
                "vendorId", PayloadValues.uuid(vendorId),
                "institutionId", PayloadValues.uuid(institutionId),
                "cardCount", cardCount,
                "fileFormat", "EMV");
        payload.put(timestampKey, timestamp);
        if (errorCategory != null) {
            payload.put("failureStage", "TRANSMISSION");
            payload.put("errorCategory", errorCategory);
        }
        enqueue(institutionId, eventType, EventTypes.AGGREGATE_EMBOSSING_FILE, fileId, priority, payload);
    }

    private void enqueue(UUID institutionId, String eventType, String aggregateType, UUID aggregateId,
                         NotificationPriority priority, Map<String, Object> payload) {
        writer.enqueue(OutboxEventCommand.builder()
                .institutionId(institutionId)
                .eventType(eventType)
                .aggregateType(aggregateType)
                .aggregateId(aggregateId)
                .priority(priority)
                .idempotencyKey(IdempotencyKeys.of(institutionId, eventType, aggregateId))
                .payload(payload)
                .build());
    }

    private static Map<String, Object> batchPayload(UUID batchId, String batchReference, String productId,
                                                    UUID vendorId, UUID institutionId, int cardCount) {
        return PayloadValues.entries(
                "batchId", PayloadValues.uuid(batchId),
                "batchReference", batchReference,
                "batchType", "BULK",
                "productId", productId,
                "vendorId", PayloadValues.uuid(vendorId),
                "institutionId", PayloadValues.uuid(institutionId),
                "cardCount", cardCount);
    }

    public record CardRef(UUID cardId, String customerId, String lastFour) {
        private Map<String, Object> toMap() {
            return PayloadValues.entries(
                    "cardId", PayloadValues.uuid(cardId),
                    "customerId", customerId,
                    "lastFour", lastFour);
        }
    }
}
