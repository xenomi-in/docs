package com.lps.cms.embossing.perso.report;

import java.util.LinkedHashMap;
import java.util.Map;
import java.util.Set;
import java.util.UUID;

import com.lps.cms.embossing.perso.api.report.CourierEventRequest;
import com.lps.cms.embossing.perso.api.report.VendorResultRequest;
import com.lps.cms.embossing.perso.event.mapper.PayloadValues;
import com.lps.cms.embossing.perso.exception.EmbossingErrorCodes;
import com.lps.cms.embossing.perso.exception.EmbossingRequestException;
import com.lps.cms.embossing.perso.exception.FulfilmentConflictException;

/** Maps one normalized provider report onto the CMS command and the production-request status. */
final class FulfilmentCommandPlan {

    private FulfilmentCommandPlan() {
    }

    static Decision vendor(VendorResultRequest report, ReportSubject subject, Set<String> priorCommands) {
        requireSame(report.institutionId(), subject.institutionId(), "Institution does not match the request path");
        requireSame(report.cardId(), subject.cardId(), "Card does not match the production request");
        requireSame(report.vendorId(), subject.vendorId(), "Vendor does not match the production request");
        return switch (report.result()) {
            case ACCEPTED -> accepted(subject, report.vendorReference());
            case PERSONALIZED -> personalized(subject, priorCommands, report);
            case FILE_COMMITTED -> fileCommitted(subject, priorCommands, report.fileReference());
            case FAILED -> failed(subject, report);
        };
    }

    static Decision courier(CourierEventRequest report, ReportSubject subject, Set<String> priorCommands) {
        requireSame(report.institutionId(), subject.institutionId(), "Institution does not match the request path");
        requireSame(report.cardId(), subject.cardId(), "Card does not match the production request");
        if (subject.courierProviderId() != null && !subject.courierProviderId().equals(report.courierProviderId())) {
            throw new EmbossingRequestException(EmbossingErrorCodes.REQUEST_INVALID,
                    "Courier does not match the waybill on the production request");
        }
        return switch (report.eventType()) {
            case DISPATCHED -> dispatched(subject, priorCommands, report);
            case DELIVERED -> delivered(subject, priorCommands, report);
            case RETURNED -> returned(subject, priorCommands, report);
            case REDISPATCHED -> redispatch(subject, priorCommands, report);
        };
    }

    private static Decision accepted(ReportSubject subject, String vendorReference) {
        requireStatus(subject, Set.of("FILE_GENERATED", "SENT_TO_VENDOR"), FulfilmentCommands.START_PERSONALIZATION);
        return new Decision(FulfilmentCommands.START_PERSONALIZATION, "VENDOR_ACCEPTED",
                evidence("vendorReference", vendorReference));
    }

    private static Decision personalized(ReportSubject subject, Set<String> priorCommands, VendorResultRequest report) {
        requirePrior(priorCommands, FulfilmentCommands.START_PERSONALIZATION, FulfilmentCommands.CONFIRM_PERSONALIZATION);
        requireStatus(subject, Set.of("VENDOR_ACCEPTED"), FulfilmentCommands.CONFIRM_PERSONALIZATION);
        String sequence = report.verifiedCardSequenceNumber();
        if (sequence == null || !sequence.matches("[0-9]{3}")) {
            throw new EmbossingRequestException(EmbossingErrorCodes.REQUEST_INVALID,
                    "Verified card sequence must be three digits");
        }
        Map<String, Object> evidence = evidence(
                "verifiedCardSequenceNumber", sequence,
                "vendorReference", report.vendorReference());
        return new Decision(FulfilmentCommands.CONFIRM_PERSONALIZATION, "IN_PRODUCTION", evidence);
    }

    private static Decision fileCommitted(ReportSubject subject, Set<String> priorCommands, String fileReference) {
        requirePrior(priorCommands, FulfilmentCommands.CONFIRM_PERSONALIZATION, FulfilmentCommands.CONFIRM_BUREAU_FILE);
        requireStatus(subject, Set.of("IN_PRODUCTION"), FulfilmentCommands.CONFIRM_BUREAU_FILE);
        if (priorCommands.contains(FulfilmentCommands.CONFIRM_BUREAU_FILE)) {
            throw new FulfilmentConflictException(EmbossingErrorCodes.REQUEST_STATUS_CHANGED,
                    "A bureau file was already reported for this production request");
        }
        if (subject.personalisationFileId() == null) {
            throw new FulfilmentConflictException(EmbossingErrorCodes.REQUEST_STATUS_CHANGED,
                    "Personalisation file is not available for this production request");
        }
        if (fileReference != null && !fileReference.isBlank()
                && !fileReference.equals(subject.personalisationFileId().toString())) {
            throw new EmbossingRequestException(EmbossingErrorCodes.REQUEST_INVALID,
                    "File reference does not match the personalisation file");
        }
        return new Decision(FulfilmentCommands.CONFIRM_BUREAU_FILE, null, evidence(
                "personalisationFileId", PayloadValues.uuid(subject.personalisationFileId()),
                "fileName", subject.fileName()));
    }

    private static Decision failed(ReportSubject subject, VendorResultRequest report) {
        requireStatus(subject, Set.of("FILE_GENERATED", "SENT_TO_VENDOR", "VENDOR_ACCEPTED"),
                FulfilmentCommands.FAIL_PERSONALIZATION);
        if (report.failureCode() == null || !report.failureCode().matches("[A-Z0-9_]{1,50}")) {
            throw new EmbossingRequestException(EmbossingErrorCodes.REQUEST_INVALID,
                    "Failure code must contain only uppercase letters, digits, and underscores");
        }
        return new Decision(FulfilmentCommands.FAIL_PERSONALIZATION, "VENDOR_REJECTED", evidence(
                "failureCode", report.failureCode(),
                "failureMessage", blankToNull(report.failureMessage()),
                "vendorReference", report.vendorReference()));
    }

    private static Decision dispatched(ReportSubject subject, Set<String> priorCommands, CourierEventRequest report) {
        requirePrior(priorCommands, FulfilmentCommands.CONFIRM_BUREAU_FILE, FulfilmentCommands.CONFIRM_DISPATCH);
        requireStatus(subject, Set.of("IN_PRODUCTION"), FulfilmentCommands.CONFIRM_DISPATCH);
        requireAwb(subject, report.awbNumber());
        return new Decision(FulfilmentCommands.CONFIRM_DISPATCH, null, courierEvidence(report));
    }

    private static Decision delivered(ReportSubject subject, Set<String> priorCommands, CourierEventRequest report) {
        requirePrior(priorCommands, FulfilmentCommands.CONFIRM_DISPATCH, FulfilmentCommands.CONFIRM_DELIVERY);
        if (priorCommands.contains(FulfilmentCommands.CONFIRM_RETURN)) {
            throw new FulfilmentConflictException(EmbossingErrorCodes.REQUEST_STATUS_CHANGED,
                    "Production request is not ready for " + FulfilmentCommands.CONFIRM_DELIVERY);
        }
        requireStatus(subject, Set.of("IN_PRODUCTION"), FulfilmentCommands.CONFIRM_DELIVERY);
        requireAwb(subject, report.awbNumber());
        return new Decision(FulfilmentCommands.CONFIRM_DELIVERY, "COMPLETED", courierEvidence(report));
    }

    private static Decision returned(ReportSubject subject, Set<String> priorCommands, CourierEventRequest report) {
        requirePrior(priorCommands, FulfilmentCommands.CONFIRM_DISPATCH, FulfilmentCommands.CONFIRM_RETURN);
        if (priorCommands.contains(FulfilmentCommands.CONFIRM_DELIVERY)) {
            throw new FulfilmentConflictException(EmbossingErrorCodes.REQUEST_STATUS_CHANGED,
                    "A delivered shipment cannot be reported as returned");
        }
        requireStatus(subject, Set.of("IN_PRODUCTION"), FulfilmentCommands.CONFIRM_RETURN);
        requireAwb(subject, report.awbNumber());
        return new Decision(FulfilmentCommands.CONFIRM_RETURN, null, courierEvidence(report));
    }

    private static Decision redispatch(ReportSubject subject, Set<String> priorCommands, CourierEventRequest report) {
        requirePrior(priorCommands, FulfilmentCommands.CONFIRM_RETURN, FulfilmentCommands.REDISPATCH_SAME_PLASTIC);
        requireStatus(subject, Set.of("IN_PRODUCTION"), FulfilmentCommands.REDISPATCH_SAME_PLASTIC);
        if (subject.waybillNumber() != null && subject.waybillNumber().equals(report.awbNumber())) {
            throw new EmbossingRequestException(EmbossingErrorCodes.REQUEST_INVALID,
                    "Redispatch requires a new airway bill");
        }
        return new Decision(FulfilmentCommands.REDISPATCH_SAME_PLASTIC, null, courierEvidence(report));
    }

    private static void requireAwb(ReportSubject subject, String awbNumber) {
        if (subject.waybillNumber() != null && !subject.waybillNumber().equals(awbNumber)) {
            throw new EmbossingRequestException(EmbossingErrorCodes.REQUEST_INVALID,
                    "Airway bill does not match the waybill on the production request");
        }
    }

    private static void requireStatus(ReportSubject subject, Set<String> allowed, String command) {
        if (!allowed.contains(subject.requestStatus())) {
            throw new FulfilmentConflictException(EmbossingErrorCodes.REQUEST_STATUS_CHANGED,
                    "Production request is not ready for " + command);
        }
    }

    private static void requirePrior(Set<String> priorCommands, String required, String command) {
        if (!priorCommands.contains(required)) {
            throw new FulfilmentConflictException(EmbossingErrorCodes.REQUEST_STATUS_CHANGED,
                    "Production request is not ready for " + command);
        }
    }

    private static void requireSame(UUID reported, UUID actual, String message) {
        if (!reported.equals(actual)) {
            throw new EmbossingRequestException(EmbossingErrorCodes.REQUEST_INVALID, message);
        }
    }

    private static Map<String, Object> courierEvidence(CourierEventRequest report) {
        return evidence(
                "courierProviderId", PayloadValues.uuid(report.courierProviderId()),
                "shipmentId", PayloadValues.uuid(report.shipmentId()),
                "awbNumber", report.awbNumber(),
                "shipmentAttempt", report.shipmentAttempt(),
                "locationReference", blankToNull(report.locationReference()),
                "evidenceReference", blankToNull(report.evidenceReference()));
    }

    private static Map<String, Object> evidence(Object... keyValues) {
        Map<String, Object> evidence = new LinkedHashMap<>();
        for (int i = 0; i < keyValues.length; i += 2) {
            evidence.put((String) keyValues[i], keyValues[i + 1]);
        }
        return evidence;
    }

    private static String blankToNull(String value) {
        return value == null || value.isBlank() ? null : value;
    }

    record Decision(String command, String nextRequestStatus, Map<String, Object> evidence) {
    }
}
