package com.lps.cms.embossing.perso.report;

import java.util.List;
import java.util.Optional;
import java.util.Set;
import java.util.UUID;

import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.stereotype.Repository;

/** Reads the production request for a provider report and stores the deduplicated inbox row. */
@Repository
public class FulfilmentReportStore {

    private final JdbcTemplate jdbc;

    public FulfilmentReportStore(JdbcTemplate jdbc) {
        this.jdbc = jdbc;
    }

    public Optional<ReportSubject> findSubject(UUID institutionId, UUID productionRequestId) {
        List<ReportSubject> rows = jdbc.query("""
                SELECT r.institution_id, r.production_request_id, r.card_id, r.vendor_id, r.request_status,
                       item.personalisation_file_id, item.file_name, item.waybill_number_snapshot,
                       item.courier_provider_id
                FROM cms_v2.card_production_request r
                LEFT JOIN LATERAL (
                    SELECT f.personalisation_file_id, f.file_name, i.waybill_number_snapshot, w.courier_provider_id
                    FROM cms_v2.personalisation_batch_item i
                    LEFT JOIN cms_v2.courier_waybill w
                      ON w.institution_id = i.institution_id AND w.courier_waybill_id = i.courier_waybill_id
                    LEFT JOIN cms_v2.personalisation_file f
                      ON f.institution_id = i.institution_id AND f.batch_id = i.batch_id
                    WHERE i.institution_id = r.institution_id
                      AND i.production_request_id = r.production_request_id
                    ORDER BY i.added_at DESC
                    LIMIT 1
                ) item ON true
                WHERE r.institution_id = ? AND r.production_request_id = ?
                """, (rs, row) -> new ReportSubject(
                rs.getObject("institution_id", UUID.class),
                rs.getObject("production_request_id", UUID.class),
                rs.getObject("card_id", UUID.class),
                rs.getObject("vendor_id", UUID.class),
                rs.getString("request_status"),
                rs.getObject("personalisation_file_id", UUID.class),
                rs.getString("file_name"),
                rs.getString("waybill_number_snapshot"),
                rs.getObject("courier_provider_id", UUID.class)), institutionId, productionRequestId);
        return rows.stream().findFirst();
    }

    public Set<String> priorCommands(UUID institutionId, UUID productionRequestId, String sourceEventId) {
        List<String> commands = jdbc.query("""
                SELECT command_name
                FROM cms_v2.provider_inbox_event
                WHERE institution_id = ? AND production_request_id = ? AND source_event_id <> ?
                """, (rs, row) -> rs.getString("command_name"), institutionId, productionRequestId, sourceEventId);
        return Set.copyOf(commands);
    }

    public Optional<UUID> insert(InboxDraft draft) {
        List<UUID> inserted = jdbc.query("""
                INSERT INTO cms_v2.provider_inbox_event (
                    inbox_event_id, source, source_event_id, institution_id, event_type, command_name,
                    production_request_id, card_id, payload, payload_sha256, processing_status,
                    received_at, processed_at)
                VALUES (uuidv7(), ?, ?, ?, ?, ?, ?, ?, CAST(? AS jsonb), ?, 'PROCESSED', now(), now())
                ON CONFLICT (source, source_event_id) DO NOTHING
                RETURNING inbox_event_id
                """, (rs, row) -> rs.getObject("inbox_event_id", UUID.class),
                draft.source(), draft.sourceEventId(), draft.institutionId(), draft.eventType(),
                draft.commandName(), draft.productionRequestId(), draft.cardId(), draft.payloadJson(),
                draft.payloadSha256());
        return inserted.stream().findFirst();
    }

    public Optional<InboxEvent> findInbox(String source, String sourceEventId) {
        List<InboxEvent> rows = jdbc.query("""
                SELECT inbox_event_id, institution_id, event_type, command_name,
                       production_request_id, card_id, payload_sha256
                FROM cms_v2.provider_inbox_event
                WHERE source = ? AND source_event_id = ?
                """, (rs, row) -> new InboxEvent(
                rs.getObject("inbox_event_id", UUID.class),
                rs.getObject("institution_id", UUID.class),
                rs.getString("event_type"),
                rs.getString("command_name"),
                rs.getObject("production_request_id", UUID.class),
                rs.getObject("card_id", UUID.class),
                rs.getString("payload_sha256")), source, sourceEventId);
        return rows.stream().findFirst();
    }

    public int advanceStatus(UUID institutionId, UUID productionRequestId, String currentStatus, String nextStatus) {
        return jdbc.update("""
                UPDATE cms_v2.card_production_request
                SET request_status = ?, updated_at = now()
                WHERE institution_id = ? AND production_request_id = ? AND request_status = ?
                """, nextStatus, institutionId, productionRequestId, currentStatus);
    }

    public record InboxDraft(
            String source,
            String sourceEventId,
            UUID institutionId,
            String eventType,
            String commandName,
            UUID productionRequestId,
            UUID cardId,
            String payloadJson,
            String payloadSha256) {
    }
}
