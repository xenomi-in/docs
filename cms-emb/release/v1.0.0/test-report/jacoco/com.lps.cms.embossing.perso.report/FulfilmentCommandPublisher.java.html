package com.lps.cms.embossing.perso.report;

import java.util.Map;
import java.util.UUID;

import com.lps.cms.embossing.perso.event.model.IdempotencyKeys;
import com.lps.cms.embossing.perso.event.model.NotificationPriority;
import com.lps.cms.embossing.perso.event.model.OutboxEventCommand;
import com.lps.cms.embossing.perso.event.writer.OutboxEventWriter;
import org.springframework.stereotype.Component;

/** Writes one CMS fulfilment command into the shared outbox in the caller's transaction. */
@Component
public class FulfilmentCommandPublisher {

    static final String AGGREGATE_CARD = "CARD";

    private final OutboxEventWriter writer;

    public FulfilmentCommandPublisher(OutboxEventWriter writer) {
        this.writer = writer;
    }

    public UUID publish(UUID institutionId, UUID cardId, String command, String externalEventId,
                        Map<String, Object> payload) {
        return writer.enqueue(OutboxEventCommand.builder()
                .institutionId(institutionId)
                .eventType(command)
                .aggregateType(AGGREGATE_CARD)
                .aggregateId(cardId)
                .correlationId(externalEventId)
                .priority(FulfilmentCommands.FAIL_PERSONALIZATION.equals(command)
                        ? NotificationPriority.HIGH : NotificationPriority.MEDIUM)
                .idempotencyKey(IdempotencyKeys.of(institutionId, command, externalEventId))
                .payload(payload)
                .build());
    }
}
