package com.lps.cms.embossing.perso.report;

import java.nio.charset.StandardCharsets;
import java.security.MessageDigest;
import java.security.NoSuchAlgorithmException;
import java.time.Instant;
import java.util.HexFormat;
import java.util.LinkedHashMap;
import java.util.Map;
import java.util.UUID;

import tools.jackson.core.JacksonException;
import tools.jackson.databind.json.JsonMapper;
import com.lps.cms.embossing.perso.api.report.CourierEventRequest;
import com.lps.cms.embossing.perso.api.report.FulfilmentCommandResponse;
import com.lps.cms.embossing.perso.api.report.VendorResultRequest;
import com.lps.cms.embossing.perso.event.mapper.PayloadValues;
import com.lps.cms.embossing.perso.exception.EmbossingErrorCodes;
import com.lps.cms.embossing.perso.exception.EmbossingSystemException;
import com.lps.cms.embossing.perso.exception.FulfilmentConflictException;
import com.lps.cms.embossing.perso.exception.ResourceNotFoundException;
import com.lps.cms.embossing.perso.report.FulfilmentCommandPlan.Decision;
import com.lps.cms.embossing.perso.report.FulfilmentReportStore.InboxDraft;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

/** Accepts a vendor or courier report and enqueues the matching CMS fulfilment command. */
@Service
public class FulfilmentReportService {

    private final FulfilmentReportStore store;
    private final FulfilmentCommandPublisher publisher;
    private final JsonMapper objectMapper;

    public FulfilmentReportService(FulfilmentReportStore store, FulfilmentCommandPublisher publisher,
                                   JsonMapper objectMapper) {
        this.store = store;
        this.publisher = publisher;
        this.objectMapper = objectMapper;
    }

    @Transactional
    public FulfilmentCommandResponse reportVendor(UUID institutionId, VendorResultRequest report) {
        ReportSubject subject = subject(institutionId, report.productionRequestId());
        Decision decision = FulfilmentCommandPlan.vendor(report, subject,
                store.priorCommands(institutionId, report.productionRequestId(), report.externalEventId()));
        Map<String, Object> payload = commandPayload(report.externalEventId(), decision.command(), subject,
                subject.vendorId(), report.occurredAt(), decision.evidence());
        return commit(FulfilmentCommands.SOURCE_VENDOR, report.result().name(), report.externalEventId(),
                subject, decision, payload);
    }

    @Transactional
    public FulfilmentCommandResponse reportCourier(UUID institutionId, CourierEventRequest report) {
        ReportSubject subject = subject(institutionId, report.productionRequestId());
        Decision decision = FulfilmentCommandPlan.courier(report, subject,
                store.priorCommands(institutionId, report.productionRequestId(), report.externalEventId()));
        Map<String, Object> payload = commandPayload(report.externalEventId(), decision.command(), subject,
                subject.vendorId(), report.eventTime(), decision.evidence());
        return commit(FulfilmentCommands.SOURCE_COURIER, report.eventType().name(), report.externalEventId(),
                subject, decision, payload);
    }

    private FulfilmentCommandResponse commit(String source, String eventType, String externalEventId,
                                             ReportSubject subject, Decision decision, Map<String, Object> payload) {
        String json = toJson(payload);
        String hash = sha256(json);
        var inserted = store.insert(new InboxDraft(source, externalEventId, subject.institutionId(), eventType,
                decision.command(), subject.productionRequestId(), subject.cardId(), json, hash));
        if (inserted.isEmpty()) {
            return duplicate(source, externalEventId, subject, decision, hash);
        }
        String requestStatus = advance(subject, decision);
        UUID commandEventId = publisher.publish(subject.institutionId(), subject.cardId(), decision.command(),
                externalEventId, payload);
        return new FulfilmentCommandResponse(inserted.get(), decision.command(), subject.cardId(),
                subject.productionRequestId(), requestStatus, commandEventId, false);
    }

    private FulfilmentCommandResponse duplicate(String source, String externalEventId, ReportSubject subject,
                                                Decision decision, String hash) {
        InboxEvent existing = store.findInbox(source, externalEventId)
                .orElseThrow(() -> new EmbossingSystemException(EmbossingErrorCodes.UNEXPECTED,
                        "Provider event disappeared after a duplicate insert"));
        if (!existing.institutionId().equals(subject.institutionId())
                || !existing.productionRequestId().equals(subject.productionRequestId())
                || !existing.cardId().equals(subject.cardId())
                || !existing.commandName().equals(decision.command())
                || !existing.payloadSha256().equals(hash)) {
            throw new FulfilmentConflictException(EmbossingErrorCodes.REQUEST_STATUS_CHANGED,
                    "Provider event id was reused with a different report");
        }
        return new FulfilmentCommandResponse(existing.inboxEventId(), existing.commandName(), existing.cardId(),
                existing.productionRequestId(), subject.requestStatus(), null, true);
    }

    private String advance(ReportSubject subject, Decision decision) {
        if (decision.nextRequestStatus() == null) {
            return subject.requestStatus();
        }
        int updated = store.advanceStatus(subject.institutionId(), subject.productionRequestId(),
                subject.requestStatus(), decision.nextRequestStatus());
        if (updated != 1) {
            throw new FulfilmentConflictException(EmbossingErrorCodes.REQUEST_STATUS_CHANGED,
                    "Production request status changed before the command was recorded");
        }
        return decision.nextRequestStatus();
    }

    private ReportSubject subject(UUID institutionId, UUID productionRequestId) {
        return store.findSubject(institutionId, productionRequestId)
                .orElseThrow(() -> new ResourceNotFoundException("Production request was not found"));
    }

    private static Map<String, Object> commandPayload(String externalEventId, String command, ReportSubject subject,
                                                      UUID vendorId, Instant occurredAt, Map<String, Object> evidence) {
        Map<String, Object> payload = new LinkedHashMap<>();
        payload.put("schemaVersion", 1);
        payload.put("externalEventId", externalEventId);
        payload.put("eventType", command);
        payload.put("institutionId", PayloadValues.uuid(subject.institutionId()));
        payload.put("cardId", PayloadValues.uuid(subject.cardId()));
        payload.put("productionRequestId", PayloadValues.uuid(subject.productionRequestId()));
        payload.put("vendorId", PayloadValues.uuid(vendorId));
        payload.put("occurredAt", PayloadValues.instant(occurredAt));
        payload.put("evidence", evidence);
        return payload;
    }

    private String toJson(Map<String, Object> payload) {
        try {
            return objectMapper.writeValueAsString(payload);
        } catch (JacksonException ex) {
            throw new EmbossingSystemException(EmbossingErrorCodes.OUTBOX_WRITE_FAILED,
                    "Fulfilment command payload could not be serialized");
        }
    }

    private static String sha256(String value) {
        try {
            byte[] digest = MessageDigest.getInstance("SHA-256").digest(value.getBytes(StandardCharsets.UTF_8));
            return HexFormat.of().formatHex(digest);
        } catch (NoSuchAlgorithmException ex) {
            throw new EmbossingSystemException(EmbossingErrorCodes.UNEXPECTED, "SHA-256 is not available");
        }
    }
}
