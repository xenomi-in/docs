package com.lps.cms.embossing.perso.cms.card;

import java.util.List;
import java.util.UUID;

import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.stereotype.Repository;

/** Loads every card line for one batch in a single read. Never updates {@code cms_v2.card}. */
@Repository
public class CardFileStore {

    private final JdbcTemplate jdbc;

    public CardFileStore(JdbcTemplate jdbc) {
        this.jdbc = jdbc;
    }

    public List<CardFileLine> loadBatch(UUID batchId) {
        return jdbc.query("""
                SELECT c.card_id, c.institution_id, c.pan_last_four, c.embossed_name_line1, c.embossed_name_line2,
                       c.embossed_name_line3, c.embossed_name_line4, c.issue_date, c.expiry_date,
                       c.card_sequence_number, i.institution_code, p.product_code,
                       cu.title, cu.first_name, cu.middle_name, cu.last_name, cu.mobile_number, cu.email,
                       a.address_line_1, a.address_line_2, a.city, a.state_code, a.country_code, a.postal_code,
                       seq.reserved_sequence, item.waybill_number_snapshot
                FROM cms_v2.personalisation_batch_item item
                JOIN cms_v2.card c ON c.institution_id = item.institution_id AND c.card_id = item.card_id
                JOIN cms_v2.institution i ON i.institution_id = c.institution_id
                JOIN cms_v2.card_product p ON p.product_id = c.product_id
                JOIN cms_v2.customer_reference cu ON cu.customer_reference_id = c.customer_reference_id
                LEFT JOIN cms_v2.address_reference a ON a.address_reference_id = c.shipping_address_reference_id
                LEFT JOIN cms_v2.physical_sequence_reservation seq
                  ON seq.institution_id = c.institution_id AND seq.card_id = c.card_id
                WHERE item.batch_id = ?
                ORDER BY item.item_sequence
                """, (rs, row) -> new CardFileLine(
                rs.getObject("card_id", UUID.class),
                rs.getObject("institution_id", UUID.class),
                rs.getString("institution_code"),
                rs.getString("product_code"),
                rs.getString("pan_last_four"),
                rs.getString("embossed_name_line1"),
                rs.getString("embossed_name_line2"),
                rs.getString("embossed_name_line3"),
                rs.getString("embossed_name_line4"),
                rs.getDate("issue_date") == null ? null : rs.getDate("issue_date").toLocalDate(),
                rs.getDate("expiry_date").toLocalDate(),
                rs.getString("card_sequence_number"),
                rs.getString("reserved_sequence"),
                rs.getString("title"),
                rs.getString("first_name"),
                rs.getString("middle_name"),
                rs.getString("last_name"),
                rs.getString("mobile_number"),
                rs.getString("email"),
                rs.getString("address_line_1"),
                rs.getString("address_line_2"),
                rs.getString("city"),
                rs.getString("state_code"),
                rs.getString("country_code"),
                rs.getString("postal_code"),
                rs.getString("waybill_number_snapshot")), batchId);
    }
}
