package com.lps.cms.embossing.tools;

import org.bouncycastle.jce.provider.BouncyCastleProvider;
import org.bouncycastle.openpgp.*;
import org.bouncycastle.openpgp.operator.PBESecretKeyDecryptor;
import org.bouncycastle.openpgp.operator.jcajce.JcaKeyFingerprintCalculator;
import org.bouncycastle.openpgp.operator.jcajce.JcePBESecretKeyDecryptorBuilder;
import org.bouncycastle.openpgp.operator.jcajce.JcePublicKeyDataDecryptorFactoryBuilder;

import java.io.*;
import java.nio.file.Files;
import java.nio.file.Path;
import java.security.Security;

/**
 * Decrypts PGP-encrypted embossing files for local development and verification.
 *
 * <p>Usage: {@code DecryptGpgUtil <file.gpg> <priv-key.asc> [passphrase] [output-file]}
 *
 * <p>If passphrase is omitted or {@code "-"}, an empty passphrase is used (only for unprotected keys).
 * Passphrase-protected keys must supply the real passphrase or decryption fails when unlocking
 * the secret key.
 *
 * @author LogicPoint Systems
 */
public class DecryptGpgUtil {

    public static void main(String[] args) throws Exception {
        if (args.length < 2) {
            System.err.println("Usage: DecryptGpgUtil <file.gpg> <priv-key.asc> [passphrase] [output-file]");
            System.exit(1);
        }

        Security.addProvider(new BouncyCastleProvider());

        Path gpgFile  = Path.of(args[0]);
        Path privFile = Path.of(args[1]);
        String passphrase = args.length >= 3 ? args[2] : null;
        if ("-".equals(passphrase)) passphrase = null;
        Path outFile  = args.length >= 4 ? Path.of(args[3]) : null;

        byte[] decrypted = decrypt(gpgFile, privFile, passphrase);

        if (outFile != null) {
            Files.write(outFile, decrypted);
            System.out.println("Decrypted content written to: " + outFile.toAbsolutePath());
        } else {
            System.out.println("=== Decrypted content ===");
            System.out.println(new String(decrypted));
        }
    }

    public static byte[] decrypt(Path gpgFile, Path privKeyFile, String passphrase) throws Exception {
        try (InputStream in = PGPUtil.getDecoderStream(new BufferedInputStream(Files.newInputStream(gpgFile)))) {
            PGPObjectFactory factory = new PGPObjectFactory(in, new JcaKeyFingerprintCalculator());

            PGPEncryptedDataList encDataList = findEncryptedDataList(factory);
            if (encDataList == null) {
                throw new IllegalArgumentException("No encrypted data found in: " + gpgFile);
            }

            PGPPublicKeyEncryptedData encData = findEncryptedData(encDataList);
            if (encData == null) {
                throw new IllegalArgumentException("No public-key encrypted data found in: " + gpgFile);
            }

            PGPSecretKey secretKey = loadSecretKey(privKeyFile, encData.getKeyID());
            char[] passphraseChars = passphrase != null ? passphrase.toCharArray() : new char[0];
            PBESecretKeyDecryptor decryptor = new JcePBESecretKeyDecryptorBuilder()
                    .setProvider("BC")
                    .build(passphraseChars);
            PGPPrivateKey privateKey;
            try {
                privateKey = secretKey.extractPrivateKey(decryptor);
            } catch (PGPException ex) {
                if (ex.getMessage() != null && ex.getMessage().contains("checksum mismatch")) {
                    throw new IllegalArgumentException(
                            "Failed to unlock private key — wrong or missing passphrase for " + privKeyFile,
                            ex);
                }
                throw ex;
            }

            try (InputStream clearStream = encData.getDataStream(
                    new JcePublicKeyDataDecryptorFactoryBuilder().setProvider("BC").build(privateKey))) {

                PGPObjectFactory plainFactory = new PGPObjectFactory(clearStream, new JcaKeyFingerprintCalculator());
                return extractLiteralData(plainFactory);
            }
        }
    }

    private static PGPSecretKey loadSecretKey(Path privKeyFile, long encryptionKeyId) throws Exception {
        try (InputStream in = PGPUtil.getDecoderStream(new BufferedInputStream(Files.newInputStream(privKeyFile)))) {
            PGPSecretKeyRingCollection keyRings = new PGPSecretKeyRingCollection(in, new JcaKeyFingerprintCalculator());
            PGPSecretKey match = keyRings.getSecretKey(encryptionKeyId);
            if (match != null) {
                return match;
            }
        }
        throw new IllegalArgumentException(
                "No secret key for encryption key id %016X in: %s".formatted(encryptionKeyId, privKeyFile));
    }

    private static PGPEncryptedDataList findEncryptedDataList(PGPObjectFactory factory) throws Exception {
        Object obj = factory.nextObject();
        if (obj instanceof PGPEncryptedDataList list) {
            return list;
        }
        obj = factory.nextObject();
        if (obj instanceof PGPEncryptedDataList list) {
            return list;
        }
        return null;
    }

    private static PGPPublicKeyEncryptedData findEncryptedData(PGPEncryptedDataList list) {
        for (PGPEncryptedData data : list) {
            if (data instanceof PGPPublicKeyEncryptedData pubData) {
                return pubData;
            }
        }
        return null;
    }

    private static byte[] extractLiteralData(PGPObjectFactory factory) throws Exception {
        Object obj;
        while ((obj = factory.nextObject()) != null) {
            if (obj instanceof PGPCompressedData compressed) {
                PGPObjectFactory innerFactory = new PGPObjectFactory(
                        compressed.getDataStream(), new JcaKeyFingerprintCalculator());
                return extractLiteralData(innerFactory);
            }
            if (obj instanceof PGPLiteralData literal) {
                try (InputStream dataIn = literal.getInputStream()) {
                    return dataIn.readAllBytes();
                }
            }
        }
        throw new IllegalStateException("No literal data found in decrypted stream");
    }
}
