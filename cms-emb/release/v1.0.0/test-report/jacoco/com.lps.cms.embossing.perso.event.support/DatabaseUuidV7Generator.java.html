package com.lps.cms.embossing.perso.event.support;

import java.util.UUID;

import com.lps.cms.embossing.perso.exception.EmbossingErrorCodes;
import com.lps.cms.embossing.perso.exception.EmbossingSystemException;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.stereotype.Component;

@Component
public class DatabaseUuidV7Generator {

    private final JdbcTemplate jdbcTemplate;

    public DatabaseUuidV7Generator(JdbcTemplate jdbcTemplate) {
        this.jdbcTemplate = jdbcTemplate;
    }

    public UUID generate() {
        UUID value = jdbcTemplate.queryForObject("SELECT uuidv7()", UUID.class);
        if (value == null) {
            throw new EmbossingSystemException(EmbossingErrorCodes.UUID_GENERATION_FAILED,
                    "PostgreSQL did not return a UUIDv7 value");
        }
        return value;
    }
}
