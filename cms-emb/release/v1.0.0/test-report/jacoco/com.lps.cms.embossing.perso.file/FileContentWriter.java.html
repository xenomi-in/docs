package com.lps.cms.embossing.perso.file;

import com.lps.cms.embossing.perso.file.format.EmbossingDetailLineFormatter;
import com.lps.cms.embossing.perso.file.format.EmbossingFooterFormatter;
import com.lps.cms.embossing.perso.file.format.EmbossingHeaderFormatter;
import com.lps.cms.embossing.perso.file.model.EmbossingDetailLineParams;
import com.lps.cms.embossing.perso.file.model.EmbossingFileData;
import org.springframework.stereotype.Component;

import java.io.BufferedWriter;
import java.io.IOException;
import java.io.OutputStream;
import java.io.OutputStreamWriter;
import java.nio.charset.StandardCharsets;

/**
 * Writes the EMB header, DET lines, and END footer as UTF-8.
 * The supplied stream is flushed and left open for the caller to close.
 */
@Component
public class FileContentWriter {

    private static final int BUFFER_SIZE = 64 * 1024;

    private final EmbossingHeaderFormatter headerFormatter = new EmbossingHeaderFormatter();
    private final EmbossingDetailLineFormatter detailFormatter = new EmbossingDetailLineFormatter();
    private final EmbossingFooterFormatter footerFormatter = new EmbossingFooterFormatter();

    public void write(EmbossingFileData data, OutputStream out) throws IOException {
        // The caller owns the OpenPGP stream and must close it exactly once to write its
        // integrity packet. Closing this writer would close that stream prematurely.
        BufferedWriter writer = new BufferedWriter(new OutputStreamWriter(out, StandardCharsets.UTF_8), BUFFER_SIZE);
        writer.write(headerFormatter.format(data.header()));
        writer.write('\n');
        for (EmbossingDetailLineParams line : (Iterable<EmbossingDetailLineParams>) data.details()::iterator) {
            writer.write(detailFormatter.format(line));
            writer.write('\n');
        }
        writer.write(footerFormatter.format(data.footer()));
        writer.write('\n');
        writer.flush();
    }
}
