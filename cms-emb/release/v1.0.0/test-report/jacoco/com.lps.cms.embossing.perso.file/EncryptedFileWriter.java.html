package com.lps.cms.embossing.perso.file;

import com.lps.cms.embossing.perso.exception.EmbossingErrorCodes;
import com.lps.cms.embossing.perso.exception.EmbossingException;
import com.lps.cms.embossing.perso.exception.EmbossingSystemException;
import com.lps.cms.embossing.perso.exception.FileGenerationException;
import com.lps.cms.embossing.perso.file.crypto.PgpEncryptionTarget;
import com.lps.cms.embossing.perso.file.crypto.StreamingPgpEncryptor;
import com.lps.cms.embossing.perso.file.model.EmbossingFileData;
import lombok.RequiredArgsConstructor;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Component;

import java.io.OutputStream;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.StandardOpenOption;
import java.security.DigestOutputStream;
import java.security.MessageDigest;
import java.time.LocalDateTime;
import java.time.ZoneOffset;
import java.time.format.DateTimeFormatter;
import java.util.HexFormat;

/**
 * Writes one ASCII-armored OpenPGP personalisation file under {@code embossing.output-path}
 * and returns SHA-256 checksums of the cleartext and the ciphertext.
 * A failed write removes the partial file.
 */
@Component
@RequiredArgsConstructor
public class EncryptedFileWriter {

    private static final DateTimeFormatter FILE_DATE = DateTimeFormatter.ofPattern("yyyyMMdd");

    private final FileContentWriter contentWriter;
    private final StreamingPgpEncryptor pgpEncryptor;

    @Value("${embossing.output-path:./embossing-output}")
    private Path outputBasePath;

    public WrittenFile write(String batchKey, EmbossingFileData data, PgpEncryptionTarget encryptionTarget) {
        Path outputPath = null;
        try {
            int recordCount = data.footer().recordCount();
            outputPath = resolveOutputPath(batchKey, data, recordCount);
            String baseFileName = outputPath.getFileName().toString();
            int dot = baseFileName.lastIndexOf('.');
            if (dot > 0) {
                baseFileName = baseFileName.substring(0, dot);
            }
            Files.createDirectories(outputPath.getParent());
            MessageDigest encryptedDigest = MessageDigest.getInstance("SHA-256");
            MessageDigest clearDigest = MessageDigest.getInstance("SHA-256");
            try (OutputStream fileOut = Files.newOutputStream(outputPath, StandardOpenOption.CREATE_NEW);
                 DigestOutputStream encryptedDigestOut = new DigestOutputStream(fileOut, encryptedDigest);
                 OutputStream encryptedOut = pgpEncryptor.wrap(encryptedDigestOut, baseFileName, encryptionTarget)) {
                // FileContentWriter flushes but never closes this stream. Closing it here finalises
                // the OpenPGP integrity packet before the ciphertext digest is read.
                contentWriter.write(data, new DigestOutputStream(encryptedOut, clearDigest));
            }
            return new WrittenFile(outputPath, recordCount, Files.size(outputPath),
                    HexFormat.of().formatHex(clearDigest.digest()),
                    HexFormat.of().formatHex(encryptedDigest.digest()), encryptionTarget.fingerprint());
        } catch (EmbossingException ex) {
            deletePartialFile(outputPath, ex);
            throw ex;
        } catch (Exception ex) {
            deletePartialFile(outputPath, ex);
            throw new FileGenerationException(EmbossingErrorCodes.FILE_WRITE_FAILED,
                    "Embossing file could not be written", ex);
        }
    }

    private static void deletePartialFile(Path outputPath, Exception failure) {
        if (outputPath == null) {
            return;
        }
        try {
            Files.deleteIfExists(outputPath);
        } catch (Exception cleanup) {
            failure.addSuppressed(cleanup);
        }
    }

    private Path resolveOutputPath(String batchKey, EmbossingFileData data, int recordCount) {
        String date = LocalDateTime.now(ZoneOffset.UTC).format(FILE_DATE);
        String bankName = blank(data.header().bankName()) ? "BAT" : data.header().bankName();
        String bin = blank(data.header().plasticCode()) ? batchKey.substring(0, Math.min(8, batchKey.length())) : data.header().plasticCode();
        for (int sequence = 1; sequence <= 9999; sequence++) {
            Path outputPath = resolveSafeOutputPath(bankName, bin, recordCount, date, sequence);
            if (Files.notExists(outputPath)) {
                return outputPath;
            }
        }
        throw new FileGenerationException(EmbossingErrorCodes.OUTPUT_PATH_EXHAUSTED,
                "Unable to allocate unique embossing output path for batch " + batchKey);
    }

    private Path resolveSafeOutputPath(String bankName, String bin, int recordCount, String date, int sequence) {
        Path base = outputBasePath.toAbsolutePath().normalize();
        String filename = String.join("_", bankName, bin, "P", "V", "D" + recordCount, "B", date,
                String.valueOf(sequence)) + ".gpg";
        Path candidate = base.resolve(bankName).resolve(date).resolve(filename).toAbsolutePath().normalize();
        if (!candidate.startsWith(base)) {
            throw new EmbossingSystemException(EmbossingErrorCodes.OUTPUT_PATH_INVALID,
                    "Embossing output path escapes configured output root");
        }
        return candidate;
    }

    private static boolean blank(String value) {
        return value == null || value.isBlank();
    }

    public record WrittenFile(Path outputPath, int recordCount, long fileSizeBytes, String clearChecksum,
                              String encryptedChecksum, String pgpKeyFingerprint) {}
}
