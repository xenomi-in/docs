package com.lps.cms.embossing.perso.file;

import com.lps.cms.embossing.perso.cms.card.CardFileLine;
import com.lps.cms.embossing.perso.exception.EmbossingErrorCodes;
import com.lps.cms.embossing.perso.exception.FileGenerationException;
import com.lps.cms.embossing.perso.file.model.EmbossingDetailLineParams;
import com.lps.cms.embossing.perso.file.model.EmbossingFileData;
import com.lps.cms.embossing.perso.file.model.EmbossingFooterParams;
import com.lps.cms.embossing.perso.file.model.EmbossingHeaderParams;
import com.lps.cms.embossing.perso.file.data.SensitiveCardDataProvider;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Component;

import java.time.format.DateTimeFormatter;
import java.util.List;

/**
 * Builds {@link EmbossingFileData} for one reserved batch.
 * PAN, CVV, and track data are taken from {@link SensitiveCardDataProvider}.
 */
@Component
@RequiredArgsConstructor
public class EmbossingDataFactory {

    private static final DateTimeFormatter YYMM = DateTimeFormatter.ofPattern("yyMM");
    private static final DateTimeFormatter MMDDHHMMSS = DateTimeFormatter.ofPattern("MMddHHmmss");

    private final SensitiveCardDataProvider sensitiveCardDataProvider;

    public EmbossingFileData load(String requestType, List<CardFileLine> lines) {
        if (lines.isEmpty()) {
            throw new FileGenerationException(EmbossingErrorCodes.BATCH_EMPTY, "Batch has no card lines");
        }
        CardFileLine first = lines.getFirst();
        String newOrReplacement = "REPLACEMENT".equalsIgnoreCase(requestType) ? "R" : "N";
        EmbossingHeaderParams header = new EmbossingHeaderParams(
                "EMB",
                truncate(first.institutionCode(), 3),
                java.time.LocalDateTime.now().format(MMDDHHMMSS),
                truncate(first.institutionId().toString(), 10),
                newOrReplacement,
                truncate(first.productCode(), 6),
                truncate(first.productCode(), 6));

        return new EmbossingFileData(header, lines.stream().map(line -> detail(line, first.productCode())),
                new EmbossingFooterParams("END", lines.size()));
    }

    private EmbossingDetailLineParams detail(CardFileLine card, String expectedProductCode) {
        if (!expectedProductCode.equals(card.productCode())) {
            throw new FileGenerationException(EmbossingErrorCodes.BATCH_PRODUCT_MISMATCH,
                    "Batch contains multiple products; card=" + card.cardId());
        }
        SensitiveCardDataProvider.SensitiveCardData sensitive = sensitiveCardDataProvider.resolve(card);
        validateSensitiveData(card.cardId(), sensitive);
        String sequence = card.reservedSequence() != null ? card.reservedSequence() : card.cardSequence();
        return new EmbossingDetailLineParams(
                    "DET",
                    card.waybillNumber(),
                    sensitive.pan(),
                    sequence == null ? null : sequence.substring(Math.max(0, sequence.length() - 2)),
                    card.embossedName1(),
                    card.issueDate() == null ? null : card.issueDate().format(YYMM),
                    card.expiryDate().format(YYMM),
                    sensitive.cvv2(),
                    null,
                    sensitive.track2(),
                    sensitive.cvv1(),
                    null,
                    card.panLastFour(),
                    card.title(),
                    card.address1(),
                    card.address2(),
                    null,
                    null,
                    card.city(),
                    card.state(),
                    card.country(),
                    card.postalCode(),
                    card.firstName(),
                    card.middleName(),
                    card.lastName(),
                    card.embossedName1(),
                    truncate(card.productCode(), 20),
                    "N",
                    "P",
                    "N",
                    null,
                    card.embossedName3(),
                    card.embossedName4(),
                    card.address1(),
                    card.address2(),
                    null,
                    null,
                    card.city(),
                    card.state(),
                    card.country(),
                    card.postalCode(),
                    card.mobile(),
                    card.mobile(),
                    card.email(),
                    truncate(card.productCode(), 6),
                    truncate(card.productCode(), 6),
                    null,
                    null,
                    null,
                    null);
    }

    private static void validateSensitiveData(java.util.UUID cardId,
                                              SensitiveCardDataProvider.SensitiveCardData sensitive) {
        if (sensitive == null || sensitive.pan() == null || !sensitive.pan().matches("[0-9]{12,19}")) {
            throw new FileGenerationException(EmbossingErrorCodes.SENSITIVE_DATA_INVALID,
                    "Secure export returned an invalid PAN for card=" + cardId);
        }
        if (sensitive.cvv1() == null || !sensitive.cvv1().matches("[0-9]{3,4}")
                || sensitive.cvv2() == null || !sensitive.cvv2().matches("[0-9]{3,4}")) {
            throw new FileGenerationException(EmbossingErrorCodes.SENSITIVE_DATA_INVALID,
                    "Secure export returned invalid CVV data for card=" + cardId);
        }
        if (sensitive.track2() == null || sensitive.track2().isBlank()) {
            throw new FileGenerationException(EmbossingErrorCodes.SENSITIVE_DATA_INVALID,
                    "Secure export did not return Track 2 for card=" + cardId);
        }
    }

    private static String truncate(String value, int max) {
        if (value == null) {
            return "";
        }
        return value.length() <= max ? value : value.substring(0, max);
    }
}
