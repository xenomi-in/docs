package com.lps.cms.embossing.perso.pipeline;

import com.lps.cms.embossing.perso.exception.EmbossingException;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.boot.context.event.ApplicationReadyEvent;
import org.springframework.context.event.EventListener;
import org.springframework.scheduling.annotation.Scheduled;
import org.springframework.stereotype.Component;

/**
 * Runs {@link Pipeline} on {@code embossing.schedule.cron}.
 * When {@code embossing.schedule.run-on-startup} is enabled, it also runs once the application is ready.
 */
@Slf4j
@Component
@RequiredArgsConstructor
public class EmbossingPipelineScheduler {

    private final Pipeline pipeline;

    @Value("${embossing.schedule.enabled:true}")
    private boolean enabled;

    @Value("${embossing.schedule.run-on-startup:false}")
    private boolean runOnStartup;

    @EventListener(ApplicationReadyEvent.class)
    public void onApplicationReady() {
        if (!enabled || !runOnStartup) {
            return;
        }
        log.info("Embossing pipeline triggered on startup");
        execute();
    }

    @Scheduled(cron = "${embossing.schedule.cron:0 0 0 * * *}")
    public void run() {
        if (!enabled) {
            return;
        }
        log.info("Embossing pipeline triggered by schedule");
        execute();
    }

    private void execute() {
        try {
            pipeline.execute();
        } catch (EmbossingException ex) {
            log.error("Embossing pipeline failed code={}", ex.code(), ex);
        } catch (RuntimeException ex) {
            log.error("Embossing pipeline failed", ex);
        }
    }
}
