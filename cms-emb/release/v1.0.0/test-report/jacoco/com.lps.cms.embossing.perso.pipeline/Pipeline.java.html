package com.lps.cms.embossing.perso.pipeline;

import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Service;

import com.lps.cms.embossing.perso.event.correlation.JobRun;
import com.lps.cms.embossing.perso.exception.EmbossingException;
import com.lps.cms.embossing.perso.file.transmit.FileTransmitter;

/**
 * Runs one fulfilment cycle: reserve eligible requests, generate their files,
 * then transmit files that are ready for SFTP.
 */
@Slf4j
@Service
@RequiredArgsConstructor
public class Pipeline {

    private final FulfilmentJob fulfilmentJob;
    private final FileTransmitter fileTransmitter;

    public void execute() {
        JobRun.start();
        try {
            var batches = fulfilmentJob.claimAndBatch();
            int generated = 0;
            for (var batch : batches) {
                try {
                    if (fulfilmentJob.generateOne(batch.batchId(), batch.requestType(), batch.institutionId())) {
                        generated++;
                    }
                } catch (EmbossingException ex) {
                    log.error("File generation failed for batch {} code={}", batch.batchId(), ex.code(), ex);
                    fulfilmentJob.markGenerationFailed(batch.batchId(), batch.institutionId(), ex.getMessage());
                } catch (RuntimeException ex) {
                    log.error("File generation failed for batch {}", batch.batchId(), ex);
                    fulfilmentJob.markGenerationFailed(batch.batchId(), batch.institutionId(), "File generation failed");
                }
            }
            int transmitted = transmitReadyFiles();
            log.info("Embossing pipeline completed — correlationId={} batches={} generated={} transmitted={}",
                    JobRun.current(), batches.size(), generated, transmitted);
        } finally {
            JobRun.clear();
        }
    }

    private int transmitReadyFiles() {
        try {
            return fileTransmitter.transmitReady();
        } catch (RuntimeException ex) {
            log.error("SFTP transmission failed", ex);
            return 0;
        }
    }
}
