/**
 * Reserves eligible production requests into personalisation batches and writes
 * one encrypted file for a batch that is still reserved.
 */
@Slf4j
@Service
@RequiredArgsConstructor
public class FulfilmentJob {

    private final ProductionRequestStore requests;
    private final PersonalisationBatchStore batches;
    private final WaybillStore waybills;
    private final CardFileStore cardFiles;
    private final EmbossingDataFactory dataFactory;
    private final EncryptedFileWriter encryptedFileWriter;
    private final PgpPublicKeyResolver pgpPublicKeyResolver;
    private final EmbossingEventSink events;
    private final ProductionAttemptRepository attempts;
    private final ReconciliationExceptionRepository exceptions;

    @Value("${embossing.v2.claim-limit:500}")
    private int claimLimit;

    @Value("${embossing.output-path:./embossing-output}")
    private Path outputBasePath;

    @Transactional
    public List<CreatedBatch> claimAndBatch() {
        List<EligibleRequest> locked = requests.lockEligible(claimLimit);
        if (requests.reserve(ids(locked)) != locked.size()) {
            throw new FulfilmentConflictException(EmbossingErrorCodes.REQUEST_RESERVATION_LOST,
                    "Eligible request lock was not fully reserved");
        }
        Map<String, List<EligibleRequest>> groups = new LinkedHashMap<>();
        for (EligibleRequest request : locked) {
            groups.computeIfAbsent(groupKey(request), key -> new ArrayList<>()).add(request);
        }
        List<CreatedBatch> created = new ArrayList<>();
        for (List<EligibleRequest> group : groups.values()) {
            EligibleRequest first = group.getFirst();
            var configuration = batches.activeConfiguration(first.institutionId(), first.configurationId());
            if (configuration.isEmpty()) {
                requests.release(ids(group));
                recordException(first.institutionId(), "BATCH_CONFIGURATION_VERSION_MISSING", "BATCH_CONFIGURATION",
                        first.configurationId().toString(), "No ACTIVE personalisation batch configuration version");
                continue;
            }
            int maximumBatchSize = configuration.get().maximumBatchSize();
            for (int start = 0; start < group.size(); start += maximumBatchSize) {
                createBatch(group.subList(start, Math.min(group.size(), start + maximumBatchSize)),
                        configuration.get()).ifPresent(created::add);
            }
        }
        return created;
    }

    private java.util.Optional<CreatedBatch> createBatch(List<EligibleRequest> group, ActiveConfiguration configuration) {
        EligibleRequest first = group.getFirst();
        List<Waybill> available = waybills.lockAvailable(first.institutionId(), group.size());
        if (available.size() < group.size()) {
            List<EligibleRequest> unfilled = group.subList(available.size(), group.size());
            requests.release(ids(unfilled));
            unfilled.forEach(request -> recordException(request.institutionId(), "WAYBILL_UNAVAILABLE",
                    "PRODUCTION_REQUEST", request.productionRequestId().toString(),
                    "No AVAILABLE waybill; request left ELIGIBLE"));
        }
        if (available.isEmpty()) {
            return java.util.Optional.empty();
        }
        List<EligibleRequest> paired = group.subList(0, available.size());
        waybills.reserve(available.stream().map(Waybill::waybillId).toList());

        UUID batchId = UUID.randomUUID();
        String reference = "V2-" + LocalDate.now(ZoneOffset.UTC) + "-" + batchId.toString().substring(0, 8);
        batches.insertBatch(batchId, first.institutionId(), first.vendorId(), configuration.configurationVersionId(),
                reference, paired.size());
        List<NewBatchItem> items = new ArrayList<>(paired.size());
        for (int index = 0; index < paired.size(); index++) {
            EligibleRequest request = paired.get(index);
            Waybill waybill = available.get(index);
            UUID itemId = UUID.randomUUID();
            items.add(new NewBatchItem(itemId, request.institutionId(), request.productionRequestId(),
                    request.cardId(), waybill.waybillId(), waybill.waybillNumber(), index + 1));
            saveAttempt(request, itemId);
        }
        batches.insertItems(batchId, items);
        publishBatchCreated(batchId, reference, first);
        log.info("Created personalisation batch {} with {} item(s)", batchId, paired.size());
        return java.util.Optional.of(new CreatedBatch(batchId, first.institutionId(), first.requestType()));
    }

    @Transactional
    public void markGenerationFailed(UUID batchId, UUID institutionId, String message) {
        batches.updateStatus(batchId, "GENERATION_FAILED");
        recordException(institutionId, "FILE_GENERATION_FAILED", "PRODUCTION_BATCH", batchId.toString(), message);
        batches.header(batchId).ifPresent(batch -> events.batchFailed(
                institutionId, batchId, batch.batchReference(),
                batch.productId() == null ? null : batch.productId().toString(),
                batch.vendorId(), batch.itemCount()));
    }

    /**
     * Writes one reserved batch. A batch that lost eligibility is cancelled in this same
     * transaction and reported as skipped, so the cancellation commits instead of rolling back
     * with the exception.
     */
    @Transactional
    public boolean generateOne(UUID batchId, String requestType, UUID institutionId) {
        try {
            writeReservedBatch(batchId, requestType, institutionId);
            return true;
        } catch (BatchNoLongerEligibleException ex) {
            log.info("Skipped cancelled or ineligible personalisation batch {} code={}", batchId, ex.code());
            return false;
        }
    }

    private void writeReservedBatch(UUID batchId, String requestType, UUID institutionId) {
        List<ReservedBatchItem> lockedItems = batches.lockReservedItems(batchId, institutionId);
        if (lockedItems.isEmpty() || lockedItems.stream().anyMatch(item -> !"RESERVED_FOR_BATCH".equals(item.requestStatus()))) {
            batches.cancelReserved(batchId, institutionId);
            throw new BatchNoLongerEligibleException(batchId);
        }
        batches.updateStatus(batchId, "GENERATING");
        BatchHeader batch = batches.header(batchId)
                .orElseThrow(() -> new EmbossingSystemException(EmbossingErrorCodes.BATCH_NOT_FOUND,
                        "Personalisation batch not found: " + batchId));
        var lines = cardFiles.loadBatch(batchId);
        if (lines.size() != lockedItems.size()) {
            throw new FileGenerationException(EmbossingErrorCodes.BATCH_LINE_MISMATCH,
                    "Card file lines do not match reserved batch items for " + batchId);
        }
        var data = dataFactory.load(requestType, lines);
        EncryptedFileWriter.WrittenFile result = encryptedFileWriter.write(batchId.toString(), data,
                pgpPublicKeyResolver.resolve(institutionId, batch.vendorId()));
        String relative = outputBasePath.toAbsolutePath().normalize()
                .relativize(result.outputPath().toAbsolutePath().normalize())
                .toString().replace('\\', '/');
        UUID fileId = UUID.randomUUID();
        batches.insertFile(fileId, institutionId, batchId, result.outputPath().getFileName().toString(),
                result.recordCount(), "sha256:" + result.clearChecksum(),
                "sha256:" + result.encryptedChecksum(), relative);
        batches.updateStatus(batchId, "FILE_GENERATED");
        events.fileGenerated(institutionId, fileId, batchId, batch.vendorId(),
                result.recordCount(), java.time.Instant.now().toString());
        waybills.assign(lockedItems.stream().map(ReservedBatchItem::waybillId).toList());
        requests.markFileGenerated(lockedItems.stream().map(ReservedBatchItem::productionRequestId).toList());
    }

    private void publishBatchCreated(UUID batchId, String reference, EligibleRequest first) {
        var cards = batches.noticeCards(batchId);
        events.batchCreated(first.institutionId(), batchId, reference,
                first.productId().toString(), first.vendorId(), cards.size(),
                cards.stream().map(card -> new CardRef(card.cardId(), card.customerRef(), card.lastFour())).toList());
    }

    private void saveAttempt(EligibleRequest request, UUID itemId) {
        int next = attempts.findFirstByInstitutionIdAndProductionRequestIdOrderByAttemptNumberDesc(
                request.institutionId(), request.productionRequestId()).map(a -> a.getAttemptNumber() + 1).orElse(1);
        ProductionAttempt attempt = new ProductionAttempt();
        attempt.setAttemptId(UUID.randomUUID());
        attempt.setInstitutionId(request.institutionId());
        attempt.setProductionRequestId(request.productionRequestId());
        attempt.setCardId(request.cardId());
        attempt.setAttemptNumber(next);
        attempt.setBatchItemId(itemId);
        attempt.setAttemptStatus("BATCHED");
        attempt.setStartedAt(OffsetDateTime.now(ZoneOffset.UTC));
        attempts.save(attempt);
    }

    private void recordException(UUID institutionId, String code, String aggregateType, String aggregateId, String detail) {
        if (exceptions.existsByExceptionCodeAndAggregateIdAndResolvedAtIsNull(code, aggregateId)) {
            return;
        }
        ReconciliationException row = new ReconciliationException();
        row.setExceptionId(UUID.randomUUID());
        row.setInstitutionId(institutionId);
        row.setExceptionCode(code);
        row.setAggregateType(aggregateType);
        row.setAggregateId(aggregateId);
        row.setDetail(safe(detail));
        row.setDetectedAt(OffsetDateTime.now(ZoneOffset.UTC));
        exceptions.save(row);
    }

    private static List<UUID> ids(List<EligibleRequest> requests) {
        return requests.stream().map(EligibleRequest::productionRequestId).toList();
    }

    private static String safe(String message) {
        if (message == null) {
            return "unknown";
        }
        return message.length() <= 1000 ? message : message.substring(0, 1000);
    }

    private static String groupKey(EligibleRequest request) {
        return request.institutionId() + ":" + request.vendorId() + ":" + request.configurationId() + ":"
                + request.productId() + ":" + request.requestType();
    }

    public record CreatedBatch(UUID batchId, UUID institutionId, String requestType) {}

}
