package com.lps.cms.embossing.perso.file.format;

import com.lps.cms.embossing.perso.exception.EmbossingErrorCodes;
import com.lps.cms.embossing.perso.exception.FileGenerationException;
import com.lps.cms.embossing.perso.file.model.EmbossingHeaderParams;

/**
 * Formats one EMB header line. Each field is truncated to its maximum length.
 * A fixed branch marker, {@code B}, is written between the date and the bank identifier.
 */
public class EmbossingHeaderFormatter {

    private static final String RECORD_TYPE = "EMB";
    private static final String BRANCH_CUSTOMER = "B";
    private static final String DEFAULT_NEW_REPLACEMENT = "N";

    private static final int LEN_RECORD_TYPE = 3;
    private static final int LEN_BANK_NAME = 3;
    private static final int LEN_DATE = 10;
    private static final int LEN_BANK_ID = 10;
    private static final int LEN_NEW_REPLACEMENT = 1;
    private static final int LEN_PRODUCT_CODE = 6;
    private static final int LEN_PLASTIC_CODE = 6;

    public String format(EmbossingHeaderParams params) {
        String recordType = max(defaultValue(params.recordType(), RECORD_TYPE), LEN_RECORD_TYPE);
        String bankName = max(params.bankName(), LEN_BANK_NAME);
        String date = max(params.date(), LEN_DATE);
        String bankId = max(params.bankId(), LEN_BANK_ID);
        String productCode = max(params.productCode(), LEN_PRODUCT_CODE);
        String plasticCode = max(params.plasticCode(), LEN_PLASTIC_CODE);
        String newOrReplacement = max(defaultValue(params.newOrReplacement(), DEFAULT_NEW_REPLACEMENT), LEN_NEW_REPLACEMENT);

        return String.join("|",
                recordType,
                bankName,
                date,
                BRANCH_CUSTOMER,
                bankId,
                newOrReplacement,
                productCode,
                plasticCode);
    }

    private static String max(String value, int maxLength) {
        if (value == null) return "";
        validateFileField(value);
        return value.length() <= maxLength ? value : value.substring(0, maxLength);
    }

    private static void validateFileField(String value) {
        if (value.indexOf('|') >= 0 || value.indexOf('~') >= 0 || value.indexOf('\r') >= 0
                || value.indexOf('\n') >= 0 || !value.chars().allMatch(ch -> ch >= 0x20 && ch <= 0x7E)) {
            throw new FileGenerationException(EmbossingErrorCodes.FILE_CONTENT_INVALID,
                    "Embossing header fields must be printable ASCII without delimiters");
        }
    }

    private static String defaultValue(String value, String fallback) {
        return value == null || value.isBlank() ? fallback : value;
    }
}
