package com.lps.cms.embossing.perso.file.format;

import com.lps.cms.embossing.perso.file.model.EmbossingFooterParams;

/**
 * Formats one END footer line. The record type is at most 3 characters
 * and the detail count is at most 9 digits.
 */
public class EmbossingFooterFormatter {

    private static final String RECORD_TYPE = "END";
    private static final int LEN_RECORD_TYPE = 3;
    private static final int LEN_RECORD_COUNT = 9;

    public String format(EmbossingFooterParams params) {
        String recordType = params.recordType() != null ? params.recordType() : RECORD_TYPE;
        return max(recordType, LEN_RECORD_TYPE) + "|" + max(String.valueOf(params.recordCount()), LEN_RECORD_COUNT);
    }

    private static String max(String value, int maxLength) {
        if (value == null) return "";
        return value.length() <= maxLength ? value : value.substring(0, maxLength);
    }
}
