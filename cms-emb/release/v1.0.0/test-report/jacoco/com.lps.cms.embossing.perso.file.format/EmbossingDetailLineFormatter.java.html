package com.lps.cms.embossing.perso.file.format;

import com.lps.cms.embossing.perso.exception.EmbossingErrorCodes;
import com.lps.cms.embossing.perso.exception.FileGenerationException;
import com.lps.cms.embossing.perso.file.model.EmbossingDetailLineParams;

/**
 * Formats one DET record. Values longer than the field maximum are truncated.
 * Shorter values are written unchanged, and the line ends with {@code ~}.
 */
public class EmbossingDetailLineFormatter {

    private static final String RECORD_TYPE = "DET";
    private static final String DELIMITER = "|";
    private static final String END_DELIMITER = "~";

    private static final int LEN_AIRWAY_BILL = 30;
    private static final int LEN_CARD_NUMBER = 19;
    private static final int LEN_CARD_SEQ = 2;
    private static final int LEN_CARD_NAME = 27;
    private static final int LEN_START_DATE = 4;
    private static final int LEN_EXPIRY_DATE = 4;
    private static final int LEN_CVC2 = 3;
    private static final int LEN_TRACK1 = 76;
    private static final int LEN_TRACK2 = 37;
    private static final int LEN_ICVV = 3;
    private static final int LEN_CRN = 16;
    private static final int LEN_ARN_CIF = 16;
    private static final int LEN_SHORT = 5;
    private static final int LEN_ADDRESS = 30;
    private static final int LEN_EMBOSSED_NAME = 35;
    private static final int LEN_TEMPLATE_ID = 20;
    private static final int LEN_ONE = 1;
    private static final int LEN_APP_PREF_NAME = 16;
    private static final int LEN_LINE_EMB = 27;
    private static final int LEN_DELIVERY_ADDRESS = 50;
    private static final int LEN_DELIVERY_OTHER = 30;
    private static final int LEN_DELIVERY_EMAIL = 100;
    private static final int LEN_PLASTIC_PRODUCT = 6;
    private static final int LEN_IMAGE = 20;
    private static final int LEN_PROV1 = 100;
    private static final int LEN_PROV2 = 50;
    private static final int LEN_PROV3 = 25;

    public String format(EmbossingDetailLineParams p) {
        return (p.recordType() != null ? p.recordType() : RECORD_TYPE) + DELIMITER
                + pad(p.airwayBillNumber(), LEN_AIRWAY_BILL)
                + DELIMITER + pad(p.cardNumber(), LEN_CARD_NUMBER)
                + DELIMITER + pad(p.cardSequenceNumber(), LEN_CARD_SEQ)
                + DELIMITER + pad(p.cardName(), LEN_CARD_NAME)
                + DELIMITER + pad(p.startDate(), LEN_START_DATE)
                + DELIMITER + pad(p.expiryDate(), LEN_EXPIRY_DATE)
                + DELIMITER + pad(p.cvc2(), LEN_CVC2)
                + DELIMITER + pad(p.track1Data(), LEN_TRACK1)
                + DELIMITER + pad(p.track2Data(), LEN_TRACK2)
                + DELIMITER + pad(p.iCvv(), LEN_ICVV)
                + DELIMITER + pad(p.crnProxyNumber(), LEN_CRN)
                + DELIMITER + pad(p.arnCifId(), LEN_ARN_CIF)
                + DELIMITER + pad(p.salutation(), LEN_SHORT)
                + DELIMITER + pad(p.address1(), LEN_ADDRESS)
                + DELIMITER + pad(p.address2(), LEN_ADDRESS)
                + DELIMITER + pad(p.address3(), LEN_ADDRESS)
                + DELIMITER + pad(p.address4(), LEN_ADDRESS)
                + DELIMITER + pad(p.city(), LEN_ADDRESS)
                + DELIMITER + pad(p.state(), LEN_ADDRESS)
                + DELIMITER + pad(p.country(), LEN_ADDRESS)
                + DELIMITER + pad(p.postalCode(), LEN_ADDRESS)
                + DELIMITER + pad(p.firstName(), LEN_ADDRESS)
                + DELIMITER + pad(p.middleName(), LEN_ADDRESS)
                + DELIMITER + pad(p.lastName(), LEN_ADDRESS)
                + DELIMITER + pad(p.embossedName(), LEN_EMBOSSED_NAME)
                + DELIMITER + pad(p.templateId(), LEN_TEMPLATE_ID)
                + DELIMITER + pad(p.typeOfDelivery(), LEN_ONE)
                + DELIMITER + pad(p.cardHolderStatus(), LEN_ONE)
                + DELIMITER + pad(p.newCustomer(), LEN_ONE)
                + DELIMITER + pad(p.applicationPreferredName(), LEN_APP_PREF_NAME)
                + DELIMITER + pad(p.thirdLineEmbossing(), LEN_LINE_EMB)
                + DELIMITER + pad(p.fourthLineEmbossing(), LEN_LINE_EMB)
                + DELIMITER + pad(p.deliveryAddress1(), LEN_DELIVERY_ADDRESS)
                + DELIMITER + pad(p.deliveryAddress2(), LEN_DELIVERY_ADDRESS)
                + DELIMITER + pad(p.deliveryAddress3(), LEN_DELIVERY_ADDRESS)
                + DELIMITER + pad(p.deliveryAddress4(), LEN_DELIVERY_ADDRESS)
                + DELIMITER + pad(p.deliveryCity(), LEN_DELIVERY_OTHER)
                + DELIMITER + pad(p.deliveryState(), LEN_DELIVERY_OTHER)
                + DELIMITER + pad(p.deliveryCountry(), LEN_DELIVERY_OTHER)
                + DELIMITER + pad(p.deliveryPostcode(), LEN_DELIVERY_OTHER)
                + DELIMITER + pad(p.deliveryContact(), LEN_DELIVERY_OTHER)
                + DELIMITER + pad(p.deliveryMobileNo(), LEN_DELIVERY_OTHER)
                + DELIMITER + pad(p.deliveryEmailId(), LEN_DELIVERY_EMAIL)
                + DELIMITER + pad(p.plasticCode(), LEN_PLASTIC_PRODUCT)
                + DELIMITER + pad(p.productCode(), LEN_PLASTIC_PRODUCT)
                + DELIMITER + DELIMITER  // two empty placeholders per spec
                + DELIMITER + pad(p.imageFileName(), LEN_IMAGE)
                + DELIMITER + pad(p.provisionalField1(), LEN_PROV1)
                + DELIMITER + pad(p.provisionalField2(), LEN_PROV2)
                + DELIMITER + pad(p.provisionalField3(), LEN_PROV3)
                + END_DELIMITER;
    }

    private static String pad(String value, int length) {
        if (value == null) value = "";
        validateFileField(value);
        if (value.length() >= length) return value.substring(0, length);
        return value + "".repeat(length - value.length());
    }

    private static void validateFileField(String value) {
        if (value.indexOf('|') >= 0 || value.indexOf('~') >= 0 || value.indexOf('\r') >= 0
                || value.indexOf('\n') >= 0) {
            throw new FileGenerationException(EmbossingErrorCodes.FILE_CONTENT_INVALID,
                    "Embossing field contains a record delimiter or line break");
        }
        for (int index = 0; index < value.length(); index++) {
            char character = value.charAt(index);
            if (character < 0x20 || character > 0x7E) {
                throw new FileGenerationException(EmbossingErrorCodes.FILE_CONTENT_INVALID,
                        "Embossing file fields must be printable ASCII");
            }
        }
    }
}
