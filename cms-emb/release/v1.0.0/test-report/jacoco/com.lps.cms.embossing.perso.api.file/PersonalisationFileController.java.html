package com.lps.cms.embossing.perso.api.file;

import java.util.List;
import java.util.UUID;

import com.lps.cms.embossing.perso.api.ApiResponseEnvelope;
import com.lps.cms.embossing.perso.api.ApiResponseHelper;
import com.lps.cms.embossing.perso.api.batch.PersonalisationQueryService;
import org.springframework.http.ResponseEntity;
import org.springframework.validation.annotation.Validated;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RestController;

@Validated
@RestController
@RequestMapping("/cms/api/v2/institutions/{institutionId}")
public class PersonalisationFileController {

    private final PersonalisationQueryService service;

    public PersonalisationFileController(PersonalisationQueryService service) {
        this.service = service;
    }

    @GetMapping("/personalisation-batches/{batchId}/files")
    public ResponseEntity<ApiResponseEnvelope<List<PersonalisationFileResponse>>> list(
            @PathVariable UUID institutionId,
            @PathVariable UUID batchId) {
        return ApiResponseHelper.ok(service.listFiles(institutionId, batchId));
    }

    @GetMapping("/personalisation-files/{fileId}")
    public ResponseEntity<ApiResponseEnvelope<PersonalisationFileResponse>> get(
            @PathVariable UUID institutionId,
            @PathVariable UUID fileId) {
        return ApiResponseHelper.ok(service.getFile(institutionId, fileId));
    }
}
