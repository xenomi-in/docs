package com.lps.cms.embossing.perso.api.report;

import java.util.UUID;

import com.lps.cms.embossing.perso.api.ApiResponseEnvelope;
import com.lps.cms.embossing.perso.api.ApiResponseHelper;
import com.lps.cms.embossing.perso.report.FulfilmentReportService;
import jakarta.validation.Valid;
import org.springframework.http.ResponseEntity;
import org.springframework.validation.annotation.Validated;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RestController;

@Validated
@RestController
@RequestMapping("/cms/api/v2/institutions/{institutionId}/courier-events")
public class CourierEventController {

    private final FulfilmentReportService reports;

    public CourierEventController(FulfilmentReportService reports) {
        this.reports = reports;
    }

    @PostMapping
    public ResponseEntity<ApiResponseEnvelope<FulfilmentCommandResponse>> report(
            @PathVariable UUID institutionId,
            @Valid @RequestBody CourierEventRequest request) {
        return ApiResponseHelper.ok(reports.reportCourier(institutionId, request));
    }
}
