package com.lps.cms.embossing.perso.api.report;

import java.time.Instant;
import java.util.UUID;

import com.lps.cms.embossing.perso.report.VendorOutcome;
import jakarta.validation.constraints.NotBlank;
import jakarta.validation.constraints.NotNull;
import jakarta.validation.constraints.Size;

public record VendorResultRequest(
        @NotBlank @Size(max = 255) String externalEventId,
        @NotNull UUID institutionId,
        @NotNull UUID vendorId,
        @NotNull UUID productionRequestId,
        @NotNull UUID cardId,
        @NotNull VendorOutcome result,
        @Size(max = 3) String verifiedCardSequenceNumber,
        @Size(max = 255) String vendorReference,
        @Size(max = 255) String fileReference,
        @Size(max = 50) String failureCode,
        @Size(max = 500) String failureMessage,
        @NotNull Instant occurredAt) {
}
