package com.lps.cms.embossing.perso.api.report;

import java.time.Instant;
import java.util.UUID;

import com.lps.cms.embossing.perso.report.CourierOutcome;
import jakarta.validation.constraints.Min;
import jakarta.validation.constraints.NotBlank;
import jakarta.validation.constraints.NotNull;
import jakarta.validation.constraints.Size;

public record CourierEventRequest(
        @NotBlank @Size(max = 255) String externalEventId,
        @NotNull UUID institutionId,
        @NotNull UUID courierProviderId,
        UUID shipmentId,
        @NotNull UUID cardId,
        @NotNull UUID productionRequestId,
        @NotBlank @Size(max = 200) String awbNumber,
        @NotNull @Min(1) Integer shipmentAttempt,
        @NotNull CourierOutcome eventType,
        @NotNull Instant eventTime,
        @Size(max = 255) String locationReference,
        @Size(max = 255) String evidenceReference) {
}
