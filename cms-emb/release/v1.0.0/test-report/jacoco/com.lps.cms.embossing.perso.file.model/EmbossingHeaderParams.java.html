package com.lps.cms.embossing.perso.file.model;

/**
 * Fields for one EMB header. Lengths are maximums; the header formatter truncates longer values.
 */
public record EmbossingHeaderParams(

        /** Record type. Maximum 3 characters. Blank values become {@code EMB}. */
        String recordType,
        /** Bank name. Maximum 3 characters. */
        String bankName,
        /** File timestamp in {@code MMDDHHMMSS}. Maximum 10 characters. */
        String date,
        /** Bank identifier. Maximum 10 characters. */
        String bankId,
        /** New or replacement indicator. Maximum 1 character: {@code N} or {@code R}. Blank values become {@code N}. */
        String newOrReplacement,
        /** Product code. Maximum 6 characters. */
        String productCode,
        /** Plastic code. Maximum 6 characters. */
        String plasticCode
) {}
