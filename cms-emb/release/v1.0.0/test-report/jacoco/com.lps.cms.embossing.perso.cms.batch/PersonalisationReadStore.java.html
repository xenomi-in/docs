package com.lps.cms.embossing.perso.cms.batch;

import java.sql.ResultSet;
import java.sql.SQLException;
import java.sql.Timestamp;
import java.time.OffsetDateTime;
import java.time.ZoneOffset;
import java.util.List;
import java.util.Optional;
import java.util.UUID;

import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.stereotype.Repository;

/** Reads personalisation batches and files for one institution. */
@Repository
public class PersonalisationReadStore {

    private final JdbcTemplate jdbc;

    public PersonalisationReadStore(JdbcTemplate jdbc) {
        this.jdbc = jdbc;
    }

    public List<BatchRow> listBatches(UUID institutionId, String status, int limit, long offset) {
        if (status == null) {
            return jdbc.query("""
                    SELECT batch_id, institution_id, vendor_id, batch_configuration_version_id,
                           batch_reference, batch_status, item_count, created_at, updated_at
                    FROM cms_v2.personalisation_batch
                    WHERE institution_id = ?
                    ORDER BY created_at DESC
                    LIMIT ? OFFSET ?
                    """, this::batch, institutionId, limit, offset);
        }
        return jdbc.query("""
                SELECT batch_id, institution_id, vendor_id, batch_configuration_version_id,
                       batch_reference, batch_status, item_count, created_at, updated_at
                FROM cms_v2.personalisation_batch
                WHERE institution_id = ? AND batch_status = ?
                ORDER BY created_at DESC
                LIMIT ? OFFSET ?
                """, this::batch, institutionId, status, limit, offset);
    }

    public long countBatches(UUID institutionId, String status) {
        Number count = status == null
                ? jdbc.queryForObject("""
                        SELECT count(*) FROM cms_v2.personalisation_batch WHERE institution_id = ?
                        """, Number.class, institutionId)
                : jdbc.queryForObject("""
                        SELECT count(*) FROM cms_v2.personalisation_batch
                        WHERE institution_id = ? AND batch_status = ?
                        """, Number.class, institutionId, status);
        return count == null ? 0 : count.longValue();
    }

    public Optional<BatchRow> findBatch(UUID institutionId, UUID batchId) {
        List<BatchRow> rows = jdbc.query("""
                SELECT batch_id, institution_id, vendor_id, batch_configuration_version_id,
                       batch_reference, batch_status, item_count, created_at, updated_at
                FROM cms_v2.personalisation_batch
                WHERE institution_id = ? AND batch_id = ?
                """, this::batch, institutionId, batchId);
        return rows.stream().findFirst();
    }

    public List<FileRow> listFiles(UUID institutionId, UUID batchId) {
        return jdbc.query("""
                SELECT personalisation_file_id, institution_id, batch_id, file_sequence, file_name,
                       schema_version, record_count, file_status, generated_at
                FROM cms_v2.personalisation_file
                WHERE institution_id = ? AND batch_id = ?
                ORDER BY file_sequence
                """, this::file, institutionId, batchId);
    }

    public Optional<FileRow> findFile(UUID institutionId, UUID fileId) {
        List<FileRow> rows = jdbc.query("""
                SELECT personalisation_file_id, institution_id, batch_id, file_sequence, file_name,
                       schema_version, record_count, file_status, generated_at
                FROM cms_v2.personalisation_file
                WHERE institution_id = ? AND personalisation_file_id = ?
                """, this::file, institutionId, fileId);
        return rows.stream().findFirst();
    }

    private BatchRow batch(ResultSet rs, int row) throws SQLException {
        return new BatchRow(
                rs.getObject("batch_id", UUID.class),
                rs.getObject("institution_id", UUID.class),
                rs.getObject("vendor_id", UUID.class),
                rs.getObject("batch_configuration_version_id", UUID.class),
                rs.getString("batch_reference"),
                rs.getString("batch_status"),
                rs.getInt("item_count"),
                utc(rs, "created_at"),
                utc(rs, "updated_at"));
    }

    private FileRow file(ResultSet rs, int row) throws SQLException {
        return new FileRow(
                rs.getObject("personalisation_file_id", UUID.class),
                rs.getObject("institution_id", UUID.class),
                rs.getObject("batch_id", UUID.class),
                rs.getInt("file_sequence"),
                rs.getString("file_name"),
                rs.getString("schema_version"),
                rs.getInt("record_count"),
                rs.getString("file_status"),
                utc(rs, "generated_at"));
    }

    private static OffsetDateTime utc(ResultSet rs, String column) throws SQLException {
        Timestamp timestamp = rs.getTimestamp(column);
        return timestamp == null ? null : timestamp.toInstant().atOffset(ZoneOffset.UTC);
    }

    public record BatchRow(
            UUID batchId,
            UUID institutionId,
            UUID vendorId,
            UUID configurationVersionId,
            String batchReference,
            String batchStatus,
            int itemCount,
            OffsetDateTime createdAt,
            OffsetDateTime updatedAt) {
    }

    public record FileRow(
            UUID fileId,
            UUID institutionId,
            UUID batchId,
            int fileSequence,
            String fileName,
            String schemaVersion,
            int recordCount,
            String fileStatus,
            OffsetDateTime generatedAt) {
    }
}
