package com.lps.cms.embossing.perso.cms.batch;

import java.util.List;
import java.util.Optional;
import java.util.UUID;

import com.lps.cms.embossing.perso.cms.batch.BatchRecords.ActiveConfiguration;
import com.lps.cms.embossing.perso.cms.batch.BatchRecords.BatchHeader;
import com.lps.cms.embossing.perso.cms.batch.BatchRecords.NewBatchItem;
import com.lps.cms.embossing.perso.cms.batch.BatchRecords.NoticeCard;
import com.lps.cms.embossing.perso.cms.batch.BatchRecords.ReservedBatchItem;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.stereotype.Repository;

/**
 * Reads the active batch configuration and persists personalisation batches, items, and file rows.
 */
@Repository
public class PersonalisationBatchStore {

    private final JdbcTemplate jdbc;

    public PersonalisationBatchStore(JdbcTemplate jdbc) {
        this.jdbc = jdbc;
    }

    public Optional<ActiveConfiguration> activeConfiguration(UUID institutionId, UUID configurationId) {
        List<ActiveConfiguration> configurations = jdbc.query("""
                SELECT batch_configuration_version_id, maximum_batch_size, file_format_profile_reference,
                       transmission_channel, transmission_profile_reference
                FROM cms_v2.personalisation_batch_configuration_version
                WHERE institution_id = ? AND batch_configuration_id = ?
                  AND version_status = 'ACTIVE'
                  AND effective_from <= now()
                  AND (effective_to IS NULL OR effective_to > now())
                ORDER BY version_number DESC
                LIMIT 1
                """, (rs, row) -> new ActiveConfiguration(
                rs.getObject("batch_configuration_version_id", UUID.class),
                rs.getInt("maximum_batch_size"),
                rs.getString("file_format_profile_reference"),
                rs.getString("transmission_channel"),
                rs.getString("transmission_profile_reference")), institutionId, configurationId);
        return configurations.stream().findFirst();
    }

    public void insertBatch(UUID batchId, UUID institutionId, UUID vendorId, UUID versionId,
                            String reference, int itemCount) {
        jdbc.update("""
                INSERT INTO cms_v2.personalisation_batch (
                    batch_id, institution_id, vendor_id, batch_configuration_version_id, batch_reference,
                    scheduled_at, cutoff_at, batch_status, item_count, created_at, updated_at, created_by, updated_by)
                VALUES (?, ?, ?, ?, ?, clock_timestamp(), clock_timestamp(), 'CREATED', ?, now(), now(), 'vendor-service', 'vendor-service')
                """, batchId, institutionId, vendorId, versionId, reference, itemCount);
    }

    public void insertItems(UUID batchId, List<NewBatchItem> items) {
        jdbc.batchUpdate("""
                INSERT INTO cms_v2.personalisation_batch_item (
                    batch_item_id, institution_id, batch_id, production_request_id, card_id,
                    courier_waybill_id, waybill_number_snapshot, item_sequence, item_status, added_at)
                VALUES (?, ?, ?, ?, ?, ?, ?, ?, 'RESERVED', now())
                """, items, items.size(), (statement, item) -> {
            statement.setObject(1, item.batchItemId());
            statement.setObject(2, item.institutionId());
            statement.setObject(3, batchId);
            statement.setObject(4, item.productionRequestId());
            statement.setObject(5, item.cardId());
            statement.setObject(6, item.waybillId());
            statement.setString(7, item.waybillNumber());
            statement.setInt(8, item.sequence());
        });
    }

    public List<ReservedBatchItem> lockReservedItems(UUID batchId, UUID institutionId) {
        return jdbc.query("""
                SELECT i.batch_item_id, i.institution_id, i.production_request_id, i.card_id,
                       i.waybill_number_snapshot, i.item_sequence, i.courier_waybill_id, r.request_status
                FROM cms_v2.personalisation_batch_item i
                JOIN cms_v2.card_production_request r
                  ON r.institution_id = i.institution_id
                 AND r.production_request_id = i.production_request_id
                WHERE i.batch_id = ? AND i.institution_id = ?
                ORDER BY i.item_sequence
                FOR UPDATE OF i, r
                """, (rs, row) -> new ReservedBatchItem(
                rs.getObject("batch_item_id", UUID.class),
                rs.getObject("institution_id", UUID.class),
                rs.getObject("production_request_id", UUID.class),
                rs.getObject("card_id", UUID.class),
                rs.getString("waybill_number_snapshot"),
                rs.getInt("item_sequence"),
                rs.getObject("courier_waybill_id", UUID.class),
                rs.getString("request_status")), batchId, institutionId);
    }

    public void cancelReserved(UUID batchId, UUID institutionId) {
        jdbc.update("""
                UPDATE cms_v2.personalisation_batch
                SET batch_status = 'CANCELLED', updated_at = now(), updated_by = 'vendor-service'
                WHERE batch_id = ? AND institution_id = ? AND batch_status IN ('CREATED', 'GENERATING')
                """, batchId, institutionId);
        jdbc.update("""
                UPDATE cms_v2.courier_waybill w
                SET waybill_status = 'AVAILABLE', reserved_at = NULL
                FROM cms_v2.personalisation_batch_item i
                WHERE i.batch_id = ? AND i.institution_id = ?
                  AND w.courier_waybill_id = i.courier_waybill_id
                  AND w.institution_id = i.institution_id
                  AND w.waybill_status = 'RESERVED'
                """, batchId, institutionId);
        jdbc.update("""
                UPDATE cms_v2.card_production_request r
                SET request_status = 'ELIGIBLE', updated_at = now()
                FROM cms_v2.personalisation_batch_item i
                WHERE i.batch_id = ? AND i.institution_id = ?
                  AND r.institution_id = i.institution_id
                  AND r.production_request_id = i.production_request_id
                  AND r.request_status = 'RESERVED_FOR_BATCH'
                """, batchId, institutionId);
    }

    public void updateStatus(UUID batchId, String status) {
        jdbc.update("""
                UPDATE cms_v2.personalisation_batch
                SET batch_status = ?, updated_at = now(), updated_by = 'vendor-service'
                WHERE batch_id = ?
                """, status, batchId);
    }

    public Optional<BatchHeader> header(UUID batchId) {
        List<BatchHeader> rows = jdbc.query("""
                SELECT b.batch_reference, b.vendor_id, b.institution_id, b.item_count,
                       (SELECT c.product_id
                        FROM cms_v2.personalisation_batch_item i
                        JOIN cms_v2.card c
                          ON c.institution_id = i.institution_id AND c.card_id = i.card_id
                        WHERE i.batch_id = b.batch_id
                        ORDER BY i.item_sequence
                        LIMIT 1) AS product_id
                FROM cms_v2.personalisation_batch b
                WHERE b.batch_id = ?
                """, (rs, row) -> new BatchHeader(
                rs.getString("batch_reference"),
                rs.getObject("vendor_id", UUID.class),
                rs.getObject("institution_id", UUID.class),
                rs.getInt("item_count"),
                rs.getObject("product_id", UUID.class)), batchId);
        return rows.stream().findFirst();
    }

    public List<NoticeCard> noticeCards(UUID batchId) {
        return jdbc.query("""
                SELECT c.card_id, cu.external_customer_id, c.pan_last_four, c.product_id
                FROM cms_v2.personalisation_batch_item i
                JOIN cms_v2.card c ON c.institution_id = i.institution_id AND c.card_id = i.card_id
                JOIN cms_v2.customer_reference cu ON cu.customer_reference_id = c.customer_reference_id
                WHERE i.batch_id = ?
                ORDER BY i.item_sequence
                """, (rs, row) -> new NoticeCard(
                rs.getObject("card_id", UUID.class),
                rs.getString("external_customer_id"),
                rs.getString("pan_last_four"),
                rs.getObject("product_id", UUID.class)), batchId);
    }

    public void insertFile(UUID fileId, UUID institutionId, UUID batchId, String fileName,
                           int recordCount, String clearHash, String encryptedHash, String objectRef) {
        jdbc.update("""
                INSERT INTO cms_v2.personalisation_file (
                    personalisation_file_id, institution_id, batch_id, file_sequence, file_name, schema_version,
                    record_count, clear_payload_hash, encrypted_payload_hash, secure_object_reference,
                    file_status, generated_at)
                VALUES (?, ?, ?, 1, ?, 'EMB-1', ?, ?, ?, ?, 'ENCRYPTED', now())
                """, fileId, institutionId, batchId, fileName, recordCount, clearHash, encryptedHash, objectRef);
    }
}
