package com.lps.cms.embossing.perso.exception;

import com.lps.cms.embossing.perso.V2EmbossingModule;
import com.lps.cms.embossing.perso.api.ApiErrorCode;
import com.lps.cms.embossing.perso.api.ApiErrorEnvelope;
import com.lps.cms.embossing.perso.api.ApiResponseHelper;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.http.converter.HttpMessageNotReadableException;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;
import org.springframework.web.method.annotation.MethodArgumentTypeMismatchException;
import org.springframework.web.servlet.resource.NoResourceFoundException;

/**
 * Maps personalisation failures onto the cms-core error envelope.
 * Unexpected failures are logged and returned as {@link ApiErrorCode#INTERNAL_ERROR}.
 */
@RestControllerAdvice(basePackageClasses = V2EmbossingModule.class)
public class EmbossingExceptionHandler {

    private static final Logger log = LoggerFactory.getLogger(EmbossingExceptionHandler.class);

    @ExceptionHandler(EmbossingException.class)
    public ResponseEntity<ApiErrorEnvelope> handleEmbossing(EmbossingException exception) {
        switch (exception.kind()) {
            case INTERNAL -> log.error("Embossing failure code={}", exception.code(), exception);
            case NOT_FOUND, BAD_REQUEST ->
                    log.debug("Embossing rejection code={} message={}", exception.code(), exception.getMessage());
            case CONFLICT, UNPROCESSABLE ->
                    log.warn("Embossing rejection code={} message={}", exception.code(), exception.getMessage());
        }
        return ApiResponseHelper.status(status(exception.kind()), exception.code(), exception.getMessage());
    }

    @ExceptionHandler({
            MethodArgumentNotValidException.class,
            MethodArgumentTypeMismatchException.class,
            HttpMessageNotReadableException.class
    })
    public ResponseEntity<ApiErrorEnvelope> handleInvalidRequest(Exception exception) {
        log.warn("Embossing request could not be accepted");
        return ApiResponseHelper.status(HttpStatus.BAD_REQUEST, ApiErrorCode.VALIDATION_ERROR.name(),
                "Request could not be accepted");
    }

    @ExceptionHandler(NoResourceFoundException.class)
    public ResponseEntity<ApiErrorEnvelope> handleMissing(NoResourceFoundException exception) {
        return ApiResponseHelper.status(HttpStatus.NOT_FOUND, ApiErrorCode.NOT_FOUND.name(),
                "The requested resource was not found");
    }

    @ExceptionHandler(Exception.class)
    public ResponseEntity<ApiErrorEnvelope> handleUnexpected(Exception exception) {
        log.error("Unhandled embossing failure", exception);
        return ApiResponseHelper.status(HttpStatus.INTERNAL_SERVER_ERROR, ApiErrorCode.INTERNAL_ERROR.name(),
                "The embossing request could not be completed");
    }

    private static HttpStatus status(EmbossingErrorKind kind) {
        return switch (kind) {
            case BAD_REQUEST -> HttpStatus.BAD_REQUEST;
            case NOT_FOUND -> HttpStatus.NOT_FOUND;
            case CONFLICT -> HttpStatus.CONFLICT;
            case UNPROCESSABLE -> HttpStatus.UNPROCESSABLE_ENTITY;
            case INTERNAL -> HttpStatus.INTERNAL_SERVER_ERROR;
        };
    }
}
