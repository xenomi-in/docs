package com.lps.cms.embossing.perso.file.crypto;

import java.io.FilterOutputStream;
import java.io.IOException;
import java.io.OutputStream;
import java.security.SecureRandom;
import java.util.Date;

import org.bouncycastle.bcpg.ArmoredOutputStream;
import org.bouncycastle.bcpg.CompressionAlgorithmTags;
import org.bouncycastle.openpgp.PGPCompressedDataGenerator;
import org.bouncycastle.openpgp.PGPEncryptedData;
import org.bouncycastle.openpgp.PGPEncryptedDataGenerator;
import org.bouncycastle.openpgp.PGPLiteralData;
import org.bouncycastle.openpgp.PGPLiteralDataGenerator;
import org.bouncycastle.openpgp.operator.jcajce.JcePGPDataEncryptorBuilder;
import org.bouncycastle.openpgp.operator.jcajce.JcePublicKeyKeyEncryptionMethodGenerator;
import org.bouncycastle.jce.provider.BouncyCastleProvider;
import org.springframework.stereotype.Component;

/**
 * Encrypts a stream to one verified vendor public key using AES-256, an integrity packet,
 * ZIP compression, and ASCII armor.
 */
@Component
public class StreamingPgpEncryptor {

    private static final int BUFFER_SIZE = 64 * 1024;

    static {
        if (java.security.Security.getProvider(BouncyCastleProvider.PROVIDER_NAME) == null) {
            java.security.Security.addProvider(new BouncyCastleProvider());
        }
    }

    public OutputStream wrap(OutputStream target, String literalFileName, PgpEncryptionTarget encryptionTarget)
            throws Exception {
        OutputStream armored = new ArmoredOutputStream(target);
        PGPEncryptedDataGenerator encryptedGenerator = new PGPEncryptedDataGenerator(
                new JcePGPDataEncryptorBuilder(PGPEncryptedData.AES_256)
                        .setWithIntegrityPacket(true)
                        .setSecureRandom(new SecureRandom())
                        .setProvider(BouncyCastleProvider.PROVIDER_NAME));
        encryptedGenerator.addMethod(new JcePublicKeyKeyEncryptionMethodGenerator(encryptionTarget.publicKey())
                .setProvider(BouncyCastleProvider.PROVIDER_NAME));
        OutputStream encrypted = encryptedGenerator.open(armored, new byte[BUFFER_SIZE]);
        PGPCompressedDataGenerator compressedGenerator = new PGPCompressedDataGenerator(CompressionAlgorithmTags.ZIP);
        OutputStream compressed = compressedGenerator.open(encrypted);
        PGPLiteralDataGenerator literalGenerator = new PGPLiteralDataGenerator();
        OutputStream literal = literalGenerator.open(compressed, PGPLiteralData.BINARY,
                literalFileName == null || literalFileName.isBlank() ? "personalisation" : literalFileName,
                new Date(), new byte[BUFFER_SIZE]);

        return new FilterOutputStream(literal) {
            private boolean closed;

            @Override
            public void close() throws IOException {
                if (closed) {
                    return;
                }
                closed = true;
                IOException failure = null;
                for (OutputStream stream : java.util.List.of(literal, compressed, encrypted, armored)) {
                    try {
                        stream.close();
                    } catch (IOException exception) {
                        if (failure == null) {
                            failure = exception;
                        } else {
                            failure.addSuppressed(exception);
                        }
                    }
                }
                if (failure != null) {
                    throw failure;
                }
            }
        };
    }
}
