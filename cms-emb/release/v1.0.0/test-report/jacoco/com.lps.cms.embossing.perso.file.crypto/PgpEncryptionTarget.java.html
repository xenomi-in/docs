package com.lps.cms.embossing.perso.file.crypto;

import java.util.Objects;
import java.util.UUID;

import org.bouncycastle.openpgp.PGPPublicKey;

/** The verified active public key to use for one tenant/vendor output file. */
public record PgpEncryptionTarget(
        UUID institutionId,
        UUID vendorId,
        UUID keyId,
        String fingerprint,
        PGPPublicKey publicKey) {

    public PgpEncryptionTarget {
        Objects.requireNonNull(institutionId, "institutionId");
        Objects.requireNonNull(vendorId, "vendorId");
        Objects.requireNonNull(keyId, "keyId");
        Objects.requireNonNull(fingerprint, "fingerprint");
        Objects.requireNonNull(publicKey, "publicKey");
    }
}
