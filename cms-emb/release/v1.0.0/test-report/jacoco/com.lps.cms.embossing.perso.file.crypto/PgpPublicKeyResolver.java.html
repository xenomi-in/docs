package com.lps.cms.embossing.perso.file.crypto;

import java.io.ByteArrayInputStream;
import java.io.InputStream;
import java.security.MessageDigest;
import java.util.HexFormat;
import java.util.List;
import java.util.UUID;

import com.lps.cms.embossing.perso.exception.EmbossingErrorCodes;
import com.lps.cms.embossing.perso.exception.FileGenerationException;
import org.bouncycastle.openpgp.PGPPublicKey;
import org.bouncycastle.openpgp.PGPPublicKeyRing;
import org.bouncycastle.openpgp.PGPUtil;
import org.bouncycastle.openpgp.operator.jcajce.JcaKeyFingerprintCalculator;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.stereotype.Repository;

/**
 * Loads the single active public key for an institution and vendor from
 * {@code cms_v2.vendor_pgp_encryption_key} and verifies its fingerprint.
 */
@Repository
public class PgpPublicKeyResolver {

    private final JdbcTemplate jdbc;

    public PgpPublicKeyResolver(JdbcTemplate jdbc) {
        this.jdbc = jdbc;
    }

    public PgpEncryptionTarget resolve(UUID institutionId, UUID vendorId) {
        List<KeyRow> rows = jdbc.query("""
                SELECT vendor_pgp_encryption_key_id, fingerprint, armored_public_key
                FROM cms_v2.vendor_pgp_encryption_key
                WHERE institution_id = ?
                  AND vendor_id = ?
                  AND key_status = 'ACTIVE'
                  AND effective_from <= now()
                  AND (effective_to IS NULL OR effective_to > now())
                ORDER BY effective_from DESC
                LIMIT 2
                """, (rs, rowNum) -> new KeyRow(
                rs.getObject("vendor_pgp_encryption_key_id", UUID.class),
                rs.getString("fingerprint"),
                rs.getString("armored_public_key")), institutionId, vendorId);
        if (rows.size() != 1) {
            throw new FileGenerationException(EmbossingErrorCodes.PGP_KEY_UNAVAILABLE,
                    "Expected exactly one active PGP encryption key for tenant="
                            + institutionId + " vendor=" + vendorId + "; found " + rows.size());
        }
        KeyRow row = rows.getFirst();
        PGPPublicKey key = parseEncryptionKey(row.armoredPublicKey());
        String actualFingerprint = HexFormat.of().withUpperCase().formatHex(key.getFingerprint());
        if (!MessageDigest.isEqual(actualFingerprint.getBytes(java.nio.charset.StandardCharsets.US_ASCII),
                row.fingerprint().trim().toUpperCase(java.util.Locale.ROOT)
                        .getBytes(java.nio.charset.StandardCharsets.US_ASCII))) {
            throw new FileGenerationException(EmbossingErrorCodes.PGP_KEY_INVALID,
                    "Configured PGP key fingerprint does not match key material for key=" + row.keyId());
        }
        return new PgpEncryptionTarget(institutionId, vendorId, row.keyId(), actualFingerprint, key);
    }

    private static PGPPublicKey parseEncryptionKey(String armoredKey) {
        if (armoredKey == null || armoredKey.isBlank()) {
            throw new FileGenerationException(EmbossingErrorCodes.PGP_KEY_INVALID,
                    "Configured PGP public key material is empty");
        }
        try (InputStream input = PGPUtil.getDecoderStream(
                new ByteArrayInputStream(armoredKey.getBytes(java.nio.charset.StandardCharsets.US_ASCII)))) {
            PGPPublicKeyRing ring = new PGPPublicKeyRing(input, new JcaKeyFingerprintCalculator());
            for (java.util.Iterator<PGPPublicKey> keys = ring.getPublicKeys(); keys.hasNext();) {
                PGPPublicKey key = keys.next();
                if (key.isEncryptionKey() && !key.hasRevocation()) {
                    return key;
                }
            }
            throw new FileGenerationException(EmbossingErrorCodes.PGP_KEY_INVALID,
                    "Configured PGP key ring has no usable encryption key");
        } catch (FileGenerationException exception) {
            throw exception;
        } catch (Exception exception) {
            throw new FileGenerationException(EmbossingErrorCodes.PGP_KEY_INVALID,
                    "Configured PGP public key is invalid", exception);
        }
    }

    private record KeyRow(UUID keyId, String fingerprint, String armoredPublicKey) {
    }
}
