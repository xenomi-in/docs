package com.lps.cms.embossing;

import com.lps.cms.embossing.infra.config.OpenApiConfig;
import com.lps.cms.embossing.perso.V2EmbossingModule;
import org.springframework.boot.persistence.autoconfigure.EntityScan;
import org.springframework.boot.SpringApplication;
import org.springframework.boot.autoconfigure.SpringBootApplication;
import org.springframework.data.jpa.repository.config.EnableJpaRepositories;
import org.springframework.scheduling.annotation.EnableScheduling;

/**
 * Spring Boot entry point. Component scanning, JPA, and scheduling are limited to
 * {@link com.lps.cms.embossing.perso.V2EmbossingModule} and OpenAPI configuration.
 *
 * @author LogicPoint Systems
 */
@SpringBootApplication(scanBasePackageClasses = {V2EmbossingModule.class, OpenApiConfig.class})
@EntityScan(basePackageClasses = V2EmbossingModule.class)
@EnableJpaRepositories(basePackageClasses = V2EmbossingModule.class)
@EnableScheduling
public class EmbossingApplication {

	public static void main(String[] args) {
		SpringApplication.run(EmbossingApplication.class, args);
	}

}
