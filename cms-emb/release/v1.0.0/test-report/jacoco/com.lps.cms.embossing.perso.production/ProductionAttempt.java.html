package com.lps.cms.embossing.perso.production;

import jakarta.persistence.Column;
import jakarta.persistence.Entity;
import jakarta.persistence.Id;
import jakarta.persistence.Table;
import lombok.Getter;
import lombok.Setter;

import java.time.OffsetDateTime;
import java.util.UUID;

@Entity
@Table(name = "production_attempt", schema = "cms_v2")
@Getter
@Setter
public class ProductionAttempt {

    @Id
    @Column(name = "attempt_id")
    private UUID attemptId;

    @Column(name = "institution_id", nullable = false)
    private UUID institutionId;

    @Column(name = "production_request_id", nullable = false)
    private UUID productionRequestId;

    @Column(name = "card_id", nullable = false)
    private UUID cardId;

    @Column(name = "attempt_number", nullable = false)
    private int attemptNumber;

    @Column(name = "batch_item_id")
    private UUID batchItemId;

    @Column(name = "vendor_reference", length = 255)
    private String vendorReference;

    @Column(name = "attempt_status", nullable = false, length = 32)
    private String attemptStatus;

    @Column(name = "failure_code", length = 50)
    private String failureCode;

    @Column(name = "failure_message", length = 500)
    private String failureMessage;

    @Column(name = "started_at", nullable = false)
    private OffsetDateTime startedAt;

    @Column(name = "completed_at")
    private OffsetDateTime completedAt;
}
