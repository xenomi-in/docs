package com.lps.cms.embossing.perso.file.transmit;

import java.io.IOException;
import java.nio.file.Path;

import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import net.schmizz.sshj.SSHClient;
import net.schmizz.sshj.sftp.SFTPClient;
import net.schmizz.sshj.transport.verification.PromiscuousVerifier;
import net.schmizz.sshj.userauth.keyprovider.KeyProvider;
import org.springframework.stereotype.Component;

/** Opens one SSH session, writes one file under the remote drop directory, and disconnects. */
@Slf4j
@Component
@RequiredArgsConstructor
public class SshjSftpFileUploader implements SftpFileUploader {

    private final SftpTransmissionProperties properties;

    @Override
    public void upload(Path localFile, String remoteFileName) throws IOException {
        SSHClient ssh = new SSHClient();
        try {
            ssh.setConnectTimeout(properties.getConnectionTimeoutMs());
            if (!properties.isStrictHostKeyChecking()) {
                ssh.addHostKeyVerifier(new PromiscuousVerifier());
            }
            ssh.connect(properties.getHost(), properties.getPort());
            authenticate(ssh);
            String remotePath = remoteDirectory() + "/" + remoteFileName;
            try (SFTPClient sftp = ssh.newSFTPClient()) {
                sftp.put(localFile.toAbsolutePath().toString(), remotePath);
            }
            log.info("SFTP upload succeeded host={}:{} remote={}", properties.getHost(), properties.getPort(), remotePath);
        } finally {
            disconnect(ssh);
        }
    }

    private void authenticate(SSHClient ssh) throws IOException {
        String privateKeyPath = properties.getPrivateKeyPath();
        if (privateKeyPath != null && !privateKeyPath.isBlank()) {
            KeyProvider keyProvider = ssh.loadKeys(privateKeyPath);
            ssh.authPublickey(properties.getUsername(), keyProvider);
            return;
        }
        ssh.authPassword(properties.getUsername(), properties.getPassword());
    }

    private String remoteDirectory() {
        String path = properties.getRemotePath();
        if (path == null || path.isBlank()) {
            return "/upload";
        }
        return path.endsWith("/") ? path.substring(0, path.length() - 1) : path;
    }

    private static void disconnect(SSHClient ssh) {
        try {
            if (ssh.isConnected()) {
                ssh.disconnect();
            }
        } catch (IOException ex) {
            log.warn("Error disconnecting SSH session", ex);
        }
    }
}
