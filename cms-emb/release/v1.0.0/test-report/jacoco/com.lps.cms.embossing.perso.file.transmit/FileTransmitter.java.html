package com.lps.cms.embossing.perso.file.transmit;

import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.time.Instant;

import com.lps.cms.embossing.perso.event.catalog.EmbossingEventSink;
import com.lps.cms.embossing.perso.file.transmit.FileTransmissionStore.ReadyFile;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Service;

/**
 * Uploads encrypted personalisation files whose batch configuration uses SFTP.
 * Each file is transmitted independently.
 */
@Slf4j
@Service
public class FileTransmitter {

    private final FileTransmissionStore files;
    private final SftpFileUploader uploader;
    private final SftpTransmissionProperties properties;
    private final EmbossingEventSink events;
    private final Path outputBasePath;

    public FileTransmitter(FileTransmissionStore files, SftpFileUploader uploader,
                           SftpTransmissionProperties properties, EmbossingEventSink events,
                           @Value("${embossing.output-path:./embossing-output}") Path outputBasePath) {
        this.files = files;
        this.uploader = uploader;
        this.properties = properties;
        this.events = events;
        this.outputBasePath = outputBasePath;
    }

    public int transmitReady() {
        if (!properties.isEnabled()) {
            log.info("SFTP transmission is disabled");
            return 0;
        }
        if (properties.getHost() == null || properties.getHost().isBlank()
                || properties.getUsername() == null || properties.getUsername().isBlank()) {
            log.error("SFTP transmission is enabled without a host and username");
            return 0;
        }
        int sent = 0;
        for (ReadyFile file : files.readySftpFiles()) {
            if (send(file)) {
                sent++;
            }
        }
        return sent;
    }

    private boolean send(ReadyFile file) {
        try {
            Path localFile = localFile(file);
            uploadWithRetry(localFile, file.fileName());
            files.markSent(file);
            events.fileTransmitted(file.institutionId(), file.fileId(), file.batchId(), file.vendorId(),
                    file.recordCount(), Instant.now().toString());
            log.info("Transmitted personalisation file {} for batch {}", file.fileId(), file.batchId());
            return true;
        } catch (RuntimeException | IOException ex) {
            log.error("SFTP transmission failed for file {} batch {}", file.fileId(), file.batchId(), ex);
            files.markFailed(file, ex.getMessage());
            events.transmissionFailed(file.institutionId(), file.fileId(), file.batchId(), file.vendorId(),
                    file.recordCount(), "SFTP_UPLOAD_FAILED");
            return false;
        }
    }

    private void uploadWithRetry(Path localFile, String remoteFileName) throws IOException {
        int attempts = Math.max(properties.getMaxRetries(), 1);
        IOException lastFailure = null;
        for (int attempt = 1; attempt <= attempts; attempt++) {
            try {
                uploader.upload(localFile, remoteFileName);
                return;
            } catch (IOException ex) {
                lastFailure = ex;
                log.warn("SFTP upload attempt {}/{} failed for {}", attempt, attempts, remoteFileName);
            }
        }
        throw lastFailure == null ? new IOException("SFTP upload failed") : lastFailure;
    }

    private Path localFile(ReadyFile file) {
        Path base = outputBasePath.toAbsolutePath().normalize();
        Path local = base.resolve(file.objectReference()).normalize();
        if (!local.startsWith(base) || !Files.isRegularFile(local)) {
            throw new IllegalStateException("Personalisation file is not in the output directory: " + file.fileId());
        }
        return local;
    }
}
