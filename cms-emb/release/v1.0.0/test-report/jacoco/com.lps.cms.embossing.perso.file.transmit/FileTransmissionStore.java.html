package com.lps.cms.embossing.perso.file.transmit;

import java.util.List;
import java.util.UUID;

import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.stereotype.Repository;
import org.springframework.transaction.annotation.Transactional;

/** Reads encrypted SFTP files and records the result of one upload attempt. */
@Repository
public class FileTransmissionStore {

    private final JdbcTemplate jdbc;

    public FileTransmissionStore(JdbcTemplate jdbc) {
        this.jdbc = jdbc;
    }

    public List<ReadyFile> readySftpFiles() {
        return jdbc.query("""
                SELECT f.personalisation_file_id, f.institution_id, f.batch_id, b.vendor_id,
                       f.file_name, f.secure_object_reference, f.record_count
                FROM cms_v2.personalisation_file f
                JOIN cms_v2.personalisation_batch b
                  ON b.batch_id = f.batch_id AND b.institution_id = f.institution_id
                JOIN cms_v2.personalisation_batch_configuration_version v
                  ON v.batch_configuration_version_id = b.batch_configuration_version_id
                 AND v.institution_id = b.institution_id
                WHERE f.file_status = 'ENCRYPTED'
                  AND v.transmission_channel = 'SFTP'
                ORDER BY f.generated_at
                """, (rs, row) -> new ReadyFile(
                rs.getObject("personalisation_file_id", UUID.class),
                rs.getObject("institution_id", UUID.class),
                rs.getObject("batch_id", UUID.class),
                rs.getObject("vendor_id", UUID.class),
                rs.getString("file_name"),
                rs.getString("secure_object_reference"),
                rs.getInt("record_count")));
    }

    @Transactional
    public void markSent(ReadyFile file) {
        int updated = jdbc.update("""
                UPDATE cms_v2.personalisation_file
                SET file_status = 'SENT'
                WHERE personalisation_file_id = ? AND institution_id = ? AND file_status = 'ENCRYPTED'
                """, file.fileId(), file.institutionId());
        if (updated != 1) {
            throw new IllegalStateException("Personalisation file was no longer waiting for transmission: " + file.fileId());
        }
        jdbc.update("""
                UPDATE cms_v2.personalisation_batch
                SET batch_status = 'SENT', updated_at = now(), updated_by = 'vendor-service'
                WHERE batch_id = ? AND institution_id = ?
                  AND batch_status IN ('FILE_GENERATED', 'TRANSMISSION_FAILED')
                """, file.batchId(), file.institutionId());
        jdbc.update("""
                UPDATE cms_v2.card_production_request r
                SET request_status = 'SENT_TO_VENDOR', updated_at = now()
                FROM cms_v2.personalisation_batch_item i
                WHERE i.batch_id = ? AND i.institution_id = ?
                  AND r.institution_id = i.institution_id
                  AND r.production_request_id = i.production_request_id
                  AND r.request_status = 'FILE_GENERATED'
                """, file.batchId(), file.institutionId());
        insertAttempt(file, "SUCCEEDED", null, null);
    }

    @Transactional
    public void markFailed(ReadyFile file, String errorMessage) {
        jdbc.update("""
                UPDATE cms_v2.personalisation_batch
                SET batch_status = 'TRANSMISSION_FAILED', updated_at = now(), updated_by = 'vendor-service'
                WHERE batch_id = ? AND institution_id = ?
                  AND batch_status IN ('FILE_GENERATED', 'TRANSMISSION_FAILED')
                """, file.batchId(), file.institutionId());
        insertAttempt(file, "FAILED", "SFTP_UPLOAD_FAILED", trim(errorMessage));
    }

    private void insertAttempt(ReadyFile file, String status, String errorCode, String errorMessage) {
        Integer next = jdbc.queryForObject("""
                SELECT coalesce(max(attempt_number), 0) + 1
                FROM cms_v2.file_transmission_attempt
                WHERE personalisation_file_id = ?
                """, Integer.class, file.fileId());
        jdbc.update("""
                INSERT INTO cms_v2.file_transmission_attempt (
                    transmission_attempt_id, institution_id, personalisation_file_id, attempt_number,
                    transmission_channel, remote_file_name, transmission_status, started_at, completed_at,
                    error_code, error_message)
                VALUES (?, ?, ?, ?, 'SFTP', ?, ?, now(), now(), ?, ?)
                """, UUID.randomUUID(), file.institutionId(), file.fileId(), next == null ? 1 : next,
                file.fileName(), status, errorCode, errorMessage);
    }

    private static String trim(String message) {
        if (message == null || message.isBlank()) {
            return "SFTP upload failed";
        }
        return message.length() <= 500 ? message : message.substring(0, 500);
    }

    public record ReadyFile(
            UUID fileId,
            UUID institutionId,
            UUID batchId,
            UUID vendorId,
            String fileName,
            String objectReference,
            int recordCount) {
    }
}
