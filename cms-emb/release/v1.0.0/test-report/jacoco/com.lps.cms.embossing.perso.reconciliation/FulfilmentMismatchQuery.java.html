package com.lps.cms.embossing.perso.reconciliation;

import java.util.List;

import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.stereotype.Repository;

/** Reads fulfilment mismatches. It does not repair them. */
@Repository
public class FulfilmentMismatchQuery {

    private final JdbcTemplate jdbc;

    public FulfilmentMismatchQuery(JdbcTemplate jdbc) {
        this.jdbc = jdbc;
    }

    public List<String> find() {
        return jdbc.query("""
                SELECT 'FILE_SENT_REQUEST_NOT_SENT:' || f.personalisation_file_id
                FROM cms_v2.personalisation_file f
                JOIN cms_v2.personalisation_batch_item i ON i.batch_id = f.batch_id
                JOIN cms_v2.card_production_request r ON r.production_request_id = i.production_request_id
                WHERE f.file_status = 'SENT'
                  AND r.request_status NOT IN ('SENT_TO_VENDOR', 'VENDOR_ACCEPTED', 'IN_PRODUCTION', 'COMPLETED')
                UNION ALL
                SELECT 'RESERVED_WITHOUT_ITEM:' || r.production_request_id
                FROM cms_v2.card_production_request r
                LEFT JOIN cms_v2.personalisation_batch_item i ON i.production_request_id = r.production_request_id
                WHERE r.request_status = 'RESERVED_FOR_BATCH' AND i.batch_item_id IS NULL
                """, (rs, row) -> rs.getString(1));
    }
}
