package com.lps.cms.embossing.perso.reconciliation;

import lombok.RequiredArgsConstructor;
import org.springframework.scheduling.annotation.Scheduled;
import org.springframework.stereotype.Component;

import java.time.OffsetDateTime;
import java.time.ZoneOffset;
import java.util.UUID;

/** Records each unresolved fulfilment mismatch reported by {@link FulfilmentMismatchQuery}. */
@Component
@RequiredArgsConstructor
public class ReconciliationJob {

    private final FulfilmentMismatchQuery mismatches;
    private final ReconciliationExceptionRepository exceptions;

    @Scheduled(cron = "${embossing.v2.reconcile-cron:0 0 * * * *}")
    public void reconcile() {
        for (String mismatch : mismatches.find()) {
            int split = mismatch.indexOf(':');
            String code = mismatch.substring(0, split);
            String id = mismatch.substring(split + 1);
            if (exceptions.existsByExceptionCodeAndAggregateIdAndResolvedAtIsNull(code, id)) {
                continue;
            }
            ReconciliationException row = new ReconciliationException();
            row.setExceptionId(UUID.randomUUID());
            row.setExceptionCode(code);
            row.setAggregateType("PRODUCTION_REQUEST");
            row.setAggregateId(id);
            row.setDetail(mismatch);
            row.setDetectedAt(OffsetDateTime.now(ZoneOffset.UTC));
            exceptions.save(row);
        }
    }
}
