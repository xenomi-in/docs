package com.lps.cms.embossing.perso.cms.request;

import java.util.List;
import java.util.UUID;

import com.lps.cms.embossing.perso.cms.PgUuidArray;
import com.lps.cms.embossing.perso.exception.EmbossingErrorCodes;
import com.lps.cms.embossing.perso.exception.FulfilmentConflictException;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.stereotype.Repository;

/** Claims and advances {@code cms_v2.card_production_request} rows for one fulfilment run. */
@Repository
public class ProductionRequestStore {

    private final JdbcTemplate jdbc;

    public ProductionRequestStore(JdbcTemplate jdbc) {
        this.jdbc = jdbc;
    }

    public List<EligibleRequest> lockEligible(int limit) {
        return jdbc.query("""
                SELECT r.production_request_id, r.institution_id, r.card_id, r.vendor_id,
                       r.batch_configuration_id, r.request_type, c.product_id
                FROM cms_v2.card_production_request r
                JOIN cms_v2.card c ON c.institution_id = r.institution_id AND c.card_id = r.card_id
                WHERE r.request_status = 'ELIGIBLE'
                  AND r.card_data_ready AND r.address_ready AND r.approval_ready
                  AND r.switch_onboarding_ready
                  AND r.eligible_at <= now()
                ORDER BY r.eligible_at, r.requested_at, r.production_request_id
                FOR UPDATE OF r SKIP LOCKED
                LIMIT ?
                """, (rs, row) -> new EligibleRequest(
                rs.getObject("production_request_id", UUID.class),
                rs.getObject("institution_id", UUID.class),
                rs.getObject("card_id", UUID.class),
                rs.getObject("vendor_id", UUID.class),
                rs.getObject("batch_configuration_id", UUID.class),
                rs.getString("request_type"),
                rs.getObject("product_id", UUID.class)), limit);
    }

    public int reserve(List<UUID> requestIds) {
        if (requestIds.isEmpty()) {
            return 0;
        }
        return jdbc.update("""
                UPDATE cms_v2.card_production_request
                SET request_status = 'RESERVED_FOR_BATCH', updated_at = now()
                WHERE production_request_id = ANY(?)
                  AND request_status = 'ELIGIBLE'
                """, PgUuidArray.of(jdbc, requestIds));
    }

    public void release(List<UUID> requestIds) {
        if (requestIds.isEmpty()) {
            return;
        }
        jdbc.update("""
                UPDATE cms_v2.card_production_request
                SET request_status = 'ELIGIBLE', updated_at = now()
                WHERE production_request_id = ANY(?)
                  AND request_status = 'RESERVED_FOR_BATCH'
                """, PgUuidArray.of(jdbc, requestIds));
    }

    public void markFileGenerated(List<UUID> requestIds) {
        if (requestIds.isEmpty()) {
            return;
        }
        int updated = jdbc.update("""
                UPDATE cms_v2.card_production_request
                SET request_status = 'FILE_GENERATED', updated_at = now()
                WHERE production_request_id = ANY(?)
                  AND request_status = 'RESERVED_FOR_BATCH'
                """, PgUuidArray.of(jdbc, requestIds));
        if (updated != requestIds.size()) {
            throw new FulfilmentConflictException(EmbossingErrorCodes.REQUEST_STATUS_CHANGED,
                    "Refusing to mark file generated; reserved request set changed");
        }
    }
}
