package com.lps.cms.embossing.perso.event.model;

import java.nio.charset.StandardCharsets;
import java.util.StringJoiner;
import java.util.UUID;

/**
 * Deterministic outbox idempotency keys built from domain identity. The key is unique per
 * {@code (institution, eventType, domain fact)} so a replay produces the same key and is
 * de-duplicated by {@code uk_outbox_idempotency}.
 */
public final class IdempotencyKeys {

    private IdempotencyKeys() {
    }

    public static UUID of(UUID institutionId, String eventType, Object... domainParts) {
        StringJoiner joiner = new StringJoiner("|");
        joiner.add("cms-v2-outbox");
        joiner.add(String.valueOf(institutionId));
        joiner.add(eventType);
        for (Object part : domainParts) {
            joiner.add(part == null ? "" : String.valueOf(part));
        }
        return UUID.nameUUIDFromBytes(joiner.toString().getBytes(StandardCharsets.UTF_8));
    }
}
