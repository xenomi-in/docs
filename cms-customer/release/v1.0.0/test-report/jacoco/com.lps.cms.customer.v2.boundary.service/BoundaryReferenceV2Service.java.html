package com.lps.cms.customer.v2.boundary.service;

    @Transactional
    public CustomerReferenceV2Response createCustomer(UUID institutionId, CreateCustomerReferenceV2Request request) {
        String actor = tenantAccess.require(institutionId);
        Instant now = Instant.now();
        try {
            return customerResponse(customers.saveAndFlush(CustomerReferenceV2.builder()
                    .institutionId(institutionId).sourceSystem(normalize(request.sourceSystem()))
                    .externalCustomerId(request.externalCustomerId().trim())
                    .customerType(request.customerType()).title(trimToNull(request.title()))
                    .firstName(request.firstName().trim()).middleName(trimToNull(request.middleName()))
                    .lastName(request.lastName().trim()).dateOfBirth(request.dateOfBirth())
                    .mobileCountryCode(trimToNull(request.mobileCountryCode()))
                    .mobileNumber(trimToNull(request.mobileNumber()))
                    .email(normalizeEmail(request.email()))
                    .status(request.status() == null ? ReferenceStatus.ACTIVE : request.status())
                    .createdAt(now).updatedAt(now).createdBy(actor).updatedBy(actor).build()));
        } catch (DataIntegrityViolationException ex) {
            throw new V2ConflictException("Customer reference already exists for institution, sourceSystem and externalCustomerId", ex);
        }
    }

    @Transactional(readOnly = true)
    public CustomerReferenceV2Response getCustomer(UUID institutionId, UUID id) {
        tenantAccess.require(institutionId);
        return customerResponse(customers.findByIdAndInstitutionId(id, institutionId)
                .orElseThrow(() -> new ResourceNotFoundException("CustomerReference", "customerReferenceId", id)));
    }

    @Transactional(readOnly = true)
    public CustomerReferenceV2Response resolveCustomer(UUID institutionId, String sourceSystem, String externalId) {
        tenantAccess.require(institutionId);
        return customerResponse(customers.findByInstitutionIdAndSourceSystemAndExternalCustomerId(
                institutionId, normalize(sourceSystem), externalId.trim())
                .orElseThrow(() -> new ResourceNotFoundException("CustomerReference", "externalCustomerId", externalId)));
    }

    @Transactional(readOnly = true)
    public PagedResponseV2<CustomerReferenceV2Response> listCustomers(
            UUID institutionId, ReferenceStatus status, Integer page, Integer limit) {
        tenantAccess.require(institutionId);
        var pageable = page(page, limit);
        var result = status == null ? customers.findByInstitutionId(institutionId, pageable)
                : customers.findByInstitutionIdAndStatus(institutionId, status, pageable);
        return PagedResponseV2.from(result.map(this::customerResponse));
    }

    @Transactional
    public CustomerReferenceV2Response updateCustomer(
            UUID institutionId, UUID id, UpdateReferenceStatusV2Request request) {
        String actor = tenantAccess.require(institutionId);
        var entity = customers.findByIdAndInstitutionId(id, institutionId)
                .orElseThrow(() -> new ResourceNotFoundException("CustomerReference", "customerReferenceId", id));
        validateReferenceTransition(entity.getStatus(), request.status());
        entity.setStatus(request.status()); entity.setUpdatedAt(Instant.now()); entity.setUpdatedBy(actor);
        return customerResponse(customers.save(entity));
    }

    @Transactional
    public CustomerReferenceV2Response updateCustomerProfile(
            UUID institutionId, UUID id, UpdateCustomerProfileV2Request request) {
        String actor = tenantAccess.require(institutionId);
        requireCustomerProfileChange(request);
        var entity = customers.findByIdAndInstitutionId(id, institutionId)
                .orElseThrow(() -> new ResourceNotFoundException(
                        "CustomerReference", "customerReferenceId", id));
        if (entity.getStatus() == ReferenceStatus.CLOSED) {
            throw new V2BusinessRuleException("A CLOSED customer profile cannot be updated");
        }
        if (request.title() != null) entity.setTitle(trimToNull(request.title()));
        if (request.firstName() != null) entity.setFirstName(request.firstName().trim());
        if (request.middleName() != null) entity.setMiddleName(trimToNull(request.middleName()));
        if (request.lastName() != null) entity.setLastName(request.lastName().trim());
        if (request.dateOfBirth() != null) entity.setDateOfBirth(request.dateOfBirth());
        if (request.mobileCountryCode() != null) entity.setMobileCountryCode(request.mobileCountryCode().trim());
        if (request.mobileNumber() != null) entity.setMobileNumber(request.mobileNumber().trim());
        if (request.email() != null) entity.setEmail(normalizeEmail(request.email()));
        entity.setUpdatedAt(Instant.now());
        entity.setUpdatedBy(actor);
        return customerResponse(customers.save(entity));
    }

    @Transactional
    public CustomerAddressReferenceV2Response createAddress(
            UUID institutionId, UUID customerId, CreateCustomerAddressReferenceV2Request request) {
        String actor = tenantAccess.require(institutionId);
        var customer = customers.findByIdAndInstitutionId(customerId, institutionId)
                .orElseThrow(() -> new ResourceNotFoundException(
                        "CustomerReference", "customerReferenceId", customerId));
        if (customer.getStatus() != ReferenceStatus.ACTIVE) {
            throw new V2BusinessRuleException("Customer reference must be ACTIVE");
        }
        try {
            Instant now = Instant.now();
            var entity = CustomerAddressReferenceV2.builder()
                    .institutionId(institutionId)
                    .customerReferenceId(customerId)
                    .sourceSystem(normalize(request.sourceSystem()))
                    .externalAddressId(request.externalAddressId().trim())
                    .addressType(request.addressType())
                    .addressLine1(request.addressLine1().trim())
                    .addressLine2(trimToNull(request.addressLine2()))
                    .landmark(trimToNull(request.landmark()))
                    .city(request.city().trim())
                    .district(trimToNull(request.district()))
                    .stateCode(normalize(request.stateCode()))
                    .postalCode(request.postalCode().trim())
                    .countryCode(normalize(request.countryCode()))
                    .primary(Boolean.TRUE.equals(request.primary()))
                    .status(request.status() == null ? ReferenceStatus.ACTIVE : request.status())
                    .createdAt(now).updatedAt(now).createdBy(actor).updatedBy(actor)
                    .build();
            return addressResponse(addresses.saveAndFlush(entity));
        } catch (DataIntegrityViolationException ex) {
            throw new V2ConflictException(
                    "Address reference already exists for institution, sourceSystem and externalAddressId", ex);
        }
    }

        requireCustomer(institutionId, customerId);
        var pageable = page(page, limit);
        var result = status == null
                ? addresses.findByInstitutionIdAndCustomerReferenceId(institutionId, customerId, pageable)
                : addresses.findByInstitutionIdAndCustomerReferenceIdAndStatus(

    }

    @Transactional
    public CustomerAddressReferenceV2Response updateAddress(
            UUID institutionId, UUID customerId, UUID addressId,
            UpdateCustomerAddressV2Request request) {
        String actor = tenantAccess.require(institutionId);
        requireAddressChange(request);
        var entity = addressEntity(institutionId, customerId, addressId);
        if (entity.getStatus() == ReferenceStatus.CLOSED) {
            throw new V2BusinessRuleException("A CLOSED address cannot be updated");
        }
        if (request.addressType() != null) entity.setAddressType(request.addressType());
        if (request.addressLine1() != null) entity.setAddressLine1(request.addressLine1().trim());
        if (request.addressLine2() != null) entity.setAddressLine2(trimToNull(request.addressLine2()));
        if (request.landmark() != null) entity.setLandmark(trimToNull(request.landmark()));
        if (request.city() != null) entity.setCity(request.city().trim());
        if (request.district() != null) entity.setDistrict(trimToNull(request.district()));
        if (request.stateCode() != null) entity.setStateCode(normalize(request.stateCode()));
        if (request.postalCode() != null) entity.setPostalCode(request.postalCode().trim());
        if (request.countryCode() != null) entity.setCountryCode(normalize(request.countryCode()));
        if (request.primary() != null) entity.setPrimary(request.primary());
        entity.setUpdatedAt(Instant.now());
        entity.setUpdatedBy(actor);
        try {
            return addressResponse(addresses.saveAndFlush(entity));
        } catch (DataIntegrityViolationException ex) {
            throw new V2ConflictException("Only one ACTIVE primary address is allowed per customer", ex);
        }
    }

    @Transactional
    public CustomerAddressReferenceV2Response updateAddressStatus(

        entity.setUpdatedAt(Instant.now());

        return addressResponse(addresses.save(entity));
    }

    @Transactional
    public CustomerAccountLinkV2Response linkAccount(
            UUID institutionId, UUID customerId, CreateCustomerAccountLinkV2Request request) {
        String actor = tenantAccess.require(institutionId);
        var customer = customers.findByIdAndInstitutionId(customerId, institutionId)
                .orElseThrow(() -> new ResourceNotFoundException("CustomerReference", "customerReferenceId", customerId));
        var account = accounts.findByIdAndInstitutionId(request.externalAccountReferenceId(), institutionId)
                .orElseThrow(() -> new ResourceNotFoundException("ExternalAccountReference",
                        "externalAccountReferenceId", request.externalAccountReferenceId()));
        if (customer.getStatus() != ReferenceStatus.ACTIVE || account.getStatus() != ReferenceStatus.ACTIVE) {
            throw new V2BusinessRuleException("Customer and account references must both be ACTIVE");
        }
        validateRelationship(account.getAccountType(), request.relationshipType());
        if (links.findByInstitutionIdAndCustomerReferenceIdAndExternalAccountReferenceIdAndStatus(
                institutionId, customerId, account.getId(), LinkStatus.ACTIVE).isPresent()) {
            throw new V2ConflictException("An active customer-account relationship already exists");
        }
        Instant now = Instant.now();
        var link = CustomerAccountLinkV2.builder()
                .institutionId(institutionId).customerReferenceId(customerId)
                .externalAccountReferenceId(account.getId()).relationshipType(request.relationshipType())
                .primary(request.primary()).status(LinkStatus.ACTIVE)
                .effectiveFrom(request.effectiveFrom() == null ? now : request.effectiveFrom())
                .createdAt(now).updatedAt(now).createdBy(actor).updatedBy(actor).build();
        return linkResponse(links.saveAndFlush(link), account.getAccountType());
    }

        }
        return PagedResponseV2.from(links.findByInstitutionIdAndCustomerReferenceId(
                institutionId, customerId, page(page, limit)).map(this::linkResponse));
    }

    @Transactional
    public CustomerAccountLinkV2Response updateLinkStatus(UUID institutionId, UUID customerId, UUID accountId,
            UpdateCustomerAccountLinkStatusV2Request request) {
        String actor = tenantAccess.require(institutionId);
        var link = links.findByInstitutionIdAndCustomerReferenceIdAndExternalAccountReferenceIdAndStatus(
                        institutionId, customerId, accountId, LinkStatus.ACTIVE)
                .orElseThrow(() -> new ResourceNotFoundException("ActiveCustomerAccountLink", "externalAccountReferenceId", accountId));
        if (request.status() == LinkStatus.ACTIVE) return linkResponse(link);
        Instant effectiveTo = request.effectiveTo() == null ? Instant.now() : request.effectiveTo();
        if (!effectiveTo.isAfter(link.getEffectiveFrom())) {
            throw new V2BusinessRuleException("effectiveTo must be later than effectiveFrom");
        }
        link.setStatus(request.status()); link.setEffectiveTo(effectiveTo);
        link.setUpdatedAt(Instant.now()); link.setUpdatedBy(actor);
        return linkResponse(links.save(link));
    }

    private static PageRequest page(Integer page, Integer limit) {
        int p = page == null ? 1 : page;
        int l = limit == null ? 20 : limit;
        if (p < 1 || l < 1 || l > 100) throw new BadRequestException("page must be >= 1 and limit must be between 1 and 100");
        return PageRequest.of(p - 1, l, Sort.by(Sort.Direction.DESC, "createdAt"));
    }
    private static String normalize(String value) { return value.trim().toUpperCase(java.util.Locale.ROOT); }
    private static String normalizeEmail(String value) {
        String normalized = trimToNull(value);
        return normalized == null ? null : normalized.toLowerCase(java.util.Locale.ROOT);
    }
    private static String trimToNull(String value) {
        if (value == null) return null;
        String trimmed = value.trim();
        return trimmed.isEmpty() ? null : trimmed;
    }
    private CustomerReferenceV2 requireCustomer(UUID institutionId, UUID customerId) {
        return customers.findByIdAndInstitutionId(customerId, institutionId)
                .orElseThrow(() -> new ResourceNotFoundException(
                        "CustomerReference", "customerReferenceId", customerId));
    }
    private CustomerAddressReferenceV2 addressEntity(UUID institutionId, UUID customerId, UUID addressId) {
        requireCustomer(institutionId, customerId);
        return addresses.findByIdAndInstitutionIdAndCustomerReferenceId(addressId, institutionId, customerId)
                .orElseThrow(() -> new ResourceNotFoundException(
                        "CustomerAddress", "addressReferenceId", addressId));
    }
    private static void requireCustomerProfileChange(UpdateCustomerProfileV2Request request) {
        if (request.title() == null && request.firstName() == null && request.middleName() == null
                && request.lastName() == null && request.dateOfBirth() == null
                && request.mobileCountryCode() == null && request.mobileNumber() == null
                && request.email() == null) {
            throw new BadRequestException("At least one customer profile field must be provided");
        }
    }
    private static void requireAddressChange(UpdateCustomerAddressV2Request request) {
        if (request.addressType() == null && request.addressLine1() == null
                && request.addressLine2() == null && request.landmark() == null
                && request.city() == null && request.district() == null
                && request.stateCode() == null && request.postalCode() == null
                && request.countryCode() == null && request.primary() == null) {
            throw new BadRequestException("At least one address field must be provided");
        }
    }
    private static void validateRelationship(AccountType accountType, CustomerAccountRelationshipType relationship) {
        boolean valid = accountType == AccountType.CREDIT_CARD_ACCOUNT
                ? relationship == CustomerAccountRelationshipType.PRIMARY_HOLDER
                    || relationship == CustomerAccountRelationshipType.SUPPLEMENTARY_HOLDER
                    || relationship == CustomerAccountRelationshipType.AUTHORIZED_USER
                : relationship == CustomerAccountRelationshipType.PREPAID_OWNER
                    || relationship == CustomerAccountRelationshipType.AUTHORIZED_USER;
        if (!valid) throw new V2BusinessRuleException("relationshipType is not valid for accountType " + accountType);
    }
    private static void validateReferenceTransition(ReferenceStatus current, ReferenceStatus requested) {
        if (current == requested) return;
        boolean allowed = switch (current) {
            case ACTIVE -> requested == ReferenceStatus.INACTIVE || requested == ReferenceStatus.BLOCKED
                    || requested == ReferenceStatus.CLOSED;
            case INACTIVE, BLOCKED -> requested == ReferenceStatus.ACTIVE || requested == ReferenceStatus.CLOSED;
            case CLOSED -> false;
        };
        if (!allowed) throw new V2BusinessRuleException("Status transition " + current + " -> " + requested + " is not allowed");
    }
    private CustomerReferenceV2Response customerResponse(CustomerReferenceV2 e) {
        return new CustomerReferenceV2Response(e.getId(), e.getInstitutionId(), e.getSourceSystem(),
                e.getExternalCustomerId(), e.getCustomerType(), e.getTitle(), e.getFirstName(),
                e.getMiddleName(), e.getLastName(), e.getDateOfBirth(), e.getMobileCountryCode(),
                e.getMobileNumber(), e.getEmail(), e.getStatus(), e.getCreatedAt(), e.getUpdatedAt(),
                e.getCreatedBy(), e.getUpdatedBy());
    }
    private ExternalAccountReferenceV2Response accountResponse(ExternalAccountReferenceV2 e) {
        return new ExternalAccountReferenceV2Response(e.getId(), e.getInstitutionId(), e.getSourceSystem(),
                e.getExternalAccountId(), e.getAccountType(), e.getStatus(), e.getCreatedAt(), e.getUpdatedAt(), e.getCreatedBy(), e.getUpdatedBy());
    }
    private CustomerAddressReferenceV2Response addressResponse(CustomerAddressReferenceV2 e) {
        return new CustomerAddressReferenceV2Response(e.getId(), e.getInstitutionId(),
                e.getCustomerReferenceId(), e.getSourceSystem(), e.getExternalAddressId(),
                e.getAddressType(), e.getAddressLine1(), e.getAddressLine2(), e.getLandmark(),
                e.getCity(), e.getDistrict(), e.getStateCode(), e.getPostalCode(), e.getCountryCode(),
                e.isPrimary(), e.getStatus(), e.getCreatedAt(), e.getUpdatedAt(), e.getCreatedBy(),
                e.getUpdatedBy());
    }
    private CustomerAccountLinkV2Response linkResponse(CustomerAccountLinkV2 link) {
        AccountType type = accounts.findByIdAndInstitutionId(
                        link.getExternalAccountReferenceId(), link.getInstitutionId())
                .orElseThrow(() -> new ResourceNotFoundException("ExternalAccountReference",
                        "externalAccountReferenceId", link.getExternalAccountReferenceId())).getAccountType();
        return linkResponse(link, type);
    }
    private CustomerAccountLinkV2Response linkResponse(CustomerAccountLinkV2 link, AccountType type) {
        return new CustomerAccountLinkV2Response(link.getId(), link.getInstitutionId(), link.getCustomerReferenceId(),
                link.getExternalAccountReferenceId(), type, link.getRelationshipType(), link.isPrimary(),
                link.getStatus(), link.getEffectiveFrom(), link.getEffectiveTo(), link.getCreatedAt(),
                link.getUpdatedAt(), link.getCreatedBy(), link.getUpdatedBy());
    }
