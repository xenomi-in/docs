package com.lps.cms.customer.v2.boundary.controller;

import com.lps.cms.customer.v2.common.ApiSuccessEnvelopeV2;
import com.lps.cms.customer.v2.common.ApiResponseV2;
import com.lps.cms.customer.v2.boundary.dto.*;
import com.lps.cms.customer.v2.boundary.model.ReferenceStatus;
import com.lps.cms.customer.v2.boundary.service.BoundaryReferenceV2Service;
import com.lps.cms.customer.v2.common.PagedResponseV2;
import io.swagger.v3.oas.annotations.Operation;
import io.swagger.v3.oas.annotations.tags.Tag;
import jakarta.validation.Valid;
import lombok.RequiredArgsConstructor;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.*;
import java.util.UUID;

@RestController
@RequiredArgsConstructor
@RequestMapping("/api/v2/institutions/{institutionId}/customers")
@Tag(name = "V2 - Customers", description = "Institution-scoped customer profiles, addresses and account relationships")
public class CustomerBoundaryV2Controller {
    private final BoundaryReferenceV2Service service;

    @PostMapping
    @Operation(operationId = "createCustomerV2", summary = "Onboard a customer",
            description = "Creates an institution-scoped customer profile and returns the UUID used by CMS Core as customerReferenceId.")
    public ResponseEntity<ApiSuccessEnvelopeV2<CustomerReferenceV2Response>> create(
            @PathVariable UUID institutionId, @Valid @RequestBody CreateCustomerReferenceV2Request request) {
        return ResponseEntity.status(HttpStatus.CREATED).body(ApiResponseV2.success(service.createCustomer(institutionId, request)));
    }

    @GetMapping
    @Operation(operationId = "listCustomersV2", summary = "List customers")
    public ApiSuccessEnvelopeV2<PagedResponseV2<CustomerReferenceV2Response>> list(
            @PathVariable UUID institutionId, @RequestParam(required = false) ReferenceStatus status,
            @RequestParam(required = false) Integer page, @RequestParam(required = false) Integer limit) {
        return ApiResponseV2.success(service.listCustomers(institutionId, status, page, limit));
    }

    @GetMapping("/resolve")
    @Operation(operationId = "resolveCustomerReferenceV2", summary = "Resolve a customer reference by external identity")
    public ApiSuccessEnvelopeV2<CustomerReferenceV2Response> resolve(
            @PathVariable UUID institutionId, @RequestParam String sourceSystem,
            @RequestParam String externalCustomerId) {
        return ApiResponseV2.success(service.resolveCustomer(institutionId, sourceSystem, externalCustomerId));
    }

    @GetMapping("/{customerReferenceId}")
    @Operation(operationId = "getCustomerV2", summary = "Get a customer profile by UUID")
    public ApiSuccessEnvelopeV2<CustomerReferenceV2Response> get(
            @PathVariable UUID institutionId, @PathVariable UUID customerReferenceId) {
        return ApiResponseV2.success(service.getCustomer(institutionId, customerReferenceId));
    }

    @PostMapping("/{customerReferenceId}/actions/update-status")
    @Operation(operationId = "updateCustomerReferenceStatusV2", summary = "Change a customer-reference lifecycle status")
    public ApiSuccessEnvelopeV2<CustomerReferenceV2Response> updateStatus(
            @PathVariable UUID institutionId, @PathVariable UUID customerReferenceId,
            @Valid @RequestBody UpdateReferenceStatusV2Request request) {
        return ApiResponseV2.success(service.updateCustomer(institutionId, customerReferenceId, request));
    }

    @PostMapping("/{customerReferenceId}/actions/update-profile")
    @Operation(operationId = "updateCustomerProfileV2", summary = "Update customer profile details",
            description = "Updates mutable customer PII while preserving source identity and audit history.")
    public ApiSuccessEnvelopeV2<CustomerReferenceV2Response> updateProfile(
            @PathVariable UUID institutionId, @PathVariable UUID customerReferenceId,
            @Valid @RequestBody UpdateCustomerProfileV2Request request) {
        return ApiResponseV2.success(service.updateCustomerProfile(
                institutionId, customerReferenceId, request));
    }

    @PostMapping("/{customerReferenceId}/accounts")
    @Operation(operationId = "linkCustomerAccountV2", summary = "Link a credit account or prepaid wallet to a customer",
            description = "Creates an active, institution-scoped relationship. CREDIT_CARD_ACCOUNT supports primary/supplementary holders; PREPAID_WALLET supports prepaid owners.")
    public ResponseEntity<ApiSuccessEnvelopeV2<CustomerAccountLinkV2Response>> linkAccount(
            @PathVariable UUID institutionId, @PathVariable UUID customerReferenceId,
            @Valid @RequestBody CreateCustomerAccountLinkV2Request request) {
        return ResponseEntity.status(HttpStatus.CREATED)
                .body(ApiResponseV2.success(service.linkAccount(institutionId, customerReferenceId, request)));
    }

    @PostMapping("/{customerReferenceId}/addresses")
    @Operation(operationId = "createCustomerAddressV2", summary = "Add an address to a customer",
            description = "Stores the postal address in customer-service and returns the addressReferenceId used by CMS Core and card fulfilment.")
    public ResponseEntity<ApiSuccessEnvelopeV2<CustomerAddressReferenceV2Response>> addAddress(
            @PathVariable UUID institutionId, @PathVariable UUID customerReferenceId,
            @Valid @RequestBody CreateCustomerAddressReferenceV2Request request) {
        return ResponseEntity.status(HttpStatus.CREATED)
                .body(ApiResponseV2.success(service.createAddress(
                        institutionId, customerReferenceId, request)));
    }

    @GetMapping("/{customerReferenceId}/addresses")
    @Operation(operationId = "listCustomerAddressesV2", summary = "List customer addresses")
    public ApiSuccessEnvelopeV2<PagedResponseV2<CustomerAddressReferenceV2Response>> listAddresses(
            @PathVariable UUID institutionId, @PathVariable UUID customerReferenceId,
            @RequestParam(required = false) ReferenceStatus status,
            @RequestParam(required = false) Integer page,
            @RequestParam(required = false) Integer limit) {
        return ApiResponseV2.success(service.listAddresses(
                institutionId, customerReferenceId, status, page, limit));
    }

    @GetMapping("/{customerReferenceId}/addresses/{addressReferenceId}")
    @Operation(operationId = "getCustomerAddressV2", summary = "Get a customer address")
    public ApiSuccessEnvelopeV2<CustomerAddressReferenceV2Response> getAddress(
            @PathVariable UUID institutionId, @PathVariable UUID customerReferenceId,
            @PathVariable UUID addressReferenceId) {
        return ApiResponseV2.success(service.getAddress(
                institutionId, customerReferenceId, addressReferenceId));
    }

    @PostMapping("/{customerReferenceId}/addresses/{addressReferenceId}/actions/update-details")
    @Operation(operationId = "updateCustomerAddressV2", summary = "Update customer address details")
    public ApiSuccessEnvelopeV2<CustomerAddressReferenceV2Response> updateAddress(
            @PathVariable UUID institutionId, @PathVariable UUID customerReferenceId,
            @PathVariable UUID addressReferenceId,
            @Valid @RequestBody UpdateCustomerAddressV2Request request) {
        return ApiResponseV2.success(service.updateAddress(
                institutionId, customerReferenceId, addressReferenceId, request));
    }

    @PostMapping("/{customerReferenceId}/addresses/{addressReferenceId}/actions/update-status")
    @Operation(operationId = "updateCustomerAddressStatusV2", summary = "Change customer address status",
            description = "Preserves address history; no DELETE operation is provided.")
    public ApiSuccessEnvelopeV2<CustomerAddressReferenceV2Response> updateAddressStatus(
            @PathVariable UUID institutionId, @PathVariable UUID customerReferenceId,
            @PathVariable UUID addressReferenceId,
            @Valid @RequestBody UpdateReferenceStatusV2Request request) {
        return ApiResponseV2.success(service.updateAddressStatus(
                institutionId, customerReferenceId, addressReferenceId, request));
    }

    @GetMapping("/{customerReferenceId}/accounts")
    @Operation(operationId = "listCustomerAccountsV2", summary = "List accounts linked to a customer")
    public ApiSuccessEnvelopeV2<PagedResponseV2<CustomerAccountLinkV2Response>> listAccounts(
            @PathVariable UUID institutionId, @PathVariable UUID customerReferenceId,
            @RequestParam(required = false) Integer page, @RequestParam(required = false) Integer limit) {
        return ApiResponseV2.success(service.listCustomerAccounts(institutionId, customerReferenceId, page, limit));
    }

    @PostMapping("/{customerReferenceId}/accounts/{externalAccountReferenceId}/actions/update-status")
    @Operation(operationId = "updateCustomerAccountLinkStatusV2", summary = "Close or deactivate a customer-account relationship",
            description = "Preserves relationship history; no DELETE operation is provided.")
    public ApiSuccessEnvelopeV2<CustomerAccountLinkV2Response> updateAccountLinkStatus(
            @PathVariable UUID institutionId, @PathVariable UUID customerReferenceId,
            @PathVariable UUID externalAccountReferenceId,
            @Valid @RequestBody UpdateCustomerAccountLinkStatusV2Request request) {
        return ApiResponseV2.success(service.updateLinkStatus(
                institutionId, customerReferenceId, externalAccountReferenceId, request));
    }
}
