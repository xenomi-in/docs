package com.lps.cms.customer.v2.boundary.controller;

import com.lps.cms.customer.v2.common.ApiSuccessEnvelopeV2;
import com.lps.cms.customer.v2.common.ApiResponseV2;
import com.lps.cms.customer.v2.boundary.dto.*;
import com.lps.cms.customer.v2.boundary.model.ReferenceStatus;
import com.lps.cms.customer.v2.boundary.service.BoundaryReferenceV2Service;
import com.lps.cms.customer.v2.common.PagedResponseV2;
import io.swagger.v3.oas.annotations.Operation;
import io.swagger.v3.oas.annotations.tags.Tag;
import jakarta.validation.Valid;
import lombok.RequiredArgsConstructor;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.*;
import java.util.UUID;

@RestController
@RequiredArgsConstructor
@RequestMapping("/api/v2/institutions/{institutionId}/accounts")
@Tag(name = "V2 - Account Boundary", description = "Institution-scoped references to accounts owned by an accounting or core-banking system")
public class AccountBoundaryV2Controller {
    private final BoundaryReferenceV2Service service;

    @PostMapping
    @Operation(operationId = "createExternalAccountReferenceV2", summary = "Create an external account reference",
            description = "Registers an opaque account identity. Balances, transaction history and PAN data are not stored.")
    public ResponseEntity<ApiSuccessEnvelopeV2<ExternalAccountReferenceV2Response>> create(
            @PathVariable UUID institutionId, @Valid @RequestBody CreateExternalAccountReferenceV2Request request) {
        return ResponseEntity.status(HttpStatus.CREATED).body(ApiResponseV2.success(service.createAccount(institutionId, request)));
    }

    @GetMapping
    @Operation(operationId = "listExternalAccountReferencesV2", summary = "List external account references")
    public ApiSuccessEnvelopeV2<PagedResponseV2<ExternalAccountReferenceV2Response>> list(
            @PathVariable UUID institutionId, @RequestParam(required = false) ReferenceStatus status,
            @RequestParam(required = false) Integer page, @RequestParam(required = false) Integer limit) {
        return ApiResponseV2.success(service.listAccounts(institutionId, status, page, limit));
    }

    @GetMapping("/resolve")
    @Operation(operationId = "resolveExternalAccountReferenceV2", summary = "Resolve an account reference by external identity")
    public ApiSuccessEnvelopeV2<ExternalAccountReferenceV2Response> resolve(
            @PathVariable UUID institutionId, @RequestParam String sourceSystem,
            @RequestParam String externalAccountId) {
        return ApiResponseV2.success(service.resolveAccount(institutionId, sourceSystem, externalAccountId));
    }

    @GetMapping("/{externalAccountReferenceId}")
    @Operation(operationId = "getExternalAccountReferenceV2", summary = "Get an external account reference by UUID")
    public ApiSuccessEnvelopeV2<ExternalAccountReferenceV2Response> get(
            @PathVariable UUID institutionId, @PathVariable UUID externalAccountReferenceId) {
        return ApiResponseV2.success(service.getAccount(institutionId, externalAccountReferenceId));
    }

    @PostMapping("/{externalAccountReferenceId}/actions/update-status")
    @Operation(operationId = "updateExternalAccountReferenceStatusV2", summary = "Change an account-reference lifecycle status")
    public ApiSuccessEnvelopeV2<ExternalAccountReferenceV2Response> updateStatus(
            @PathVariable UUID institutionId, @PathVariable UUID externalAccountReferenceId,
            @Valid @RequestBody UpdateReferenceStatusV2Request request) {
        return ApiResponseV2.success(service.updateAccount(institutionId, externalAccountReferenceId, request));
    }

    @GetMapping("/{externalAccountReferenceId}/customers")
    @Operation(operationId = "listAccountCustomersV2", summary = "List customers linked to a credit account or prepaid wallet")
    public ApiSuccessEnvelopeV2<PagedResponseV2<CustomerAccountLinkV2Response>> listCustomers(
            @PathVariable UUID institutionId, @PathVariable UUID externalAccountReferenceId,
            @RequestParam(required = false) Integer page, @RequestParam(required = false) Integer limit) {
        return ApiResponseV2.success(service.listAccountCustomers(institutionId, externalAccountReferenceId, page, limit));
    }
}
