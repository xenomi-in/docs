package com.lps.cms.customer.v2.boundary.dto;

import io.swagger.v3.oas.annotations.media.Schema;
import jakarta.validation.constraints.Email;
import jakarta.validation.constraints.Past;
import jakarta.validation.constraints.Pattern;
import jakarta.validation.constraints.Size;
import java.time.LocalDate;

public record UpdateCustomerProfileV2Request(
        @Size(max = 20) @Schema(example = "MR") String title,
        @Size(min = 1, max = 100) @Schema(example = "Pavan") String firstName,
        @Size(max = 100) String middleName,
        @Size(min = 1, max = 100) @Schema(example = "Kumar") String lastName,
        @Past @Schema(example = "1990-05-15") LocalDate dateOfBirth,
        @Pattern(regexp = "^\\+[1-9][0-9]{0,3}$") @Schema(example = "+91") String mobileCountryCode,
        @Pattern(regexp = "^[0-9]{6,15}$") @Schema(example = "9876543210") String mobileNumber,
        @Email @Size(max = 254) @Schema(example = "pavan@example.com") String email) {}
