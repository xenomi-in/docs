package com.lps.cms.customer.v2.boundary.dto;

import com.lps.cms.customer.v2.boundary.model.ReferenceStatus;
import com.lps.cms.customer.v2.boundary.model.CustomerTypeV2;
import io.swagger.v3.oas.annotations.media.Schema;
import jakarta.validation.constraints.Email;
import jakarta.validation.constraints.NotBlank;
import jakarta.validation.constraints.NotNull;
import jakarta.validation.constraints.Past;
import jakarta.validation.constraints.Pattern;
import jakarta.validation.constraints.Size;
import java.time.LocalDate;

public record CreateCustomerReferenceV2Request(
        @NotBlank @Size(max = 50) @Pattern(regexp = "^[A-Za-z0-9._-]+$")
        @Schema(description = "System that owns the customer record", example = "CORE_BANKING")
        String sourceSystem,
        @NotBlank @Size(max = 120)
        @Schema(description = "Customer identifier in the originating system", example = "CUST-100045")
        String externalCustomerId,
        @NotNull
        @Schema(description = "Customer classification", example = "INDIVIDUAL")
        CustomerTypeV2 customerType,
        @Size(max = 20)
        @Schema(description = "Customer title", example = "MR")
        String title,
        @NotBlank @Size(max = 100)
        @Schema(description = "Customer first or given name", example = "Pavan")
        String firstName,
        @Size(max = 100)
        @Schema(description = "Customer middle name")
        String middleName,
        @NotBlank @Size(max = 100)
        @Schema(description = "Customer last or family name", example = "Kumar")
        String lastName,
        @Past
        @Schema(description = "Date of birth for an individual customer", example = "1990-05-15")
        LocalDate dateOfBirth,
        @Pattern(regexp = "^\\+[1-9][0-9]{0,3}$")
        @Schema(description = "E.164 country calling code", example = "+91")
        String mobileCountryCode,
        @Pattern(regexp = "^[0-9]{6,15}$")
        @Schema(description = "Mobile subscriber number without the country calling code", example = "9876543210")
        String mobileNumber,
        @Email @Size(max = 254)
        @Schema(description = "Customer email address", example = "pavan@example.com")
        String email,
        @Schema(description = "Initial reference lifecycle status", example = "ACTIVE")
        ReferenceStatus status) {}
