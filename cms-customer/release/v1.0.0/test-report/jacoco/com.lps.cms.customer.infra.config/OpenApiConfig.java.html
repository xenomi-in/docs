package com.lps.cms.customer.infra.config;

import io.swagger.v3.oas.models.OpenAPI;
import io.swagger.v3.oas.models.Components;
import io.swagger.v3.oas.models.info.Info;
import io.swagger.v3.oas.models.info.Contact;
import io.swagger.v3.oas.models.security.SecurityRequirement;
import io.swagger.v3.oas.models.security.SecurityScheme;
import io.swagger.v3.oas.models.tags.Tag;
import io.swagger.v3.oas.models.parameters.Parameter;
import io.swagger.v3.oas.models.media.StringSchema;
import io.swagger.v3.oas.models.media.Content;
import io.swagger.v3.oas.models.media.MediaType;
import io.swagger.v3.oas.models.media.Schema;
import io.swagger.v3.oas.models.responses.ApiResponse;
import io.swagger.v3.oas.models.headers.Header;
import io.swagger.v3.core.converter.ModelConverters;
import com.lps.cms.customer.v2.common.ApiErrorEnvelopeV2;
import org.springdoc.core.models.GroupedOpenApi;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import java.util.List;
import java.util.Map;

@Configuration
public class OpenApiConfig {
    @Bean
    OpenAPI customerOpenApi() {
        return new OpenAPI()
                .components(new Components().addSecuritySchemes("bearerAuth", new SecurityScheme()
                        .type(SecurityScheme.Type.HTTP)
                        .scheme("bearer")
                        .bearerFormat("JWT")
                        .description("Keycloak access token for audience cms-api")))
                .addSecurityItem(new SecurityRequirement().addList("bearerAuth"))
                .info(new Info()
                        .title("Customer Service API")
                        .version("2.0.0")
                        .description("Customer profiles, postal addresses and institution-scoped account relationships for the CMS platform.")
                        .contact(new Contact().name("CMS API Team")));
    }

    @Bean
    GroupedOpenApi customerV2OpenApi() {
        return GroupedOpenApi.builder().group("customer-v2").displayName("Customer Service API V2")
                .pathsToMatch("/api/v2/**")
                .addOpenApiCustomizer(api -> {
                    api.setInfo(new Info().title("Customer Service API V2").version("2.0.0")
                            .description("""
                                    Institution-scoped APIs for onboarding customer profiles, maintaining postal addresses,
                                    registering external credit-account references and managing customer-account relationships.

                                    Customer Service owns customer and address PII. CMS Core stores only the returned
                                    `customerReferenceId`, `addressReferenceId` and `externalAccountReferenceId` UUIDs.

                                    Authentication uses a bearer JWT. Read operations require `customer:read` or
                                    `customer:manage`; customer writes require `customer:manage`. Account reads require
                                    `account:read` or `account:manage`; account writes require `account:manage`.
                                    """)
                            .contact(new Contact().name("CMS API Team")));
                    api.setTags(List.of(
                            new Tag().name("V2 - Customers")
                                    .description("Customer onboarding, profile maintenance, postal addresses and account links"),
                            new Tag().name("V2 - Account Boundary")
                                    .description("Opaque credit-account references owned by accounting or core banking")));
                    if (api.getComponents() == null) api.setComponents(new Components());
                    api.getComponents().addSecuritySchemes("bearerAuth", new SecurityScheme()
                            .type(SecurityScheme.Type.HTTP).scheme("bearer").bearerFormat("JWT")
                            .description("Keycloak access token containing institution context and required permissions"));
                    api.addSecurityItem(new SecurityRequirement().addList("bearerAuth"));
                    ModelConverters.getInstance().readAll(ApiErrorEnvelopeV2.class)
                            .forEach(api.getComponents()::addSchemas);
                    if (api.getPaths() != null) api.getPaths().values().forEach(path ->
                            path.readOperationsMap().forEach((method, operation) -> {
                                describePathParameters(operation.getParameters());
                                operation.addParametersItem(new Parameter().in("header").name("X-Request-Id")
                                        .required(false).description("Optional caller-generated correlation ID. The service returns the accepted or generated value in the response header and envelope metadata.")
                                        .schema(new StringSchema().maxLength(128).pattern("^[A-Za-z0-9._:-]+$")));
                                boolean write = method == io.swagger.v3.oas.models.PathItem.HttpMethod.POST
                                        || method == io.swagger.v3.oas.models.PathItem.HttpMethod.PUT;
                                if (write) operation.addParametersItem(new Parameter().in("header").name("Idempotency-Key")
                                        .required(true).description("Unique replay-protection key for this institution, route and request body; retained for 24 hours by default")
                                        .schema(new StringSchema().minLength(1).maxLength(128)
                                                .pattern("^[A-Za-z0-9._:-]+$")));
                                if (operation.getResponses() != null) {
                                    operation.getResponses().putIfAbsent("400", error("Validation or idempotency-key failure"));
                                    operation.getResponses().putIfAbsent("401", error("Bearer token is missing or invalid"));
                                    operation.getResponses().putIfAbsent("403", error("Permission or tenant access denied"));
                                    operation.getResponses().putIfAbsent("404", error("Tenant-scoped reference was not found"));
                                    if (write) operation.getResponses().putIfAbsent("409", error("Conflict or idempotency replay conflict"));
                                    operation.getResponses().putIfAbsent("422", error("Business rule violation"));
                                    operation.getResponses().putIfAbsent("429", error("Request rate limit exceeded"));
                                    operation.getResponses().putIfAbsent("500", error("Unexpected internal error"));
                                    operation.getResponses().values().forEach(response -> {
                                        response.addHeaderObject("X-Request-Id", new Header()
                                                .description("Correlation identifier for this request")
                                                .schema(new StringSchema()));
                                        if (write) response.addHeaderObject("Idempotency-Replayed", new Header()
                                                .description("True when the stored response was replayed")
                                                .schema(new Schema<>().type("boolean")));
                                    });
                                }
                            }));
                }).build();
    }

    private static void describePathParameters(List<Parameter> parameters) {
        if (parameters == null) return;
        Map<String, String> descriptions = Map.of(
                "institutionId", "Institution UUID that defines the tenant boundary",
                "customerReferenceId", "Customer UUID generated by Customer Service",
                "addressReferenceId", "Customer-address UUID generated by Customer Service",
                "externalAccountReferenceId", "Account-reference UUID generated by Customer Service");
        parameters.stream()
                .filter(parameter -> "path".equals(parameter.getIn()))
                .forEach(parameter -> {
                    String description = descriptions.get(parameter.getName());
                    if (description != null) parameter.setDescription(description);
                });
    }

    private static ApiResponse error(String description) {
        return new ApiResponse().description(description).content(new Content().addMediaType("application/json",
                new MediaType().schema(new Schema<>().$ref("#/components/schemas/ApiErrorEnvelopeV2"))));
    }
}
