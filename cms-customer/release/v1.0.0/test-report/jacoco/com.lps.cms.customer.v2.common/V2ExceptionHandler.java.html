package com.lps.cms.customer.v2.common;

import com.lps.cms.customer.api.exception.BadRequestException;
import com.lps.cms.customer.api.exception.ResourceNotFoundException;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.validation.ConstraintViolationException;
import org.springframework.core.Ordered;
import org.springframework.core.annotation.Order;
import org.springframework.http.*;
import org.springframework.security.access.AccessDeniedException;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.*;
import java.util.List;

@Order(Ordered.HIGHEST_PRECEDENCE)
@RestControllerAdvice(basePackages = "com.lps.cms.customer.v2")
public class V2ExceptionHandler {
    @ExceptionHandler(ResourceNotFoundException.class)
    ResponseEntity<ApiErrorEnvelopeV2> notFound(ResourceNotFoundException ex, HttpServletRequest request) {
        return error(request, 404, "NOT_FOUND", ex.getMessage(), null);
    }
    @ExceptionHandler(V2ConflictException.class)
    ResponseEntity<ApiErrorEnvelopeV2> conflict(V2ConflictException ex, HttpServletRequest request) {
        return error(request, 409, "CONFLICT", ex.getMessage(), null);
    }
    @ExceptionHandler(V2BusinessRuleException.class)
    ResponseEntity<ApiErrorEnvelopeV2> business(V2BusinessRuleException ex, HttpServletRequest request) {
        return error(request, 422, "BUSINESS_RULE_VIOLATION", ex.getMessage(), null);
    }
    @ExceptionHandler({BadRequestException.class, ConstraintViolationException.class})
    ResponseEntity<ApiErrorEnvelopeV2> badRequest(Exception ex, HttpServletRequest request) {
        return error(request, 400, "BAD_REQUEST", ex.getMessage(), null);
    }
    @ExceptionHandler(MethodArgumentNotValidException.class)
    ResponseEntity<ApiErrorEnvelopeV2> validation(MethodArgumentNotValidException ex, HttpServletRequest request) {
        List<Object> details = ex.getBindingResult().getFieldErrors().stream()
                .map(error -> (Object) java.util.Map.of("field", error.getField(), "message", error.getDefaultMessage()))
                .toList();
        return error(request, 400, "VALIDATION_ERROR", "One or more fields are invalid", details);
    }
    @ExceptionHandler(AccessDeniedException.class)
    ResponseEntity<ApiErrorEnvelopeV2> denied(AccessDeniedException ex, HttpServletRequest request) {
        return error(request, 403, "ACCESS_DENIED", "The authenticated subject is not allowed to access this resource", null);
    }
    @ExceptionHandler(Exception.class)
    ResponseEntity<ApiErrorEnvelopeV2> internal(Exception ex, HttpServletRequest request) {
        return error(request, 500, "INTERNAL_ERROR", "An unexpected internal error occurred", null);
    }
    private static ResponseEntity<ApiErrorEnvelopeV2> error(HttpServletRequest request, int status,
            String code, String message, List<?> details) {
        return ResponseEntity.status(status).body(ApiResponseV2.error(request, code, message, details));
    }
}
