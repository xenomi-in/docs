package com.lps.cms.customer.v2.common;

import org.springframework.data.domain.Page;
import java.util.List;

public record PagedResponseV2<T>(List<T> content, int page, int size, long totalElements,
                                  int totalPages, boolean first, boolean last) {
    public static <T> PagedResponseV2<T> from(Page<T> page) {
        return new PagedResponseV2<>(page.getContent(), page.getNumber() + 1, page.getSize(),
                page.getTotalElements(), page.getTotalPages(), page.isFirst(), page.isLast());
    }
}
