package com.lps.cms.customer.v2.common;

import jakarta.servlet.http.HttpServletRequest;
import org.springframework.web.context.request.RequestContextHolder;
import org.springframework.web.context.request.ServletRequestAttributes;
import java.time.Instant;
import java.util.List;
import java.util.UUID;

public final class ApiResponseV2 {
    private ApiResponseV2() {}

    public static <T> ApiSuccessEnvelopeV2<T> success(T data) {
        return new ApiSuccessEnvelopeV2<>(true, data, meta(currentRequest()));
    }

    public static ApiErrorEnvelopeV2 error(HttpServletRequest request, String code, String message, List<?> details) {
        return new ApiErrorEnvelopeV2(false, new ApiErrorInfoV2(code, message, details), meta(request));
    }

    public static ApiErrorEnvelopeV2 error(String code, String message, List<?> details) {
        return error(currentRequest(), code, message, details);
    }

    private static ApiMetaV2 meta(HttpServletRequest request) {
        Object value = request == null ? null : request.getAttribute(RequestIdFilter.ATTRIBUTE);
        return new ApiMetaV2(value == null ? UUID.randomUUID().toString() : value.toString(), Instant.now().toString());
    }

    private static HttpServletRequest currentRequest() {
        return RequestContextHolder.getRequestAttributes() instanceof ServletRequestAttributes attributes
                ? attributes.getRequest() : null;
    }
}
