package com.lps.cms.customer.v2.common.idempotency;

import org.springframework.beans.factory.annotation.Value;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.stereotype.Repository;
import java.time.OffsetDateTime;
import java.util.List;

@Repository
public class IdempotencyRepositoryV2 {
    private final JdbcTemplate jdbc;
    private final long retentionHours;
    public IdempotencyRepositoryV2(JdbcTemplate jdbc,
            @Value("${customer.idempotency.retention-hours:24}") long retentionHours) {
        this.jdbc = jdbc; this.retentionHours = retentionHours;
    }
    public Claim claim(Identity id, String requestHash) {
        jdbc.update("DELETE FROM api_idempotency_record WHERE expires_at <= CURRENT_TIMESTAMP");
        int inserted = jdbc.update("""
                INSERT INTO api_idempotency_record
                    (tenant_scope, actor_id, http_method, request_path, idempotency_key_hash, request_hash, expires_at)
                VALUES (?, ?, ?, ?, ?, ?, ?) ON CONFLICT DO NOTHING
                """, id.tenant(), id.actor(), id.method(), id.path(), id.keyHash(), requestHash,
                OffsetDateTime.now().plusHours(retentionHours));
        if (inserted == 1) return new Claim(State.STARTED, null);
        List<Stored> records = jdbc.query("""
                SELECT request_hash, processing_status, response_status, response_content_type,
                       response_location, response_body FROM api_idempotency_record
                 WHERE tenant_scope=? AND actor_id=? AND http_method=? AND request_path=?
                   AND idempotency_key_hash=? AND expires_at>CURRENT_TIMESTAMP
                """, (rs, row) -> new Stored(rs.getString(1), rs.getString(2),
                (Integer) rs.getObject(3), rs.getString(4), rs.getString(5), rs.getBytes(6)),
                id.tenant(), id.actor(), id.method(), id.path(), id.keyHash());
        if (records.isEmpty()) return claim(id, requestHash);
        Stored stored = records.getFirst();
        if (!stored.requestHash().equals(requestHash)) return new Claim(State.CONFLICT, stored);
        return "COMPLETED".equals(stored.processingStatus())
                ? new Claim(State.REPLAY, stored) : new Claim(State.IN_PROGRESS, stored);
    }
    public void complete(Identity id, int status, String contentType, String location, byte[] body) {
        jdbc.update("""
                UPDATE api_idempotency_record SET processing_status='COMPLETED', response_status=?,
                    response_content_type=?, response_location=?, response_body=?, completed_at=CURRENT_TIMESTAMP
                 WHERE tenant_scope=? AND actor_id=? AND http_method=? AND request_path=? AND idempotency_key_hash=?
                """, status, contentType, location, body, id.tenant(), id.actor(), id.method(), id.path(), id.keyHash());
    }
    public void delete(Identity id) {
        jdbc.update("DELETE FROM api_idempotency_record WHERE tenant_scope=? AND actor_id=? AND http_method=? AND request_path=? AND idempotency_key_hash=?",
                id.tenant(), id.actor(), id.method(), id.path(), id.keyHash());
    }
    public record Identity(String tenant, String actor, String method, String path, String keyHash) {}
    public record Stored(String requestHash, String processingStatus, Integer status, String contentType, String location, byte[] body) {}
    public record Claim(State state, Stored stored) {}
    public enum State { STARTED, REPLAY, CONFLICT, IN_PROGRESS }
}
