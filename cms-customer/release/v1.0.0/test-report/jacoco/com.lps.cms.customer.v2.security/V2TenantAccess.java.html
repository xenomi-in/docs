package com.lps.cms.customer.v2.security;

import org.springframework.security.access.AccessDeniedException;
import org.springframework.security.core.context.SecurityContextHolder;
import org.springframework.security.oauth2.jwt.Jwt;
import org.springframework.stereotype.Component;
import java.util.UUID;

@Component
public class V2TenantAccess {
    public String require(UUID institutionId) {
        if (!(SecurityContextHolder.getContext().getAuthentication().getPrincipal() instanceof Jwt jwt)) {
            throw new AccessDeniedException("A signed JWT tenant context is required");
        }
        boolean platformAccess = Boolean.TRUE.equals(jwt.getClaim("platform_access"));
        String claim = jwt.getClaimAsString("institution_id");
        if (claim == null || claim.isBlank()) {
            claim = jwt.getClaimAsString("institution-id");
        }
        if (!platformAccess && (claim == null || !institutionId.toString().equals(claim))) {
            throw new AccessDeniedException("Institution access denied");
        }
        return jwt.getSubject();
    }
}
