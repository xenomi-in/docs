package com.lps.isoswitch.exception;

/**
 * Exception thrown when API-related errors occur.
 * This exception is used for HTTP communication issues,
 * API response errors, and other API-related problems.
 *
 * @author Logicpoint Systems
 * @version 1.0
 * @since 2025-06-15
 */
public class ApiException extends BaseException {
    private final int statusCode;

    /**
     * Creates a new ApiException with the specified error code and message.
     *
     * @param errorCode The error code
     * @param errorMessage The error message
     */
    public ApiException(String errorCode, String errorMessage, int statusCode) {
        super(errorCode, errorMessage);
        this.statusCode = statusCode;
    }

    /**
     * Creates a new ApiException with the specified error code, message, and cause.
     *
     * @param errorCode The error code
     * @param errorMessage The error message
     * @param cause The cause of the exception
     */
    public ApiException(String errorCode, String errorMessage, Throwable cause, int statusCode) {
        super(errorCode, errorMessage, cause);
        this.statusCode = statusCode;
    }

    public int getStatusCode() {
        return statusCode;
    }
}
