package com.lps.isoswitch.exception;

/**
 * Exception thrown when message processing errors occur.
 * This exception is used for ISO8583 message parsing errors,
 * validation failures, and other message processing issues.
 *
 * @author Logicpoint Systems
 * @version 1.0
 * @since 2025-06-15
 */
public class MessageProcessingException extends BaseException {
    /**
     * Creates a new MessageProcessingException with the specified error code and message.
     *
     * @param errorCode The error code
     * @param errorMessage The error message
     */
    public MessageProcessingException(String errorCode, String errorMessage) {
        super(errorCode, errorMessage);
    }

    /**
     * Creates a new MessageProcessingException with the specified error code, message, and cause.
     *
     * @param errorCode The error code
     * @param errorMessage The error message
     * @param cause The cause of the exception
     */
    public MessageProcessingException(String errorCode, String errorMessage, Throwable cause) {
        super(errorCode, errorMessage, cause);
    }
}
