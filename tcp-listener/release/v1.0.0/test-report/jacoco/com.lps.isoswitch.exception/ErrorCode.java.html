package com.lps.isoswitch.exception;

/**
 * Enumeration of error codes used throughout the application.
 */
public enum ErrorCode {
    // Network-related errors (1000-1999)
    NETWORK_CONNECTION_FAILED("1000", "Failed to establish network connection"),
    NETWORK_TIMEOUT("1001", "Network operation timed out"),
    NETWORK_DISCONNECTED("1002", "Network connection was disconnected"),
    NETWORK_INVALID_RESPONSE("1003", "Invalid response received from network"),
    NETWORK_SEND_FAILED("1004", "Failed to send message over network"),
    NETWORK_RECEIVE_FAILED("1005", "Failed to receive message from network"),

    // Message processing errors (2000-2999)
    MESSAGE_PACKING_FAILED("2000", "Failed to pack ISO message"),
    MESSAGE_UNPACKING_FAILED("2001", "Failed to unpack ISO message"),
    MESSAGE_VALIDATION_FAILED("2002", "Message validation failed"),
    MESSAGE_FIELD_MISSING("2003", "Required field is missing in message"),
    MESSAGE_FIELD_INVALID("2004", "Invalid field value in message"),
    MESSAGE_TYPE_UNKNOWN("2005", "Unknown message type"),
    MESSAGE_FORMAT_INVALID("2006", "Invalid message format"),
    MESSAGE_PROCESSING_FAILED("2007", "Failed to process message"),

    // API-related errors (3000-3999)
    API_CALL_FAILED("3000", "API call failed"),
    API_TIMEOUT("3001", "API call timed out"),
    API_INVALID_RESPONSE("3002", "Invalid response from API"),
    API_AUTHENTICATION_FAILED("3003", "API authentication failed"),
    API_RATE_LIMIT_EXCEEDED("3004", "API rate limit exceeded"),
    API_SERVICE_UNAVAILABLE("3005", "API service is unavailable"),

    // Configuration errors (4000-4999)
    CONFIG_INVALID("4000", "Invalid configuration"),
    CONFIG_MISSING("4001", "Required configuration is missing"),
    CONFIG_LOAD_FAILED("4002", "Failed to load configuration"),

    // System errors (5000-5999)
    SYSTEM_ERROR("5000", "Internal system error"),
    SYSTEM_RESOURCE_UNAVAILABLE("5001", "Required system resource is unavailable"),
    SYSTEM_INITIALIZATION_FAILED("5002", "System initialization failed"),
    SYSTEM_SHUTDOWN_FAILED("5003", "System shutdown failed");

    private final String code;
    private final String message;

    ErrorCode(String code, String message) {
        this.code = code;
        this.message = message;
    }

    public String getCode() {
        return code;
    }

    public String getMessage() {
        return message;
    }
}
