package com.lps.isoswitch.tcpconnector.channel;

import io.netty.channel.ChannelHandlerContext;

import org.jpos.iso.ISOException;
import org.jpos.iso.ISOMsg;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import com.lps.isoswitch.config.RawDataLogging;

/**
 * HEX Length Channel implementation for ISO8583 message handling.
 * This channel uses hexadecimal-encoded length fields similar to jPOS HEX channel.
 *
 * Features:
 * - Hexadecimal length field (e.g., "0080" for 128 bytes)
 * - Configurable length field size (typically 4 characters)
 * - Length field is separate from message data
 * - Compatible with jPOS HEX channel format
 *
 * @author Logicpoint Systems
 * @version 1.0
 * @since 2025-06-15
 */
public class HexLengthChannel extends AbstractISOChannel {

    private static final Logger log = LoggerFactory.getLogger(HexLengthChannel.class);

    // Default HEX length configuration
    private static final int DEFAULT_HEX_LENGTH_SIZE = 4;
    private static final boolean DEFAULT_INCLUDE_LENGTH_FIELD = false;
    private static final boolean DEFAULT_LENGTH_IN_ASCII = false;
    private static final boolean DEFAULT_LENGTH_IN_HEX = true;

    /**
     * Constructs a new HexLengthChannel with default configuration.
     */
    public HexLengthChannel() {
        super();
        configureHexLength();
    }

    /**
     * Constructs a new HexLengthChannel with specified maximum message length.
     *
     * @param maxMessageLength The maximum message length in bytes
     */
    public HexLengthChannel(int maxMessageLength) {
        super(maxMessageLength);
        configureHexLength();
    }

    /**
     * Constructs a new HexLengthChannel with custom length field size.
     *
     * @param maxMessageLength The maximum message length in bytes
     * @param lengthFieldSize The size of the HEX length field
     */
    public HexLengthChannel(int maxMessageLength, int lengthFieldSize) {
        super(maxMessageLength);
        setLengthFieldConfig(lengthFieldSize, DEFAULT_INCLUDE_LENGTH_FIELD,
                           DEFAULT_LENGTH_IN_HEX, DEFAULT_LENGTH_IN_ASCII);
    }

    /**
     * Configures the channel for HEX length handling.
     */
    private void configureHexLength() {
        setLengthFieldConfig(DEFAULT_HEX_LENGTH_SIZE, DEFAULT_INCLUDE_LENGTH_FIELD,
                           DEFAULT_LENGTH_IN_HEX, DEFAULT_LENGTH_IN_ASCII);
    }

    /**
     * Called when a connection is established.
     *
     * @param ctx The channel handler context
     */
    @Override
    protected void onConnect(ChannelHandlerContext ctx) {
        log.info("HEX Length Channel connected to: {}", ctx.channel().remoteAddress());
    }

    /**
     * Called when a connection is closed.
     *
     * @param ctx The channel handler context
     */
    @Override
    protected void onDisconnect(ChannelHandlerContext ctx) {
        log.info("HEX Length Channel disconnected from: {}", ctx.channel().remoteAddress());
    }

    /**
     * Called when a complete ISO8583 message is received.
     *
     * @param ctx The channel handler context
     * @param isoMsg The received ISO8583 message
     */
    @Override
    protected void onMessage(ChannelHandlerContext ctx, ISOMsg isoMsg) {
        try {
        log.info("Received ISO8583 message via HEX Length Channel - MTI: {}, STAN: {}",
                isoMsg.getMTI(), isoMsg.getString(11));

        // Log message details for debugging
        RawDataLogging.isoMessage("HEX channel ISO message", isoMsg);

        // Delegate to message handler or process directly
        handleISOMessage(ctx, isoMsg);
        } catch (ISOException e) {
            log.error("Error processing ISO8583 message", e);
        }
    }

    /**
     * Called when an error occurs.
     *
     * @param ctx The channel handler context
     * @param cause The error
     */
    @Override
    protected void onError(ChannelHandlerContext ctx, Throwable cause) {
        log.error("HEX Length Channel error", cause);
        // Close the connection on error
        ctx.close();
    }

    /**
     * Handles the received ISO8583 message.
     * This method should be overridden by subclasses to provide specific message handling.
     *
     * @param ctx The channel handler context
     * @param isoMsg The ISO8583 message to handle
     */
    protected void handleISOMessage(ChannelHandlerContext ctx, ISOMsg isoMsg) {
        try {
            // Default implementation - log the message
            log.info("Handling ISO8583 message - MTI: {}", isoMsg.getMTI());
        } catch (ISOException e) {
            log.error("Error processing ISO8583 message", e);
        }
    }

    /**
     * Gets the channel type description.
     *
     * @return The channel type description
     */
    public String getChannelType() {
        return "HEX Length Channel";
    }

    /**
     * Gets the current length field configuration.
     *
     * @return A string describing the length field configuration
     */
    public String getLengthFieldConfig() {
        return String.format("HEX Length Field: %d bytes, Include: %s",
                           lengthFieldSize, includeLengthField);
    }
}
