package com.lps.isoswitch.tcpconnector.channel;

import io.netty.channel.ChannelHandlerContext;

import org.jpos.iso.ISOException;
import org.jpos.iso.ISOMsg;
import org.jpos.iso.ISOPackager;
import org.jpos.iso.packager.GenericPackager;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import com.lps.isoswitch.config.RawDataLogging;

import java.io.InputStream;

/**
 * RuPay Channel implementation for ISO8583 message handling.
 * This channel is specifically configured for RuPay protocol requirements
 * with 2-byte binary length fields.
 *
 * Features:
 * - 2-byte binary length field (similar to jPOS binary channel)
 * - RuPay-specific ISO8583 packager configuration
 * - Length field is separate from message data
 * - Compatible with RuPay network requirements
 *
 * @author Logicpoint Systems
 * @version 1.0
 * @since 2025-06-15
 */
public class RuPayChannel extends AbstractISOChannel {

    private static final Logger log = LoggerFactory.getLogger(RuPayChannel.class);

    // RuPay-specific configuration
    private static final int RUPAY_LENGTH_FIELD_SIZE = 2;
    private static final int RUPAY_MAX_MESSAGE_LENGTH = 6144;
    private static final boolean RUPAY_INCLUDE_LENGTH_FIELD = false;
    private static final boolean RUPAY_LENGTH_IN_ASCII = false;
    private static final boolean RUPAY_LENGTH_IN_HEX = false;

    /**
     * Constructs a new RuPayChannel with default RuPay configuration.
     */
    public RuPayChannel() {
        super(RUPAY_MAX_MESSAGE_LENGTH);
        configureRuPayChannel();
        initializeRuPayPackager();
    }

    /**
     * Constructs a new RuPayChannel with custom maximum message length.
     *
     * @param maxMessageLength The maximum message length in bytes
     */
    public RuPayChannel(int maxMessageLength) {
        super(maxMessageLength);
        configureRuPayChannel();
        initializeRuPayPackager();
    }

    /**
     * Configures the channel for RuPay protocol requirements.
     */
    private void configureRuPayChannel() {
        setLengthFieldConfig(RUPAY_LENGTH_FIELD_SIZE, RUPAY_INCLUDE_LENGTH_FIELD,
                           RUPAY_LENGTH_IN_HEX, RUPAY_LENGTH_IN_ASCII);
        setTimeout(30000, 30000); // 30 seconds timeout
    }

    /**
     * Initializes the RuPay ISO8583 packager.
     */
    private void initializeRuPayPackager() {
        try {
            InputStream is = getClass().getResourceAsStream("/RuPayISO8583.xml");
            if (is == null) {
                log.error("Could not find RuPayISO8583.xml in classpath");
                throw new RuntimeException("RuPay packager configuration file not found");
            }
            ISOPackager ruPayPackager = new GenericPackager(is);
            setPackager(ruPayPackager);
            is.close();
            log.info("RuPay ISO8583 packager initialized successfully");
        } catch (Exception e) {
            log.error("Error initializing RuPay ISO8583 packager", e);
            throw new RuntimeException("Failed to initialize RuPay packager", e);
        }
    }

    /**
     * Called when a connection is established.
     *
     * @param ctx The channel handler context
     */
    @Override
    protected void onConnect(ChannelHandlerContext ctx) {
        log.info("RuPay Channel connected to: {}", ctx.channel().remoteAddress());
    }

    /**
     * Called when a connection is closed.
     *
     * @param ctx The channel handler context
     */
    @Override
    protected void onDisconnect(ChannelHandlerContext ctx) {
        log.info("RuPay Channel disconnected from: {}", ctx.channel().remoteAddress());
    }

    /**
     * Called when a complete ISO8583 message is received.
     *
     * @param ctx The channel handler context
     * @param isoMsg The received ISO8583 message
     */
    @Override
    protected void onMessage(ChannelHandlerContext ctx, ISOMsg isoMsg) {
        try {
            log.info("Received RuPay ISO8583 message - MTI: {}, STAN: {}",
                    isoMsg.getMTI(), isoMsg.getString(11));

        // Log message details for debugging
        RawDataLogging.isoMessage("RuPay channel ISO message", isoMsg);

        // Delegate to message handler or process directly
        handleRuPayMessage(ctx, isoMsg);
        } catch (ISOException e) {
            log.error("Error processing RuPay ISO8583 message", e);
        }
    }

    /**
     * Called when an error occurs.
     *
     * @param ctx The channel handler context
     * @param cause The error
     */
    @Override
    protected void onError(ChannelHandlerContext ctx, Throwable cause) {
        log.error("RuPay Channel error", cause);
        // Close the connection on error
        ctx.close();
    }

    /**
     * Handles the received RuPay ISO8583 message.
     * This method should be overridden by subclasses to provide specific RuPay message handling.
     *
     * @param ctx The channel handler context
     * @param isoMsg The RuPay ISO8583 message to handle
     */
    protected void handleRuPayMessage(ChannelHandlerContext ctx, ISOMsg isoMsg) {
        try {



        // Default implementation - log the message
        log.info("Handling RuPay ISO8583 message - MTI: {}", isoMsg.getMTI());

        // Check if it's a network message (0800) or financial message
        String mti = isoMsg.getMTI();
        if (mti != null && mti.endsWith("800")) {
            log.info("Processing RuPay network message");
            handleNetworkMessage(ctx, isoMsg);
        } else {
            log.info("Processing RuPay financial message");
            handleFinancialMessage(ctx, isoMsg);
        }
    } catch (ISOException e) {
        log.error("Error processing RuPay ISO8583 message", e);
    }
    }

    /**
     * Handles RuPay network messages (MTI 0800).
     *
     * @param ctx The channel handler context
     * @param isoMsg The network message
     */
    protected void handleNetworkMessage(ChannelHandlerContext ctx, ISOMsg isoMsg) {
        try {
            log.info("Handling RuPay network message - MTI: {}", isoMsg.getMTI());
            // Default implementation - can be overridden by subclasses
        } catch (ISOException e) {
            log.error("Error processing RuPay ISO8583 message", e);
        }
    }

    /**
     * Handles RuPay financial messages.
     *
     * @param ctx The channel handler context
     * @param isoMsg The financial message
     */
    protected void handleFinancialMessage(ChannelHandlerContext ctx, ISOMsg isoMsg) {
        try {
            log.info("Handling RuPay financial message - MTI: {}", isoMsg.getMTI());
            // Default implementation - can be overridden by subclasses
        } catch (ISOException e) {
            log.error("Error processing RuPay ISO8583 message", e);
        }
    }

    /**
     * Gets the channel type description.
     *
     * @return The channel type description
     */
    public String getChannelType() {
        return "RuPay Channel";
    }

    /**
     * Gets the current length field configuration.
     *
     * @return A string describing the length field configuration
     */
    public String getLengthFieldConfig() {
        return String.format("RuPay Length Field: %d bytes (binary), Include: %s",
                           lengthFieldSize, includeLengthField);
    }

    /**
     * Gets the RuPay-specific configuration information.
     *
     * @return A string describing the RuPay configuration
     */
    public String getRuPayConfig() {
        return String.format("RuPay Channel - Max Length: %d, Timeout: %dms",
                           maxMessageLength, readTimeout);
    }
}
