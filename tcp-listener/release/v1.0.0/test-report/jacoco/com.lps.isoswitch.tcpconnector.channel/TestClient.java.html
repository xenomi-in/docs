package com.lps.isoswitch.tcpconnector.channel;

import java.util.concurrent.Executors;
import java.util.concurrent.ScheduledExecutorService;
import java.util.concurrent.TimeUnit;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

public class TestClient {

    private static final Logger log = LoggerFactory.getLogger(TestClient.class);

    private RuPayClientChannel client;
    private ScheduledExecutorService healthCheckExecutor;
    
    public static void main(String[] args) {
        TestClient testClient = new TestClient();
        testClient.startConnection();
    }

    public void startConnection() {
        // Create client
        client = new RuPayClientChannel();
        
        // Set up connection monitoring
        setupConnectionMonitoring();
        
        // Connect to server
        client.connectToRuPayServer("192.168.0.105", 9052)
            .thenAccept(success -> {
                if (success) {
                    log.info("Connected to RuPay server");
                    performSignOn();
                } else {
                    log.error("Failed to connect to RuPay server");
                }
            });
    }

    private void performSignOn() {
        client.signOn().thenAccept(success -> {
            if (success) {
                log.info("Sign-on successful");
                startHealthChecks();
            } else {
                log.error("Sign-on failed");
            }
        });
    }
    
    private void setupConnectionMonitoring() {
        // Monitor connection status
        ScheduledExecutorService monitorExecutor = Executors.newSingleThreadScheduledExecutor();
        monitorExecutor.scheduleAtFixedRate(() -> {
            if (!client.isConnected()) {
                log.warn("Connection lost - attempting reconnection");
            }
            
            if (client.isConnected() && !client.isSignedOn()) {
                log.warn("Connected but not signed on - attempting sign-on");
                performSignOn();
            }
        }, 30, 30, TimeUnit.SECONDS);
    }
    
    private void startHealthChecks() {
        // Send periodic echo messages
        healthCheckExecutor = Executors.newSingleThreadScheduledExecutor();
        healthCheckExecutor.scheduleAtFixedRate(() -> {
            if (client.isConnected() && client.isSignedOn()) {
                client.echo().thenAccept(success -> {
                    if (!success) {
                        log.warn("Echo failed - connection may be unhealthy");
                    }
                });
            }
        }, 60, 60, TimeUnit.SECONDS);
    }
    
    public void stopConnection() {
        if (healthCheckExecutor != null) {
            healthCheckExecutor.shutdown();
        }
        
        if (client != null) {
            client.signOff().thenRun(() -> {
                client.disconnect().thenRun(() -> {
                    client.shutdown();
                });
            });
        }
    }
}
