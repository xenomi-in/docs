package com.lps.isoswitch.tcpconnector.channel;

/**
 * Configuration class for different channel types and their length-based message reading settings.
 * This class provides predefined configurations for various ISO8583 channel types similar to jPOS.
 * 
 * @author Logicpoint Systems
 * @version 1.0
 * @since 2025-06-15
 */
public class ChannelConfig {

    /**
     * Channel type enumeration for different message framing protocols.
     */
    public enum ChannelType {
        ASCII_LENGTH("ASCII Length Channel", 4, false, false, true),
        HEX_LENGTH("HEX Length Channel", 4, false, true, false),
        BINARY_2BYTE("Binary 2-byte Length Channel", 2, false, false, false),
        BINARY_4BYTE("Binary 4-byte Length Channel", 4, false, false, false),
        RUPAY("RuPay Channel", 2, false, false, false),
        VISA("Visa Channel", 2, false, false, false),
        MASTERCARD("MasterCard Channel", 2, false, false, false);

        private final String description;
        private final int lengthFieldSize;
        private final boolean includeLengthField;
        private final boolean lengthInHex;
        private final boolean lengthInAscii;

        ChannelType(String description, int lengthFieldSize, boolean includeLengthField, 
                   boolean lengthInHex, boolean lengthInAscii) {
            this.description = description;
            this.lengthFieldSize = lengthFieldSize;
            this.includeLengthField = includeLengthField;
            this.lengthInHex = lengthInHex;
            this.lengthInAscii = lengthInAscii;
        }

        public String getDescription() {
            return description;
        }

        public int getLengthFieldSize() {
            return lengthFieldSize;
        }

        public boolean isIncludeLengthField() {
            return includeLengthField;
        }

        public boolean isLengthInHex() {
            return lengthInHex;
        }

        public boolean isLengthInAscii() {
            return lengthInAscii;
        }
    }

    /**
     * Default configuration values for different channel types.
     */
    public static class Defaults {
        // Maximum message lengths
        public static final int DEFAULT_MAX_MESSAGE_LENGTH = 8192;
        public static final int RUPAY_MAX_MESSAGE_LENGTH = 6144;
        public static final int VISA_MAX_MESSAGE_LENGTH = 8192;
        public static final int MASTERCARD_MAX_MESSAGE_LENGTH = 8192;
        
        // Timeouts
        public static final int DEFAULT_READ_TIMEOUT = 30000;
        public static final int DEFAULT_WRITE_TIMEOUT = 30000;
        public static final int DEFAULT_CONNECT_TIMEOUT = 10000;
        
        // Thread pool settings
        public static final int DEFAULT_CORE_POOL_SIZE = 10;
        public static final int DEFAULT_MAX_POOL_SIZE = 20;
        public static final int DEFAULT_QUEUE_CAPACITY = 100;
        public static final long DEFAULT_KEEP_ALIVE_TIME = 60L;
    }

    /**
     * Creates a channel configuration for the specified channel type.
     *
     * @param channelType The type of channel
     * @return The channel configuration
     */
    public static ChannelConfiguration createConfiguration(ChannelType channelType) {
        return new ChannelConfiguration(channelType);
    }

    /**
     * Creates a RuPay-specific channel configuration.
     *
     * @return The RuPay channel configuration
     */
    public static ChannelConfiguration createRuPayConfiguration() {
        return new ChannelConfiguration(ChannelType.RUPAY)
                .withMaxMessageLength(Defaults.RUPAY_MAX_MESSAGE_LENGTH)
                .withReadTimeout(Defaults.DEFAULT_READ_TIMEOUT)
                .withWriteTimeout(Defaults.DEFAULT_WRITE_TIMEOUT);
    }

    /**
     * Creates a Visa-specific channel configuration.
     *
     * @return The Visa channel configuration
     */
    public static ChannelConfiguration createVisaConfiguration() {
        return new ChannelConfiguration(ChannelType.VISA)
                .withMaxMessageLength(Defaults.VISA_MAX_MESSAGE_LENGTH)
                .withReadTimeout(Defaults.DEFAULT_READ_TIMEOUT)
                .withWriteTimeout(Defaults.DEFAULT_WRITE_TIMEOUT);
    }

    /**
     * Creates a MasterCard-specific channel configuration.
     *
     * @return The MasterCard channel configuration
     */
    public static ChannelConfiguration createMasterCardConfiguration() {
        return new ChannelConfiguration(ChannelType.MASTERCARD)
                .withMaxMessageLength(Defaults.MASTERCARD_MAX_MESSAGE_LENGTH)
                .withReadTimeout(Defaults.DEFAULT_READ_TIMEOUT)
                .withWriteTimeout(Defaults.DEFAULT_WRITE_TIMEOUT);
    }

    /**
     * Creates an ASCII length channel configuration.
     *
     * @return The ASCII length channel configuration
     */
    public static ChannelConfiguration createAsciiLengthConfiguration() {
        return new ChannelConfiguration(ChannelType.ASCII_LENGTH)
                .withMaxMessageLength(Defaults.DEFAULT_MAX_MESSAGE_LENGTH)
                .withReadTimeout(Defaults.DEFAULT_READ_TIMEOUT)
                .withWriteTimeout(Defaults.DEFAULT_WRITE_TIMEOUT);
    }

    /**
     * Creates a HEX length channel configuration.
     *
     * @return The HEX length channel configuration
     */
    public static ChannelConfiguration createHexLengthConfiguration() {
        return new ChannelConfiguration(ChannelType.HEX_LENGTH)
                .withMaxMessageLength(Defaults.DEFAULT_MAX_MESSAGE_LENGTH)
                .withReadTimeout(Defaults.DEFAULT_READ_TIMEOUT)
                .withWriteTimeout(Defaults.DEFAULT_WRITE_TIMEOUT);
    }

    /**
     * Channel configuration builder class.
     */
    public static class ChannelConfiguration {
        private final ChannelType channelType;
        private int maxMessageLength = Defaults.DEFAULT_MAX_MESSAGE_LENGTH;
        private int readTimeout = Defaults.DEFAULT_READ_TIMEOUT;
        private int writeTimeout = Defaults.DEFAULT_WRITE_TIMEOUT;
        private int connectTimeout = Defaults.DEFAULT_CONNECT_TIMEOUT;
        private int corePoolSize = Defaults.DEFAULT_CORE_POOL_SIZE;
        private int maxPoolSize = Defaults.DEFAULT_MAX_POOL_SIZE;
        private int queueCapacity = Defaults.DEFAULT_QUEUE_CAPACITY;
        private long keepAliveTime = Defaults.DEFAULT_KEEP_ALIVE_TIME;

        public ChannelConfiguration(ChannelType channelType) {
            this.channelType = channelType;
        }

        public ChannelConfiguration withMaxMessageLength(int maxMessageLength) {
            this.maxMessageLength = maxMessageLength;
            return this;
        }

        public ChannelConfiguration withReadTimeout(int readTimeout) {
            this.readTimeout = readTimeout;
            return this;
        }

        public ChannelConfiguration withWriteTimeout(int writeTimeout) {
            this.writeTimeout = writeTimeout;
            return this;
        }

        public ChannelConfiguration withConnectTimeout(int connectTimeout) {
            this.connectTimeout = connectTimeout;
            return this;
        }

        public ChannelConfiguration withCorePoolSize(int corePoolSize) {
            this.corePoolSize = corePoolSize;
            return this;
        }

        public ChannelConfiguration withMaxPoolSize(int maxPoolSize) {
            this.maxPoolSize = maxPoolSize;
            return this;
        }

        public ChannelConfiguration withQueueCapacity(int queueCapacity) {
            this.queueCapacity = queueCapacity;
            return this;
        }

        public ChannelConfiguration withKeepAliveTime(long keepAliveTime) {
            this.keepAliveTime = keepAliveTime;
            return this;
        }

        // Getters
        public ChannelType getChannelType() {
            return channelType;
        }

        public int getMaxMessageLength() {
            return maxMessageLength;
        }

        public int getReadTimeout() {
            return readTimeout;
        }

        public int getWriteTimeout() {
            return writeTimeout;
        }

        public int getConnectTimeout() {
            return connectTimeout;
        }

        public int getCorePoolSize() {
            return corePoolSize;
        }

        public int getMaxPoolSize() {
            return maxPoolSize;
        }

        public int getQueueCapacity() {
            return queueCapacity;
        }

        public long getKeepAliveTime() {
            return keepAliveTime;
        }

        public int getLengthFieldSize() {
            return channelType.getLengthFieldSize();
        }

        public boolean isIncludeLengthField() {
            return channelType.isIncludeLengthField();
        }

        public boolean isLengthInHex() {
            return channelType.isLengthInHex();
        }

        public boolean isLengthInAscii() {
            return channelType.isLengthInAscii();
        }

        public String getDescription() {
            return channelType.getDescription();
        }

        @Override
        public String toString() {
            return String.format("ChannelConfiguration{type=%s, maxLength=%d, readTimeout=%d, writeTimeout=%d}",
                    channelType.getDescription(), maxMessageLength, readTimeout, writeTimeout);
        }
    }

    /**
     * Utility method to get channel configuration description.
     *
     * @param channelType The channel type
     * @return A formatted description of the channel configuration
     */
    public static String getChannelDescription(ChannelType channelType) {
        return String.format("%s - Length: %d bytes, Format: %s, Include: %s",
                channelType.getDescription(),
                channelType.getLengthFieldSize(),
                channelType.isLengthInAscii() ? "ASCII" : 
                channelType.isLengthInHex() ? "HEX" : "BINARY",
                channelType.isIncludeLengthField() ? "Yes" : "No");
    }

    /**
     * Utility method to validate channel configuration.
     *
     * @param config The channel configuration to validate
     * @return true if valid, false otherwise
     */
    public static boolean isValidConfiguration(ChannelConfiguration config) {
        if (config == null) return false;
        if (config.getMaxMessageLength() <= 0) return false;
        if (config.getReadTimeout() <= 0) return false;
        if (config.getWriteTimeout() <= 0) return false;
        if (config.getCorePoolSize() <= 0) return false;
        if (config.getMaxPoolSize() < config.getCorePoolSize()) return false;
        if (config.getQueueCapacity() <= 0) return false;
        if (config.getKeepAliveTime() <= 0) return false;
        return true;
    }
}
