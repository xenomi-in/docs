package com.lps.isoswitch.tcpconnector.channel;

import io.netty.channel.ChannelHandlerContext;
import io.netty.channel.ChannelPipeline;
import io.netty.handler.codec.LengthFieldBasedFrameDecoder;
import io.netty.handler.codec.LengthFieldPrepender;
import io.netty.handler.codec.bytes.ByteArrayDecoder;
import io.netty.handler.logging.LogLevel;
import io.netty.handler.logging.LoggingHandler;

import org.jpos.iso.ISOException;
import org.jpos.iso.ISOMsg;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import com.lps.isoswitch.config.RawDataLogging;

import java.util.concurrent.CompletableFuture;
import java.util.concurrent.TimeUnit;

/**
 * ISO8583 Client Channel implementation that provides client connection capabilities.
 * This class extends AbstractISOChannel and adds client-specific functionality
 * for connecting to ISO8583 servers.
 *
 * Features:
 * - Client connection management
 * - Automatic reconnection
 * - Length-based message framing
 * - ISO8583 message handling
 * - Connection status monitoring
 * - Easy-to-use client API
 *
 * @author Logicpoint Systems
 * @version 1.0
 * @since 2025-06-15
 */
public abstract class ISOClientChannel extends AbstractISOChannel {

    private static final Logger log = LoggerFactory.getLogger(ISOClientChannel.class);

    // Client-specific configuration
    private static final int DEFAULT_MAX_FRAME_LENGTH = 8192;
    private static final int DEFAULT_LENGTH_FIELD_OFFSET = 0;
    private static final int DEFAULT_LENGTH_ADJUSTMENT = 0;
    private static final int DEFAULT_INITIAL_BYTES_TO_STRIP = 0;
    private static final boolean DEFAULT_INCLUDE_LENGTH_FIELD_LENGTH = false;

    // Channel context for sending messages
    private ChannelHandlerContext currentContext;

    /**
     * Constructs a new ISOClientChannel with default configuration.
     */
    public ISOClientChannel() {
        super();
    }

    /**
     * Constructs a new ISOClientChannel with specified maximum message length.
     *
     * @param maxMessageLength The maximum message length in bytes
     */
    public ISOClientChannel(int maxMessageLength) {
        super(maxMessageLength);
    }

    /**
     * Connects to the specified host and port as a client.
     * This method provides a simplified connection interface.
     *
     * @param host The host address to connect to
     * @param port The port number to connect to
     * @return CompletableFuture that completes when connection is established
     */
    public CompletableFuture<Boolean> connectToServer(String host, int port) {
        log.info("Connecting to ISO8583 server at {}:{}", host, port);
        return connect(host, port);
    }

    /**
     * Connects to the specified host and port with timeout.
     *
     * @param host The host address to connect to
     * @param port The port number to connect to
     * @param timeoutMs Connection timeout in milliseconds
     * @return CompletableFuture that completes when connection is established
     */
    public CompletableFuture<Boolean> connectToServer(String host, int port, long timeoutMs) {
        log.info("Connecting to ISO8583 server at {}:{} with timeout {}ms", host, port, timeoutMs);

        CompletableFuture<Boolean> connectFuture = connect(host, port);

        // Add timeout
        CompletableFuture<Boolean> timeoutFuture = new CompletableFuture<>();
        connectFuture.orTimeout(timeoutMs, TimeUnit.MILLISECONDS)
                    .whenComplete((result, throwable) -> {
                        if (throwable != null) {
                            log.error("Connection timeout to {}:{}", host, port);
                            timeoutFuture.complete(false);
                        } else {
                            timeoutFuture.complete(result);
                        }
                    });

        return timeoutFuture;
    }

    /**
     * Sends an ISO8583 message to the connected server.
     *
     * @param isoMsg The ISO8583 message to send
     * @return CompletableFuture that completes when message is sent
     */
    public CompletableFuture<Boolean> sendMessageAsync(ISOMsg isoMsg) {
        CompletableFuture<Boolean> sendFuture = new CompletableFuture<>();

        try {
            if (!isConnected()) {
                log.error("Cannot send message - not connected to server");
                sendFuture.complete(false);
                return sendFuture;
            }

            if (currentContext == null) {
                log.error("Cannot send message - no active channel context");
                sendFuture.complete(false);
                return sendFuture;
            }

            // Send the message
            sendMessage(currentContext, isoMsg);
            log.info("Message sent successfully - MTI: {}, STAN: {}",
                    isoMsg.getMTI(), isoMsg.getString(11));
            sendFuture.complete(true);

        } catch (Exception e) {
            log.error("Error sending message", e);
            sendFuture.completeExceptionally(e);
        }

        return sendFuture;
    }

    /**
     * Sends an ISO8583 message and waits for a response.
     *
     * @param isoMsg The ISO8583 message to send
     * @param timeoutMs Response timeout in milliseconds
     * @return CompletableFuture that completes with the response message
     */
    public CompletableFuture<ISOMsg> sendMessageAndWaitForResponse(ISOMsg isoMsg, long timeoutMs) {
        CompletableFuture<ISOMsg> responseFuture = new CompletableFuture<>();

        try {
            if (!isConnected()) {
                log.error("Cannot send message - not connected to server");
                responseFuture.completeExceptionally(new IllegalStateException("Not connected"));
                return responseFuture;
            }

            // Store the future to be completed when response is received
            String stan = isoMsg.getString(11);
            if (stan != null) {
                // This would require a response tracking mechanism
                // For now, we'll just send the message
                sendMessageAsync(isoMsg).thenAccept(success -> {
                    if (success) {
                        log.info("Message sent, waiting for response with STAN: {}", stan);
                        // In a real implementation, you'd store the future and complete it when response arrives
                    } else {
                        responseFuture.completeExceptionally(new RuntimeException("Failed to send message"));
                    }
                });
            } else {
                responseFuture.completeExceptionally(new IllegalArgumentException("STAN is required for response tracking"));
            }

        } catch (Exception e) {
            log.error("Error sending message and waiting for response", e);
            responseFuture.completeExceptionally(e);
        }

        return responseFuture;
    }

    /**
     * Initiates a sign-on request to the server.
     *
     * @return CompletableFuture that completes when sign-on is successful
     */
    public CompletableFuture<Boolean> signOn() {
        CompletableFuture<Boolean> signOnFuture = new CompletableFuture<>();

        try {
            ISOMsg signOnMsg = createSignOnMessage();
            log.info("Initiating sign-on request");

            sendMessageAsync(signOnMsg).thenAccept(success -> {
                if (success) {
                    log.info("Sign-on request sent successfully");
                    // Response will be handled in onMessage
                    signOnFuture.complete(true);
                } else {
                    log.error("Failed to send sign-on request");
                    signOnFuture.complete(false);
                }
            });

        } catch (Exception e) {
            log.error("Error creating sign-on message", e);
            signOnFuture.completeExceptionally(e);
        }

        return signOnFuture;
    }

    /**
     * Initiates a sign-off request to the server.
     *
     * @return CompletableFuture that completes when sign-off is successful
     */
    public CompletableFuture<Boolean> signOff() {
        CompletableFuture<Boolean> signOffFuture = new CompletableFuture<>();

        try {
            ISOMsg signOffMsg = createSignOffMessage();
            log.info("Initiating sign-off request");

            sendMessageAsync(signOffMsg).thenAccept(success -> {
                if (success) {
                    log.info("Sign-off request sent successfully");
                    // Response will be handled in onMessage
                    signOffFuture.complete(true);
                } else {
                    log.error("Failed to send sign-off request");
                    signOffFuture.complete(false);
                }
            });

        } catch (Exception e) {
            log.error("Error creating sign-off message", e);
            signOffFuture.completeExceptionally(e);
        }

        return signOffFuture;
    }

    /**
     * Sends an echo request to the server.
     *
     * @return CompletableFuture that completes when echo is successful
     */
    public CompletableFuture<Boolean> echo() {
        CompletableFuture<Boolean> echoFuture = new CompletableFuture<>();

        try {
            ISOMsg echoMsg = createEchoMessage();
            log.info("Sending echo request");

            sendMessageAsync(echoMsg).thenAccept(success -> {
                if (success) {
                    log.info("Echo request sent successfully");
                    // Response will be handled in onMessage
                    echoFuture.complete(true);
                } else {
                    log.error("Failed to send echo request");
                    echoFuture.complete(false);
                }
            });

        } catch (Exception e) {
            log.error("Error creating echo message", e);
            echoFuture.completeExceptionally(e);
        }

        return echoFuture;
    }

    /**
     * Configures the Netty pipeline for the client channel.
     * Adds length-based framing and logging handlers.
     *
     * @param pipeline The channel pipeline to configure
     */
    @Override
    protected void configurePipeline(ChannelPipeline pipeline) {
        // Add length-based frame decoder
        pipeline.addLast(new LengthFieldBasedFrameDecoder(
                getMaxFrameLength(),
                getLengthFieldOffset(),
                getLengthFieldSize(),
                getLengthAdjustment(),
                getInitialBytesToStrip()));

        // Add byte array decoder
        pipeline.addLast(new ByteArrayDecoder());

        // Add length field prepender for outgoing messages
        pipeline.addLast(new LengthFieldPrepender(
                getLengthFieldSize(),
                getIncludeLengthFieldLength()));

        // Add logging handler
        if (RawDataLogging.isEnabled()) {
            pipeline.addLast(new LoggingHandler(LogLevel.DEBUG));
        }

        // Add this channel handler
        pipeline.addLast(this);
    }

    /**
     * Called when a connection is established.
     * Stores the channel context for sending messages.
     *
     * @param ctx The channel handler context
     */
    @Override
    protected void onConnect(ChannelHandlerContext ctx) {
        this.currentContext = ctx;
        log.info("ISO8583 client channel connected to: {}", ctx.channel().remoteAddress());
    }

    /**
     * Called when a connection is closed.
     * Clears the channel context.
     *
     * @param ctx The channel handler context
     */
    @Override
    protected void onDisconnect(ChannelHandlerContext ctx) {
        this.currentContext = null;
        log.info("ISO8583 client channel disconnected from: {}",
                ctx != null ? ctx.channel().remoteAddress() : "unknown");
    }

    /**
     * Called when a complete ISO8583 message is received.
     *
     * @param ctx The channel handler context
     * @param isoMsg The received ISO8583 message
     */
    @Override
    protected void onMessage(ChannelHandlerContext ctx, ISOMsg isoMsg) {
        try {
        log.info("Received ISO8583 message from server - MTI: {}, STAN: {}",
                isoMsg.getMTI(), isoMsg.getString(11));

        // Handle the message based on MTI
        String mti = isoMsg.getMTI();
        if (mti != null) {
            if (mti.endsWith("800")) {
                handleNetworkMessage(ctx, isoMsg);
            } else {
                handleFinancialMessage(ctx, isoMsg);
            }
        } else {
                log.warn("Received message without MTI");
            }
        } catch (ISOException e) {
            log.error("Error processing ISO8583 message", e);
        }
    }

    /**
     * Called when an error occurs.
     *
     * @param ctx The channel handler context
     * @param cause The error
     */
    @Override
    protected void onError(ChannelHandlerContext ctx, Throwable cause) {
        log.error("ISO8583 client channel error", cause);
        // Close the connection on error
        if (ctx != null) {
            ctx.close();
        }
    }

    // Abstract methods to be implemented by subclasses

    /**
     * Handles network messages (MTI 0800).
     *
     * @param ctx The channel handler context
     * @param isoMsg The network message
     */
    protected abstract void handleNetworkMessage(ChannelHandlerContext ctx, ISOMsg isoMsg);

    /**
     * Handles financial messages.
     *
     * @param ctx The channel handler context
     * @param isoMsg The financial message
     */
    protected abstract void handleFinancialMessage(ChannelHandlerContext ctx, ISOMsg isoMsg);

    // Configuration methods that can be overridden

    /**
     * Gets the length field size.
     *
     * @return The length field size
     */
    protected int getLengthFieldSize() {
        return lengthFieldSize;
    }

    /**
     * Gets the maximum frame length for the decoder.
     *
     * @return The maximum frame length
     */
    protected int getMaxFrameLength() {
        return DEFAULT_MAX_FRAME_LENGTH;
    }

    /**
     * Gets the length field offset.
     *
     * @return The length field offset
     */
    protected int getLengthFieldOffset() {
        return DEFAULT_LENGTH_FIELD_OFFSET;
    }

    /**
     * Gets the length adjustment.
     *
     * @return The length adjustment
     */
    protected int getLengthAdjustment() {
        return DEFAULT_LENGTH_ADJUSTMENT;
    }

    /**
     * Gets the initial bytes to strip.
     *
     * @return The initial bytes to strip
     */
    protected int getInitialBytesToStrip() {
        return DEFAULT_INITIAL_BYTES_TO_STRIP;
    }

    /**
     * Gets whether to include length field length.
     *
     * @return Whether to include length field length
     */
    protected boolean getIncludeLengthFieldLength() {
        return DEFAULT_INCLUDE_LENGTH_FIELD_LENGTH;
    }

    // Message creation methods that can be overridden

    /**
     * Creates a sign-on message.
     *
     * @return The sign-on message
     */
    protected ISOMsg createSignOnMessage() {
        try {
        ISOMsg signOnMsg = new ISOMsg();
        signOnMsg.setPackager(getPackager());
        signOnMsg.setMTI("0800");
        signOnMsg.set(70, "001"); // Sign-on request
        signOnMsg.set(7, getTransmissionDateTime());
        signOnMsg.set(11, getNextStan());
        return signOnMsg;
        } catch (ISOException e) {
            log.error("Error creating sign-on message", e);
            return null;
        }
    }

    /**
     * Creates a sign-off message.
     *
     * @return The sign-off message
     */
    protected ISOMsg createSignOffMessage() {
        try {
        ISOMsg signOffMsg = new ISOMsg();
        signOffMsg.setPackager(getPackager());
        signOffMsg.setMTI("0800");
        signOffMsg.set(70, "002"); // Sign-off request
        signOffMsg.set(7, getTransmissionDateTime());
        signOffMsg.set(11, getNextStan());
        return signOffMsg;
        } catch (ISOException e) {
            log.error("Error creating sign-off message", e);
            return null;
        }
    }

    /**
     * Creates an echo message.
     *
     * @return The echo message
     */
    protected ISOMsg createEchoMessage() {
        try {
        ISOMsg echoMsg = new ISOMsg();
        echoMsg.setPackager(getPackager());
        echoMsg.setMTI("0800");
        echoMsg.set(70, "301"); // Echo request
        echoMsg.set(7, getTransmissionDateTime());
        echoMsg.set(11, getNextStan());
        return echoMsg;
        } catch (ISOException e) {
            log.error("Error creating echo message", e);
            return null;
        }
    }

    // Utility methods

    /**
     * Gets the current transmission date and time in MMDDhhmmss format.
     *
     * @return The transmission date and time string
     */
    protected String getTransmissionDateTime() {
        return java.time.LocalDateTime.now().format(java.time.format.DateTimeFormatter.ofPattern("MMddHHmmss"));
    }

    /**
     * Gets the next System Trace Audit Number (STAN).
     * This is a simple implementation - in production, this should be thread-safe and persistent.
     *
     * @return The next STAN
     */
    protected String getNextStan() {
        return String.format("%06d", (int) (Math.random() * 999999));
    }

    /**
     * Gets the client connection status information.
     *
     * @return A string describing the client connection status
     */
    public String getClientStatus() {
        return String.format("ISO8583 Client - %s, Context: %s",
                getConnectionStatus(),
                currentContext != null ? "Active" : "Inactive");
    }
}
