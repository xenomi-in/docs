package com.lps.isoswitch.tcpconnector.channel;

import io.netty.channel.ChannelHandler.Sharable;
import io.netty.channel.ChannelHandlerContext;

import org.jpos.iso.ISOException;
import org.jpos.iso.ISOMsg;
import org.jpos.iso.ISOPackager;
import org.jpos.iso.packager.GenericPackager;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import com.lps.isoswitch.config.RawDataLogging;

import java.io.InputStream;

/**
 * RuPay Client Channel implementation for connecting to RuPay servers.
 * This class provides RuPay-specific client functionality with proper
 * length-based message framing and connection management.
 *
 * Features:
 * - RuPay-specific configuration (2-byte binary length fields)
 * - Client connection management
 * - Automatic reconnection
 * - ISO8583 message handling
 * - Network and financial message processing
 * - Sign-on/Sign-off/Echo functionality
 *
 * @author Logicpoint Systems
 * @version 1.0
 * @since 2025-06-15
 */
@Sharable
public class RuPayClientChannel extends ISOClientChannel {

    private static final Logger log = LoggerFactory.getLogger(RuPayClientChannel.class);

    // RuPay-specific configuration
    private static final int RUPAY_LENGTH_FIELD_SIZE = 2;
    private static final int RUPAY_MAX_MESSAGE_LENGTH = 6144;
    private static final boolean RUPAY_INCLUDE_LENGTH_FIELD = false;
    private static final boolean RUPAY_LENGTH_IN_ASCII = false;
    private static final boolean RUPAY_LENGTH_IN_HEX = false;

    // Connection status
    private boolean signedOn = false;

    /**
     * Constructs a new RuPayClientChannel with default RuPay configuration.
     */
    public RuPayClientChannel() {
        super(RUPAY_MAX_MESSAGE_LENGTH);
        configureRuPayChannel();
        initializeRuPayPackager();
    }

    /**
     * Constructs a new RuPayClientChannel with custom maximum message length.
     *
     * @param maxMessageLength The maximum message length in bytes
     */
    public RuPayClientChannel(int maxMessageLength) {
        super(maxMessageLength);
        configureRuPayChannel();
        initializeRuPayPackager();
    }

    /**
     * Configures the channel for RuPay protocol requirements.
     */
    private void configureRuPayChannel() {
        setLengthFieldConfig(RUPAY_LENGTH_FIELD_SIZE, RUPAY_INCLUDE_LENGTH_FIELD,RUPAY_LENGTH_IN_HEX, RUPAY_LENGTH_IN_ASCII);
        setTimeout(30000, 30000); // 30 seconds timeout
        setReconnectConfig(true, 5000); // Auto-reconnect with 5 second delay
    }

    /**
     * Initializes the RuPay ISO8583 packager.
     */
    private void initializeRuPayPackager() {
        try {
            InputStream is = getClass().getResourceAsStream("/RuPayISO8583.xml");
            if (is == null) {
                log.error("Could not find RuPayISO8583.xml in classpath");
                throw new RuntimeException("RuPay packager configuration file not found");
            }
            ISOPackager ruPayPackager = new GenericPackager(is);
            setPackager(ruPayPackager);
            is.close();
            log.info("RuPay ISO8583 packager initialized successfully");
        } catch (Exception e) {
            log.error("Error initializing RuPay ISO8583 packager", e);
            throw new RuntimeException("Failed to initialize RuPay packager", e);
        }
    }

    /**
     * Connects to a RuPay server with the specified host and port.
     *
     * @param host The host address to connect to
     * @param port The port number to connect to
     * @return CompletableFuture that completes when connection is established
     */
    public java.util.concurrent.CompletableFuture<Boolean> connectToRuPayServer(String host, int port) {
        log.info("Connecting to RuPay server at {}:{}", host, port);
        return connect(host, port);
    }

    /**
     * Connects to a RuPay server with timeout.
     *
     * @param host The host address to connect to
     * @param port The port number to connect to
     * @param timeoutMs Connection timeout in milliseconds
     * @return CompletableFuture that completes when connection is established
     */
    public java.util.concurrent.CompletableFuture<Boolean> connectToRuPayServer(String host, int port, long timeoutMs) {
        log.info("Connecting to RuPay server at {}:{} with timeout {}ms", host, port, timeoutMs);

        java.util.concurrent.CompletableFuture<Boolean> connectFuture = connect(host, port);

        // Add timeout
        java.util.concurrent.CompletableFuture<Boolean> timeoutFuture = new java.util.concurrent.CompletableFuture<>();
        connectFuture.orTimeout(timeoutMs, java.util.concurrent.TimeUnit.MILLISECONDS)
                    .whenComplete((result, throwable) -> {
                        if (throwable != null) {
                            log.error("Connection timeout to RuPay server {}:{}", host, port);
                            timeoutFuture.complete(false);
                        } else {
                            timeoutFuture.complete(result);
                        }
                    });

        return timeoutFuture;
    }

    /**
     * Called when a connection is established.
     *
     * @param ctx The channel handler context
     */
    @Override
    protected void onConnect(ChannelHandlerContext ctx) {
        log.info("RuPay client channel connected to: {}", ctx.channel().remoteAddress());
        signedOn = false; // Reset sign-on status on new connection
    }

    /**
     * Called when a connection is closed.
     *
     * @param ctx The channel handler context
     */
    @Override
    protected void onDisconnect(ChannelHandlerContext ctx) {
        log.info("RuPay client channel disconnected from: {}",
                ctx != null ? ctx.channel().remoteAddress() : "unknown");
        signedOn = false; // Reset sign-on status on disconnection
    }

    /**
     * Called when a complete ISO8583 message is received.
     *
     * @param ctx The channel handler context
     * @param isoMsg The received ISO8583 message
     */
    @Override
    protected void onMessage(ChannelHandlerContext ctx, ISOMsg isoMsg) {
        try {
            log.info("Received RuPay ISO8583 message - MTI: {}, STAN: {}",
                    isoMsg.getMTI(), isoMsg.getString(11));

            // Log message details for debugging
            RawDataLogging.isoMessage("RuPay client channel ISO message", isoMsg);

            // Handle the message based on MTI
            String mti = isoMsg.getMTI();
            if (mti != null) {
                if (mti.endsWith("800")) {
                    handleNetworkMessage(ctx, isoMsg);
                } else {
                    handleFinancialMessage(ctx, isoMsg);
                }
            } else {
                log.warn("Received RuPay message without MTI");
            }
        } catch (Exception e) {
            log.error("Error processing RuPay message", e);
            onError(ctx, e);
        }
    }

    /**
     * Called when an error occurs.
     *
     * @param ctx The channel handler context
     * @param cause The error
     */
    @Override
    protected void onError(ChannelHandlerContext ctx, Throwable cause) {
        log.error("RuPay client channel error", cause);
        // Close the connection on error
        if (ctx != null) {
            ctx.close();
        }
    }

    /**
     * Handles network messages (MTI 0800).
     *
     * @param ctx The channel handler context
     * @param isoMsg The network message
     */
    @Override
    protected void handleNetworkMessage(ChannelHandlerContext ctx, ISOMsg isoMsg) {
        try {
            log.info("Handling RuPay network message - MTI: {}", isoMsg.getMTI());

            // Check network management information code
            String networkCode = isoMsg.getString(70);
            if (networkCode != null) {
                switch (networkCode) {
                    case "001": // Sign-on
                        handleSignOnResponse(ctx, isoMsg);
                        break;
                    case "002": // Sign-off
                        handleSignOffResponse(ctx, isoMsg);
                        break;
                    case "301": // Echo
                        handleEchoResponse(ctx, isoMsg);
                        break;
                    default:
                        log.info("Unknown network management code: {}", networkCode);
                }
            }
        } catch (Exception e) {
            log.error("Error handling RuPay network message", e);
        }
    }

    /**
     * Handles financial messages.
     *
     * @param ctx The channel handler context
     * @param isoMsg The financial message
     */
    @Override
    protected void handleFinancialMessage(ChannelHandlerContext ctx, ISOMsg isoMsg) {
        try {
            log.info("Handling RuPay financial message - MTI: {}", isoMsg.getMTI());

            // Check if we're signed on
            if (!signedOn) {
                log.warn("Received financial message but not signed on");
                // Send decline response
                sendDeclineResponse(ctx, isoMsg, "96", "Not signed on");
                return;
            }

            // Process the financial message
            processFinancialMessage(ctx, isoMsg);

        } catch (Exception e) {
            log.error("Error handling RuPay financial message", e);
        }
    }

    /**
     * Handles sign-on response.
     *
     * @param ctx The channel handler context
     * @param isoMsg The sign-on response message
     */
    protected void handleSignOnResponse(ChannelHandlerContext ctx, ISOMsg isoMsg) {
        try {
            String responseCode = isoMsg.getString(39);
            if ("00".equals(responseCode)) {
                signedOn = true;
                log.info("RuPay sign-on successful");
            } else {
                signedOn = false;
                log.error("RuPay sign-on failed with response code: {}", responseCode);
            }
        } catch (Exception e) {
            log.error("Error handling sign-on response", e);
        }
    }

    /**
     * Handles sign-off response.
     *
     * @param ctx The channel handler context
     * @param isoMsg The sign-off response message
     */
    protected void handleSignOffResponse(ChannelHandlerContext ctx, ISOMsg isoMsg) {
        try {
            String responseCode = isoMsg.getString(39);
            if ("00".equals(responseCode)) {
                signedOn = false;
                log.info("RuPay sign-off successful");
            } else {
                log.error("RuPay sign-off failed with response code: {}", responseCode);
            }
        } catch (Exception e) {
            log.error("Error handling sign-off response", e);
        }
    }

    /**
     * Handles echo response.
     *
     * @param ctx The channel handler context
     * @param isoMsg The echo response message
     */
    protected void handleEchoResponse(ChannelHandlerContext ctx, ISOMsg isoMsg) {
        try {
            String responseCode = isoMsg.getString(39);
            if ("00".equals(responseCode)) {
                log.info("RuPay echo successful");
            } else {
                log.error("RuPay echo failed with response code: {}", responseCode);
            }
        } catch (Exception e) {
            log.error("Error handling echo response", e);
        }
    }

    /**
     * Processes financial messages.
     * This method should be overridden by subclasses to provide specific financial message handling.
     *
     * @param ctx The channel handler context
     * @param isoMsg The financial message
     */
    protected void processFinancialMessage(ChannelHandlerContext ctx, ISOMsg isoMsg) {
        try {
            log.info("Processing RuPay financial message - MTI: {}", isoMsg.getMTI());
            // Default implementation - can be overridden by subclasses
        } catch (ISOException e) {
            log.error("Error processing RuPay financial message", e);
        }
    }

    /**
     * Sends a decline response.
     *
     * @param ctx The channel handler context
     * @param originalMsg The original message
     * @param responseCode The response code
     * @param errorMessage The error message
     */
    protected void sendDeclineResponse(ChannelHandlerContext ctx, ISOMsg originalMsg, String responseCode, String errorMessage) {
        try {
            log.error("Sending RuPay decline response. Error: {}", errorMessage);

            ISOMsg responseMsg = new ISOMsg();
            responseMsg.setPackager(getPackager());

            // Set response MTI (change last digit to 1 for response)
            String mti = originalMsg.getMTI();
            int mtiNumber = Integer.parseInt(mti) + 10;
            responseMsg.setMTI("0" + mtiNumber);

            // Copy essential fields from original message
            copyEssentialFields(originalMsg, responseMsg);

            // Set response code
            responseMsg.set(39, responseCode);

            // Set additional fields based on response code
            switch (responseCode) {
                case "96":
                    responseMsg.set(44, "SYSTEM ERROR");
                    break;
                case "91":
                    responseMsg.set(44, "SERVICE UNAVAILABLE");
                    break;
                case "05":
                    responseMsg.set(44, "DECLINED");
                    break;
                case "12":
                    responseMsg.set(44, "INVALID TRANSACTION");
                    break;
                default:
                    responseMsg.set(44, "DECLINED");
            }

            // Send the response
            sendMessage(ctx, responseMsg);
            log.info("RuPay decline response sent successfully");

        } catch (Exception e) {
            log.error("Error sending RuPay decline response", e);
        }
    }

    /**
     * Copies essential fields from the original message to the response message.
     *
     * @param originalMsg The original ISO message
     * @param responseMsg The response ISO message
     */
    protected void copyEssentialFields(ISOMsg originalMsg, ISOMsg responseMsg) {
        try {
            // Copy essential fields that should be present in response
            if (originalMsg.hasField(2)) responseMsg.set(2, originalMsg.getString(2));
            if (originalMsg.hasField(3)) responseMsg.set(3, originalMsg.getString(3));
            if (originalMsg.hasField(4)) responseMsg.set(4, originalMsg.getString(4));
            if (originalMsg.hasField(7)) responseMsg.set(7, originalMsg.getString(7));
            if (originalMsg.hasField(11)) responseMsg.set(11, originalMsg.getString(11));
            if (originalMsg.hasField(12)) responseMsg.set(12, originalMsg.getString(12));
            if (originalMsg.hasField(13)) responseMsg.set(13, originalMsg.getString(13));
            if (originalMsg.hasField(14)) responseMsg.set(14, originalMsg.getString(14));
            if (originalMsg.hasField(22)) responseMsg.set(22, originalMsg.getString(22));
            if (originalMsg.hasField(25)) responseMsg.set(25, originalMsg.getString(25));
            if (originalMsg.hasField(35)) responseMsg.set(35, originalMsg.getString(35));
            if (originalMsg.hasField(37)) responseMsg.set(37, originalMsg.getString(37));
            if (originalMsg.hasField(41)) responseMsg.set(41, originalMsg.getString(41));
            if (originalMsg.hasField(42)) responseMsg.set(42, originalMsg.getString(42));
            if (originalMsg.hasField(43)) responseMsg.set(43, originalMsg.getString(43));
            if (originalMsg.hasField(49)) responseMsg.set(49, originalMsg.getString(49));
            if (originalMsg.hasField(55)) responseMsg.set(55, originalMsg.getString(55));
        } catch (Exception e) {
            log.error("Error copying essential fields", e);
        }
    }

    /**
     * Checks if the client is signed on.
     *
     * @return true if signed on, false otherwise
     */
    public boolean isSignedOn() {
        return signedOn;
    }

    /**
     * Gets the RuPay client status information.
     *
     * @return A string describing the RuPay client status
     */
    public String getRuPayClientStatus() {
        return String.format("RuPay Client - %s, Signed On: %s",
                getConnectionStatus(), signedOn ? "Yes" : "No");
    }
}
