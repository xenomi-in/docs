package com.lps.isoswitch.tcpconnector.channel;

import io.netty.buffer.Unpooled;
import io.netty.channel.ChannelHandlerContext;

import io.netty.channel.ChannelFuture;
import io.netty.channel.ChannelFutureListener;
import io.netty.channel.EventLoopGroup;
import io.netty.channel.nio.NioEventLoopGroup;
import io.netty.channel.socket.nio.NioSocketChannel;
import io.netty.channel.ChannelInitializer;
import io.netty.channel.socket.SocketChannel;
import io.netty.channel.ChannelPipeline;
import org.jpos.iso.ISOMsg;
import org.jpos.iso.ISOPackager;
import org.jpos.iso.ISOException;
import org.slf4j.Logger;

/**
 * Abstract base class for ISO8583 channels that provides length-based message reading.
 * This class implements message framing similar to jPOS channels with support for
 * different length formats (ASCII, HEX, BINARY) and client connection capabilities.
 * 
 * Features:
 * - Length-based message framing
 * - Support for different length formats (2-byte, 4-byte, ASCII, HEX)
 * - Message buffering for incomplete messages
 * - Proper error handling and logging
 * - Thread-safe message processing
 * - Client connection management
 * - Automatic reconnection handling
 * 
 * @author Logicpoint Systems
 * @version 1.0
 * @since 2025-06-15
 */
public abstract class AbstractISOChannel extends ChannelInboundHandlerAdapter {

    private static final Logger log = LoggerFactory.getLogger(AbstractISOChannel.class);
    
    // Length format constants
    protected static final int LENGTH_FORMAT_2BYTE = 2;
    protected static final int LENGTH_FORMAT_4BYTE = 4;
    protected static final int LENGTH_FORMAT_ASCII = 1;
    protected static final int LENGTH_FORMAT_HEX = 2;
    
    // Default configuration
    protected static final int DEFAULT_MAX_MESSAGE_LENGTH = 8192;
    protected static final int DEFAULT_READ_TIMEOUT = 30000;
    protected static final int DEFAULT_WRITE_TIMEOUT = 30000;
    protected static final int DEFAULT_CONNECT_TIMEOUT = 10000;
    protected static final int DEFAULT_RECONNECT_DELAY = 5000;
    
    // Channel state
    protected final AtomicInteger messageCounter = new AtomicInteger(0);
    protected ByteBuffer messageBuffer = ByteBuffer.allocate(DEFAULT_MAX_MESSAGE_LENGTH);
    protected boolean connected = false;
    protected long lastActivity = System.currentTimeMillis();
    
    // Client connection management
    protected Bootstrap bootstrap;
    protected EventLoopGroup eventLoopGroup;
    protected String host;
    protected int port;
    protected final AtomicBoolean isConnecting = new AtomicBoolean(false);
    protected final AtomicBoolean isReconnecting = new AtomicBoolean(false);
    protected ScheduledExecutorService reconnectExecutor;
    protected final AtomicBoolean isShuttingDown = new AtomicBoolean(false);
    
    // Configuration
    protected int lengthFieldSize = LENGTH_FORMAT_2BYTE;
    protected int maxMessageLength = DEFAULT_MAX_MESSAGE_LENGTH;
    protected int readTimeout = DEFAULT_READ_TIMEOUT;
    protected int writeTimeout = DEFAULT_WRITE_TIMEOUT;
    protected int connectTimeout = DEFAULT_CONNECT_TIMEOUT;
    protected int reconnectDelay = DEFAULT_RECONNECT_DELAY;
    protected boolean includeLengthField = false;
    protected boolean lengthInHex = false;
    protected boolean lengthInAscii = false;
    protected boolean autoReconnect = true;
    
    // ISO8583 packager
    protected ISOPackager packager;
    
    /**
     * Constructs a new AbstractISOChannel with default configuration.
     */
    public AbstractISOChannel() {
        this(DEFAULT_MAX_MESSAGE_LENGTH);
    }
    
    /**
     * Constructs a new AbstractISOChannel with specified maximum message length.
     *
     * @param maxMessageLength The maximum message length in bytes
     */
    public AbstractISOChannel(int maxMessageLength) {
        this.maxMessageLength = maxMessageLength;
        this.messageBuffer = ByteBuffer.allocate(maxMessageLength);
        this.eventLoopGroup = new NioEventLoopGroup();
        this.reconnectExecutor = Executors.newSingleThreadScheduledExecutor();
    }
    
    /**

    }

    /**
     * Sets the length field configuration.
     *
     * @param lengthFieldSize Size of the length field (2 or 4 bytes)
     * @param includeLengthField Whether to include length field in message
     * @param lengthInHex Whether length is in hexadecimal format
     * @param lengthInAscii Whether length is in ASCII format

    public void setLengthFieldConfig(int lengthFieldSize, boolean includeLengthField, 
                                   boolean lengthInHex, boolean lengthInAscii) {
        this.lengthFieldSize = lengthFieldSize;
        this.includeLengthField = includeLengthField;
        this.lengthInHex = lengthInHex;
        this.lengthInAscii = lengthInAscii;
    }

    public void setTimeout(int readTimeout, int writeTimeout) {
        this.readTimeout = readTimeout;
        this.writeTimeout = writeTimeout;
    }

    public void setConnectTimeout(int connectTimeout) {
        this.connectTimeout = connectTimeout;
    }
    
    /**
     * Sets reconnection configuration.
     *
     * @param autoReconnect Whether to automatically reconnect
     * @param reconnectDelay Reconnection delay in milliseconds
     */
    public void setReconnectConfig(boolean autoReconnect, int reconnectDelay) {
        this.autoReconnect = autoReconnect;
        this.reconnectDelay = reconnectDelay;
    }
    
    /**
     * Connects to the specified host and port as a client.
     *
     * @param host The host address to connect to
     * @param port The port number to connect to
     * @return CompletableFuture that completes when connection is established
     */
    public CompletableFuture<Boolean> connect(String host, int port) {
        this.host = host;
        this.port = port;
        
        CompletableFuture<Boolean> connectFuture = new CompletableFuture<>();
        
        if (isConnecting.compareAndSet(false, true)) {
            try {
                log.info("Connecting to {}:{}", host, port);
                
                // Create bootstrap if not exists
                if (bootstrap == null) {
                    bootstrap = new Bootstrap();
                    bootstrap.group(eventLoopGroup)
                            .channel(NioSocketChannel.class)
                            .handler(new ChannelInitializer<SocketChannel>() {
                                @Override
                                protected void initChannel(SocketChannel ch) throws Exception {
                                    ChannelPipeline pipeline = ch.pipeline();
                                    configurePipeline(pipeline);
                                }
                            });
                }
                
                // Connect to server
                bootstrap.connect(host, port).addListener((ChannelFutureListener) future -> {
                    if (future.isSuccess()) {
                        log.info("Successfully connected to {}:{}", host, port);
                        connected = true;
                        lastActivity = System.currentTimeMillis();
                        onConnect(future.channel().pipeline().context(this));
                        connectFuture.complete(true);
                    } else {
                        log.error("Failed to connect to {}:{}", host, port, future.cause());
                        connected = false;
                        connectFuture.complete(false);
                        
                        if (autoReconnect) {
                            scheduleReconnect();
                        }
                    }
                    isConnecting.set(false);
                });
                
            } catch (Exception e) {
                log.error("Error during connection attempt", e);
                isConnecting.set(false);
                connectFuture.completeExceptionally(e);
            }
        } else {
            connectFuture.complete(false);
        }
        
        return connectFuture;
    }
    
    /**
     * Disconnects from the server.
     *
     * @return CompletableFuture that completes when disconnection is finished
     */
    public CompletableFuture<Void> disconnect() {
        CompletableFuture<Void> disconnectFuture = new CompletableFuture<>();
        
        try {
            log.info("Disconnecting from {}:{}", host, port);
            
            if (connected) {
                // Close the channel
                if (bootstrap != null && bootstrap.config().group() != null) {
                    bootstrap.config().group().shutdownGracefully();
                }
                
                connected = false;
                onDisconnect(null);
                log.info("Disconnected from {}:{}", host, port);
            }
            
            disconnectFuture.complete(null);
            
        } catch (Exception e) {
            log.error("Error during disconnection", e);
            disconnectFuture.completeExceptionally(e);
        }
        
        return disconnectFuture;
    }
    
    /**
     * Schedules a reconnection attempt using virtual threads.
     */
    protected void scheduleReconnect() {
        // Check if we're shutting down
        if (isShuttingDown.get()) {
            log.info("Skipping reconnection - channel is shutting down");
            return;
        }
        
        if (isReconnecting.compareAndSet(false, true)) {
            log.info("Scheduling reconnection to {}:{} in {} ms using virtual threads", host, port, reconnectDelay);
            
            // Use virtual thread executor for reconnection scheduling
            CompletableFuture.runAsync(() -> {
                try {
                    Thread.sleep(reconnectDelay);
                    
                    // Check shutdown state before attempting reconnection
                    if (isShuttingDown.get()) {
                        log.info("Cancelling reconnection - channel is shutting down");
                        isReconnecting.set(false);
                        return;
                    }
                    
                    if (!connected && autoReconnect) {
                        log.info("Attempting to reconnect to {}:{} using virtual threads", host, port);
                        connect(host, port).thenAccept(success -> {
                            if (!success && autoReconnect && !isShuttingDown.get()) {
                                scheduleReconnect();
                            }
                        });
                    }
                } catch (Exception e) {
                    log.error("Error during reconnection attempt", e);
                    if (autoReconnect && !isShuttingDown.get()) {
                        scheduleReconnect();
                    }
                } finally {
                    isReconnecting.set(false);
                }
            }, reconnectExecutor);
        }
    }
    
    /**
     * Configures the Netty pipeline for the channel.
     * This method should be overridden by subclasses to add specific handlers.
     *
     * @param pipeline The channel pipeline to configure
     */
    protected void configurePipeline(ChannelPipeline pipeline) {
        // Add this channel handler to the pipeline
        pipeline.addLast(this);
    }
    
    /**
     * Called when a connection is established with the server.
     * Resets the reconnection flag and logs the connection status.
     *
     * @param ctx The channel handler context
     */
    @Override
    public void channelActive(ChannelHandlerContext ctx) {
        this.connected = true;
        this.lastActivity = System.currentTimeMillis();
        log.info("Channel connected to: {}", ctx.channel().remoteAddress());
        onConnect(ctx);
    }

    /**
     * Called when the connection with the server is lost.
     * Initiates reconnection process if not already reconnecting.
     *
     * @param ctx The channel handler context
     */
    @Override
    public void channelInactive(ChannelHandlerContext ctx) {
        this.connected = false;
        log.info("Channel disconnected from: {}", ctx.channel().remoteAddress());
        onDisconnect(ctx);
        
        if (autoReconnect && !isReconnecting.get()) {
            scheduleReconnect();
        }
    }

    /**
     * Main message reading method that handles length-based message framing.
     *
     * @param ctx The channel handler context
     * @param msg The received message
     */
    @Override
    public void channelRead(ChannelHandlerContext ctx, Object msg) {
        try {
            ByteBuf buf = (ByteBuf) msg;
            byte[] data = new byte[buf.readableBytes()];
            buf.readBytes(data);
            
            log.debug("Received {} bytes from channel", data.length);
            
            // Process the received data
            processReceivedData(ctx, data);
            
        } catch (Exception e) {
            log.error("Error reading from channel", e);
            onError(ctx, e);
        } finally {
            if (msg instanceof ByteBuf) {
                ((ByteBuf) msg).release();
            }
        }
    }
    
    /**
     * Processes received data with length-based framing.
     *
     * @param ctx The channel handler context
     * @param data The received data
     */
    protected void processReceivedData(ChannelHandlerContext ctx, byte[] data) {
        // Add data to buffer
        messageBuffer.put(data);
        messageBuffer.flip();
        
        // Process complete messages
        while (messageBuffer.remaining() >= getMinimumMessageLength()) {
            // Check if we have enough data to read the length
            if (messageBuffer.remaining() < lengthFieldSize) {
                break;
            }
            
            // Mark current position
            messageBuffer.mark();
            
            // Read message length
            int messageLength = readMessageLength();
            
            // Validate message length
            if (messageLength <= 0 || messageLength > maxMessageLength) {
                log.error("Invalid message length: {}", messageLength);
                messageBuffer.reset();
                break;
            }
            
            // Check if we have the complete message
            int totalLength = includeLengthField ? messageLength : messageLength + lengthFieldSize;
            if (messageBuffer.remaining() < totalLength) {
                messageBuffer.reset();
                break;
            }
            
            // Extract the complete message
            byte[] messageData = new byte[messageLength];
            if (includeLengthField) {
                // Skip length field and read message
                messageBuffer.position(messageBuffer.position() + lengthFieldSize);
                messageBuffer.get(messageData);
            } else {
                // Read message directly
                messageBuffer.get(messageData);
            }
            
            // Process the complete message
            processCompleteMessage(ctx, messageData);
            
            // Compact buffer for next message
            messageBuffer.compact();
        }
    }
    
    /**
     * Reads the message length from the buffer based on the configured format.
     *
     * @return The message length
     */
    protected int readMessageLength() {
        if (lengthInAscii) {
            return readAsciiLength();
        } else if (lengthInHex) {
            return readHexLength();
        } else {
            return readBinaryLength();
        }
    }
    
    /**
     * Reads ASCII length from the buffer.
     *
     * @return The message length
     */
    protected int readAsciiLength() {
        byte[] lengthBytes = new byte[lengthFieldSize];
        messageBuffer.get(lengthBytes);
        String lengthStr = new String(lengthBytes, StandardCharsets.US_ASCII);
        return Integer.parseInt(lengthStr);
    }
    
    /**
     * Reads hexadecimal length from the buffer.
     *
     * @return The message length
     */
    protected int readHexLength() {
        byte[] lengthBytes = new byte[lengthFieldSize];
        messageBuffer.get(lengthBytes);
        String lengthStr = new String(lengthBytes, StandardCharsets.US_ASCII);
        return Integer.parseInt(lengthStr, 16);
    }
    
    /**
     * Reads binary length from the buffer.
     *
     * @return The message length
     */
    protected int readBinaryLength() {
        if (lengthFieldSize == 2) {
            return messageBuffer.getShort() & 0xFFFF;
        } else if (lengthFieldSize == 4) {
            return messageBuffer.getInt();
        } else {
            throw new IllegalArgumentException("Unsupported length field size: " + lengthFieldSize);
        }
    }
    
    /**
     * Gets the minimum message length required to start processing.
     *
     * @return The minimum message length
     */
    protected int getMinimumMessageLength() {
        return lengthFieldSize;
    }
    
    /**
     * Processes a complete ISO8583 message.
     *
     * @param ctx The channel handler context
     * @param messageData The complete message data
     */
    protected void processCompleteMessage(ChannelHandlerContext ctx, byte[] messageData) {
        try {
            log.debug("Processing complete message of {} bytes", messageData.length);
            
            // Parse ISO8583 message
            ISOMsg isoMsg = parseISOMessage(messageData);
            
            // Handle the message
            onMessage(ctx, isoMsg);
            
            // Update activity timestamp
            lastActivity = System.currentTimeMillis();
            messageCounter.incrementAndGet();
            
        } catch (Exception e) {
            log.error("Error processing message", e);
            onError(ctx, e);
        }
    }
    
    /**
     * Parses ISO8583 message from byte array.
     *
     * @param messageData The message data
     * @return The parsed ISO8583 message
     * @throws ISOException if parsing fails
     */
    protected ISOMsg parseISOMessage(byte[] messageData) throws ISOException {
        if (packager == null) {
            throw new ISOException("Packager not set");
        }
        
        ISOMsg isoMsg = new ISOMsg();
        isoMsg.setPackager(packager);
        isoMsg.unpack(messageData);
        
        return isoMsg;
    }
    
    /**
     * Sends an ISO8583 message with proper length prefixing.
     *
     * @param ctx The channel handler context
     * @param isoMsg The ISO8583 message to send
     * @throws ISOException if packing fails
     */
    public void sendMessage(ChannelHandlerContext ctx, ISOMsg isoMsg) throws ISOException {
        if (!connected) {
            throw new IllegalStateException("Channel not connected");
        }
        
        byte[] messageData = isoMsg.pack();
        byte[] framedMessage = frameMessage(messageData);
        
        log.debug("Sending message of {} bytes (framed: {} bytes)", 
                 messageData.length, framedMessage.length);
        
        ByteBuf buf = Unpooled.wrappedBuffer(framedMessage);
        ctx.writeAndFlush(buf);
        
        lastActivity = System.currentTimeMillis();
    }
    
    /**
     * Sends an ISO8583 message using the current channel context.
     *
     * @param isoMsg The ISO8583 message to send
     * @throws ISOException if packing fails
     */
    public void sendMessage(ISOMsg isoMsg) throws ISOException {
        if (bootstrap == null || bootstrap.config().group() == null) {
            throw new IllegalStateException("Channel not initialized");
        }
        
        // This is a simplified version - in practice, you'd need to store the channel context
        throw new UnsupportedOperationException("Use sendMessage(ctx, isoMsg) with channel context");
    }
    
    /**
     * Frames a message with length prefix according to the channel configuration.
     *
     * @param messageData The message data to frame
     * @return The framed message
     */
    protected byte[] frameMessage(byte[] messageData) {
        int messageLength = messageData.length;
        byte[] lengthBytes = encodeLength(messageLength);
        
        if (includeLengthField) {
            // Include length field in the message
            byte[] framedMessage = new byte[lengthBytes.length + messageData.length];
            System.arraycopy(lengthBytes, 0, framedMessage, 0, lengthBytes.length);
            System.arraycopy(messageData, 0, framedMessage, lengthBytes.length, messageData.length);
            return framedMessage;
        } else {
            // Length field is separate
            byte[] framedMessage = new byte[lengthBytes.length + messageData.length];
            System.arraycopy(lengthBytes, 0, framedMessage, 0, lengthBytes.length);
            System.arraycopy(messageData, 0, framedMessage, lengthBytes.length, messageData.length);
            return framedMessage;
        }
    }
    
    /**
     * Encodes the message length according to the channel configuration.
     *
     * @param length The message length
     * @return The encoded length bytes
     */
    protected byte[] encodeLength(int length) {
        if (lengthInAscii) {
            return encodeAsciiLength(length);
        } else if (lengthInHex) {
            return encodeHexLength(length);
        } else {
            return encodeBinaryLength(length);
        }
    }
    
    /**
     * Encodes length in ASCII format.
     *
     * @param length The message length
     * @return The ASCII encoded length
     */
    protected byte[] encodeAsciiLength(int length) {
        String lengthStr = String.format("%0" + lengthFieldSize + "d", length);
        return lengthStr.getBytes(StandardCharsets.US_ASCII);
    }
    
    /**
     * Encodes length in hexadecimal format.
     *
     * @param length The message length
     * @return The hexadecimal encoded length
     */
    protected byte[] encodeHexLength(int length) {
        String lengthStr = String.format("%0" + (lengthFieldSize * 2) + "x", length);
        return lengthStr.getBytes(StandardCharsets.US_ASCII);
    }
    
    /**
     * Encodes length in binary format.
     *
     * @param length The message length
     * @return The binary encoded length
     */
    protected byte[] encodeBinaryLength(int length) {
        ByteBuffer buffer = ByteBuffer.allocate(lengthFieldSize);
        if (lengthFieldSize == 2) {
            buffer.putShort((short) length);
        } else if (lengthFieldSize == 4) {
            buffer.putInt(length);
        } else {
            throw new IllegalArgumentException("Unsupported length field size: " + lengthFieldSize);
        }
        return buffer.array();
    }
    
    /**
     * Called when an exception occurs.
     *
     * @param ctx The channel handler context
     * @param cause The exception
     */
    @Override
    public void exceptionCaught(ChannelHandlerContext ctx, Throwable cause) {
        log.error("Channel exception", cause);
        onError(ctx, cause);
    }
    
    /**
     * Shuts down the channel and releases resources.
     *
     * @return CompletableFuture that completes when shutdown is finished
     */
    public CompletableFuture<Void> shutdown() {
        CompletableFuture<Void> shutdownFuture = new CompletableFuture<>();
        
        try {
            log.info("Shutting down channel");
            
            // Disconnect if connected
            if (connected) {
                disconnect().thenRun(() -> {
                    try {
                        // Shutdown event loop group
                        if (eventLoopGroup != null) {
                            eventLoopGroup.shutdownGracefully();
                        }
                        
                        // Shutdown reconnect executor
                        if (reconnectExecutor != null) {
                            reconnectExecutor.shutdown();
                            if (!reconnectExecutor.awaitTermination(5, TimeUnit.SECONDS)) {
                                reconnectExecutor.shutdownNow();
                            }
                        }
                        
                        log.info("Channel shutdown completed");
                        shutdownFuture.complete(null);
                        
                    } catch (Exception e) {
                        log.error("Error during channel shutdown", e);
                        shutdownFuture.completeExceptionally(e);
                    }
                });
            } else {
                // Shutdown event loop group
                if (eventLoopGroup != null) {
                    eventLoopGroup.shutdownGracefully();
                }
                
                // Shutdown reconnect executor
                if (reconnectExecutor != null) {
                    reconnectExecutor.shutdown();
                    if (!reconnectExecutor.awaitTermination(5, TimeUnit.SECONDS)) {
                        reconnectExecutor.shutdownNow();
                    }
                }
                
                log.info("Channel shutdown completed");
                shutdownFuture.complete(null);
            }
            
        } catch (Exception e) {
            log.error("Error during channel shutdown", e);
            shutdownFuture.completeExceptionally(e);
        }
        
        return shutdownFuture;
    }
