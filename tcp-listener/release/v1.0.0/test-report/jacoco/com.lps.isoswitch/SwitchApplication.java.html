package com.lps.isoswitch;

import java.util.concurrent.ExecutorService;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.beans.factory.annotation.Qualifier;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.boot.CommandLineRunner;
import org.springframework.boot.SpringApplication;
import org.springframework.boot.autoconfigure.SpringBootApplication;
import org.springframework.context.annotation.Bean;
import org.springframework.context.ApplicationListener;
import org.springframework.context.event.ContextClosedEvent;
import org.springframework.scheduling.annotation.EnableScheduling;

import com.lps.isoswitch.config.SchemeConfig;
import com.lps.isoswitch.helper.SchemeConnHelpr;
import com.lps.isoswitch.service.SchemeRegistryService;
import com.lps.isoswitch.tcpconnector.SchemesConnector;

import jakarta.annotation.PreDestroy;

/**
 * Main application class for the ISO8583 Switch.
 * This class handles both server and client modes of operation.
 * Uses virtual threads for improved scalability and resource utilization.
 * 
 * @author Logicpoint Systems
 * @version 2.0
 * @since 2025-06-15
 *
 * Coding Standards:
 * 1. All constants should be UPPER_CASE
 * 2. All instance variables should be private
 * 3. All public methods should have JavaDoc
 * 4. Use meaningful variable names
 * 5. Keep methods focused and small
 * 6. Handle exceptions appropriately
 * 7. Use proper logging levels
 * 8. Use virtual threads for I/O operations
 */
@SpringBootApplication
@EnableScheduling
public class SwitchApplication implements ApplicationListener<ContextClosedEvent> {
    private static final Logger log = LoggerFactory.getLogger(SwitchApplication.class);
    
    /** Application mode (server/client) */
    @Value("${app.mode:client}")
    private String appMode;

    /** Connection parameters */
    @Value("${tcp.client.host:}")
    private String host;
    @Value("${tcp.client.port:#{null}}")
    private Integer port;
    @Value("${tcp.client.connectionId:}")
    private String connectionId;
  
    /** Scheme connection helper */
    @Autowired
    private SchemeConnHelpr schemeConnHelper;

    @Autowired
    private SchemeRegistryService schemeRegistryService;

    @Autowired
    @Qualifier("tcpConnectionExecutor")
    private ExecutorService tcpConnectionExecutor;

    /**
     * Main application entry point
     * @param args Command line arguments
     */
    public static void main(String[] args) {
        SpringApplication.run(SwitchApplication.class, args);
    }

    /**
     * Handles Spring Boot shutdown event.
     *
     * @param event The context closed event
     */
    @Override
    public void onApplicationEvent(ContextClosedEvent event) {
        log.info("SwitchApplication received shutdown event");
        // The ShutdownCoordinator will handle the actual shutdown
    }

    /**
     * PreDestroy method for additional cleanup.
     */
    @PreDestroy
    public void onPreDestroy() {
        log.info("SwitchApplication PreDestroy called");
    }

    /**
     * Command line runner bean that initializes the application based on mode
     * @return CommandLineRunner instance
     */
    @Bean
    public CommandLineRunner commandLineRunner() {
        return args -> {
            log.info("Starting application in {} mode", appMode);
            
            if ("server".equalsIgnoreCase(appMode)) {
                startServerMode();
            } else if ("client".equalsIgnoreCase(appMode)) {
                startClientMode();
            } else {
                handleInvalidMode();
            }
        };
    }

    /**
     * Starts the application in server mode
     */
    private void startServerMode() {
        log.info("TCP Server started successfully");
    }

    /**
     * Starts the application in client mode
     */
    private void startClientMode() {
        log.info("Message processing started in client mode using virtual threads");
        StratClient();
    }

    /**
     * Handles invalid application mode
     */
    private void handleInvalidMode() {
        log.error("Invalid application mode: {}. Must be either 'server' or 'client'", appMode);
        System.exit(1);
    }
  
    /**
     * Starts the client using injected virtual thread executor.
     */
    public void StratClient() {
        SchemeConfig.ResolvedSchemeSettings settings = schemeRegistryService.getActiveSchemeSettings(host, port, connectionId);
        SchemesConnector connector = schemeRegistryService.getActiveConnector();
        connector.setSchemeConnHelper(schemeConnHelper);

        // Validate connection details
        if (!validateConnectionDetails(settings.host(), settings.port())) {
            return;
        }

        // Start connection using injected virtual thread executor
        startConnection(tcpConnectionExecutor, connector, settings);
    }

    /**
     * Validates connection details
     * @return true if valid, false otherwise
     */
    private boolean validateConnectionDetails(String resolvedHost, Integer resolvedPort) {
        if (resolvedHost == null || resolvedPort == null) {
            log.error("Connection details not properly configured. Host: {}, Port: {}", resolvedHost, resolvedPort);
            return false;
        }
        return true;
    }

    /**
     * Starts the connection using virtual threads for improved scalability.
     * @param executorService Virtual thread executor service
     * @param connector scheme connector bean
     */
    private void startConnection(
            ExecutorService executorService,
            SchemesConnector connector,
            SchemeConfig.ResolvedSchemeSettings settings) {
        log.info("Starting {} client connection to {}:{} using virtual threads",
                settings.scheme(), settings.host(), settings.port());
        executorService.submit(() -> {
            try {   
                connector.connect(settings.host(), settings.port(), settings.connectionId());
            } catch (InterruptedException e) {
                log.error("Connection interrupted", e);
                Thread.currentThread().interrupt();
            } catch (Exception e) {
                log.error("Failed to start client connection", e);
            }
        });
    }
}
