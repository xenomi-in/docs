package com.lps.isoswitch.jpos;

import org.jpos.iso.ISOComponent;
import org.jpos.iso.ISOException;
import org.jpos.iso.ISOFieldPackager;
import org.jpos.iso.ISOUtil;
import com.lps.isoswitch.config.RawDataLogging;

/*
 * Author: Logic point systems
 * Date: 15-07-2025
 * Version: 1.0.0
 * Description: This class is used to read the chip data into a byte array and unpack it from a byte array.
 * currently not used, future use.
 */
public class ChipDataPackager extends ISOFieldPackager {
    @Override
    public int getMaxPackedLength() {
        return 999; // or set as per RuPay spec
    }

    @Override
    public byte[] pack(ISOComponent c) throws ISOException {
        String data = (String) c.getValue();
        return ISOUtil.hex2byte(data); // Convert HEX string to byte[]
    }

    @Override
    public int unpack(ISOComponent c, byte[] b, int offset) throws ISOException {
        int len = b.length - offset;
        String hexData = ISOUtil.hexString(b, offset, len);
        c.setValue(hexData); // Store as HEX String
        RawDataLogging.object("Unpacked chip data length=" + len, hexData);
        return len;
    }
}
