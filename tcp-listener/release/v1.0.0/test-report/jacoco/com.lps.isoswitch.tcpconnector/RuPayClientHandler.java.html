/**
 * RuPayClientHandler is responsible for handling TCP connections and processing
 * ISO8583 messages for RuPay transactions. It implements concurrent message
 * processing with proper error handling and response management using virtual
 * threads for improved scalability.
 *
 * Key features: - Concurrent message processing using virtual threads -
 * Automatic reconnection handling - ISO8583 message parsing and response
 * generation - Error handling with appropriate decline responses - Network
 * message (0800) handling - Graceful shutdown support
 *
 * @author Logicpoint Systems
 * @version 2.0
 * @since 2025-06-15
 */
@Component
@ChannelHandler.Sharable
public class RuPayClientHandler extends ChannelInboundHandlerAdapter implements SchemeManagementHandler {

    private static final Logger log = LoggerFactory.getLogger(RuPayClientHandler.class);
    private final SwitchDestinationEnum destination;
    private final ISOPackager packager;
    private volatile ChannelHandlerContext currentContext;

    // Virtual thread executor for handling concurrent requests
    private final ExecutorService requestExecutor;
    @Value("${max.concurrent.requests:1000}")
    private int MAX_CONCURRENT_REQUESTS; // Increased for virtual threads
    @Value("${request.timeout.seconds:30}")
    private int REQUEST_TIMEOUT_SECONDS;

    // Counter for active requests
    private final AtomicInteger activeRequests = new AtomicInteger(0);

    // Response codes for different scenarios
    private static final String RESPONSE_CODE_SYSTEM_ERROR = "96";  // System malfunction
    private static final String RESPONSE_CODE_TIMEOUT = "91";       // Issuer or switch inoperative
    private static final String RESPONSE_CODE_DECLINE = "05";       // Do not honor
    private static final String RESPONSE_CODE_INVALID_TRANSACTION = "12"; // Invalid transaction

    private final ApiService apiService;
    private final IsoResponseMapper responseMapper;
    private final SchemeNetworkMessageProcessor networkMessageProcessor;
    private final ConnectionStatusService connectionStatusService;

    // Shutdown related fields
    private final AtomicBoolean isShuttingDown = new AtomicBoolean(false);
    private final AtomicReference<CompletableFuture<Void>> shutdownFuture = new AtomicReference<>();
    private static final int SHUTDOWN_TIMEOUT_SECONDS = 30;
    private static final int SHUTDOWN_CHECK_INTERVAL_MS = 100;

    /**
     * Constructs a new RuPayClientHandler. Initializes virtual thread executor
     * for concurrent request processing.
     *
     * @param apiService The API service for processing ISO messages
     * @param isoMessageExecutor Virtual thread executor for ISO message
     * processing
     */
    public RuPayClientHandler(
            ApiService apiService,
            IsoResponseMapper responseMapper,
            SchemeNetworkMessageProcessor networkMessageProcessor,
            ConnectionStatusService connectionStatusService,
            SchemeConfig schemeConfig,
            @Qualifier("isoMessageExecutor") ExecutorService isoMessageExecutor) {
        this.apiService = apiService;
        this.responseMapper = responseMapper;
        this.networkMessageProcessor = networkMessageProcessor;
        this.connectionStatusService = connectionStatusService;
        this.destination = SwitchDestinationEnum.RUPAY;
        this.requestExecutor = isoMessageExecutor;
        try {
            String packagerPath = schemeConfig.resolve(
                    destination,
                    null,
                    null,
                    null,
                    null,
                    null,
                    null,
                    null,
                    null,
                    null).packager();
            InputStream is = getClass().getResourceAsStream(packagerPath);
            if (is == null) {
                log.error("Could not find {} in classpath", packagerPath);
                throw new ISOException("Packager configuration file not found");
            }
            packager = new GenericPackager(is);
            is.close();
        } catch (ISOException e) {
            log.error("Error initializing ISO8583 packager", e);
            throw new RuntimeException("Failed to initialize ISO8583 packager", e);
        } catch (Exception e) {
            log.error("Error loading packager configuration", e);
            throw new RuntimeException("Failed to load packager configuration", e);
        }
    }

    @Override
    public SwitchDestinationEnum getSupportedScheme() {
        return destination;
    }

    /**
     * Called when a connection is established with the server. Resets the
     * reconnection flag and logs the connection status.
     *
     * @param ctx The channel handler context
     */
    @Override
    public void channelActive(ChannelHandlerContext ctx) {
        this.currentContext = ctx;
        connectionStatusService.updateConnectionStatus(destination, true);
        log.info("Connection established with server");
    }

    /**
     * Called when the connection with the server is lost. Initiates
     * reconnection process if not already reconnecting.
     *
     * @param ctx The channel handler context
     */
    @Override
    public void channelInactive(ChannelHandlerContext ctx) {
        log.info("Connection lost with server");
        connectionStatusService.updateConnectionStatus(destination, false);

        if (isShuttingDown.get()) {
            shutdownThreadPool();
        }
    }

    /**
     * Processes incoming messages from the TCP connection. Handles concurrent
     * message processing using virtual threads for improved scalability.
     *
     * @param ctx The channel handler context
     * @param msg The received message
     */
    @Override
    public void channelRead(ChannelHandlerContext ctx, Object msg) {
        // Check if shutdown is in progress
        if (isShuttingDown.get()) {
            log.warn("Rejecting new request as shutdown is in progress");
            return;
        }

        if (activeRequests.get() >= MAX_CONCURRENT_REQUESTS) {
            log.warn("Maximum concurrent requests limit reached. Rejecting new request.");
            return;
        }

        ByteBuf buf = Unpooled.wrappedBuffer((byte[]) msg);
        byte[] data = new byte[buf.readableBytes()];
        buf.getBytes(0, data);
        RawDataLogging.bytes("Inbound RuPay frame", data);
        try {
            activeRequests.incrementAndGet();
            log.info("Received message in RuPayClientHandler. Active requests: {}", activeRequests.get());

            // Process message using virtual threads for improved scalability
            CompletableFuture.runAsync(() -> processMessage(ctx, data), requestExecutor)
                    .orTimeout(REQUEST_TIMEOUT_SECONDS, TimeUnit.SECONDS)
                    .whenComplete((result, error) -> {
                        if (error != null) {
                            log.error("Error processing message", error);
                            if (error instanceof TimeoutException) {
                                log.error("Request timed out after {} seconds", REQUEST_TIMEOUT_SECONDS);
                            }
                        }
                        activeRequests.decrementAndGet();
                    });
        } catch (Exception e) {
            log.error("Error in channelRead", e);
            activeRequests.decrementAndGet();
        } finally {
            buf.release();
        }
    }

    /**
     * Processes an ISO8583 message, handling both financial and network
     * messages. For financial messages, it calls the API service; for network
     * messages (0800), it handles them directly.
     *
     * @param ctx The channel handler context
     * @param data The raw message data
     */
    private void processMessage(ChannelHandlerContext ctx, byte[] data) {
        ISOMsg isoMsg = new ISOMsg();
        try {
            isoMsg.setPackager(packager);
            isoMsg.unpack(data);
            RawDataLogging.isoMessage("Inbound RuPay ISO message", isoMsg);

            if (networkMessageProcessor.shouldRouteToSwitchCore(isoMsg, destination)) {
                processMessage(ctx, isoMsg);
            } else {
                handleLocalNetworkMessage(ctx, isoMsg);
            }
        } catch (ISOException e) {
            log.error("Error processing ISO8583 message", e);
            RawDataLogging.isoMessage("Invalid inbound RuPay ISO message", isoMsg);
            sendDeclineResponse(ctx, isoMsg, RESPONSE_CODE_INVALID_TRANSACTION,
                    "Invalid ISO message format: " + e.getMessage());
        } catch (Exception e) {
            log.error("Error processing message", e);
            sendDeclineResponse(ctx, isoMsg, RESPONSE_CODE_SYSTEM_ERROR,
                    "System error: " + e.getMessage());
        }
    }

    /**
     * Processes an ISO8583 message, handling both financial and network
     * messages. For financial messages, it calls the API service; for network
     * messages (0800), it handles them directly.
     *
     * @param ctx The channel handler context
     * @param isoMsg The received ISO message
     */
    private void processMessage(ChannelHandlerContext ctx, ISOMsg isoMsg) {
        try {
            if (!connectionStatusService.isReadyForTransactions(destination)) {
                log.error("Not ready for transactions. Connection or sign-on not established.");
                sendDeclineResponse(ctx, isoMsg, RESPONSE_CODE_SYSTEM_ERROR, "Not ready for transactions");
                return;
            }

            log.info("Processing financial message");
            apiService.processIsoMessage(isoMsg, destination.name())
                    .thenAcceptAsync(response -> {
                        try {
                            ISOMsg responseMsg = responseMapper.convertApiResponse(response, packager);
                            byte[] responseBytes = responseMsg.pack();
                            RawDataLogging.bytes("Outbound RuPay response frame", responseBytes);
                            RawDataLogging.isoMessage("Outbound RuPay ISO response", responseMsg);
                            ChannelHandlerContext activeContext = resolveContext(ctx);
                            if (activeContext == null) {
                                throw new IllegalStateException("No active channel available to send financial response");
                            }
                            activeContext.writeAndFlush(frameMessage(activeContext, responseBytes));
                            connectionStatusService.recordTransaction(destination, true);
                            log.info("Financial message response sent successfully");
                        } catch (Exception e) {
                            log.error("Error sending response", e);
                            handleError(ctx, e, isoMsg);
                        }
                    }, requestExecutor)
                    .exceptionally(throwable -> {
                        log.error("Error processing message - API", throwable);
                        handleError(ctx, throwable, isoMsg);
                        return null;
                    });
        } catch (Exception e) {
            log.error("Error processing message", e);
            handleError(ctx, e, isoMsg);
        }
    }

    /**
     * Handles local network messages and writes generated responses to the
     * active channel.
     *
     * @param ctx The channel handler context
     * @param isoMsg The ISO message to process
     */
    private void handleLocalNetworkMessage(ChannelHandlerContext ctx, ISOMsg isoMsg) {
        try {
            log.info("Processing network message in RuPayClientHandler");
            var response = networkMessageProcessor.handleLocalNetworkMessage(isoMsg, destination);
            if (response.isPresent()) {
                ISOMsg responseMsg = response.get();
                byte[] responseData = responseMsg.pack();
                RawDataLogging.isoMessage("Outbound local network ISO response", responseMsg);
                ChannelHandlerContext activeContext = resolveContext(ctx);
                if (activeContext == null) {
                    log.error("No active channel available to send network message response");
                    return;
                }
                activeContext.writeAndFlush(frameMessage(activeContext, responseData))
                        .addListener(future -> {
                            if (!future.isSuccess()) {
                                log.error("Failed to send network message response", future.cause());
                            } else {
                                log.info("Successfully sent network message response");
                            }
                        });
            }
        } catch (ISOException e) {
            log.error("Error processing network message", e);
            try {
                sendDeclineResponse(ctx, isoMsg, RESPONSE_CODE_SYSTEM_ERROR,
                        "Error processing network message: " + e.getMessage());
            } catch (Exception ex) {
                log.error("Error sending decline response for network message", ex);
            }
        }
    }

    /**
     * Sends a decline response when the API service is unavailable or
     * encounters an error. Creates a proper ISO response message with
     * appropriate response code and error details.
     *
     * @param ctx The channel handler context
     * @param originalMsg The original ISO message
     * @param responseCode The response code to send
     * @param errorMessage The error message for logging
     */
    private void sendDeclineResponse(ChannelHandlerContext ctx, ISOMsg originalMsg, String responseCode, String errorMessage) {
        try {
            log.error("Sending decline response. Error: {}", errorMessage);
            ChannelHandlerContext activeContext = resolveContext(ctx);
            if (activeContext == null) {
                log.error("No active channel available to send decline response");
                return;
            }

            ISOMsg responseMsg = responseMapper.createDeclineResponse(originalMsg, packager, responseCode);
            byte[] responseData = responseMsg.pack();
            RawDataLogging.isoMessage("Outbound decline ISO response", responseMsg);
            activeContext.writeAndFlush(frameMessage(activeContext, responseData))
                    .addListener(future -> {
                        if (!future.isSuccess()) {
                            log.error("Failed to send decline response", future.cause());
                        } else {
                            log.info("Successfully sent decline response with code: {}", responseCode);
                        }
                    });
        } catch (ISOException e) {
            log.error("Error creating decline response", e);
        }
    }

    /**
     * Handles exceptions that occur during channel operations. Logs the error
     * and attempts reconnection if appropriate.
     *
     * @param ctx The channel handler context
     * @param cause The exception that occurred
     */
    @Override
    public void exceptionCaught(ChannelHandlerContext ctx, Throwable cause) {
        log.error("Channel exception", cause);
        
        // Handle specific timeout exceptions
        if (cause instanceof ReadTimeoutException) {
            log.error("Read timeout detected - connection may be dead");
            connectionStatusService.updateConnectionStatus(destination, false);
        } else if (cause instanceof WriteTimeoutException) {
            log.error("Write timeout detected - connection may be dead");
            connectionStatusService.updateConnectionStatus(destination, false);
        }
        ctx.close();
    }
    
    /**
     * Handles idle state events for connection monitoring.
     *
     * @param ctx The channel handler context
     * @param evt The idle state event
     */
    @Override
    public void userEventTriggered(ChannelHandlerContext ctx, Object evt) {
        if (evt instanceof IdleStateEvent) {
            IdleStateEvent idleEvent = (IdleStateEvent) evt;
            log.warn("Idle state detected: {}", idleEvent.state());
            
            switch (idleEvent.state()) {
                case READER_IDLE:
                    log.warn("Reader idle - no data received for 60 seconds");
                    // Connection appears to be dead, close it
                    connectionStatusService.updateConnectionStatus(destination, false);
                    ctx.close();
                    break;
                case WRITER_IDLE:
                    log.warn("Writer idle - no data sent for 60 seconds");
                    // Send a ping to check if connection is alive
                    sendConnectionTest(ctx);
                    break;
                case ALL_IDLE:
                    log.warn("All idle - no activity for 60 seconds");
                    // Connection appears to be dead, close it
                    connectionStatusService.updateConnectionStatus(destination, false);
                    ctx.close();
                    break;
            }
        }
    }
    
    /**
     * Sends a connection test message to verify the connection is still alive.
     *
     * @param ctx The channel handler context
     */
    private void sendConnectionTest(ChannelHandlerContext ctx) {
        try {
            if (!ctx.channel().isActive()) {
                log.warn("Cannot send connection test - channel is not active");
                return;
            }
            
            ISOMsg testMsg = networkMessageProcessor.createEchoTestRequest(packager, destination);
            byte[] testData = testMsg.pack();
            
            log.debug("Sending connection test message");
            ctx.writeAndFlush(frameMessage(ctx, testData))
                .addListener(future -> {
                    if (!future.isSuccess()) {
                        log.error("Failed to send connection test", future.cause());
                        connectionStatusService.updateConnectionStatus(destination, false);
                        ctx.close();
                    } else {
                        log.debug("Connection test sent successfully");
                    }
                });
                
        } catch (Exception e) {
            log.error("Error sending connection test", e);
            connectionStatusService.updateConnectionStatus(destination, false);
            ctx.close();
        }
    }

    /**
     * Initiates a graceful shutdown of the handler. This method will: 1. Stop
     * accepting new requests 2. Wait for existing requests to complete 3. Close
     * the connection 4. Shutdown the thread pool
     *
     * @return CompletableFuture that completes when shutdown is finished
     */
    public CompletableFuture<Void> shutdown() {
        if (isShuttingDown.compareAndSet(false, true)) {
            log.info("Initiating graceful shutdown of RuPayClientHandler");

            CompletableFuture<Void> future = new CompletableFuture<>();
            shutdownFuture.set(future);

            // Stop accepting new requests
            log.info("Stopping acceptance of new requests");

            // Wait for active requests to complete
            CompletableFuture.runAsync(() -> {
                try {
                    waitForActiveRequests();

                    // Close the channel if it's still open
                    ChannelHandlerContext activeContext = resolveContext(null);
                    if (activeContext != null && activeContext.channel().isActive()) {
                        log.info("Closing active channel");
                        activeContext.close().sync();
                    }

                    // Shutdown the thread pool
                    shutdownThreadPool();

                    log.info("Graceful shutdown completed successfully");
                    future.complete(null);
                } catch (Exception e) {
                    log.error("Error during graceful shutdown", e);
                    future.completeExceptionally(e);
                }
            });

            return future;
        }

        return shutdownFuture.get();
    }

    /**
     * Waits for all active requests to complete or timeout.
     *
     * @throws InterruptedException if the wait is interrupted
     */
    private void waitForActiveRequests() throws InterruptedException {
        long startTime = System.currentTimeMillis();
        long timeoutTime = startTime + (SHUTDOWN_TIMEOUT_SECONDS * 1000);

        while (activeRequests.get() > 0) {
            if (System.currentTimeMillis() > timeoutTime) {
                log.warn("Timeout waiting for active requests to complete. Active requests: {}",
                        activeRequests.get());
                break;
            }

            log.info("Waiting for {} active requests to complete", activeRequests.get());
            Thread.sleep(SHUTDOWN_CHECK_INTERVAL_MS);
        }
    }

    /**
     * Shuts down the thread pool gracefully.
     */
    private void shutdownThreadPool() {
        log.info("Request executor is managed by Spring and will be closed during application shutdown");
    }

    public boolean isShuttingDown() {
        return isShuttingDown.get();
    }

    /**
     * Handles errors by sending a decline response and updating transaction
     * metrics.
     *
     * @param ctx The channel handler context
     * @param error The error that occurred
     */
    private void handleError(ChannelHandlerContext ctx, Throwable error, ISOMsg isoMsg) {
        try {
            if (error instanceof ApiException) {
                sendDeclineResponse(ctx, isoMsg, "05", "API Error");
            } else if (error instanceof NetworkException) {
                sendDeclineResponse(ctx, isoMsg, "91", "Network Error");
            } else {
                sendDeclineResponse(ctx, isoMsg, "96", "System Error");
            }
            connectionStatusService.recordTransaction(destination, false);
        } catch (Exception e) {
            log.error("Error handling error response", e);
        }
    }

    /**
     * Initiates a sign-on request to the server. This method creates and sends
     * a network management message (0800) with sign-on type.
     *
     * @return CompletableFuture<Boolean> that completes with true if sign-on
     * was successful
     */
    public CompletableFuture<Boolean> initiateSignOn() {
        CompletableFuture<Boolean> signOnFuture = new CompletableFuture<>();

        try {
            ChannelHandlerContext activeContext = resolveContext(null);
            if (activeContext == null) {
                log.error("Cannot initiate sign-on without an active channel");
                signOnFuture.complete(false);
                return signOnFuture;
            }
            ISOMsg signOnMsg = networkMessageProcessor.createSignOnRequest(packager, destination);
            log.info("Initiating sign-on request");
            byte[] requestData = signOnMsg.pack();

            activeContext.writeAndFlush(frameMessage(activeContext, requestData))
                    .addListener(future -> {
                        if (!future.isSuccess()) {
                            log.error("Failed to send sign-on request", future.cause());
                            signOnFuture.complete(false);
                        } else {
                            log.info("Sign-on request sent successfully");
                            // Response will be handled in channelRead
                        }
                    });
        } catch (Exception e) {
            log.error("Error initiating sign-on", e);
            signOnFuture.complete(false);
        }

        return signOnFuture;
    }

    @Override
    public CompletableFuture<Boolean> sendIsoMessage(ISOMsg isoMsg) {
        CompletableFuture<Boolean> sendFuture = new CompletableFuture<>();

        try {
            ChannelHandlerContext activeContext = resolveContext(null);
            if (activeContext == null) {
                log.error("Cannot send ISO message without an active channel");
                sendFuture.complete(false);
                return sendFuture;
            }

            isoMsg.setPackager(packager);
            byte[] requestData = isoMsg.pack();

            activeContext.writeAndFlush(frameMessage(activeContext, requestData))
                    .addListener(future -> {
                        if (!future.isSuccess()) {
                            log.error("Failed to send ISO message", future.cause());
                            connectionStatusService.recordTransaction(destination, false);
                            sendFuture.complete(false);
                        } else {
                            log.info("ISO message sent successfully");
                            connectionStatusService.recordTransaction(destination, true);
                            sendFuture.complete(true);
                        }
                    });
        } catch (Exception e) {
            log.error("Error sending ISO message", e);
            connectionStatusService.recordTransaction(destination, false);
            sendFuture.complete(false);
        }

        return sendFuture;
    }

    private ChannelHandlerContext resolveContext(ChannelHandlerContext preferredContext) {
        if (preferredContext != null && preferredContext.channel().isActive()) {
            return preferredContext;
        }
        if (currentContext != null && currentContext.channel().isActive()) {
            return currentContext;
        }
        return null;
    }

    private ByteBuf frameMessage(ChannelHandlerContext ctx, byte[] payload) {
        ByteBuf framed = ctx.alloc().buffer(AppConstants.RuPay_FIELD_LEGTH_PREPEND + payload.length);
        framed.writeShort(payload.length);
        framed.writeBytes(payload);
        return framed;
    }
