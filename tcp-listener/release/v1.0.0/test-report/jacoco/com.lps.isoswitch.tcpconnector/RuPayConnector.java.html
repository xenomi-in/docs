package com.lps.isoswitch.tcpconnector;

import org.springframework.stereotype.Component;

import com.lps.isoswitch.utils.AppConstants;
import com.lps.isoswitch.config.RawDataLogging;
import com.lps.isoswitch.utils.Enums.StatusEnum;
import com.lps.isoswitch.utils.Enums.SwitchDestinationEnum;

import io.netty.channel.ChannelPipeline;

/**
 * @author Logicpoint Systems
 * @version 1.0
 * @since 2025-06-15
 * Spring-managed connector for the live RuPay TCP connection.
 * This class extends the SchemesConnector class and implements the Connector interface.
 * It is responsible for adding the necessary channel handlers to the pipeline and for sending messages to the RuPay server.
     *
 */
@Component
public class RuPayConnector extends SchemesConnector {

    private final RuPayClientHandler ruPayClientHandler;

    public RuPayConnector(RuPayClientHandler ruPayClientHandler) {
        this.ruPayClientHandler = ruPayClientHandler;
    }

    @Override
    public void addPipelines(ChannelPipeline pipeline) {
        this.clientHandler = ruPayClientHandler;

        pipeline.addLast(createFrameDecoder(
                AppConstants.RuPay_MAX_FRAME_LENGTH,
                AppConstants.RuPay_FIELD_OFFSET,
                AppConstants.RuPay_FIELD_LENGTH,
                AppConstants.RuPay_ADJUSTMENT,
                AppConstants.RuPay_INITIAL_BYTES_STRIP));
        pipeline.addLast(createByteArrayDecoder());
        if (RawDataLogging.isEnabled()) {
            pipeline.addLast(createLoggingHandler());
        }
        pipeline.addLast(this.clientHandler);
    }

    @Override
    public SwitchDestinationEnum getSupportedScheme() {
        return SwitchDestinationEnum.RUPAY;
    }

    @Override
    protected StatusEnum sendMsg(byte[] msg) {
        throw new UnsupportedOperationException("RuPayConnector.sendMsg is not implemented");
    }
}
