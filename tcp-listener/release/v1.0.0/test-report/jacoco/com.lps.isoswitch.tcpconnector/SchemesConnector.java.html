/**
 * Abstract implementation of Connector for scheme connections. This class
 * provides the base implementation for managing TCP connections to different
 * payment schemes. It handles connection establishment, shutdown, and
 * connection status management.
 *
 * @author LogicPoint Systems
 * @version 1.0
 */
@EnableScheduling
public abstract class SchemesConnector implements Connector {

    private static final Logger log = LoggerFactory.getLogger(SchemesConnector.class);
    private static final int MAX_FRAME_LENGTH = 8192;
    private static final int LENGTH_FIELD_OFFSET = 0;
    private static final int LENGTH_FIELD_LENGTH = 2;
    private static final int LENGTH_ADJUSTMENT = 0;
    private static final int INITIAL_BYTES_TO_STRIP = 2;
    
    @Autowired
    protected SchemeConfig schemeConfig;

    @Autowired
    protected TransportSecurityProperties transportSecurity;

    @Autowired
    protected TransportTlsContextFactory tlsContextFactory;

    @PostConstruct
    public void init() {
        SchemeConfig.ResolvedSchemeSettings settings = schemeConfig.resolve(
                getSupportedScheme(),
                host,
                port,
                connectionId,
                connectTimeoutMillis,
                readTimeoutSeconds,
                writeTimeoutSeconds,
                idleTimeoutSeconds,
                keepAliveEnabled,
                reconnectDelayMillis);
        log.info("Initializing SchemesConnector for {} with host={}, port={}, connectionId={}",
                settings.scheme(), settings.host(), settings.port(), settings.connectionId());
    }

    /**
     * Channel group for managing all active connections
     */
    private final ChannelGroup defaultGroup = new DefaultChannelGroup(GlobalEventExecutor.INSTANCE);

    /**
     * Current connection status

    protected StatusEnum connStatus = StatusEnum.INACTIVE;

    /**
     * Event loop group for handling I/O operations
     */
    private EventLoopGroup eventLoopGroup = new NioEventLoopGroup();

    /**

    protected String connectionId;

    @Value("${tcp.client.read.timeout:30}")

    @Value("${tcp.client.write.timeout:30}")

    @Value("${tcp.client.idle.timeout:60}")

    /**
     * {@inheritDoc} Establishes a TCP connection to the specified host and
     * port.
     */
    @Override
    public void connect(String host, Integer port, String connectionId) throws InterruptedException {
        started.set(true);
        if (shuttingDown.get()) {
            log.info("Skipping connect for {} because shutdown is in progress", connectionId);
            return;
        }
        if (!connecting.compareAndSet(false, true)) {
            log.info("Connection attempt already in progress for {}", connectionId);
            return;
        }
        if (host == null || port == null) {
            log.error("Invalid connection parameters: host={}, port={}", host, port);
            connecting.set(false);
            return;
        }

        SchemeConfig.ResolvedSchemeSettings settings = schemeConfig.resolve(
                getSupportedScheme(),
                host,
                port,
                connectionId,
                connectTimeoutMillis,
                readTimeoutSeconds,
                writeTimeoutSeconds,
                idleTimeoutSeconds,
                keepAliveEnabled,
                reconnectDelayMillis);

        int effectiveConnectTimeoutMillis = settings.connectTimeout() != null
                ? settings.connectTimeout()
                : connectTimeoutMillis;
        int effectiveReadTimeoutSeconds = settings.readTimeout() != null
                ? settings.readTimeout()
                : readTimeoutSeconds;
        int effectiveWriteTimeoutSeconds = settings.writeTimeout() != null
                ? settings.writeTimeout()
                : writeTimeoutSeconds;
        int effectiveIdleTimeoutSeconds = settings.idleTimeout() != null
                ? settings.idleTimeout()
                : idleTimeoutSeconds;
        boolean effectiveKeepAliveEnabled = settings.keepAliveEnabled() != null
                ? settings.keepAliveEnabled()
                : keepAliveEnabled;
        int effectiveReconnectDelayMillis = settings.reconnectDelay() != null
                ? settings.reconnectDelay()
                : reconnectDelayMillis;

        this.host = settings.host();
        this.port = settings.port();
        this.connectionId = settings.connectionId();
        this.reconnectDelayMillis = effectiveReconnectDelayMillis;
        
        log.info("Connecting to {}:{}", this.host, this.port);
        
        // Check if already connected
        if (connStatus == StatusEnum.ACTIVE) {
            log.info("Already Connected.");
            connecting.set(false);
            return;
        }

        try {
            final SslContext schemeSslContext = transportSecurity.getScheme().isEnabled()
                    ? tlsContextFactory.createClientContext("Scheme", transportSecurity.getScheme())
                    : null;
            // Create a new Bootstrap instance with enhanced connection detection
            bootstrap = new Bootstrap();
            bootstrap.group(this.eventLoopGroup)
                    .remoteAddress(this.host, this.port)
                    .channel(NioSocketChannel.class)
                    // Configure TCP options for better connection detection
                    .option(ChannelOption.SO_KEEPALIVE, effectiveKeepAliveEnabled)
                    .option(ChannelOption.TCP_NODELAY, true)
                    .option(ChannelOption.CONNECT_TIMEOUT_MILLIS, effectiveConnectTimeoutMillis)
                    .handler(new ChannelInitializer<SocketChannel>() {
                        @Override
                        protected void initChannel(SocketChannel socketChannel) throws Exception {
                            ChannelPipeline pipeline = socketChannel.pipeline();

                            if (schemeSslContext != null) {
                                SSLEngine sslEngine = schemeSslContext.newEngine(
                                        socketChannel.alloc(), SchemesConnector.this.host, SchemesConnector.this.port);
                                SSLParameters sslParameters = sslEngine.getSSLParameters();
                                sslParameters.setEndpointIdentificationAlgorithm("HTTPS");
                                sslEngine.setSSLParameters(sslParameters);
                                pipeline.addFirst("schemeTls", new SslHandler(sslEngine));
                            }
                            
                            // Add timeout handlers for connection detection
                            if (effectiveReadTimeoutSeconds > 0) {
                                pipeline.addLast(new ReadTimeoutHandler(effectiveReadTimeoutSeconds));
                            }
                            if (effectiveWriteTimeoutSeconds > 0) {
                                pipeline.addLast(new WriteTimeoutHandler(effectiveWriteTimeoutSeconds));
                            }
                            if (effectiveIdleTimeoutSeconds > 0) {
                                pipeline.addLast(new IdleStateHandler(
                                        effectiveIdleTimeoutSeconds,
                                        effectiveIdleTimeoutSeconds,
                                        effectiveIdleTimeoutSeconds));
                            }

                            addPipelines(pipeline);
                        }
                    });

            // Establish the connection
            this.future = bootstrap.connect().sync();
            
            defaultGroup.add(future.channel());
            
            // Add a listener to handle connection success or failure
            future.addListener((ChannelFutureListener) channelFuture -> {
                log.info("Connection attempt result: {}", channelFuture.isSuccess());
                if (channelFuture.isSuccess()) {
                    connectionSucess(future, connectionId);
                } else {
                    connectionFailed(future, connectionId);
                }
                connecting.set(false);
            });
        } catch (Exception e) {
            log.error("Connection error: {}", e.getMessage());
            connectionFailed(null, connectionId);
            connecting.set(false);
        }
    }

    /**
     * Adds necessary channel handlers to the pipeline. This method must be
     * implemented by concrete classes to add scheme-specific handlers.
     *
     * @param pipeline The channel pipeline to add handlers to
     */
    @Override
    public void addPipelines(ChannelPipeline pipeline) {
        pipeline.addLast(createFrameDecoder(
                MAX_FRAME_LENGTH,
                LENGTH_FIELD_OFFSET,
                LENGTH_FIELD_LENGTH,
                LENGTH_ADJUSTMENT,
                INITIAL_BYTES_TO_STRIP));
        pipeline.addLast(createByteArrayDecoder());
        if (RawDataLogging.isEnabled()) {
            pipeline.addLast(createLoggingHandler());
        }
        pipeline.addLast(this.clientHandler);
    }

    protected LengthFieldBasedFrameDecoder createFrameDecoder(
            int maxFrameLength,
            int lengthFieldOffset,
            int lengthFieldLength,
            int lengthAdjustment,
            int initialBytesToStrip) {
        return new LengthFieldBasedFrameDecoder(
                maxFrameLength,
                lengthFieldOffset,
                lengthFieldLength,
                lengthAdjustment,
                initialBytesToStrip);
    }

    protected ByteArrayDecoder createByteArrayDecoder() {
        return new ByteArrayDecoder();
    }

    protected LoggingHandler createLoggingHandler() {
        return new LoggingHandler(LogLevel.INFO);
    }

    /**
     * {@inheritDoc} Gracefully shuts down the connection and releases
     * resources.
     */
    @Override
    public void shutdown() throws InterruptedException {
        shuttingDown.set(true);
        closeResources();
        this.connStatus = StatusEnum.INACTIVE;
    }

    private void closeResources() throws InterruptedException {
        try {
            if (future != null && future.channel() != null && future.channel().isOpen()) {
                future.channel().close().sync();
            }
            defaultGroup.close().awaitUninterruptibly();
            this.eventLoopGroup.shutdownGracefully().sync();
        } catch (InterruptedException e) {
            log.info("Exception caught:{}", e);
            throw e;
        }
    }

    /**
     * {@inheritDoc} Handles successful connection establishment.
     */
    @Override
    public void connectionSucess(ChannelFuture future, String connectionId) {

        log.info("Successfully connect.{}", connectionId);

        this.connStatus = StatusEnum.ACTIVE;
        connecting.set(false);

        //schConHlpr.updateStatus(connectionId, this.connStatus);
    }

    /**
     * {@inheritDoc} Handles connection failure and attempts reconnection.
     */
    @Override
    public void connectionFailed(ChannelFuture future, String connectionId) {
        if (shuttingDown.get()) {
            log.info("Skipping reconnect for {} because shutdown is in progress", connectionId);
            return;
        }

        log.error("Failed to connect.{}", connectionId);

        this.connStatus = StatusEnum.INACTIVE;
        connecting.set(false);

        schConHlpr.updateStatus(connectionId, this.connStatus);

        int effectiveReconnectDelayMillis = Math.max(0, reconnectDelayMillis);
        log.info("Trying to re-connect after {} ms.", effectiveReconnectDelayMillis);

        try {
            Thread.sleep(effectiveReconnectDelayMillis);
        } catch (Exception e) {
            log.error("Exception caught:{}", e);
        }

        try {
            this.connect(this.host, this.port, this.connectionId);
        } catch (Exception e) {
            log.error("Exception caught:{}", e);
        }
    }

    /**
     * Reconnects to the specified host and port.
     *
     * @param host The host address to connect to
     * @param port The port number to connect to
     * @param connectionId Unique identifier for this connection
     */
    public void reconnect(String host, Integer port, String connectionId) {
        if (shuttingDown.get()) {
            log.info("Skipping reconnect for {} because shutdown is in progress", connectionId);
            return;
        }
        if (connecting.get()) {
            log.debug("Skipping reconnect for {} because another connect is already in progress", connectionId);
            return;
        }

        try {
            closeResources();
        } catch (Exception e) {
            log.info("Exception caught:{}", e);
        }

        try {
            cleanData();
        } catch (Exception e) {
            log.info("Exception caught:{}", e);
        }

        try {
            this.connect(host, port, connectionId);
        } catch (Exception e) {
            log.info("Exception caught:{}", e);
        }
    }

    /**
     * Cleans up connection-related data. Resets the connection status and
     * creates a new event loop group.
     */
    protected void cleanData() {
        connStatus = StatusEnum.INACTIVE;
        eventLoopGroup = new NioEventLoopGroup();
        future = null;
    }

    /**
     * Attempts to reconnect using the current connection parameters.
     */
    public void reconnect() {
        if (shuttingDown.get()) {
            log.info("Skipping reconnect because shutdown is in progress");
            return;
        }
        if (connecting.get()) {
            log.debug("Skipping reconnect because another connect is already in progress");
            return;
        }

        try {
            closeResources();
        } catch (Exception e) {
            log.info("Exception caught:{}", e);
        }

        try {
            cleanData();
        } catch (Exception e) {
            log.info("Exception caught:{}", e);
        }

        try {
            this.connect(this.host, this.port, this.connectionId);
        } catch (Exception e) {
            log.info("Exception caught:{}", e);
        }
    }

    protected abstract StatusEnum sendMsg(byte[] msg);

    /**
     * Periodically checks the connection status and attempts reconnection if
     * necessary. This method runs at configurable intervals.
     */
    @Scheduled(
            initialDelayString = "${tcp.client.connection.check.initial.delay:30000}",
            fixedDelayString = "${tcp.client.connection.check.interval:30000}")
    public void doCheckConnStatus() {
        try {
            if (shuttingDown.get()) {
                return;
            }
            if (!started.get()) {
                log.debug("Skipping connection status check because connector startup has not begun");
                return;
            }
            if (!"client".equalsIgnoreCase(appMode)) {
                return;
            }
            if (connecting.get()) {
                log.debug("Skipping connection status check because a connection attempt is already in progress");
                return;
            }
            log.debug("Checking for connection status.......");

            // Check if future is null or not successful
            if (this.future == null || !this.future.isSuccess()) {
                log.warn("Connection future is null or not successful - attempting reconnection");
                this.reconnect();
                return;
            }

            // Check if channel is registered and active
            if (!this.future.channel().isRegistered() || !this.future.channel().isActive()) {
                log.warn("Channel is not registered or not active - attempting reconnection");
                this.reconnect();
                return;
            }

            // Check if channel is writable (indicates connection health)
            if (!this.future.channel().isWritable()) {
                log.warn("Channel is not writable - connection may be dead");
                this.reconnect();
                return;
            }

            // Additional socket-level health check
            if (!isSocketConnected()) {
                log.warn("Socket-level check indicates connection is dead - attempting reconnection");
                this.reconnect();
                return;
            }

            log.debug("Connection status check passed - connection appears healthy");

        } catch (Exception e) {
            log.error("Error during connection status check", e);
            // If there's an exception during status check, assume connection is dead
            this.reconnect();
        }
    }

    /**
     * Performs a socket-level connection health check.
     * This method attempts to determine if the underlying socket is still connected
     * by checking various socket properties and attempting a lightweight operation.
     *
     * @return true if the socket appears to be connected, false otherwise
     */
    private boolean isSocketConnected() {
        try {
            if (this.future == null || this.future.channel() == null) {
                return false;
            }

            // Check if channel is open and active
            if (!this.future.channel().isOpen() || !this.future.channel().isActive()) {
                return false;
            }

            // Check if channel is writable
            if (!this.future.channel().isWritable()) {
                return false;
            }

            // Additional check: try to get remote address
            // If this throws an exception, the connection is likely dead
            try {
                this.future.channel().remoteAddress();
            } catch (Exception e) {
                log.debug("Remote address check failed - connection may be dead", e);
                return false;
            }

            return true;

        } catch (Exception e) {
            log.debug("Socket health check failed", e);
            return false;
        }
    }
