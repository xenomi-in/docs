package com.lps.isoswitch.controller;

import com.lps.isoswitch.model.IsoRequest;
import com.lps.isoswitch.service.SwitchCoreOutboundService;
import com.lps.isoswitch.utils.Enums.SwitchDestinationEnum;
import org.jpos.iso.ISOException;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RestController;

import java.util.Map;
import java.util.concurrent.TimeUnit;
import java.util.concurrent.TimeoutException;

/**
 * API used by Switch Core to send ISO8583 requests to the connected scheme.
 */
@RestController
@RequestMapping("/api/v1/switch-core")
public class SwitchCoreOutboundController {

    private static final Logger log = LoggerFactory.getLogger(SwitchCoreOutboundController.class);

    private final SwitchCoreOutboundService outboundService;

    @Value("${switch-core.outbound.ack.timeout.seconds:10}")
    private int ackTimeoutSeconds;

    public SwitchCoreOutboundController(SwitchCoreOutboundService outboundService) {
        this.outboundService = outboundService;
    }

    @PostMapping("/iso8583")
    public ResponseEntity<Map<String, Object>> sendIso8583(@RequestBody IsoRequest request) {
        return sendIso8583(null, request);
    }

    @PostMapping("/{scheme}/iso8583")
    public ResponseEntity<Map<String, Object>> sendIso8583(
            @PathVariable String scheme,
            @RequestBody IsoRequest request) {
        try {
            SwitchDestinationEnum destination = resolveDestination(scheme, request);
            boolean sent = outboundService.sendToScheme(request, destination)
                    .get(ackTimeoutSeconds, TimeUnit.SECONDS);

            if (!sent) {
                return ResponseEntity.status(HttpStatus.SERVICE_UNAVAILABLE).body(Map.of(
                        "status", "error",
                        "message", "Failed to write ISO8583 message to scheme socket",
                        "scheme", destination.name()
                ));
            }

            return ResponseEntity.ok(Map.of(
                    "status", "success",
                    "message", "ISO8583 message written to scheme socket",
                    "scheme", destination.name(),
                    "mti", request.getMti(),
                    "uniqueTransactionId", valueOrEmpty(request.getUniqueTransactionId())
            ));
        } catch (IllegalArgumentException | ISOException e) {
            log.error("Invalid outbound ISO8583 request for scheme {}", scheme, e);
            return ResponseEntity.badRequest().body(Map.of(
                    "status", "error",
                    "message", e.getMessage(),
                    "scheme", scheme
            ));
        } catch (TimeoutException e) {
            log.error("Timed out waiting for outbound ISO8583 write acknowledgement for scheme {}", scheme, e);
            return ResponseEntity.status(HttpStatus.GATEWAY_TIMEOUT).body(Map.of(
                    "status", "error",
                    "message", "Timed out waiting for socket write acknowledgement",
                    "scheme", scheme
            ));
        } catch (Exception e) {
            log.error("Error sending outbound ISO8583 request for scheme {}", scheme, e);
            return ResponseEntity.internalServerError().body(Map.of(
                    "status", "error",
                    "message", "Failed to send ISO8583 message: " + e.getMessage(),
                    "scheme", scheme
            ));
        }
    }

    private SwitchDestinationEnum resolveDestination(String pathScheme, IsoRequest request) {
        String requestScheme = request == null ? null : request.getScheme();
        if (pathScheme != null && !pathScheme.isBlank()) {
            return SwitchDestinationEnum.from(pathScheme);
        }
        return SwitchDestinationEnum.from(requestScheme);
    }

    private String valueOrEmpty(String value) {
        return value == null ? "" : value;
    }
}
