package com.lps.isoswitch.controller;

import com.lps.isoswitch.tcpconnector.SchemeManagementHandler;
import com.lps.isoswitch.service.ConnectionStatusService;
import com.lps.isoswitch.service.SchemeRegistryService;
import com.lps.isoswitch.utils.Enums.SwitchDestinationEnum;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.*;

import java.util.Map;
import java.util.concurrent.CompletableFuture;

/**
 * Controller for managing connection states and sign-on/sign-off operations.
 * Provides REST endpoints for connection management operations.
 *
 * @author Logicpoint Systems
 * @version 1.0
 * @since 2025-06-15
 */
@RestController
@RequestMapping("/api/v1/connection")
public class ConnectionManagementController {

    private static final Logger log = LoggerFactory.getLogger(ConnectionManagementController.class);

    @Autowired
    private SchemeRegistryService schemeRegistryService;

    @Autowired
    private ConnectionStatusService connectionStatusService;

    /**
     * Initiates a sign-on request for the specified scheme.
     *
     * @param scheme The card scheme (e.g., RUPAY)
     * @return ResponseEntity with operation status
     */
    @PostMapping("/{scheme}/sign-on")
    public ResponseEntity<Map<String, Object>> signOn(@PathVariable String scheme) {
        try {
            SwitchDestinationEnum destination = SwitchDestinationEnum.from(scheme);

            // Check if connection is established
            Map<String, Object> metrics = connectionStatusService.getMetrics(destination);
            boolean connectionStatus = (boolean) metrics.get("isConnected");
            if (!connectionStatus) {
                return ResponseEntity.badRequest().body(Map.of(
                        "status", "error",
                        "message", "Cannot initiate sign-on: Connection not established",
                        "scheme", scheme,
                        "connectionStatus", connectionStatus
                ));
            }

            SchemeManagementHandler handler = schemeRegistryService.getManagementHandler(destination);
            CompletableFuture<Boolean> signOnFuture = handler.initiateSignOn();

            return ResponseEntity.ok(Map.of(
                    "status", "success",
                    "message", "Sign-on request initiated",
                    "scheme", scheme,
                    "connectionStatus", connectionStatus
            ));
        } catch (IllegalArgumentException e) {
            log.error("Invalid scheme: {}", scheme);
            return ResponseEntity.badRequest().body(Map.of(
                    "status", "error",
                    "message", "Invalid scheme: " + scheme
            ));
        } catch (Exception e) {
            log.error("Error initiating sign-on", e);
            return ResponseEntity.internalServerError().body(Map.of(
                    "status", "error",
                    "message", "Failed to initiate sign-on: " + e.getMessage()
            ));
        }
    }

    /**
     * Initiates a sign-off request for the specified scheme.
     *
     * @param scheme The card scheme (e.g., RUPAY)
     * @return ResponseEntity with operation status
     */
    @PostMapping("/{scheme}/sign-off")
    public ResponseEntity<Map<String, Object>> signOff(@PathVariable String scheme) {
        try {
            SwitchDestinationEnum destination = SwitchDestinationEnum.from(scheme);

            // Check if connection is established
            Map<String, Object> metrics = connectionStatusService.getMetrics(destination);
            boolean connectionStatus = (boolean) metrics.get("isConnected");

            // Handle null connection status
            if (!connectionStatus) {
                return ResponseEntity.badRequest().body(Map.of(
                        "status", "error",
                        "message", "Cannot initiate sign-off: Connection not established",
                        "scheme", scheme,
                        "connectionStatus", connectionStatus
                ));
            }

            SchemeManagementHandler handler = schemeRegistryService.getManagementHandler(destination);
            CompletableFuture<Boolean> signOffFuture = handler.initiateSignOff();

            return ResponseEntity.ok(Map.of(
                    "status", "success",
                    "message", "Sign-off request initiated",
                    "scheme", scheme,
                    "connectionStatus", connectionStatus
            ));
        } catch (IllegalArgumentException e) {
            log.error("Invalid scheme: {}", scheme);
            return ResponseEntity.badRequest().body(Map.of(
                    "status", "error",
                    "message", "Invalid scheme: " + scheme
            ));
        } catch (Exception e) {
            log.error("Error initiating sign-off", e);
            return ResponseEntity.internalServerError().body(Map.of(
                    "status", "error",
                    "message", "Failed to initiate sign-off: " + e.getMessage()
            ));
        }
    }

    @PostMapping("/{scheme}/dke/{dkeType}")
    public ResponseEntity<Map<String, Object>> dke(@PathVariable String scheme, @PathVariable String dkeType) {
        try {
            SwitchDestinationEnum destination = SwitchDestinationEnum.from(scheme);
            Map<String, Object> metrics = connectionStatusService.getMetrics(destination);
            boolean connectionStatus = (boolean) metrics.get("isConnected");
            if (!connectionStatus) {
                return ResponseEntity.badRequest().body(Map.of(
                        "status", "error",
                        "message", "Cannot initiate DKE " + dkeType + ": Connection not established",
                        "scheme", scheme,
                        "connectionStatus", connectionStatus
                ));
            }
            SchemeManagementHandler handler = schemeRegistryService.getManagementHandler(destination);
            CompletableFuture<Boolean> dkeFuture = handler.initiateDke(dkeType);
            return ResponseEntity.ok(Map.of(
                    "status", "success",
                    "message", "DKE " + dkeType + " request initiated",
                    "scheme", scheme,
                    "connectionStatus", connectionStatus
            ));
        } catch (IllegalArgumentException e) {
            log.error("Invalid scheme: {}", scheme);
            return ResponseEntity.badRequest().body(Map.of(
                    "status", "error",
                    "message", "Invalid scheme: " + scheme
            ));
        } catch (Exception e) {
            log.error("Error initiating DKE " + dkeType, e);
            return ResponseEntity.internalServerError().body(Map.of(
                    "status", "error",
                    "message", "Failed to initiate DKE " + dkeType + ": " + e.getMessage()
            ));
        }
    }

    /**
     * Gets the current connection and sign-on status for the specified scheme.
     *
     * @param scheme The card scheme (e.g., RUPAY)
     * @return ResponseEntity with connection status
     */
    @GetMapping("/{scheme}/status")
    public ResponseEntity<Map<String, Object>> getStatus(@PathVariable String scheme) {
        try {
            SwitchDestinationEnum destination = SwitchDestinationEnum.from(scheme);
            Map<String, Object> metrics = connectionStatusService.getMetrics(destination);

            return ResponseEntity.ok(metrics);
        } catch (IllegalArgumentException e) {
            log.error("Invalid scheme: {}", scheme);
            return ResponseEntity.badRequest().body(Map.of(
                    "status", "error",
                    "message", "Invalid scheme: " + scheme
            ));
        } catch (Exception e) {
            log.error("Error getting connection status", e);
            return ResponseEntity.internalServerError().body(Map.of(
                    "status", "error",
                    "message", "Failed to get connection status: " + e.getMessage()
            ));
        }
    }
}
