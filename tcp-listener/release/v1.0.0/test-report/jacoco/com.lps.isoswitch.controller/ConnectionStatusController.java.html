package com.lps.isoswitch.controller;

import com.lps.isoswitch.service.ConnectionStatusService;
import com.lps.isoswitch.utils.Enums.SwitchDestinationEnum;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.*;

import java.util.Map;

/**
 * Controller for exposing connection status and metrics.
 * This controller provides endpoints to check connection status,
 * sign-on status, and transaction metrics.
 *
 * @author Logicpoint Systems
 * @version 1.0
 * @since 2025-06-15
 */
@RestController
@RequestMapping("/api/v1/connection-status")
public class ConnectionStatusController {
    private static final Logger log = LoggerFactory.getLogger(ConnectionStatusController.class);
    private final ConnectionStatusService connectionStatusService;

    public ConnectionStatusController(ConnectionStatusService connectionStatusService) {
        this.connectionStatusService = connectionStatusService;
    }

    /**
     * Gets the status and metrics for a specific scheme.
     *
     * @param scheme The scheme to get status for
     * @return The status and metrics for the scheme
     */
    @GetMapping("/{scheme}")
    public ResponseEntity<Map<String, Object>> getSchemeStatus(
            @PathVariable("scheme") SwitchDestinationEnum scheme) {
        log.debug("Getting status for scheme: {}", scheme);
        return ResponseEntity.ok(connectionStatusService.getMetrics(scheme));
    }

    /**
     * Gets the status and metrics for all schemes.
     *
     * @return The status and metrics for all schemes
     */
    @GetMapping
    public ResponseEntity<Map<SwitchDestinationEnum, Map<String, Object>>> getAllStatus() {
        log.debug("Getting status for all schemes");
        return ResponseEntity.ok(connectionStatusService.getAllMetrics());
    }

    /**
     * Checks if a scheme is ready for transactions.
     *
     * @param scheme The scheme to check
     * @return true if ready for transactions, false otherwise
     */
    @GetMapping("/{scheme}/ready")
    public ResponseEntity<Boolean> isReadyForTransactions(
            @PathVariable("scheme") SwitchDestinationEnum scheme) {
        log.debug("Checking if scheme is ready for transactions: {}", scheme);
        return ResponseEntity.ok(connectionStatusService.isReadyForTransactions(scheme));
    }
}
