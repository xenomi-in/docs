package com.lps.isoswitch.service;

import java.util.Map;

import org.jpos.iso.ISOException;
import org.jpos.iso.ISOMsg;
import org.jpos.iso.ISOPackager;
import org.springframework.stereotype.Component;

/**
 * Converts outbound responses into jPOS {@link ISOMsg} instances.
 *
 * <p>This component keeps response construction out of the TCP handler. It
 * covers two response paths:
 * API responses returned by Switch Core and local decline responses generated
 * by the listener when parsing, routing, or downstream calls fail.</p>
 */
@Component
public class IsoResponseMapper {

    /**
     * Builds an ISO8583 response from the JSON-like map returned by Switch Core.
     *
     * <p>Expected shape:
     * {@code {"mti": "0210", "fields": {"39": "00", ...}}}. Field keys are
     * converted to ISO field numbers and values are written as strings.</p>
     *
     * @param response Switch Core response payload
     * @param packager scheme-specific packager to attach to the response
     * @return packed-message-ready ISO response
     * @throws ISOException when the response does not contain a valid fields map
     * or cannot be converted to an ISO message
     */
    public ISOMsg convertApiResponse(Map<String, Object> response, ISOPackager packager) throws ISOException {
        ISOMsg isoMsg = new ISOMsg();
        isoMsg.setPackager(packager);

        Object mti = response.get("mti");
        if (mti != null) {
            isoMsg.setMTI(String.valueOf(mti));
        }

        Object fields = response.get("fields");
        if (!(fields instanceof Map<?, ?> responseFields)) {
            throw new ISOException("API response does not contain a valid fields map");
        }

        for (Map.Entry<?, ?> entry : responseFields.entrySet()) {
            int fieldNumber = Integer.parseInt(String.valueOf(entry.getKey()));
            Object fieldValue = entry.getValue();
            if (fieldValue != null) {
                isoMsg.set(fieldNumber, String.valueOf(fieldValue));
            }
        }

        return isoMsg;
    }

    /**
     * Creates a local decline response using key fields from the original
     * request so the scheme can correlate the response with the failed request.
     *
     * @param originalMsg original inbound ISO message
     * @param packager scheme-specific packager to attach to the response
     * @param responseCode ISO field 39 response code
     * @return decline response ready to pack and send over TCP
     * @throws ISOException when the response MTI or fields cannot be set
     */
    public ISOMsg createDeclineResponse(ISOMsg originalMsg, ISOPackager packager, String responseCode) throws ISOException {
        ISOMsg responseMsg = new ISOMsg();
        responseMsg.setPackager(packager);

        String mti = originalMsg.getMTI();
        int mtiNumber = Integer.parseInt(mti) + 10;
        responseMsg.setMTI("0" + mtiNumber);

        copyEssentialFields(originalMsg, responseMsg);
        responseMsg.set(39, responseCode);
        responseMsg.set(44, responseDescription(responseCode));

        return responseMsg;
    }

    /**
     * Copies only correlation and terminal/merchant fields that are safe and
     * useful for error responses.
     */
    private void copyEssentialFields(ISOMsg originalMsg, ISOMsg responseMsg) throws ISOException {
        int[] essentialFields = {2, 3, 4, 7, 11, 12, 13, 32, 37, 41, 42, 49};

        for (int field : essentialFields) {
            if (originalMsg.hasField(field)) {
                responseMsg.set(field, originalMsg.getString(field));
            }
        }
    }

    /**
     * Human-readable text for field 44 on locally generated decline responses.
     */
    private String responseDescription(String responseCode) {
        return switch (responseCode) {
            case "96" -> "SYSTEM ERROR";
            case "91" -> "SERVICE UNAVAILABLE";
            case "05" -> "DECLINED";
            case "12" -> "INVALID TRANSACTION";
            default -> "DECLINED";
        };
    }
}
