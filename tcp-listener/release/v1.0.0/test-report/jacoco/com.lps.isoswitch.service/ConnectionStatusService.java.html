package com.lps.isoswitch.service;

import com.lps.isoswitch.config.SchemeConfig;
import com.lps.isoswitch.utils.Enums.SwitchDestinationEnum;
import io.micrometer.core.instrument.Gauge;
import io.micrometer.core.instrument.MeterRegistry;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Service;

import java.time.LocalDateTime;
import java.util.Map;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.atomic.AtomicBoolean;
import java.util.concurrent.atomic.AtomicInteger;

/**
 * Service for tracking connection and sign-on status of different schemes.
 * This service maintains the state of connections and sign-on status,
 * and provides metrics for monitoring.
 *
 * @author Logicpoint Systems
 * @version 1.0
 * @since 2025-06-15
 */
@Service
public class ConnectionStatusService {
    private static final Logger log = LoggerFactory.getLogger(ConnectionStatusService.class);

    // Connection status tracking
    private final static Map<SwitchDestinationEnum, AtomicBoolean> connectionStatus = new ConcurrentHashMap<>();
    private final static Map<SwitchDestinationEnum, AtomicBoolean> signOnStatus = new ConcurrentHashMap<>();
    private final static Map<SwitchDestinationEnum, LocalDateTime> lastSignOnTime = new ConcurrentHashMap<>();
    private final static Map<SwitchDestinationEnum, LocalDateTime> lastSignOffTime = new ConcurrentHashMap<>();
    
    // Metrics tracking
    private final static Map<SwitchDestinationEnum, AtomicInteger> totalTransactions = new ConcurrentHashMap<>();
    private final static Map<SwitchDestinationEnum, AtomicInteger> successfulTransactions = new ConcurrentHashMap<>();
    private final static Map<SwitchDestinationEnum, AtomicInteger> failedTransactions = new ConcurrentHashMap<>();
    private final static Map<SwitchDestinationEnum, LocalDateTime> lastTransactionTime = new ConcurrentHashMap<>();

    public ConnectionStatusService(
            MeterRegistry meterRegistry,
            SchemeConfig schemeConfig,
            @Value("${app.site-id:unknown}") String siteId,
            @Value("${app.listener-id:unknown}") String listenerId) {
        SwitchDestinationEnum scheme = schemeConfig.getActiveScheme();
        SchemeConfig.BaseConfig schemeSettings = schemeConfig.getConfig(scheme);
        String connectionId = schemeSettings == null || isBlank(schemeSettings.getConnectionId())
                ? scheme.name()
                : schemeSettings.getConnectionId();
        String endpointHost = schemeSettings == null || isBlank(schemeSettings.getHost())
                ? "unknown"
                : schemeSettings.getHost();
        String endpointPort = schemeSettings == null || schemeSettings.getPort() == null
                ? "unknown"
                : schemeSettings.getPort().toString();

        AtomicBoolean connected = connectionStatus.computeIfAbsent(
                scheme, ignored -> new AtomicBoolean(false));
        AtomicBoolean signedOn = signOnStatus.computeIfAbsent(
                scheme, ignored -> new AtomicBoolean(false));

        Gauge.builder("switch.endpoint.tcp.connected", connected,
                        status -> status.get() ? 1.0 : 0.0)
                .description("TCP connection status: 1=connected, 0=disconnected")
                .tag("site_id", siteId)
                .tag("listener_id", listenerId)
                .tag("connection_id", connectionId)
                .tag("scheme", scheme.name())
                .tag("endpoint_host", endpointHost)
                .tag("endpoint_port", endpointPort)
                .register(meterRegistry);

        Gauge.builder("switch.endpoint.signed.on", signedOn,
                        status -> status.get() ? 1.0 : 0.0)
                .description("Scheme sign-on status: 1=signed on, 0=signed off")
                .tag("site_id", siteId)
                .tag("listener_id", listenerId)
                .tag("connection_id", connectionId)
                .tag("scheme", scheme.name())
                .tag("endpoint_host", endpointHost)
                .tag("endpoint_port", endpointPort)
                .register(meterRegistry);

        Gauge.builder("switch.endpoint.ready", this,
                        ignored -> isReadyForTransactions(scheme) ? 1.0 : 0.0)
                .description("Endpoint transaction readiness: 1=ready, 0=not ready")
                .tag("site_id", siteId)
                .tag("listener_id", listenerId)
                .tag("connection_id", connectionId)
                .tag("scheme", scheme.name())
                .tag("endpoint_host", endpointHost)
                .tag("endpoint_port", endpointPort)
                .register(meterRegistry);
    }

    /**
     * Updates the connection status for a scheme.
     *
     * @param scheme The scheme to update
     * @param isConnected The new connection status
     */
    public void updateConnectionStatus(SwitchDestinationEnum scheme, boolean isConnected) {
        connectionStatus.computeIfAbsent(scheme, k -> new AtomicBoolean(false))
                       .set(isConnected);
        if (!isConnected) {
            signOnStatus.computeIfAbsent(scheme, k -> new AtomicBoolean(false))
                    .set(false);
        }
        log.info("Connection status updated for {}: {}", scheme, isConnected);
    }

    /**
     * Updates the sign-on status for a scheme.
     *
     * @param scheme The scheme to update
     * @param isSignedOn The new sign-on status
     */
    public void updateSignOnStatus(SwitchDestinationEnum scheme, boolean isSignedOn) {
        signOnStatus.computeIfAbsent(scheme, k -> new AtomicBoolean(false))
                   .set(isSignedOn);
        if (isSignedOn) {
            lastSignOnTime.put(scheme, LocalDateTime.now());
        } else {
            lastSignOffTime.put(scheme, LocalDateTime.now());
        }
        log.info("Sign-on status updated for {}: {}", scheme, isSignedOn);
    }

    /**
     * Checks if a scheme is both connected and signed on.
     *
     * @param scheme The scheme to check
     * @return true if both connected and signed on, false otherwise
     */
    public boolean isReadyForTransactions(SwitchDestinationEnum scheme) {
        boolean isConnected = connectionStatus.getOrDefault(scheme, new AtomicBoolean(false)).get();
        boolean isSignedOn = signOnStatus.getOrDefault(scheme, new AtomicBoolean(false)).get();
        return isConnected && isSignedOn;
    }

    /**
     * Records a transaction attempt and its result.
     *
     * @param scheme The scheme where the transaction occurred
     * @param isSuccessful Whether the transaction was successful
     */
    public void recordTransaction(SwitchDestinationEnum scheme, boolean isSuccessful) {
        totalTransactions.computeIfAbsent(scheme, k -> new AtomicInteger(0)).incrementAndGet();
        if (isSuccessful) {
            successfulTransactions.computeIfAbsent(scheme, k -> new AtomicInteger(0)).incrementAndGet();
        } else {
            failedTransactions.computeIfAbsent(scheme, k -> new AtomicInteger(0)).incrementAndGet();
        }
        lastTransactionTime.put(scheme, LocalDateTime.now());
    }

    /**
     * Gets the current metrics for a scheme.
     *
     * @param scheme The scheme to get metrics for
     * @return A map containing the metrics
     */
    public Map<String, Object> getMetrics(SwitchDestinationEnum scheme) {
        Map<String, Object> metrics = new ConcurrentHashMap<>();
        metrics.put("isConnected", connectionStatus.getOrDefault(scheme, new AtomicBoolean(false)).get());
        metrics.put("isSignedOn", signOnStatus.getOrDefault(scheme, new AtomicBoolean(false)).get());
        //metrics.put("lastSignOnTime", lastSignOnTime.get(scheme));
       // metrics.put("lastSignOffTime", lastSignOffTime.get(scheme));
        metrics.put("totalTransactions", totalTransactions.getOrDefault(scheme, new AtomicInteger(0)).get());
        metrics.put("successfulTransactions", successfulTransactions.getOrDefault(scheme, new AtomicInteger(0)).get());
        metrics.put("failedTransactions", failedTransactions.getOrDefault(scheme, new AtomicInteger(0)).get());
       // metrics.put("lastTransactionTime", lastTransactionTime.get(scheme));
        return metrics;
    }

    /**
     * Gets the current metrics for all schemes.
     *
     * @return A map containing metrics for all schemes
     */
    public Map<SwitchDestinationEnum, Map<String, Object>> getAllMetrics() {
        Map<SwitchDestinationEnum, Map<String, Object>> allMetrics = new ConcurrentHashMap<>();
        for (SwitchDestinationEnum scheme : SwitchDestinationEnum.values()) {
            allMetrics.put(scheme, getMetrics(scheme));
        }
        return allMetrics;
    }

    private boolean isBlank(String value) {
        return value == null || value.isBlank();
    }
}
