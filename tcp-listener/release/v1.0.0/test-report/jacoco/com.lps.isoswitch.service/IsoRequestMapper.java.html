package com.lps.isoswitch.service;

import com.lps.isoswitch.model.IsoRequest;
import org.jpos.iso.ISOException;
import org.jpos.iso.ISOMsg;
import org.jpos.iso.ISOPackager;
import org.springframework.stereotype.Component;

import java.util.LinkedHashMap;
import java.util.Map;

/**
 * Converts Switch Core API payloads into jPOS ISO8583 messages.
 */
@Component
public class IsoRequestMapper {

    public ISOMsg convertApiRequest(IsoRequest request, ISOPackager packager) throws ISOException {
        if (request == null) {
            throw new ISOException("Request body is required");
        }
        if (request.getMti() == null || request.getMti().isBlank()) {
            throw new ISOException("MTI is required");
        }

        ISOMsg isoMsg = new ISOMsg();
        isoMsg.setPackager(packager);
        isoMsg.setMTI(request.getMti());

        Map<String, String> fields = new LinkedHashMap<>();
        if (request.getFields() != null) {
            fields.putAll(request.getFields());
        }
        putIfAbsent(fields, "2", request.getPan());
        putIfAbsent(fields, "3", request.getProcessingCode());
        putIfAbsent(fields, "4", request.getAmount());
        putIfAbsent(fields, "7", request.getTransmissionDateTime());
        putIfAbsent(fields, "11", request.getStan());
        putIfAbsent(fields, "12", request.getLocalTransactionTime());
        putIfAbsent(fields, "13", request.getLocalTransactionDate());
        putIfAbsent(fields, "18", request.getMerchantType());
        putIfAbsent(fields, "22", request.getPosEntryMode());
        putIfAbsent(fields, "25", request.getPosConditionCode());
        putIfAbsent(fields, "32", request.getAcquiringInstitutionId());
        putIfAbsent(fields, "33", request.getForwardingInstitutionId());
        putIfAbsent(fields, "37", request.getRrn());
        putIfAbsent(fields, "38", request.getAuthorizationId());
        putIfAbsent(fields, "39", request.getResponseCode());
        putIfAbsent(fields, "42", request.getCardAcceptorId());
        putIfAbsent(fields, "43", request.getCardAcceptorName());
        putIfAbsent(fields, "48", request.getAdditionalData());
        putIfAbsent(fields, "49", request.getCurrencyCode());
        putIfAbsent(fields, "90", request.getOriginalData());
        putIfAbsent(fields, "95", request.getReplacementAmount());
        putIfAbsent(fields, "100", request.getReceivingInstitutionId());
        putIfAbsent(fields, "102", request.getAccountId1());
        putIfAbsent(fields, "103", request.getAccountId2());

        for (Map.Entry<String, String> entry : fields.entrySet()) {
            String value = entry.getValue();
            if (value == null) {
                continue;
            }
            int fieldNumber = parseFieldNumber(entry.getKey());
            isoMsg.set(fieldNumber, value);
        }

        return isoMsg;
    }

    private void putIfAbsent(Map<String, String> fields, String fieldNumber, String value) {
        if (value != null && !value.isBlank()) {
            fields.putIfAbsent(fieldNumber, value);
        }
    }

    private int parseFieldNumber(String fieldNumber) throws ISOException {
        try {
            int parsed = Integer.parseInt(fieldNumber);
            if (parsed < 1 || parsed > 128) {
                throw new ISOException("ISO field number out of range: " + fieldNumber);
            }
            return parsed;
        } catch (NumberFormatException e) {
            throw new ISOException("Invalid ISO field number: " + fieldNumber, e);
        }
    }
}
