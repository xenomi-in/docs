package com.lps.isoswitch.service;

import org.springframework.beans.factory.annotation.Value;
import org.springframework.http.MediaType;
import org.springframework.stereotype.Service;
import org.springframework.web.reactive.function.client.WebClient;
import org.springframework.web.reactive.function.client.WebClientResponseException;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import tools.jackson.databind.ObjectMapper;
import org.jpos.iso.ISOMsg;
import org.jpos.iso.ISOException;
import com.lps.isoswitch.config.SchemeConfig;
import com.lps.isoswitch.config.RawDataLogging;
import com.lps.isoswitch.utils.Enums.SwitchDestinationEnum;

import java.util.HashMap;
import java.util.Map;
import java.util.concurrent.CompletableFuture;
import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;
import java.util.concurrent.TimeUnit;
import java.time.Duration;
import tools.jackson.databind.node.ObjectNode;
import reactor.core.scheduler.Schedulers;
import reactor.util.retry.Retry;

/**
 * Service class for handling API communications with external services.
 * This service manages all HTTP communications with external APIs,
 * including request/response handling, error management, and retry logic.
 *
 * @author Logicpoint Systems
 * @version 1.0
 * @since 2025-06-15
 */
@Service
public final class ApiService {
    private static final Logger log = LoggerFactory.getLogger(ApiService.class);
    private static final int API_TIMEOUT_SECONDS = 30;
    
    private final WebClient webClient;
    private final ObjectMapper objectMapper;
    private final ExecutorService executorService;
    private final SchemeConfig schemeConfig;
    
    @Value("${api.service.url}")
    private String apiServiceUrl;
    
    @Value("${api.service.url.dke}")
    private String apiServiceUrlDke;

    @Value("${app.site-id:}")
    private String siteId;

    @Value("${app.listener-id:switchTcpListener}")
    private String listenerId;
    
    /**
     * Constructs a new ApiService with the required dependencies.
     *
     * @param webClient Configured WebClient with connection pooling
     * @param objectMapper ObjectMapper for JSON processing
     */
    public ApiService(WebClient webClient, ObjectMapper objectMapper, SchemeConfig schemeConfig) {
        this.webClient = webClient;
        this.objectMapper = objectMapper;
        this.schemeConfig = schemeConfig;
        this.executorService = Executors.newVirtualThreadPerTaskExecutor();
    }
    
    /**
     * Sends ISO8583 message to API service for processing.
     * Uses CompletableFuture for asynchronous operation and proper timeout handling.
     *
     * @param isoMsg The ISO8583 message to process
     * @param scheme The scheme name (e.g., "RuPay", "Visa")
     * @return CompletableFuture containing the API response
     * @throws ISOException if there's an error processing the ISO message
     */
    public CompletableFuture<Map<String, Object>> processIsoMessage(ISOMsg isoMsg, String scheme) {
        return CompletableFuture.supplyAsync(() -> {
            try {
                boolean dkeRequest = isDkeRequest(isoMsg);
                String resolvedConnectionId = resolveConnectionId(scheme);
                // Convert ISOMsg to JSON request
                ObjectNode jsonRequest = objectMapper.createObjectNode();
                jsonRequest.put("scheme", scheme);
                jsonRequest.put("mti", isoMsg.getMTI());
                String uniqueTxnId = isoMsg.getMTI()+isoMsg.getString(7)+isoMsg.getString(37);
                jsonRequest.put("uniqueTransactionId", uniqueTxnId);
                
                ObjectNode fields = objectMapper.createObjectNode();
                // Add all ISO fields to JSON
                for (int i = 1; i <= 128; i++) {
                    if (isoMsg.hasField(i)) {
                        fields.put(String.valueOf(i), isoMsg.getString(i));
                    }
                }
                jsonRequest.set("fields", fields);
                RawDataLogging.object("ISO request JSON for " + scheme, jsonRequest);
                //Map<String, Object> requestMap = convertIsoToMap(isoMsg, scheme);
                //log.info("Sending request to API service for scheme {}: {}", scheme, requestMap);
                String apiUrl = apiServiceUrl;
                if (dkeRequest) {
                    apiUrl = apiServiceUrlDke+"/"+scheme;
                }
                
                return webClient.post()
                    .uri(apiUrl)
                    .contentType(MediaType.APPLICATION_JSON)
                    .header("site_id", siteId)
                    .header("listener_id", listenerId)
                    .header("connection_id", resolvedConnectionId)
                    .bodyValue(jsonRequest)
                    .retrieve()
                    .bodyToMono(Map.class)
                    //.retryWhen(Retry.backoff(3, Duration.ofSeconds(1))
                        //.filter(throwable -> !(throwable instanceof IllegalArgumentException)))
                    .timeout(Duration.ofSeconds(API_TIMEOUT_SECONDS))
                    .doOnSuccess(response -> log.info("Received response from API for scheme {}: {}", scheme, response.get(39)))
                    .doOnError(error -> log.error("Error calling API service for scheme {}: {}", scheme, error.getMessage()))
                    .subscribeOn(Schedulers.boundedElastic())
                    .block();

                
            } catch (WebClientResponseException e) {
                log.error("Error processing ISO message for API with scheme {}. Status={}",
                        scheme,
                        e.getStatusCode(),
                        e);
                throw new RuntimeException("Failed to process ISO message", e);
            } catch (Exception e) {
                log.error("Error processing ISO message for API with scheme {}", scheme, e);
                throw new RuntimeException("Failed to process ISO message", e);
            }
        }, executorService);
    }
    
    /**
     * Converts ISOMsg to a Map representation.
     * Uses Java 17 features for better readability and maintainability.
     *
     * @param isoMsg The ISO8583 message to convert
     * @param scheme The scheme name
     * @return Map containing the ISO message fields
     * @throws ISOException if there's an error accessing ISO message fields
     */
    private Map<String, Object> convertIsoToMap(ISOMsg isoMsg, String scheme) throws ISOException {
        var result = new HashMap<String, Object>();
        
        // Add MTI and scheme
        result.put("mti", isoMsg.getMTI());
        result.put("scheme", scheme);
        
        // Add all fields using enhanced for loop
        for (int i = 1; i <= 128; i++) {
            if (isoMsg.hasField(i)) {
                result.put(String.valueOf(i), isoMsg.getString(i));
            }
        }
        
        return result;
    }

    private String resolveConnectionId(String scheme) {
        try {
            return schemeConfig.resolve(
                    SwitchDestinationEnum.from(scheme),
                    null,
                    null,
                    null,
                    null,
                    null,
                    null,
                    null,
                    null,
                    null).connectionId();
        } catch (Exception ex) {
            log.warn("Falling back to scheme name for connection_id because resolution failed for {}", scheme, ex);
            return scheme;
        }
    }

    private boolean isDkeRequest(ISOMsg isoMsg) throws ISOException {
        return "0800".equals(isoMsg.getMTI())
                && ("184".equals(isoMsg.getString(70)) || "185".equals(isoMsg.getString(70)));
    }

    /**
     * Shuts down the executor service when the application stops.
     * This method should be called during application shutdown.
     */
    public void shutdown() {
        try {
            log.info("Shutting down ApiService executor service");
            executorService.shutdown();
            if (!executorService.awaitTermination(60, TimeUnit.SECONDS)) {
                executorService.shutdownNow();
            }
        } catch (InterruptedException e) {
            log.error("Error shutting down executor service", e);
            Thread.currentThread().interrupt();
            executorService.shutdownNow();
        }
    }

    /**
     * Processes a transaction message.
     *
     * @param requestMsg The ISO message to process
     * @return The response ISO message
     * @throws ISOException if there's an error processing the message
     */
    public ISOMsg processTransaction(ISOMsg requestMsg) throws ISOException {
        log.info("Processing transaction message");
        // Add transaction processing logic here
        ISOMsg responseMsg = new ISOMsg();
        responseMsg.setMTI("0210");
        responseMsg.set(39, "00");
        return responseMsg;
    }
}
