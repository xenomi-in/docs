package com.lps.isoswitch.service;

import java.util.EnumMap;
import java.util.List;
import java.util.Map;

import org.springframework.stereotype.Service;

import com.lps.isoswitch.config.SchemeConfig;
import com.lps.isoswitch.tcpconnector.SchemeManagementHandler;
import com.lps.isoswitch.tcpconnector.SchemesConnector;
import com.lps.isoswitch.utils.Enums.SwitchDestinationEnum;

/**
 * Central registry for resolving scheme-specific connectors and handlers.
 */
@Service
public class SchemeRegistryService {

    private final Map<SwitchDestinationEnum, SchemesConnector> connectors;
    private final Map<SwitchDestinationEnum, SchemeManagementHandler> handlers;
    private final SchemeConfig schemeConfig;

    public SchemeRegistryService(
            List<SchemesConnector> connectorBeans,
            List<SchemeManagementHandler> handlerBeans,
            SchemeConfig schemeConfig) {
        this.schemeConfig = schemeConfig;
        this.connectors = new EnumMap<>(SwitchDestinationEnum.class);
        this.handlers = new EnumMap<>(SwitchDestinationEnum.class);

        for (SchemesConnector connector : connectorBeans) {
            connectors.put(connector.getSupportedScheme(), connector);
        }

        for (SchemeManagementHandler handler : handlerBeans) {
            handlers.put(handler.getSupportedScheme(), handler);
        }
    }

    public SwitchDestinationEnum getActiveScheme() {
        return schemeConfig.getActiveScheme();
    }

    public SchemeConfig.ResolvedSchemeSettings getActiveSchemeSettings(
            String fallbackHost,
            Integer fallbackPort,
            String fallbackConnectionId) {
        return schemeConfig.resolve(
                getActiveScheme(),
                fallbackHost,
                fallbackPort,
                fallbackConnectionId,
                null,
                null,
                null,
                null,
                null,
                null);
    }

    public SchemesConnector getActiveConnector() {
        return getConnector(getActiveScheme());
    }

    public SchemesConnector getConnector(SwitchDestinationEnum scheme) {
        SchemesConnector connector = connectors.get(scheme);
        if (connector == null) {
            throw new IllegalStateException("No connector registered for scheme: " + scheme);
        }
        return connector;
    }

    public SchemeManagementHandler getManagementHandler(SwitchDestinationEnum scheme) {
        SchemeManagementHandler handler = handlers.get(scheme);
        if (handler == null) {
            throw new IllegalStateException("No management handler registered for scheme: " + scheme);
        }
        return handler;
    }

    public List<SchemesConnector> getAllConnectors() {
        return List.copyOf(connectors.values());
    }
}
