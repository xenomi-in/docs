package com.lps.isoswitch.service;

import com.lps.isoswitch.config.SchemeConfig;
import com.lps.isoswitch.model.IsoRequest;
import com.lps.isoswitch.tcpconnector.SchemeManagementHandler;
import com.lps.isoswitch.utils.Enums.SwitchDestinationEnum;
import org.jpos.iso.ISOException;
import org.jpos.iso.ISOMsg;
import org.jpos.iso.ISOPackager;
import org.jpos.iso.packager.GenericPackager;
import org.springframework.stereotype.Service;

import java.io.InputStream;
import java.util.EnumMap;
import java.util.Map;
import java.util.concurrent.CompletableFuture;

/**
 * Accepts Switch Core requests, packs them as ISO8583, and writes them to the
 * scheme socket that is already connected.
 */
@Service
public class SwitchCoreOutboundService {

    private final SchemeRegistryService schemeRegistryService;
    private final SchemeConfig schemeConfig;
    private final IsoRequestMapper isoRequestMapper;
    private final Map<SwitchDestinationEnum, ISOPackager> packagers = new EnumMap<>(SwitchDestinationEnum.class);

    public SwitchCoreOutboundService(
            SchemeRegistryService schemeRegistryService,
            SchemeConfig schemeConfig,
            IsoRequestMapper isoRequestMapper) {
        this.schemeRegistryService = schemeRegistryService;
        this.schemeConfig = schemeConfig;
        this.isoRequestMapper = isoRequestMapper;
    }

    public CompletableFuture<Boolean> sendToScheme(IsoRequest request, SwitchDestinationEnum scheme) throws ISOException {
        ISOMsg isoMsg = isoRequestMapper.convertApiRequest(request, getPackager(scheme));
        SchemeManagementHandler handler = schemeRegistryService.getManagementHandler(scheme);
        return handler.sendIsoMessage(isoMsg);
    }

    private synchronized ISOPackager getPackager(SwitchDestinationEnum scheme) throws ISOException {
        ISOPackager packager = packagers.get(scheme);
        if (packager != null) {
            return packager;
        }

        String packagerPath = schemeConfig.resolve(
                scheme,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null).packager();
        try (InputStream inputStream = getClass().getResourceAsStream(packagerPath)) {
            if (inputStream == null) {
                throw new ISOException("Packager configuration file not found: " + packagerPath);
            }
            ISOPackager loadedPackager = new GenericPackager(inputStream);
            packagers.put(scheme, loadedPackager);
            return loadedPackager;
        } catch (ISOException e) {
            throw e;
        } catch (Exception e) {
            throw new ISOException("Failed to load packager: " + packagerPath, e);
        }
    }
}
