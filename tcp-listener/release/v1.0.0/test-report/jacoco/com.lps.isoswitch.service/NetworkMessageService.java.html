package com.lps.isoswitch.service;

import com.lps.isoswitch.config.SchemeConfig;
import com.lps.isoswitch.utils.Enums.SwitchDestinationEnum;
import org.jpos.iso.ISOMsg;
import org.jpos.iso.ISOException;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.stereotype.Service;

/**
 * Service class for handling network messages in the ISO8583 Switch.
 * This service manages different types of network messages (sign-on, sign-off, echo, key exchange)
 * for various card schemes (RuPay, Visa, MasterCard).
 *
 * @author Logicpoint Systems
 * @version 1.0
 * @since 2025-06-15
 */
@Service
public class NetworkMessageService {
    
    private static final Logger log = LoggerFactory.getLogger(NetworkMessageService.class);
    
    // Response codes
    private static final String RESPONSE_APPROVED = "00";
    private static final String RESPONSE_DECLINED = "05";
    
    // Network message type field
    private static final int NETWORK_MESSAGE_TYPE_FIELD = 70;
    
    private final SchemeConfig schemeConfig;
    
    /**
     * Constructor receives scheme configuration so field 70 network codes can
     * differ per scheme without changing Java code.
     */
    public NetworkMessageService(SchemeConfig schemeConfig) {
        this.schemeConfig = schemeConfig;
    }
    
    /**
     * Handles network messages based on their type and scheme.
     * 
     * @param requestMsg The incoming network message
     * @param scheme The card scheme (RuPay, Visa, MasterCard)
     * @return ISOMsg containing the response
     * @throws ISOException if there's an error processing the message
     */
    public ISOMsg handleNetworkMessage(ISOMsg requestMsg, String scheme) throws ISOException {
        if (!requestMsg.hasField(NETWORK_MESSAGE_TYPE_FIELD)) {
            throw new ISOException("Network message type field (70) is missing");
        }
        
        SwitchDestinationEnum destination = SwitchDestinationEnum.from(scheme);
        SchemeConfig.NetworkCodes networkCodes = schemeConfig.getNetworkCodes(destination);
        String networkMessageType = requestMsg.getString(NETWORK_MESSAGE_TYPE_FIELD);
        log.info("Processing {} network message type: {}", destination, networkMessageType);
        
        if (networkCodes.getSignOn().equals(networkMessageType)) {
            return handleSignOn(requestMsg, destination);
        }
        if (networkCodes.getSignOff().equals(networkMessageType)) {
            return handleSignOff(requestMsg, destination);
        }
        if (networkCodes.getEcho().equals(networkMessageType)) {
            return handleEcho(requestMsg, destination);
        }
        if (networkCodes.getDkeZpk().equals(networkMessageType)
                || networkCodes.getDkeZak().equals(networkMessageType)) {
            return handleKeyExchange(requestMsg, destination);
        }
        if (networkCodes.getCutover().equals(networkMessageType)) {
            return handleCutover(requestMsg, destination);
        }
        if (networkCodes.getNetworkManagement().equals(networkMessageType)) {
            return handleNetworkManagement(requestMsg, destination);
        }

        return handleUnknownMessage(requestMsg, networkMessageType);
    }

    /**
     * Creates the response shell shared by all local network-management
     * responses: response MTI plus fields that must be echoed back.
     */
    private ISOMsg createBaseResponse(ISOMsg requestMsg) throws ISOException {
        ISOMsg responseMsg = new ISOMsg();
        responseMsg.setPackager(requestMsg.getPackager());

        String mti = requestMsg.getMTI();
        int mtiNumber = Integer.parseInt(mti) + 10;
        responseMsg.setMTI("0" + mtiNumber);

        copyEssentialFields(requestMsg, responseMsg);
        return responseMsg;
    }

    /**
     * Copies fields that should be mirrored in a network-management response.
     */
    private void copyEssentialFields(ISOMsg requestMsg, ISOMsg responseMsg) throws ISOException {
        int[] essentialFields = {2, 3, 4, 7, 11, 12, 13, 32, 37, 41, 42, 49, NETWORK_MESSAGE_TYPE_FIELD};

        for (int field : essentialFields) {
            if (requestMsg.hasField(field)) {
                responseMsg.set(field, requestMsg.getString(field));
            }
        }
    }

    /**
     * Handles configured sign-on requests for the active scheme.
     */
    private ISOMsg handleSignOn(ISOMsg requestMsg, SwitchDestinationEnum scheme) throws ISOException {
        log.info("Processing {} sign-on request", scheme);
        ISOMsg responseMsg = createBaseResponse(requestMsg);
        responseMsg.set(39, RESPONSE_APPROVED);
        log.info("{} SIGN-ON SUCCESSFUL", scheme);
        return responseMsg;
    }

    /**
     * Handles configured sign-off requests for the active scheme.
     */
    private ISOMsg handleSignOff(ISOMsg requestMsg, SwitchDestinationEnum scheme) throws ISOException {
        log.info("Processing {} sign-off request", scheme);
        ISOMsg responseMsg = createBaseResponse(requestMsg);
        responseMsg.set(39, RESPONSE_APPROVED);
        log.info("{} SIGN-OFF SUCCESSFUL", scheme);
        return responseMsg;
    }

    /**
     * Handles configured echo requests for the active scheme.
     */
    private ISOMsg handleEcho(ISOMsg requestMsg, SwitchDestinationEnum scheme) throws ISOException {
        log.info("Processing {} echo request", scheme);
        ISOMsg responseMsg = createBaseResponse(requestMsg);
        responseMsg.set(39, RESPONSE_APPROVED);
        log.info("{} ECHO RESPONSE", scheme);
        return responseMsg;
    }

    /**
     * Handles configured DKE/key-exchange requests for the active scheme.
     */
    private ISOMsg handleKeyExchange(ISOMsg requestMsg, SwitchDestinationEnum scheme) throws ISOException {
        log.info("Processing {} key exchange request", scheme);
        ISOMsg responseMsg = createBaseResponse(requestMsg);
        responseMsg.set(39, RESPONSE_APPROVED);
        log.info("{} KEY EXCHANGE SUCCESSFUL", scheme);
        return responseMsg;
    }

    /**
     * Handles configured cutover requests for schemes that use them.
     */
    private ISOMsg handleCutover(ISOMsg requestMsg, SwitchDestinationEnum scheme) throws ISOException {
        log.info("Processing {} cutover request", scheme);
        ISOMsg responseMsg = createBaseResponse(requestMsg);
        responseMsg.set(39, RESPONSE_APPROVED);
        log.info("{} CUTOVER SUCCESSFUL", scheme);
        return responseMsg;
    }

    /**
     * Handles configured generic network-management requests for schemes that
     * require a separate field 70 code.
     */
    private ISOMsg handleNetworkManagement(ISOMsg requestMsg, SwitchDestinationEnum scheme) throws ISOException {
        log.info("Processing {} network management request", scheme);
        ISOMsg responseMsg = createBaseResponse(requestMsg);
        responseMsg.set(39, RESPONSE_APPROVED);
        log.info("{} NETWORK MANAGEMENT SUCCESSFUL", scheme);
        return responseMsg;
    }

    /**
     * Declines unconfigured or unsupported network-management message types.
     */
    private ISOMsg handleUnknownMessage(ISOMsg requestMsg, String messageType) throws ISOException {
        log.warn("Received unknown network message type: {}", messageType);
        ISOMsg responseMsg = createBaseResponse(requestMsg);
        responseMsg.set(39, RESPONSE_DECLINED);
        return responseMsg;
    }
}
