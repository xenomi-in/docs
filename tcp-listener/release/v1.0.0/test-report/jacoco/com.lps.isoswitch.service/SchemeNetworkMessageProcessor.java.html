package com.lps.isoswitch.service;

import com.lps.isoswitch.config.SchemeConfig;
import com.lps.isoswitch.utils.Enums.SwitchDestinationEnum;
import org.jpos.iso.ISOException;
import org.jpos.iso.ISOMsg;
import org.jpos.iso.ISOPackager;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.stereotype.Service;

import java.text.SimpleDateFormat;
import java.util.Date;
import java.util.Optional;

/**
 * Encapsulates scheme network-management decisions for the TCP listener.
 *
 * <p>The Netty handler should only parse bytes, dispatch messages, frame
 * responses, and write to the socket. This service owns the business rules for
 * network-management MTIs: whether a message is handled locally, whether it is
 * routed to Switch Core, how local sign-on/sign-off responses update listener
 * readiness, and how outbound sign-on/sign-off/DKE/echo requests are built.</p>
 */
@Service
public class SchemeNetworkMessageProcessor {

    private static final Logger log = LoggerFactory.getLogger(SchemeNetworkMessageProcessor.class);

    private static final String MTI_NETWORK_REQUEST_SUFFIX = "800";
    private static final String MTI_NETWORK_RESPONSE_SUFFIX = "810";
    private static final String RESPONSE_APPROVED = "00";
    private static final int FIELD_RESPONSE_CODE = 39;
    private static final int FIELD_NETWORK_CODE = 70;

    private final SchemeConfig schemeConfig;
    private final NetworkMessageService networkMessageService;
    private final ConnectionStatusService connectionStatusService;

    public SchemeNetworkMessageProcessor(
            SchemeConfig schemeConfig,
            NetworkMessageService networkMessageService,
            ConnectionStatusService connectionStatusService) {
        this.schemeConfig = schemeConfig;
        this.networkMessageService = networkMessageService;
        this.connectionStatusService = connectionStatusService;
    }

    /**
     * Decides whether an inbound ISO message should go to Switch Core.
     *
     * <p>Financial messages always go to Switch Core. Network-management DKE
     * requests configured as switch-core DKE codes also go to Switch Core.
     * Other network-management messages are handled locally by this listener.</p>
     *
     * @param isoMsg inbound ISO message
     * @param scheme active scheme used to resolve field 70 code values
     * @return true when the handler should invoke the API path
     * @throws ISOException when MTI or field access fails
     */
    public boolean shouldRouteToSwitchCore(ISOMsg isoMsg, SwitchDestinationEnum scheme) throws ISOException {
        return !isNetworkMessage(isoMsg) || isSwitchCoreDkeRequest(isoMsg, scheme);
    }

    /**
     * Handles a locally processed network-management message.
     *
     * <p>For request MTIs, this delegates to {@link NetworkMessageService} and
     * returns a response for the TCP handler to write. For response MTIs, it
     * updates connection/sign-on state and returns empty because no socket
     * response is required.</p>
     *
     * @param isoMsg inbound network-management message
     * @param scheme active scheme
     * @return optional response message to send back over TCP
     * @throws ISOException when network-message processing fails
     */
    public Optional<ISOMsg> handleLocalNetworkMessage(ISOMsg isoMsg, SwitchDestinationEnum scheme) throws ISOException {
        if (isoMsg.getMTI().endsWith(MTI_NETWORK_REQUEST_SUFFIX)) {
            ISOMsg responseMsg = networkMessageService.handleNetworkMessage(isoMsg, scheme.name());
            updateSignOnStatusFromNetworkResult(isoMsg, responseMsg, scheme);
            return Optional.of(responseMsg);
        }

        updateStatusFromNetworkResponse(isoMsg, scheme);
        return Optional.empty();
    }

    /**
     * Creates an outbound sign-on request using the scheme-specific field 70 code.
     */
    public ISOMsg createSignOnRequest(ISOPackager packager, SwitchDestinationEnum scheme) throws ISOException {
        return createNetworkRequest(packager, schemeConfig.getNetworkCodes(scheme).getSignOn());
    }

    /**
     * Creates an outbound sign-off request using the scheme-specific field 70 code.
     */
    public ISOMsg createSignOffRequest(ISOPackager packager, SwitchDestinationEnum scheme) throws ISOException {
        return createNetworkRequest(packager, schemeConfig.getNetworkCodes(scheme).getSignOff());
    }

    /**
     * Creates an outbound echo-test request using the scheme-specific field 70 code.
     */
    public ISOMsg createEchoTestRequest(ISOPackager packager, SwitchDestinationEnum scheme) throws ISOException {
        return createNetworkRequest(packager, schemeConfig.getNetworkCodes(scheme).getEcho());
    }

    /**
     * Creates an outbound DKE request for supported key types.
     *
     * @param packager scheme-specific packager
     * @param dkeType ZPK or ZAK
     * @return network-management request with the correct field 70 value
     */
    public ISOMsg createDkeRequest(ISOPackager packager, String dkeType, SwitchDestinationEnum scheme) throws ISOException {
        SchemeConfig.NetworkCodes networkCodes = schemeConfig.getNetworkCodes(scheme);
        if ("ZPK".equalsIgnoreCase(dkeType)) {
            return createNetworkRequest(packager, networkCodes.getDkeZpk());
        }
        if ("ZAK".equalsIgnoreCase(dkeType)) {
            return createNetworkRequest(packager, networkCodes.getDkeZak());
        }
        throw new IllegalArgumentException("Invalid DKE type: " + dkeType);
    }

    /**
     * Network-management MTIs end with 800 for requests or 810 for responses.
     */
    private boolean isNetworkMessage(ISOMsg isoMsg) throws ISOException {
        String mti = isoMsg.getMTI();
        return mti.endsWith(MTI_NETWORK_REQUEST_SUFFIX) || mti.endsWith(MTI_NETWORK_RESPONSE_SUFFIX);
    }

    /**
     * Configured switch-core DKE field 70 values are passed to Switch Core.
     */
    private boolean isSwitchCoreDkeRequest(ISOMsg isoMsg, SwitchDestinationEnum scheme) throws ISOException {
        SchemeConfig.NetworkCodes networkCodes = schemeConfig.getNetworkCodes(scheme);
        String networkCode = isoMsg.getString(FIELD_NETWORK_CODE);
        return networkCodes.getSwitchCoreDkeZpk().equals(networkCode)
                || networkCodes.getSwitchCoreDkeZak().equals(networkCode);
    }

    /**
     * Builds the common ISO0800 skeleton used by sign-on, sign-off, DKE, and
     * echo-test requests.
     */
    private ISOMsg createNetworkRequest(ISOPackager packager, String networkCode) throws ISOException {
        ISOMsg networkMsg = new ISOMsg();
        networkMsg.setPackager(packager);
        networkMsg.setMTI("0800");
        networkMsg.set(7, transmissionDateTime());
        networkMsg.set(11, nextStan());
        networkMsg.set(FIELD_NETWORK_CODE, networkCode);
        return networkMsg;
    }

    /**
     * Updates listener readiness after locally answering sign-on/sign-off
     * requests from the scheme.
     */
    private void updateSignOnStatusFromNetworkResult(
            ISOMsg requestMsg,
            ISOMsg responseMsg,
            SwitchDestinationEnum scheme) throws ISOException {
        if (!RESPONSE_APPROVED.equals(responseMsg.getString(FIELD_RESPONSE_CODE))) {
            return;
        }

        SchemeConfig.NetworkCodes networkCodes = schemeConfig.getNetworkCodes(scheme);
        String networkCode = requestMsg.getString(FIELD_NETWORK_CODE);
        if (networkCodes.getSignOn().equals(networkCode)) {
            log.info("Updating sign-on status for {}", scheme);
            connectionStatusService.updateSignOnStatus(scheme, true);
        } else if (networkCodes.getSignOff().equals(networkCode)) {
            log.info("Updating sign-off status for {}", scheme);
            connectionStatusService.updateSignOnStatus(scheme, false);
        }
    }

    /**
     * Updates listener state when the scheme sends a network-management response
     * to a request initiated by this listener.
     */
    private void updateStatusFromNetworkResponse(ISOMsg isoMsg, SwitchDestinationEnum scheme) throws ISOException {
        if (!RESPONSE_APPROVED.equals(isoMsg.getString(FIELD_RESPONSE_CODE))) {
            return;
        }

        SchemeConfig.NetworkCodes networkCodes = schemeConfig.getNetworkCodes(scheme);
        String networkCode = isoMsg.getString(FIELD_NETWORK_CODE);
        if (networkCodes.getSignOn().equals(networkCode)) {
            log.info("Updating sign-on status for {}", scheme);
            connectionStatusService.updateSignOnStatus(scheme, true);
        } else if (networkCodes.getSignOff().equals(networkCode)) {
            log.info("Updating sign-off status for {}", scheme);
            connectionStatusService.updateSignOnStatus(scheme, false);
        } else if (networkCodes.getEcho().equals(networkCode)) {
            log.debug("Received echo test response for {} - connection is healthy", scheme);
            connectionStatusService.updateConnectionStatus(scheme, true);
        }
    }

    /**
     * ISO8583 transmission date/time format: MMDDhhmmss.
     */
    private String transmissionDateTime() {
        return new SimpleDateFormat("MMddHHmmss").format(new Date());
    }

    /**
     * Generates a six-digit STAN for listener-initiated network messages.
     */
    private String nextStan() {
        return String.format("%06d", (int) (Math.random() * 1000000));
    }
}
