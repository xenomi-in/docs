package com.lps.isoswitch.utils;

/**
 * Enums class for the ISO Switch application.
 * This class contains all application-wide enums used
 * throughout the codebase.
 *
 * @author Logicpoint Systems
 * @version 1.0
 * @since 2025-06-15
 */
public class Enums {

	public enum StatusEnum {
		INACTIVE, ACTIVE, SUCCESS, FAIL, NONE
	}
    public enum SwitchDestinationEnum {
		RUPAY, VISA, MC;

        public static SwitchDestinationEnum from(String value) {
            if (value == null || value.isBlank()) {
                throw new IllegalArgumentException("Scheme value cannot be blank");
            }

            return switch (value.trim().toUpperCase()) {
                case "RUPAY" -> RUPAY;
                case "VISA" -> VISA;
                case "MC", "MASTERCARD" -> MC;
                default -> throw new IllegalArgumentException("Unsupported scheme: " + value);
            };
        }
    }
}
