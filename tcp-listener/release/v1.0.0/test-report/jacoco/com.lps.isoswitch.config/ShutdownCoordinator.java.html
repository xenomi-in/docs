package com.lps.isoswitch.config;

import java.util.concurrent.atomic.AtomicBoolean;

import com.lps.isoswitch.service.ApiService;
import com.lps.isoswitch.service.SchemeRegistryService;
import com.lps.isoswitch.tcpconnector.RuPayClientHandler;
import com.lps.isoswitch.tcpconnector.SchemesConnector;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.context.ApplicationListener;
import org.springframework.context.event.ContextClosedEvent;
import org.springframework.stereotype.Component;

import jakarta.annotation.PreDestroy;
import java.util.concurrent.CompletableFuture;
import java.util.concurrent.TimeUnit;

/**
 * Coordinates graceful shutdown of all application components.
 * Ensures proper cleanup of resources when the application is stopped.
 * 
 * @author Logicpoint Systems
 * @version 1.0
 * @since 2025-06-15
 */
@Component
public class ShutdownCoordinator implements ApplicationListener<ContextClosedEvent> {

    private static final Logger log = LoggerFactory.getLogger(ShutdownCoordinator.class);
    private static final int SHUTDOWN_TIMEOUT_SECONDS = 30;
    private final AtomicBoolean shutdownStarted = new AtomicBoolean(false);
    
    @Autowired
    private ApiService apiService;
    
    @Autowired
    private RuPayClientHandler ruPayClientHandler;

    @Autowired
    private SchemeRegistryService schemeRegistryService;

    /**
     * Handles Spring Boot application shutdown event.
     * Coordinates graceful shutdown of all components.
     *
     * @param event The context closed event
     */
    @Override
    public void onApplicationEvent(ContextClosedEvent event) {
        log.info("Received Spring Boot shutdown event. Initiating graceful shutdown...");
        performGracefulShutdown();
    }

    /**
     * PreDestroy method as backup for shutdown coordination.
     * Called when the Spring context is being destroyed.
     */
    @PreDestroy
    public void onPreDestroy() {
        log.info("PreDestroy called. Ensuring graceful shutdown...");
        performGracefulShutdown();
    }

    /**
     * Performs graceful shutdown of all application components.
     */
    public void performGracefulShutdown() {
        if (!shutdownStarted.compareAndSet(false, true)) {
            log.info("Graceful shutdown already in progress");
            return;
        }
        try {
            log.info("Starting graceful shutdown sequence...");
            
            // Step 1: Stop accepting new requests
            log.info("Step 1: Stopping acceptance of new requests");
            
            // Step 2: Shutdown TCP handlers
            shutdownTcpHandlers();

            // Step 3: Shutdown scheme connectors
            shutdownSchemeConnectors();
            
            // Step 4: Shutdown API service
            shutdownApiService();

            // Step 5: Wait for completion
            waitForShutdownCompletion();
            
            log.info("Graceful shutdown completed successfully");
            
        } catch (Exception e) {
            log.error("Error during graceful shutdown", e);
        }
    }

    /**
     * Shuts down TCP client handlers.
     */
    private void shutdownTcpHandlers() {
        log.info("Shutting down TCP client handlers...");
        
        try {
            if (ruPayClientHandler != null) {
                CompletableFuture<Void> handlerShutdown = ruPayClientHandler.shutdown();
                if (handlerShutdown != null) {
                    handlerShutdown.get(SHUTDOWN_TIMEOUT_SECONDS, TimeUnit.SECONDS);
                    log.info("RuPayClientHandler shutdown completed");
                }
            }
            
        } catch (Exception e) {
            log.error("Error shutting down TCP handlers", e);
        }
    }

    private void shutdownSchemeConnectors() {
        log.info("Shutting down scheme connectors...");

        for (SchemesConnector connector : schemeRegistryService.getAllConnectors()) {
            try {
                connector.setShuttingDown(true);
                connector.shutdown();
                log.info("Connector shutdown completed for {}", connector.getSupportedScheme());
            } catch (Exception e) {
                log.error("Error shutting down connector for {}", connector.getSupportedScheme(), e);
            }
        }
    }

    /**
     * Shuts down the API service.
     */
    private void shutdownApiService() {
        log.info("Shutting down API service...");
        
        try {
            if (apiService != null) {
                apiService.shutdown();
                log.info("API service shutdown completed");
            }
        } catch (Exception e) {
            log.error("Error shutting down API service", e);
        }
    }

    /**
     * Shuts down the schemes connector.
     */
    private void shutdownSchemesConnector() {
        log.info("Shutting down schemes connector...");
        
        try {
            //if (schemesConnector != null) {
            //    schemesConnector.shutdown();
            //    log.info("Schemes connector shutdown completed");
            //}
        } catch (Exception e) {
            log.error("Error shutting down schemes connector", e);
        }
    }

    /**
     * Waits for shutdown completion with timeout.
     */
    private void waitForShutdownCompletion() {
        log.info("Waiting for shutdown completion...");
        
        try {
            // Wait for any remaining operations to complete
            Thread.sleep(2000);
            log.info("Shutdown wait completed");
        } catch (InterruptedException e) {
            log.warn("Shutdown wait interrupted", e);
            Thread.currentThread().interrupt();
        }
    }
}
