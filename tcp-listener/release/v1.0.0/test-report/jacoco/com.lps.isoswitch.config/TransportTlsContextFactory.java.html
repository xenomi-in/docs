package com.lps.isoswitch.config;

import java.io.InputStream;
import java.security.KeyStore;

import javax.net.ssl.KeyManagerFactory;
import javax.net.ssl.TrustManagerFactory;

import org.springframework.core.io.Resource;
import org.springframework.core.io.ResourceLoader;
import org.springframework.stereotype.Component;

import io.netty.handler.ssl.SslContext;
import io.netty.handler.ssl.SslContextBuilder;
import io.netty.handler.ssl.SslProvider;

/** Builds client-side Netty TLS contexts from configured key/trust stores. */
@Component
public class TransportTlsContextFactory {

    private final ResourceLoader resourceLoader;

    public TransportTlsContextFactory(ResourceLoader resourceLoader) {
        this.resourceLoader = resourceLoader;
    }

    public SslContext createClientContext(
            String connectionName,
            TransportSecurityProperties.TlsEndpoint endpoint) {
        if (!endpoint.isEnabled()) {
            throw new IllegalStateException(connectionName + " TLS is disabled");
        }
        requireText(endpoint.getKeyStore(), connectionName + " key store");
        requireText(endpoint.getKeyStorePassword(), connectionName + " key-store password");
        requireText(endpoint.getTrustStore(), connectionName + " trust store");
        requireText(endpoint.getTrustStorePassword(), connectionName + " trust-store password");

        try {
            KeyStore keyStore = loadStore(endpoint.getKeyStore(), endpoint.getKeyStoreType(),
                    endpoint.getKeyStorePassword());
            KeyManagerFactory keyManagers = KeyManagerFactory.getInstance(KeyManagerFactory.getDefaultAlgorithm());
            String keyPassword = hasText(endpoint.getKeyPassword())
                    ? endpoint.getKeyPassword()
                    : endpoint.getKeyStorePassword();
            keyManagers.init(keyStore, keyPassword.toCharArray());

            KeyStore trustStore = loadStore(endpoint.getTrustStore(), endpoint.getTrustStoreType(),
                    endpoint.getTrustStorePassword());
            TrustManagerFactory trustManagers = TrustManagerFactory.getInstance(
                    TrustManagerFactory.getDefaultAlgorithm());
            trustManagers.init(trustStore);

            return SslContextBuilder.forClient()
                    .sslProvider(SslProvider.JDK)
                    .protocols("TLSv1.3", "TLSv1.2")
                    .keyManager(keyManagers)
                    .trustManager(trustManagers)
                    .build();
        } catch (Exception exception) {
            throw new IllegalStateException("Unable to initialize " + connectionName + " TLS material", exception);
        }
    }

    private KeyStore loadStore(String location, String type, String password) throws Exception {
        Resource resource = resourceLoader.getResource(location);
        if (!resource.exists() || !resource.isReadable()) {
            throw new IllegalStateException("TLS store is not readable: " + location);
        }
        KeyStore store = KeyStore.getInstance(type);
        try (InputStream input = resource.getInputStream()) {
            store.load(input, password.toCharArray());
        }
        return store;
    }

    private void requireText(String value, String name) {
        if (!hasText(value)) {
            throw new IllegalStateException(name + " is required");
        }
    }

    private boolean hasText(String value) {
        return value != null && !value.isBlank();
    }
}
