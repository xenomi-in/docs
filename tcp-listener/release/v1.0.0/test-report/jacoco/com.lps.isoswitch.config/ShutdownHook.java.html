package com.lps.isoswitch.config;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Component;

import jakarta.annotation.PostConstruct;

/**
 * Runtime shutdown hook as a backup for graceful shutdown.
 * Ensures cleanup even if Spring shutdown doesn't work properly.
 * 
 * @author Logicpoint Systems
 * @version 1.0
 * @since 2025-06-15
 */
@Component
public class ShutdownHook {

    private static final Logger log = LoggerFactory.getLogger(ShutdownHook.class);
    
    @Autowired
    private ShutdownCoordinator shutdownCoordinator;

    /**
     * Registers the shutdown hook with the JVM.
     */
    @PostConstruct
    public void registerShutdownHook() {
        Runtime.getRuntime().addShutdownHook(new Thread(() -> {
            log.info("JVM shutdown hook triggered");
            try {
                shutdownCoordinator.performGracefulShutdown();
            } catch (Exception e) {
                log.error("Error in shutdown hook", e);
            }
        }, "shutdown-hook"));
        
        log.info("Shutdown hook registered successfully");
    }
}
