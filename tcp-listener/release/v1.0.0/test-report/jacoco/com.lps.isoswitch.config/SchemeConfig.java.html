package com.lps.isoswitch.config;

import lombok.Data;
import org.springframework.boot.context.properties.ConfigurationProperties;
import org.springframework.context.annotation.Configuration;
import java.util.List;

import com.lps.isoswitch.utils.Enums.SwitchDestinationEnum;

/**
 * Configuration class for card scheme settings.
 * This class provides configuration for different card schemes
 * (RuPay, Visa, MasterCard) including their specific settings
 * and connection parameters.
 *
 * @author Logicpoint Systems
 * @version 1.0
 * @since 2025-06-15
 */
@Data
@Configuration
@ConfigurationProperties(prefix = "scheme")
public class SchemeConfig {
    private String active = "RUPAY";
    private RuPayConfig rupay;
    private VisaConfig visa;
    private McConfig mc;

    public void setRupay(RuPayConfig rupay) {
        this.rupay = rupay;
    }

    public void setVisa(VisaConfig visa) {
        this.visa = visa;
    }
    public RuPayConfig getRupay() {
        return  this.rupay;
    }

    public VisaConfig getVisa() {
        return this.visa;
    }

    public void setMc(McConfig mc) {
        this.mc = mc;
    }

    public McConfig getMc() {
        return this.mc;
    }

    public SwitchDestinationEnum getActiveScheme() {
        return SwitchDestinationEnum.from(active);
    }

    public BaseConfig getConfig(SwitchDestinationEnum scheme) {
        return switch (scheme) {
            case RUPAY -> rupay;
            case VISA -> visa;
            case MC -> mc;
        };
    }

    public NetworkCodes getNetworkCodes(SwitchDestinationEnum scheme) {
        BaseConfig config = getConfig(scheme);
        if (config == null) {
            throw new IllegalStateException("No configuration found for scheme: " + scheme);
        }
        return config.getNetworkCodes().withDefaults(defaultNetworkCodes());
    }

    public ResolvedSchemeSettings resolve(
            SwitchDestinationEnum scheme,
            String fallbackHost,
            Integer fallbackPort,
            String fallbackConnectionId,
            Integer fallbackConnectTimeout,
            Integer fallbackReadTimeout,
            Integer fallbackWriteTimeout,
            Integer fallbackIdleTimeout,
            Boolean fallbackKeepAliveEnabled,
            Integer fallbackReconnectDelay) {
        BaseConfig config = getConfig(scheme);
        if (config == null) {
            throw new IllegalStateException("No configuration found for scheme: " + scheme);
        }

        String resolvedConnectionId = isBlank(config.getConnectionId())
                ? (isBlank(fallbackConnectionId) ? scheme.name() : fallbackConnectionId)
                : config.getConnectionId();

        return new ResolvedSchemeSettings(
                scheme,
                firstNonBlank(config.getHost(), fallbackHost),
                config.getPort() != null ? config.getPort() : fallbackPort,
                resolvedConnectionId,
                firstNonBlank(config.getPackager(), defaultPackagerPath(scheme)),
                firstNonNull(config.getConnectTimeout(), fallbackConnectTimeout),
                firstNonNull(config.getReadTimeout(), fallbackReadTimeout),
                firstNonNull(config.getWriteTimeout(), fallbackWriteTimeout),
                firstNonNull(config.getIdleTimeout(), fallbackIdleTimeout),
                firstNonNull(config.getKeepAliveEnabled(), fallbackKeepAliveEnabled),
                firstNonNull(config.getReconnectDelay(), fallbackReconnectDelay));
    }

    private String firstNonBlank(String primary, String fallback) {
        return isBlank(primary) ? fallback : primary;
    }

    private <T> T firstNonNull(T primary, T fallback) {
        return primary != null ? primary : fallback;
    }

    private boolean isBlank(String value) {
        return value == null || value.isBlank();
    }

    private String defaultPackagerPath(SwitchDestinationEnum scheme) {
        return switch (scheme) {
            case RUPAY -> "/RuPayISO8583.xml";
            case VISA -> "/VisaISO8583.xml";
            case MC -> "/MasterCardISO8583.xml";
        };
    }

    private NetworkCodes defaultNetworkCodes() {
        NetworkCodes defaults = new NetworkCodes();
        defaults.setSignOn("001");
        defaults.setSignOff("002");
        defaults.setEcho("301");
        defaults.setCutover("201");
        defaults.setDkeZpk("164");
        defaults.setDkeZak("165");
        defaults.setNetworkManagement("181");
        defaults.setSwitchCoreDkeZpk("184");
        defaults.setSwitchCoreDkeZak("185");
        return defaults;
    }

        public NetworkCodes withDefaults(NetworkCodes defaults) {
            NetworkCodes resolved = new NetworkCodes();
            resolved.setSignOn(firstNonBlank(signOn, defaults.getSignOn()));
            resolved.setSignOff(firstNonBlank(signOff, defaults.getSignOff()));
            resolved.setEcho(firstNonBlank(echo, defaults.getEcho()));
            resolved.setCutover(firstNonBlank(cutover, defaults.getCutover()));
            resolved.setDkeZpk(firstNonBlank(dkeZpk, defaults.getDkeZpk()));
            resolved.setDkeZak(firstNonBlank(dkeZak, defaults.getDkeZak()));
            resolved.setNetworkManagement(firstNonBlank(networkManagement, defaults.getNetworkManagement()));
            resolved.setSwitchCoreDkeZpk(firstNonBlank(switchCoreDkeZpk, defaults.getSwitchCoreDkeZpk()));
            resolved.setSwitchCoreDkeZak(firstNonBlank(switchCoreDkeZak, defaults.getSwitchCoreDkeZak()));
            return resolved;
        }

        private String firstNonBlank(String primary, String fallback) {
            return primary == null || primary.isBlank() ? fallback : primary;
        }
    }

    @Data

        public RuPayConfig() {

                "0800"  // Network management request
            ));
        }
    }
