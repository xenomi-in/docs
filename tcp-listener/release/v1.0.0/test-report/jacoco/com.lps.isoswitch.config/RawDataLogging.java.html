package com.lps.isoswitch.config;

import java.io.ByteArrayOutputStream;
import java.io.PrintStream;
import java.nio.charset.StandardCharsets;
import java.util.Arrays;

import org.jpos.iso.ISOMsg;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.core.env.Environment;
import org.springframework.stereotype.Component;

import jakarta.annotation.PostConstruct;

/**
 * Explicitly gated raw payment-message logging for local troubleshooting.
 * Raw logging is never available while the {@code prod} profile is active.
 */
@Component
public final class RawDataLogging {

    private static final Logger rawLog = LoggerFactory.getLogger("com.lps.isoswitch.rawdata");
    private static volatile boolean enabled;

    private final boolean requested;
    private final Environment environment;

    public RawDataLogging(
            @Value("${security.raw-data-logging.enabled:false}") boolean requested,
            Environment environment) {
        this.requested = requested;
        this.environment = environment;
    }

    @PostConstruct
    void configure() {
        enabled = false;
        boolean production = Arrays.stream(environment.getActiveProfiles())
                .anyMatch("prod"::equalsIgnoreCase);
        if (production && requested) {
            throw new IllegalStateException(
                    "Raw payment-data logging cannot be enabled with the prod profile");
        }
        enabled = requested && !production;
        if (enabled) {
            rawLog.warn("RAW PAYMENT-DATA LOGGING IS ENABLED FOR NON-PRODUCTION TROUBLESHOOTING");
        }
    }

    public static boolean isEnabled() {
        return enabled && rawLog.isDebugEnabled();
    }

    public static void bytes(String label, byte[] value) {
        if (isEnabled() && value != null) {
            rawLog.debug("{}: {}", label, new String(value, StandardCharsets.ISO_8859_1));
        }
    }

    public static void object(String label, Object value) {
        if (isEnabled()) {
            rawLog.debug("{}: {}", label, value);
        }
    }

    public static void isoMessage(String label, ISOMsg message) {
        if (!isEnabled() || message == null) {
            return;
        }
        ByteArrayOutputStream output = new ByteArrayOutputStream();
        try (PrintStream stream = new PrintStream(output, true, StandardCharsets.UTF_8)) {
            message.dump(stream, label + ": ");
        }
        rawLog.debug("{}", output.toString(StandardCharsets.UTF_8));
    }
}
