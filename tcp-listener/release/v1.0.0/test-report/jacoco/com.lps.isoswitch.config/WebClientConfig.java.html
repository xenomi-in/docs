package com.lps.isoswitch.config;

import io.netty.channel.ChannelOption;
import io.netty.handler.timeout.ReadTimeoutHandler;
import io.netty.handler.timeout.WriteTimeoutHandler;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.beans.factory.ObjectProvider;
import org.springframework.http.client.reactive.ReactorClientHttpConnector;
import org.springframework.web.reactive.function.client.WebClient;
import reactor.netty.http.client.HttpClient;
import reactor.netty.resources.ConnectionProvider;

import java.time.Duration;
import java.util.concurrent.TimeUnit;

/**
 * Configuration for WebClient with connection pooling.
 * 
 * @author Logicpoint Systems
 * @version 1.0
 * @since 2025-06-15
 */
@Configuration
public class WebClientConfig {

    private final TransportSecurityProperties transportSecurity;
    private final TransportTlsContextFactory tlsContextFactory;
    private final ObjectProvider<SwitchCoreOAuthExchangeFilter> switchCoreOAuthFilter;

    public WebClientConfig(
            TransportSecurityProperties transportSecurity,
            TransportTlsContextFactory tlsContextFactory,
            ObjectProvider<SwitchCoreOAuthExchangeFilter> switchCoreOAuthFilter) {
        this.transportSecurity = transportSecurity;
        this.tlsContextFactory = tlsContextFactory;
        this.switchCoreOAuthFilter = switchCoreOAuthFilter;
    }

    @Value("${webclient.connection.pool.max-connections:200}")
    private int maxConnections;

    @Value("${webclient.connection.pool.acquire-timeout:30}")
    private int acquireTimeout;

    @Value("${webclient.connection.pool.max-idle-time:60}")
    private int maxIdleTime;

    @Value("${webclient.timeout.read:30}")
    private int readTimeout;

    @Value("${webclient.timeout.write:30}")
    private int writeTimeout;

    @Value("${webclient.timeout.connect:10}")
    private int connectTimeout;

    @Bean
    public WebClient webClient() {
        // Configure connection provider with pooling
        ConnectionProvider connectionProvider = ConnectionProvider.builder("api-connection-pool")
            .maxConnections(maxConnections)
            .pendingAcquireTimeout(Duration.ofSeconds(acquireTimeout))
            .maxIdleTime(Duration.ofSeconds(maxIdleTime))
            .metrics(true) // Enable metrics for monitoring
            .build();

        // Configure HTTP client with timeouts
        HttpClient httpClient = HttpClient.create(connectionProvider)
            .option(ChannelOption.CONNECT_TIMEOUT_MILLIS, connectTimeout * 1000)
            .doOnConnected(conn -> 
                conn.addHandlerLast(new ReadTimeoutHandler(readTimeout, TimeUnit.SECONDS))
                    .addHandlerLast(new WriteTimeoutHandler(writeTimeout, TimeUnit.SECONDS))
            );

        if (transportSecurity.getSwitchCore().isEnabled()) {
            httpClient = httpClient.secure(ssl -> ssl
                    .sslContext(tlsContextFactory.createClientContext(
                            "Switch-core", transportSecurity.getSwitchCore()))
                    .handshakeTimeout(Duration.ofSeconds(connectTimeout)));
        }

        WebClient.Builder builder = WebClient.builder()
            .clientConnector(new ReactorClientHttpConnector(httpClient));
        switchCoreOAuthFilter.ifAvailable(builder::filter);
        return builder.build();
    }
}
