package com.lps.isoswitch.config;

import java.net.URI;

import org.springframework.beans.factory.annotation.Value;
import org.springframework.boot.ApplicationArguments;
import org.springframework.boot.ApplicationRunner;
import org.springframework.context.annotation.Profile;
import org.springframework.stereotype.Component;

/** Prevents a production listener from starting with plaintext payment paths. */
@Component
@Profile("prod")
public class ProductionTransportSecurityValidator implements ApplicationRunner {

    private final TransportSecurityProperties properties;
    private final TransportTlsContextFactory tlsContextFactory;
    private final String transactionUrl;
    private final String keyExchangeUrl;

    public ProductionTransportSecurityValidator(
            TransportSecurityProperties properties,
            TransportTlsContextFactory tlsContextFactory,
            @Value("${api.service.url}") String transactionUrl,
            @Value("${api.service.url.dke}") String keyExchangeUrl) {
        this.properties = properties;
        this.tlsContextFactory = tlsContextFactory;
        this.transactionUrl = transactionUrl;
        this.keyExchangeUrl = keyExchangeUrl;
    }

    @Override
    public void run(ApplicationArguments args) {
        requireHttps(transactionUrl, "Switch-core transaction URL");
        requireHttps(keyExchangeUrl, "Switch-core key-exchange URL");
        if (!properties.getSwitchCore().isEnabled()) {
            throw new IllegalStateException("Switch-core mTLS must be enabled in production");
        }
        if (!properties.getScheme().isEnabled()) {
            throw new IllegalStateException("Scheme TLS must be enabled in production");
        }
        // Read and parse both identities and trust stores during startup. This
        // prevents a production instance from becoming ready with unusable TLS.
        tlsContextFactory.createClientContext("Switch-core", properties.getSwitchCore());
        tlsContextFactory.createClientContext("Scheme", properties.getScheme());
    }

    private void requireHttps(String value, String name) {
        try {
            if (!"https".equalsIgnoreCase(URI.create(value).getScheme())) {
                throw new IllegalStateException(name + " must use HTTPS in production");
            }
        } catch (IllegalArgumentException exception) {
            throw new IllegalStateException(name + " is invalid", exception);
        }
    }
}
