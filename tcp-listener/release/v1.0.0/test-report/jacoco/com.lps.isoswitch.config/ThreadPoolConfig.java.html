package com.lps.isoswitch.config;

import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.scheduling.concurrent.ThreadPoolTaskExecutor;

import java.util.concurrent.Executor;
import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

import jakarta.annotation.PreDestroy;

/**
 * Configuration class for thread pool settings.
 * This class provides configuration for various thread pools
 * used in the application for concurrent processing.
 * 
 * Virtual threads are used for I/O-bound operations to improve scalability
 * and resource utilization, while traditional thread pools are maintained
 * for CPU-intensive operations.
 *
 * @author Logicpoint Systems
 * @version 2.0
 * @since 2025-06-15
 */
@Configuration
public class ThreadPoolConfig {

    private static final Logger log = LoggerFactory.getLogger(ThreadPoolConfig.class);
    private ExecutorService virtualThreadExecutor;
    private ExecutorService cpuIntensiveExecutor;
    private ExecutorService isoMessageExecutor;
    private ExecutorService tcpConnectionExecutor;
    private ExecutorService scheduledTaskExecutor;
    private ThreadPoolTaskExecutor legacyThreadPoolExecutor;

    /**
     * Virtual thread executor for I/O-bound operations.
     * Ideal for API calls, database operations, and network I/O.
     * 
     * @return Virtual thread executor service
     */
    @Bean(name = "virtualThreadExecutor")
    public ExecutorService virtualThreadExecutor() {
        this.virtualThreadExecutor = Executors.newVirtualThreadPerTaskExecutor();
        return this.virtualThreadExecutor;
    }

    /**
     * Traditional thread pool executor for CPU-intensive operations.
     * Used for message processing, data transformation, and calculations.
     * 
     * @return Traditional thread pool executor
     */
    @Bean(name = "cpuIntensiveExecutor")
    public ExecutorService cpuIntensiveExecutor() {
        this.cpuIntensiveExecutor = Executors.newFixedThreadPool(
            Runtime.getRuntime().availableProcessors(),
            r -> {
                Thread t = new Thread(r, "cpu-intensive-");
                t.setPriority(Thread.NORM_PRIORITY);
                return t;
            }
        );
        return this.cpuIntensiveExecutor;
    }

    /**
     * ISO message processing executor using virtual threads.
     * Optimized for I/O-bound ISO8583 message processing.
     * 
     * @return Virtual thread executor for ISO messages
     */
    @Bean(name = "isoMessageExecutor")
    public ExecutorService isoMessageExecutor() {
        this.isoMessageExecutor = Executors.newVirtualThreadPerTaskExecutor();
        return this.isoMessageExecutor;
    }

    /**
     * TCP connection management executor using virtual threads.
     * Handles connection establishment, maintenance, and cleanup.
     * 
     * @return Virtual thread executor for TCP operations
     */
    @Bean(name = "tcpConnectionExecutor")
    public ExecutorService tcpConnectionExecutor() {
        this.tcpConnectionExecutor = Executors.newVirtualThreadPerTaskExecutor();
        return this.tcpConnectionExecutor;
    }

    /**
     * Scheduled task executor using virtual threads.
     * For periodic tasks like health checks, monitoring, and cleanup.
     * 
     * @return Virtual thread scheduled executor
     */
    @Bean(name = "scheduledTaskExecutor")
    public ExecutorService scheduledTaskExecutor() {
        this.scheduledTaskExecutor = Executors.newVirtualThreadPerTaskExecutor();
        return this.scheduledTaskExecutor;
    }

    /**
     * Legacy thread pool executor for backward compatibility.
     * Maintained for components that haven't been migrated yet.
     * 
     * @return Traditional thread pool task executor
     */
    @Bean(name = "legacyThreadPoolExecutor")
    public Executor legacyThreadPoolExecutor() {
        this.legacyThreadPoolExecutor = new ThreadPoolTaskExecutor();
        this.legacyThreadPoolExecutor.setCorePoolSize(10);
        this.legacyThreadPoolExecutor.setMaxPoolSize(20);
        this.legacyThreadPoolExecutor.setQueueCapacity(500);
        this.legacyThreadPoolExecutor.setThreadNamePrefix("legacy-");
        this.legacyThreadPoolExecutor.initialize();
        return this.legacyThreadPoolExecutor;
    }

    /**
     * Shutdown hook for graceful cleanup of thread pools.
     */
    @PreDestroy
    public void shutdown() {
        log.info("Shutting down ThreadPoolConfig...");
        
        try {
            shutdownExecutor("virtualThreadExecutor", virtualThreadExecutor);
            shutdownExecutor("cpuIntensiveExecutor", cpuIntensiveExecutor);
            shutdownExecutor("isoMessageExecutor", isoMessageExecutor);
            shutdownExecutor("tcpConnectionExecutor", tcpConnectionExecutor);
            shutdownExecutor("scheduledTaskExecutor", scheduledTaskExecutor);
            shutdownLegacyExecutor();
            
            log.info("ThreadPoolConfig shutdown completed");
        } catch (Exception e) {
            log.error("Error during ThreadPoolConfig shutdown", e);
        }
    }

    /**
     * Shuts down a specific executor by name.
     *
     * @param executorName The name of the executor to shutdown
     */
    private void shutdownExecutor(String executorName, ExecutorService executorService) {
        try {
            if (executorService == null) {
                return;
            }
            log.info("Shutting down executor: {}", executorName);
            executorService.shutdown();
            if (!executorService.awaitTermination(5, java.util.concurrent.TimeUnit.SECONDS)) {
                log.warn("Executor {} did not terminate in time, forcing shutdown", executorName);
                executorService.shutdownNow();
            }
        } catch (Exception e) {
            log.error("Error shutting down executor: {}", executorName, e);
        }
    }

    private void shutdownLegacyExecutor() {
        if (legacyThreadPoolExecutor == null) {
            return;
        }
        log.info("Shutting down executor: legacyThreadPoolExecutor");
        legacyThreadPoolExecutor.shutdown();
    }
}
