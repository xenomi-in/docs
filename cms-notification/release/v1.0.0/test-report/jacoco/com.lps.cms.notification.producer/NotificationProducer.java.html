package com.lps.cms.notification.producer;

import com.cms.notification.avro.NotificationEvent;
import com.lps.cms.notification.config.NotificationProducerProperties;
import com.lps.cms.notification.config.NotificationTopicsProperties;
import com.lps.cms.notification.model.EventSource;

import io.github.resilience4j.bulkhead.Bulkhead;
import io.github.resilience4j.bulkhead.BulkheadRegistry;
import io.github.resilience4j.circuitbreaker.CircuitBreaker;
import io.github.resilience4j.circuitbreaker.CircuitBreakerRegistry;
import io.micrometer.core.instrument.MeterRegistry;
import io.micrometer.core.instrument.Timer;
import org.apache.kafka.clients.producer.RecordMetadata;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.kafka.core.KafkaTemplate;
import org.springframework.kafka.support.SendResult;
import org.springframework.stereotype.Service;

import java.time.Duration;
import java.util.concurrent.ExecutionException;
import java.util.concurrent.TimeUnit;
import java.util.concurrent.TimeoutException;
import java.util.function.Supplier;

/**
 * Synchronously publishes enriched notification events to Kafka using Avro serialization
 * and Schema Registry.
 * <p>
 * The target topic is resolved from the event's {@link EventSource}: each upstream system
 * (CMS, Switch) owns a dedicated topic so streams can be scaled, retained, and consumed
 * independently.
 * <p>
 * Each source is guarded independently so one upstream cannot degrade the other:
 * <ul>
 *   <li><b>Bulkhead</b> — caps concurrent in-flight publishes per source, protecting the
 *       servlet thread pool from being exhausted by a single upstream during a slowdown.</li>
 *   <li><b>Circuit breaker</b> — opens after sustained publish failures so callers get an
 *       immediate retryable 503 instead of holding a thread for the full delivery budget.
 *       Only {@link NotificationPublishException} is recorded as a failure; client errors
 *       (validation, oversized payloads) are raised before the guards and never trip it.</li>
 * </ul>
 */
@Service
public class NotificationProducer {

    private static final Logger log = LoggerFactory.getLogger(NotificationProducer.class);

    /** Prefix for per-source circuit breaker and bulkhead instance names. */
    static final String GUARD_NAME_PREFIX = "kafka-publish-";

    private static final String PUBLISH_TIMER = "notifications.publish";

    private final KafkaTemplate<String, NotificationEvent> kafkaTemplate;
    private final NotificationTopicsProperties topics;
    private final NotificationEventAvroMapper avroMapper;
    private final Duration publishAwaitTimeout;
    private final CircuitBreakerRegistry circuitBreakerRegistry;
    private final BulkheadRegistry bulkheadRegistry;
    private final MeterRegistry meterRegistry;

    public NotificationProducer(KafkaTemplate<String, NotificationEvent> kafkaTemplate,
                                NotificationTopicsProperties topics,
                                NotificationEventAvroMapper avroMapper,
                                NotificationProducerProperties producerProperties,
                                CircuitBreakerRegistry circuitBreakerRegistry,
                                BulkheadRegistry bulkheadRegistry,
                                MeterRegistry meterRegistry) {
        this.kafkaTemplate = kafkaTemplate;
        this.topics = topics;
        this.avroMapper = avroMapper;
        this.publishAwaitTimeout = producerProperties.publishAwaitTimeout();
        this.circuitBreakerRegistry = circuitBreakerRegistry;
        this.bulkheadRegistry = bulkheadRegistry;
        this.meterRegistry = meterRegistry;
    }

    /**
     * Maps the domain event to Avro, then sends it to the source-specific topic through the
     * source's bulkhead and circuit breaker, blocking until ack or timeout.
     * <p>
     * The await ({@code delivery.timeout.ms + awaitBuffer}) is intentionally longer than the producer's
     * own {@code delivery.timeout.ms}, so the future always resolves with the producer's authoritative
     * outcome before this wait elapses. This guarantees a failure is never reported while a record is
     * still in flight (which could otherwise be delivered later and produce a duplicate). Callers that
     * retry on failure must reuse the same {@code eventId}, which downstream consumers treat as the
     * idempotency key, since legitimate retries remain at-least-once.
     *
     * @param event enriched domain event to publish
     * @throws NotificationPublishException when the send is interrupted, fails, or times out
     * @throws io.github.resilience4j.circuitbreaker.CallNotPermittedException when the source's circuit is open
     * @throws io.github.resilience4j.bulkhead.BulkheadFullException when the source's concurrency cap is reached
     */
    public void publish(com.lps.cms.notification.model.NotificationEvent event) {
        EventSource source = event.eventSource();
        String topic = topics.eventsTopic(source);
        String partitionKey = NotificationPartitionKey.resolve(event);
        // Mapping (and payload validation) happens before the resilience guards so
        // client errors surface as 4xx and are never recorded as publish failures.
        NotificationEvent avroEvent = avroMapper.toAvro(event);

        String guardName = GUARD_NAME_PREFIX + source.name();
        CircuitBreaker circuitBreaker = circuitBreakerRegistry.circuitBreaker(guardName);
        Bulkhead bulkhead = bulkheadRegistry.bulkhead(guardName);

        if (log.isDebugEnabled()) {
            log.debug("Publishing to Kafka eventId={} topic={} partitionKey={} awaitTimeout={}",
                    event.eventId(), topic, partitionKey, publishAwaitTimeout);
        }

        Supplier<RecordMetadata> guardedSend = Bulkhead.decorateSupplier(bulkhead,
                CircuitBreaker.decorateSupplier(circuitBreaker,
                        () -> send(topic, partitionKey, avroEvent, event.eventId())));

        Timer.Sample sample = Timer.start(meterRegistry);
        try {
            RecordMetadata metadata = guardedSend.get();
            sample.stop(publishTimer(source, "success"));
            log.info("Published notification eventId={} topic={} partition={} offset={}",
                    event.eventId(), metadata.topic(), metadata.partition(), metadata.offset());
        } catch (RuntimeException ex) {
            sample.stop(publishTimer(source, "failure"));
            throw ex;
        }
    }

    private RecordMetadata send(String topic, String partitionKey, NotificationEvent avroEvent, String eventId) {
        try {
            SendResult<String, NotificationEvent> result = kafkaTemplate.send(topic, partitionKey, avroEvent)
                    .get(publishAwaitTimeout.toMillis(), TimeUnit.MILLISECONDS);
            return result.getRecordMetadata();
        } catch (InterruptedException ex) {
            Thread.currentThread().interrupt();
            log.warn("Kafka publish interrupted eventId={} topic={}", eventId, topic);
            throw new NotificationPublishException("Interrupted while publishing notification event", ex);
        } catch (ExecutionException ex) {
            log.error("Kafka publish failed eventId={} topic={}", eventId, topic, ex.getCause());
            throw new NotificationPublishException("Failed to publish notification event", ex.getCause());
        } catch (TimeoutException ex) {
            log.warn("Kafka publish await exceeded eventId={} topic={} awaitTimeout={}",
                    eventId, topic, publishAwaitTimeout);
            throw new NotificationPublishException("Timed out while publishing notification event", ex);
        }
    }

    private Timer publishTimer(EventSource source, String outcome) {
        return meterRegistry.timer(PUBLISH_TIMER, "source", source.name(), "outcome", outcome);
    }
}
