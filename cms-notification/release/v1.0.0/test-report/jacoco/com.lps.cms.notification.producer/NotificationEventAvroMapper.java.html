package com.lps.cms.notification.producer;

import com.cms.notification.avro.NotificationEvent;
import com.cms.notification.avro.NotificationMetadata;
import com.cms.notification.avro.NotificationPriority;
import tools.jackson.core.JacksonException;
import tools.jackson.databind.json.JsonMapper;
import com.lps.cms.notification.config.NotificationProducerProperties;
import com.lps.cms.notification.exception.PayloadTooLargeException;
import com.lps.cms.notification.exception.ValidationException;
import org.springframework.stereotype.Component;

import java.nio.charset.StandardCharsets;

/**
 * Maps the internal domain {@link com.lps.cms.notification.model.NotificationEvent}
 * to the Avro-generated type registered in Schema Registry.
 * <p>
 * Serializes {@code payload} to a JSON string for the Avro {@code string} field and
 * enforces the configured maximum payload size so oversized events are rejected as
 * client errors before they reach the Kafka producer.
 */
@Component
public class NotificationEventAvroMapper {

    private final JsonMapper jsonMapper;
    private final long maxPayloadBytes;

    public NotificationEventAvroMapper(JsonMapper jsonMapper,
                                       NotificationProducerProperties producerProperties) {
        this.jsonMapper = jsonMapper;
        this.maxPayloadBytes = producerProperties.maxPayloadSize().toBytes();
    }

    /**
     * Converts a domain event to its Avro representation for Kafka produce.
     *
     * @param event enriched domain event
     * @return Avro {@link NotificationEvent} ready for serialization
     * @throws ValidationException      if {@code payload} cannot be serialized to JSON
     * @throws PayloadTooLargeException if the serialized payload exceeds the configured maximum
     */
    public NotificationEvent toAvro(com.lps.cms.notification.model.NotificationEvent event) {
        String payloadJson = serializePayload(event);

        NotificationMetadata metadata = NotificationMetadata.newBuilder()
                .setTenantId(event.metadata().tenantId())
                .setTraceId(event.metadata().traceId())
                .setRetryCount(event.metadata().retryCount())
                .build();

        return NotificationEvent.newBuilder()
                .setEventId(event.eventId())
                .setEventType(event.eventType())
                .setEventSource(event.eventSource().name())
                .setEventTime(event.eventTime().toString())
                .setVersion(event.version())
                .setTenantId(event.metadata().tenantId())
                .setAggregateType(event.aggregateType())
                .setAggregateId(event.aggregateId())
                .setCustomerId(event.customerId())
                .setCorrelationId(event.correlationId())
                .setChannels(event.channels())
                .setPriority(NotificationPriority.valueOf(event.priority().name()))
                .setPayload(payloadJson)
                .setMetadata(metadata)
                .build();
    }

    private String serializePayload(com.lps.cms.notification.model.NotificationEvent event) {
        String payloadJson;
        try {
            payloadJson = jsonMapper.writeValueAsString(event.payload());
        } catch (JacksonException ex) {
            throw new ValidationException("payload must be JSON-serializable", ex);
        }

        long payloadBytes = payloadJson.getBytes(StandardCharsets.UTF_8).length;
        if (payloadBytes > maxPayloadBytes) {
            throw new PayloadTooLargeException(
                    "payload is " + payloadBytes + " bytes; maximum allowed is " + maxPayloadBytes + " bytes");
        }
        return payloadJson;
    }
}
