package com.lps.cms.notification.producer;

import org.springframework.util.StringUtils;

import com.lps.cms.notification.model.NotificationEvent;

/**
 * Resolves Kafka record keys for the per-source event topics and their aligned retry/DLQ topics.
 * <p>
 * {@code tenantId} is mandatory on every event, so every key is tenant-prefixed and the same
 * event content always yields the same key regardless of which producer sent it. Resolution order:
 * <ol>
 *   <li>{@code tenantId:customerId} — per-customer ordering within a tenant</li>
 *   <li>{@code tenantId:aggregateType:aggregateId} — per-aggregate ordering when no customer is set</li>
 *   <li>{@code tenantId} — tenant-level ordering as the fallback</li>
 * </ol>
 */
public final class NotificationPartitionKey {

    /** Separator between key segments in composite keys. */
    static final String KEY_SEPARATOR = ":";

    private NotificationPartitionKey() {
    }

    /**
     * Derives the Kafka message key for the given event.
     *
     * @param event enriched notification event (tenant always present in metadata)
     * @return deterministic, tenant-prefixed partition key (never blank)
     */
    public static String resolve(NotificationEvent event) {
        String tenantId = event.metadata().tenantId();
        if (StringUtils.hasText(event.customerId())) {
            return tenantId + KEY_SEPARATOR + event.customerId();
        }
        if (StringUtils.hasText(event.aggregateType()) && StringUtils.hasText(event.aggregateId())) {
            return tenantId + KEY_SEPARATOR + event.aggregateType() + KEY_SEPARATOR + event.aggregateId();
        }
        return tenantId;
    }
}
