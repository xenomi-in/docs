package com.lps.cms.notification.model;

import org.springframework.util.StringUtils;

import java.util.Arrays;
import java.util.Locale;
import java.util.Optional;

/**
 * Upstream systems allowed to publish notification events.
 * <p>
 * The source is a first-class routing dimension: each source publishes to its own
 * Kafka topic set (see {@code app.notification.topics.sources.*}). Adding a new
 * upstream requires a new constant here plus its topic configuration.
 */
public enum EventSource {

    /** cms-core-service events. */
    CMS,

    /** Switch (transaction switch) events. */
    SWITCH;

    /**
     * Parses a client-supplied source, tolerating case and surrounding whitespace.
     *
     * @param value raw {@code eventSource} from the request; may be null or blank
     * @return the matching source, or empty when the value is blank or unknown
     */
    public static Optional<EventSource> fromValue(String value) {
        if (!StringUtils.hasText(value)) {
            return Optional.empty();
        }
        String normalized = value.trim().toUpperCase(Locale.ROOT);
        return Arrays.stream(values())
                .filter(source -> source.name().equals(normalized))
                .findFirst();
    }
}
