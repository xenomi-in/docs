package com.lps.cms.notification.model;

import java.time.Instant;
import java.util.List;
import java.util.Map;

/**
 * Canonical notification event produced by this service and published to Kafka.
 * <p>
 * Populated by {@link com.lps.cms.notification.service.MetadataEnricher} after validation.
 *
 * @param eventId        unique event identifier
 * @param eventType      business event name
 * @param eventSource    upstream system; determines the Kafka topic the event is routed to
 * @param eventTime      server timestamp at enrichment
 * @param version        schema/event contract version
 * @param aggregateType  optional aggregate type for routing
 * @param aggregateId    optional aggregate id for routing
 * @param customerId     optional customer id for routing
 * @param correlationId  optional cross-service correlation id
 * @param channels       normalized delivery channel list (may be empty)
 * @param priority       delivery priority
 * @param payload        event body map
 * @param metadata       service-owned metadata (trace, tenant, retry count)
 */
public record NotificationEvent(
        String eventId,
        String eventType,
        EventSource eventSource,
        Instant eventTime,
        String version,
        String aggregateType,
        String aggregateId,
        String customerId,
        String correlationId,
        List<String> channels,
        NotificationPriority priority,
        Map<String, Object> payload,
        NotificationMetadata metadata
) {
}
