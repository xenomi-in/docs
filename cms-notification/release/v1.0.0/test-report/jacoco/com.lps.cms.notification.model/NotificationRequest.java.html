package com.lps.cms.notification.model;

import jakarta.validation.constraints.NotBlank;
import jakarta.validation.constraints.NotNull;
import jakarta.validation.constraints.Size;

import java.util.List;
import java.util.Map;

/**
 * Incoming notification intent from an upstream system via the REST API.
 * <p>
 * Bean Validation covers structural constraints; routing rules are enforced in
 * {@link com.lps.cms.notification.service.EventValidator}.
 *
 * @param eventId        required caller-supplied idempotency key, derived from the business fact
 *                       (e.g. transaction id + event type); duplicate submissions must reuse it
 * @param eventType      required event name (e.g. {@code TRANSACTION_APPROVED})
 * @param tenantId       required tenant/institution scope for multi-tenant routing
 * @param aggregateType  entity type when using aggregate-based routing (paired with {@code aggregateId})
 * @param aggregateId    entity id when using aggregate-based routing
 * @param customerId     customer scope for per-user ordering
 * @param eventSource    required originating system ({@code CMS} or {@code SWITCH}); determines the Kafka topic
 * @param correlationId  optional id linking related events across services
 * @param channels       delivery channels suggested to downstream consumers (max 10)
 * @param priority       delivery priority; defaults to {@link NotificationPriority#MEDIUM}
 * @param payload        required JSON-compatible event body
 */
public record NotificationRequest(
        @NotBlank String eventId,
        @NotBlank String eventType,
        @NotBlank String tenantId,
        String aggregateType,
        String aggregateId,
        String customerId,
        @NotBlank String eventSource,
        String correlationId,
        @Size(max = 10) List<@NotBlank String> channels,
        NotificationPriority priority,
        @NotNull Map<String, Object> payload
) {
}
