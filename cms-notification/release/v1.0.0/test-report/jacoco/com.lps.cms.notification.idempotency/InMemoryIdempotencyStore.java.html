package com.lps.cms.notification.idempotency;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

import java.time.Clock;
import java.time.Duration;
import java.time.Instant;
import java.util.Optional;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.ConcurrentMap;

/**
 * Process-local idempotency store for tests and single-instance local runs.
 * <p>
 * Not safe for multi-replica production: each JVM has its own map. Prefer
 * {@link RedisIdempotencyStore} whenever more than one instance may serve HTTP retries.
 */
public class InMemoryIdempotencyStore implements IdempotencyStore {

    private static final Logger log = LoggerFactory.getLogger(InMemoryIdempotencyStore.class);

    private final ConcurrentMap<String, TimedRecord> entries = new ConcurrentHashMap<>();
    private final Duration ttl;
    private final Clock clock;

    public InMemoryIdempotencyStore(Duration ttl) {
        this(ttl, Clock.systemUTC());
    }

    InMemoryIdempotencyStore(Duration ttl, Clock clock) {
        this.ttl = ttl;
        this.clock = clock;
    }

    @Override
    public Optional<IdempotencyRecord> find(IdempotencyKey key) {
        purgeIfExpired(key.asStorageKey());
        TimedRecord timed = entries.get(key.asStorageKey());
        return Optional.ofNullable(timed).map(TimedRecord::record);
    }

    @Override
    public boolean tryBegin(IdempotencyKey key) {
        purgeIfExpired(key.asStorageKey());
        Instant expiresAt = clock.instant().plus(ttl);
        IdempotencyRecord inProgress = new IdempotencyRecord(IdempotencyStatus.IN_PROGRESS, key.eventId());
        TimedRecord previous = entries.putIfAbsent(key.asStorageKey(), new TimedRecord(inProgress, expiresAt));
        boolean claimed = previous == null;
        if (claimed && log.isDebugEnabled()) {
            log.debug("Idempotency claim acquired key={}", key.asStorageKey());
        }
        return claimed;
    }

    @Override
    public void complete(IdempotencyKey key) {
        Instant expiresAt = clock.instant().plus(ttl);
        entries.put(key.asStorageKey(), new TimedRecord(
                new IdempotencyRecord(IdempotencyStatus.COMPLETED, key.eventId()),
                expiresAt));
        if (log.isDebugEnabled()) {
            log.debug("Idempotency claim completed key={}", key.asStorageKey());
        }
    }

    @Override
    public void abandon(IdempotencyKey key) {
        entries.computeIfPresent(key.asStorageKey(), (k, existing) -> {
            if (existing.record().status() == IdempotencyStatus.COMPLETED) {
                return existing;
            }
            return null;
        });
        if (log.isDebugEnabled()) {
            log.debug("Idempotency claim abandoned key={}", key.asStorageKey());
        }
    }

    private void purgeIfExpired(String storageKey) {
        TimedRecord timed = entries.get(storageKey);
        if (timed != null && timed.expiresAt().isBefore(clock.instant())) {
            entries.remove(storageKey, timed);
        }
    }

    private record TimedRecord(IdempotencyRecord record, Instant expiresAt) {
    }
}
