package com.lps.cms.notification.idempotency;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.data.redis.core.StringRedisTemplate;
import org.springframework.data.redis.core.script.DefaultRedisScript;

import java.time.Duration;
import java.util.List;
import java.util.Optional;

/**
 * Redis-backed idempotency store safe for multi-instance deployments.
 * <p>
 * Uses {@code SET key value NX EX} for atomic claim. Values are compact:
 * {@code IN_PROGRESS:{eventId}} / {@code COMPLETED:{eventId}}.
 * Abandon deletes only when the key is still in-progress (Lua) so a race cannot
 * wipe a completed marker after a successful publish.
 */
public class RedisIdempotencyStore implements IdempotencyStore {

    private static final Logger log = LoggerFactory.getLogger(RedisIdempotencyStore.class);

    private static final String IN_PROGRESS_PREFIX = "IN_PROGRESS:";
    private static final String COMPLETED_PREFIX = "COMPLETED:";

    /**
     * Delete only if value is still IN_PROGRESS for this event (avoid clobbering COMPLETED).
     */
    private static final DefaultRedisScript<Long> ABANDON_SCRIPT = new DefaultRedisScript<>(
            """
                    local current = redis.call('GET', KEYS[1])
                    if current == false then
                      return 0
                    end
                    if string.sub(current, 1, 12) == 'IN_PROGRESS:' then
                      return redis.call('DEL', KEYS[1])
                    end
                    return 0
                    """,
            Long.class
    );

    private final StringRedisTemplate redisTemplate;
    private final String keyPrefix;
    private final Duration ttl;

    public RedisIdempotencyStore(StringRedisTemplate redisTemplate, String keyPrefix, Duration ttl) {
        this.redisTemplate = redisTemplate;
        this.keyPrefix = keyPrefix.endsWith(":") ? keyPrefix : keyPrefix + ":";
        this.ttl = ttl;
    }

    @Override
    public Optional<IdempotencyRecord> find(IdempotencyKey key) {
        String value = redisTemplate.opsForValue().get(redisKey(key));
        return Optional.ofNullable(value).flatMap(this::parse);
    }

    @Override
    public boolean tryBegin(IdempotencyKey key) {
        Boolean claimed = redisTemplate.opsForValue().setIfAbsent(
                redisKey(key),
                IN_PROGRESS_PREFIX + key.eventId(),
                ttl
        );
        boolean acquired = Boolean.TRUE.equals(claimed);
        if (acquired && log.isDebugEnabled()) {
            log.debug("Idempotency claim acquired key={}", key.asStorageKey());
        }
        return acquired;
    }

    @Override
    public void complete(IdempotencyKey key) {
        redisTemplate.opsForValue().set(
                redisKey(key),
                COMPLETED_PREFIX + key.eventId(),
                ttl
        );
        if (log.isDebugEnabled()) {
            log.debug("Idempotency claim completed key={}", key.asStorageKey());
        }
    }

    @Override
    public void abandon(IdempotencyKey key) {
        Long deleted = redisTemplate.execute(ABANDON_SCRIPT, List.of(redisKey(key)));
        if (log.isDebugEnabled()) {
            log.debug("Idempotency claim abandon key={} deleted={}", key.asStorageKey(), deleted);
        }
    }

    private String redisKey(IdempotencyKey key) {
        return keyPrefix + key.asStorageKey();
    }

    private Optional<IdempotencyRecord> parse(String value) {
        if (value.startsWith(COMPLETED_PREFIX)) {
            return Optional.of(new IdempotencyRecord(
                    IdempotencyStatus.COMPLETED,
                    value.substring(COMPLETED_PREFIX.length())));
        }
        if (value.startsWith(IN_PROGRESS_PREFIX)) {
            return Optional.of(new IdempotencyRecord(
                    IdempotencyStatus.IN_PROGRESS,
                    value.substring(IN_PROGRESS_PREFIX.length())));
        }
        log.warn("Ignoring unrecognized idempotency value: {}", value);
        return Optional.empty();
    }
}
