package com.lps.cms.notification.idempotency;

import jakarta.validation.constraints.NotBlank;
import jakarta.validation.constraints.NotNull;
import org.springframework.boot.context.properties.ConfigurationProperties;
import org.springframework.validation.annotation.Validated;

import java.time.Duration;

/**
 * Idempotency cache tuning bound from {@code app.notification.idempotency.*}.
 *
 * @param store             backend: {@code redis} (shared, prod) or {@code memory} (single JVM / tests)
 * @param keyPrefix         Redis key namespace prefix
 * @param ttl               how long completed (and in-progress) claims are retained — must cover client retries
 * @param inProgressWait    max time a concurrent duplicate waits for the owner to complete
 * @param inProgressPoll    poll interval while waiting on {@link IdempotencyStatus#IN_PROGRESS}
 */
@Validated
@ConfigurationProperties(prefix = "app.notification.idempotency")
public record IdempotencyProperties(
        @NotNull
        StoreType store,
        @NotBlank
        String keyPrefix,
        @NotNull
        Duration ttl,
        @NotNull
        Duration inProgressWait,
        @NotNull
        Duration inProgressPoll
) {

    public enum StoreType {
        REDIS,
        MEMORY
    }
}
