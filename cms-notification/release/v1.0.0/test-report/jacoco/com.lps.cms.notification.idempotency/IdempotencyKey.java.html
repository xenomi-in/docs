package com.lps.cms.notification.idempotency;

import com.lps.cms.notification.model.EventSource;

/**
 * Business key that uniquely identifies a notification publish intent for HTTP retries.
 * <p>
 * {@code institutionId} comes from the gateway {@code X-Institution-Id} header (currently
 * required to match body {@code tenantId}). Combined with source and caller-supplied
 * {@code eventId}, two identical submits resolve to the same key.
 */
public record IdempotencyKey(
        String institutionId,
        EventSource source,
        String eventId
) {

    public IdempotencyKey {
        if (institutionId == null || institutionId.isBlank()) {
            throw new IllegalArgumentException("institutionId must not be blank");
        }
        if (source == null) {
            throw new IllegalArgumentException("source must not be null");
        }
        if (eventId == null || eventId.isBlank()) {
            throw new IllegalArgumentException("eventId must not be blank");
        }
        institutionId = institutionId.trim();
        eventId = eventId.trim();
    }

    /**
     * Stable Redis / map key segment: {@code {institution}:{source}:{eventId}}.
     */
    public String asStorageKey() {
        return institutionId + ':' + source.name() + ':' + eventId;
    }
}
