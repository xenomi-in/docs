package com.lps.cms.notification.controller;

import jakarta.validation.Valid;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RequestHeader;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RestController;

import com.lps.cms.notification.config.InstitutionHeaders;
import com.lps.cms.notification.model.NotificationRequest;
import com.lps.cms.notification.service.NotificationService;
import com.lps.cms.notification.service.PublishResult;

/**
 * REST ingress for upstream systems (CMS, Switch, and others) to submit notification events.
 * <p>
 * Expects {@code cms-api-gateway} to stamp {@link InstitutionHeaders#INSTITUTION_ID} on every call.
 */
@RestController
@RequestMapping("/notifications")
public class NotificationController {

    private static final Logger log = LoggerFactory.getLogger(NotificationController.class);

    /** Set on idempotent replays so callers/ops can distinguish first accept from retry. */
    public static final String IDEMPOTENT_REPLAY_HEADER = "X-Idempotent-Replay";

    private final NotificationService notificationService;

    public NotificationController(NotificationService notificationService) {
        this.notificationService = notificationService;
    }

    /**
     * Accepts a notification intent, publishes it to Kafka (once per idempotency key), and
     * returns {@code 202 Accepted}. The event id is returned in {@code X-Event-Id}.
     *
     * @param institutionId gateway institution id; must match body {@code tenantId}
     * @param request       validated notification payload from the caller
     * @return empty body with {@code 202} and {@code X-Event-Id} on success
     */
    @PostMapping("/publish")
    public ResponseEntity<Void> publish(
            @RequestHeader(InstitutionHeaders.INSTITUTION_ID) String institutionId,
            @RequestBody @Valid NotificationRequest request) {
        PublishResult result = notificationService.handle(institutionId, request);
        log.info("Accepted notification eventId={} eventType={} eventSource={} replay={}",
                result.eventId(), request.eventType(), request.eventSource(), result.replay());

        ResponseEntity.BodyBuilder response = ResponseEntity.accepted()
                .header("X-Event-Id", result.eventId());
        if (result.replay()) {
            response.header(IDEMPOTENT_REPLAY_HEADER, "true");
        }
        return response.build();
    }
}
