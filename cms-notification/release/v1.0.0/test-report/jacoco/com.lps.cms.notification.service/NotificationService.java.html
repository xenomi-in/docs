package com.lps.cms.notification.service;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.slf4j.MDC;
import org.springframework.stereotype.Service;

import com.lps.cms.notification.config.InstitutionHeaders;
import com.lps.cms.notification.config.TraceIdFilter;
import com.lps.cms.notification.exception.IdempotencyConflictException;
import com.lps.cms.notification.idempotency.IdempotencyKey;
import com.lps.cms.notification.idempotency.IdempotencyProperties;
import com.lps.cms.notification.idempotency.IdempotencyRecord;
import com.lps.cms.notification.idempotency.IdempotencyStatus;
import com.lps.cms.notification.idempotency.IdempotencyStore;
import com.lps.cms.notification.model.NotificationEvent;
import com.lps.cms.notification.model.NotificationRequest;
import com.lps.cms.notification.producer.NotificationProducer;

import java.time.Duration;
import java.util.Optional;

/**
 * Orchestrates the ingest pipeline: validate, enrich, idempotency gate, publish to Kafka.
 * <p>
 * HTTP retries with the same {@code (institutionId, eventSource, eventId)} return the original
 * acceptance without a second Kafka produce. Kafka producer idempotence alone does not provide
 * that guarantee.
 */
@Service
public class NotificationService {

    private static final Logger log = LoggerFactory.getLogger(NotificationService.class);

    private final EventValidator validator;
    private final MetadataEnricher enricher;
    private final NotificationProducer producer;
    private final IdempotencyStore idempotencyStore;
    private final Duration inProgressWait;
    private final Duration inProgressPoll;

    public NotificationService(EventValidator validator,
                               MetadataEnricher enricher,
                               NotificationProducer producer,
                               IdempotencyStore idempotencyStore,
                               IdempotencyProperties idempotencyProperties) {
        this.validator = validator;
        this.enricher = enricher;
        this.producer = producer;
        this.idempotencyStore = idempotencyStore;
        this.inProgressWait = idempotencyProperties.inProgressWait();
        this.inProgressPoll = idempotencyProperties.inProgressPoll();
    }

    /**
     * Processes a single publish request end-to-end under the institution-scoped idempotency key.
     *
     * @param institutionId gateway {@code X-Institution-Id} (must match body {@code tenantId})
     * @param request       incoming notification intent
     * @return event id and whether this call was an idempotent replay
     */
    public PublishResult handle(String institutionId, NotificationRequest request) {
        String normalizedInstitutionId = institutionId.trim();
        MDC.put(InstitutionHeaders.INSTITUTION_ID_MDC_KEY, normalizedInstitutionId);

        if (log.isDebugEnabled()) {
            log.debug("Handling publish eventType={} institutionId={}",
                    request.eventType(), normalizedInstitutionId);
        }

        validator.validate(normalizedInstitutionId, request);
        NotificationEvent event = enricher.enrich(request);
        MDC.put(TraceIdFilter.EVENT_ID_MDC_KEY, event.eventId());

        IdempotencyKey key = new IdempotencyKey(
                normalizedInstitutionId,
                event.eventSource(),
                event.eventId());

        Optional<PublishResult> earlyReplay = completedReplay(key);
        if (earlyReplay.isPresent()) {
            return earlyReplay.get();
        }

        long deadlineNanos = System.nanoTime() + inProgressWait.toNanos();
        while (!idempotencyStore.tryBegin(key)) {
            Optional<PublishResult> replay = completedReplay(key);
            if (replay.isPresent()) {
                return replay.get();
            }
            if (System.nanoTime() >= deadlineNanos) {
                throw new IdempotencyConflictException(
                        "Another request is publishing this event; please retry with the same eventId");
            }
            sleep(inProgressPoll);
        }

        try {
            producer.publish(event);
            idempotencyStore.complete(key);
            return PublishResult.accepted(event.eventId());
        } catch (RuntimeException ex) {
            idempotencyStore.abandon(key);
            throw ex;
        }
    }

    private Optional<PublishResult> completedReplay(IdempotencyKey key) {
        Optional<IdempotencyRecord> record = idempotencyStore.find(key);
        if (record.isEmpty() || record.get().status() != IdempotencyStatus.COMPLETED) {
            return Optional.empty();
        }
        log.info("Idempotent replay eventId={} institutionId={} source={}",
                record.get().eventId(), key.institutionId(), key.source());
        return Optional.of(PublishResult.replay(record.get().eventId()));
    }

    private static void sleep(Duration duration) {
        try {
            Thread.sleep(Math.max(1L, duration.toMillis()));
        } catch (InterruptedException ex) {
            Thread.currentThread().interrupt();
            throw new IdempotencyConflictException(
                    "Interrupted while waiting for an in-flight publish of this event");
        }
    }
}
