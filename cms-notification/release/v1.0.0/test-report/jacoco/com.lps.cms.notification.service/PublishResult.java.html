package com.lps.cms.notification.service;

/**
 * Outcome of a publish request after the idempotency gate.
 *
 * @param eventId caller-supplied event id
 * @param replay  {@code true} when this HTTP call reused a prior successful publish (no Kafka send)
 */
public record PublishResult(String eventId, boolean replay) {

    public static PublishResult accepted(String eventId) {
        return new PublishResult(eventId, false);
    }

    public static PublishResult replay(String eventId) {
        return new PublishResult(eventId, true);
    }
}
