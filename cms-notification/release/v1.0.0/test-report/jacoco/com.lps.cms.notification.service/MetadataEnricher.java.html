package com.lps.cms.notification.service;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.slf4j.MDC;
import org.springframework.stereotype.Component;
import org.springframework.util.StringUtils;

import com.lps.cms.notification.config.TraceIdFilter;
import com.lps.cms.notification.exception.ValidationException;
import com.lps.cms.notification.model.EventSource;
import com.lps.cms.notification.model.NotificationEvent;
import com.lps.cms.notification.model.NotificationMetadata;
import com.lps.cms.notification.model.NotificationPriority;
import com.lps.cms.notification.model.NotificationRequest;

import java.time.Instant;
import java.util.ArrayList;
import java.util.List;
import java.util.UUID;

/**
 * Builds a canonical {@link com.lps.cms.notification.model.NotificationEvent} from an API request
 * by applying defaults, timestamps, trace correlation, and service-owned metadata.
 * <p>
 * {@code eventId}, {@code eventSource}, and {@code tenantId} are caller-supplied and mandatory
 * (enforced upstream); this enricher never invents identity or routing values.
 */
@Component
public class MetadataEnricher {

    private static final Logger log = LoggerFactory.getLogger(MetadataEnricher.class);

    private static final String EVENT_VERSION = "1.0";

    /**
     * Enriches a validated request into a publish-ready domain event.
     * <p>
     * Generates {@code traceId} when absent from MDC; resolves {@code eventSource} to an
     * {@link EventSource}; sets {@code retryCount} to zero for first publish.
     *
     * @param req validated notification request
     * @return fully populated event ready for Avro mapping and Kafka publish
     */
    public NotificationEvent enrich(NotificationRequest req) {
        String traceId = MDC.get(TraceIdFilter.TRACE_ID_MDC_KEY);
        if (!StringUtils.hasText(traceId)) {
            traceId = UUID.randomUUID().toString();
            MDC.put(TraceIdFilter.TRACE_ID_MDC_KEY, traceId);
            log.debug("Generated traceId for request without {}", TraceIdFilter.TRACE_ID_HEADER);
        }

        if (!StringUtils.hasText(req.eventId())) {
            log.debug("Generated eventId for request without client-supplied eventId");
        }
        EventSource eventSource = EventSource.fromValue(req.eventSource())
                .orElseThrow(() -> new ValidationException("eventSource must be a known source"));

        List<String> channels = req.channels() == null
                ? List.of()
                : req.channels().stream()
                .map(String::trim)
                .toList();
        NotificationPriority priority = req.priority() == null ? NotificationPriority.MEDIUM : req.priority();

        return new NotificationEvent(
                req.eventId().trim(),
                req.eventType().trim(),
                eventSource,
                Instant.now(),
                EVENT_VERSION,
                trimToNull(req.aggregateType()),
                trimToNull(req.aggregateId()),
                trimToNull(req.customerId()),
                trimToNull(req.correlationId()),
                channels,
                priority,
                req.payload(),
                new NotificationMetadata(req.tenantId().trim(), traceId, 0)
        );
    }

    private static String trimToNull(String value) {
        if (!StringUtils.hasText(value)) {
            return null;
        }
        return value.trim();
    }

}
