package com.lps.cms.notification.service;

import org.springframework.stereotype.Component;
import org.springframework.util.StringUtils;

import com.lps.cms.notification.exception.ValidationException;
import com.lps.cms.notification.model.EventSource;
import com.lps.cms.notification.model.NotificationRequest;

import java.util.Arrays;
import java.util.stream.Collectors;

import com.lps.cms.notification.config.InstitutionHeaders;

/**
 * Validates business rules beyond Bean Validation annotations on {@link NotificationRequest}.
 * <p>
 * Ensures {@code eventSource} resolves to a routable {@link EventSource}, aggregate identifiers
 * are consistent, and body {@code tenantId} matches the gateway institution id.
 */
@Component
public class EventValidator {

    /**
     * Validates the request; throws {@link ValidationException} with a descriptive message on failure.
     *
     * @param institutionId gateway institution id (already trimmed)
     * @param request       notification intent to validate
     * @throws ValidationException when a required field or combination rule is violated
     */
    public void validate(String institutionId, NotificationRequest request) {
        if (EventSource.fromValue(request.eventSource()).isEmpty()) {
            String allowed = Arrays.stream(EventSource.values())
                    .map(Enum::name)
                    .collect(Collectors.joining(", "));
            throw new ValidationException("eventSource must be one of: " + allowed);
        }

        String tenantId = request.tenantId() == null ? "" : request.tenantId().trim();
        if (!institutionId.equals(tenantId)) {
            throw new ValidationException(
                    "tenantId must match " + InstitutionHeaders.INSTITUTION_ID
                            + " header (institution and tenant are currently the same identity)");
        }

        boolean hasAggregateType = StringUtils.hasText(request.aggregateType());
        boolean hasAggregateId = StringUtils.hasText(request.aggregateId());

        if (hasAggregateType ^ hasAggregateId) {
            throw new ValidationException("aggregateType and aggregateId must be provided together");
        }
    }
}
