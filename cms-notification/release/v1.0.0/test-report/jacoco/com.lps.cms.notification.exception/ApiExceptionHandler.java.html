package com.lps.cms.notification.exception;

import io.github.resilience4j.bulkhead.BulkheadFullException;
import io.github.resilience4j.circuitbreaker.CallNotPermittedException;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.slf4j.MDC;
import org.springframework.http.HttpHeaders;
import org.springframework.http.HttpStatus;
import org.springframework.http.ProblemDetail;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.MissingRequestHeaderException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;

import com.lps.cms.notification.config.TraceIdFilter;
import com.lps.cms.notification.producer.NotificationPublishException;

/**
 * Maps domain and infrastructure failures to RFC 9457 {@link ProblemDetail} responses
 * with consistent HTTP status codes for API clients.
 * <p>
 * Retryable infrastructure failures (publish errors, open circuit, saturated bulkhead)
 * carry a {@code Retry-After} header so upstream retry behavior is part of the contract.
 */
@RestControllerAdvice
public class ApiExceptionHandler {

    private static final Logger log = LoggerFactory.getLogger(ApiExceptionHandler.class);

    /** Suggested wait after a publish failure or timeout. */
    private static final String RETRY_AFTER_PUBLISH_FAILURE_SECONDS = "5";
    /** Suggested wait while a circuit is open; aligned with waitDurationInOpenState. */
    private static final String RETRY_AFTER_CIRCUIT_OPEN_SECONDS = "10";
    /** Suggested wait when a source's concurrency cap is reached; contention is transient. */
    private static final String RETRY_AFTER_BULKHEAD_FULL_SECONDS = "1";
    /** Suggested wait when another in-flight publish holds the idempotency claim. */
    private static final String RETRY_AFTER_IDEMPOTENCY_CONFLICT_SECONDS = "1";

    /** Business-rule validation failures from {@link com.lps.cms.notification.service.EventValidator}. */
    @ExceptionHandler(ValidationException.class)
    public ProblemDetail handleValidationFailure(ValidationException ex) {
        log.warn("Bad request: {}", ex.getMessage());
        return ProblemDetail.forStatusAndDetail(HttpStatus.BAD_REQUEST, ex.getMessage());
    }

    /** Missing gateway institution header. */
    @ExceptionHandler(MissingRequestHeaderException.class)
    public ProblemDetail handleMissingHeader(MissingRequestHeaderException ex) {
        String detail = "Required request header '" + ex.getHeaderName() + "' is not present";
        log.warn("Bad request: {}", detail);
        return ProblemDetail.forStatusAndDetail(HttpStatus.BAD_REQUEST, detail);
    }

    /**
     * Concurrent duplicate submit while the first publish is still in flight and did not
     * finish within the wait budget.
     */
    @ExceptionHandler(IdempotencyConflictException.class)
    public ResponseEntity<ProblemDetail> handleIdempotencyConflict(IdempotencyConflictException ex) {
        log.warn("Returning 409 for idempotency conflict eventId={}: {}", mdcEventId(), ex.getMessage());
        ProblemDetail problem = ProblemDetail.forStatusAndDetail(HttpStatus.CONFLICT, ex.getMessage());
        problem.setTitle("Publish already in progress");
        return withRetryAfter(problem, RETRY_AFTER_IDEMPOTENCY_CONFLICT_SECONDS);
    }

    /** Jakarta Bean Validation failures on the request body (e.g. missing {@code eventId}). */
    @ExceptionHandler(MethodArgumentNotValidException.class)
    public ProblemDetail handleValidation(MethodArgumentNotValidException ex) {
        String detail = ex.getBindingResult().getFieldErrors().stream()
                .findFirst()
                .map(err -> err.getField() + ": " + err.getDefaultMessage())
                .orElse("Validation failed");
        log.warn("Request validation failed: {}", detail);
        return ProblemDetail.forStatusAndDetail(HttpStatus.BAD_REQUEST, detail);
    }

    /** Serialized payload exceeds the configured maximum size. */
    @ExceptionHandler(PayloadTooLargeException.class)
    public ProblemDetail handlePayloadTooLarge(PayloadTooLargeException ex) {
        log.warn("Rejecting oversized payload eventId={}: {}", mdcEventId(), ex.getMessage());
        ProblemDetail problem = ProblemDetail.forStatusAndDetail(HttpStatus.PAYLOAD_TOO_LARGE, ex.getMessage());
        problem.setTitle("Payload too large");
        return problem;
    }

    /** Kafka publish timeout or broker errors; clients should retry with backoff. */
    @ExceptionHandler(NotificationPublishException.class)
    public ResponseEntity<ProblemDetail> handlePublishFailure(NotificationPublishException ex) {
        log.warn("Returning 503 for publish failure eventId={}: {}", mdcEventId(), ex.getMessage());
        ProblemDetail problem = ProblemDetail.forStatusAndDetail(
                HttpStatus.SERVICE_UNAVAILABLE,
                "Notification event could not be published. Please retry with the same eventId."
        );
        problem.setTitle("Notification publish failed");
        return withRetryAfter(problem, RETRY_AFTER_PUBLISH_FAILURE_SECONDS);
    }

    /** Circuit open for the event's source: Kafka is degraded, fail fast without holding a thread. */
    @ExceptionHandler(CallNotPermittedException.class)
    public ResponseEntity<ProblemDetail> handleCircuitOpen(CallNotPermittedException ex) {
        log.warn("Returning 503 (circuit open) eventId={}: {}", mdcEventId(), ex.getMessage());
        ProblemDetail problem = ProblemDetail.forStatusAndDetail(
                HttpStatus.SERVICE_UNAVAILABLE,
                "Publishing is temporarily suspended for this event source. Please retry with the same eventId."
        );
        problem.setTitle("Notification publishing suspended");
        return withRetryAfter(problem, RETRY_AFTER_CIRCUIT_OPEN_SECONDS);
    }

    /** Concurrency cap reached for the event's source; the caller should back off briefly. */
    @ExceptionHandler(BulkheadFullException.class)
    public ResponseEntity<ProblemDetail> handleBulkheadFull(BulkheadFullException ex) {
        log.warn("Returning 429 (bulkhead full) eventId={}: {}", mdcEventId(), ex.getMessage());
        ProblemDetail problem = ProblemDetail.forStatusAndDetail(
                HttpStatus.TOO_MANY_REQUESTS,
                "Too many concurrent publishes for this event source. Please retry with the same eventId."
        );
        problem.setTitle("Publish concurrency limit reached");
        return withRetryAfter(problem, RETRY_AFTER_BULKHEAD_FULL_SECONDS);
    }

    /** Catch-all for uncategorized server errors. */
    @ExceptionHandler(Exception.class)
    public ProblemDetail handleUnexpected(Exception ex) {
        log.error("Unhandled notification API error eventId={}", mdcEventId(), ex);
        ProblemDetail problem = ProblemDetail.forStatusAndDetail(
                HttpStatus.INTERNAL_SERVER_ERROR,
                "Unexpected server error"
        );
        problem.setTitle("Internal server error");
        return problem;
    }

    private static ResponseEntity<ProblemDetail> withRetryAfter(ProblemDetail problem, String retryAfterSeconds) {
        return ResponseEntity.status(problem.getStatus())
                .header(HttpHeaders.RETRY_AFTER, retryAfterSeconds)
                .body(problem);
    }

    private static String mdcEventId() {
        return MDC.get(TraceIdFilter.EVENT_ID_MDC_KEY);
    }
}
