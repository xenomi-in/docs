package com.lps.cms.notification.exception;

/**
 * Serialized event payload exceeds the configured maximum
 * ({@code app.notification.producer.max-payload-size}).
 * <p>
 * Enforced before publish so oversized events fail with HTTP 413 instead of
 * surfacing as a misleading Kafka broker error.
 */
public class PayloadTooLargeException extends RuntimeException {

    public PayloadTooLargeException(String message) {
        super(message);
    }
}
