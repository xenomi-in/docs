package com.lps.cms.notification.config;

import com.lps.cms.notification.model.EventSource;

import jakarta.validation.Valid;
import jakarta.validation.constraints.NotBlank;
import org.springframework.boot.context.properties.ConfigurationProperties;
import org.springframework.validation.annotation.Validated;

import java.util.EnumSet;
import java.util.Map;
import java.util.Set;

/**
 * Per-source Kafka topic names bound from {@code app.notification.topics.sources.*}.
 * <p>
 * Each {@link EventSource} owns a full topic set (main + retry tiers + DLQ) so upstream
 * streams are isolated and can be scaled, retained, and consumed independently. This
 * service publishes only to {@link TopicSet#events()}; retry and DLQ topics are
 * provisioned for downstream consumers.
 * <p>
 * Binding fails at startup when any {@link EventSource} lacks a topic set, so a
 * misconfigured deployment can never accept an event it cannot route.
 *
 * @param sources topic set per event source; every {@link EventSource} must be present
 */
@Validated
@ConfigurationProperties(prefix = "app.notification.topics")
public record NotificationTopicsProperties(
        Map<EventSource, @Valid TopicSet> sources
) {

    public NotificationTopicsProperties {
        Set<EventSource> missing = EnumSet.allOf(EventSource.class);
        if (sources != null) {
            missing.removeAll(sources.keySet());
        }
        if (!missing.isEmpty()) {
            throw new IllegalStateException(
                    "Missing Kafka topic configuration under app.notification.topics.sources for: " + missing);
        }
        sources = Map.copyOf(sources);
    }

    /**
     * Resolves the primary events topic for the given source.
     *
     * @param source upstream event source
     * @return topic name this service publishes the event to
     */
    public String eventsTopic(EventSource source) {
        return sources.get(source).events();
    }

    /**
     * Kafka topic names owned by a single event source.
     *
     * @param events primary notification event topic (this service publishes here)
     * @param retry1 first retry topic (consumer-owned)
     * @param retry2 second retry topic (consumer-owned)
     * @param retry3 third retry topic (consumer-owned)
     * @param dlq    dead-letter topic for exhausted retries
     */
    public record TopicSet(
            @NotBlank
            String events,
            @NotBlank
            String retry1,
            @NotBlank
            String retry2,
            @NotBlank
            String retry3,
            @NotBlank
            String dlq
    ) {
    }
}
