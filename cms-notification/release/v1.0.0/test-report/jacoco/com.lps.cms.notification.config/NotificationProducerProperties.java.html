package com.lps.cms.notification.config;

import jakarta.validation.constraints.NotNull;
import org.springframework.boot.context.properties.ConfigurationProperties;
import org.springframework.util.unit.DataSize;
import org.springframework.validation.annotation.Validated;

import java.time.Duration;

/**
 * Producer tuning bound from {@code app.notification.producer.*} in application YAML.
 * <p>
 * {@code deliveryTimeout} is the single source of truth for how long a publish may take:
 * it is applied to the Kafka producer's {@code delivery.timeout.ms} and the application waits
 * {@code deliveryTimeout + awaitBuffer} on the send future. Because the application always waits
 * longer than the producer's own delivery deadline, the future resolves with the producer's
 * authoritative outcome before the wait elapses, so a failure can never be reported while a
 * record is still in flight (which would otherwise be delivered later and cause a duplicate).
 *
 * @param deliveryTimeout maximum time the Kafka producer may spend delivering a record (drives {@code delivery.timeout.ms})
 * @param awaitBuffer     extra time the application waits beyond {@code deliveryTimeout} so the producer always resolves first
 * @param maxPayloadSize  maximum serialized payload size accepted for a single event; oversized payloads are rejected
 *                        with HTTP 413 before publish instead of failing at the broker's {@code max.request.size}
 */
@Validated
@ConfigurationProperties(prefix = "app.notification.producer")
public record NotificationProducerProperties(
        @NotNull
        Duration deliveryTimeout,
        @NotNull
        Duration awaitBuffer,
        @NotNull
        DataSize maxPayloadSize
) {

    /**
     * @return how long the application blocks on the send future: {@code deliveryTimeout + awaitBuffer}
     */
    public Duration publishAwaitTimeout() {
        return deliveryTimeout.plus(awaitBuffer);
    }
}
