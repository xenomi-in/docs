package com.logicpoint.iso8583.validation.bin.impl;

import com.logicpoint.iso8583.validation.bin.BinValidator;
import com.logicpoint.iso8583.validation.ValidationResult;
import com.logicpoint.iso8583.service.CacheInitializationService;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import java.util.ArrayList;
import java.util.List;
import java.util.Map;

/**
 * Concrete implementation of BinValidator with comprehensive BIN validation logic.
 */
@Service
public class BinValidatorImpl implements BinValidator {
    
    Logger log = LoggerFactory.getLogger(BinValidatorImpl.class);

    @Autowired
    private CacheInitializationService cacheService;
    
    @Override
    public ValidationResult validate(String bin) {
        List<String> errors = new ArrayList<>();
        
        Map<String, CacheInitializationService.BinInstitutionData> binCache = cacheService.loadBinInstitutionCache();
        CacheInitializationService.BinInstitutionData binData = binCache.get(bin);
        log.debug(" BIN Data: {}", binCache.keySet());
        if (binData == null) {
            errors.add("INVALID_BIN: BIN not found");
            return new ValidationResult(errors, new ArrayList<>());
        }

        if (!binData.isBinActive()) {
            errors.add("BLOCKED_BIN: BIN is blocked or suspended");
        }

        // Additional BIN validations
        if (!bin.matches("\\d{6}")) {
            errors.add("INVALID_BIN: BIN must be exactly 6 digits");
        }
        log.debug("BIN Validation Result: {}", errors);
        return new ValidationResult(errors, new ArrayList<>());
    }
    @Override
    public String getValidatorType() {
        return "BIN_VALIDATOR";
    }
}
