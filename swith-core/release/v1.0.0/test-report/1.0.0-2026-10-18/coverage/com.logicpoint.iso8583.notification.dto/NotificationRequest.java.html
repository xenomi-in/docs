package com.logicpoint.iso8583.notification.dto;

import com.fasterxml.jackson.annotation.JsonInclude;
import com.fasterxml.jackson.annotation.JsonIgnore;
import jakarta.validation.Valid;
import jakarta.validation.constraints.AssertTrue;
import jakarta.validation.constraints.NotBlank;
import jakarta.validation.constraints.NotEmpty;
import jakarta.validation.constraints.NotNull;
import java.util.List;

@JsonInclude(JsonInclude.Include.NON_NULL)
public record NotificationRequest(
        @NotBlank String eventType,
        @NotNull @Valid DeclinedTransactionPayload payload,
        String eventId,
        String tenantId,
        String customerId,
        String aggregateType,
        String aggregateId,
        @NotBlank String eventSource,
        String correlationId,
        @NotEmpty List<NotificationChannel> channels,
        @NotNull NotificationPriority priority
) {

    public static final String TRANSACTION_DECLINED = "TRANSACTION_DECLINED";
    public static final String SWITCH_EVENT_SOURCE = "SWITCH";

    @AssertTrue(message = "eventSource must be SWITCH")
    @JsonIgnore
    public boolean isSwitchEventSource() {
        return SWITCH_EVENT_SOURCE.equals(eventSource);
    }

    @AssertTrue(message = "customerId, tenantId, or aggregateType + aggregateId is required")
    @JsonIgnore
    public boolean hasRoutingIdentifier() {
        return hasText(customerId)
                || hasText(tenantId)
                || (hasText(aggregateType) && hasText(aggregateId));
    }

    private boolean hasText(String value) {
        return value != null && !value.isBlank();
    }
}
