package com.logicpoint.iso8583.notification.mapper;

import com.logicpoint.iso8583.context.RequestContext;
import com.logicpoint.iso8583.context.TransactionChannel;
import com.logicpoint.iso8583.notification.dto.DeclinedTransactionPayload;
import com.logicpoint.iso8583.notification.dto.NotificationChannel;
import com.logicpoint.iso8583.notification.dto.NotificationPriority;
import com.logicpoint.iso8583.notification.dto.NotificationRequest;
import com.logicpoint.iso8583.validation.TransactionType;
import java.time.Clock;
import java.time.OffsetDateTime;
import java.util.List;
import java.util.Map;
import java.util.Set;
import org.springframework.stereotype.Component;

@Component
public class DeclinedTransactionNotificationMapper {

    private static final List<NotificationChannel> DEFAULT_CHANNELS = List.of(
            NotificationChannel.EMAIL,
            NotificationChannel.SMS
    );
    private static final String CHANNEL_ATM = "ATM";
    private static final String CHANNEL_POS = "POS";
    private static final String CHANNEL_ECOMMERCE = "E-comm";
    private static final Set<TransactionType> ATM_TRANSACTION_TYPES = Set.of(
            TransactionType.CASH_WITHDRAWAL,
            TransactionType.CASH_WITHDRAWAL_CONTACTLESS,
            TransactionType.BALANCE_ENQUIRY,
            TransactionType.BALANCE_ENQUIRY_CL,
            TransactionType.MINI_STATEMENT,
            TransactionType.PIN_CHANGE,
            TransactionType.STATEMENT_REQUEST,
            TransactionType.CASH_DEPOSIT,
            TransactionType.SERVICE_CREATION,
            TransactionType.SERVICE_CREATION_CONTACTLESS,
            TransactionType.MOBILE_NUMBER_UPDATE,
            TransactionType.CHEQUEBOOK_REQUEST,
            TransactionType.AADHAR_INQUIRY,
            TransactionType.BIOMETIC_VERIFICATION
    );
    private static final Set<TransactionType> ECOMMERCE_TRANSACTION_TYPES = Set.of(
            TransactionType.ECOMMERCE,
            TransactionType.MOTO,
            TransactionType.IVR_PURCHASE,
            TransactionType.EMI_PURCHASE,
            TransactionType.SI_REGISTRATION,
            TransactionType.SI_EDIT,
            TransactionType.SI_TRANSACTION,
            TransactionType.SI_DEREGISTRATION,
            TransactionType.SI_REVOKEALL,
            TransactionType.SI_PAUSE,
            TransactionType.SI_START,
            TransactionType.SI_PDN
    );
    private final Clock clock;

    public DeclinedTransactionNotificationMapper() {
        this(Clock.systemUTC());
    }

    DeclinedTransactionNotificationMapper(Clock clock) {
        this.clock = clock;
    }

    public NotificationRequest toNotificationRequest(RequestContext requestContext, Map<Integer, String> responseFields, String traceId) {
        DeclinedTransactionPayload payload = new DeclinedTransactionPayload(
                requestContext.getUniqueTransactionId(),
                field(requestContext, responseFields, 11),
                field(requestContext, responseFields, 37),
                field(requestContext, responseFields, 2),
                requestContext.getPanHash(),
                field(requestContext, responseFields, 3),
                field(requestContext, responseFields, 4),
                field(requestContext, responseFields, 49),
                field(requestContext, responseFields, 42),
                field(requestContext, responseFields, 41),
                resolveChannel(requestContext, responseFields),
                responseFields.get(39),
                resolveDeclineReason(requestContext),
                OffsetDateTime.now(clock),
                requestContext.getRequestId(),
                requestContext.getInstitutionId(),
                field(requestContext, responseFields, 32)
        );

        return new NotificationRequest(
                NotificationRequest.TRANSACTION_DECLINED,
                payload,
                null,
                requestContext.getInstitutionId(),
                requestContext.getPanHash(),
                "TRANSACTION",
                requestContext.getUniqueTransactionId(),
                NotificationRequest.SWITCH_EVENT_SOURCE,
                firstNonBlank(field(requestContext, responseFields, 37), traceId),
                DEFAULT_CHANNELS,
                NotificationPriority.MEDIUM
        );
    }

    private String resolveDeclineReason(RequestContext requestContext) {
        return requestContext.getInternalResponseCode() == null
                ? "ISSUER_DECLINED"
                : requestContext.getInternalResponseCode().name();
    }

    private String resolveChannel(RequestContext requestContext, Map<Integer, String> responseFields) {
        TransactionChannel resolvedChannel = requestContext.getChannel();
        if (resolvedChannel != null && resolvedChannel != TransactionChannel.UNKNOWN) {
            return switch (resolvedChannel) {
                case ATM -> CHANNEL_ATM;
                case ECOMMERCE -> CHANNEL_ECOMMERCE;
                case POS -> CHANNEL_POS;
                case UNKNOWN -> CHANNEL_POS;
            };
        }

        // Compatibility fallback for contexts created outside the main processing pipeline.
        TransactionType transactionType = requestContext.getTransactionType();
        if (ATM_TRANSACTION_TYPES.contains(transactionType)) {
            return CHANNEL_ATM;
        }
        if (ECOMMERCE_TRANSACTION_TYPES.contains(transactionType)) {
            return CHANNEL_ECOMMERCE;
        }

        String posEntryMode = field(requestContext, responseFields, 22);
        String posConditionCode = field(requestContext, responseFields, 25);
        String processingCode = field(requestContext, responseFields, 3);
        if (isEcommerce(posEntryMode, posConditionCode)) {
            return CHANNEL_ECOMMERCE;
        }
        if (isAtm(processingCode, posEntryMode, posConditionCode)) {
            return CHANNEL_ATM;
        }
        return CHANNEL_POS;
    }

    private boolean isEcommerce(String posEntryMode, String posConditionCode) {
        return startsWithAny(posEntryMode, "001", "011", "061", "810", "811")
                || "59".equals(posConditionCode)
                || "08".equals(posConditionCode)
                || "06".equals(posConditionCode)
                || "07".equals(posConditionCode);
    }

    private boolean isAtm(String processingCode, String posEntryMode, String posConditionCode) {
        return startsWithAny(processingCode, "01", "30", "31", "38", "40", "91")
                || startsWithAny(posEntryMode, "010", "020")
                || "51".equals(posConditionCode)
                || "52".equals(posConditionCode);
    }

    private boolean startsWithAny(String value, String... prefixes) {
        if (value == null || value.isBlank()) {
            return false;
        }
        for (String prefix : prefixes) {
            if (value.startsWith(prefix)) {
                return true;
            }
        }
        return false;
    }

    private String field(RequestContext requestContext, Map<Integer, String> responseFields, int fieldNumber) {
        String responseValue = responseFields.get(fieldNumber);
        return responseValue != null ? responseValue : requestContext.getFields().get(fieldNumber);
    }

    private String firstNonBlank(String value, String fallback) {
        return value == null || value.isBlank() ? fallback : value;
    }
}
