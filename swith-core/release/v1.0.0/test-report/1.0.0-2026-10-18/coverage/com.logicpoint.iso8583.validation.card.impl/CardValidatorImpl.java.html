package com.logicpoint.iso8583.validation.card.impl;

import com.logicpoint.iso8583.validation.card.CardValidator;
import com.logicpoint.iso8583.validation.ValidationResult;
import com.logicpoint.iso8583.validation.orchestrator.CardDataContext;

import org.springframework.stereotype.Service;

import com.logicpoint.iso8583.entity.CardData;
import com.logicpoint.iso8583.service.CardDataService;

import org.springframework.beans.factory.annotation.Autowired;

import java.util.ArrayList;
import java.util.List;
import java.time.LocalDate;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

/**
 * @author LogicPoint Systems   
 * @version 1.0
 * @since 2025-07-16
 * 
 * This class implements the CardValidator interface and provides comprehensive card validation logic.
 * It includes methods for validating card numbers, expiry dates, Luhn algorithm for card number validation and checks if the card number is present in the database and status is active.

 */
@Service
public class CardValidatorImpl implements CardValidator {

    private static final Logger log = LoggerFactory.getLogger(CardValidatorImpl.class);

    @Autowired
    private CardDataService cardDataService;

    @Override
    public ValidationResult validate(String cardData) {
        List<String> errors = new ArrayList<>();

        // Basic card number validation
        ValidationResult cardNumberResult = validateCardNumber(cardData);
        if (!cardNumberResult.isValid()) {
            errors.addAll(cardNumberResult.getErrors());
        }

        return new ValidationResult(errors, new ArrayList<>());
    }

    @Override
    public ValidationResult validateCardNumber(String cardNumber) {
        List<String> errors = new ArrayList<>();

        if (cardNumber == null || cardNumber.trim().isEmpty()) {
            errors.add("INVALID_CARD: Card number is required");
            return new ValidationResult(errors, new ArrayList<>());
        }

        // Remove spaces and dashes
        String cleanCardNumber = cardNumber.replaceAll("[\\s-]", "");

        // Check length (13-19 digits)
        if (cleanCardNumber.length() < 13 || cleanCardNumber.length() > 19) {
            errors.add("INVALID_CARD: Card number must be 13-19 digits");
            return new ValidationResult(errors, new ArrayList<>());
        }

        // Check if all characters are digits
        if (!cleanCardNumber.matches("\\d+")) {
            errors.add("INVALID_CARD: Card number must contain only digits");
            return new ValidationResult(errors, new ArrayList<>());
        }

        // Luhn algorithm validation
        if (!isValidLuhn(cleanCardNumber)) {
            errors.add("INVALID_CARD: Card number failed checksum validation");
            return new ValidationResult(errors, new ArrayList<>());
        }
       

       
        return new ValidationResult(errors, new ArrayList<>());
    }

    @Override
    public ValidationResult validateExpiryDate(String expiryDate,String cardNumber) {
        List<String> errors = new ArrayList<>();

        if (expiryDate == null || expiryDate.trim().isEmpty()) {
            errors.add("INVALID_CARD: Expiry date is required");
            return new ValidationResult(errors, new ArrayList<>());
        }

        if (!expiryDate.matches("\\d{4}")) {
            errors.add("INVALID_CARD: Expiry date must be in MMYY format");
            return new ValidationResult(errors, new ArrayList<>());
        }

        try {
            String expDate = "20" + expiryDate.substring(0, 2) + "-" + expiryDate.substring(2) + "-01";
            log.debug("Expiry date: {}", expDate);
            LocalDate expiry = LocalDate.parse(expDate);
            LocalDate now = LocalDate.now();

            if (expiry.isBefore(now)) {
                errors.add("EXPIRED_CARD: Card has expired");
                return new ValidationResult(errors, new ArrayList<>());
            }
        } catch (Exception e) {
            errors.add("INVALID_CARD: Invalid expiry date format");
            e.printStackTrace();
            return new ValidationResult(errors, new ArrayList<>());
        }
        // Check if card number is present in the database and status is active
        CardDataContext cardDataContext = cardDataService.fetchCardData(cardNumber);
        CardData cardData = cardDataContext.isValid() ? cardDataContext.getCardData() : null;

        if (cardData == null) {
            errors.add("INVALID_CARD: Card not found in DB");
            return new ValidationResult(errors, new ArrayList<>());
        } else if (!cardData.getStatus().equalsIgnoreCase("ACTIVE")) {
            errors.add("INVALID_CARD: Card is not active");
            return new ValidationResult(errors, new ArrayList<>());
        } else if (!cardData.getExpiryDate().equals(expiryDate)) {
            errors.add("INVALID_CARD: Card expiry date does not match");
            return new ValidationResult(errors, new ArrayList<>());
        }
        return new ValidationResult(errors, new ArrayList<>());
    }

    @Override
    public ValidationResult validateCvv(String cvv) {
        List<String> errors = new ArrayList<>();

        if (cvv == null || cvv.trim().isEmpty()) {
            errors.add("INVALID_CARD: CVV is required");
            return new ValidationResult(errors, new ArrayList<>());
        }

        if (!cvv.matches("\\d{3,4}")) {
            errors.add("INVALID_CARD: CVV must be 3-4 digits");
        }

        return new ValidationResult(errors, new ArrayList<>());
    }

    @Override
    public String getValidatorType() {
        return "CARD_VALIDATOR";
    }

    /**
     * Implements Luhn algorithm for card number validation.
     */
    private boolean isValidLuhn(String cardNumber) {
        int sum = 0;
        boolean alternate = false;

        for (int i = cardNumber.length() - 1; i >= 0; i--) {
            int n = Integer.parseInt(cardNumber.substring(i, i + 1));
            if (alternate) {
                n *= 2;
                if (n > 9) {
                    n = (n % 10) + 1;
                }
            }
            sum += n;
            alternate = !alternate;
        }

        return (sum % 10 == 0);
    }
}
