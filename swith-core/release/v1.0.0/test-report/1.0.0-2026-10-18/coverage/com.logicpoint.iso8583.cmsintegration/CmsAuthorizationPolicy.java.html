package com.logicpoint.iso8583.cmsintegration;

import static com.logicpoint.iso8583.cmsintegration.CmsContracts.*;
import com.logicpoint.iso8583.context.RequestContext;
import java.util.Map;

public final class CmsAuthorizationPolicy {
    private CmsAuthorizationPolicy() {}
    public static void enforce(Profile profile, RequestContext request, String homeCountry, boolean familyBlocked) {
        CmsProfileRules.validate(profile);
        if(familyBlocked || profile.effectiveStatus()!=Status.ACTIVE) deny("Card is restricted");
        Map<Integer,String> fields=request.getFields();
        String country=fields.get(19);
        if(country==null || !country.matches("[0-9]{3}")) deny("Acquirer country is required");
        Geography geography=homeCountry.equals(country)?Geography.DOMESTIC:Geography.INTERNATIONAL;
        if(geography==Geography.INTERNATIONAL && !profile.internationalEnabled()) deny("International use disabled");
        Channel channel=channel(request);
        boolean physical=channel==Channel.ATM || channel==Channel.POS_CONTACT || channel==Channel.CONTACTLESS_POS;
        if(physical && !profile.physicalEnabled() || !physical && !profile.digitalEnabled()) deny("Card representation is not ready");
        Control control=profile.controls().stream().filter(c->c.geography()==geography && c.channel()==channel)
                .findFirst().orElseThrow(()->CmsIntegrationException.bad("Control dimension missing"));
        if(!control.enabled()) deny("Channel disabled");
        String amount=fields.get(4);
        if(amount==null || !amount.matches("[0-9]{1,12}") || !profile.currencyCode().equals(fields.get(49)))
            deny("Amount or limit currency unsupported");
        // ISO DE4 is already in currency minor units; never round through a floating-point value.
        long minor=Long.parseLong(amount);
        if(minor>control.perTransactionLimitMinor()) deny("Per-transaction limit exceeded");
        String entry=fields.get(22);
        if(physical && (entry==null || entry.length()<2)) deny("POS entry mode required");
        boolean contactless=entry!=null && (entry.startsWith("07") || entry.startsWith("91"));
        if(contactless && (!control.contactlessEnabled() || minor>control.contactlessLimitMinor()))
            deny("Contactless disabled or limit exceeded");
    }
    static Channel channel(RequestContext request) {
        String type=request.getTransactionType().name();
        if(type.equals("MOTO") || type.equals("IVR_PURCHASE")) return Channel.MOTO;
        if(type.startsWith("SI_") || type.equals("ECOMMERCE") || type.equals("EMI_PURCHASE") || type.equals("CARD_EMI_PURCHASE")) return Channel.ECOM;
        if(type.startsWith("CASH_WITHDRAWAL")) return Channel.ATM;
        if(type.startsWith("PURCHASE") || type.startsWith("CASH_AT_POS") || type.equals("PREAUTH")) return pos(request);
        return switch(request.getChannel()) {
            case ATM -> Channel.ATM;
            case POS -> pos(request);
            case ECOMMERCE -> Channel.ECOM;
            default -> throw CmsIntegrationException.bad("Unsupported authorization channel");
        };
    }
    private static Channel pos(RequestContext request) {
        String entry=request.getFields().get(22);
        return entry!=null && (entry.startsWith("07") || entry.startsWith("91")) ? Channel.CONTACTLESS_POS : Channel.POS_CONTACT;
    }
    private static void deny(String message) { throw CmsIntegrationException.bad(message); }
}
