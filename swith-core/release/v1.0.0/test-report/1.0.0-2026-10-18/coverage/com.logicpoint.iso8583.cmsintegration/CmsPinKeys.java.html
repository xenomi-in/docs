package com.logicpoint.iso8583.cmsintegration;

import java.util.*;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.stereotype.Component;

@Component
public class CmsPinKeys {
    private final JdbcTemplate db;
    public CmsPinKeys(JdbcTemplate db) { this.db=db; }
    public String resolve(UUID institution,String reference,String purpose) {
        var rows=db.queryForList("""
            select k.key_value,k.is_tr31 from cms_pin_key_binding b
            join cms_institution_mapping m on m.cms_institution_id=b.cms_institution_id
            join key_management k on k.key_id=b.key_id
            where b.cms_institution_id=? and b.reference=? and b.purpose=?
              and k.institution_id=m.switch_institution_id
              and k.key_status='A' and k.valid_from<=now() and k.valid_to>=now()
              and ((b.purpose='PVK' and k.key_type='PVK')
                or (b.purpose='INPUT' and k.key_type in ('ZPK','PEK')))
            """,institution,reference,purpose);
        if(rows.size()!=1) throw CmsIntegrationException.unavailable("Approved PIN key binding is unavailable");
        var row=rows.getFirst();
        byte[] bytes=(byte[])row.get("key_value");
        if(bytes==null || bytes.length==0) throw CmsIntegrationException.unavailable("PIN key is unavailable");
        // Initial adapter supports the repository's existing cryptogram command format.
        if(Boolean.TRUE.equals(row.get("is_tr31")))
            throw CmsIntegrationException.unavailable("TR-31 PIN keys require a certified command adapter");
        return HexFormat.of().withUpperCase().formatHex(bytes);
    }
}
