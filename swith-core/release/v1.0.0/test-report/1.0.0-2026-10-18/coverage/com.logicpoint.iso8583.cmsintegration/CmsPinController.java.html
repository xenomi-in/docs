package com.logicpoint.iso8583.cmsintegration;

import tools.jackson.databind.json.JsonMapper;

import tools.jackson.databind.*;
import io.swagger.v3.oas.annotations.Operation;
import io.swagger.v3.oas.annotations.Parameter;
import io.swagger.v3.oas.annotations.media.Content;
import io.swagger.v3.oas.annotations.media.ExampleObject;
import io.swagger.v3.oas.annotations.media.Schema;
import io.swagger.v3.oas.annotations.responses.ApiResponse;
import io.swagger.v3.oas.annotations.tags.Tag;
import io.swagger.v3.oas.annotations.security.SecurityRequirement;
import jakarta.validation.Validator;
import org.springframework.web.bind.annotation.*;
import java.util.UUID;
import static com.logicpoint.iso8583.cmsintegration.CmsPinContracts.*;

@RestController
@RequestMapping("/internal/v1/institutions/{institutionId}")
@SecurityRequirement(name="cmsServiceBearer")
@Tag(name="CMS PIN operations", description="Separate PIN establishment after committed card registration. Requires the CMS service JWT scope switch:card-authorization:write and mTLS when enabled.")
public class CmsPinController {
    private final CmsPinOperationService service; private final JsonMapper json; private final Validator validator;
    public CmsPinController(CmsPinOperationService service,JsonMapper json,Validator validator){this.service=service;this.json=json.rebuild().enable(DeserializationFeature.FAIL_ON_UNKNOWN_PROPERTIES).build();this.validator=validator;}
    @PostMapping("/pan-families/{panFamilyReference}/pin-operations")
    @Operation(summary="Submit a serialized, versioned PIN operation",
            description="Supports CONFIGURE_PIN_PROFILE, INITIALIZE_SELECTED_PIN, GENERATE_INITIAL_PIN, CHANGE_PIN and RESET_PIN. "
                    + "Path identities must match the body. Replay the identical request with the same operationId. "
                    + "Configuration completes without changing the PIN version. Initialization, change and reset require "
                    + "protectedPinMaterial: RSA-OAEP-256/A256GCM JWE with kid=cms-pin-v1 (configurable). "
                    + "The envelope binds all command identities/versions and expires within five minutes. "
                    + "Change requires currentPin and newPin; reset/initialization require only newPin. Each contains "
                    + "encryptedPinBlock, format (ISO_0:16 hex digits, ISO_4:32), and keyBindingReference. "
                    + "authorizationReference is required. Clear PINs are never accepted. Generation omits the envelope. "
                    + "The Visa/ISO-0 adapter uses explicit PVK/PVKI and trusted one-use authorization grants. Disabled/unavailable HSM returns 503. IBM, ISO_4, TR-31 and generated mailers are not implemented. No automatic pending completion is implemented.",
            requestBody=@io.swagger.v3.oas.annotations.parameters.RequestBody(required=true,
                    content=@Content(mediaType="application/json", schema=@Schema(implementation=PinCommand.class),
                    examples={@ExampleObject(name="CONFIGURE_SELECTED", description="Requires a registered family with no existing PIN profile. Key alias is illustrative.", value=CmsPinApiExamples.CONFIGURE_SELECTED_REQUEST),
                    @ExampleObject(name="INITIALIZE_SELECTED", description="Requires a configured HSM executor; encrypted placeholders must be replaced with a real JWE where supplied.", value=CmsPinApiExamples.INITIALIZE_SELECTED_REQUEST),
                    @ExampleObject(name="CONFIGURE_GENERATED", description="Requires a registered family with no existing PIN profile. Key alias is illustrative.", value=CmsPinApiExamples.CONFIGURE_GENERATED_REQUEST),
                    @ExampleObject(name="GENERATE_INITIAL", description="Requires a configured HSM executor; encrypted placeholders must be replaced with a real JWE where supplied.", value=CmsPinApiExamples.GENERATE_INITIAL_REQUEST),
                    @ExampleObject(name="CHANGE", description="Requires an already established active PIN version 1. Replace the encrypted placeholder with a bound JWE containing PIN blocks and authorization.", value=CmsPinApiExamples.CHANGE_REQUEST),
                    @ExampleObject(name="RESET", description="Requires an already established active PIN version 2. Replace the encrypted placeholder with a bound JWE containing PIN blocks and authorization.", value=CmsPinApiExamples.RESET_REQUEST)})))
    @ApiResponse(responseCode="200", description="Stored receipt. Execution success examples require configured HSM and authorization grants; they are not produced by the default deployment. Examples use ISO-8601 timestamps; timestamp-enabled Jackson configuration emits numeric Unix seconds.",
            content=@Content(mediaType="application/json", schema=@Schema(implementation=PinOperationReceipt.class),
                    examples={@ExampleObject(name="CONFIGURE_SELECTED", value=CmsPinApiExamples.CONFIGURE_SELECTED_RESPONSE),
                    @ExampleObject(name="INITIALIZE_SELECTED", value=CmsPinApiExamples.INITIALIZE_SELECTED_RESPONSE),
                    @ExampleObject(name="CONFIGURE_GENERATED", value=CmsPinApiExamples.CONFIGURE_GENERATED_RESPONSE),
                    @ExampleObject(name="GENERATE_INITIAL", value=CmsPinApiExamples.GENERATE_INITIAL_RESPONSE),
                    @ExampleObject(name="CHANGE", value=CmsPinApiExamples.CHANGE_RESPONSE),
                    @ExampleObject(name="RESET", value=CmsPinApiExamples.RESET_RESPONSE)}))
    @ApiResponse(responseCode="400", description="Invalid fields or encrypted PIN envelope")
    @ApiResponse(responseCode="409", description="Stale version, replay conflict or invalid establishment state")
    @ApiResponse(responseCode="503", description="PIN key, HSM or authorization integration unavailable; no success receipt is committed")
    public JsonNode submit(
            @Parameter(description="Registered CMS institution UUID", example="11111111-1111-4111-8111-111111111111") @PathVariable UUID institutionId,
            @Parameter(description="Registered PAN family UUID; equals registration panFamilyId", example="55555555-5555-4555-8555-555555555555") @PathVariable UUID panFamilyReference,
            @Parameter(description="Must equal operationId in the chosen request example", example="70000000-0000-4000-8000-000000000001") @RequestHeader("Idempotency-Key") UUID key,
            @RequestBody JsonNode body) {
        PinCommand command;
        try {
            if(body==null || !body.isObject()
                    || !body.path("expectedPinVersion").isIntegralNumber()
                    || !body.path("expectedPinVersion").canConvertToLong()
                    || !body.path("policyProfileVersion").isIntegralNumber()
                    || !body.path("policyProfileVersion").canConvertToLong()) throw new IllegalArgumentException();
            command=json.treeToValue(body,PinCommand.class);
            if(command==null || !validator.validate(command).isEmpty()) throw new IllegalArgumentException();
            if(!key.equals(command.operationId())) throw CmsIntegrationException.bad("Idempotency-Key must equal operationId");
        } catch(CmsIntegrationException e){throw e;} catch(Exception e){throw CmsIntegrationException.bad("Invalid PIN command fields");}
        return service.submit(institutionId,panFamilyReference,command);
    }
    @GetMapping("/pin-operations/{operationId}")
    @Operation(summary="Read PIN operation status", description="Returns the stored receipt without invoking the HSM. No body or Idempotency-Key is required. Pending operations currently do not advance. Unknown operations return 404.")
    @ApiResponse(responseCode="200", description="Stored receipt; no PIN or verification value is returned",
            content=@Content(mediaType="application/json", schema=@Schema(implementation=PinOperationReceipt.class),
                    examples={@ExampleObject(name="Completed generation (requires HSM adapter)", value=CmsPinApiExamples.GENERATE_INITIAL_RESPONSE),
                            @ExampleObject(name="Completed configuration", value=CmsPinApiExamples.CONFIGURE_SELECTED_RESPONSE)}))
    public JsonNode get(
            @Parameter(example="11111111-1111-4111-8111-111111111111") @PathVariable UUID institutionId,
            @Parameter(example="70000000-0000-4000-8000-000000000004") @PathVariable UUID operationId){return service.get(institutionId,operationId);}
}
