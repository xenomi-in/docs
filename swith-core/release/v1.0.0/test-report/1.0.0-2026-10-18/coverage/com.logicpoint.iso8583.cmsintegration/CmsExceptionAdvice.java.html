package com.logicpoint.iso8583.cmsintegration;

import org.springframework.core.annotation.Order;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.*;
import java.util.Map;

@Order(-100)
@RestControllerAdvice(assignableTypes={CmsCardController.class,CmsPinController.class})
public class CmsExceptionAdvice {
    private static final org.slf4j.Logger log = org.slf4j.LoggerFactory.getLogger(CmsExceptionAdvice.class);
    @ExceptionHandler(CmsIntegrationException.class)
    ResponseEntity<?> known(CmsIntegrationException e) {
        log.warn("CMS integration rejected status={} reason={}", e.status, e.getMessage());
        return ResponseEntity.status(e.status).body(Map.of("success",false,"message",e.getMessage()));
    }
    @ExceptionHandler({org.springframework.http.converter.HttpMessageNotReadableException.class,
            org.springframework.web.bind.MissingRequestHeaderException.class,
            org.springframework.web.method.annotation.MethodArgumentTypeMismatchException.class})
    ResponseEntity<?> bad(Exception e) { log.warn("CMS integration invalid request exceptionType={}", e.getClass().getSimpleName()); return ResponseEntity.badRequest().body(Map.of("success",false,"message","Invalid request")); }
    @ExceptionHandler(Exception.class)
    ResponseEntity<?> unexpected(Exception e) {
        java.sql.SQLException sql = null;
        Throwable cause = e;
        for (int depth = 0; cause != null && depth < 20; depth++, cause = cause.getCause()) {
            if (cause instanceof java.sql.SQLException failure) {
                sql = failure;
                break;
            }
        }
        log.warn("CMS integration failed exceptionType={} reasonCode={} sqlState={} sqlErrorCode={}",
                e.getClass().getSimpleName(), diagnosticCode(e),
                sql == null ? "none" : sql.getSQLState(), sql == null ? 0 : sql.getErrorCode());
        // Hibernate and JSON exceptions may include rejected sensitive values. Never echo or log their messages.
        return ResponseEntity.status(503).body(Map.of("success",false,"message","Integration temporarily unavailable; query command receipt before retrying"));
    }

    private static String diagnosticCode(Throwable failure) {
        for (Throwable cause = failure; cause != null; cause = cause.getCause()) {
            String message = cause.getMessage();
            if (message == null) continue;
            if (message.startsWith("CMS registration requires encryption")) return "CHD_PERSISTENCE_MODE_INVALID";
            if (message.startsWith("PAN encryption could not")) return "CHD_CARD_KEY_UNAVAILABLE";
            if (message.startsWith("Expiry encryption could not")) return "CHD_CARD_KEY_UNAVAILABLE";
            if (message.startsWith("Protected PAN lookup could not")) return "CHD_PAN_LOOKUP_UNAVAILABLE";
        }
        return "UNCLASSIFIED_FAILURE";
    }
}
