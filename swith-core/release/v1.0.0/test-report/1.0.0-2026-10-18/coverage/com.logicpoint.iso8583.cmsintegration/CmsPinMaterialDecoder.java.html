package com.logicpoint.iso8583.cmsintegration;

import tools.jackson.databind.json.JsonMapper;

import static com.logicpoint.iso8583.cmsintegration.CmsPinContracts.*;
import tools.jackson.databind.*;
import tools.jackson.core.StreamReadFeature;
import com.nimbusds.jose.*;
import com.nimbusds.jose.crypto.RSADecrypter;
import jakarta.validation.Validator;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.core.io.ResourceLoader;
import org.springframework.stereotype.Component;
import java.security.KeyStore;
import java.security.interfaces.RSAPrivateKey;
import java.time.Instant;

/** Removes only transport encryption; PIN blocks remain encrypted for the HSM. */
@Component
public class CmsPinMaterialDecoder {
    private final ResourceLoader resources;
    private final JsonMapper json;
    private final Validator validator;
    private final String path, password, alias, kid;

    public CmsPinMaterialDecoder(ResourceLoader resources, JsonMapper json, Validator validator,
            @Value("${switch.cms.pin-keystore:}") String path,
            @Value("${switch.cms.pin-keystore-password:}") String password,
            @Value("${switch.cms.pin-key-alias:cms-pin}") String alias,
            @Value("${switch.cms.pin-key-id:cms-pin-v1}") String kid) {
        this.resources=resources;
        this.json=json.rebuild()
                .enable(DeserializationFeature.FAIL_ON_UNKNOWN_PROPERTIES)
                .enable(DeserializationFeature.FAIL_ON_NULL_FOR_PRIMITIVES)
                .enable(DeserializationFeature.FAIL_ON_MISSING_CREATOR_PROPERTIES)
                .enable(DeserializationFeature.FAIL_ON_TRAILING_TOKENS)
                .enable(StreamReadFeature.STRICT_DUPLICATE_DETECTION)
                .disable(MapperFeature.ALLOW_COERCION_OF_SCALARS).build();
        this.validator=validator; this.path=path; this.password=password; this.alias=alias; this.kid=kid;
    }

    public PinMaterial decode(PinCommand command) {
        String compact=command.protectedPinMaterial();
        if(compact==null || compact.isBlank() || compact.length()>16384)
            throw CmsIntegrationException.bad("Protected PIN material is required");
        if(path.isBlank()) throw CmsIntegrationException.unavailable("PIN recipient key is not configured");
        RSAPrivateKey key;
        try {
            KeyStore store=KeyStore.getInstance("PKCS12");
            try(var input=resources.getResource(path).getInputStream()) { store.load(input,password.toCharArray()); }
            key=(RSAPrivateKey)store.getKey(alias,password.toCharArray());
            if(key==null || key.getModulus().bitLength()<2048) throw new IllegalArgumentException();
        } catch(Exception e) {
            throw CmsIntegrationException.unavailable("PIN recipient key is unavailable");
        }
        try {
            JWEObject jwe=JWEObject.parse(compact);
            var header=jwe.getHeader();
            if(!JWEAlgorithm.RSA_OAEP_256.equals(header.getAlgorithm())
                    || !EncryptionMethod.A256GCM.equals(header.getEncryptionMethod())
                    || header.getCompressionAlgorithm()!=null
                    || !kid.equals(header.getKeyID())
                    || (header.getCriticalParams()!=null && !header.getCriticalParams().isEmpty()))
                throw new IllegalArgumentException();
            jwe.decrypt(new RSADecrypter(key));
            // currentPin is optional for reset/initialization; require all binding fields explicitly.
            JsonNode node=json.readTree(jwe.getPayload().toString());
            if(!node.isObject()) throw new IllegalArgumentException();
            if(!node.has("currentPin")) ((tools.jackson.databind.node.ObjectNode)node).putNull("currentPin");
            PinMaterial material=json.treeToValue(node,PinMaterial.class);
            Instant now=Instant.now();
            if(!validator.validate(material).isEmpty()
                    || !command.operationId().equals(material.operationId())
                    || !command.institutionId().equals(material.institutionId())
                    || !command.panFamilyReference().equals(material.panFamilyReference())
                    || command.operation()!=material.operation()
                    || command.policyProfileVersion()!=material.policyProfileVersion()
                    || command.expectedPinVersion()!=material.expectedPinVersion()
                    || !material.expiresAt().isAfter(now) || material.expiresAt().isAfter(now.plusSeconds(300)))
                throw new IllegalArgumentException();
            if(command.operation()==Operation.CHANGE_PIN) {
                validateBlock(material.currentPin());
            } else if(command.operation()!=Operation.RESET_PIN && command.operation()!=Operation.INITIALIZE_SELECTED_PIN
                    || material.currentPin()!=null) throw new IllegalArgumentException();
            validateBlock(material.newPin());
            return material;
        } catch(Exception e) {
            // Never propagate parser/HSM input text or cryptographic exception details.
            throw CmsIntegrationException.bad("Invalid or expired protected PIN material");
        }
    }

    private void validateBlock(EncryptedPin pin) {
        if(pin==null || pin.format()==null || pin.encryptedPinBlock()==null) throw new IllegalArgumentException();
        int length=pin.format()==PinBlockFormat.ISO_0?16:32;
        if(!pin.encryptedPinBlock().matches("[0-9A-Fa-f]{"+length+"}")) throw new IllegalArgumentException();
    }
}
