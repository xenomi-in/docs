package com.logicpoint.iso8583.cmsintegration;

import io.swagger.v3.oas.annotations.media.Schema;
import jakarta.validation.Valid;
import jakarta.validation.constraints.*;
import java.time.Instant;
import java.util.UUID;

/** CMS-to-Switch PIN contract. Clear PINs and verification values are never returned. */
public final class CmsPinContracts {
    private CmsPinContracts() {}
    public enum Operation { CONFIGURE_PIN_PROFILE, INITIALIZE_SELECTED_PIN, GENERATE_INITIAL_PIN, CHANGE_PIN, RESET_PIN }
    public enum Status { PENDING, COMPLETED, FAILED }
    public enum Establishment { CUSTOMER_SELECTED, HSM_GENERATED }
    public enum Delivery { NONE, PIN_MAILER }
    public enum VerificationMethod { VISA_PVV, IBM_3624_OFFSET }

    @Schema(description="Configuration only. CUSTOMER_SELECTED requires NONE delivery; HSM_GENERATED requires PIN_MAILER. The runtime currently supports VISA_PVV; configured profiles also drive transaction verification.")
    public record PinProfile(@NotNull Establishment establishment, @NotNull Delivery delivery,
            @NotNull VerificationMethod verificationMethod,
            @Schema(description="Institution-scoped cms_pin_key_binding alias pinning an exact key_management row; resolved at execution", example="issuer10.visa.pvk.v1")
            @NotBlank @Pattern(regexp="[A-Za-z0-9_.:-]{1,100}") String keyBindingReference,
            @Schema(description="Explicit Visa PIN verification key index shared by establishment and verification", example="0")
            @NotNull @Min(0) @Max(6) Integer pvki) {
        public PinProfile(Establishment establishment, Delivery delivery, VerificationMethod method, String reference) {
            this(establishment,delivery,method,reference,0);
        }
    }

    public record PinCommand(
            @Schema(description="Unique command ID; must equal Idempotency-Key") @NotNull UUID operationId,
            @Schema(description="Must equal path institutionId") @NotNull UUID institutionId,
            @Schema(description="Must equal path family and registration panFamilyId") @NotNull UUID panFamilyReference,
            @Schema(description="Must increase for configuration; must match stored profile for execution", example="1") @Positive long policyProfileVersion,
            @Schema(description="Must match active PIN version; initially 0", example="0") @PositiveOrZero long expectedPinVersion,
            @NotNull Operation operation,
            @Schema(description="Required for CONFIGURE_PIN_PROFILE; omit for all other operations") @Valid PinProfile profile,
            @Schema(description="Compact RSA-OAEP-256/A256GCM JWE. Required for initialization, change and reset; omit for configuration and generation.")
            @Size(max=16384) String protectedPinMaterial) {
        @Override public String toString() { return "PinCommand[operationId=" + operationId + "]"; }
    }

    public enum PinBlockFormat { ISO_0, ISO_4 }
    public record EncryptedPin(@NotBlank String encryptedPinBlock, @NotNull PinBlockFormat format,
            @NotBlank @Pattern(regexp="[A-Za-z0-9_.:-]{1,100}") String keyBindingReference) {
        @Override public String toString() { return "EncryptedPin[REDACTED]"; }
    }
    public record PinMaterial(@NotNull UUID operationId, @NotNull UUID institutionId,
            @NotNull UUID panFamilyReference, @Positive long policyProfileVersion,
            @PositiveOrZero long expectedPinVersion, @NotNull Operation operation,
            @NotNull Instant expiresAt,
            @NotBlank @Pattern(regexp="[A-Za-z0-9_.:-]{1,200}") String authorizationReference,
            @Valid EncryptedPin currentPin, @NotNull @Valid EncryptedPin newPin) {
        @Override public String toString() { return "PinMaterial[REDACTED]"; }
    }

    public record PinOperationReceipt(UUID operationId, UUID institutionId, UUID panFamilyReference,
            Operation operation, Status status, long policyProfileVersion,
            @Schema(description="Active PIN version; configuration and pending/failed execution leave it unchanged") long pinVersion,
            @Schema(description="Required on completed initial generation; identifies a mailer, not delivery confirmation", nullable=true, example="MAILER-20260923-000001") String mailerReference,
            @Schema(description="Executor-defined failure code; no standardized catalogue yet", nullable=true) String failureCode,
            @Schema(description="Receipt creation instant; ISO-8601 or Unix seconds according to Jackson configuration") Instant updatedAt) {}
}
