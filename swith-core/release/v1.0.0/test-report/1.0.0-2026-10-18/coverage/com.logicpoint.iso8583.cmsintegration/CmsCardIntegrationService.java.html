package com.logicpoint.iso8583.cmsintegration;

import static com.logicpoint.iso8583.cmsintegration.CmsContracts.*;
import tools.jackson.databind.JsonNode;
import tools.jackson.databind.json.JsonMapper;
import com.logicpoint.iso8583.entity.CardData;
import com.logicpoint.iso8583.repository.CardDataRepository;
import com.logicpoint.iso8583.service.CardDataEncryptionService;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import java.nio.charset.StandardCharsets;
import java.security.MessageDigest;
import java.time.LocalDateTime;
import java.util.*;

@Service
public class CmsCardIntegrationService {
    private final JdbcTemplate db;
    private final JsonMapper json;
    private final CardDataRepository cards;
    private final CardDataEncryptionService crypto;
    private final CmsMaterialDecoder materials;
    public CmsCardIntegrationService(JdbcTemplate db, JsonMapper json, CardDataRepository cards,
            CardDataEncryptionService crypto, CmsMaterialDecoder materials) {
        this.db=db; this.json=json; this.cards=cards; this.crypto=crypto; this.materials=materials;
    }
    @Transactional
    public JsonNode register(UUID tenant, UUID cardId, Registration request) {
        String hash=hash("REGISTER:"+cardId+":"+encode(request));
        JsonNode replay=replay(tenant,request.commandId(),hash);
        if(replay!=null) return replay;
        // Tenant-level serialization also orders registration, family blocks and multi-card cutover.
        lock(tenant);
        db.queryForList("select pg_advisory_xact_lock(hashtextextended(?,0))","cms-pin-family:"+tenant+":"+request.panFamilyId());
        CmsProfileRules.validate(request.profile());
        if(request.profile().effectiveStatus()!=Status.PENDING || request.profile().physicalEnabled()
                || request.profile().digitalEnabled()) throw CmsIntegrationException.bad("Register as PENDING with use disabled");
        if(!db.queryForList("select 1 from cms_card_profile where cms_institution_id=? and cms_card_id=?",tenant,cardId).isEmpty())
            throw CmsIntegrationException.conflict("Card already registered under another command");
        JsonNode material=materials.decode(request.protectedMaterial(),tenant,cardId,request.commandId());
        String pan=material.path("pan").asText();
        // Match the longest configured BIN prefix; only prefixes are sent to SQL, never full PAN.
        List<Integer> owners=db.query("select institution_id from bin_details where bin_number in (?,?) order by length(bin_number) desc limit 1",
                (rs,n)->rs.getInt(1),pan.substring(0,6),pan.substring(0,8));
        if(owners.size()!=1) throw CmsIntegrationException.bad("PAN BIN does not belong to the mapped institution");
        Map<String,Object> mapping=ensureInstitutionMapping(tenant, request.profile().currencyCode(), owners.getFirst());
        int institution=((Number)mapping.get("switch_institution_id")).intValue();
        List<Long> accounts=db.query("select switch_account_id from cms_account_mapping where cms_institution_id=? and cms_account_reference_id=?",
                (rs,n)->rs.getLong(1),tenant,request.accountReferenceId());
        if(accounts.isEmpty()) {
            db.update("insert into cms_account_mapping(cms_institution_id,cms_account_reference_id,switch_account_id) values(?,?,nextval('cms_switch_account_id_seq'))",
                    tenant, request.accountReferenceId());
            accounts=db.query("select switch_account_id from cms_account_mapping where cms_institution_id=? and cms_account_reference_id=?",
                    (rs,n)->rs.getLong(1),tenant,request.accountReferenceId());
        }
        if(accounts.size()!=1) throw CmsIntegrationException.bad("Account mapping could not be created");
        if(request.predecessorCardId()!=null) {
            Map<String,Object> source=row(tenant,request.predecessorCardId());
            Long account=db.queryForObject("select account_id from card_data where auth_id=?",Long.class,source.get("auth_id"));
            if(!accounts.getFirst().equals(account)) throw CmsIntegrationException.bad("Predecessor account mismatch");
        }
        if(!mapping.get("currency_code").toString().equals(request.profile().currencyCode()))
            throw CmsIntegrationException.bad("Profile currency differs from institution currency");
        if(owners.getFirst()!=institution) throw CmsIntegrationException.bad("PAN BIN does not belong to the mapped institution");
        String hmac=crypto.computePanLookupHmac(pan,institution);
        if(hmac==null) throw CmsIntegrationException.unavailable("Institution PAN lookup key is unavailable");
        List<UUID> families=db.query("select distinct pan_family_id from cms_card_profile where cms_institution_id=? and pan_lookup_hmac=?",
                (rs,n)->rs.getObject(1,UUID.class),tenant,hmac);
        if(!families.isEmpty() && !families.equals(List.of(request.panFamilyId())))
            throw CmsIntegrationException.conflict("Same PAN must retain its family identity");
        if(!db.queryForList("select 1 from cms_card_profile where cms_institution_id=? and pan_family_id=? and pan_lookup_hmac<>?",
                tenant,request.panFamilyId(),hmac).isEmpty()) throw CmsIntegrationException.conflict("New PAN requires a new PAN family");
        if(!db.queryForList("select 1 from card_data d where d.pan_lookup_hmac=? and not exists(select 1 from cms_card_profile p where p.auth_id=d.auth_id)",hmac).isEmpty())
            throw CmsIntegrationException.conflict("Legacy PAN requires explicit enrollment before CMS registration");
        // Reserve the identity before the first INSERT; card_id is NOT NULL in the deployed schema.
        // No PAN or PIN material is included in this uncommitted seed row.
        Long authId=db.queryForObject("""
            with identity_value as (
                select nextval(pg_get_serial_sequence('card_data','auth_id')) as id
            )
            insert into card_data(auth_id,card_id,pan,expiry_date,status,account_id,card_type,card_scheme,
                created_at,updated_at,pin_retry_count,max_pin_retry,pin_blocked,atc,arqc_validation_required)
            overriding system value
            select id,cast(id as integer),'','','PENDING',?,?,?,now(),now(),0,3,false,0,true
            from identity_value returning auth_id
            """,Long.class,accounts.getFirst(),material.path("cardType").asText(),material.path("cardScheme").asText());
        CardData card=cards.findById(authId).orElseThrow();
        card.setPan(pan); card.setExpiryDate(material.path("expiryYYMM").asText());
        // PIN establishment is a separate family-level operation. Registration
        // must not import a CMS-computed verification value.
        card.setPvv_offest(null);
        // Switch resolves PVK/CVK selection from its own shared HSM catalogue.
        // Ignore legacy CMS aliases instead of persisting foreign identifiers.
        card.setKey_ref(null); card.setCvv_key_ref(null);
        crypto.prepareProtectedForPersistence(card,institution);
        cards.saveAndFlush(card);
        db.update("insert into cms_card_profile(cms_institution_id,cms_card_id,auth_id,pan_lookup_hmac,pan_family_id,expiry_yymm,physical_sequence,authorization_revision,profile,predecessor_card_id) values(?,?,?,?,?,?,?,1,?::jsonb,?)",
                tenant,cardId,card.getAuth_id(),hmac,request.panFamilyId(),material.path("expiryYYMM").asText(),
                request.physicalSequence(),encode(request.profile()),request.predecessorCardId());
        return finish(tenant,request.commandId(),hash,new Receipt(request.commandId(),cardId,true,1,Status.PENDING));
    }
    @Transactional
    public JsonNode update(UUID tenant, UUID cardId, Update request) {
        String hash=hash("UPDATE:"+cardId+":"+encode(request));
        JsonNode replay=replay(tenant,request.commandId(),hash);
        if(replay!=null) return replay;
        lock(tenant);
        if(request.physicalSequence()!=null) {
            Map<String,Object> source=row(tenant,cardId);
            Object sequence=source.get("physical_sequence");
            if(sequence!=null && !sequence.equals(request.physicalSequence()))
                throw CmsIntegrationException.conflict("Existing physical sequence is immutable; register a successor");
            if(sequence==null) {
                Profile prior=read(source.get("profile").toString());
                if(request.profile().lifecycleVersion()<=prior.lifecycleVersion())
                    throw CmsIntegrationException.conflict("Assigning physical sequence requires a new lifecycle version");
                db.update("update cms_card_profile set physical_sequence=? where cms_institution_id=? and cms_card_id=?",request.physicalSequence(),tenant,cardId);
            }
        }
        apply(tenant,cardId,request.expectedRevision(),request.authorizationRevision(),request.profile(),request.commandId());
        return finish(tenant,request.commandId(),hash,new Receipt(request.commandId(),cardId,true,
                request.authorizationRevision(),request.profile().effectiveStatus()));
    }
    @Transactional
    public JsonNode cutover(UUID tenant, Cutover request) {
        String hash=hash("CUTOVER:"+encode(request));
        JsonNode replay=replay(tenant,request.commandId(),hash);
        if(replay!=null) return replay;
        lock(tenant);
        if(request.predecessorCardId().equals(request.successorCardId())
                || request.predecessor().effectiveStatus()!=Status.REPLACED)
            throw CmsIntegrationException.bad("Cutover requires distinct cards and REPLACED predecessor");
        Map<String,Object> old=row(tenant,request.predecessorCardId());
        Map<String,Object> next=row(tenant,request.successorCardId());
        if(!request.predecessorCardId().equals(next.get("predecessor_card_id"))
                || CmsProfileRules.terminal(read(old.get("profile").toString()).effectiveStatus()))
            throw CmsIntegrationException.conflict("Invalid predecessor relationship or already retired source");
        List<Long> accountIds=db.query("select account_id from card_data where auth_id in (?,?)",
                (rs,n)->rs.getLong(1),old.get("auth_id"),next.get("auth_id"));
        if(accountIds.size()!=2 || !accountIds.get(0).equals(accountIds.get(1)))
            throw CmsIntegrationException.conflict("Cutover cards must belong to the same mapped account");
        apply(tenant,request.predecessorCardId(),request.predecessorExpectedRevision(),
                Math.addExact(request.predecessorExpectedRevision(),1),request.predecessor(),request.commandId());
        apply(tenant,request.successorCardId(),request.successorExpectedRevision(),
                Math.addExact(request.successorExpectedRevision(),1),request.successor(),request.commandId());
        return finish(tenant,request.commandId(),hash,Map.of("commandId",request.commandId(),"applied",true,
                "predecessorAppliedRevision",request.predecessorExpectedRevision()+1,
                "successorAppliedRevision",request.successorExpectedRevision()+1));
    }
    private void apply(UUID tenant, UUID card, long expected, long revision, Profile next, UUID command) {
        Map<String,Object> existing=row(tenant,card);
        Profile old=read(existing.get("profile").toString());
        if(next.physicalEnabled() && existing.get("physical_sequence")==null)
            throw CmsIntegrationException.bad("Physical enablement requires a sequence");
        CmsProfileRules.transition(old,next,((Number)existing.get("authorization_revision")).longValue(),expected,revision);
        if(!old.currencyCode().equals(next.currencyCode())) throw CmsIntegrationException.bad("Card currency is immutable");
        UUID family=(UUID)existing.get("pan_family_id");
        if(next.effectiveStatus()==Status.ACTIVE && blocked(tenant,family))
            throw CmsIntegrationException.conflict("PAN family has a permanent security block");
        if(CmsProfileRules.permanent(next.statusReason())) {
            db.update("insert into cms_pan_family_block(cms_institution_id,pan_family_id,reason,command_id) values(?,?,?,?) on conflict(cms_institution_id,pan_family_id) do nothing",
                    tenant,family,next.statusReason(),command);
        }
        db.update("update cms_card_profile set profile=?::jsonb,authorization_revision=?,updated_at=now() where cms_institution_id=? and cms_card_id=?",
                encode(next),revision,tenant,card);
        // Only lifecycle status is updated: do not overwrite PIN retries, ATC or Switch-owned counters.
        db.update("update card_data set status=?,updated_at=now() where auth_id=?",next.effectiveStatus().name(),existing.get("auth_id"));
    }
    @Transactional(readOnly=true)
    public JsonNode get(UUID tenant, UUID card) {
        Map<String,Object> row=row(tenant,card);
        return json.valueToTree(Map.of("cardId",card,"authorizationRevision",row.get("authorization_revision"),
                "profile",read(row.get("profile").toString()),"familyBlocked",blocked(tenant,(UUID)row.get("pan_family_id"))));
    }
    @Transactional(readOnly=true)
    public JsonNode command(UUID tenant, UUID command) {
        List<String> rows=db.query("select receipt::text from cms_card_command where cms_institution_id=? and command_id=?",
                (rs,n)->rs.getString(1),tenant,command);
        if(rows.isEmpty()) throw new CmsIntegrationException(404,"Command not found");
        return tree(rows.getFirst());
    }
    private JsonNode replay(UUID tenant, UUID command, String hash) {
        db.queryForList("select pg_advisory_xact_lock(hashtextextended(?,0))","cms-command:"+tenant+":"+command);
        List<Map<String,Object>> prior=db.queryForList("select request_hash,receipt::text from cms_card_command where cms_institution_id=? and command_id=?",tenant,command);
        if(prior.isEmpty()) return null;
        if(!hash.equals(prior.getFirst().get("request_hash"))) throw CmsIntegrationException.conflict("Command ID payload conflict");
        return tree(prior.getFirst().get("receipt").toString());
    }
    private JsonNode finish(UUID tenant, UUID command, String hash, Object receipt) {
        String body=encode(receipt);
        db.update("insert into cms_card_command(cms_institution_id,command_id,request_hash,receipt) values(?,?,?,?::jsonb)",tenant,command,hash,body);
        return tree(body);
    }
    private void lock(UUID tenant) { db.queryForList("select pg_advisory_xact_lock(hashtextextended(?,0))","cms-tenant:"+tenant); }
    private Map<String,Object> row(UUID tenant, UUID card) {
        List<Map<String,Object>> rows=db.queryForList("select * from cms_card_profile where cms_institution_id=? and cms_card_id=?",tenant,card);
        if(rows.isEmpty()) throw new CmsIntegrationException(404,"Card not registered");
        return rows.getFirst();
    }
    private Map<String,Object> mapping(UUID tenant) {
        List<Map<String,Object>> rows=db.queryForList("select * from cms_institution_mapping where cms_institution_id=?",tenant);
        if(rows.isEmpty()) throw CmsIntegrationException.bad("Institution mapping is required");
        return rows.getFirst();
    }
    private Map<String,Object> ensureInstitutionMapping(UUID tenant, String currency, int binOwner) {
        List<Map<String,Object>> rows=db.queryForList("select * from cms_institution_mapping where cms_institution_id=?",tenant);
        if(!rows.isEmpty() && ((Number)rows.getFirst().get("switch_institution_id")).intValue()!=binOwner) {
            if(!db.queryForList("select 1 from cms_card_profile where cms_institution_id=? limit 1",tenant).isEmpty())
                throw CmsIntegrationException.conflict("Existing card institution mapping differs from PAN BIN owner");
            if(!db.queryForList("select 1 from cms_institution_mapping where switch_institution_id=? and cms_institution_id<>?",binOwner,tenant).isEmpty())
                throw CmsIntegrationException.conflict("PAN BIN owner is already mapped to another CMS institution");
            Map<String,Object> configured=configuredInstitution(binOwner,currency);
            db.update("update cms_institution_mapping set switch_institution_id=?,home_country_code=?,currency_code=? where cms_institution_id=?",
                    binOwner, configured.get("country"), currency, tenant);
            rows=db.queryForList("select * from cms_institution_mapping where cms_institution_id=?",tenant);
        } else if(rows.isEmpty()) {
            Map<String,Object> configured=configuredInstitution(binOwner,currency);
            if(!db.queryForList("select 1 from cms_institution_mapping where switch_institution_id=?",binOwner).isEmpty())
                throw CmsIntegrationException.conflict("PAN BIN owner is already mapped to another CMS institution");
            db.update("insert into cms_institution_mapping(cms_institution_id,switch_institution_id,home_country_code,currency_code) values(?,?,?,?)",
                    tenant, binOwner, configured.get("country"), currency);
            rows=db.queryForList("select * from cms_institution_mapping where cms_institution_id=?",tenant);
        }
        return rows.getFirst();
    }
    private Map<String,Object> configuredInstitution(int binOwner, String currency) {
        var institutions=db.queryForList("select institution_country,currency from institutions where institution_id=?",binOwner);
        if(institutions.size()!=1) throw CmsIntegrationException.bad("BIN owner institution is not configured");
        var configured=institutions.getFirst();
        if(!currency.equals(Objects.toString(configured.get("currency"), "").trim()))
            throw CmsIntegrationException.bad("Profile currency differs from BIN owner institution currency");
        String country=Objects.toString(configured.get("institution_country"), "").trim();
        if(country.length()!=3) throw CmsIntegrationException.bad("BIN owner institution requires a three-character country code");
        return Map.of("country",country);
    }
    private boolean blocked(UUID tenant, UUID family) {
        return !db.queryForList("select 1 from cms_pan_family_block where cms_institution_id=? and pan_family_id=?",tenant,family).isEmpty();
    }
    private String encode(Object value) { try { return json.writeValueAsString(value); } catch(Exception e) { throw CmsIntegrationException.bad("Invalid payload"); } }
    private JsonNode tree(String value) { try { return json.readTree(value); } catch(Exception e) { throw CmsIntegrationException.unavailable("Invalid stored receipt"); } }
    private Profile read(String value) { try { return json.readValue(value,Profile.class); } catch(Exception e) { throw CmsIntegrationException.unavailable("Invalid stored profile"); } }
    private static String hash(String value) { try { return HexFormat.of().formatHex(MessageDigest.getInstance("SHA-256").digest(value.getBytes(StandardCharsets.UTF_8))); } catch(Exception e) { throw new IllegalStateException(e); } }
}
