package com.logicpoint.iso8583.cmsintegration;

import tools.jackson.databind.json.JsonMapper;

import static com.logicpoint.iso8583.cmsintegration.CmsContracts.*;
import tools.jackson.databind.*;
import jakarta.validation.Validator;
import io.swagger.v3.oas.annotations.Operation;
import io.swagger.v3.oas.annotations.responses.ApiResponse;
import io.swagger.v3.oas.annotations.responses.ApiResponses;
import io.swagger.v3.oas.annotations.security.SecurityRequirement;
import io.swagger.v3.oas.annotations.tags.Tag;
import org.springframework.web.bind.annotation.*;
import java.util.UUID;

@RestController
@RequestMapping("/internal/v1/institutions/{institutionId}")
@Tag(name = "CMS card authorization integration", description = "Authenticated CMS lifecycle and authorization-profile synchronization")
@SecurityRequirement(name = "cmsServiceBearer")
public class CmsCardController {
    private final CmsCardIntegrationService service;
    private final JsonMapper json;
    private final Validator validator;
    public CmsCardController(CmsCardIntegrationService service,JsonMapper json,Validator validator) {
        this.service=service; this.json=json.rebuild().enable(DeserializationFeature.FAIL_ON_UNKNOWN_PROPERTIES).build(); this.validator=validator;
    }
    @PutMapping("/cards/{cardId}/registration")
    @Operation(summary = "Register a CMS card", description = "Creates the Switch card_data identity and its initial PENDING authorization profile from recipient-encrypted card material.")
    @ApiResponses({@ApiResponse(responseCode = "200", description = "Registration receipt"),
            @ApiResponse(responseCode = "400", description = "Invalid profile, envelope, mapping, or idempotency key"),
            @ApiResponse(responseCode = "409", description = "Command or identity conflict")})
    public JsonNode register(@PathVariable UUID institutionId,@PathVariable UUID cardId,
            @RequestHeader("Idempotency-Key") UUID key,@RequestBody JsonNode body) {
        Registration request=parse(body,Registration.class); match(key,request.commandId());
        return service.register(institutionId,cardId,request);
    }
    @PutMapping("/cards/{cardId}/authorization-profile")
    @Operation(summary = "Update card authorization profile", description = "Atomically applies the complete status, representation, geography, channel and limit profile.")
    @ApiResponses({@ApiResponse(responseCode = "200", description = "Update receipt"),
            @ApiResponse(responseCode = "400", description = "Invalid profile or lifecycle transition"),
            @ApiResponse(responseCode = "409", description = "Stale revision or command conflict")})
    public JsonNode update(@PathVariable UUID institutionId,@PathVariable UUID cardId,
            @RequestHeader("Idempotency-Key") UUID key,@RequestBody JsonNode body) {
        Update request=parse(body,Update.class); match(key,request.commandId());
        return service.update(institutionId,cardId,request);
    }
    @PostMapping("/card-cutovers")
    @Operation(summary = "Cut over a replacement", description = "Atomically retires the predecessor and applies the successor profile.")
    @ApiResponses({@ApiResponse(responseCode = "200", description = "Cutover receipt"),
            @ApiResponse(responseCode = "400", description = "Invalid cutover"),
            @ApiResponse(responseCode = "409", description = "Revision, family, account or predecessor conflict")})
    public JsonNode cutover(@PathVariable UUID institutionId,@RequestHeader("Idempotency-Key") UUID key,
            @RequestBody JsonNode body) {
        Cutover request=parse(body,Cutover.class); match(key,request.commandId());
        return service.cutover(institutionId,request);
    }
    @GetMapping("/cards/{cardId}/authorization-profile")
    @Operation(summary = "Read the applied authorization profile")
    @ApiResponses({@ApiResponse(responseCode = "200", description = "Current profile"),
            @ApiResponse(responseCode = "404", description = "Card is not registered")})
    public JsonNode get(@PathVariable UUID institutionId,@PathVariable UUID cardId) {
        return service.get(institutionId,cardId);
    }
    @GetMapping("/card-commands/{commandId}")
    @Operation(summary = "Read a command receipt", description = "Used by CMS to reconcile a timeout or retry without repeating a lifecycle effect.")
    @ApiResponses({@ApiResponse(responseCode = "200", description = "Stored receipt"),
            @ApiResponse(responseCode = "404", description = "Command is not known")})
    public JsonNode command(@PathVariable UUID institutionId,@PathVariable UUID commandId) {
        return service.command(institutionId,commandId);
    }
    private <T> T parse(JsonNode body,Class<T> type) {
        try {
            T result=json.treeToValue(body,type);
            if(result==null || !validator.validate(result).isEmpty()) throw new IllegalArgumentException();
            return result;
        } catch(Exception e) { throw CmsIntegrationException.bad("Invalid request fields"); }
    }
    private void match(UUID header,UUID command) { if(!header.equals(command)) throw CmsIntegrationException.bad("Idempotency-Key must equal commandId"); }
}
