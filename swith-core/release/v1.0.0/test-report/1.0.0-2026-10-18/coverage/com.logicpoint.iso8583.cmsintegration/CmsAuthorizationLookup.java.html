package com.logicpoint.iso8583.cmsintegration;

import static com.logicpoint.iso8583.cmsintegration.CmsContracts.*;
import tools.jackson.databind.json.JsonMapper;
import com.logicpoint.iso8583.context.RequestContext;
import com.logicpoint.iso8583.entity.CardData;
import com.logicpoint.iso8583.repository.CardDataRepository;
import com.logicpoint.iso8583.service.CardDataEncryptionService;
import jakarta.persistence.EntityManager;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import java.util.*;

@Service
public class CmsAuthorizationLookup {
    private final JdbcTemplate db;
    private final JsonMapper json;
    private final CardDataRepository cards;
    private final CardDataEncryptionService crypto;
    private final EntityManager em;
    public CmsAuthorizationLookup(JdbcTemplate db,JsonMapper json,CardDataRepository cards,
            CardDataEncryptionService crypto,EntityManager em) {
        this.db=db; this.json=json; this.cards=cards; this.crypto=crypto; this.em=em;
    }
    @Transactional(readOnly=true)
    public CardData lookup(RequestContext request, String hmac, int institution) {
        List<Map<String,Object>> rows=db.queryForList("""
            select p.*,m.home_country_code,exists(select 1 from cms_pan_family_block b
                where b.cms_institution_id=p.cms_institution_id and b.pan_family_id=p.pan_family_id) as family_blocked
            from cms_card_profile p join cms_institution_mapping m using(cms_institution_id)
            where m.switch_institution_id=? and p.pan_lookup_hmac=?
            """,institution,hmac);
        if(rows.isEmpty()) return null; // Legacy cards continue through their existing path.
        String expiry=request.getFields().get(14), sequence=request.getFields().get(23);
        if(expiry==null || !expiry.matches("[0-9]{2}(0[1-9]|1[0-2])")) throw CmsIntegrationException.bad("Expiry evidence required");
        if(sequence!=null && !sequence.matches("[0-9]{3}")) throw CmsIntegrationException.bad("Invalid sequence evidence");
        List<Map<String,Object>> matches=rows.stream().filter(r->expiry.equals(r.get("expiry_yymm")))
                .filter(r->sequence==null || sequence.equals(r.get("physical_sequence"))).toList();
        if(matches.size()!=1) throw CmsIntegrationException.bad("Card evidence is unknown or ambiguous");
        Map<String,Object> row=matches.getFirst();
        Profile profile;
        try { profile=json.readValue(row.get("profile").toString(),Profile.class); }
        catch(Exception e) { throw CmsIntegrationException.unavailable("Authorization profile is unreadable"); }
        CmsAuthorizationPolicy.enforce(profile,request,row.get("home_country_code").toString(),Boolean.TRUE.equals(row.get("family_blocked")));
        CardData card=cards.findById(((Number)row.get("auth_id")).longValue()).orElseThrow(()->CmsIntegrationException.unavailable("Card material missing"));
        // Do not hydrate clear fields into a managed entity: Hibernate dirty checking could persist them.
        em.detach(card);
        crypto.hydrateDecryptedFields(card,institution);
        card.setStatus(profile.effectiveStatus().name());
        request.setAttribute("cmsCardId",row.get("cms_card_id"));
        request.setAttribute("cmsAuthorizationRevision",row.get("authorization_revision"));
        return card;
    }
    public boolean isManaged(String hmac) {
        return !db.queryForList("select 1 from cms_card_profile where pan_lookup_hmac=? limit 1",hmac).isEmpty();
    }
    public boolean isManaged(Long authId) {
        return !db.queryForList("select 1 from cms_card_profile where auth_id=?",authId).isEmpty();
    }
    public void updatePinCounters(CardData card) {
        db.update("update card_data set pin_retry_count=?,pin_blocked=?,updated_at=now() where auth_id=?",
                card.getPin_retry_count(),card.getPin_blocked(),card.getAuth_id());
    }
}
