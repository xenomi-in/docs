package com.logicpoint.iso8583.cmsintegration;

import static com.logicpoint.iso8583.cmsintegration.CmsPinContracts.*;
import tools.jackson.databind.json.JsonMapper;
import com.logicpoint.iso8583.entity.CardData;
import com.logicpoint.iso8583.repository.CardDataRepository;
import com.logicpoint.iso8583.service.CardDataEncryptionService;
import jakarta.persistence.EntityManager;
import java.util.*;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.stereotype.Component;

/** Access only while holding the institution/family transaction locks. */
@Component
public class CmsPinState {
    final JdbcTemplate db;
    final JsonMapper json;
    final CardDataRepository cards;
    final CardDataEncryptionService crypto;
    final EntityManager em;
    public CmsPinState(JdbcTemplate db,JsonMapper json,CardDataRepository cards,CardDataEncryptionService crypto,EntityManager em) {
        this.db=db; this.json=json; this.cards=cards; this.crypto=crypto; this.em=em;
    }
    public record Family(UUID institution,UUID family) {}
    JdbcTemplate jdbc() { return db; }
    JsonMapper mapper() { return json; }
    public Family family(Long authId) {
        var rows=db.queryForList("select cms_institution_id,pan_family_id from cms_card_profile where auth_id=?",authId);
        return rows.isEmpty()?null:new Family((UUID)rows.getFirst().get("cms_institution_id"),(UUID)rows.getFirst().get("pan_family_id"));
    }
    public void lock(Family f) {
        db.queryForList("select pg_advisory_xact_lock(hashtextextended(?,0))","cms-tenant:"+f.institution());
        lockFamily(f);
    }
    public void lockFamily(Family f) {
        db.queryForList("select pg_advisory_xact_lock(hashtextextended(?,0))","cms-pin-family:"+f.institution()+":"+f.family());
    }
    public boolean configured(Family f) {
        return !db.queryForList("select 1 from cms_pin_profile where cms_institution_id=? and pan_family_id=?",f.institution(),f.family()).isEmpty();
    }
    public Map<String,Object> profile(Family f) {
        var rows=db.queryForList("select * from cms_pin_profile where cms_institution_id=? and pan_family_id=? for update",f.institution(),f.family());
        if(rows.isEmpty()) throw CmsIntegrationException.conflict("PIN profile is not configured");
        return rows.getFirst();
    }
    public PinProfile decode(Map<String,Object> row) {
        try{return json.readValue(row.get("profile").toString(),PinProfile.class);}
        catch(Exception e){throw CmsIntegrationException.unavailable("Invalid stored PIN profile");}
    }
    public CardData card(Family f) {
        var ids=db.query("select auth_id from cms_card_profile where cms_institution_id=? and pan_family_id=? order by auth_id limit 1",
                (rs,n)->rs.getLong(1),f.institution(),f.family());
        if(ids.isEmpty()) throw CmsIntegrationException.conflict("PAN family is not registered");
        CardData card=cards.findById(ids.getFirst()).orElseThrow();
        em.refresh(card); em.detach(card);
        Integer institution=db.queryForObject("select switch_institution_id from cms_institution_mapping where cms_institution_id=?",Integer.class,f.institution());
        crypto.hydrateDecryptedFields(card,institution);
        return card;
    }
    public boolean blocked(Family f) {
        return !db.queryForList("""
            select 1 from cms_card_profile p join card_data d on d.auth_id=p.auth_id
            where p.cms_institution_id=? and p.pan_family_id=?
              and (d.pin_blocked=true or d.pin_retry_count>=coalesce(d.max_pin_retry,3))
            """,f.institution(),f.family()).isEmpty()
            || !db.queryForList("select 1 from cms_pan_family_block where cms_institution_id=? and pan_family_id=?",f.institution(),f.family()).isEmpty();
    }
    public void counters(Family f,boolean success) {
        // All siblings share the most restrictive retry state.
        if(success) {
            db.update("update card_data set pin_retry_count=0,pin_blocked=false where auth_id in (select auth_id from cms_card_profile where cms_institution_id=? and pan_family_id=?)",f.institution(),f.family());
        } else {
            db.update("""
                with attempts as (
                    select coalesce(max(d.pin_retry_count),0)+1 as n
                    from card_data d join cms_card_profile p on p.auth_id=d.auth_id
                    where p.cms_institution_id=? and p.pan_family_id=?
                )
                update card_data d set pin_retry_count=a.n,
                    pin_blocked=(a.n>=coalesce(d.max_pin_retry,3)),updated_at=now()
                from attempts a where d.auth_id in
                    (select auth_id from cms_card_profile where cms_institution_id=? and pan_family_id=?)
                """,f.institution(),f.family(),f.institution(),f.family());
        }
    }
}
