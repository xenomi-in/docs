package com.logicpoint.iso8583.cmsintegration;

import jakarta.validation.Valid;
import jakarta.validation.constraints.*;
import java.util.List;
import java.util.UUID;

public final class CmsContracts {
    private CmsContracts() {}
    public enum Status { PENDING, ACTIVE, FROZEN, BLOCKED, EXPIRED, CLOSED, REPLACED }
    public enum Geography { DOMESTIC, INTERNATIONAL }
    public enum Channel { ATM, POS_CONTACT, ECOM, MOTO, CONTACTLESS_POS }
    public record Control(@NotNull Geography geography, @NotNull Channel channel,
            boolean enabled, @PositiveOrZero long perTransactionLimitMinor,
            boolean contactlessEnabled, @PositiveOrZero long contactlessLimitMinor) {}
    public record Profile(@NotNull Status effectiveStatus, @Size(max=40) String statusReason,
            @Positive long lifecycleVersion, @Positive int controlVersion,
            boolean internationalEnabled, boolean physicalEnabled, boolean digitalEnabled,
            @Pattern(regexp="[0-9]{3}") @NotNull String currencyCode,
            @NotNull @Size(min=10,max=10) List<@Valid Control> controls) {}
    // Provisioning material is encrypted to the Switch recipient key. Never retained in command receipts.
    public record Registration(@NotNull UUID commandId, @NotNull UUID accountReferenceId, UUID predecessorCardId,
            @NotNull UUID panFamilyId, @Pattern(regexp="[0-9]{3}") String physicalSequence,
            @NotBlank @Size(max=16384) String protectedMaterial,
            @NotNull @Valid Profile profile) {
        @Override public String toString() { return "Registration[commandId=" + commandId + "]"; }
    }
    public record Update(@NotNull UUID commandId, @Positive long expectedRevision,
            @Positive long authorizationRevision, @Pattern(regexp="[0-9]{3}") String physicalSequence,
            @NotNull @Valid Profile profile) {
        public Update(UUID commandId,long expectedRevision,long authorizationRevision,Profile profile) {
            this(commandId,expectedRevision,authorizationRevision,null,profile);
        }
    }
    public record Cutover(@NotNull UUID commandId, @NotNull UUID predecessorCardId,
            @NotNull UUID successorCardId, @Positive long predecessorExpectedRevision,
            @Positive long successorExpectedRevision, @NotNull @Valid Profile predecessor,
            @NotNull @Valid Profile successor) {}
    public record Receipt(UUID commandId, UUID cardId, boolean applied, long appliedRevision,
            Status effectiveStatus) {}
}
