package com.logicpoint.iso8583.cmsintegration;

import static com.logicpoint.iso8583.cmsintegration.CmsPinContracts.*;
import com.logicpoint.iso8583.service.KeyManagementService;
import com.logicpoint.iso8583.responsecode.InternalResCode;
import tools.jackson.databind.JsonNode;
import java.util.*;
import java.nio.charset.StandardCharsets;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

@Service
public class CmsPinTransactionService {
    private final CmsPinState state;
    private final CmsPinHsm hsm;
    private final CmsPinKeys keys;
    private final KeyManagementService networkKeys;
    private final CmsPinOperationService operations;
    public CmsPinTransactionService(CmsPinState state,CmsPinHsm hsm,CmsPinKeys keys,
            KeyManagementService networkKeys,CmsPinOperationService operations) {
        this.state=state;this.hsm=hsm;this.keys=keys;this.networkKeys=networkKeys;this.operations=operations;
    }
    public boolean managed(Long authId) { return state.family(authId)!=null; }

    @Transactional
    public InternalResCode verify(Long authId,String pinBlock) {
        var f=state.family(authId);
        if(f==null) return null;
        state.lockFamily(f);
        // Unenrolled existing CMS cards retain their prior verification path.
        if(!state.configured(f)) return null;
        var row=state.profile(f);
        if(((Number)row.get("active_pin_version")).longValue()==0) return InternalResCode.RESTRICTED_CARD;
        if(state.blocked(f)) return InternalResCode.RESTRICTED_CARD;
        var profile=state.decode(row);
        var card=state.card(f);
        boolean valid=hsm.verify(card.getPan(),pinBlock,networkKeys.getWorkingKeyForCard(card.getPan(),"ZPK"),
                keys.resolve(f.institution(),profile.keyBindingReference(),"PVK"),profile,card.getPvv_offest());
        state.counters(f,valid);
        return valid?InternalResCode.APPROVED:InternalResCode.INVALID_PIN;
    }

    @Transactional
    public JsonNode change(Long authId,String transactionId,String currentBlock,String newBlock) {
        var f=state.family(authId);
        if(f==null) throw CmsIntegrationException.bad("CMS card required");
        if(transactionId==null || transactionId.isBlank()) throw CmsIntegrationException.bad("Stable transaction identity required");
        UUID id=UUID.nameUUIDFromBytes(("network-pin:"+f.institution()+":"+authId+":"+transactionId).getBytes(StandardCharsets.UTF_8));
        String hash=CmsPinGrantVerifier.hash("network:"+authId+":"+transactionId+":"+currentBlock+":"+newBlock);
        state.jdbc().queryForList("select pg_advisory_xact_lock(hashtextextended(?,0))","cms-pin-operation:"+f.institution()+":"+id);
        state.lock(f);
        var prior=state.jdbc().queryForList("select request_hash,receipt::text from cms_pin_operation where cms_institution_id=? and operation_id=?",f.institution(),id);
        if(!prior.isEmpty()) {
            if(!hash.equals(prior.getFirst().get("request_hash"))) throw CmsIntegrationException.conflict("Transaction PIN payload conflict");
            try { return state.mapper().readTree(prior.getFirst().get("receipt").toString()); }
            catch(Exception e) { throw CmsIntegrationException.unavailable("PIN receipt is unreadable"); }
        }
        var row=state.profile(f);
        long version=((Number)row.get("active_pin_version")).longValue();
        if(version==0) throw CmsIntegrationException.conflict("PIN is not established");
        var profile=state.decode(row);
        var command=new PinCommand(id,f.institution(),f.family(),((Number)row.get("policy_profile_version")).longValue(),
                version,Operation.CHANGE_PIN,null,null);
        CmsPinOperationExecutor.Result result;
        if(state.blocked(f)) result=CmsPinOperationExecutor.Result.failed("PIN_BLOCKED");
        else {
            var card=state.card(f);
            String pek=networkKeys.getWorkingKeyForCard(card.getPan(),"ZPK");
            String pvk=keys.resolve(f.institution(),profile.keyBindingReference(),"PVK");
            if(!hsm.verify(card.getPan(),currentBlock,pek,pvk,profile,card.getPvv_offest())) {
                state.counters(f,false);
                result=CmsPinOperationExecutor.Result.failed("INVALID_PIN");
            } else result=CmsPinOperationExecutor.Result.completed(hsm.derive(card.getPan(),newBlock,pek,pvk,profile),null);
        }
        return operations.commitNetwork(command,hash,result);
    }
}
