package com.logicpoint.iso8583.cmsintegration;

public class CmsIntegrationException extends RuntimeException {
    public final int status;
    public CmsIntegrationException(int status, String message) { super(message); this.status=status; }
    public static CmsIntegrationException bad(String message) { return new CmsIntegrationException(400,message); }
    public static CmsIntegrationException conflict(String message) { return new CmsIntegrationException(409,message); }
    public static CmsIntegrationException unavailable(String message) { return new CmsIntegrationException(503,message); }
}
