package com.logicpoint.iso8583.cmsintegration;

import static com.logicpoint.iso8583.cmsintegration.CmsPinContracts.*;
import org.springframework.stereotype.Component;

@Component
public class CmsFuturexPinExecutor implements CmsPinOperationExecutor {
    private final CmsPinState state;
    private final CmsPinKeys keys;
    private final CmsPinHsm hsm;
    public CmsFuturexPinExecutor(CmsPinState state,CmsPinKeys keys,CmsPinHsm hsm) {this.state=state;this.keys=keys;this.hsm=hsm;}
    @Override public Result execute(PinCommand command,PinProfile profile,PinMaterial material) {
        hsm.supported(profile);
        if(command.operation()==Operation.GENERATE_INITIAL_PIN)
            throw CmsIntegrationException.unavailable("PIN generation and mailer integration is not configured");
        if(material==null || material.newPin()==null || material.newPin().format()!=PinBlockFormat.ISO_0
                || (material.currentPin()!=null && material.currentPin().format()!=PinBlockFormat.ISO_0))
            throw CmsIntegrationException.bad("PIN adapter supports encrypted ISO_0 blocks only");
        var family=new CmsPinState.Family(command.institutionId(),command.panFamilyReference());
        var card=state.card(family);
        String pvk=keys.resolve(command.institutionId(),profile.keyBindingReference(),"PVK");
        String newKey=keys.resolve(command.institutionId(),material.newPin().keyBindingReference(),"INPUT");
        if(command.operation()==Operation.CHANGE_PIN) {
            if(state.blocked(family)) return Result.failed("PIN_BLOCKED");
            if(material.currentPin()==null) throw CmsIntegrationException.bad("Current PIN is required");
            String oldKey=keys.resolve(command.institutionId(),material.currentPin().keyBindingReference(),"INPUT");
            if(!hsm.verify(card.getPan(),material.currentPin().encryptedPinBlock(),oldKey,pvk,profile,card.getPvv_offest())) {
                state.counters(family,false);
                return Result.failed("INVALID_PIN");
            }
        }
        // GOFF is deterministic for the supplied encrypted PIN and pinned PVK.
        // It has no mailer/random-generation external effect to duplicate on retry.
        return Result.completed(hsm.derive(card.getPan(),material.newPin().encryptedPinBlock(),newKey,pvk,profile),null);
    }
}
