package com.logicpoint.iso8583.cmsintegration;

import static com.logicpoint.iso8583.cmsintegration.CmsContracts.*;
import java.util.HashSet;
import java.util.Set;

public final class CmsProfileRules {
    private CmsProfileRules() {}
    public static boolean permanent(String reason) {
        return reason != null && Set.of("LOST","STOLEN","COMPROMISED","FRAUD_SUSPECTED").contains(reason);
    }
    public static boolean terminal(Status status) {
        return status==Status.CLOSED || status==Status.REPLACED || status==Status.EXPIRED;
    }
    public static void validate(Profile profile) {
        if (profile==null || profile.effectiveStatus()==null || profile.controls()==null
                || profile.controls().size()!=10 || profile.lifecycleVersion()<1 || profile.controlVersion()<1
                || profile.currencyCode()==null || !profile.currencyCode().matches("[0-9]{3}"))
            throw CmsIntegrationException.bad("Complete authorization profile is required");
        Set<String> dimensions=new HashSet<>();
        if(profile.effectiveStatus()==Status.ACTIVE && profile.statusReason()!=null)
            throw CmsIntegrationException.bad("ACTIVE must not carry a restriction reason");
        for(Control c:profile.controls()) {
            if(c==null || c.geography()==null || c.channel()==null || c.perTransactionLimitMinor()<0
                    || c.contactlessLimitMinor()<0 || c.contactlessLimitMinor()>c.perTransactionLimitMinor()
                    || !dimensions.add(c.geography()+":"+c.channel()))
                throw CmsIntegrationException.bad("Invalid or duplicate control dimension");
        }
        if (permanent(profile.statusReason()) && profile.effectiveStatus()!=Status.BLOCKED
                && !terminal(profile.effectiveStatus()))
            throw CmsIntegrationException.bad("Permanent security reason requires a restrictive status");
    }
    public static void transition(Profile old, Profile next, long current, long expected, long revision) {
        validate(next);
        if(current!=expected || revision!=Math.addExact(current,1))
            throw CmsIntegrationException.conflict("Stale or nonconsecutive authorization revision");
        if(terminal(old.effectiveStatus()) && next.effectiveStatus()!=old.effectiveStatus())
            throw CmsIntegrationException.conflict("Terminal card cannot be reopened");
        if(permanent(old.statusReason()) && !permanent(next.statusReason()) && !terminal(next.effectiveStatus()))
            throw CmsIntegrationException.conflict("Permanent security restriction cannot be released");
        if(next.lifecycleVersion()<old.lifecycleVersion() || next.controlVersion()<old.controlVersion())
            throw CmsIntegrationException.conflict("Domain versions cannot decrease");
        boolean lifecycleChanged=old.effectiveStatus()!=next.effectiveStatus()
                || !java.util.Objects.equals(old.statusReason(),next.statusReason())
                || old.physicalEnabled()!=next.physicalEnabled() || old.digitalEnabled()!=next.digitalEnabled();
        boolean controlsChanged=old.internationalEnabled()!=next.internationalEnabled()
                || !old.controls().equals(next.controls()) || !old.currencyCode().equals(next.currencyCode());
        if(lifecycleChanged && next.lifecycleVersion()<=old.lifecycleVersion()
                || controlsChanged && next.controlVersion()<=old.controlVersion())
            throw CmsIntegrationException.conflict("Changed domains require a newer domain version");
    }
}
