package com.logicpoint.iso8583.cmsintegration;

import static com.logicpoint.iso8583.cmsintegration.CmsPinContracts.*;
import com.logicpoint.iso8583.hsm.*;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Component;

/** Stateless VISA/ISO-0 operations using the existing Futurex RPIN/GOFF dialect. */
@Component
public class CmsPinHsm {
    private final SimpleNettyHsmClient client;
    private final boolean enabled;
    public CmsPinHsm(SimpleNettyHsmClient client,@Value("${switch.cms.pin-hsm-enabled:false}") boolean enabled) {
        this.client=client; this.enabled=enabled;
    }
    public void supported(PinProfile profile) {
        if(!enabled) throw CmsIntegrationException.unavailable("CMS PIN HSM execution is disabled");
        if(profile.verificationMethod()!=VerificationMethod.VISA_PVV || profile.pvki()==null
                || profile.pvki()<0 || profile.pvki()>6)
            throw CmsIntegrationException.bad("PIN adapter requires VISA_PVV with explicit PVKI 0..6");
    }
    public boolean verify(String pan,String block,String pek,String pvk,PinProfile profile,String pvv) {
        supported(profile);
        if(pvv==null || !pvv.matches("[0-9]{4}")) throw CmsIntegrationException.conflict("Active Visa PVV is unavailable");
        FuturexHsmCmd cmd=base("RPIN",pan,block,pek,pvk,profile);
        cmd.addField("FW",block); cmd.addField("BX",pvv);
        var response=send(cmd,"RPIN");
        String result=response.getField("BB");
        if(!"Y".equals(result) && !"N".equals(result)) throw CmsIntegrationException.unavailable("Invalid PIN verification response");
        return "Y".equals(result);
    }
    public String derive(String pan,String block,String pek,String pvk,PinProfile profile) {
        supported(profile);
        String value=send(base("GOFF",pan,block,pek,pvk,profile),"GOFF").getField("BX");
        if(value==null || !value.matches("[0-9]{4}")) throw CmsIntegrationException.unavailable("Invalid PIN derivation response");
        return value;
    }
    private FuturexHsmCmd base(String operation,String pan,String block,String pek,String pvk,PinProfile profile) {
        if(pan==null || !pan.matches("[0-9]{16}|[0-9]{19}") || block==null || !block.matches("[0-9A-Fa-f]{16}")
                || pek==null || !pek.matches("[0-9A-Fa-f]+") || pvk==null || !pvk.matches("[0-9A-Fa-f]+"))
            throw CmsIntegrationException.bad("Invalid ISO-0 PIN input or key");
        String withoutCheck=pan.substring(0,pan.length()-1);
        FuturexHsmCmd cmd=new FuturexHsmCmd(operation);
        cmd.addField("BF","3"); cmd.addField("BY",profile.pvki().toString());
        cmd.addField("AW","1"); cmd.addField("AL",block); cmd.addField("AX",pek); cmd.addField("CA",pvk);
        cmd.addField("BZ",withoutCheck.substring(withoutCheck.length()-11));
        cmd.addField("AK",withoutCheck.substring(withoutCheck.length()-12));
        return cmd;
    }
    private FuturexHsmCmd send(FuturexHsmCmd command,String expected) {
        try {
            FuturexHsmCmd response=FuturexHsmCmd.parse(client.sendMessage(command.build()));
            // Parser stores the first AO token in commandId, not the field map.
            if(!("AO"+expected).equals(response.getCommandId())) throw new IllegalArgumentException();
            return response;
        } catch(Exception e) { throw CmsIntegrationException.unavailable("PIN HSM command failed"); }
    }
}
