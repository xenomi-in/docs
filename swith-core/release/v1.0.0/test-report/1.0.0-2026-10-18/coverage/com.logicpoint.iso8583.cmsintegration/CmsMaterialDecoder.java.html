package com.logicpoint.iso8583.cmsintegration;

import tools.jackson.databind.JsonNode;
import tools.jackson.databind.json.JsonMapper;
import com.nimbusds.jose.*;
import com.nimbusds.jose.crypto.RSADecrypter;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.core.io.ResourceLoader;
import org.springframework.stereotype.Component;
import java.security.KeyStore;
import java.security.interfaces.RSAPrivateKey;
import java.time.Instant;
import java.util.Set;
import java.util.UUID;

@Component
public class CmsMaterialDecoder {
    private final ResourceLoader resources;
    private final JsonMapper json;
    private final String path, password, alias;
    public CmsMaterialDecoder(ResourceLoader resources, JsonMapper json,
            @Value("${switch.cms.material-keystore:}") String path,
            @Value("${switch.cms.material-keystore-password:}") String password,
            @Value("${switch.cms.material-key-alias:cms-onboarding}") String alias) {
        this.resources=resources; this.json=json; this.path=path; this.password=password; this.alias=alias;
    }
    public JsonNode decode(String compact, UUID tenant, UUID card, UUID command) {
        if (path.isBlank()) throw CmsIntegrationException.unavailable("Provisioning recipient key is not configured");
        try {
            JWEObject jwe=JWEObject.parse(compact);
            if (!JWEAlgorithm.RSA_OAEP_256.equals(jwe.getHeader().getAlgorithm())
                    || !EncryptionMethod.A256GCM.equals(jwe.getHeader().getEncryptionMethod())
                    || jwe.getHeader().getCompressionAlgorithm()!=null) throw new IllegalArgumentException();
            KeyStore store=KeyStore.getInstance("PKCS12");
            try(var input=resources.getResource(path).getInputStream()) { store.load(input,password.toCharArray()); }
            jwe.decrypt(new RSADecrypter((RSAPrivateKey)store.getKey(alias,password.toCharArray())));
            JsonNode data=json.readTree(jwe.getPayload().toString());
            Set<String> allowed=Set.of("institutionId","cardId","commandId","expiresAt","pan","expiryYYMM",
                    "cardType","cardScheme","cvkReference");
            data.propertyNames().iterator().forEachRemaining(k -> { if(!allowed.contains(k)) throw new IllegalArgumentException(); });
            if(data.hasNonNull("cvkReference") && !data.path("cvkReference").asText().matches("[A-Za-z0-9_.:-]{1,100}")) throw new IllegalArgumentException();
            Instant expiry=Instant.parse(data.path("expiresAt").asText());
            if (!tenant.toString().equals(data.path("institutionId").asText())
                    || !card.toString().equals(data.path("cardId").asText())
                    || !command.toString().equals(data.path("commandId").asText())
                    || !expiry.isAfter(Instant.now()) || expiry.isAfter(Instant.now().plusSeconds(300))
                    || !data.path("pan").asText().matches("[0-9]{16}|[0-9]{19}")
                    || !data.path("expiryYYMM").asText().matches("[0-9]{2}(0[1-9]|1[0-2])")
                    || !data.path("cardType").asText().matches("[CDP]")
                    || !data.path("cardScheme").asText().matches("[VMR]")) throw new IllegalArgumentException();
            return data;
        } catch (Exception ex) {
            // Do not attach decryption/parser exceptions: they can contain provisioning material.
            throw CmsIntegrationException.bad("Invalid or expired protected provisioning material");
        }
    }
}
