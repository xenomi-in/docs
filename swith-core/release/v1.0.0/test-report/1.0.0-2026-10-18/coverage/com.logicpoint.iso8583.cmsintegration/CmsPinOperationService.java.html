package com.logicpoint.iso8583.cmsintegration;

import tools.jackson.databind.json.JsonMapper;

import tools.jackson.databind.*;
import com.logicpoint.iso8583.entity.CardData;
import com.logicpoint.iso8583.repository.CardDataRepository;
import com.logicpoint.iso8583.service.CardDataEncryptionService;
import org.springframework.beans.factory.ObjectProvider;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import static com.logicpoint.iso8583.cmsintegration.CmsPinContracts.*;
import java.time.Instant;
import java.util.*;

@Service
public class CmsPinOperationService {
    private final JdbcTemplate db;
    private final JsonMapper json;
    private final CardDataRepository cards;
    private final CardDataEncryptionService crypto;
    private final CmsPinOperationExecutor executor;
    private final CmsPinMaterialDecoder materials;
    private final CmsPinAuthorizationVerifier authorization;

    public CmsPinOperationService(JdbcTemplate db,JsonMapper json,CardDataRepository cards,
            CardDataEncryptionService crypto,ObjectProvider<CmsPinOperationExecutor> executors,
            CmsPinMaterialDecoder materials,ObjectProvider<CmsPinAuthorizationVerifier> authorizations) {
        this.db=db; this.json=json; this.cards=cards; this.crypto=crypto;
        this.executor=executors.getIfAvailable();
        this.materials=materials;
        this.authorization=authorizations.getIfAvailable();
    }

    @Transactional
    public JsonNode submit(UUID tenant,UUID family,PinCommand command) {
        if(!tenant.equals(command.institutionId()) || !family.equals(command.panFamilyReference()))
            throw CmsIntegrationException.bad("Command institution and PAN family must match the resource path");
        String fingerprint=hash(command);
        db.queryForList("select pg_advisory_xact_lock(hashtextextended(?,0))","cms-pin-operation:"+tenant+":"+command.operationId());
        // Coordinate registration/cutover while collecting family members for persistence.
        db.queryForList("select pg_advisory_xact_lock(hashtextextended(?,0))","cms-tenant:"+tenant);
        lock(tenant,family);
        List<Map<String,Object>> prior=db.queryForList("select request_hash,receipt::text from cms_pin_operation where cms_institution_id=? and operation_id=?",tenant,command.operationId());
        if(!prior.isEmpty()) {
            if(!fingerprint.equals(prior.getFirst().get("request_hash"))) throw CmsIntegrationException.conflict("Operation ID payload conflict");
            return tree(prior.getFirst().get("receipt").toString());
        }
        if(db.queryForList("select 1 from cms_card_profile where cms_institution_id=? and pan_family_id=? limit 1",tenant,family).isEmpty())
            throw new CmsIntegrationException(404,"PAN family is not registered");
        PinOperationReceipt receipt=command.operation()==Operation.CONFIGURE_PIN_PROFILE
                ? configure(tenant,family,command) : execute(tenant,family,command);
        String encoded=encode(receipt);
        db.update("insert into cms_pin_operation(cms_institution_id,operation_id,pan_family_id,operation_type,request_hash,status,receipt) values(?,?,?,?,?,?,?::jsonb)",
                tenant,command.operationId(),family,command.operation().name(),fingerprint,receipt.status().name(),encoded);
        return tree(encoded);
    }

    private PinOperationReceipt configure(UUID tenant,UUID family,PinCommand command) {
        if(command.profile()==null || command.protectedPinMaterial()!=null)
            throw CmsIntegrationException.bad("CONFIGURE_PIN_PROFILE requires profile and no protectedPinMaterial");
        List<Map<String,Object>> rows=profile(tenant,family);
        long currentPin=rows.isEmpty()?0:number(rows.getFirst(),"active_pin_version");
        long currentPolicy=rows.isEmpty()?0:number(rows.getFirst(),"policy_profile_version");
        expected(command,currentPin);
        if(command.policyProfileVersion()<=currentPolicy) throw CmsIntegrationException.conflict("Policy/profile version must increase");
        if(currentPin>0 && !command.profile().equals(read(rows.getFirst().get("profile").toString(),PinProfile.class)))
            throw CmsIntegrationException.conflict("Active PIN profile is immutable; cryptographic migration is required");
        if(command.profile().establishment()==Establishment.CUSTOMER_SELECTED && command.profile().delivery()!=Delivery.NONE)
            throw CmsIntegrationException.bad("Customer-selected PIN cannot request a PIN mailer");
        if(command.profile().establishment()==Establishment.HSM_GENERATED && command.profile().delivery()!=Delivery.PIN_MAILER)
            throw CmsIntegrationException.bad("HSM-generated initial PIN requires PIN_MAILER delivery");
        db.update("insert into cms_pin_profile(cms_institution_id,pan_family_id,policy_profile_version,active_pin_version,profile) values(?,?,?,?,?::jsonb) on conflict(cms_institution_id,pan_family_id) do update set policy_profile_version=excluded.policy_profile_version,profile=excluded.profile,updated_at=now()",
                tenant,family,command.policyProfileVersion(),currentPin,encode(command.profile()));
        return receipt(command,Status.COMPLETED,currentPin,null,null);
    }

    private PinOperationReceipt execute(UUID tenant,UUID family,PinCommand command) {
        if(command.profile()!=null) throw CmsIntegrationException.bad("Only CONFIGURE_PIN_PROFILE carries profile");
        List<Map<String,Object>> rows=profile(tenant,family);
        if(rows.isEmpty()) throw CmsIntegrationException.conflict("PIN profile is not configured");
        Map<String,Object> row=rows.getFirst();
        long policy=number(row,"policy_profile_version"), current=number(row,"active_pin_version");
        if(command.policyProfileVersion()!=policy) throw CmsIntegrationException.conflict("Stale policy/profile version");
        expected(command,current);
        PinProfile configured=read(row.get("profile").toString(),PinProfile.class);
        if(command.operation()==Operation.INITIALIZE_SELECTED_PIN && configured.establishment()!=Establishment.CUSTOMER_SELECTED)
            throw CmsIntegrationException.conflict("Configured profile does not allow customer PIN selection");
        if(command.operation()==Operation.GENERATE_INITIAL_PIN && configured.establishment()!=Establishment.HSM_GENERATED)
            throw CmsIntegrationException.conflict("Configured profile does not allow generated initial PIN");
        if((command.operation()==Operation.INITIALIZE_SELECTED_PIN || command.operation()==Operation.GENERATE_INITIAL_PIN) && current!=0)
            throw CmsIntegrationException.conflict("Initial PIN is already established");
        if((command.operation()==Operation.CHANGE_PIN || command.operation()==Operation.RESET_PIN) && current==0)
            throw CmsIntegrationException.conflict("PIN must be established before change or reset");
        PinMaterial material=null;
        if(command.operation()==Operation.GENERATE_INITIAL_PIN) {
            if(command.protectedPinMaterial()!=null)
                throw CmsIntegrationException.bad("GENERATE_INITIAL_PIN does not accept protectedPinMaterial");
        } else {
            material=materials.decode(command);
        }
        if(executor==null || (material!=null && authorization==null))
            throw CmsIntegrationException.unavailable("PIN HSM or customer authorization integration is not configured");
        if(material!=null) authorization.verify(command,material);
        CmsPinOperationExecutor.Result result=executor.execute(command,configured,material);
        if(result==null || !result.completed())
            throw CmsIntegrationException.unavailable("PIN execution outcome requires reconciliation; retry the same operation ID");
        return applyResult(command,result,current);
    }

    private PinOperationReceipt applyResult(PinCommand command,CmsPinOperationExecutor.Result result,long current) {
        UUID tenant=command.institutionId(), family=command.panFamilyReference();
        if(result.failureCode()!=null) return receipt(command,Status.FAILED,current,null,result.failureCode());
        if(result.verificationValue()==null || !result.verificationValue().matches("[0-9]{4,12}"))
            throw CmsIntegrationException.unavailable("HSM returned no valid verification value");
        if(command.operation()==Operation.GENERATE_INITIAL_PIN && (result.mailerReference()==null || result.mailerReference().isBlank()))
            throw CmsIntegrationException.unavailable("Generated PIN has no mailer reference");
        long next=Math.addExact(current,1);
        persistVerificationValue(tenant,family,result.verificationValue());
        db.update("update cms_pin_profile set active_pin_version=?,updated_at=now() where cms_institution_id=? and pan_family_id=?",next,tenant,family);
        return receipt(command,Status.COMPLETED,next,result.mailerReference(),null);
    }

    /** Called by the transaction entry point while holding the same family lock. */
    @Transactional(propagation=org.springframework.transaction.annotation.Propagation.MANDATORY)
    public JsonNode commitNetwork(PinCommand command,String fingerprint,CmsPinOperationExecutor.Result result) {
        var row=profile(command.institutionId(),command.panFamilyReference()).getFirst();
        long current=number(row,"active_pin_version");
        expected(command,current);
        if(number(row,"policy_profile_version")!=command.policyProfileVersion())
            throw CmsIntegrationException.conflict("Stale policy/profile version");
        String encoded=encode(applyResult(command,result,current));
        db.update("insert into cms_pin_operation(cms_institution_id,operation_id,pan_family_id,operation_type,request_hash,status,receipt) values(?,?,?,?,?,?,?::jsonb)",
                command.institutionId(),command.operationId(),command.panFamilyReference(),command.operation().name(),
                fingerprint,tree(encoded).path("status").asText(),encoded);
        return tree(encoded);
    }

    private void persistVerificationValue(UUID tenant,UUID family,String value) {
        Integer institution=db.queryForObject("select switch_institution_id from cms_institution_mapping where cms_institution_id=?",Integer.class,tenant);
        List<Long> ids=db.query("select auth_id from cms_card_profile where cms_institution_id=? and pan_family_id=?",(rs,n)->rs.getLong(1),tenant,family);
        for(Long id:ids) {
            CardData card=cards.findById(id).orElseThrow();
            crypto.hydrateDecryptedFields(card,institution);
            card.setOld_pvv_offest(card.getPvv_offest()); card.setPvv_offest(value);
            card.setPin_retry_count(0); card.setPin_blocked(false);
            crypto.prepareProtectedForPersistence(card,institution); cards.save(card);
        }
        cards.flush();
    }

    @Transactional(readOnly=true)
    public JsonNode get(UUID tenant,UUID operation) {
        List<String> rows=db.query("select receipt::text from cms_pin_operation where cms_institution_id=? and operation_id=?",(rs,n)->rs.getString(1),tenant,operation);
        if(rows.isEmpty()) throw new CmsIntegrationException(404,"PIN operation not found");
        return tree(rows.getFirst());
    }
    private List<Map<String,Object>> profile(UUID t,UUID f){return db.queryForList("select * from cms_pin_profile where cms_institution_id=? and pan_family_id=? for update",t,f);}
    private long number(Map<String,Object> row,String key){return ((Number)row.get(key)).longValue();}
    private void expected(PinCommand c,long actual){if(c.expectedPinVersion()!=actual) throw CmsIntegrationException.conflict("Stale expected PIN version");}
    private void lock(UUID t,UUID f){db.queryForList("select pg_advisory_xact_lock(hashtextextended(?,0))","cms-pin-family:"+t+":"+f);}
    private PinOperationReceipt receipt(PinCommand c,Status s,long v,String mailer,String failure){return new PinOperationReceipt(c.operationId(),c.institutionId(),c.panFamilyReference(),c.operation(),s,c.policyProfileVersion(),v,mailer,failure,Instant.now());}
    private String hash(Object o){try{byte[] d=java.security.MessageDigest.getInstance("SHA-256").digest(encode(o).getBytes(java.nio.charset.StandardCharsets.UTF_8));return java.util.HexFormat.of().formatHex(d);}catch(Exception e){throw new IllegalStateException(e);}}
    private String encode(Object o){try{return json.writeValueAsString(o);}catch(Exception e){throw CmsIntegrationException.bad("Invalid PIN command");}}
    private JsonNode tree(String s){try{return json.readTree(s);}catch(Exception e){throw CmsIntegrationException.unavailable("Invalid stored PIN operation");}}
    private <T>T read(String s,Class<T> t){try{return json.readValue(s,t);}catch(Exception e){throw CmsIntegrationException.unavailable("Invalid stored PIN profile");}}
}
