package com.logicpoint.iso8583.validation.transaction;

import com.logicpoint.iso8583.context.RequestContext;
import com.logicpoint.iso8583.validation.core.Validator;
import com.logicpoint.iso8583.validation.ValidationResult;
import com.logicpoint.iso8583.validation.TransactionType;
import java.util.Map;
/**
 * Interface for transaction type validation operations.
 * Extends the core Validator interface for transaction-specific validations.
 */
public interface TransactionTypeValidator extends Validator<TransactionType> {
    
    /**
     * Validates transaction type and returns missing required fields.
     * 
     * @param transactionType The transaction type to validate
     * @return ValidationResult with transaction type and missing fields
     */
    ValidationResult validateTransactionType(TransactionType transactionType, Map<Integer, String> transactionData);

    default ValidationResult validateTransactionType(TransactionType transactionType, RequestContext requestContext) {
        return validateTransactionType(transactionType, requestContext != null ? requestContext.getFields() : null);
    }
    
    /**
     * Validates transaction amount limits for the given type.
     * 
     * @param transactionType The transaction type
     * @param amount The transaction amount
     * @return ValidationResult with validation status
     */
    ValidationResult validateAmountLimits(TransactionType transactionType, Double amount);
    
    /**
     * Validates transaction velocity limits for the given type.
     * 
     * @param transactionType The transaction type
     * @param cardNumber The card number for velocity checking
     * @return ValidationResult with validation status
     */
    ValidationResult validateVelocityLimits(TransactionType transactionType, String cardNumber);
}
