package com.logicpoint.iso8583.validation;
import com.logicpoint.iso8583.validation.SchemeValidatorService;
import java.util.Map;
import java.util.List;
import java.util.Optional;
import com.logicpoint.iso8583.util.Scheme;

public class TestValidation {
    public static void main(String[] args) throws Exception {
        SchemeValidatorService validator = new SchemeValidatorService("validation_rules");

        Map<String, String> isoFields = Map.of(
            "2", "5399999999999999",
            "3", "000000",
            "7", "0603153025",
            "11", "123456",
            "22", "",
            "25", "51",
            "48","08500202",
            "120", "00100207002003GDN"
        );

        Scheme scheme = Scheme.RUPAY;

        Optional<TransactionIdentificationRule> rule = validator.identify(scheme, isoFields);
        if (rule.isPresent()) {
            System.out.println(" Identified as: " + rule.get().transactionType);
            List<String> missing = rule.get().validateFields(isoFields);
            if (missing.isEmpty()) {
                System.out.println("All required fields present.");
            } else {
                System.out.println(" Missing fields: " + missing);
            }
        } else {
            System.out.println(" Could not identify transaction.");
        }
    }
}
