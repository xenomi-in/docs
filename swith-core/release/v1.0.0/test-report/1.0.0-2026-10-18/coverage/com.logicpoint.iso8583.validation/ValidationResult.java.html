package com.logicpoint.iso8583.validation;

import java.util.List;
import java.util.ArrayList;
import java.util.Collections;
import java.util.Map;
import java.util.HashMap;

public class ValidationResult {
    private final List<String> errors;
    private final List<String> warnings;
    private final boolean isValid;
    private  String responseCode;
    private final String responseMessage;
    private final Map<String, Object> updatedDataElements;

    public ValidationResult(List<String> errors, List<String> warnings) {
        this.errors = new ArrayList<>(errors);
        this.warnings = new ArrayList<>(warnings);
        this.isValid = errors.isEmpty();
        this.responseCode = determineResponseCode(errors);
        this.responseMessage = determineResponseMessage(errors);
        this.updatedDataElements = new HashMap<>();
    }
    
    public ValidationResult(List<String> errors, List<String> warnings, Map<String, Object> updatedDataElements) {
        this.errors = new ArrayList<>(errors);
        this.warnings = new ArrayList<>(warnings);
        this.isValid = errors.isEmpty();
        this.responseCode = determineResponseCode(errors);
        this.responseMessage = determineResponseMessage(errors);
        this.updatedDataElements = new HashMap<>(updatedDataElements);
    }

    public List<String> getErrors() { 
        return Collections.unmodifiableList(errors); 
    }
    
    public List<String> getWarnings() { 
        return Collections.unmodifiableList(warnings); 
    }
    
    public boolean isValid() { 
        return isValid; 
    }
    
    public String getResponseCode() { 
        return responseCode; 
    }
    
    public void setResponseCode(String responseCode) {
        this.responseCode = responseCode;
    }
    
    public String getResponseMessage() { 
        return responseMessage; 
    }
    
    /**
     * Gets the updated data elements map.
     */
    public Map<String, Object> getUpdatedDataElements() {
        return Collections.unmodifiableMap(updatedDataElements);
    }
    
    /**
     * Gets a specific updated data element by key.
     */
    public Object getUpdatedDataElement(String key) {
        return updatedDataElements.get(key);
    }
    
    /**
     * Checks if there are any updated data elements.
     */
    public boolean hasUpdatedDataElements() {
        return !updatedDataElements.isEmpty();
    }

    private String determineResponseCode(List<String> errors) {
        if (errors.isEmpty()) return "00"; // Approval
        
        for (String error : errors) {
            if (error.contains("EXPIRED_CARD")) return "54";
            if (error.contains("INVALID_CARD")) return "14";
            if (error.contains("BLOCKED_BIN")) return "05";
            if (error.contains("AMOUNT_LIMIT")) return "61";
            if (error.contains("VELOCITY_LIMIT")) return "65";
            if (error.contains("INVALID_SERVICE_CODE")) return "14";
            if (error.contains("MISSING_EXPIRY_DATE")) return "30";
            if (error.contains("INVALID_TRANSACTION_TYPE")) return "12";
            if (error.contains("DUPLICATE_TRANSACTION")) return "94";
            if (error.contains("MANDATORY_TAG_MISSING")) return "30";
        }
        return "96"; // System malfunction
    }

    private String determineResponseMessage(List<String> errors) {
        if (errors.isEmpty()) return "APPROVED";
        return String.join("; ", errors);
    }

    /**
     * Creates a new Builder instance.
     */
    public static Builder builder() {
        return new Builder();
    }

    /**
     * Builder class for ValidationResult.
     */
    public static class Builder {
        private final List<String> errors = new ArrayList<>();
        private final List<String> warnings = new ArrayList<>();
        private final Map<String, Object> updatedDataElements = new HashMap<>();

        /**
         * Adds a single error to the validation result.
         */
        public Builder addError(String error) {
            if (error != null && !error.trim().isEmpty()) {
                errors.add(error.trim());
            }
            return this;
        }

        /**
         * Adds multiple errors to the validation result.
         */
        public Builder addErrors(List<String> errors) {
            if (errors != null) {
                for (String error : errors) {
                    addError(error);
                }
            }
            return this;
        }

        /**
         * Adds a single warning to the validation result.
         */
        public Builder addWarning(String warning) {
            if (warning != null && !warning.trim().isEmpty()) {
                warnings.add(warning.trim());
            }
            return this;
        }

        /**
         * Adds multiple warnings to the validation result.
         */
        public Builder addWarnings(List<String> warnings) {
            if (warnings != null) {
                for (String warning : warnings) {
                    addWarning(warning);
                }
            }
            return this;
        }

        /**
         * Adds errors from another ValidationResult.
         */
        public Builder addErrors(ValidationResult other) {
            if (other != null) {
                addErrors(other.getErrors());
            }
            return this;
        }

        /**
         * Adds warnings from another ValidationResult.
         */
        public Builder addWarnings(ValidationResult other) {
            if (other != null) {
                addWarnings(other.getWarnings());
            }
            return this;
        }

        /**
         * Adds both errors and warnings from another ValidationResult.
         */
        public Builder addValidationResult(ValidationResult other) {
            if (other != null) {
                addErrors(other.getErrors());
                addWarnings(other.getWarnings());
            }
            return this;
        }

        /**
         * Adds an updated data element.
         */
        public Builder addUpdatedDataElement(String key, Object value) {
            if (key != null && value != null) {
                updatedDataElements.put(key, value);
            }
            return this;
        }
        
        /**
         * Adds multiple updated data elements.
         */
        public Builder addUpdatedDataElements(Map<String, Object> dataElements) {
            if (dataElements != null) {
                updatedDataElements.putAll(dataElements);
            }
            return this;
        }
        
        /**
         * Clears all errors and warnings.
         */
        public Builder clear() {
            errors.clear();
            warnings.clear();
            updatedDataElements.clear();
            return this;
        }

        /**
         * Clears only errors.
         */
        public Builder clearErrors() {
            errors.clear();
            return this;
        }

        /**
         * Clears only warnings.
         */
        public Builder clearWarnings() {
            warnings.clear();
            return this;
        }

        /**
         * Sets a custom response code for the validation result.
         */
        public Builder setResponseCode(String responseCode) {
            // This will be handled in the build method
            return this;
        }

        /**
         * Builds and returns the ValidationResult.
         */
        public ValidationResult build() {
            ValidationResult result = new ValidationResult(errors, warnings, updatedDataElements);
            return result;
        }

        /**
         * Builds a ValidationResult with a specific response code.
         */
        public ValidationResult buildWithResponseCode(String responseCode) {
            ValidationResult result = build();
            // Note: This would require modifying the ValidationResult constructor
            // to accept a custom response code. For now, we'll use the default logic.
            return result;
        }

        /**
         * Checks if the current builder has any errors.
         */
        public boolean hasErrors() {
            return !errors.isEmpty();
        }

        /**
         * Checks if the current builder has any warnings.
         */
        public boolean hasWarnings() {
            return !warnings.isEmpty();
        }

        /**
         * Gets the current error count.
         */
        public int getErrorCount() {
            return errors.size();
        }

        /**
         * Gets the current warning count.
         */
        public int getWarningCount() {
            return warnings.size();
        }
    }

    @Override
    public String toString() {
        return "ValidationResult{" +
                "isValid=" + isValid +
                ", errorCount=" + errors.size() +
                ", warningCount=" + warnings.size() +
                ", responseCode='" + responseCode + '\'' +
                ", responseMessage='" + responseMessage + '\'' +
                ", updatedDataElements=" + updatedDataElements +
                '}';
    }
}
