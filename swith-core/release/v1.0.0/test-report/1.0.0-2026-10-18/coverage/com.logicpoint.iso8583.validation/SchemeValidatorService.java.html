package com.logicpoint.iso8583.validation;

import tools.jackson.databind.json.JsonMapper;
import java.util.List;
import java.util.Map;
import java.util.HashMap;
import java.util.Optional;
import java.util.Arrays;
import java.io.InputStream;
import java.io.IOException;
import com.logicpoint.iso8583.util.Scheme;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

/**
 * Service for validating ISO 8583 messages based on scheme-specific rules.
 * This service loads validation rules from JSON files and applies them to
 * incoming messages to identify transaction types and validate required fields.
 *
 * The validation rules are stored in JSON files under the specified directory,
 * with one file per payment scheme (e.g., rupay_rules.json, visa_rules.json).
 *
 * @author LogicPoint Systems
 * @version 1.0
 */
public class SchemeValidatorService {

    Logger log = LoggerFactory.getLogger(SchemeValidatorService.class);

    private final Map<Scheme, List<TransactionIdentificationRule>> schemeRules;

    /**
     * Constructs a new SchemeValidatorService.
     *
     * @param rulesDir Directory containing the validation rule JSON files
     * @throws IOException if there's an error loading the rule files
     */
    public SchemeValidatorService(String rulesDir) throws IOException {
        this.schemeRules = loadAllSchemeRules(rulesDir);
    }

    /**
     * Loads validation rules for all supported schemes from JSON files.
     *
     * @param baseDir Base directory containing the rule files
     * @return Map of schemes to their validation rules
     * @throws IOException if there's an error reading the rule files
     */
    private Map<Scheme, List<TransactionIdentificationRule>> loadAllSchemeRules(String baseDir) throws IOException {
        Map<Scheme, List<TransactionIdentificationRule>> map = new HashMap<>();
        JsonMapper mapper = JsonMapper.builderWithJackson2Defaults().build();
        try {
            for (Scheme scheme : Scheme.values()) {
                String file = baseDir + "/" + scheme.name().toLowerCase() + "_rules.json"; 
                InputStream in = getClass().getClassLoader().getResourceAsStream(file);
                if (in == null) continue;
                List<TransactionIdentificationRule> rules = Arrays.asList(
                    mapper.readValue(in, TransactionIdentificationRule[].class)
                );
                map.put(scheme, rules);
                log.info("scheme: "+scheme + " No of rules: "+rules.size());
            }
            return map;
        } catch (Exception e) {
            e.printStackTrace();
            throw new IOException("Failed to load scheme rules", e);
        }
    }

    
    /**
     * Identifies the transaction type based on the message fields.
     *
     * @param scheme The payment scheme
     * @param fields Map of field numbers to their values
     * @return Optional containing the matching transaction rule, if found
     */
    public Optional<TransactionIdentificationRule> identify(Scheme scheme, Map<String, String> fields) {
        
        List<TransactionIdentificationRule> rules = schemeRules.get(scheme);
        if (rules == null) return Optional.empty();
        log.info("Loaded {} identification rules for scheme {}", rules.size(), scheme);
        return rules.stream().filter(rule -> rule.matches(fields)).findFirst();
    }

    /**
     * Validates the required fields for a transaction.
     *
     * @param scheme The payment scheme
     * @param fields Map of field numbers to their values
     * @return List of missing required fields, or empty list if all required fields are present
     */
    public List<String> validate(Scheme scheme, Map<String, String> fields) {
        return identify(scheme, fields)
            .map(rule -> rule.validateFields(fields))
            .orElse(List.of("Transaction type not identified."));
    }
}
