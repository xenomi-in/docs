package com.logicpoint.iso8583.validation;

import java.util.Map;
import java.util.List;


/**
 * Represents a condition for validating TLV (Tag-Length-Value) fields in ISO 8583 messages.
 * Each condition specifies a tag, its expected value or list of valid values, and whether
 * the field is required.
 *
 * @author LogicPoint Systems
 * @version 1.0
 */

public class TlvCondition {
   
    private String field;
    private String tag;
    private List<String> values;
    private Integer valueStart;
    private Integer valueLength;

    /**
     * Checks if the given message fields match this condition.
     *
     * @param fields Map of field numbers to their values
     * @return true if the condition is met, false otherwise
     */
    public boolean matches(Map<String, String> fields) {
        String tlvData = fields.get(field);
        
        if (tlvData == null || tlvData.trim().isEmpty()) {
            return false;
        }

        Map<String, String> parsedTlv;
        try {
            parsedTlv = TlvUtils.parseTlv(tlvData);
        } catch (IllegalArgumentException ex) {
            return false;
        }

        String tagValue = parsedTlv.get(tag);
        if (tagValue == null) {
            return false;
        }

        String comparisonValue = extractComparisonValue(tagValue);
        return comparisonValue != null
                && (values == null || values.isEmpty() || values.contains(comparisonValue));
    }

    private String extractComparisonValue(String tagValue) {
        if (valueStart == null && valueLength == null) {
            return tagValue;
        }
        if (valueStart == null || valueLength == null || valueStart < 0 || valueLength <= 0) {
            return null;
        }

        int end = valueStart + valueLength;
        return end <= tagValue.length() ? tagValue.substring(valueStart, end) : null;
    }

    // Getters and setters for JSON deserialization
    public String getField() {
        return field;
    }

    public void setField(String field) {
        this.field = field;
    }

    public String getTag() {
        return tag;
    }

    public void setTag(String tag) {
        this.tag = tag;
    }

    public List<String> getValues() {
        return values;
    }

    public void setValues(List<String> values) {
        this.values = values;
    }

    public Integer getValueStart() {
        return valueStart;
    }

    public void setValueStart(Integer valueStart) {
        this.valueStart = valueStart;
    }

    public Integer getValueLength() {
        return valueLength;
    }

    public void setValueLength(Integer valueLength) {
        this.valueLength = valueLength;
    }
}
