package com.logicpoint.iso8583.validation;
import java.util.List;
import java.util.Map;
import java.util.ArrayList;

/**
 * Represents a rule for identifying and validating ISO 8583 transactions.
 * Each rule defines the conditions that must be met to identify a specific
 * transaction type and the fields that are required for that transaction.
 *
 * @author LogicPoint Systems
 * @version 1.0
 */
public class TransactionIdentificationRule {
    public String processingCode;
    public List<String> posEntryMode;
    public List<String> posConditionCode;
    public TransactionType transactionType;
    public List<String> requiredFields;
    public List<TlvCondition> tlvConditions;
    public List<TlvCondition> requiredTlvTags;

    public boolean matches(Map<String, String> fields) {
        if (processingCode != null && !processingCode.equals(fields.get("3"))) return false;
        if (!(posEntryMode.isEmpty() || posEntryMode.contains(fields.get("22")))) return false;
        if (!(posConditionCode.isEmpty() || posConditionCode.contains(fields.get("25")))) return false;
        
        if (tlvConditions != null) {
            for (TlvCondition cond : tlvConditions) {
                if (!cond.matches(fields)) return false;
            }
        }

        return true;
    }

     /**
     * Validates that all required fields are present in the message.
     *
     * @param fields Map of field numbers to their values
     * @return List of missing required fields, or empty list if all required fields are present
     */
    public List<String> validateFields(Map<String, String> fields) {
        List<String> missingFields = new ArrayList<>();
        if (requiredFields != null) {
            for (String field : requiredFields) {
                if (!fields.containsKey(field) || fields.get(field) == null || fields.get(field).trim().isEmpty()) {
                    missingFields.add(field);
                }
            }
        }
        if (requiredTlvTags != null) {
            for (TlvCondition requiredTag : requiredTlvTags) {
                if (!requiredTag.matches(fields)) {
                    missingFields.add(requiredTag.getField() + "." + requiredTag.getTag());
                }
            }
        }
        return missingFields;
    }
}
