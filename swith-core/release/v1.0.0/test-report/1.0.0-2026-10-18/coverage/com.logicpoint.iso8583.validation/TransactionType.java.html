package com.logicpoint.iso8583.validation;

/**
 * Enumeration of possible transaction types in ISO 8583 messages.
 * Each transaction type represents a specific kind of financial transaction
 * that can be processed by the system.
 *
 * @author LogicPoint Systems
 * @version 1.0
 */
public enum TransactionType {

    /**
     * Purchase transaction - used for regular purchases
     */
    PURCHASE,
    
    /**
     * Purchase transaction - used for contactless purchases
     */
    PURCHASE_CONTACTLESS,

    /**
     * Purchase transaction - used for contact purchases
     */
    PURCHASE_CONTACT,
    
    /**
     * Purchase transaction - used for transit purchases
     */
    PURCHASE_TRANSIT,

    /* 
     * MOTO transaction - used for MOTO purchases
     */
    MOTO,

    /**
     * Purchase transaction - used for contactless purchases
     */
    PURCHASE_FALLBACK,
    /**
     * IVR purchase transaction - used for IVR purchases
     */
    IVR_PURCHASE,

    /* * Cash withdrawal transaction - used for ATM withdrawals
     */
    CASH_WITHDRAWAL,

    // ATM CASH WITHDRAWAL in FaLLback
    CASH_WITHDRAWAL_FALLBACK,
    /* * Cash withdrawal contactless transaction - used for ATM withdrawals
     */
    CASH_WITHDRAWAL_CONTACTLESS,

    /* * Cash at POS transaction - used for POS withdrawals
     */
    CASH_AT_POS,

    /* * Cash at POS contactless transaction - used for POS withdrawals
     */
    CASH_AT_POS_CL,

    /* * E-COMMERCE transaction - used for online purchase( CNP)
     */
    ECOMMERCE,

     /* * Pre -Authorization transaction 
     */
    PREAUTH,

    /* * Loyalty transaction 
     */
    LOYALTY,

    /**
     * Balance inquiry transaction - used to check account balance
     */
    BALANCE_ENQUIRY,
    /**
     * Balance inquiry transaction - used to check account balance
     * Contactless balance inquiry transaction - used to check account balance
     */
    BALANCE_ENQUIRY_CL,
    /**
     * Mini statement transaction - used to get recent transaction history
     */
    MINI_STATEMENT,

    /**
     * Fund transfer transaction - used to transfer funds between accounts
     */
    FUND_TRANSFER,

    /**
     * Bill payment transaction - used to pay bills
     */
    BILL_PAYMENT,

    /**
     * Reversal transaction - used to reverse a previous transaction
     */
    REVERSAL,

    /**
     * PIN change transaction - used to change card PIN
     */
    PIN_CHANGE,

    /**
     * Card to card transfer transaction
     */
    CARD_TO_CARD_TRANSFER,

    /**
     * Service creation transaction
     */
    SERVICE_CREATION,   

    /**
     * Service creation transaction - used for contactless service creation
     */
    SERVICE_CREATION_CONTACTLESS,
    /**
     * Purchase with cashback transaction
     */
    PURCHASE_WITHCASHBACK,

    /**
     * Refund transaction
     */
    REFUND,

    /**
     * OCT transaction
     */
    OCT,

    /**
     * Aadhar inquiry transaction
     */
    AADHAR_INQUIRY,

    /**
     * EMI purchase transaction (e-commerce)
     */
    EMI_PURCHASE,
    /**
     * POS EMI purchase transaction 
     */
    CARD_EMI_PURCHASE,

    /**
     * Account verification transaction
     */
    ACCOUNT_VERIFICATION,

    /**
     * Mobile number update transaction
     */
    MOBILE_NUMBER_UPDATE,

    /**
     * Chequebook request transaction
     */
    CHEQUEBOOK_REQUEST,

    /**
     * Money load transaction
     */
    MONEY_LOAD,

    /**
     * ML balance update transaction
     */
    ML_BALANCEUPDATE,

    /**
     * Beneficiary account validation transaction
     */
    BENIFICIARY_ACCOUNT_VALIDATION,

    /**
     * Cash deposit transaction
     */
    CASH_DEPOSIT,

    /**
     * Statement request transaction
     */
    STATEMENT_REQUEST,

    /**
     * Standing instruction registration transaction
     */
    SI_REGISTRATION,
    SI_EDIT,
    SI_TRANSACTION,
    SI_DEREGISTRATION,
    SI_REVOKEALL,
    SI_PAUSE,
    SI_START,
    SI_PDN,

    
    BIOMETIC_VERIFICATION,
    

    /**
     * Unknown transaction type
     */
    UNKNOWN;

    private String code;

    /**
     * Constructs a new transaction type with the specified code.
     *
     * @param code The transaction type code
     */
  

    /**
     * Gets the transaction type code.
     *
     * @return The transaction type code
     */
    public String getCode() {
        return code;
    }

    /**
     * Gets the transaction type for a given code.
     *
     * @param code The transaction type code to look up
     * @return The corresponding TransactionType, or null if not found
     */
    public static TransactionType fromCode(String code) {
        for (TransactionType type : values()) {
            if (type.code.equals(code)) {
                return type;
            }
        }
        return null;
    }

    /**
     * Safely converts a string to a TransactionType.
     * Returns UNKNOWN if the string doesn't match any transaction type.
     *
     * @param name The name to convert
     * @return The corresponding TransactionType, or UNKNOWN if not found
     */
    public static TransactionType fromStringSafe(String name) {
        try {
            return valueOf(name);
        } catch (IllegalArgumentException e) {
            return UNKNOWN;
        }
    }
}
