package com.logicpoint.iso8583.validation;

import java.util.Map;
import java.util.LinkedHashMap;

/**
 * Utility class for parsing and handling TLV (Tag-Length-Value) data in ISO 8583 messages.
 * This class provides methods to parse TLV formatted strings into a map of tag-value pairs.
 *
 * @author LogicPoint Systems
 * @version 1.0
 */
public class TlvUtils {

    /**
     * Parses a TLV formatted string into a map of tag-value pairs.
     * The input string should be in the format: TTTLLLVVV... where:
     * TTT is the 3-digit tag
     * LLL is the 3-digit length
     * VVV is the value of the specified length
     *
     * @param tlvData The TLV formatted string to parse
     * @return Map containing tag-value pairs
     * @throws IllegalArgumentException if the TLV data is malformed
     */
    public static Map<String, String> parseTlv(String tlvData) {
        Map<String, String> result = new LinkedHashMap<>();
        int index = 0;

        while (index < tlvData.length()) {
            if (index + 6 > tlvData.length()) {
                throw new IllegalArgumentException("Invalid TLV data: incomplete tag or length");
            }

            String tag = tlvData.substring(index, index + 3);
            int length = Integer.parseInt(tlvData.substring(index + 3, index + 6));
            index += 6;

            if (index + length > tlvData.length()) {
                throw new IllegalArgumentException("Invalid TLV data: value length exceeds available data");
            }

            String value = tlvData.substring(index, index + length);
            result.put(tag, value);
            index += length;
        }

        return result;
    }

    /**
     * Validates if a string is in valid TLV format.
     * Checks if the string length is valid and if all length fields
     * are properly formatted.
     *
     * @param tlvData The TLV string to validate
     * @return true if the string is in valid TLV format, false otherwise
     */
    public static boolean isValidTlvFormat(String tlvData) {
        if (tlvData == null || tlvData.length() < 6) {
            return false;
        }

        int index = 0;
        while (index < tlvData.length()) {
            if (index + 6 > tlvData.length()) {
                return false;
            }

            try {
                int length = Integer.parseInt(tlvData.substring(index + 3, index + 6));
                if (length < 0 || index + 6 + length > tlvData.length()) {
                    return false;
                }
                index += 6 + length;
            } catch (NumberFormatException e) {
                return false;
            }
        }

        return true;
    }
}
