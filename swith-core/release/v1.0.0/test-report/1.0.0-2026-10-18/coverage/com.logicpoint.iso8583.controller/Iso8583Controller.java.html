package com.logicpoint.iso8583.controller;

import com.logicpoint.iso8583.model.Iso8583Message;
import com.logicpoint.iso8583.model.DownstreamHealthResponse;
import com.logicpoint.iso8583.config.RawDataLogging;
import com.logicpoint.iso8583.service.Iso8583ProcessorService;
import com.logicpoint.iso8583.service.RoutingHealthCheckService;
import io.swagger.v3.oas.annotations.Operation;
import io.swagger.v3.oas.annotations.Parameter;
import io.swagger.v3.oas.annotations.media.Content;
import io.swagger.v3.oas.annotations.media.Schema;
import io.swagger.v3.oas.annotations.responses.ApiResponse;
import io.swagger.v3.oas.annotations.responses.ApiResponses;
import io.swagger.v3.oas.annotations.tags.Tag;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.http.ResponseEntity;
import org.springframework.scheduling.annotation.Async;
import org.springframework.web.bind.annotation.*;
import org.springframework.util.StringUtils;

import java.util.HashMap;
import java.util.Map;
import java.util.concurrent.CompletableFuture;
/*
 * The controller uses asynchronous processing with CompletableFuture to handle high throughput.
 * It validates incoming ISO 8583 messages, routes them to appropriate processors based on transaction type,
 * and handles responses asynchronously using Spring's @Async capabilities.
 * The controller also exposes health check endpoints for monitoring service status.
 */
@RestController
@RequestMapping("/api/v1/iso8583")
@Tag(name = "ISO 8583 Message Processing", description = "APIs for processing ISO 8583 financial messages")
public class Iso8583Controller {

    private static final Logger log = LoggerFactory.getLogger(Iso8583Controller.class);
    private final Iso8583ProcessorService processorService;
    private final RoutingHealthCheckService routingHealthCheckService;

    /**
     * Constructs the controller with injected processor service.
     *
     * @param processorService The ISO8583 processor service
     */
    @Autowired
    public Iso8583Controller(Iso8583ProcessorService processorService,
                             RoutingHealthCheckService routingHealthCheckService) {
        this.processorService = processorService;
        this.routingHealthCheckService = routingHealthCheckService;
    }

    /**
     * Processes an ISO 8583 message asynchronously.
     *
     * @param request The ISO8583 message request
     * @param apiVersion API version header (default 1.0)
     * @return CompletableFuture with response entity
     */
    @PostMapping("/transaction")
    @Operation(
        summary = "Process ISO 8583 message",
        description = "Processes an ISO 8583 financial message and returns the response"
    )
    @ApiResponses(value = {
        @ApiResponse(
            responseCode = "200",
            description = "Message processed successfully",
            content = @Content(schema = @Schema(implementation = Iso8583Message.class))
        ),
        @ApiResponse(
            responseCode = "400",
            description = "Invalid message format"
        ),
        @ApiResponse(
            responseCode = "500",
            description = "Internal server error"
        )
    })
    @Async("iso8583TxnExecutor")
    public CompletableFuture<ResponseEntity<Iso8583Message>> processMessage(
            @Parameter(description = "ISO 8583 message to process", required = true)
            @RequestBody Iso8583Message request,
            @RequestHeader(value = "X-API-Version", defaultValue = "1.0") String apiVersion,
            @RequestHeader(value = "X-Site-Id", required = false) String siteId,
            @RequestHeader(value = "X-Listener-Id", required = false) String listenerId,
            @RequestHeader(value = "X-Connection-Id", required = false) String connectionId,
            @RequestHeader Map<String, String> headers) {
        enrichRequestMetadata(request, siteId, listenerId, connectionId, headers);
        log.info("Received ISO8583 request mti={} txnId={} apiVersion={}",
                request == null ? null : request.getMti(),
                request == null ? null : request.getUniqueTransactionId(), apiVersion);
        RawDataLogging.value("Inbound ISO8583 transaction", request);
        return processorService.processMessage(request)
                .thenApply(ResponseEntity::ok)
                .exceptionally(throwable -> {
                    log.error("Error processing message", throwable);
                    return ResponseEntity.internalServerError().build();
                });
    }

    /**
     * Health check endpoint.
     *
     * @return ResponseEntity with health status
     */
    @GetMapping("/health")
    @Operation(
        summary = "Health check endpoint",
        description = "Returns the health status of the ISO 8583 processing service"
    )
    @ApiResponses(value = {
        @ApiResponse(
            responseCode = "200",
            description = "Service is healthy"
        )
    })
    public ResponseEntity<String> healthCheck() {
        return ResponseEntity.ok("ISO 8583 Processing Service is healthy");
    }

    @GetMapping("/health/routing")
    @Operation(
        summary = "Routing readiness endpoint",
        description = "Returns machine-readable readiness so the upstream listener can decide whether to route transactions"
    )
    @ApiResponses(value = {
        @ApiResponse(
            responseCode = "200",
            description = "Service is routable",
            content = @Content(schema = @Schema(implementation = DownstreamHealthResponse.class))
        ),
        @ApiResponse(
            responseCode = "503",
            description = "Service is not routable",
            content = @Content(schema = @Schema(implementation = DownstreamHealthResponse.class))
        )
    })
    public ResponseEntity<DownstreamHealthResponse> routingHealthCheck() {
        DownstreamHealthResponse response = routingHealthCheckService.getRoutingHealth();
        return response.isRoutable()
                ? ResponseEntity.ok(response)
                : ResponseEntity.status(503).body(response);
    }

    /**
     * Gets the service version.
     *
     * @return ResponseEntity with version string
     */
    @GetMapping("/version")
    public ResponseEntity<String> getVersion() {
        return ResponseEntity.ok("1.0.0");
    }
    /**
     * Handles the ISO 8583 network message for dynamic key exchange.
     * This endpoint can be used by network participants to exchange keys.
     *
     * @param request The incoming ISO8583 message containing the key exchange data
     * @return The response ISO8583 message with the result of the key exchange
     */
    @PostMapping("/key-exchange/{scheme}")
    @Operation(
        summary = "Network message handler for dynamic key exchange",
        description = "Processes an ISO8583 network message to facilitate dynamic key exchange"
    )
    @ApiResponses(value = {
        @ApiResponse(
            responseCode = "200",
            description = "Key exchange processed successfully"
        ),
        @ApiResponse(
            responseCode = "400",
            description = "Bad request or key exchange error"
        ),
        @ApiResponse(
            responseCode = "500",
            description = "Internal server error"
        )
    })
    public ResponseEntity<Iso8583Message> handleKeyExchange(
            @Parameter(description = "Network ISO8583 key exchange message", required = true)
            @RequestBody Iso8583Message request,
            @PathVariable(value = "scheme") String scheme,
            @RequestHeader(value = "X-API-Version", defaultValue = "1.0") String apiVersion,
            @RequestHeader(value = "X-Site-Id", required = false) String siteId,
            @RequestHeader(value = "X-Listener-Id", required = false) String listenerId,
            @RequestHeader(value = "X-Connection-Id", required = false) String connectionId,
            @RequestHeader Map<String, String> headers) {
        enrichRequestMetadata(request, siteId, listenerId, connectionId, headers);
        log.info("Received network key exchange mti={} txnId={} scheme={} apiVersion={}",
                request == null ? null : request.getMti(),
                request == null ? null : request.getUniqueTransactionId(), scheme, apiVersion);
        RawDataLogging.value("Inbound ISO8583 network key exchange", request);
        try {
            Iso8583Message response = processorService.processNetworkKeyExchange(request, scheme);
            return ResponseEntity.ok(response);
        } catch (IllegalArgumentException ex) {
            log.error("Key exchange failed due to bad request: {}", ex.getMessage());
            return ResponseEntity.badRequest().build();
        } catch (Exception ex) {
            log.error("Internal error during dynamic key exchange", ex);
            return ResponseEntity.internalServerError().build();
        }
    }

    private void enrichRequestMetadata(Iso8583Message request,
                                       String siteId,
                                       String listenerId,
                                       String connectionId,
                                       Map<String, String> headers) {
        if (request == null) {
            return;
        }

        Map<String, String> metadata = request.getMetadata() == null
                ? new HashMap<>()
                : new HashMap<>(request.getMetadata());

        putIfAbsent(metadata, "site_id", firstNonBlank(
                siteId,
                resolveHeader(headers, "X-Site-Id", "site_id", "site-id", "siteid")
        ));
        putIfAbsent(metadata, "listener_id", firstNonBlank(
                listenerId,
                resolveHeader(headers, "X-Listener-Id", "listener_id", "listener-id", "listenerid")
        ));
        putIfAbsent(metadata, "connection_id", firstNonBlank(
                connectionId,
                resolveHeader(headers, "X-Connection-Id", "connection_id", "connection-id", "connectionid", "tcp_connection_id", "tcp-connection-id")
        ));
        request.setMetadata(metadata);
    }

    private void putIfAbsent(Map<String, String> metadata, String key, String value) {
        if (value != null && !value.isBlank() && !metadata.containsKey(key)) {
            metadata.put(key, value);
        }
    }

    private String firstNonBlank(String... values) {
        if (values == null) {
            return null;
        }
        for (String value : values) {
            if (StringUtils.hasText(value)) {
                return value;
            }
        }
        return null;
    }

    private String resolveHeader(Map<String, String> headers, String... names) {
        if (headers == null || headers.isEmpty() || names == null) {
            return null;
        }
        for (String name : names) {
            for (Map.Entry<String, String> entry : headers.entrySet()) {
                if (entry.getKey() != null && entry.getKey().equalsIgnoreCase(name) && StringUtils.hasText(entry.getValue())) {
                    return entry.getValue();
                }
            }
        }
        return null;
    }
}
