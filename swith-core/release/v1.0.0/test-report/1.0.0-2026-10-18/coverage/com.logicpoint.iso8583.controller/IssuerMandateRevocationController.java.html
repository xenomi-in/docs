package com.logicpoint.iso8583.controller;

import com.logicpoint.iso8583.model.IssuerMandateRevocationRequest;
import com.logicpoint.iso8583.model.IssuerMandateRevocationResult;
import com.logicpoint.iso8583.service.IssuerMandateRevocationService;
import io.swagger.v3.oas.annotations.Operation;
import io.swagger.v3.oas.annotations.tags.Tag;
import jakarta.validation.Valid;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RestController;

/** CMS/CRM API for issuer-originated standing-instruction mandate revocation. */
@RestController
@RequestMapping("/api/v1/issuer-mandates/revocations")
@Tag(name = "Issuer Mandate Revocation")
public class IssuerMandateRevocationController {

    private final IssuerMandateRevocationService revocationService;

    public IssuerMandateRevocationController(IssuerMandateRevocationService revocationService) {
        this.revocationService = revocationService;
    }

    @PostMapping
    @Operation(summary = "Originate an issuer-side mandate revocation")
    public ResponseEntity<IssuerMandateRevocationResult> revoke(
            @Valid @RequestBody IssuerMandateRevocationRequest request) {
        return ResponseEntity.ok(revocationService.revoke(request));
    }
}
