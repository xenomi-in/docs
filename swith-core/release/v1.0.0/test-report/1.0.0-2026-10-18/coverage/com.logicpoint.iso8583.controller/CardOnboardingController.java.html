package com.logicpoint.iso8583.controller;

import com.logicpoint.iso8583.entity.CardData;
import com.logicpoint.iso8583.model.CardOnboardingRequest;
import com.logicpoint.iso8583.model.CardOnboardingResponse;
import com.logicpoint.iso8583.repository.CardDataRepository;
import com.logicpoint.iso8583.service.CacheInitializationService;
import com.logicpoint.iso8583.service.CardDataEncryptionService;
import jakarta.validation.Valid;
import java.time.LocalDateTime;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.transaction.annotation.Transactional;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RestController;

@RestController
@RequestMapping("/api/v1/cards")
public class CardOnboardingController {

    private final CardDataRepository cardDataRepository;
    private final CardDataEncryptionService cardDataEncryptionService;
    private final CacheInitializationService cacheInitializationService;

    public CardOnboardingController(CardDataRepository cardDataRepository,
                                    CardDataEncryptionService cardDataEncryptionService,
                                    CacheInitializationService cacheInitializationService) {
        this.cardDataRepository = cardDataRepository;
        this.cardDataEncryptionService = cardDataEncryptionService;
        this.cacheInitializationService = cacheInitializationService;
    }

    @PostMapping("/onboard")
    @Transactional
    public ResponseEntity<CardOnboardingResponse> onboardCard(@Valid @RequestBody CardOnboardingRequest request) {
        String pan = request.getPan().trim();
        Integer institutionId = request.getInstitutionId() != null
                ? request.getInstitutionId()
                : resolveInstitutionId(pan);

        if (institutionId == null) {
            return ResponseEntity.badRequest()
                    .body(CardOnboardingResponse.failure("Institution could not be resolved from PAN BIN"));
        }

        String panLookupHmac = cardDataEncryptionService.computePanLookupHmac(pan, institutionId);
        CardData existingCard = panLookupHmac == null
                ? cardDataRepository.findByPan(pan)
                : cardDataRepository.findByPanLookupHmac(panLookupHmac);
        if (existingCard != null) {
            return ResponseEntity.status(HttpStatus.CONFLICT)
                    .body(CardOnboardingResponse.failure("Card already exists"));
        }

        CardData cardData = toCardData(request);
        cardDataEncryptionService.prepareForPersistence(cardData, institutionId);
        CardData saved = cardDataRepository.save(cardData);
        cacheInitializationService.refreshCache("cardDataCache");

        return ResponseEntity.status(HttpStatus.CREATED).body(CardOnboardingResponse.success(
                saved.getAuth_id(),
                saved.getCard_id(),
                institutionId,
                saved.getMasked_pan(),
                saved.getPan_lookup_hmac(),
                saved.getPan_key_version(),
                saved.getExpiry_key_version(),
                saved.getPvv_key_version()));
    }

    private CardData toCardData(CardOnboardingRequest request) {
        LocalDateTime now = LocalDateTime.now();
        CardData cardData = new CardData();
        cardData.setCard_id(request.getCardId());
        cardData.setPan(request.getPan().trim());
        cardData.setAccount_id(request.getAccountId());
        cardData.setCard_proxy_number(request.getCardProxyNumber());
        cardData.setCard_type(normalizeChar(request.getCardType()));
        cardData.setCard_scheme(normalizeChar(request.getCardScheme()));
        cardData.setIsDualInterface(normalizeChar(request.getDualInterface()));
        cardData.setExpiryDate(request.getExpiryDate());
        cardData.setPvv_offest(request.getPvvOffset());
        cardData.setKey_ref(request.getKeyRef());
        cardData.setCvv(request.getCvv());
        cardData.setCvv_key_ref(request.getCvvKeyRef());
        cardData.setEmv_aip(request.getEmvAip());
        cardData.setEmv_capk_id(request.getEmvCapkId());
        cardData.setArqc_validation_required(Boolean.TRUE.equals(request.getArqcValidationRequired()));
        cardData.setAtc(request.getAtc() == null ? 0 : request.getAtc());
        cardData.setPin_retry_count(request.getPinRetryCount() == null ? 0 : request.getPinRetryCount());
        cardData.setMax_pin_retry(request.getMaxPinRetry() == null ? 3 : request.getMaxPinRetry());
        cardData.setPin_blocked(Boolean.TRUE.equals(request.getPinBlocked()));
        cardData.setLast_auth_status(null);
        cardData.setLast_auth_timestamp(null);
        cardData.setCreated_at(now);
        cardData.setUpdated_at(now);
        cardData.setStatus(request.getStatus() == null || request.getStatus().isBlank()
                ? "ACTIVE"
                : request.getStatus().trim().toUpperCase());
        return cardData;
    }

    private Character normalizeChar(Character value) {
        return value == null ? null : Character.toUpperCase(value);
    }

    private Integer resolveInstitutionId(String pan) {
        if (pan == null || pan.length() < 6) {
            return null;
        }
        CacheInitializationService.BinInstitutionData binData = cacheInitializationService
                .loadBinInstitutionCache()
                .get(pan.substring(0, 6));
        if (binData == null || binData.getBinDetails() == null) {
            return null;
        }
        return binData.getBinDetails().getInstitution_id();
    }
}
