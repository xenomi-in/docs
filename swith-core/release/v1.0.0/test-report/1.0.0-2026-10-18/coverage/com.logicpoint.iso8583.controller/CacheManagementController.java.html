package com.logicpoint.iso8583.controller;

import com.logicpoint.iso8583.service.CacheUpdateService;
import com.logicpoint.iso8583.service.CacheInitializationService;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.*;
import com.logicpoint.iso8583.service.CacheInitializationService.BinInstitutionData;
import java.util.HashMap;
import java.util.Map;

/**
 * REST Controller for cache management operations.
 * Provides endpoints to refresh caches and get cache status.
 */
@RestController
@RequestMapping("/api/cache")
public class CacheManagementController {

    @Autowired
    private CacheUpdateService cacheUpdateService;

    @Autowired
    private CacheInitializationService cacheInitializationService;

    /**
     * Refresh all caches
     */
    @PostMapping("/refresh/all")
    public ResponseEntity<Map<String, String>> refreshAllCaches() {
        try {
            cacheUpdateService.refreshAllCaches();
            Map<String, String> response = new HashMap<>();
            response.put("status", "success");
            response.put("message", "All caches refreshed successfully");
            return ResponseEntity.ok(response);
        } catch (Exception e) {
            Map<String, String> response = new HashMap<>();
            response.put("status", "error");
            response.put("message", "Failed to refresh caches: " + e.getMessage());
            return ResponseEntity.internalServerError().body(response);
        }
    }

    /**
     * Refresh BIN and Institution cache
     */
    @PostMapping("/refresh/bin-institution")
    public ResponseEntity<Map<String, String>> refreshBinInstitutionCache() {
        try {
            cacheUpdateService.refreshBinInstitutionCache();
            Map<String, String> response = new HashMap<>();
            response.put("status", "success");
            response.put("message", "BIN and Institution cache refreshed successfully");
            return ResponseEntity.ok(response);
        } catch (Exception e) {
            Map<String, String> response = new HashMap<>();
            response.put("status", "error");
            response.put("message", "Failed to refresh BIN cache: " + e.getMessage());
            return ResponseEntity.internalServerError().body(response);
        }
    }

    /**
     * Refresh Card Data cache
     */
    @PostMapping("/refresh/card-data")
    public ResponseEntity<Map<String, String>> refreshCardDataCache() {
        try {
            cacheUpdateService.refreshCardDataCache();
            Map<String, String> response = new HashMap<>();
            response.put("status", "success");
            response.put("message", "Card Data cache refreshed successfully");
            return ResponseEntity.ok(response);
        } catch (Exception e) {
            Map<String, String> response = new HashMap<>();
            response.put("status", "error");
            response.put("message", "Failed to refresh Card cache: " + e.getMessage());
            return ResponseEntity.internalServerError().body(response);
        }
    }

    /**
     * Get cache status and statistics
     */
    @GetMapping("/status")
    public ResponseEntity<Map<String, Object>> getCacheStatus() {
        try {
            Map<String, Object> status = new HashMap<>();
            
            // Get cache sizes
            @SuppressWarnings("unchecked")
                Map<String, BinInstitutionData> binCache = cacheInitializationService.loadBinInstitutionCache();
            
            status.put("binInstitutionCache", Map.of(
                "size", binCache.size(),
                "status", "active"
            ));
            
          
            
            status.put("lastUpdated", System.currentTimeMillis());
            status.put("status", "healthy");
            
            return ResponseEntity.ok(status);
        } catch (Exception e) {
            Map<String, Object> status = new HashMap<>();
            status.put("status", "error");
            status.put("message", "Failed to get cache status: " + e.getMessage());
            return ResponseEntity.internalServerError().body(status);
        }
    }

    /**
     * Clear specific cache
     */
    @DeleteMapping("/{cacheName}")
    public ResponseEntity<Map<String, String>> clearCache(@PathVariable String cacheName) {
        try {
            switch (cacheName.toLowerCase()) {
                case "bin-institution":
                    cacheUpdateService.refreshBinInstitutionCache();
                    break;
                case "card-data":
                    cacheUpdateService.refreshCardDataCache();
                    break;
                case "all":
                    cacheUpdateService.refreshAllCaches();
                    break;
                default:
                    Map<String, String> response = new HashMap<>();
                    response.put("status", "error");
                    response.put("message", "Unknown cache: " + cacheName);
                    return ResponseEntity.badRequest().body(response);
            }
            
            Map<String, String> response = new HashMap<>();
            response.put("status", "success");
            response.put("message", "Cache '" + cacheName + "' cleared successfully");
            return ResponseEntity.ok(response);
        } catch (Exception e) {
            Map<String, String> response = new HashMap<>();
            response.put("status", "error");
            response.put("message", "Failed to clear cache: " + e.getMessage());
            return ResponseEntity.internalServerError().body(response);
        }
    }
}
