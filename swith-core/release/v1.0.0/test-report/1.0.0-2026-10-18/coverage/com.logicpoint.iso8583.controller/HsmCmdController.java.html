    @PostMapping("/hsm/chd-pan/encrypt")
    public Map<String, Object> encryptPan(@RequestBody PanEncryptionRequest request) {
      Map<String, Object> response = new HashMap<>();

      if (request == null || request.getPan() == null || request.getPan().isBlank()) {
        response.put("success", false);
        response.put("message", "PAN is required");
        return response;
      }

      String pan = request.getPan().trim();
      if (pan.length() < 6) {
        response.put("success", false);
        response.put("message", "PAN must contain at least 6 digits");
        return response;
      }

      Integer institutionId = resolveInstitutionId(pan);
      if (institutionId == null) {
        response.put("success", false);
        response.put("message", "Institution could not be resolved from PAN BIN");
        return response;
      }

      EncryptedFieldValue encryptedPan = cardDataEncryptionService.encryptPan(
              pan,
              institutionId,
              "api|pan|" + institutionId);
      if (encryptedPan == null) {
        response.put("success", false);
        response.put("message", "PAN encryption could not be completed");
        return response;
      }

      response.put("success", true);
      response.put("institutionId", institutionId);
      response.put("maskedPan", cardDataEncryptionService.maskPanForDisplay(pan));
      response.put("panLookupHmac", cardDataEncryptionService.computePanLookupHmac(pan, institutionId));
      response.put("panCt", encryptedPan.getCiphertext());
      response.put("panIv", encryptedPan.getIv());
      response.put("panTag", encryptedPan.getTag());
      response.put("panKeyVersion", encryptedPan.getKeyVersion());
      return response;
    }

    private Integer resolveInstitutionId(String pan) {
      CacheInitializationService.BinInstitutionData binData = cacheInitializationService
              .loadBinInstitutionCache()
              .get(pan.substring(0, 6));
      if (binData == null || binData.getBinDetails() == null) {
        return null;
      }
      return binData.getBinDetails().getInstitution_id();
    }

    public static class PanEncryptionRequest {
      private String pan;

      public String getPan() {
        return pan;
      }

      public void setPan(String pan) {
        this.pan = pan;
      }
    }

}
