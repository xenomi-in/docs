package com.logicpoint.iso8583.service;

import com.logicpoint.iso8583.config.PosDataAuthorizationProperties;
import com.logicpoint.iso8583.context.RequestContext;
import com.logicpoint.iso8583.util.Scheme;
import com.logicpoint.iso8583.validation.TransactionType;
import java.util.ArrayList;
import java.util.List;
import java.util.Map;
import java.util.stream.Collectors;
import org.springframework.stereotype.Service;

/**
 * Validates fixed-position POS data fields using scheme-configured authorization rules.
 */
@Service
public class PosDataAuthorizationValidationService {

    private final PosDataAuthorizationProperties properties;

    public PosDataAuthorizationValidationService(PosDataAuthorizationProperties properties) {
        this.properties = properties;
    }

    public List<String> validate(RequestContext requestContext, Scheme scheme, TransactionType transactionType) {
        List<String> errors = new ArrayList<>();
        if (!properties.isEnabled() || requestContext == null || scheme == null || transactionType == null) {
            return errors;
        }

        PosDataAuthorizationProperties.SchemeRule schemeRule = properties.getRule(scheme.name());
        if (schemeRule == null || !schemeRule.isEnabled()) {
            return errors;
        }

        PosDataAuthorizationProperties.TransactionRule transactionRule =
                schemeRule.getTransactionRules().get(transactionType);
        if (transactionRule == null) {
            return errors;
        }

        String posData = requestContext.getFields().get(schemeRule.getFieldNumber());
        if (isBlank(posData)) {
            if (transactionRule.isRequired()) {
                errors.add("MISSING_DE" + schemeRule.getFieldNumber() + ": POS data is required for "
                        + scheme + " " + transactionType);
            }
            return errors;
        }

        if (posData.length() < schemeRule.getMinLength()) {
            errors.add("INVALID_DE" + schemeRule.getFieldNumber() + ": POS data too short; expected at least "
                    + schemeRule.getMinLength() + " characters, got " + posData.length());
            return errors;
        }

        for (Map.Entry<String, PosDataAuthorizationProperties.SubfieldRule> entry
                : transactionRule.getSubfields().entrySet()) {
            validateSubfield(schemeRule.getFieldNumber(), posData, entry.getKey(), entry.getValue(), scheme,
                    transactionType, errors);
        }

        return errors;
    }

    private void validateSubfield(int fieldNumber,
                                  String posData,
                                  String subfieldName,
                                  PosDataAuthorizationProperties.SubfieldRule rule,
                                  Scheme scheme,
                                  TransactionType transactionType,
                                  List<String> errors) {
        int position = rule.getPosition();
        if (position < 0 || position >= posData.length()) {
            errors.add("INVALID_DE" + fieldNumber + ": " + displayName(subfieldName, rule)
                    + " is not present at configured position " + position);
            return;
        }

        String value = String.valueOf(posData.charAt(position)).toUpperCase();
        if (!rule.getAllowedValues().isEmpty() && !rule.getAllowedValues().contains(value)) {
            String allowed = rule.getAllowedValues().stream().sorted().collect(Collectors.joining(","));
            errors.add("INVALID_DE" + fieldNumber + ": " + displayName(subfieldName, rule)
                    + " expected one of [" + allowed + "] for " + scheme + " " + transactionType
                    + ", got " + value);
        }
    }

    private String displayName(String subfieldName, PosDataAuthorizationProperties.SubfieldRule rule) {
        return isBlank(rule.getDescription()) ? subfieldName : rule.getDescription();
    }

    private boolean isBlank(String value) {
        return value == null || value.trim().isEmpty();
    }
}
