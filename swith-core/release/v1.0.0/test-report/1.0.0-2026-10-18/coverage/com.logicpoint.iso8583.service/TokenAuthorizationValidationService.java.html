package com.logicpoint.iso8583.service;

import com.logicpoint.iso8583.config.TokenAuthorizationProperties;
import com.logicpoint.iso8583.context.RequestContext;
import com.logicpoint.iso8583.util.Scheme;
import com.logicpoint.iso8583.validation.TransactionType;
import java.util.ArrayList;
import java.util.List;
import java.util.Map;
import java.util.Set;
import org.springframework.stereotype.Service;

/**
 * Validates tokenized ISO8583 authorization indicators without depending on TSP APIs.
 */
@Service
public class TokenAuthorizationValidationService {

    public static final String ATTR_TOKENIZED_AUTHORIZATION = "tokenized_authorization";
    public static final String ATTR_TOKEN_VALIDATION_RESULT = "token_validation_result";

    private final TokenAuthorizationProperties properties;

    public TokenAuthorizationValidationService(TokenAuthorizationProperties properties) {
        this.properties = properties;
    }

    public List<String> validate(RequestContext requestContext, Scheme scheme, TransactionType transactionType) {
        List<String> errors = new ArrayList<>();
        if (!properties.isEnabled() || requestContext == null || scheme == null || transactionType == null) {
            return errors;
        }

        TokenAuthorizationProperties.SchemeRule rule = properties.getRule(scheme.name());
        if (rule == null || !rule.isEnabled() || !rule.getTransactionTypes().contains(transactionType)) {
            return errors;
        }

        boolean tokenized = isTokenized(requestContext, rule);
        requestContext.setAttribute(ATTR_TOKENIZED_AUTHORIZATION, tokenized);
        if (!tokenized && !rule.isValidateWhenTokenDataPresent()) {
            return errors;
        }

        String tokenData = requestContext.getFields().get(rule.getTokenDataField());
        if (isBlank(tokenData)) {
            if (tokenized && rule.isRequireTokenValidationResult()) {
                errors.add("MISSING_TOKEN_DATA: DE" + rule.getTokenDataField()
                        + " is required for tokenized " + scheme + " " + transactionType);
            }
            return errors;
        }

        Map<String, String> tokenTags;
        try {
            tokenTags = requestContext.getParsedTlv(rule.getTokenDataField());
        } catch (Exception ex) {
            errors.add("INVALID_TOKEN_DATA_FORMAT: Unable to parse DE" + rule.getTokenDataField()
                    + " token data as TLV");
            return errors;
        }

        String tokenValidationResult = normalize(tokenTags.get(rule.getTokenValidationResultTag()));
        requestContext.setAttribute(ATTR_TOKEN_VALIDATION_RESULT, tokenValidationResult);
        if (isBlank(tokenValidationResult)) {
            errors.add("MISSING_TOKEN_VALIDATION_RESULT: DE" + rule.getTokenDataField()
                    + " tag " + rule.getTokenValidationResultTag()
                    + " is required when token data is present for " + scheme + " " + transactionType);
            return errors;
        }

        if (!isBlank(tokenValidationResult)) {
            validateTokenResult(rule, tokenValidationResult, errors);
        }

        return errors;
    }

    private boolean isTokenized(RequestContext requestContext, TokenAuthorizationProperties.SchemeRule rule) {
        String posData = requestContext.getFields().get(rule.getPosDataField());
        int position = rule.getTokenIndicatorPosition();
        if (isBlank(posData) || position < 0 || position >= posData.length()) {
            return false;
        }
        String indicator = String.valueOf(posData.charAt(position)).toUpperCase();
        return rule.getTokenIndicatorValues().contains(indicator);
    }

    private void validateTokenResult(TokenAuthorizationProperties.SchemeRule rule,
                                     String tokenValidationResult,
                                     List<String> errors) {
        Set<String> approvedValues = rule.getApprovedTokenValidationValues();
        Set<String> declinedValues = rule.getDeclinedTokenValidationValues();
        if (declinedValues.contains(tokenValidationResult)) {
            errors.add("TOKEN_VALIDATION_FAILED: Token validation result " + tokenValidationResult
                    + " indicates token validation failure");
            return;
        }

        if (!approvedValues.isEmpty() && !approvedValues.contains(tokenValidationResult)) {
            errors.add("INVALID_TOKEN_VALIDATION_RESULT: Token validation result " + tokenValidationResult
                    + " is not configured as an approved value");
        }
    }

    private String normalize(String value) {
        return isBlank(value) ? null : value.trim().toUpperCase();
    }

    private boolean isBlank(String value) {
        return value == null || value.trim().isEmpty();
    }
}
