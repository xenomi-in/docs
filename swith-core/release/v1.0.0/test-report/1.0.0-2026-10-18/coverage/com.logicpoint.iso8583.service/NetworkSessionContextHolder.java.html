package com.logicpoint.iso8583.service;

/**
 * Thread-local holder for optional listener/session metadata.
 * This keeps session-scoped key resolution additive and non-invasive.
 */
public final class NetworkSessionContextHolder {

    private static final ThreadLocal<NetworkSessionContext> CURRENT = new ThreadLocal<>();

    private NetworkSessionContextHolder() {
    }

    public static void set(String siteId, String listenerId, String connectionId, String scheme) {
        if (isBlank(siteId) || isBlank(listenerId)) {
            CURRENT.remove();
            return;
        }
        CURRENT.set(new NetworkSessionContext(siteId, listenerId, connectionId, scheme));
    }

    public static NetworkSessionContext get() {
        return CURRENT.get();
    }

    public static void clear() {
        CURRENT.remove();
    }

    private static boolean isBlank(String value) {
        return value == null || value.isBlank();
    }

    public record NetworkSessionContext(String siteId, String listenerId, String connectionId, String scheme) {
    }
}
