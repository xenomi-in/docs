package com.logicpoint.iso8583.service;

public class HsmWrappedKeyResult {
    private final String wrappedKey;
    private final String kcv;
    private final String importedKeyUnderMfk;

    public HsmWrappedKeyResult(String wrappedKey, String kcv, String importedKeyUnderMfk) {
        this.wrappedKey = wrappedKey;
        this.kcv = kcv;
        this.importedKeyUnderMfk = importedKeyUnderMfk;
    }

    public String getWrappedKey() {
        return wrappedKey;
    }

    public String getKcv() {
        return kcv;
    }

    public String getImportedKeyUnderMfk() {
        return importedKeyUnderMfk;
    }
}
