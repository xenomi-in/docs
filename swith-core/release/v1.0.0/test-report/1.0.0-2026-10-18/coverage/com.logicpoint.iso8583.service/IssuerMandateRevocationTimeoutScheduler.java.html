package com.logicpoint.iso8583.service;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.scheduling.annotation.Scheduled;
import org.springframework.stereotype.Component;

/** Marks issuer-originated revocations timed out when no 0312 arrives in time. */
@Component
public class IssuerMandateRevocationTimeoutScheduler {

    private static final Logger log = LoggerFactory.getLogger(IssuerMandateRevocationTimeoutScheduler.class);

    private final IssuerMandateRevocationLifecycleService lifecycleService;

    public IssuerMandateRevocationTimeoutScheduler(IssuerMandateRevocationLifecycleService lifecycleService) {
        this.lifecycleService = lifecycleService;
    }

    @Scheduled(fixedDelayString = "${authorization.issuer-mandate-revocation.timeout-scan-delay:60000}")
    public void markExpiredRevocations() {
        int expiredCount = lifecycleService.markExpiredPendingRevocations();
        if (expiredCount > 0) {
            log.warn("Marked {} issuer mandate revocation command(s) as timed out", expiredCount);
        }
    }
}
