    /**

    private static class TokenInfo {
        private final String accessToken;
        private final LocalDateTime expiresAt;
        
        public TokenInfo(String accessToken, LocalDateTime expiresAt) {
            this.accessToken = accessToken;
            this.expiresAt = expiresAt;
        }
        
        public String getAccessToken() {
            return accessToken;
        }
        
        public LocalDateTime getExpiresAt() {
            return expiresAt;
        }
        
        public boolean isExpired() {
            return LocalDateTime.now().isAfter(expiresAt);
        }

        public boolean shouldRefresh(long thresholdSeconds) {
            return LocalDateTime.now().plusSeconds(thresholdSeconds).isAfter(expiresAt);
        }
    }

        log.debug("Starting authorization request for amount: {} {}", amount, currency);
        
        try {
            // Get valid access token
            String accessToken = getValidAccessToken();
            
            // Build authorization request
            AccountingAuthorizationRequest request = buildAuthorizationRequest(
                transactionData, amount, currency, mti, network,transactionType);
                String jsonString = accountingJsonMapper.writeValueAsString(request);
            RawDataLogging.value("Accounting authorization request", jsonString);
            // Prepare headers
            HttpHeaders headers = new HttpHeaders();
            headers.setContentType(MediaType.APPLICATION_JSON);
            headers.setBearerAuth(accessToken);
            
            HttpEntity<AccountingAuthorizationRequest> entity = new HttpEntity<>(request, headers);
            
            // Make API call
            String url = apiProperties.getBaseUrl() + apiProperties.getAuthorizationEndpoint();
            log.debug("Calling authorization API: {}", url);
            
            ResponseEntity<AccountingAuthorizationResponse> response = restTemplate.exchange(
                url, HttpMethod.POST, entity, AccountingAuthorizationResponse.class);
                //JsonMapper objectMapper = JsonMapper.builderWithJackson2Defaults().build();
                String jsonResString = accountingJsonMapper.writeValueAsString(response);
                RawDataLogging.value("Accounting authorization response", jsonResString);
            if (response.getStatusCode().is2xxSuccessful() && response.getBody() != null) {
                log.info("Authorization successful. Transaction ID: {}, Response Code: {}", 
                    response.getBody().getTransactionId(), response.getBody().getResponseCode());
                    RawDataLogging.value("Accounting authorization response body", response.getBody());
                return response.getBody();
            } else {
                log.error("Authorization failed with status: {}", response.getStatusCode());
                throw new RuntimeException("Authorization API returned unsuccessful response");
            }
            
        } catch (HttpClientErrorException e) {
            AccountingAuthorizationResponse errorResponse = parseAuthorizationError(e);
            if (errorResponse != null) {
                return errorResponse;
            }
            return null;
        } catch (HttpServerErrorException e) {
            log.error("Server error during authorization: status={}", e.getStatusCode());
            RawDataLogging.value("Accounting authorization error body", e.getResponseBodyAsString());
            throw new RuntimeException("Authorization service unavailable: " + e.getMessage(), e);
        } catch (ResourceAccessException e) {
            log.error("Network error during authorization: {}", e.getMessage());
            throw new RuntimeException("Network error during authorization: " + e.getMessage(), e);
        } catch (Exception e) {
            log.error("Unexpected error during authorization: {}", e.getMessage(), e);
            throw new RuntimeException("Authorization failed: " + e.getMessage(), e);
        }
    }

    private AccountingAuthorizationResponse parseAuthorizationError(HttpClientErrorException exception) {
        log.warn("Accounting authorization declined: {} - {}",
                exception.getStatusCode(), exception.getResponseBodyAsString());
        return parseAuthorizationErrorBody(exception.getResponseBodyAsString());
    }

    AccountingAuthorizationResponse parseAuthorizationErrorBody(String responseBody) {
        if (responseBody == null || responseBody.isBlank()) {
            return null;
        }
        try {
            AccountingAuthorizationResponse errorResponse = accountingJsonMapper.readValue(
                    responseBody, AccountingAuthorizationResponse.class);
            if (errorResponse.getErrorCode() != null && errorResponse.getResponseCode() == null) {
                errorResponse.setResponseCode(errorResponse.getErrorCode());
            }
            log.info("Parsed accounting decline - ErrorCode: {}, ErrorMessage: {}, TransactionId: {}",
                    errorResponse.getErrorCode(), errorResponse.getErrorMessage(), errorResponse.getTransactionId());
            return errorResponse;
        } catch (Exception parseException) {
            log.error("Failed to parse accounting error response body: {}", parseException.getMessage());
            return null;
        }
    }
    
    /**
     * Gets a valid access token, refreshing if necessary
     */
    private String getValidAccessToken() {
        TokenInfo tokenInfo = tokenCache.get(DEFAULT_TOKEN_CACHE_KEY);
        long refreshThresholdSeconds = Math.max(0, apiProperties.getTokenRefreshThreshold());

        if (tokenInfo == null || tokenInfo.shouldRefresh(refreshThresholdSeconds)) {
            synchronized (tokenRefreshLock) {
                tokenInfo = tokenCache.get(DEFAULT_TOKEN_CACHE_KEY);
                if (tokenInfo == null || tokenInfo.shouldRefresh(refreshThresholdSeconds)) {
                    log.debug("Token missing, expired, or within refresh threshold; refreshing access token");
                    tokenInfo = refreshAccessToken();
                    tokenCache.put(DEFAULT_TOKEN_CACHE_KEY, tokenInfo);
                } else {
                    log.debug("Using access token refreshed by another in-flight request");
                }
            }
        }

        return tokenInfo.getAccessToken();
    }
    
    /**
     * Refreshes the access token by calling login API
     */
    @Retryable(
        value = {HttpServerErrorException.class, ResourceAccessException.class},
        maxAttempts = 3,
        backoff = @Backoff(delay = 1000, multiplier = 2)
    )
    private TokenInfo refreshAccessToken() {
        try {
            AccountingLoginRequest loginRequest = new AccountingLoginRequest();
            loginRequest.setClientId(apiProperties.getUsername());
            loginRequest.setSecret(apiProperties.getPassword());
            
            // 3. Convert the POJO to a JSON string
            String jsonString = accountingJsonMapper.writeValueAsString(loginRequest);

            HttpHeaders headers = new HttpHeaders();
            headers.setContentType(MediaType.APPLICATION_JSON);
            
            HttpEntity<AccountingLoginRequest> entity = new HttpEntity<>(loginRequest, headers);
          
            RawDataLogging.value("Accounting login request body",
                    jsonString.replace(apiProperties.getPassword(), "****"));
            String url = apiProperties.getBaseUrl() + apiProperties.getLoginEndpoint();
            log.debug("Calling login API: {}", url);
            
            ResponseEntity<AccountingLoginResponse> response = restTemplate.exchange(
                url, HttpMethod.POST, entity, AccountingLoginResponse.class);
            // Access tokens are never logged, including in diagnostic raw-data mode.
            if (response.getStatusCode().is2xxSuccessful() && response.getBody() != null) {
                AccountingLoginResponse loginResponse = response.getBody();
                LocalDateTime expiresAt = LocalDateTime.now().plusSeconds(loginResponse.getExpiresIn());
                
                log.info("Successfully obtained access token, expires at: {}", expiresAt);
                return new TokenInfo(loginResponse.getAccessToken(), expiresAt);
            } else {
                throw new RuntimeException("Login API returned unsuccessful response");
            }
            
        } catch (Exception e) {
            log.error("Failed to refresh access token: {}", e.getMessage(), e);
            throw new RuntimeException("Failed to obtain access token: " + e.getMessage(), e);
        }
    }

    String resolveAuthenticationMethod(Map<Integer, String> transactionData, String transactionType) {
        if (transactionType != null && ATM_TRANSACTION_TYPES.contains(transactionType)) {
            return "ATM";
        }
        if (transactionType != null && ECOMMERCE_TRANSACTION_TYPES.contains(transactionType)) {
            return "E-comm";
        }

        String posEntryMode = transactionData.get(22);
        String posConditionCode = transactionData.get(25);
        String processingCode = transactionData.get(3);
        if (startsWithAny(posEntryMode, "001", "011", "061", "810", "811")
                || "59".equals(posConditionCode) || "08".equals(posConditionCode)
                || "06".equals(posConditionCode) || "07".equals(posConditionCode)) {
            return "E-comm";
        }
        if (startsWithAny(processingCode, "01", "30", "31", "38", "40", "91")
                || startsWithAny(posEntryMode, "010", "020")
                || "51".equals(posConditionCode) || "52".equals(posConditionCode)) {
            return "ATM";
        }
        return "POS";
    }

    private boolean startsWithAny(String value, String... prefixes) {
        if (value == null || value.isBlank()) {
            return false;
        }
        for (String prefix : prefixes) {
            if (value.startsWith(prefix)) {
                return true;
            }
        }
        return false;
    }

    private void applyStandingInstructionData(AccountingAuthorizationRequest request,
                                              Map<Integer, String> transactionData,
                                              String transactionType) {
        if (transactionType == null || !transactionType.startsWith("SI_")) {
            return;
        }

        String de48 = transactionData.get(48);
        if (de48 == null || de48.isBlank()) {
            return;
        }

        Map<String, String> de48Tags;
        try {
            de48Tags = TlvUtils.parseTlv(de48);
        } catch (Exception ex) {
            log.warn("Unable to parse DE48 for SI accounting data: {}", ex.getMessage());
            return;
        }

        String siData = de48Tags.get("085");
        if (siData == null || siData.length() < 49) {
            return;
        }

        AccountingAuthorizationRequest.SiReferenceData referenceData =
                new AccountingAuthorizationRequest.SiReferenceData();
        referenceData.setRegistrationIndicator(siData.substring(0, 2));
        referenceData.setNoOfSITxns(siData.substring(2, 5));
        referenceData.setPreferredInitiationDate(siData.substring(5, 7));
        referenceData.setSequenceNumber(resolveSiSequenceNumber(siData, de48Tags));
        request.setSiReference(referenceData);

        if (Set.of("SI_REGISTRATION", "SI_EDIT").contains(transactionType)) {
            AccountingAuthorizationRequest.SiRegistrationData registrationData =
                    new AccountingAuthorizationRequest.SiRegistrationData();
            registrationData.setMinAmount(siData.substring(7, 19));
            registrationData.setMaxAmount(siData.substring(19, 31));
            registrationData.setStartDate(formatSiDate(siData.substring(31, 39)));
            registrationData.setEndDate(formatSiDate(siData.substring(39, 47)));
            registrationData.setFrequency(formatSiFrequency(siData.substring(48, 49)));
            request.setSiRegistration(registrationData);
        }
    }

    private String resolveSiSequenceNumber(String siData, Map<String, String> de48Tags) {
        String sequenceFromTag = de48Tags.get("089");
        if (sequenceFromTag != null && !sequenceFromTag.isBlank()) {
            return sequenceFromTag.trim();
        }
        return siData.substring(2, 5);
    }

    private String formatSiDate(String ddMMyyyy) {
        if (ddMMyyyy == null || ddMMyyyy.isBlank() || "00000000".equals(ddMMyyyy)) {
            return null;
        }
        return ddMMyyyy.substring(4, 8) + "-" + ddMMyyyy.substring(2, 4) + "-" + ddMMyyyy.substring(0, 2);
    }

    private String formatSiFrequency(String frequencyIndicator) {
        if (frequencyIndicator == null || frequencyIndicator.isBlank()) {
            return null;
        }
        return String.format("%02d", Integer.parseInt(frequencyIndicator));
    }

    private AccountingAuthorizationRequest buildAuthorizationRequest(
            RequestContext requestContext,
            BigDecimal amount,
            String currency,
            String mti,
            String network,
            String transactionType) {
        Map<Integer, String> transactionData = requestContext.getFields();
        AccountingAuthorizationRequest request = buildAuthorizationRequest(
                transactionData,
                amount,
                currency,
                mti,
                network,
                transactionType);
        Map<String, String> de48Tlv = requestContext.getParsedTlv(48);
        if (transactionType != null && (transactionType.equals("SERVICE_CREATION") || transactionType.equals("SERVICE_CREATION_CONTACTLESS"))) {
            AccountingAuthorizationRequest.Ncmc ncmc = new AccountingAuthorizationRequest.Ncmc();
            ncmc.setServiceID(de48Tlv.get("084"));
            ncmc.setType(de48Tlv.get("079"));
            request.setNcmc(ncmc);
        } else if ("MONEY_LOAD".equals(transactionType)) {
            AccountingAuthorizationRequest.Ncmc ncmc = new AccountingAuthorizationRequest.Ncmc();
            ncmc.setId(de48Tlv.get("079"));
            ncmc.setType(de48Tlv.get("082"));
            ncmc.setServiceID(de48Tlv.get("084"));
            request.setNcmc(ncmc);
        } else if ("ML_BALANCEUPDATE".equals(transactionType)) {
            AccountingAuthorizationRequest.Ncmc ncmc = new AccountingAuthorizationRequest.Ncmc();
            ncmc.setType(de48Tlv.get("082"));
            request.setNcmc(ncmc);
        }
        request.setNetwork(requestContext.getScheme() != null ? requestContext.getScheme().name() : network);
        request.setTransactionId(requestContext.getUniqueTransactionId());
        return request;
    }

 */
@Retryable(
    value = {HttpServerErrorException.class, ResourceAccessException.class},
    maxAttempts = 3,
    backoff = @Backoff(delay = 1000, multiplier = 2)
)
public AccountingAuthorizationResponse authorizeReversal(
        RequestContext requestContext,
        BigDecimal amount,
        String currency,
        String originalMti,
        String network,
        String originalAccountingTxnId) {

    log.debug("Starting reversal request for original transaction: {}", originalAccountingTxnId);

    try {
        String accessToken = getValidAccessToken();

        AccountingAuthorizationRequest request = buildReversalRequest(
            requestContext, amount, currency, originalMti, network, originalAccountingTxnId);

        String jsonString = accountingJsonMapper.writeValueAsString(request);
        RawDataLogging.value("Accounting reversal request", jsonString);
