    /**
     * Processes a reversal request.
     *
     * @param fields Map of ISO 8583 field numbers to their values
     * @param scheme The payment scheme
     * @return Processed message fields with response codes
     */
    public Map<Integer, String> processReversalRequest(RequestContext requestContext) {
        Map<Integer, String> fields = requestContext.getFields();
        Scheme scheme = requestContext.getScheme();
        log.info("Processing reversal request for scheme: {}", scheme);
        responseAssembler.setResponseMti(requestContext);
        
        try {
            // Extract key fields for reversal processing
            String processingCode = fields.get(3);
            String posEntryMode = fields.get(22);
            String posConditionCode = fields.get(25);
            String de48 = fields.getOrDefault(48, "");
            String de90 = fields.getOrDefault(90, ""); // Original Data Elements
            String de120Tag001 = fields.getOrDefault(120, "");
            String rrn = fields.get(37);
            String stan = fields.get(11);
            String terminalId = fields.get(41);
            String transmissionTime = fields.get(7);
            
           
            
            // Validate required fields for reversal
            if (rrn == null || rrn.trim().isEmpty()) {
                log.error("Missing RRN for reversal request");
                responseAssembler.applyResponseCode(requestContext, InternalResCode.FORMAT_ERROR);
                return fields;
            }
            
            if (stan == null || stan.trim().isEmpty()) {
                log.error("Missing STAN for reversal request");
                responseAssembler.applyResponseCode(requestContext, InternalResCode.FORMAT_ERROR);
                return fields;
            }
            
            // Validate DE90 (Original Data Elements) if present
            if (de90 != null && !de90.trim().isEmpty()) {
                if (!validateDe90(de90, fields)) {
                    log.error("DE90 validation failed for reversal request");
                    responseAssembler.applyResponseCode(requestContext, InternalResCode.FORMAT_ERROR);
                    return fields;
                }
            }
            
            // Find the original transaction to reverse
            Transaction originalTransaction = findOriginalTransaction(rrn, stan, terminalId, transmissionTime,de90.substring(0,4));
            
            if (originalTransaction == null) {
                log.warn("Original transaction not found for reversal - RRN: {}, STAN: {}", rrn, stan);
                responseAssembler.applyResponseCode(requestContext, InternalResCode.INVALID_TRANSACTION);
                return fields;
            }
            
            // Check if transaction is already reversed
            if (isTransactionAlreadyReversed(originalTransaction)) {
                log.warn("Transaction already reversed - RRN: {}, STAN: {}", rrn, stan);
                responseAssembler.applyResponseCode(requestContext, InternalResCode.DUPLICATE_TRANSACTION);
                return fields;
            }
            
            // Check if reversal is within time limit
            if (!isReversalWithinTimeLimit(originalTransaction, transmissionTime)) {
                log.warn("Reversal request outside time limit - RRN: {}, STAN: {}", rrn, stan);
                responseAssembler.applyResponseCode(requestContext, InternalResCode.INVALID_AMOUNT);
                return fields;
            }
            
            // Process reversal based on scheme
            switch (scheme) {
                case RUPAY:
                    return processRuPayReversal(requestContext, originalTransaction, processingCode, posEntryMode, posConditionCode, de48, de90, de120Tag001);
                case VISA:
                    return processVisaReversal(requestContext, originalTransaction, processingCode, posEntryMode, posConditionCode, de90);
                case MASTERCARD:
                    return processMastercardReversal(requestContext, originalTransaction, processingCode, posEntryMode, posConditionCode, de90);
                default:
                    log.warn("Unsupported scheme for reversal: {}", scheme);
                    responseAssembler.applyResponseCode(requestContext, InternalResCode.DO_NOT_HONOR);
                    return fields;
            }
            
        } catch (Exception e) {
            log.error("Error processing reversal request", e);
            responseAssembler.applyResponseCode(requestContext, InternalResCode.SYSTEM_ERROR);
            return fields;
        }
    }

    /**
     * Validates DE90 (Original Data Elements) for reversal requests.
     * DE90 contains: Original MTI (4) + Original STAN (6) + Original Transmission Date (4) + Original Transmission Time (6)
     *
     * @param de90 Original Data Elements field
     * @param fields Current message fields
     * @return true if validation passes, false otherwise
     */
    private boolean validateDe90(String de90, Map<Integer, String> fields) {
        try {
            log.info("Validating DE90: {}", de90);
            
            // DE90 should be 20 characters: MTI(4) + STAN(6) + Date(4) + Time(6)
            if (de90.length() != 42) {
                log.error("DE90 length invalid: expected 42, got {}", de90.length());
                return false;
            }
            
            // Extract components from DE90
            // DE90 format breakdown based on specification:
            //  1-4:   Original message type (MTI)           [0,4)
            //  5-10:  Original STAN number                  [4,10)
            // 11-20:  Original Transmission date and time   [10,20)
            // 21-31:  Original acquirer ID                  [20,31)
            // 32-42:  Original forwarding institution id    [31,42)
            // Example: "02001234561231231230012345678901234567890"
            String originalMti = de90.substring(0, 4);
            String originalStan = de90.substring(4, 10);
            String originalTransDateTime = de90.substring(10, 20);
            String originalAcquirerId = de90.length() >= 31 ? de90.substring(20, 31) : null;
            String originalForwardingInstId = de90.length() >= 42 ? de90.substring(31, 42) : null;
           
            
            log.debug("DE90 components - MTI: {}, STAN: {}, DateTime: {}", 
                     originalMti, originalStan, originalTransDateTime);
            
            // Validate original MTI (should be a financial transaction)
            if (!isValidFinancialMti(originalMti)) {
                log.error("Invalid original MTI in DE90: {}", originalMti);
                return false;
            }
            
            // Validate original STAN format (should be numeric)
            if (!originalStan.matches("\\d{6}")) {
                log.error("Invalid original STAN format in DE90: {}", originalStan);
                return false;
            }
            
            // Validate original date format (MMDD)
            if (!originalTransDateTime.matches("\\d{10}")) {
                log.error("Invalid original date format in DE90: {}", originalTransDateTime);
                return false;
            }
            
          
            
            // Cross-validate with current message fields if available
            String currentStan = fields.get(11);
            if (currentStan != null && !currentStan.equals(originalStan)) {
                log.warn("DE90 STAN {} does not match current STAN {}", originalStan, currentStan);
                // This might be acceptable in some cases, so we log as warning, not error
            }
            
            log.info("DE90 validation successful");
            return true;
            
        } catch (Exception e) {
            log.error("Error validating DE90: {}", e.getMessage());
            return false;
        }
    }

    /**
     * Checks if the MTI represents a valid financial transaction.
     *
     * @param mti Message Type Indicator
     * @return true if valid financial MTI, false otherwise
     */
    private boolean isValidFinancialMti(String mti) {
        // Valid financial transaction MTIs
        return "0100".equals(mti) || "0200".equals(mti) || 
               "0101".equals(mti) || "0201".equals(mti) ||
               "0400".equals(mti) || "0420".equals(mti);
    }

    /**
     * Finds the original transaction to be reversed.
     *
     * @param rrn Retrieval Reference Number
     * @param stan System Trace Audit Number
     * @param terminalId Terminal ID
     * @param transmissionTime Transmission time
     * @return Original transaction or null if not found
     */
    private Transaction findOriginalTransaction(String rrn, String stan, String terminalId, String transmissionTime,String orgMti) {
        try {
            // First try to find by RRN , Terminal id and mti
            List<Transaction> transactions = transactionRepository.findByRrnAndTID(terminalId, rrn, transmissionTime, orgMti);
            
            if (!transactions.isEmpty()) {
                // If multiple found, prefer the most recent one
                return transactions.stream()
                        .max((t1, t2) -> t1.getDe7TransmissionDatetime().compareTo(t2.getDe7TransmissionDatetime()))
                        .orElse(null);
            }
            // Convert transmissionTime (MMDDhhmmss) into LocalDate
            // Assumes current year; adjust as needed for cross-year reversals
            LocalDate transmissionDate = null;
            if (transmissionTime != null && transmissionTime.length() >= 4) {
                int month = Integer.parseInt(transmissionTime.substring(0, 2));
                int day = Integer.parseInt(transmissionTime.substring(2, 4));
                int year = LocalDate.now().getYear();
                try {
                    transmissionDate = LocalDate.of(year, month, day);
                } catch (Exception ex) {
                    log.warn("Failed to parse transmissionTime {} to LocalDate: {}", transmissionTime, ex.getMessage());
                }
            }
            // If not found by RRN and STAN, try by terminal and time
            if (terminalId != null && transmissionTime != null) {
                List<Transaction> terminalTransactions = transactionRepository.findByTerminalAndDate(terminalId, transmissionTime);
                if (!terminalTransactions.isEmpty()) {
                    return terminalTransactions.get(0);
                }
            }
            
            return null;
        } catch (Exception e) {
            log.error("Error finding original transaction", e);
            return null;
        }
    }

    /**
     * Checks if the transaction is already reversed.
     *
     * @param transaction The transaction to check
     * @return true if already reversed, false otherwise
     */
    private boolean isTransactionAlreadyReversed(Transaction transaction) {
        // Check if there's a reversal record for this transaction
        try {
            List<TransactionResponse> reversals = transactionResponseRepository.findByOriginalData(transaction.getDe90OriginalDataElements());
            return !reversals.isEmpty();
        } catch (Exception e) {
            log.error("Error checking if transaction is already reversed", e);
            return false;
        }
    }

    /**
     * Checks if reversal is within the allowed time limit.
     *
     * @param originalTransaction The original transaction
     * @param reversalTime The reversal request time
     * @return true if within time limit, false otherwise
     */
    private boolean isReversalWithinTimeLimit(Transaction originalTransaction, String reversalTime) {
        try {
            // Parse original transaction time
            String originalTime = originalTransaction.getDe7TransmissionDatetime();
            
            // Calculate time difference (assuming MMDDhhmmss format)
            long originalTimestamp = parseTransmissionTime(originalTime);
            long reversalTimestamp = parseTransmissionTime(reversalTime);
            
            // Allow reversal within 24 hours (86400 seconds)
            long timeDifference = reversalTimestamp - originalTimestamp;
            return timeDifference <= 86400 && timeDifference >= 0;
            
        } catch (Exception e) {
            log.error("Error checking reversal time limit", e);
            return false;
        }
    }

    /**
     * Parses transmission time from MMDDhhmmss format to timestamp.
     *
     * @param transmissionTime Time in MMDDhhmmss format
     * @return Timestamp in seconds
     */
    private long parseTransmissionTime(String transmissionTime) {
        if (transmissionTime == null || transmissionTime.length() != 10) {
            throw new IllegalArgumentException("Invalid transmission time format");
        }
        
        try {
            int month = Integer.parseInt(transmissionTime.substring(0, 2));
            int day = Integer.parseInt(transmissionTime.substring(2, 4));
            int hour = Integer.parseInt(transmissionTime.substring(4, 6));
            int minute = Integer.parseInt(transmissionTime.substring(6, 8));
            int second = Integer.parseInt(transmissionTime.substring(8, 10));
            
            // Create a timestamp (simplified - assumes current year)
            java.time.LocalDateTime dateTime = java.time.LocalDateTime.now()
                    .withMonth(month)
                    .withDayOfMonth(day)
                    .withHour(hour)
                    .withMinute(minute)
                    .withSecond(second);
            
            return dateTime.toEpochSecond(java.time.ZoneOffset.UTC);
        } catch (Exception e) {
            throw new IllegalArgumentException("Error parsing transmission time: " + transmissionTime, e);
        }
    }

    /**
     * Processes RuPay reversal request.
     *
     * @param fields Message fields
     * @param originalTransaction Original transaction
     * @param processingCode Processing code
     * @param posEntryMode POS entry mode
     * @param posConditionCode POS condition code
     * @param de48 Additional data
     * @param de90 Original data elements
     * @param de120Tag001 Tag 001 from DE120
     * @return Processed fields
     */
    private Map<Integer, String> processRuPayReversal(RequestContext requestContext, Transaction originalTransaction,
                                                     String processingCode, String posEntryMode, String posConditionCode, 
                                                     String de48, String de90, String de120Tag001) {
        Map<Integer, String> fields = requestContext.getFields();
        log.info("Processing RuPay reversal request");
        
        try {
            // Identify transaction type
           // RuPayTxnType type = RuPayTxnType.identifyByNetwork(processingCode, posEntryMode, posConditionCode, de48, de120Tag001, "RuPay");
            log.info("RuPay reversal transaction type: {}");
            
            // Validate reversal amount matches original (if amount is present)
            if (fields.containsKey(4)) {
                // Convert field 4 (amount) into double, considering rightmost 2 digits as fraction part
                String field4 = fields.get(4);
                BigDecimal reversalAmountDouble = BigDecimal.ZERO;
                if (field4 != null && !field4.isEmpty() && field4.matches("\\d+")) {
                    long amountLong = Long.parseLong(field4);
                    reversalAmountDouble = BigDecimal.valueOf(amountLong / 100.0);
                }
            
                BigDecimal originalAmount = originalTransaction.getDe4AmountTransaction() != null ? 
                        originalTransaction.getDe4AmountTransaction(): null;
                
                if (originalAmount != null && reversalAmountDouble.compareTo(originalAmount) != 0) {
                    log.warn("Reversal amount {} does not match original amount {}", reversalAmountDouble, originalAmount);
                    responseAssembler.applyResponseCode(requestContext, InternalResCode.INVALID_AMOUNT);
                    return fields;
                }
            }
            
            // Create reversal transaction record
            Transaction reversalTransaction = createReversalTransaction(originalTransaction, fields);
            Transaction savedReversal = transactionRepository.save(reversalTransaction);
            
           
            if (savedReversal != null) {
                log.info("Reversal transaction saved with ID: {}", savedReversal.getTransactionId());

                if (isApplicationProcessedReversal()) {
                    log.info("Approving RuPay reversal in application based on reversal.processing-mode={}", reversalProcessingMode);
                    approveReversalInApplication(requestContext, originalTransaction);
                    return fields;
                }
                
                AccountingAuthorizationResponse authResponse = accountingApiService.authorizeReversal(
                    requestContext,
                    new BigDecimal(fields.get(4)),
                    fields.get(49),
                    de90.substring(0, 4),
                    requestContext.getScheme().name(),
                    originalTransaction.getTransactionId());

                    if (authResponse != null && !"00".equals(authResponse.getResponseCode())) {
                        responseAssembler.applyResponseCode(requestContext, InternalResCode.DO_NOT_HONOR);
                        return fields;
                    }
                // Update original transaction status
                approveReversalInApplication(requestContext, originalTransaction);
                
               
                
            } 
            
        } catch (Exception e) {
            log.error("Error processing RuPay reversal", e);
            responseAssembler.applyResponseCode(requestContext, InternalResCode.SYSTEM_ERROR);
        }
        
        return fields;
    }

    private boolean isApplicationProcessedReversal() {
        return "application".equalsIgnoreCase(reversalProcessingMode != null ? reversalProcessingMode.trim() : "");
    }

    private void approveReversalInApplication(RequestContext requestContext, Transaction originalTransaction) {
        originalTransaction.setStatus(Transaction.Status.FAILED);
        transactionRepository.save(originalTransaction);

        responseAssembler.applyResponseCode(requestContext, InternalResCode.APPROVED);
        applyReversalAuthCode(requestContext, originalTransaction);
    }

    void applyReversalAuthCode(RequestContext requestContext, Transaction originalTransaction) {
        String requestAuthCode = requestContext.getFields().get(38);
        if (requestAuthCode != null && !requestAuthCode.isBlank()) {
            requestContext.getFields().put(38, requestAuthCode);
            return;
        }

        String originalAuthCode = originalTransaction.getDe38AuthApprovalCode();
        if (originalAuthCode != null && !originalAuthCode.isBlank()) {
            requestContext.getFields().put(38, originalAuthCode);
        } else {
            requestContext.getFields().remove(38);
            log.warn("Reversal request and original transaction {} do not contain DE38; omitting DE38 in response",
                    originalTransaction.getTransactionId());
        }
    }

    /**
     * Processes Visa reversal request.
     *
     * @param fields Message fields
     * @param originalTransaction Original transaction
     * @param processingCode Processing code
     * @param posEntryMode POS entry mode
     * @param posConditionCode POS condition code
     * @param de90 Original data elements
     * @return Processed fields
     */
    private Map<Integer, String> processVisaReversal(RequestContext requestContext, Transaction originalTransaction,
                                                    String processingCode, String posEntryMode, String posConditionCode, 
                                                    String de90) {
        Map<Integer, String> fields = requestContext.getFields();
        log.info("Processing Visa reversal request");
        
        try {
            // Visa-specific reversal logic
            // Validate reversal amount
            if (fields.containsKey(4)) {
                String reversalAmount = fields.get(4);
                String originalAmount = originalTransaction.getDe4AmountTransaction() != null ? 
                        originalTransaction.getDe4AmountTransaction().toString() : null;
                
                if (originalAmount != null && !reversalAmount.equals(originalAmount)) {
                    log.warn("Visa reversal amount mismatch");
                    responseAssembler.applyResponseCode(requestContext, InternalResCode.INVALID_AMOUNT);
                    return fields;
                }
            }
            
            // Create reversal record
            Transaction reversalTransaction = createReversalTransaction(originalTransaction, fields);
            Transaction savedReversal = transactionRepository.save(reversalTransaction);
            
            if (savedReversal != null) {
                log.info("Visa reversal transaction saved with ID: {}", savedReversal.getTransactionId());
                responseAssembler.applyResponseCode(requestContext, InternalResCode.APPROVED);
                applyReversalAuthCode(requestContext, originalTransaction);
            } else {
                responseAssembler.applyResponseCode(requestContext, InternalResCode.SYSTEM_ERROR);
            }
            
        } catch (Exception e) {
            log.error("Error processing Visa reversal", e);
            responseAssembler.applyResponseCode(requestContext, InternalResCode.SYSTEM_ERROR);
        }
        
        return fields;
    }

    /**
     * Creates a reversal transaction record based on the original transaction.
     *
     * @param originalTransaction The original transaction
     * @param fields The reversal request fields
     * @return New reversal transaction
     */
    private Transaction createReversalTransaction(Transaction originalTransaction, Map<Integer, String> fields) {
        Transaction reversalTransaction = new Transaction();
        
        // Copy original transaction data
        reversalTransaction.setDe2Pan(originalTransaction.getDe2Pan());
        reversalTransaction.setDe2PanCt(originalTransaction.getDe2PanCt());
        reversalTransaction.setDe2PanIv(originalTransaction.getDe2PanIv());
        reversalTransaction.setDe2PanTag(originalTransaction.getDe2PanTag());
        reversalTransaction.setDe2PanMasked(originalTransaction.getDe2PanMasked());
        reversalTransaction.setTxnChdKeyVersion(originalTransaction.getTxnChdKeyVersion());
        reversalTransaction.setInstitutionId(originalTransaction.getInstitutionId());
        reversalTransaction.setPanLookupHmac(originalTransaction.getPanLookupHmac());
        reversalTransaction.setDe3ProcessingCode(originalTransaction.getDe3ProcessingCode());
        reversalTransaction.setDe4AmountTransaction(originalTransaction.getDe4AmountTransaction());
        reversalTransaction.setDe35Track2(null);
        reversalTransaction.setDe45Track1(null);
        reversalTransaction.setDe7TransmissionDatetime(fields.get(7)); // Use reversal time
        reversalTransaction.setDe11Stan(fields.get(11)); // Use reversal STAN
        reversalTransaction.setDe32AcquiringInstitutionId(originalTransaction.getDe32AcquiringInstitutionId());
        reversalTransaction.setDe37Rrn(originalTransaction.getDe37Rrn());
        reversalTransaction.setDe38AuthApprovalCode(originalTransaction.getDe38AuthApprovalCode()); // Use correct method name
        reversalTransaction.setDe41TerminalId(originalTransaction.getDe41TerminalId());
        reversalTransaction.setDe42CardAcceptorId(originalTransaction.getDe42CardAcceptorId());
        reversalTransaction.setDe48AdditionalDataPrivate(fields.get(48));
        reversalTransaction.setDe49CurrencyCodeTransaction(originalTransaction.getDe49CurrencyCodeTransaction());
        reversalTransaction.setDe90OriginalDataElements(fields.get(90)); // Store DE90
        
        // Set reversal-specific fields
        reversalTransaction.setMti(fields.get(0));
        reversalTransaction.setStatus(Transaction.Status.FAILED); // Use enum value
        reversalTransaction.setDe52PinBlock(null);
        // Note: originalTransactionId field doesn't exist in the entity, so we'll store it in a comment or additional field
        
        reversalTransaction.setTransactionId(transactionIdGeneratorService.nextTransactionId());
        
        return reversalTransaction;
    }

    /**
     * Adds MAC to reversal response if MAC validation is enabled.
     *
     * @param fields The response fields
     */
    private void addMacToReversal(RequestContext requestContext) {
        try {
            Map<Integer, String> fields = requestContext.getFields();
            String de38 = fields.get(38);
            if (de38 == null) de38 = "";
            
            String inputString = fields.get(0) + fields.get(2) + fields.get(3) + fields.get(4) + 
                               fields.get(11) + fields.get(12) + fields.get(37) + de38 + fields.get(39);
            String mac = MacGeneration.generateMACvalue(getMacKey(), inputString);
            
            Map<String, String> tlvMap = new java.util.HashMap<>(requestContext.getParsedTlv(48));
            tlvMap.remove("099");
            tlvMap.put("099", mac);
            
            String tlvString = packTlv(tlvMap);
            fields.put(48, tlvString);
            
            log.info("MAC added to reversal response");
            RawDataLogging.value("Reversal response MAC", mac);
        } catch (Exception e) {
            log.error("Error adding MAC to reversal response", e);
        }
    }

    /**
     * Helper to get MAC key from properties or config.
     */
    private String getMacKey() {
        // Ideally inject via @Value or config class, fallback to default
        return System.getenv().getOrDefault("MAC_KEY", "3EBF5B2F98FD3886BA4F34857504A7C8");
    }

    /**
     * Packs a map of tag-value pairs into TLV string format.
     *
     * @param tlvMap Map of tag-value pairs
     * @return TLV formatted string
     */
    private String packTlv(Map<String, String> tlvMap) {
        StringBuilder sb = new StringBuilder();
        for (Map.Entry<String, String> entry : tlvMap.entrySet()) {
            String tag = entry.getKey();
            String value = entry.getValue();
            sb.append(String.format("%03d%03d%s", Integer.parseInt(tag), value.length(), value));
        }
        return sb.toString();
    }
