package com.logicpoint.iso8583.service;

import com.logicpoint.iso8583.hsm.FuturexHsmCmd;
import com.logicpoint.iso8583.hsm.SimpleNettyHsmClient;
import io.netty.channel.Channel;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

import org.springframework.beans.factory.annotation.Value;
import org.springframework.beans.factory.ObjectProvider;
import org.springframework.scheduling.annotation.Async;
import org.springframework.scheduling.annotation.Scheduled;
import org.springframework.stereotype.Service;

// Removed imports to avoid circular dependency

import jakarta.annotation.PostConstruct;
import jakarta.annotation.PreDestroy;

import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.ConcurrentLinkedQueue;
import java.util.concurrent.atomic.AtomicBoolean;
import java.util.concurrent.atomic.AtomicLong;
import java.util.concurrent.TimeUnit;


/**
 * Service responsible for maintaining HSM connection health through keep-alive mechanism.
 * Sends periodic echo commands to prevent connections from becoming stale.
 * 
 * @author LogicPoint Systems
 * @version 1.0
 */
@Service
public class HsmKeepAliveService implements HsmKeepAliveServiceInterface {

    private static final Logger logger = LoggerFactory.getLogger(HsmKeepAliveService.class);
    private static final String HEALTH_CHECK_COMMAND = "[AOECHO;BJ0;AGEcho Test connection.;]";
    
    private final ObjectProvider<SimpleNettyHsmClient> hsmClientProvider;
    
    // Keep-alive configuration
    @Value("${hsm.futurex.keep-alive-enabled:true}")
    private boolean keepAliveEnabled;
    
    @Value("${hsm.futurex.keep-alive-interval:30000}")
    private long keepAliveInterval; // milliseconds
    
    @Value("${hsm.futurex.keep-alive-idle-timeout:60000}")
    private long idleTimeout; // milliseconds
    
    @Value("${hsm.futurex.keep-alive-echo-timeout:5000}")
    private long echoTimeout; // milliseconds
    
    // Connection tracking
    private final ConcurrentHashMap<Channel, ConnectionHealth> connectionHealthMap = new ConcurrentHashMap<>();
    private final ConcurrentLinkedQueue<Channel> connectionsToCheck = new ConcurrentLinkedQueue<>();
    private final AtomicBoolean isRunning = new AtomicBoolean(false);
    private final AtomicLong lastActivityTime = new AtomicLong(System.currentTimeMillis());

    public HsmKeepAliveService(ObjectProvider<SimpleNettyHsmClient> hsmClientProvider) {
        this.hsmClientProvider = hsmClientProvider;
    }
    
    /**
     * Tracks the health status of a connection.
     */
    private static class ConnectionHealth {
        private final AtomicLong lastActivityTime = new AtomicLong(System.currentTimeMillis());
        private final AtomicLong lastEchoTime = new AtomicLong(0);
        private final AtomicBoolean isHealthy = new AtomicBoolean(true);
        private volatile int consecutiveFailures = 0;
        private volatile int maxConsecutiveFailures = 3;
        
        public long getLastActivityTime() {
            return lastActivityTime.get();
        }
        
        public void updateActivityTime() {
            lastActivityTime.set(System.currentTimeMillis());
        }
        
        public long getLastEchoTime() {
            return lastEchoTime.get();
        }
        
        public void updateEchoTime() {
            lastEchoTime.set(System.currentTimeMillis());
            consecutiveFailures = 0;
            isHealthy.set(true);
        }
        
        public boolean isHealthy() {
            return isHealthy.get();
        }
        
        public void markUnhealthy() {
            consecutiveFailures++;
            if (consecutiveFailures >= maxConsecutiveFailures) {
                isHealthy.set(false);
            }
        }
        
        public void markHealthy() {
            consecutiveFailures = 0;
            isHealthy.set(true);
        }
        
        public int getConsecutiveFailures() {
            return consecutiveFailures;
        }
        
        public boolean needsEchoCheck(long currentTime, long interval, long idleTimeout) {
            long timeSinceLastActivity = currentTime - getLastActivityTime();
            long timeSinceLastEcho = currentTime - getLastEchoTime();
            
            // Check if connection has been idle for too long or echo is overdue
            return (timeSinceLastActivity > idleTimeout && timeSinceLastEcho > interval) || 
                   (timeSinceLastEcho > interval && isHealthy());
        }
    }

    @PostConstruct
    public void initialize() {
        if (keepAliveEnabled) {
            logger.info("HSM Keep-alive service initialized with interval: {}ms, idle timeout: {}ms", 
                       keepAliveInterval, idleTimeout);
            isRunning.set(true);
        } else {
            logger.info("HSM Keep-alive service is disabled");
        }
    }

    @PreDestroy
    public void shutdown() {
        logger.info("Shutting down HSM Keep-alive service");
        isRunning.set(false);
        connectionHealthMap.clear();
        connectionsToCheck.clear();
    }

    /**
     * Scheduled task that runs periodically to check connection health.
     * This method is called by Spring's scheduler.
     */
    @Scheduled(fixedDelayString = "${hsm.futurex.keep-alive-interval:30000}")
    public void performKeepAliveCheck() {
        if (!keepAliveEnabled || !isRunning.get()) {
            return;
        }
        
        try {
            long currentTime = System.currentTimeMillis();
            updateActivityTime();
            
            // Get current pool statistics for monitoring (simplified for keep-alive)
            logger.debug("Keep-alive check - Performing health verification");
            
            // Perform health check on HSM
            performHsmHealthCheck();
            
        } catch (Exception e) {
            logger.error("Error during keep-alive check: {}", e.getMessage(), e);
        }
    }

    /**
     * Performs a health check on the HSM by sending an echo command.
     */
    @Async
    public void performHsmHealthCheck() {
        if (!isRunning.get()) {
            return;
        }
        
        try {
            logger.debug("Performing HSM health check with echo command");

            SimpleNettyHsmClient hsmClient = hsmClientProvider.getIfAvailable();
            if (hsmClient == null) {
                logger.warn("HSM health check could not be performed: HSM client is unavailable");
                updateAllConnectionHealth(false);
                return;
            }

            String response = hsmClient.sendMessage(HEALTH_CHECK_COMMAND);
            if (response == null || response.trim().isEmpty()) {
                logger.warn("HSM health check failed: no response received");
                updateAllConnectionHealth(false);
                return;
            }

            FuturexHsmCmd responseCmd = FuturexHsmCmd.parse(response);
            if (responseCmd.isSuccess()) {
                logger.debug("HSM health check successful: {}", responseCmd.getField("AG"));
                updateAllConnectionHealth(true);
            } else {
                logger.warn("HSM health check failed: {}", responseCmd.getErroMsg());
                updateAllConnectionHealth(false);
            }
            
        } catch (Exception e) {
            logger.error("HSM health check failed: {}", e.getMessage());
            updateAllConnectionHealth(false);
        }
    }

    /**
     * Updates the health status of all tracked connections.
     * 
     * @param isHealthy true if the health check was successful
     */
    private void updateAllConnectionHealth(boolean isHealthy) {
        long currentTime = System.currentTimeMillis();
        
        connectionHealthMap.forEach((channel, health) -> {
            if (isHealthy) {
                health.updateEchoTime();
                logger.debug("Connection {} marked as healthy", channel.remoteAddress());
            } else {
                health.markUnhealthy();
                logger.warn("Connection {} marked as unhealthy (failures: {})", 
                           channel.remoteAddress(), health.getConsecutiveFailures());
                
                // If connection is consistently unhealthy, consider removing it
                if (!health.isHealthy()) {
                    logger.warn("Connection {} is consistently unhealthy, may need replacement", 
                               channel.remoteAddress());
                }
            }
        });
    }

    /**
     * Updates the last activity time for the service.
     * This should be called whenever any HSM operation is performed.
     */
    public void updateActivityTime() {
        lastActivityTime.set(System.currentTimeMillis());
    }

    /**
     * Registers a connection for health monitoring.
     * 
     * @param channel The connection channel to monitor
     */
    public void registerConnection(Channel channel) {
        if (!keepAliveEnabled) {
            return;
        }
        
        ConnectionHealth health = new ConnectionHealth();
        connectionHealthMap.put(channel, health);
        logger.debug("Registered connection {} for health monitoring", channel.remoteAddress());
    }

    /**
     * Unregisters a connection from health monitoring.
     * 
     * @param channel The connection channel to stop monitoring
     */
    public void unregisterConnection(Channel channel) {
        ConnectionHealth health = connectionHealthMap.remove(channel);
        if (health != null) {
            logger.debug("Unregistered connection {} from health monitoring", channel.remoteAddress());
        }
    }

    /**
     * Updates the activity time for a specific connection.
     * This should be called whenever a connection is used.
     * 
     * @param channel The connection that was used
     */
    public void updateConnectionActivity(Channel channel) {
        ConnectionHealth health = connectionHealthMap.get(channel);
        if (health != null) {
            health.updateActivityTime();
        }
        updateActivityTime();
    }

    /**
     * Gets the health status of all connections.
     * 
     * @return A map of connection health status
     */
    public ConcurrentHashMap<Channel, ConnectionHealth> getConnectionHealthMap() {
        return new ConcurrentHashMap<>(connectionHealthMap);
    }

    /**
     * Gets the number of healthy connections.
     * 
     * @return Number of healthy connections
     */
    public int getHealthyConnectionCount() {
        return (int) connectionHealthMap.values().stream()
                .filter(ConnectionHealth::isHealthy)
                .count();
    }

    /**
     * Gets the number of unhealthy connections.
     * 
     * @return Number of unhealthy connections
     */
    public int getUnhealthyConnectionCount() {
        return (int) connectionHealthMap.values().stream()
                .filter(health -> !health.isHealthy())
                .count();
    }

    /**
     * Gets the total number of monitored connections.
     * 
     * @return Total number of monitored connections
     */
    public int getTotalConnectionCount() {
        return connectionHealthMap.size();
    }

    /**
     * Gets keep-alive statistics as a formatted string.
     * 
     * @return Keep-alive statistics
     */
    public String getKeepAliveStats() {
        int total = getTotalConnectionCount();
        int healthy = getHealthyConnectionCount();
        int unhealthy = getUnhealthyConnectionCount();
        
        return String.format("Keep-alive stats - Total: %d, Healthy: %d, Unhealthy: %d, Enabled: %s", 
                           total, healthy, unhealthy, keepAliveEnabled);
    }

    /**
     * Forces a health check on all connections.
     * This can be called manually for immediate health verification.
     */
    public void forceHealthCheck() {
        if (keepAliveEnabled && isRunning.get()) {
            logger.info("Forcing immediate HSM health check");
            performHsmHealthCheck();
        }
    }

    /**
     * Checks if the keep-alive service is running.
     * 
     * @return true if the service is running
     */
    public boolean isRunning() {
        return isRunning.get() && keepAliveEnabled;
    }

    /**
     * Gets the last activity time of the service.
     * 
     * @return Last activity time in milliseconds
     */
    public long getLastActivityTime() {
        return lastActivityTime.get();
    }

    /**
     * Gets the time since last activity.
     * 
     * @return Time since last activity in milliseconds
     */
    public long getTimeSinceLastActivity() {
        return System.currentTimeMillis() - lastActivityTime.get();
    }
}
