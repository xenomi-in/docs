package com.logicpoint.iso8583.service;

import com.logicpoint.iso8583.config.IssuerMandateRevocationProperties;
import com.logicpoint.iso8583.entity.StandingInstructionMandate;
import com.logicpoint.iso8583.model.Iso8583Message;
import com.logicpoint.iso8583.model.IssuerMandateRevocationRequest;
import java.math.BigDecimal;
import java.math.RoundingMode;
import java.time.LocalDate;
import java.time.LocalDateTime;
import java.time.format.DateTimeFormatter;
import java.util.LinkedHashMap;
import java.util.Map;
import java.util.UUID;
import java.util.concurrent.atomic.AtomicLong;
import org.springframework.stereotype.Component;

/** Builds a network-configured issuer-originated mandate revocation message. */
@Component
public class IssuerMandateRevocationMessageFactory {

    private static final DateTimeFormatter DE7_FORMAT = DateTimeFormatter.ofPattern("MMddHHmmss");
    private static final DateTimeFormatter SI_DATE_FORMAT = DateTimeFormatter.ofPattern("ddMMyyyy");
    private static final DateTimeFormatter RRN_PREFIX_FORMAT = DateTimeFormatter.ofPattern("yyDDDHH");
    private final AtomicLong traceSequence = new AtomicLong(System.currentTimeMillis() % 1_000_000L);
    private final IssuerMandateRevocationProperties properties;

    public IssuerMandateRevocationMessageFactory(IssuerMandateRevocationProperties properties) {
        this.properties = properties;
    }

    public Iso8583Message create(StandingInstructionMandate mandate,
                                 IssuerMandateRevocationRequest command) {
        String scheme = command.scheme().name();
        IssuerMandateRevocationProperties.SchemeRule rule = requireRule(scheme);
        if (!rule.getAllowedPosConditionCodes().contains(command.posConditionCode())) {
            throw new IllegalArgumentException("Unsupported POS condition code for " + scheme);
        }

        LocalDateTime now = LocalDateTime.now();
        String stan = String.format("%06d", Math.floorMod(traceSequence.incrementAndGet(), 1_000_000L));
        String rrn = now.format(RRN_PREFIX_FORMAT) + stan.substring(1);
        String commandId = UUID.randomUUID().toString();

        Iso8583Message message = new Iso8583Message();
        message.setScheme(command.scheme());
        message.setMti(rule.getRequestMti());
        message.setUniqueTransactionId(commandId);
        message.setField(7, now.format(DE7_FORMAT));
        message.setField(11, stan);
        message.setField(22, rule.getPosEntryMode());
        message.setField(25, command.posConditionCode());
        message.setField(37, rrn);
        putIfPresent(message, 41, mandate.getTerminalId());
        putIfPresent(message, 42, mandate.getCardAcceptorId());
        putIfPresent(message, 49, mandate.getCurrencyCode());

        Map<String, String> tags = new LinkedHashMap<>();
        tags.put(rule.getEciTag(), rule.getEciValue());
        tags.put(rule.getSiDataTag(), buildSiData(mandate));
        tags.put(rule.getMandateIdTag(), mandate.getMandateId());
        message.setField(rule.getPrivateDataField(), packTlv(tags));
        message.setMetadata(Map.of(
                "source", "CMS_CRM",
                "requested_by", command.requestedBy(),
                "reason", command.reason() == null ? "" : command.reason()));
        return message;
    }

    private IssuerMandateRevocationProperties.SchemeRule requireRule(String scheme) {
        IssuerMandateRevocationProperties.SchemeRule rule = properties.getRule(scheme);
        if (!properties.isEnabled() || rule == null || !rule.isEnabled()) {
            throw new IllegalStateException("Issuer mandate revocation is not enabled for " + scheme);
        }
        return rule;
    }

    private String buildSiData(StandingInstructionMandate mandate) {
        String value = "02"
                + fixedDigits(mandate.getNumberOfInstallments(), 3, "000")
                + fixedDigits(mandate.getPreferredDay(), 2, "01")
                + amount(mandate.getMinAmount())
                + amount(mandate.getMaxAmount())
                + date(mandate.getStartDate())
                + date(mandate.getEndDate())
                + fixedValue(mandate.getNotificationMode(), "0")
                + fixedValue(mandate.getFrequencyIndicator(), "6");
        if (value.length() != 49) {
            throw new IllegalStateException("Generated DE48 tag 085 must be 49 characters");
        }
        return value;
    }

    private String amount(BigDecimal value) {
        if (value == null) {
            return "000000000000";
        }
        long minorUnits = value.setScale(2, RoundingMode.UNNECESSARY).movePointRight(2).longValueExact();
        if (minorUnits < 0 || minorUnits > 999_999_999_999L) {
            throw new IllegalArgumentException("Mandate amount cannot be represented in tag 085");
        }
        return String.format("%012d", minorUnits);
    }

    private String date(LocalDate value) {
        return value == null ? "00000000" : value.format(SI_DATE_FORMAT);
    }

    private String fixedDigits(String value, int length, String defaultValue) {
        String resolved = value == null || value.isBlank() ? defaultValue : value;
        if (!resolved.matches("\\d{" + length + "}")) {
            throw new IllegalArgumentException("Invalid persisted mandate fixed-position value: " + resolved);
        }
        return resolved;
    }

    private String fixedValue(String value, String defaultValue) {
        String resolved = value == null || value.isBlank() ? defaultValue : value;
        if (resolved.length() != 1) {
            throw new IllegalArgumentException("Invalid persisted mandate indicator: " + resolved);
        }
        return resolved;
    }

    private String packTlv(Map<String, String> tags) {
        StringBuilder packed = new StringBuilder();
        tags.forEach((tag, value) -> packed.append(tag)
                .append(String.format("%03d", value.length()))
                .append(value));
        return packed.toString();
    }

    private void putIfPresent(Iso8583Message message, int field, String value) {
        if (value != null && !value.isBlank()) {
            message.setField(field, value);
        }
    }
}
