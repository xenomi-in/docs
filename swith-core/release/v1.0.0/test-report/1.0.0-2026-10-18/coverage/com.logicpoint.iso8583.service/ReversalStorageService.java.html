package com.logicpoint.iso8583.service;

import tools.jackson.databind.json.JsonMapper;
import com.logicpoint.iso8583.entity.ReversalRequest;
import com.logicpoint.iso8583.repository.ReversalRequestRepository;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;

import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.dao.DataAccessException;
import org.springframework.scheduling.annotation.Async;
import org.springframework.stereotype.Service;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import jakarta.annotation.PostConstruct;
import jakarta.annotation.PreDestroy;

import java.io.*;
import java.nio.ByteBuffer;
import java.nio.channels.AsynchronousFileChannel;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.Paths;
import java.nio.file.StandardOpenOption;
import java.time.LocalDateTime;
import java.time.format.DateTimeFormatter;
import java.util.concurrent.CompletableFuture;
import java.util.concurrent.ConcurrentLinkedQueue;
import java.util.concurrent.Executors;
import java.util.concurrent.ScheduledExecutorService;
import java.util.concurrent.TimeUnit;
import java.util.concurrent.locks.ReentrantLock;

@Slf4j
@Service

public class ReversalStorageService {

    private static final Logger log = LoggerFactory.getLogger(ReversalStorageService.class);
    private final ReversalRequestRepository reversalRequestRepository;
    private final JsonMapper objectMapper;
    private final ReentrantLock fileLock = new ReentrantLock();

    @Autowired
    public ReversalStorageService(ReversalRequestRepository reversalRequestRepository, JsonMapper objectMapper) {
        this.reversalRequestRepository = reversalRequestRepository;
        this.objectMapper = objectMapper;
    }
    // Buffered file writes for better performance
    private final ConcurrentLinkedQueue<ReversalRequest> fileWriteQueue = new ConcurrentLinkedQueue<>();
    private final ScheduledExecutorService fileWriterExecutor = Executors.newSingleThreadScheduledExecutor(
            r -> {
                Thread t = new Thread(r, "reversal-file-writer");
                t.setDaemon(true);
                return t;
            });

    @Value("${reversal.file-storage.path:./reversals/pending}")
    private String fileStoragePath;

    @Value("${reversal.file-storage.enabled:true}")
    private boolean fileStorageEnabled;

    @Value("${reversal.file-storage.batch-size:10}")
    private int batchSize;

    @Value("${reversal.file-storage.flush-interval:5000}")
    private long flushIntervalMs;

    // Initialize batch writer
    // {
    //     fileWriterExecutor.scheduleWithFixedDelay(
    //         this::flushFileQueue, 
    //         flushIntervalMs, 
    //         flushIntervalMs, 
    //         TimeUnit.MILLISECONDS
    //     );
    // }
    @PostConstruct
    void startFileWriter() {
        long interval = (flushIntervalMs > 0) ? flushIntervalMs : 5000L;
        if (flushIntervalMs <= 0) {
            log.warn("Invalid reversal.file-storage.flush-interval={}ms. Falling back to 5000ms", flushIntervalMs);
        }

        fileWriterExecutor.scheduleWithFixedDelay(
                this::flushFileQueue,
                interval,
                interval,
                TimeUnit.MILLISECONDS
        );
    }

    @PreDestroy
    void stopFileWriter() {
        fileWriterExecutor.shutdown();
    }

    /**
     * Fast path: Try DB first (synchronous for immediate feedback) Slow path:
     * Queue for file write if DB fails (async, non-blocking)
     */
    public ReversalRequest saveReversalRequest(ReversalRequest request) {
        // Fast path: Try database first (most common case)
        ReversalRequest saved = trySaveToDatabase(request);
        if (saved != null) {
            return saved;
        }

        // Fallback: Queue for async file write (non-blocking)
        if (fileStorageEnabled) {
            fileWriteQueue.offer(request);
            log.warn("Reversal queued for file storage (DB unavailable): {}", request.getTransactionId());
            // Return immediately - file write happens async
            return request;
        }

        log.error("CRITICAL: Failed to save reversal - both DB and file storage unavailable");
        return request;
    }

    /**
     * Optimized database save with connection pool
     */
    private ReversalRequest trySaveToDatabase(ReversalRequest request) {
        try {
            // Use saveAndFlush for immediate persistence
            return reversalRequestRepository.saveAndFlush(request);
            //return null;
        } catch (DataAccessException e) {
            // Log only at debug level to avoid log spam
            if (log.isDebugEnabled()) {
                log.debug("Database save failed: {}", e.getMessage());
            }
            return null;
        } catch (Exception e) {
            log.error("Unexpected error saving to database", e);
            return null;
        }
    }

    /**
     * Batch file writes for better I/O performance
     */
    private void flushFileQueue() {
        if (fileWriteQueue.isEmpty()) {
            return;
        }

        fileLock.lock();
        try {
            Path dir = Paths.get(fileStoragePath);
            if (!Files.exists(dir)) {
                Files.createDirectories(dir);
            }

            // Process batch
            int processed = 0;
            while (!fileWriteQueue.isEmpty() && processed < batchSize) {
                ReversalRequest request = fileWriteQueue.poll();
                if (request != null) {
                    writeToFile(request, dir);
                    processed++;
                }
            }

            if (processed > 0) {
                log.debug("Flushed {} reversal requests to file", processed);
            }

        } catch (Exception e) {
            log.error("Error flushing file queue", e);
        } finally {
            fileLock.unlock();
        }
    }

    /**
     * Optimized file write using NIO
     */
    private void writeToFile(ReversalRequest request, Path dir) {
        try {
            String filename = String.format("reversal_%s_%s.json",
                    request.getTransactionId(),
                    LocalDateTime.now().format(DateTimeFormatter.ofPattern("yyyyMMddHHmmss")));

            Path filePath = dir.resolve(filename);

            // Use async file channel for non-blocking I/O
            String json = objectMapper.writeValueAsString(request);
            byte[] bytes = json.getBytes();

            // Write synchronously but in separate thread pool
            Files.write(filePath, bytes,
                    StandardOpenOption.CREATE,
                    StandardOpenOption.WRITE);

            if (request.getId() == null) {
                request.setId(-1L);
            }

        } catch (Exception e) {
            log.error("Failed to write reversal to file", e);
            // Re-queue for retry
            fileWriteQueue.offer(request);
        }
    }

    /**
     * Load pending reversals from files (called on startup/recovery)
     */
    @Async
    public CompletableFuture<Void> loadPendingReversalsFromFiles() {
        if (!fileStorageEnabled) {
            return CompletableFuture.completedFuture(null);
        }

        return CompletableFuture.runAsync(() -> {
            fileLock.lock();
            try {
                Path dir = Paths.get(fileStoragePath);
                if (!Files.exists(dir)) {
                    return;
                }

                Files.list(dir)
                        .filter(path -> path.toString().endsWith(".json"))
                        .limit(100) // Process in batches
                        .forEach(filePath -> {
                            try {
                                String json = new String(Files.readAllBytes(filePath));
                                ReversalRequest request = objectMapper.readValue(json, ReversalRequest.class);

                                if (trySaveToDatabase(request) != null) {
                                    Files.delete(filePath);
                                    log.info("Migrated reversal from file to DB: {}", filePath);
                                }
                            } catch (Exception e) {
                                log.error("Error loading reversal from file {}", filePath, e);
                            }
                        });

            } catch (Exception e) {
                log.error("Error loading pending reversals", e);
            } finally {
                fileLock.unlock();
            }
        });
    }
}
