package com.logicpoint.iso8583.service;

import io.micrometer.core.instrument.MeterRegistry;
import io.micrometer.core.instrument.Timer;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Component;

@Component
@RequiredArgsConstructor
public class ReversalMetrics {
    
    private final MeterRegistry meterRegistry;
    
    public void recordContextRegistration() {
        meterRegistry.counter("reversal.context.registered").increment();
    }
    
    public void recordReversalTriggered() {
        meterRegistry.counter("reversal.triggered").increment();
    }
    
    public Timer.Sample startReversalExecution() {
        return Timer.start(meterRegistry);
    }
    
    public void recordReversalExecution(Timer.Sample sample, boolean success) {
        sample.stop(meterRegistry.timer("reversal.execution.time", 
            "success", String.valueOf(success)));
    }
    
    public void recordFileStorageFallback() {
        meterRegistry.counter("reversal.storage.file.fallback").increment();
    }
}
