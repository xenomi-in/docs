    /**
     * Processes an ISO 8583 message based on its type and scheme.
     *
     * @param fields Map of ISO 8583 field numbers to their values
     * @param scheme The payment scheme (e.g., RUPAY, VISA)
     * @param mti Message Type Indicator
     * @return Processed message fields with response codes
     */
    public Map<Integer, String> processMessage(RequestContext requestContext) {
        try {
            Map<Integer, String> fields = requestContext.getFields();
            Scheme scheme = requestContext.getScheme();
            String mti = requestContext.getMti();
            String uniqueTransactionId = requestContext.getUniqueTransactionId();
            MessageType messageType = MessageType.fromMti(mti);
            log.info("Processing message type: {}", messageType);
            identifyTransactionTypeForPersistence(requestContext, messageType);
            requestContext.setChannel(resolveTransactionChannel(requestContext));
            // Save transaction request details
            String transactionId = transactionIdGeneratorService.nextTransactionId();
            Transaction transaction = mapTransaction(requestContext, transactionId);
            transaction.setMti(mti);
            log.info("Transaction ID: {}", transaction.getTransactionId());
            if (duplicateCheckService.isDuplicate(requestContext)) {
                return fields;
            }
            // Save both entities
            Transaction savedTxn = transactionRepository.save(transaction);
            if (savedTxn == null) {
                log.error("Failed to save transaction");
                return fields;
            }
            log.info("Saved transaction reg with ID: {}", savedTxn.getTransactionId());
            switch (messageType) {
                case FINANCIAL_REQUEST:
                    Map<Integer, String> respFields = null;
                    if (scheme == Scheme.RUPAY && macValidationEnabled && ("0100".equals(mti) || "0200".equals(mti))) {
                        log.info("MAC validation enabled for Scheme: {} and MTI: {}", scheme, mti);
                        boolean macValid = validateMAC(requestContext);
                        if (!macValid) {
                            log.warn("MAC validation failed for Scheme: {} and MTI: {}", scheme, mti);
                            responseAssembler.setResponseMti(requestContext);
                            responseAssembler.applyResponseCode(requestContext, InternalResCode.MAC_FAIL);
                            String tlvString = genMacAndUpdateTlv(requestContext, fields.get(0) + fields.get(2) + fields.get(3) + fields.get(4) + fields.get(11) + fields.get(12) + fields.get(37) + (fields.get(38) == null ? "" : fields.get(38)) + fields.get(39));
                            fields.put(48, tlvString);
                            return fields;
                        }
                        log.info("MAC validation successful for Scheme: {} and MTI: {}", scheme, mti);
                        respFields = financialRequestProcessor.process(requestContext);
                    } else {
                        log.info("MAC validation disabled for Scheme: {} and MTI: {}", scheme, mti);
                        respFields = financialRequestProcessor.process(requestContext);
                    }
                    if (scheme == Scheme.RUPAY && macValidationEnabled && ("0100".equals(mti) || "0200".equals(mti))) {
                        String tlvString = genMacAndUpdateTlv(requestContext, fields.get(0) + fields.get(2) + fields.get(3) + fields.get(4) + fields.get(11) + fields.get(12) + fields.get(37) +( fields.get(38) == null ? "" : fields.get(38)) + fields.get(39));
                        respFields.put(48, tlvString);
                    }
                    persistApprovalData(savedTxn, respFields);
                    return respFields;
                case REVERSAL_REQUEST:
                    requestContext.setTransactionType(TransactionType.REVERSAL);
                    return processReversalRequest(requestContext);
                case NETWORK_MANAGEMENT_REQUEST:
                    requestContext.setTransactionType(TransactionType.UNKNOWN);
                    return processNetworkRequest(requestContext);
                case ACTION_RESPONSE:
                    issuerMandateRevocationService.reconcile(scheme, toIso8583Message(requestContext));
                    fields.put(0, mti);
                    return fields;
                default:
                    log.warn("Unsupported message type: {}", messageType);
                    return fields;
            }
        } catch (Exception e) {
            log.error("Error processing message", e);
            throw new RuntimeException("Failed to process message", e);
        }
    }

    private void persistApprovalData(Transaction transaction, Map<Integer, String> fields) {
        if (transaction == null || fields == null || !"00".equals(fields.get(39))) {
            return;
        }

        String authCode = fields.get(38);
        if (authCode == null || authCode.isBlank()) {
            return;
        }

        transaction.setDe38AuthApprovalCode(authCode);
        transaction.setStatus(Transaction.Status.SUCCESS);
        transactionRepository.save(transaction);
    }

    private Iso8583Message toIso8583Message(RequestContext requestContext) {
        Iso8583Message message = new Iso8583Message();
        message.setScheme(requestContext.getScheme());
        message.setMti(requestContext.getMti());
        message.setUniqueTransactionId(requestContext.getUniqueTransactionId());
        message.setFields(new HashMap<>(requestContext.getFields()));
        return message;
    }

    private String genMacAndUpdateTlv(RequestContext requestContext, String inputStr) {
        RawDataLogging.value("Response MAC input", inputStr);
        RawDataLogging.value("Response DE48", requestContext.getFields().get(48));
        //String genMac = nettyHsmService.generateMac(inputStr, "RUPAY");
        String mac = nettyHsmService.generateMac(inputStr, "RUPAY");//MacGeneration.generateMACvalue(getMacKey(), inputStr);
        //String mac = "E62C83E14A0CA802"; // to simulate 
        if(macProcess.equalsIgnoreCase("invalid")) {
            String shuffledMac = "";
            while (mac.length() != 0)
            {
                int index = (int) Math.floor(Math.random() * mac.length());
                char c = mac.charAt(index);
                mac = mac.substring(0,index)+mac.substring(index+1);
                shuffledMac += c;
            }
            mac = shuffledMac;
        }
        Map<String, String> tlvMap = new HashMap<>(requestContext.getParsedTlv(48));
        tlvMap.remove("099");
       if(!macProcess.equalsIgnoreCase("omit")) {
        tlvMap.put("099", mac);
       }
        if(tlvMap.containsKey("058"))  tlvMap.remove("058"); // remove 058 tag from the tlv map in response
        if(!tlvMap.containsKey("054"))  tlvMap.put("054", "N");  // add 054 tag value with 'N' for mac failed scenario
        RawDataLogging.value("Response MAC input string", inputStr);
        RawDataLogging.value("Response TLV map after MAC update", tlvMap);
        String tlvString = packTlv(tlvMap);
        RawDataLogging.value("Response MAC", mac);
        RawDataLogging.value("Response TLV string", tlvString);
        return packTlv(tlvMap);

    }

    private String getMacKey() {
        // Ideally inject via @Value or config class, fallback to default
        return System.getenv().getOrDefault("MAC_KEY", "3EBF5B2F98FD3886BA4F34857504A7C8");
    }

    /**
     * Packs a map of tag-value pairs into TLV string format.
     *
     * @param tlvMap Map of tag-value pairs
     * @return TLV formatted string
     */
    private String packTlv(Map<String, String> tlvMap) {
        StringBuilder sb = new StringBuilder();
        for (Map.Entry<String, String> entry : tlvMap.entrySet()) {
            String tag = entry.getKey();
            String value = entry.getValue();
            sb.append(String.format("%03d%03d%s", Integer.parseInt(tag), value.length(), value));
        }
        return sb.toString();
    }

    /**
     * Maps ISO 8583 fields to a Transaction entity.
     *
     * @param requestContext request-scoped transaction context
     * @return Mapped Transaction entity
     */
    private Transaction mapTransaction(RequestContext requestContext, String transactionId) {
        Map<Integer, String> fields = requestContext.getFields();
        Transaction transaction = new Transaction();

        try {
            transaction.setTransactionId(transactionId);
            if (requestContext.getInstitutionId() != null && !requestContext.getInstitutionId().isBlank()) {
                try {
                    transaction.setInstitutionId(Integer.valueOf(requestContext.getInstitutionId()));
                } catch (NumberFormatException ex) {
                    log.warn("Unable to parse institution id {} for transaction {}", requestContext.getInstitutionId(), requestContext.getUniqueTransactionId());
                }
            }
            if (fields.containsKey(2)) {
                String pan = fields.get(2);
                transaction.setDe2Pan(pan);
                // Masked PAN remains the same
                String maskedPan = pan.substring(0, 6) + "******" + pan.substring(pan.length() - 4);
                transaction.setDe2PanMasked(maskedPan);
            }

            if (fields.containsKey(14)) {
                // Intentionally not persisted in transaction history; PAN is the only encrypted CHD field retained.
            }

            if (fields.containsKey(35)) {
                // Intentionally not persisted in transaction history.
            }

            if (fields.containsKey(45)) {
                // Intentionally not persisted in transaction history.
            }

            if (fields.containsKey(3)) {
                transaction.setDe3ProcessingCode(fields.get(3));
            }

            if (fields.containsKey(4)) {
                transaction.setDe4AmountTransaction(new BigDecimal(Long.parseLong(fields.get(4)) / 100.0));
            }

            if (fields.containsKey(7)) {
                // Parse transmission datetime from field 7 (MMDDhhmmss)
                String f7 = fields.get(7);
                transaction.setDe7TransmissionDatetime(f7); // Set actual parsed datetime
            }

            if (fields.containsKey(11)) {
                transaction.setDe11Stan(fields.get(11));
            }

            if (fields.containsKey(12)) {
                transaction.setDe12LocalTime(fields.get(12));
            }

            if (fields.containsKey(13)) {
                transaction.setDe13LocalDate(fields.get(13));
            }
            
            if (fields.containsKey(18)) {
                transaction.setDe18MerchantType(fields.get(18));
            }

            if (fields.containsKey(19)) {
                transaction.setDe19AcquiringCountryCode(fields.get(19));
            }

            if (fields.containsKey(22)) {
                transaction.setDe22PosEntryMode(fields.get(22));
            }

            if (fields.containsKey(25)) {
                transaction.setDe25PosConditionCode(fields.get(25));
            }

            if (fields.containsKey(32)) {
                transaction.setDe32AcquiringInstitutionId(fields.get(32));
            }

            if (fields.containsKey(37)) {
                transaction.setDe37Rrn(fields.get(37));
            }

            if (fields.containsKey(41)) {
                transaction.setDe41TerminalId(fields.get(41));
            }

            if (fields.containsKey(42)) {
                transaction.setDe42CardAcceptorId(fields.get(42));
            }

            if (fields.containsKey(48)) {
                transaction.setDe48AdditionalDataPrivate(fields.get(48));
            }

            if (fields.containsKey(49)) {
                transaction.setDe49CurrencyCodeTransaction(fields.get(49));
            }
            if (fields.containsKey(54)) {
                transaction.setDe54AdditionalAmounts(fields.get(54));
            }
            if (fields.containsKey(60)) {
                transaction.setDe60TerminalData(fields.get(60));
            }
            if (fields.containsKey(61)) {
                transaction.setDe61PosData(fields.get(61));
            }
            if (fields.containsKey(90)) {
                transaction.setDe90OriginalDataElements(fields.get(90));
            }

            transaction.setCreatedAt(LocalDateTime.now());
            transaction.setStatus(Transaction.Status.PENDING);
            transaction.setTransactionType(requestContext.getTransactionType());
            transaction.setSourceChannel(requestContext.getChannel().name());
            transaction.setPanLookupHmac(null);
            transaction.setDe52PinBlock(null);
            transactionChdEncryptionService.applyOnWrite(transaction, requestContext);

        } catch (Exception e) {
            log.error("Encryption failed", e);
            throw new RuntimeException("Failed to encrypt sensitive data");
        }

        return transaction;
    }

    private void identifyTransactionTypeForPersistence(
            RequestContext requestContext,
            MessageType messageType) {
        try {
            TransactionType transactionType = switch (messageType) {
                case FINANCIAL_REQUEST -> financialRequestProcessor.identifyTransactionType(requestContext);
                case REVERSAL_REQUEST -> TransactionType.REVERSAL;
                default -> TransactionType.UNKNOWN;
            };
            requestContext.setTransactionType(transactionType);
        } catch (Exception ex) {
            requestContext.setTransactionType(TransactionType.UNKNOWN);
            log.warn("Transaction type identification failed before persistence; storing as UNKNOWN: {}",
                    ex.getMessage());
        }
    }

    private TransactionChannel resolveTransactionChannel(RequestContext requestContext) {
        TransactionType type = requestContext.getTransactionType();
        if (type != null && ATM_TRANSACTION_TYPES.contains(type)) {
            return TransactionChannel.ATM;
        }
        if (type != null && ECOMMERCE_TRANSACTION_TYPES.contains(type)) {
            return TransactionChannel.ECOMMERCE;
        }

        String entryMode = requestContext.getFields().get(22);
        String conditionCode = requestContext.getFields().get(25);
        String processingCode = requestContext.getFields().get(3);
        if (startsWithAny(entryMode, "001", "011", "061", "810", "811")
                || "59".equals(conditionCode) || "08".equals(conditionCode)
                || "06".equals(conditionCode) || "07".equals(conditionCode)) {
            return TransactionChannel.ECOMMERCE;
        }
        if (startsWithAny(processingCode, "01", "30", "31", "38", "40", "91")
                || startsWithAny(entryMode, "010", "020")
                || "51".equals(conditionCode) || "52".equals(conditionCode)) {
            return TransactionChannel.ATM;
        }
        return TransactionChannel.POS;
    }

    private boolean startsWithAny(String value, String... prefixes) {
        if (value == null || value.isBlank()) {
            return false;
        }
        for (String prefix : prefixes) {
            if (value.startsWith(prefix)) {
                return true;
            }
        }
        return false;
    }

    /**
     * Processes a reversal request.
     *
     * @param fields Map of ISO 8583 field numbers to their values
     * @param scheme The payment scheme
     * @return Processed message fields with response codes
     */
    private Map<Integer, String> processReversalRequest(RequestContext requestContext) {
        Map<Integer, String> fields = requestContext.getFields();

        // Call the reversalProcessService to process the transaction
        if (reversalProcessService != null) {
            try {
                Map<Integer, String> processedFields = reversalProcessService.processReversalRequest(requestContext); //processReversal(fields, scheme);
                if (processedFields != null) {
                    fields.putAll(processedFields);
                    responseAssembler.setResponseMti(requestContext);
                    log.info("Reversal processed successfully by reversalProcessService.");
                } else {
                    log.warn("reversalProcessService returned null for reversal processing.");
                }
            } catch (Exception ex) {
                log.error("Error processing reversal via reversalProcessService", ex);
                responseAssembler.applyResponseCode(requestContext, InternalResCode.SYSTEM_ERROR);
                return fields;
            }
        } else {
            log.warn("reversalProcessService bean is not available. Skipping external reversal processing.");
            responseAssembler.applyResponseCode(requestContext, InternalResCode.SYSTEM_ERROR);
            return fields;
        }
        return fields;
    }

    /**
     * Validates MAC for the transaction.
     *
     * @param fields Map of ISO 8583 field numbers to their values
     * @param mti Message Type Indicator
     * @return true if MAC is valid, false otherwise
     */
    private boolean validateMAC(RequestContext requestContext) {
        Map<Integer, String> fields = requestContext.getFields();
        String mti = requestContext.getMti();
        // get the MAC from the fields
        String key = getMacKey();
        String inputString = mti + fields.get(2) + fields.get(3) + fields.get(4) + fields.get(11) + fields.get(12) + fields.get(37);
        Map<String, String> tlvMap = requestContext.getParsedTlv(48);

        String reqMac = tlvMap.get("099");
        if (reqMac != null) {
            RawDataLogging.value("MAC key", key);
            RawDataLogging.value("Request MAC input", inputString);
            RawDataLogging.value("Request MAC", reqMac);
            boolean macValid = nettyHsmService.verifyMac(inputString, reqMac, "RUPAY");
            log.debug("MAC validation result: {}", macValid);
            return macValid;
           /* String genMac = MacGeneration.generateMACvalue(key, inputString);
           
            RawDataLogging.value("Generated MAC", genMac);
            if (genMac != null && genMac.equals(reqMac)) {
                return true;
            } else {
                return false;
            } */
        } else {
            return false;
        }
    }

    /**
     * Processes a network management request.
     *
     * @param fields Map of ISO 8583 field numbers to their values
     * @return Processed message fields with response codes
     */
    private Map<Integer, String> processNetworkRequest(RequestContext requestContext) {
        log.info("Processing network management request");
        responseAssembler.setResponseMti(requestContext);
        responseAssembler.applyResponseCode(requestContext, InternalResCode.APPROVED);
        return requestContext.getFields();
    }

    /**
     * Builds a composite key for transaction identification.
     *
     * @param processingCode Processing code
     * @param entryMode Entry mode
     * @param posConditionCode POS condition code
     * @return Composite key string
     */
    protected String buildCompositeKey(String processingCode, String entryMode, String posConditionCode) {
        return String.format("%s|%s|%s",
                Objects.requireNonNull(processingCode, "Processing code cannot be null"),
                Objects.requireNonNull(entryMode, "Entry mode cannot be null"),
                Objects.requireNonNull(posConditionCode, "POS condition code cannot be null")
        );
    }
