package com.logicpoint.iso8583.service;

import com.logicpoint.iso8583.config.ThreeDsAuthorizationProperties;
import com.logicpoint.iso8583.context.RequestContext;
import com.logicpoint.iso8583.util.Scheme;
import com.logicpoint.iso8583.validation.TransactionType;
import java.util.ArrayList;
import java.util.Base64;
import java.util.List;
import java.util.Map;
import java.util.Set;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.stereotype.Service;

/**
 * Validates network-configured 3DS authorization indicators carried in ISO8583 fields.
 */
@Service
public class ThreeDsAuthorizationValidationService {

    private static final Logger log = LoggerFactory.getLogger(ThreeDsAuthorizationValidationService.class);

    private final ThreeDsAuthorizationProperties properties;

    public ThreeDsAuthorizationValidationService(ThreeDsAuthorizationProperties properties) {
        this.properties = properties;
    }

    public List<String> validate(RequestContext requestContext, Scheme scheme, TransactionType transactionType) {
        List<String> errors = new ArrayList<>();
        if (!properties.isEnabled() || requestContext == null || scheme == null || transactionType == null) {
            return errors;
        }

        ThreeDsAuthorizationProperties.SchemeRule rule = properties.getRule(scheme.name());
        if (rule == null || !rule.isEnabled() || !rule.getTransactionTypes().contains(transactionType)) {
            return errors;
        }

        Map<String, String> de48Tags;
        try {
            de48Tags = requestContext.getParsedTlv(48);
        } catch (Exception ex) {
            errors.add("INVALID_3DS_DE48_FORMAT: Unable to parse DE48 for 3DS authorization data");
            return errors;
        }

        String eci = normalize(de48Tags.get(rule.getEciTag()));
        if (eci == null) {
            errors.add("MISSING_3DS_ECI: DE48 tag " + rule.getEciTag() + " is required for " + scheme + " " + transactionType);
            return errors;
        }

        if (!rule.getAllowedEciValues().isEmpty() && !rule.getAllowedEciValues().contains(eci)) {
            errors.add("INVALID_3DS_ECI: DE48 tag " + rule.getEciTag() + " value " + eci
                    + " is not configured for " + scheme + " " + transactionType);
        }

        String cavv = de48Tags.get(rule.getCavvTag());
        if (!rule.isOnBehalfOf()) {
            if (rule.getCavvRequiredEciValues().contains(eci) && isBlank(cavv)) {
                errors.add("MISSING_3DS_CAVV: DE48 tag " + rule.getCavvTag() + " is required for ECI " + eci);
                return errors;
            }

            if (!isBlank(cavv)) {
                String authResultCode = extractAuthenticationResultCode(cavv);
                if (authResultCode == null) {
                    errors.add("INVALID_3DS_CAVV_FORMAT: DE48 tag " + rule.getCavvTag()
                            + " is not a valid Base64 encoded authentication value");
                } else {
                    Set<String> allowedAuthResults = rule.getCavvAuthResultByEci().get(eci);
                    if (allowedAuthResults != null && !allowedAuthResults.isEmpty()
                            && !allowedAuthResults.contains(authResultCode)) {
                        errors.add("INVALID_3DS_CAVV_ECI_MAPPING: CAVV authentication result "
                                + authResultCode + " is not valid for ECI " + eci);
                    }
                }
            }
        }

        validateOptionalTransactionIdTags(de48Tags, rule.getDsTransactionIdTags(), "DS");
        validateOptionalTransactionIdTags(de48Tags, rule.getAcsTransactionIdTags(), "ACS");
        return errors;
    }

    private void validateOptionalTransactionIdTags(Map<String, String> de48Tags, List<String> tags, String owner) {
        for (String tag : tags) {
            String value = de48Tags.get(tag);
            if (!isBlank(value)) {
                log.debug("{} transaction id found in DE48 tag {}", owner, tag);
                return;
            }
        }
    }

    private String extractAuthenticationResultCode(String cavv) {
        try {
            byte[] decoded = Base64.getDecoder().decode(cavv);
            if (decoded.length == 0) {
                return null;
            }

            String printable = new String(decoded);
            String trimmed = printable.trim();
            if (trimmed.matches("\\d{2}.*")) {
                return trimmed.substring(0, 2);
            }

            StringBuilder bcd = new StringBuilder(decoded.length * 2);
            for (byte value : decoded) {
                bcd.append(Character.forDigit((value >> 4) & 0x0F, 16));
                bcd.append(Character.forDigit(value & 0x0F, 16));
            }
            String numeric = bcd.toString().toUpperCase();
            return numeric.matches("\\d{2}.*") ? numeric.substring(0, 2) : null;
        } catch (IllegalArgumentException ex) {
            return null;
        }
    }

    private String normalize(String value) {
        return isBlank(value) ? null : value.trim().toUpperCase();
    }

    private boolean isBlank(String value) {
        return value == null || value.trim().isEmpty();
    }
}
