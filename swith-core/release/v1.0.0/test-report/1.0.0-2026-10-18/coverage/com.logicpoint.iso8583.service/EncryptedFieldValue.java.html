package com.logicpoint.iso8583.service;

public class EncryptedFieldValue {
    private final byte[] ciphertext;
    private final byte[] iv;
    private final byte[] tag;
    private final Integer keyVersion;

    public EncryptedFieldValue(byte[] ciphertext, byte[] iv, byte[] tag, Integer keyVersion) {
        this.ciphertext = ciphertext;
        this.iv = iv;
        this.tag = tag;
        this.keyVersion = keyVersion;
    }

    public byte[] getCiphertext() {
        return ciphertext;
    }

    public byte[] getIv() {
        return iv;
    }

    public byte[] getTag() {
        return tag;
    }

    public Integer getKeyVersion() {
        return keyVersion;
    }
}
