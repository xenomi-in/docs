@Async("iso8583TxnExecutor")
public CompletableFuture<Iso8583Message> processMessage(Iso8583Message request) {
    String uniqueTransactionId = request.getUniqueTransactionId();
    RequestContext requestContext = buildRequestContext(request, uniqueTransactionId);
    MDC.put("txnId", uniqueTransactionId);
    applyNetworkSessionContext(request);
    transactionOutcomeMetrics.started();

    try {
        // Register transaction for reversal tracking
        automaticReversalService.registerTransaction(requestContext);

        // Process the message
        Map<Integer, String> processedFields = messageProcessor.processMessage(requestContext);
        
        // Check if accounting was approved (stored in context)
        String responseCode = processedFields.get(39);
        boolean accountingApproved = "00".equals(responseCode);
        notificationService.publishDeclinedAuthorization(requestContext, new java.util.HashMap<>(processedFields));
        
        if (accountingApproved) {
            // Extract accounting transaction ID from processed fields
            String accountingTxnId = processedFields.get(901);
            if (accountingTxnId != null) {
                automaticReversalService.markAccountingApproved(uniqueTransactionId, accountingTxnId);
            }
        }
        
        // Create response - WRAP IN TRY-CATCH
        Iso8583Message response;
        try {
            response = buildResponse(processedFields, requestContext);
        } catch (Exception e) {
            log.error("Failed to build response", e);
            if (accountingApproved) {
                automaticReversalService.handlePostApprovalFailure(
                    uniqueTransactionId, "RESPONSE_BUILD", 
                    "Failed to build response: " + e.getMessage(), e);
            }
            throw e;
        }
        
        // Save transaction response - WRAP IN TRY-CATCH
        try {
            saveTransactionResponse(response, requestContext);
        } catch (Exception e) {
            log.error("Failed to save transaction response", e);
            if (accountingApproved) {
                automaticReversalService.handlePostApprovalFailure(
                    uniqueTransactionId, "RESPONSE_PERSIST", 
                    "Failed to persist response: " + e.getMessage(), e);
            }
            throw e;
        }
        
        // Mark response as sent (this clears reversal requirement)
        automaticReversalService.markResponseSent(uniqueTransactionId);

        transactionOutcomeMetrics.completed(
                requestContext,
                "00".equals(response.getField(39)) ? "success" : "declined",
                response.getField(39));
        
        return CompletableFuture.completedFuture(response);
        
    } catch (Exception e) {
        log.error("Error processing ISO8583 message", e);
        
        // Generic failure handler - will check if reversal needed
        automaticReversalService.handlePostApprovalFailure(
            uniqueTransactionId, "PROCESSING", 
            "Processing failed: " + e.getMessage(), e);

        transactionOutcomeMetrics.completed(
                requestContext,
                isTimeout(e) ? "timeout" : "error",
                "none");
        
        throw new RuntimeException("Failed to process ISO8583 message", e);
    } finally {
        transactionOutcomeMetrics.finished();
        NetworkSessionContextHolder.clear();
        MDC.remove("txnId");
    }

private boolean isTimeout(Throwable throwable) {
    Throwable current = throwable;
    while (current != null) {
        if (current instanceof java.util.concurrent.TimeoutException
                || current instanceof java.net.SocketTimeoutException
                || current instanceof java.net.http.HttpTimeoutException) {
            return true;
        }
        current = current.getCause();
    }
    return false;
}

private RequestContext buildRequestContext(Iso8583Message request, String uniqueTransactionId) {
    String institutionId = resolveInstitutionId(request);
    RequestContext requestContext = new RequestContext(
        uniqueTransactionId,
        uniqueTransactionId,
        institutionId,
        request.getMti(),
        request.getScheme(),
        request.getFields()
    );
    String panHash = cardDataEncryptionService.computePanLookupHmac(requestContext.getCardNumber(), parseInstitutionId(institutionId));
    requestContext.setPanHash(panHash);
    return requestContext;
}

private String resolveInstitutionId(Iso8583Message request) {
    String pan = request.getField(2);
    if (pan != null && pan.length() >= 6) {
        CacheInitializationService.BinInstitutionData binData = cacheInitializationService
                .loadBinInstitutionCache()
                .get(pan.substring(0, 6));
        if (binData != null && binData.getBinDetails() != null && binData.getBinDetails().getInstitution_id() != null) {
            return String.valueOf(binData.getBinDetails().getInstitution_id());
        }
    }
    return request.getField(32);
}

private Integer parseInstitutionId(String institutionId) {
    if (institutionId == null || institutionId.isBlank()) {
        return null;
    }
    try {
        return Integer.valueOf(institutionId);
    } catch (NumberFormatException ex) {
        return null;
    }
}

private Iso8583Message buildResponse(Map<Integer, String> processedFields, RequestContext requestContext) {
    Iso8583Message response = new Iso8583Message();
    response.setScheme(requestContext.getScheme());
    response.setMti(processedFields.get(0));
    response.setUniqueTransactionId(requestContext.getUniqueTransactionId());
    
    processedFields.remove(0);
    processedFields.remove(14);
    processedFields.remove(18);
    processedFields.remove(22);
    processedFields.remove(25);
    processedFields.remove(33);
    processedFields.remove(35);
    processedFields.remove(40);
    processedFields.remove(42);
    processedFields.remove(43);
    processedFields.remove(52);
    processedFields.remove(61);
    processedFields.remove(901);
    sanitizeResponseDe48(processedFields);
    sanitizeFailedResponseFields(processedFields);
    sanitizeResponseDe55(processedFields, requestContext);
    
    response.setFields(processedFields);
    return response;
}

private void sanitizeFailedResponseFields(Map<Integer, String> fields) {
    if ("00".equals(fields.get(39))) {
        return;
    }

    fields.remove(55);
    removeTlvTag(fields, 120, "004");
}

private void removeTlvTag(Map<Integer, String> fields, int fieldNumber, String tag) {
    String tlv = fields.get(fieldNumber);
    if (tlv == null || tlv.isBlank()) {
        return;
    }

    try {
        Map<String, String> tlvMap = TlvUtils.parseTlv(tlv);
        if (tlvMap.remove(tag) == null) {
            return;
        }

        if (tlvMap.isEmpty()) {
            fields.remove(fieldNumber);
        } else {
            fields.put(fieldNumber, packTlv(tlvMap));
        }
    } catch (Exception ex) {
        log.warn("Unable to sanitize response DE{} tag {}; removing field: {}",
                fieldNumber, tag, ex.getMessage());
        fields.remove(fieldNumber);
    }
}

private void sanitizeResponseDe48(Map<Integer, String> fields) {
    String de48 = fields.get(48);
    if (de48 == null || de48.isBlank()) {
        return;
    }

    try {
        Map<String, String> tlvMap = TlvUtils.parseTlv(de48);
        if (!tlvMap.keySet().removeAll(RESPONSE_DE48_OMIT_TAGS)) {
            return;
        }

        fields.put(48, packTlv(tlvMap));
    } catch (Exception ex) {
        log.warn("Unable to sanitize response DE48 tags: {}", ex.getMessage());
    }
}

private void sanitizeResponseDe55(Map<Integer, String> fields, RequestContext requestContext) {
    if (!Scheme.RUPAY.equals(requestContext.getScheme()) || !fields.containsKey(55)) {
        return;
    }

    try {
        ChipData chipData = new ChipDataParser().parseChipData(fields.get(55));
        String tag91 = chipData.getTagValue("91");
        if (tag91 == null || tag91.isBlank()) {
            fields.remove(55);
            return;
        }

        ChipData responseChipData = new ChipData();
        responseChipData.setTagValue("91", tag91.trim());
        fields.put(55, responseChipData.buildIccData());
    } catch (Exception ex) {
        log.warn("Unable to sanitize response DE55; removing field 55: {}", ex.getMessage());
        fields.remove(55);
    }
}

private String packTlv(Map<String, String> tlvMap) {
    StringBuilder packed = new StringBuilder();
    for (Map.Entry<String, String> entry : tlvMap.entrySet()) {
        packed.append(entry.getKey())
                .append(String.format("%03d", entry.getValue().length()))
                .append(entry.getValue());
    }
    return packed.toString();
}

private void saveTransactionResponse(Iso8583Message response, RequestContext requestContext) {
    TransactionResponse transactionResponse = new TransactionResponse();
    transactionResponse.setTransactionId(requestContext.getUniqueTransactionId());
    transactionResponse.setResponseCode(response.getField(39));
    transactionResponse.setResponseMessage("");
    transactionResponse.setAuthCode(response.getField(38));
    transactionResponse.setChipData(response.getField(55));
    transactionResponse.setMti(response.getMti());
    transactionResponse.setOriginalData(requestContext.getFields().get(90));
    transactionResponseRepository.save(transactionResponse);
    log.info("Transaction response saved with ID: {}", requestContext.getUniqueTransactionId());
}

    /**
     * Processes a Dynamic Key Exchange (DKE) ISO8583 message. This method
     * handles logic specific to DKE messages, such as key parsing, validation,
     * updating security modules, and assembling DKE-compliant response fields.
     *
     * @param request The incoming ISO8583 message representing a DKE request
     * @return The processed ISO8583 response message for the DKE operation
     */
    public Iso8583Message processNetworkKeyExchange(Iso8583Message request, String scheme) {
        String uniqueTransactionId = java.util.UUID.randomUUID().toString();
        try {
            applyNetworkSessionContext(request);
            log.info("Processing Dynamic Key Exchange message: txnId={}", uniqueTransactionId);

            String reqKcv = reqEncryptedKeyTR31.substring(reqEncryptedKeyTR31.length() - 6); // // last 6 characters of the encrypted key
            reqEncryptedKeyTR31 = reqEncryptedKeyTR31.substring(0, reqEncryptedKeyTR31.length() - 6); // // last 6 characters of the encrypted key
            String reqHeader = reqEncryptedKeyTR31.substring(0, 16); // 16 bytes of the encrypted key
            String keyType = request.getField(70); // Key Type or usage info (70) 
            if (scheme.equals("RUPAY") && keyType.equals("184")) {
                keyType = "ZPK";
                if (reqHeader.equals("B0080P0TB00S0000")) {
                    responseCode = "00";
                    reesCode = InternalResCode.APPROVED;
                } else {
                    log.error("Invalid header for " + keyType);
                    responseCode = "81";
                    reesCode = InternalResCode.CRYPTOGRAPHIC_ERROR;
                }
            } else if (scheme.equals("RUPAY") && keyType.equals("185")) {
                keyType = "ZAK";
                if (reqHeader.equals("B0080M3TC00S0000")) {
                    responseCode = "00";
                    reesCode = InternalResCode.APPROVED;
                } else {
                    log.error("Invalid header for " + keyType);
                    responseCode = "81";
                    reesCode = InternalResCode.CRYPTOGRAPHIC_ERROR;

                String[] workingKeyParts = hmsResponse.split(";");
                if (workingKeyParts.length != 3) {

                    responseCode = "81";
                    reesCode = InternalResCode.CRYPTOGRAPHIC_ERROR;

                    responseCode = "00";

                        log.error("KCV mismatch for " + keyType);

                        } else {
                            responseCode = "00";
                            reesCode = InternalResCode.APPROVED;

    private boolean rotateKeyForRequestContext(Iso8583Message request,
                                               String keyType,
                                               String scheme,
                                               String encryptedKeyTR31,
                                               String kcv) {
        String siteId = request.getMetadataValue("site_id");
        String listenerId = request.getMetadataValue("listener_id");
        String connectionId = request.getMetadataValue("connection_id");

        if (hasText(siteId) && hasText(listenerId) && hasText(connectionId)) {
            log.info("Rotating DKE key into network_session_keys for site={}, listener={}, connection={}, scheme={}, keyType={}",
                    siteId, listenerId, connectionId, scheme, keyType);
            return keyManagementService.rotateSessionNetworkKey(
                    keyType,
                    scheme,
                    siteId,
                    listenerId,
                    connectionId,
                    encryptedKeyTR31,
                    kcv
            );
        }

        log.warn("Missing site/listener/connection metadata for DKE key rotation. Falling back to legacy key_management path for scheme={}, keyType={}",
                scheme, keyType);
        return keyManagementService.rotateKey(keyType, scheme, "INACTIVE", encryptedKeyTR31, kcv);
    }

    private void applyNetworkSessionContext(Iso8583Message request) {
        if (request == null) {
            NetworkSessionContextHolder.clear();
            return;
        }

        NetworkSessionContextHolder.set(
                request.getMetadataValue("site_id"),
                request.getMetadataValue("listener_id"),
                request.getMetadataValue("connection_id"),
                request.getScheme() != null ? request.getScheme().name() : null
        );
    }

    private boolean hasText(String value) {
        return value != null && !value.isBlank();
    }
