package com.logicpoint.iso8583.service;

import com.logicpoint.iso8583.model.DownstreamHealthResponse;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Service;

import javax.sql.DataSource;
import java.sql.Connection;
import java.time.LocalDateTime;
import java.util.LinkedHashMap;
import java.util.Map;

@Service
public class RoutingHealthCheckService {

    private static final Logger log = LoggerFactory.getLogger(RoutingHealthCheckService.class);

    private final DataSource dataSource;
    private final HsmKeepAliveServiceInterface hsmKeepAliveService;

    @Value("${spring.application.name:switch-core-processor}")
    private String applicationName;

    @Value("${switch.site.id:UNKNOWN_SITE}")
    private String siteId;

    @Value("${switch.node.id:UNKNOWN_NODE}")
    private String nodeId;

    @Value("${accounting.api.enabled:true}")
    private boolean accountingApiEnabled;

    public RoutingHealthCheckService(DataSource dataSource, HsmKeepAliveServiceInterface hsmKeepAliveService) {
        this.dataSource = dataSource;
        this.hsmKeepAliveService = hsmKeepAliveService;
    }

    public DownstreamHealthResponse getRoutingHealth() {
        boolean databaseUp = isDatabaseUp();
        Map<String, Object> checks = new LinkedHashMap<>();
        checks.put("database", databaseCheck(databaseUp));
        checks.put("hsm", hsmCheck());
        checks.put("accounting", accountingCheck());

        boolean routable = databaseUp && isHsmRoutable();

        DownstreamHealthResponse response = new DownstreamHealthResponse();
        response.setServiceName(applicationName);
        response.setStatus(routable ? "UP" : "DOWN");
        response.setRoutable(routable);
        response.setTimestamp(LocalDateTime.now());
        response.setNode(Map.of(
                "siteId", siteId,
                "nodeId", nodeId
        ));
        response.setChecks(checks);
        return response;
    }

    private boolean isDatabaseUp() {
        try (Connection connection = dataSource.getConnection()) {
            return connection.isValid(2);
        } catch (Exception ex) {
            log.warn("Routing health check database validation failed: {}", ex.getMessage());
            return false;
        }
    }

    private Map<String, Object> databaseCheck(boolean up) {
        Map<String, Object> result = new LinkedHashMap<>();
        result.put("status", up ? "UP" : "DOWN");
        return result;
    }

    private Map<String, Object> hsmCheck() {
        Map<String, Object> result = new LinkedHashMap<>();
        int total = hsmKeepAliveService.getTotalConnectionCount();
        int healthy = hsmKeepAliveService.getHealthyConnectionCount();
        int unhealthy = hsmKeepAliveService.getUnhealthyConnectionCount();
        boolean keepAliveRunning = hsmKeepAliveService.isRunning();

        String status;
        if (!keepAliveRunning) {
            status = "UNKNOWN";
        } else if (total == 0) {
            status = "UNKNOWN";
        } else if (healthy > 0) {
            status = "UP";
        } else {
            status = "DOWN";
        }

        result.put("status", status);
        result.put("keepAliveRunning", keepAliveRunning);
        result.put("totalConnections", total);
        result.put("healthyConnections", healthy);
        result.put("unhealthyConnections", unhealthy);
        return result;
    }

    private Map<String, Object> accountingCheck() {
        Map<String, Object> result = new LinkedHashMap<>();
        result.put("status", accountingApiEnabled ? "CONFIGURED" : "DISABLED");
        result.put("enabled", accountingApiEnabled);
        return result;
    }

    private boolean isHsmRoutable() {
        if (!hsmKeepAliveService.isRunning()) {
            return true;
        }

        int total = hsmKeepAliveService.getTotalConnectionCount();
        if (total == 0) {
            return true;
        }

        return hsmKeepAliveService.getHealthyConnectionCount() > 0;
    }
}
