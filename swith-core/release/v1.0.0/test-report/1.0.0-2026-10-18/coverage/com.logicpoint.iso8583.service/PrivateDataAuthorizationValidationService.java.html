package com.logicpoint.iso8583.service;

import com.logicpoint.iso8583.config.PrivateDataAuthorizationProperties;
import com.logicpoint.iso8583.context.RequestContext;
import com.logicpoint.iso8583.util.Scheme;
import com.logicpoint.iso8583.validation.TransactionType;
import java.util.ArrayList;
import java.util.List;
import java.util.Map;
import org.springframework.stereotype.Service;

/**
 * Validates scheme-configured TLV tags in private authorization data fields.
 */
@Service
public class PrivateDataAuthorizationValidationService {

    private final PrivateDataAuthorizationProperties properties;

    public PrivateDataAuthorizationValidationService(PrivateDataAuthorizationProperties properties) {
        this.properties = properties;
    }

    public List<String> validate(RequestContext requestContext, Scheme scheme, TransactionType transactionType) {
        List<String> errors = new ArrayList<>();
        if (!properties.isEnabled() || requestContext == null || scheme == null || transactionType == null) {
            return errors;
        }

        PrivateDataAuthorizationProperties.SchemeRule schemeRule = properties.getRule(scheme.name());
        if (schemeRule == null || !schemeRule.isEnabled()) {
            return errors;
        }

        List<PrivateDataAuthorizationProperties.TagRule> rules =
                schemeRule.getTransactionRules().get(transactionType);
        if (rules == null || rules.isEmpty()) {
            return errors;
        }

        for (PrivateDataAuthorizationProperties.TagRule rule : rules) {
            validateRule(requestContext, scheme, transactionType, rule, errors);
        }
        return errors;
    }

    private void validateRule(RequestContext requestContext,
                              Scheme scheme,
                              TransactionType transactionType,
                              PrivateDataAuthorizationProperties.TagRule rule,
                              List<String> errors) {
        Map<String, String> tags;
        try {
            tags = requestContext.getParsedTlv(rule.getFieldNumber());
        } catch (Exception ex) {
            errors.add("INVALID_DE" + rule.getFieldNumber() + "_FORMAT: Unable to parse "
                    + description(rule) + " for " + scheme + " " + transactionType);
            return;
        }

        String value = normalize(tags.get(rule.getTag()));
        if (isBlank(value)) {
            if (rule.isRequired()) {
                errors.add("MISSING_DE" + rule.getFieldNumber() + "_TAG_" + rule.getTag() + ": "
                        + description(rule) + " is required for " + scheme + " " + transactionType);
            }
            return;
        }

        if (!rule.getAllowedValues().isEmpty() && !rule.getAllowedValues().contains(value)) {
            errors.add("INVALID_DE" + rule.getFieldNumber() + "_TAG_" + rule.getTag() + ": "
                    + description(rule) + " value " + value + " is not allowed for "
                    + scheme + " " + transactionType);
        }
    }

    private String description(PrivateDataAuthorizationProperties.TagRule rule) {
        return isBlank(rule.getDescription())
                ? "DE" + rule.getFieldNumber() + " tag " + rule.getTag()
                : rule.getDescription();
    }

    private String normalize(String value) {
        return isBlank(value) ? null : value.trim().toUpperCase();
    }

    private boolean isBlank(String value) {
        return value == null || value.trim().isEmpty();
    }
}
