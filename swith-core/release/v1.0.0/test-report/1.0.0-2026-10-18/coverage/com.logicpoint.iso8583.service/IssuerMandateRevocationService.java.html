package com.logicpoint.iso8583.service;

import com.logicpoint.iso8583.entity.StandingInstructionMandate;
import com.logicpoint.iso8583.model.Iso8583Message;
import com.logicpoint.iso8583.model.IssuerMandateRevocationRequest;
import com.logicpoint.iso8583.model.IssuerMandateRevocationResult;
import com.logicpoint.iso8583.util.Scheme;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.stereotype.Service;

/** Orchestrates the CMS/CRM trigger, outbound transport, and response reconciliation. */
@Service
public class IssuerMandateRevocationService {

    private static final Logger log = LoggerFactory.getLogger(IssuerMandateRevocationService.class);

    private final IssuerMandateRevocationLifecycleService lifecycleService;
    private final IssuerMandateRevocationMessageFactory messageFactory;
    private final IssuerActionMessageGateway messageGateway;

    public IssuerMandateRevocationService(
            IssuerMandateRevocationLifecycleService lifecycleService,
            IssuerMandateRevocationMessageFactory messageFactory,
            IssuerActionMessageGateway messageGateway) {
        this.lifecycleService = lifecycleService;
        this.messageFactory = messageFactory;
        this.messageGateway = messageGateway;
    }

    public IssuerMandateRevocationResult revoke(IssuerMandateRevocationRequest command) {
        StandingInstructionMandate mandate = lifecycleService.getMandate(command.scheme(), command.mandateId());
        Iso8583Message request = messageFactory.create(mandate, command);
        StandingInstructionMandate pending = lifecycleService.markPending(command.scheme(), command.mandateId(), request);
        try {
            log.info("Sending issuer mandate revocation: commandId={}, scheme={}, mandateId={}, requestedBy={}",
                    request.getUniqueTransactionId(), command.scheme(), command.mandateId(), command.requestedBy());
            messageGateway.send(request);
            return result(pending, request.getMti(), null);
        } catch (RuntimeException ex) {
            log.error("Issuer mandate revocation remains pending after listener dispatch failure: commandId={}",
                    request.getUniqueTransactionId(), ex);
            throw ex;
        }
    }

    public IssuerMandateRevocationResult reconcile(Scheme scheme, Iso8583Message response) {
        StandingInstructionMandate reconciled = lifecycleService.reconcile(scheme, response);
        return result(reconciled, null, response.getMti());
    }

    private IssuerMandateRevocationResult result(StandingInstructionMandate mandate,
                                                  String requestMti,
                                                  String responseMti) {
        return new IssuerMandateRevocationResult(
                mandate.getRevocationCommandId(),
                mandate.getMandateId(),
                requestMti,
                responseMti,
                mandate.getRevocationResponseCode(),
                mandate.getStatus().name());
    }
}
