package com.logicpoint.iso8583.service;

import com.logicpoint.iso8583.config.IssuerMandateRevocationProperties;
import com.logicpoint.iso8583.entity.StandingInstructionMandate;
import com.logicpoint.iso8583.entity.StandingInstructionMandate.MandateStatus;
import com.logicpoint.iso8583.model.Iso8583Message;
import com.logicpoint.iso8583.repository.StandingInstructionMandateRepository;
import com.logicpoint.iso8583.util.Scheme;
import java.time.LocalDateTime;
import java.util.List;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

/** Owns transactional state transitions for outbound mandate revocation. */
@Service
public class IssuerMandateRevocationLifecycleService {

    private final StandingInstructionMandateRepository mandateRepository;
    private final IssuerMandateRevocationProperties properties;
    private final StandingInstructionHubClient hubClient;

    public IssuerMandateRevocationLifecycleService(
            StandingInstructionMandateRepository mandateRepository,
            IssuerMandateRevocationProperties properties,
            StandingInstructionHubClient hubClient) {
        this.mandateRepository = mandateRepository;
        this.properties = properties;
        this.hubClient = hubClient;
    }

    @Transactional(readOnly = true)
    public StandingInstructionMandate getMandate(Scheme scheme, String mandateId) {
        return mandateRepository.findBySchemeAndMandateId(scheme.name(), mandateId)
                .orElseThrow(() -> new IllegalArgumentException("Mandate not found: " + mandateId));
    }

    @Transactional
    public StandingInstructionMandate markPending(Scheme scheme,
                                                  String mandateId,
                                                  Iso8583Message outboundMessage) {
        StandingInstructionMandate mandate = mandateRepository
                .lockBySchemeAndMandateId(scheme.name(), mandateId)
                .orElseThrow(() -> new IllegalArgumentException("Mandate not found: " + mandateId));
        if (mandate.getStatus() == MandateStatus.REVOKED) {
            throw new IllegalStateException("Mandate is already revoked: " + mandateId);
        }
        if (mandate.getStatus() == MandateStatus.REVOCATION_PENDING) {
            throw new IllegalStateException("Mandate revocation is already pending: " + mandateId);
        }
        if (mandate.getStatus() == MandateStatus.REVOCATION_TIMED_OUT) {
            throw new IllegalStateException("Mandate revocation has already timed out: " + mandateId);
        }

        LocalDateTime now = LocalDateTime.now();
        mandate.setRevocationPreviousStatus(mandate.getStatus());
        mandate.setStatus(MandateStatus.REVOCATION_PENDING);
        mandate.setRevocationCommandId(outboundMessage.getUniqueTransactionId());
        mandate.setRevocationRequestStan(outboundMessage.getField(11));
        mandate.setRevocationRequestRrn(outboundMessage.getField(37));
        mandate.setRevocationRequestedAt(now);
        mandate.setRevocationResponseDueAt(now.plusSeconds(properties.getResponseTimeoutSeconds()));
        mandate.setRevocationTimedOutAt(null);
        mandate.setRevocationResponseCode(null);
        return mandateRepository.save(mandate);
    }

    @Transactional
    public StandingInstructionMandate reconcile(Scheme scheme, Iso8583Message response) {
        IssuerMandateRevocationProperties.SchemeRule rule = requireRule(scheme);
        if (!rule.getResponseMti().equals(response.getMti())) {
            throw new IllegalArgumentException("Unexpected mandate revocation response MTI: " + response.getMti());
        }
        String stan = requiredField(response, 11);
        String rrn = requiredField(response, 37);
        String responseCode = requiredField(response, 39);
        StandingInstructionMandate mandate = mandateRepository
                .lockByRevocationCorrelation(scheme.name(), stan, rrn)
                .orElseThrow(() -> new IllegalArgumentException(
                        "No pending mandate revocation matches DE11/DE37"));

        if (mandate.getStatus() == MandateStatus.REVOKED
                && rule.getApprovedResponseCodes().contains(responseCode)) {
            return mandate;
        }

        mandate.setRevocationResponseCode(responseCode);
        mandate.setRevocationTimedOutAt(null);
        mandate.setLastStan(stan);
        mandate.setLastRrn(rrn);
        if (rule.getApprovedResponseCodes().contains(responseCode)) {
            mandate.setStatus(MandateStatus.REVOKED);
            mandate.setRevokedAt(LocalDateTime.now());
            StandingInstructionMandate saved = mandateRepository.save(mandate);
            hubClient.mandateRevoked(saved);
            return saved;
        }

        restorePreviousStatus(mandate);
        return mandateRepository.save(mandate);
    }

    @Transactional
    public int markExpiredPendingRevocations() {
        if (!properties.isEnabled()) {
            return 0;
        }
        LocalDateTime now = LocalDateTime.now();
        List<StandingInstructionMandate> expired = mandateRepository
                .lockExpiredRevocations(MandateStatus.REVOCATION_PENDING, now);
        for (StandingInstructionMandate mandate : expired) {
            mandate.setStatus(MandateStatus.REVOCATION_TIMED_OUT);
            mandate.setRevocationTimedOutAt(now);
            mandateRepository.save(mandate);
        }
        return expired.size();
    }

    private void restorePreviousStatus(StandingInstructionMandate mandate) {
        MandateStatus previous = mandate.getRevocationPreviousStatus();
        mandate.setStatus(previous == MandateStatus.PAUSED ? MandateStatus.PAUSED : MandateStatus.ACTIVE);
    }

    private IssuerMandateRevocationProperties.SchemeRule requireRule(Scheme scheme) {
        IssuerMandateRevocationProperties.SchemeRule rule = properties.getRule(scheme.name());
        if (rule == null || !rule.isEnabled()) {
            throw new IllegalStateException("No issuer mandate revocation rule for " + scheme);
        }
        return rule;
    }

    private String requiredField(Iso8583Message message, int field) {
        String value = message.getField(field);
        if (value == null || value.isBlank()) {
            throw new IllegalArgumentException("Mandate revocation response is missing DE" + field);
        }
        return value;
    }
}
