package com.logicpoint.iso8583.service;

import java.time.Instant;
import java.util.concurrent.atomic.AtomicLong;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Service;

@Service
public class TransactionIdGeneratorService {

    private static final Logger log = LoggerFactory.getLogger(TransactionIdGeneratorService.class);

    private final AtomicLong lastEpochNanos = new AtomicLong();

    @Value("${switch.site.id:1}")
    private int siteId;

    @Value("${switch.node.id:1}")
    private int nodeId;

    public String nextTransactionId() {
        validateConfiguration();
        long epochNanos = nextMonotonicEpochNanos();
        return siteId + String.format("%02d", nodeId) + epochNanos;
    }

    private long nextMonotonicEpochNanos() {
        while (true) {
            long candidate = currentEpochNanos();
            long previous = lastEpochNanos.get();
            long next = Math.max(candidate, previous + 1);
            if (lastEpochNanos.compareAndSet(previous, next)) {
                return next;
            }
        }
    }

    private long currentEpochNanos() {
        Instant now = Instant.now();
        return (now.getEpochSecond() * 1_000_000_000L) + now.getNano();
    }

    private void validateConfiguration() {
        if (siteId < 0 || siteId > 9) {
            throw new IllegalArgumentException("switch.site.id must be between 0 and 9");
        }
        if (nodeId < 0 || nodeId > 99) {
            throw new IllegalArgumentException("switch.node.id must be between 0 and 99");
        }
    }

    @jakarta.annotation.PostConstruct
    void logConfiguration() {
        validateConfiguration();
        log.info("Initialized transaction ID generator for site {} node {}", siteId, String.format("%02d", nodeId));
    }
}
