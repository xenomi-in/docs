package com.logicpoint.iso8583.service;

import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.scheduling.annotation.Scheduled;
import org.springframework.stereotype.Component;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

@Slf4j
@Component
@RequiredArgsConstructor
public class ReversalRecoveryScheduler {

    private static final Logger log = LoggerFactory.getLogger(ReversalRecoveryScheduler.class);
    
    private final ReversalStorageService reversalStorageService;
    private final AutomaticReversalService automaticReversalService;
    
    /**
     * Migrates file-stored reversals to DB when DB becomes available
     */
    @Scheduled(fixedDelay = 300000, initialDelay = 60000) // Every 5 minutes
    public void recoverFileStoredReversals() {
        log.debug("Running reversal recovery scheduler");
        try {
            reversalStorageService.loadPendingReversalsFromFiles();
        } catch (Exception e) {
            log.error("Error in reversal recovery scheduler", e);
        }
    }
    
    /**
     * Retries pending reversals
     */
    @Scheduled(fixedDelay = 120000, initialDelay = 60000) // Every 2 minutes
    public void retryPendingReversals() {
        log.debug("Running reversal retry scheduler");
        try {
            automaticReversalService.retryPendingReversals();
        } catch (Exception e) {
            log.error("Error in reversal retry scheduler", e);
        }
    }
}
