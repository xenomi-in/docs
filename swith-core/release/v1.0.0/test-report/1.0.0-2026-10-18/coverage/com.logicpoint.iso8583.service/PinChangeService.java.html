package com.logicpoint.iso8583.service;

import com.logicpoint.iso8583.entity.CardData;
import com.logicpoint.iso8583.validation.orchestrator.CardDataContext;
import com.logicpoint.iso8583.validation.ValidationResult;

import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

import java.time.LocalDateTime;
import java.util.Map;

import com.logicpoint.iso8583.validation.TlvUtils;

/**
 * Service for handling PIN change operations using NettyHsmService.PinChangeandVerify()
 * Updates database with new PVV offset after successful PIN change
 * 
 * @author LogicPoint Systems
 * @version 1.0
 * @since 2025-01-27
 */
@Service
public class PinChangeService {
    
    private static final Logger logger = LoggerFactory.getLogger(PinChangeService.class);
    
    @Autowired
    private NettyHsmService nettyHsmService;
    
    @Autowired
    private CardDataService cardDataService;
    @Autowired
    private com.logicpoint.iso8583.cmsintegration.CmsAuthorizationLookup cmsLookup;
    
    /**
     * Process PIN change using NettyHsmService.PinChangeandVerify()
     * 
     * @param pan Primary Account Number
     * @param currentPinBlock Current encrypted PIN block
     * @param newPinBlock New encrypted PIN block
     * @param pinMethod PIN method (VISA, IBM3624, ISO9564)
     * @param transactionId Transaction ID for logging
     * @return PinChangeResult with success status and new offset
     */
    @Transactional
    public ValidationResult processPinChange(Map<Integer, String> transactionData, CardDataContext cardContext) {
        logger.info("Processing PIN change for card");
        
        try {
            if(cmsLookup!=null && cmsLookup.isManaged(cardContext.getCardData().getAuth_id()))
                return new ValidationResult.Builder().addError("CMS PIN change requires the family transaction service").build();
            String pan = transactionData.get(2);
            String currentPinBlock = transactionData.get(52);
            Map<String, String> de120Data = TlvUtils.parseTlv(transactionData.get(120));
           String newPinBlock = de120Data.get("004"); // New PIN Block
            String pinMethod = "VISA";
            
            CardData cardData = cardContext.getCardData();
            String currentOffset = cardData.getPvv_offest();
            
            if (currentOffset == null) {
                logger.error("No PVV offset found for card: {}", maskPan(pan));
                
            ValidationResult.Builder resultBuilder = new ValidationResult.Builder();
            resultBuilder.addError("Missing PVV offset");
            return resultBuilder.build();
            }
            // Step 2: Store original offset for rollback
            String originalOffset = currentOffset;
            
            // Step 3: Call HSM PIN change
            logger.debug("Calling HSM PIN change for card: {}", maskPan(pan));
            String hsmResult = nettyHsmService.pinChangeAndVerify(
                currentPinBlock,  // Current PIN block
                pan,             // PAN
                newPinBlock,     // New PIN block
                pinMethod,       // PIN method
                currentOffset,   // Current offset
                false            // false = PIN change, true = verification
            );
            
            logger.debug("HSM PIN change result: {}", hsmResult);
            
            // Step 4: Parse HSM response
            if (hsmResult == null || hsmResult.startsWith("F")) {
                logger.warn("HSM PIN change failed for card: {}. Result: {}", maskPan(pan), hsmResult);
                return handleHsmFailure(hsmResult);
            }
            
            // Step 5: Extract result and new offset
            String[] resultParts = hsmResult.split(";");
            if (resultParts.length < 2) {
                logger.error("Invalid HSM response format: {}", hsmResult);
               // return new PinChangeResult(false, "96", "Invalid HSM response", null);
                ValidationResult.Builder resultBuilder = new ValidationResult.Builder();
                resultBuilder.addError("Invalid HSM response");
                return resultBuilder.build();
            }
            
            String pinChangeResult = resultParts[0];
            String newOffset = resultParts[1];
            
            // Step 6: Check if PIN change was successful
            if (!"Y".equals(pinChangeResult)) {
                logger.warn("PIN change failed for card: {}. HSM result: {}", maskPan(pan), pinChangeResult);
                ValidationResult.Builder resultBuilder = new ValidationResult.Builder();
                resultBuilder.addError("PIN change failed");
                return resultBuilder.build();

            }
            
            // Step 7: Update database with new PVV offset
            try {
                updateCardDataWithNewOffset(cardData, newOffset, originalOffset);
                logger.info("Successfully updated PVV offset for card: {}", maskPan(pan));
                
                ValidationResult.Builder resultBuilder = new ValidationResult.Builder();
                resultBuilder.addWarning("PIN change successful");
                return resultBuilder.build();
                
            } catch (Exception dbException) {
                logger.error("Failed to update database for card: {}", maskPan(pan), dbException);
                ValidationResult.Builder resultBuilder = new ValidationResult.Builder();
                resultBuilder.addError("Database update failed");
                return resultBuilder.build();
            }
            
        } catch (Exception e) {
            logger.error("Error during PIN change process " , e);
            ValidationResult.Builder resultBuilder = new ValidationResult.Builder();
            resultBuilder.addError("System malfunction");
            return resultBuilder.build();
        }
    }
    
    /**
     * Update card data with new PVV offset
     */
    private void updateCardDataWithNewOffset(CardData cardData, String newOffset, String originalOffset) {
        try {
            // Store old offset for audit trail
            cardData.setOld_pvv_offest(originalOffset);
            
            // Update with new offset
            cardData.setPvv_offest(newOffset);
            cardData.setUpdated_at(LocalDateTime.now());
            
            // Reset PIN retry count on successful PIN change
            cardData.setPin_retry_count(0);
            cardData.setPin_blocked(false);
            
            // Save to database
            if (!cardDataService.updateCardData(cardData)) {
                throw new RuntimeException("Failed to persist encrypted card data");
            }
            
            logger.debug("Updated card data after PIN change for card: {}", maskPan(cardData.getPan()));
            
        } catch (Exception e) {
            logger.error("Error updating card data: {}", e.getMessage());
            throw new RuntimeException("Failed to update card data", e);
        }
    }
    
    /**
     * Handle HSM failure scenarios
     */
    private ValidationResult handleHsmFailure(String hsmResult) {
        String responseCode;
        String message;
        
        if (hsmResult == null) {
            responseCode = "96";
            message = "HSM communication error";
        } else if (hsmResult.contains("Invalid PIN")) {
            responseCode = "55";
            message = "Invalid PIN";
        } else if (hsmResult.contains("PIN blocked")) {
            responseCode = "55";
            message = "PIN blocked";
        } else if (hsmResult.contains("Invalid pin method")) {
            responseCode = "30";
            message = "Invalid PIN method";
        } else {
            responseCode = "96";
            message = "HSM error: " + hsmResult;
        }
        
        ValidationResult.Builder resultBuilder = new ValidationResult.Builder();
        resultBuilder.addError( message);
        return resultBuilder.build();
    }
    
   
    
    /**
     * Mask PAN for logging
     */
    private String maskPan(String pan) {
        if (pan == null || pan.length() < 8) {
            return "****";
        }
        return pan.substring(0, 4) + "****" + pan.substring(pan.length() - 4);
    }
    
    /**
     * PIN Change Result DTO
     */
    public static class PinChangeResult {
        private boolean success;
        private String responseCode;
        private String message;
        private String newOffset;
        
        public PinChangeResult(boolean success, String responseCode, String message, String newOffset) {
            this.success = success;
            this.responseCode = responseCode;
            this.message = message;
            this.newOffset = newOffset;
        }
        
        // Getters
        public boolean isSuccess() { return success; }
        public String getResponseCode() { return responseCode; }
        public String getMessage() { return message; }
        public String getNewOffset() { return newOffset; }
        
        @Override
        public String toString() {
            return String.format("PinChangeResult{success=%s, responseCode='%s', message='%s', newOffset='%s'}", 
                success, responseCode, message, newOffset);
        }
    }
}
