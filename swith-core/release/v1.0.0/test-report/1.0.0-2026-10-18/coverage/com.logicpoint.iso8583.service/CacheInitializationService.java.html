package com.logicpoint.iso8583.service;

import com.logicpoint.iso8583.entity.BinDetails;
import com.logicpoint.iso8583.entity.Institutions;
import com.logicpoint.iso8583.entity.CardData;
import com.logicpoint.iso8583.repository.BinDetailsRepository;
import com.logicpoint.iso8583.repository.InstitutionsRepository;
import com.logicpoint.iso8583.repository.CardDataRepository;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.context.event.ApplicationReadyEvent;
import org.springframework.cache.Cache;
import org.springframework.cache.CacheManager;
import org.springframework.cache.annotation.CacheEvict;
import org.springframework.context.event.EventListener;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.scheduling.annotation.Scheduled;
import org.springframework.stereotype.Service;

import java.util.List;
import java.util.Map;
import java.util.HashMap;
import org.springframework.cache.interceptor.SimpleKey;

/**
 * Service for initializing and managing cache data for validation entities.
 * Loads data on application startup and provides scheduled cache updates.
 */
@Service
public class CacheInitializationService {

    Logger log = LoggerFactory.getLogger(CacheInitializationService.class);
    @Autowired
    private BinDetailsRepository binDetailsRepository;

    @Autowired
    private InstitutionsRepository institutionsRepository;

    @Autowired
    private CardDataRepository cardDataRepository;

    @Autowired
    private CacheManager cacheManager;

    private static final SimpleKey NO_ARG_CACHE_KEY = SimpleKey.EMPTY;

    /**
     * Initialize all caches after the application is fully ready so the first
     * live request doesn't pay the cache warm-up cost.
     */
    @EventListener(ApplicationReadyEvent.class)
    public void initializeCaches() {
        warmBinInstitutionCache();
        // warmCardDataCache();
    }

    /**
     * Load BIN and Institution data into cache
     */
    public Map<String, BinInstitutionData> loadBinInstitutionCache() {
        Cache cache = getRequiredCache("binInstitutionCache");
        Map<String, BinInstitutionData> cached = cache.get(NO_ARG_CACHE_KEY, Map.class);
        if (cached != null) {
            return cached;
        }
        return warmBinInstitutionCache();
    }

    private Map<String, BinInstitutionData> warmBinInstitutionCache() {
        log.info("Warming BIN and Institution cache on startup/refresh");
        Map<String, BinInstitutionData> loaded = buildBinInstitutionCache();
        getRequiredCache("binInstitutionCache").put(NO_ARG_CACHE_KEY, loaded);
        log.info("BIN and Institution cache warmed with {} BIN entries", loaded.size());
        return loaded;
    }

    private Map<String, BinInstitutionData> buildBinInstitutionCache() {
        Map<String, BinInstitutionData> cache = new HashMap<>();

        List<BinDetails> binDetails = binDetailsRepository.findAll();
        List<Institutions> institutions = institutionsRepository.findAll();

        Map<Long, Institutions> institutionMap = new HashMap<>();
        for (Institutions inst : institutions) {
            try {
                Long id = (Long) inst.getClass().getMethod("getInstitutionId").invoke(inst);
                institutionMap.put(id, inst);
            } catch (Exception e) {
                log.warn("Skipping institution cache entry due to reflection error: {}", e.getMessage());
            }
        }

        for (BinDetails bin : binDetails) {
            try {
                Integer institutionId = (Integer) bin.getClass().getMethod("getInstitution_id").invoke(bin);
                String binNumber = (String) bin.getClass().getMethod("getBinNumber").invoke(bin);
                Institutions institution = institutionMap.get(institutionId.longValue());
                if (institution != null) {
                    cache.put(binNumber, new BinInstitutionData(bin, institution));
                }
            } catch (Exception e) {
                log.warn("Skipping BIN cache entry due to reflection error: {}", e.getMessage());
            }
        }

        return cache;
    }

    /**
     * Load Card data into cache
     */
    public Map<String, CardData> loadCardDataCache() {
        Cache cache = getRequiredCache("cardDataCache");
        Map<String, CardData> cached = cache.get(NO_ARG_CACHE_KEY, Map.class);
        if (cached != null) {
            return cached;
        }
        return warmCardDataCache();
    }

    private Map<String, CardData> warmCardDataCache() {
        List<CardData> cardDataList = cardDataRepository.findAll();
        Map<String, CardData> cache = new HashMap<>();

        for (CardData card : cardDataList) {
            try {
                String pan = (String) card.getClass().getMethod("getPan").invoke(card);
                cache.put(pan, card);
            } catch (Exception e) {
                log.warn("Skipping card cache entry due to reflection error: {}", e.getMessage());
            }
        }

        getRequiredCache("cardDataCache").put(NO_ARG_CACHE_KEY, cache);
        return cache;
    }
    
    /**
     * Scheduled cache refresh every 5 minutes
     */
    @Scheduled(fixedRate = 300000) // 5 minutes
    @CacheEvict(value = {"binInstitutionCache", "cardDataCache"}, allEntries = true)
    public void refreshAllCaches() {
        warmBinInstitutionCache();
        warmCardDataCache();
    }

    /**
     * Manual cache refresh method
     */
    public void refreshCache(String cacheName) {
        getRequiredCache(cacheName).clear();
        switch (cacheName) {
            case "binInstitutionCache":
                warmBinInstitutionCache();
                break;
            case "cardDataCache":
                warmCardDataCache();
                break;
            default:
                log.warn("Unsupported cache refresh requested for cache: {}", cacheName);
        }
    }

    private Cache getRequiredCache(String cacheName) {
        Cache cache = cacheManager.getCache(cacheName);
        if (cache == null) {
            throw new IllegalStateException("Cache not configured: " + cacheName);
        }
        return cache;
    }

    /**
     * Data class to hold BIN and Institution data together
     */
    public static class BinInstitutionData {
        private final BinDetails binDetails;
        private final Institutions institution;

        public BinInstitutionData(BinDetails binDetails, Institutions institution) {
            this.binDetails = binDetails;
            this.institution = institution;
        }

        public BinDetails getBinDetails() { return binDetails; }
        public Institutions getInstitution() { return institution; }
        
        public boolean isBinActive() {
            try {
                String status = (String) binDetails.getClass().getMethod("getStatus").invoke(binDetails);
                return "ACTIVE".equalsIgnoreCase(status);
            } catch (Exception e) {
                return false;
            }
        }
        
        public boolean isInstitutionActive() {
            try {
                String status = (String) institution.getClass().getMethod("getStatus").invoke(institution);
                return "ACTIVE".equalsIgnoreCase(status);
            } catch (Exception e) {
                return false;
            }
        }
        
        public String getBinNumber() { 
            try {
                return (String) binDetails.getClass().getMethod("getBin_number").invoke(binDetails);
            } catch (Exception e) {
                return null;
            }
        }
        
        public String getCardScheme() { 
            try {
                return (String) binDetails.getClass().getMethod("getCard_scheme").invoke(binDetails);
            } catch (Exception e) {
                return null;
            }
        }
        
        public String getInstitutionCode() { 
            try {
                return (String) institution.getClass().getMethod("getInstitutionCode").invoke(institution);
            } catch (Exception e) {
                return null;
            }
        }
    }
}
