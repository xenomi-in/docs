package com.logicpoint.iso8583.service;

import java.security.SecureRandom;
import java.util.Arrays;
import java.util.LinkedHashMap;
import java.util.Map;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.boot.context.event.ApplicationReadyEvent;
import org.springframework.context.event.EventListener;
import org.springframework.stereotype.Service;

import com.logicpoint.iso8583.entity.KeyManagement;

@Service
public class ChdKeyProvisioningService {

    private static final Logger logger = LoggerFactory.getLogger(ChdKeyProvisioningService.class);
    private static final int KEY_SIZE_BYTES = 32;

    private final KeyManagementService keyManagementService;
    private final NettyHsmService nettyHsmService;
    private final ChdKeyCacheService chdKeyCacheService;
    private final SecureRandom secureRandom = new SecureRandom();

    @Value("${chd.encryption.enabled:false}")
    private boolean encryptionEnabled;

    @Value("${chd.encryption.auto-provision-keys:false}")
    private boolean autoProvisionKeys;

    @Value("${chd.encryption.institution-id:1}")
    private Integer institutionId;

    @Value("${chd.encryption.wrapping-key-type:DEK}")
    private String wrappingKeyType;

    @Value("${chd.encryption.card-key-algorithm:AES-256-GCM}")
    private String cardKeyAlgorithm;

    @Value("${chd.encryption.txn-key-algorithm:AES-256-GCM}")
    private String txnKeyAlgorithm;

    @Value("${chd.encryption.hmac-key-algorithm:HMAC-SHA-256}")
    private String hmacKeyAlgorithm;

    public ChdKeyProvisioningService(KeyManagementService keyManagementService,
                                     NettyHsmService nettyHsmService,
                                     ChdKeyCacheService chdKeyCacheService) {
        this.keyManagementService = keyManagementService;
        this.nettyHsmService = nettyHsmService;
        this.chdKeyCacheService = chdKeyCacheService;
    }

    @EventListener(ApplicationReadyEvent.class)
    public void ensureProvisionedKeys() {
        if (!encryptionEnabled || !autoProvisionKeys) {
            return;
        }

        provisionDefaultKeys(institutionId);
    }

    public Map<String, String> provisionDefaultKeys() {
        return provisionDefaultKeys(institutionId);
    }

    public Map<String, String> provisionDefaultKeys(Integer targetInstitutionId) {
        Map<String, String> result = new LinkedHashMap<>();
        result.put(ChdKeyCacheService.CARD_KEY_TYPE, provisionIfMissing(targetInstitutionId, ChdKeyCacheService.CARD_KEY_TYPE, cardKeyAlgorithm));
        result.put(ChdKeyCacheService.TXN_KEY_TYPE, provisionIfMissing(targetInstitutionId, ChdKeyCacheService.TXN_KEY_TYPE, txnKeyAlgorithm));
        result.put(ChdKeyCacheService.PAN_HMAC_KEY_TYPE, provisionIfMissing(targetInstitutionId, ChdKeyCacheService.PAN_HMAC_KEY_TYPE, hmacKeyAlgorithm));
        chdKeyCacheService.refresh();
        return result;
    }

    private String provisionIfMissing(Integer targetInstitutionId, String appKeyType, String algorithm) {
        KeyManagement existingKey = keyManagementService.getCurrentInstitutionKey(targetInstitutionId, appKeyType);
        if (existingKey != null) {
            logger.info("CHD key {} already exists at institution {}", appKeyType, targetInstitutionId);
            return "EXISTS";
        }

        byte[] clearKey = new byte[KEY_SIZE_BYTES];
        secureRandom.nextBytes(clearKey);

        try {
            HsmWrappedKeyResult wrappedKeyResult = nettyHsmService.wrapClearChdKey(clearKey, targetInstitutionId, wrappingKeyType, true);
            if (wrappedKeyResult == null) {
                logger.error("Unable to provision CHD key {} through HSM", appKeyType);
                return "FAILED";
            }

            keyManagementService.saveInstitutionWrappedKey(
                    targetInstitutionId,
                    appKeyType,
                    wrappedKeyResult.getWrappedKey(),
                    wrappedKeyResult.getKcv(),
                    algorithm,
                    wrappingKeyType + ":" + targetInstitutionId,
                    "SYSTEM-CHD");

            logger.info("Provisioned CHD key {} for institution {}", appKeyType, targetInstitutionId);
            return "CREATED";
        } finally {
            Arrays.fill(clearKey, (byte) 0);
        }
    }
}
