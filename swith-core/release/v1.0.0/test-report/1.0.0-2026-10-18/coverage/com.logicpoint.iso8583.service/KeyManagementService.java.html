/*

 * @description: This is the service for the key management.
 * It is used to get the working key for a specific BIN and key type.

@Service

    /**
     * Get working key for a specific BIN and key type
     */
    @Cacheable(value = "keyManagementCache", key = "#binNumber + '_' + #keyType")
    public String getWorkingKeyForBinNumber(String binNumber, String keyType) {
        try {
            // First, find the bin_details record to get the bin_id
            BinDetails binDetails = binDetailsRepository.findByBinNumber(binNumber);
            if (binDetails == null) {
                logger.warn("BIN not found in bin_details table: {}", binNumber);
                return null;
            }
            
            Integer binId = binDetails.getBin_id().intValue();
            return getWorkingKeyForBinId(binId, keyType);
            
        } catch (Exception e) {
            logger.error("Error retrieving key for BIN number: {} and key type: {}", binNumber, keyType, e);
            return null;
        }
    }
    
    /**
     * Get working key for a specific bin_id and key type
     */
    public String getWorkingKeyForBinId(Integer binId, String keyType) {
        try {
            LocalDateTime now = LocalDateTime.now();
            
            List<KeyManagement> keys = keyManagementRepository
                .findCurrentActiveKeysByBinIdAndKeyType(binId, keyType, now);
            
            if (keys.isEmpty()) {
                logger.warn("No active key found for bin_id: {} and key type: {}", binId, keyType);
                return null;
            }
            
            KeyManagement key = keys.get(0); // Should be only one current active key
            return bytesToHex(key.getKey_value());
            
        } catch (Exception e) {
            logger.error("Error retrieving key for bin_id: {} and key type: {}", binId, keyType, e);
            return null;
        }
    }
    
    /**
     * Get network-based key (like ZPK)
     */
    @Cacheable(value = "keyManagementCache", key = "'NETWORK_' + #networkName + '_' + #keyType")
    public String getNetworkKey(String networkName, String keyType) {
        try {
            LocalDateTime now = LocalDateTime.now();
            
            List<KeyManagement> keys = keyManagementRepository
                .findCurrentActiveNetworkKeys(networkName, keyType, now);
            
            if (keys.isEmpty()) {
                logger.warn("No active network key found for network: {} and key type: {}", networkName, keyType);
                return null;
            }
            
            KeyManagement key = keys.get(0);
          
            return key.getIs_tr31() ? new String(key.getKey_value()) : bytesToHex(key.getKey_value()) ;
            
        } catch (Exception e) {
            logger.error("Error retrieving network key for network: {} and key type: {}", networkName, keyType, e);
            return null;
        }
    }
    
    /**
     * Get working key for a card number with fallback logic
     */
    public String getWorkingKeyForCard(String cardNumber, String keyType) {
        try {
            String binNumber = cardNumber.substring(0, 6);
            String networkName = determineNetwork(binNumber);

            if (isNetworkBasedKey(keyType)) {
                String sessionKey = getSessionScopedNetworkKey(networkName, keyType);
                if (sessionKey != null) {
                    logger.debug("Using session-scoped network key for network: {} and key type: {}", networkName, keyType);
                    return sessionKey;
                }
            }
            
            // Try BIN-specific key first
            String binKey = getCachedWorkingKeyForBinNumber(binNumber, keyType);
            if (binKey != null) {
                return binKey;
            }
            
            // Fallback to network key for network-based keys like ZPK
            if (isNetworkBasedKey(keyType)) {
                String networkKey = getCachedNetworkKey(networkName, keyType);
                if (networkKey != null) {
                    logger.debug("Using network key for card: {} and key type: {}", maskPan(cardNumber), keyType);
                    return networkKey;
                }
            }
            
            logger.warn("No key found for card: {} and key type: {}", maskPan(cardNumber), keyType);
            return null;
            
        } catch (Exception e) {
            logger.error("Error extracting BIN from card number: {}", cardNumber, e);
            return null;
        }
    }
    
    /**
     * Determine if a key type is network-based
     */
    private boolean isNetworkBasedKey(String keyType) {
        return "ZPK".equals(keyType) || "ZAK".equals(keyType) || "KEK".equals(keyType);
    }
    
    /**
     * Determine network from BIN
     */
    private String determineNetwork(String bin) {
        if (bin.startsWith("4")) return "VISA";
        if (bin.startsWith("5") || bin.startsWith("2")) return "MASTERCARD";
        if (bin.startsWith("6") || bin.startsWith("8")) return "RUPAY";
        return "UNKNOWN";
    }

    private String getCachedWorkingKeyForBinNumber(String binNumber, String keyType) {
        String cacheKey = binNumber + "_" + keyType;
        Cache cache = cacheManager.getCache("keyManagementCache");
        if (cache != null) {
            String cached = cache.get(cacheKey, String.class);
            if (cached != null) {
                return cached;
            }
        }

        String resolved = getWorkingKeyForBinNumber(binNumber, keyType);
        if (cache != null && resolved != null) {
            cache.put(cacheKey, resolved);
        }
        return resolved;
    }

    private String getCachedNetworkKey(String networkName, String keyType) {
        String cacheKey = "NETWORK_" + networkName + "_" + keyType;
        Cache cache = cacheManager.getCache("keyManagementCache");
        if (cache != null) {
            String cached = cache.get(cacheKey, String.class);
            if (cached != null) {
                return cached;
            }
        }

        String resolved = getNetworkKey(networkName, keyType);
        if (cache != null && resolved != null) {
            cache.put(cacheKey, resolved);
        }
        return resolved;
    }

    private String getSessionScopedNetworkKey(String networkName, String keyType) {
        NetworkSessionContextHolder.NetworkSessionContext sessionContext = NetworkSessionContextHolder.get();
        if (sessionContext == null) {
            return null;
        }

        String scheme = sessionContext.scheme();
        if (scheme == null || scheme.isBlank()) {
            scheme = networkName;
        }

        return networkSessionKeyService.getActiveKeyValue(
                sessionContext.siteId(),
                sessionContext.listenerId(),
                sessionContext.connectionId(),
                scheme,
                keyType
        );
    }

    public String getCurrentSessionNetworkKey(String scheme, String keyType) {
        NetworkSessionContextHolder.NetworkSessionContext sessionContext = NetworkSessionContextHolder.get();
        if (sessionContext == null) {
            logger.warn("No network session context available for key type: {}", keyType);
            return null;
        }

        String resolvedScheme = scheme;
        if (resolvedScheme == null || resolvedScheme.isBlank()) {
            resolvedScheme = sessionContext.scheme();
        }
        if (resolvedScheme == null || resolvedScheme.isBlank()) {
            logger.warn("No scheme available for session-scoped network key lookup. keyType={}", keyType);
            return null;
        }
        resolvedScheme = resolvedScheme.trim().toUpperCase();

        String sessionKey = null;
        if (sessionContext.connectionId() != null && !sessionContext.connectionId().isBlank()) {
            sessionKey = networkSessionKeyService.getActiveKeyValue(
                    sessionContext.siteId(),
                    sessionContext.listenerId(),
                    sessionContext.connectionId(),
                    resolvedScheme,
                    keyType
            );
        }
        if (sessionKey == null) {
            sessionKey = networkSessionKeyService.getActiveKeyValue(
                    sessionContext.siteId(),
                    sessionContext.listenerId(),
                    resolvedScheme,
                    keyType
            );
        }

        if (sessionKey == null) {
            logger.warn("No session-scoped network key found for site: {}, listener: {}, scheme: {}, key type: {}",
                    sessionContext.siteId(),
                    sessionContext.listenerId(),
                    resolvedScheme,
                    keyType);
        }
        return sessionKey;
    }
    
    /**
     * Load all active keys into cache on startup
     */
    @Cacheable("keyManagementCache")
    public Map<String, String> loadAllActiveKeys() {
        logger.info("Loading all active HSM keys into cache");
        Map<String, String> keyCache = new HashMap<>();
        
        try {
            LocalDateTime now = LocalDateTime.now();
            List<KeyManagement> activeKeys = keyManagementRepository
                .findAllCurrentActiveKeys(now);
            
            for (KeyManagement key : activeKeys) {
                String cacheKey;
                
                if ("BIN".equals(key.getKey_scope())) {
                    BinDetails binDetails = binDetailsRepository.findById(key.getBin_id().longValue()).orElse(null);
                    if (binDetails != null) {
                        cacheKey = binDetails.getBinNumber() + "_" + key.getKey_type();
                    } else {
                        continue;
                    }
                } else if ("NETWORK".equals(key.getKey_scope())) {
                    cacheKey = "NETWORK_" + key.getNetwork_name() + "_" + key.getKey_type();
                } else {
                    cacheKey = "INSTITUTION_" + key.getInstitution_id() + "_" + key.getKey_type();
                }
                
                String keyValue = bytesToHex(key.getKey_value());
                keyCache.put(cacheKey, keyValue);
                
                logger.debug("Cached key: {}", cacheKey);
            }
            
            logger.info("Loaded {} active keys into cache", activeKeys.size());
            
        } catch (Exception e) {
            logger.error("Error loading keys into cache", e);
        }
        
        return keyCache;
    }

    public KeyManagement getCurrentInstitutionKey(Integer institutionId, String keyType) {
        LocalDateTime now = LocalDateTime.now();
        List<KeyManagement> keys = keyManagementRepository
            .findCurrentActiveInstitutionKeys(institutionId, keyType, now);
        return keys.isEmpty() ? null : keys.get(0);
    }

    public List<KeyManagement> getActiveInstitutionKeyVersions(Integer institutionId, String keyType) {
        return keyManagementRepository.findActiveKeysByInstitutionIdAndKeyType(
            institutionId, keyType, "A", LocalDateTime.now());
    }

    public String getCurrentInstitutionKeyValue(Integer institutionId, String keyType) {
        KeyManagement key = getCurrentInstitutionKey(institutionId, keyType);
        if (key == null || key.getKey_value() == null || key.getKey_value().length == 0) {
            return null;
        }

        return Boolean.TRUE.equals(key.getIs_tr31())
                ? new String(key.getKey_value(), StandardCharsets.US_ASCII)
                : bytesToHex(key.getKey_value());
    }

    public KeyManagement getInstitutionKeyByVersion(Integer institutionId, String keyType, Integer keyVersion) {
        if (institutionId == null || keyType == null || keyVersion == null) {
            return null;
        }

        return getActiveInstitutionKeyVersions(institutionId, keyType).stream()
            .filter(key -> keyVersion.equals(key.getKey_version()))
            .findFirst()
            .orElse(null);
    }

    @Transactional
    public KeyManagement saveInstitutionWrappedKey(Integer institutionId,
                                                   String keyType,
                                                   String wrappedKey,
                                                   String kcv,
                                                   String algorithm,
                                                   String wrappingKeyRef,
                                                   String createdBy) {
        LocalDateTime now = LocalDateTime.now();
        List<KeyManagement> currentKeys = keyManagementRepository.findCurrentActiveInstitutionKeys(institutionId, keyType, now);
        Integer nextVersion = 1;

        for (KeyManagement currentKey : currentKeys) {
            currentKey.setIs_current(false);
            currentKey.setUpdated_at(now);
            keyManagementRepository.save(currentKey);
            if (currentKey.getKey_version() != null && currentKey.getKey_version() >= nextVersion) {
                nextVersion = currentKey.getKey_version() + 1;
            }
        }

        KeyManagement key = new KeyManagement();
        key.setInstitution_id(institutionId);
        key.setKey_type(keyType);
        key.setKey_scope("INSTITUTION");
        key.setKey_status("A");
        key.setIs_current(true);
        key.setKey_version(nextVersion);
        key.setKey_value(wrappedKey.getBytes(StandardCharsets.US_ASCII));
        key.setKey_check_value(kcv);
        key.setIs_tr31(true);
        key.setValid_from(now);
        key.setValid_to(now.plusYears(2));
        key.setCreated_at(now);
        key.setUpdated_at(now);
        key.setCreated_by(createdBy);
        key.setAlgorithm(algorithm);
        key.setWrapping_key_ref(wrappingKeyRef);
        return keyManagementRepository.save(key);
    }
    
    /**
     * Rotate key (deactivate current, activate new)
     */
    @Transactional
    public boolean  rotateKey(String keyType, String scheme, String keyStatus, String newKey, String kcv) {
        // Deactivate current key
        List<KeyManagement> currentKey = keyManagementRepository.findByKeyTypeAndNetworkNameAndKeyStatus(keyType, scheme, "A");
        if (currentKey.isEmpty()) {
            logger.warn("No active key found for key type: {} and scheme: {}", keyType, scheme);
            return false;
        }
        KeyManagement currentKeyEntity = currentKey.get(0);
        currentKeyEntity.setKey_status("I");
        currentKeyEntity.setUpdated_at(LocalDateTime.now());
        keyManagementRepository.save(currentKeyEntity);
        
        // Activate new key
        KeyManagement newKeyEntity = new KeyManagement();
        newKeyEntity.setInstitution_id(currentKey.get(0).getInstitution_id());
        newKeyEntity.setKey_type(keyType);
        newKeyEntity.setKey_status("A");
        newKeyEntity.setIs_current(true);
        newKeyEntity.setKey_scope("NETWORK");
        newKeyEntity.setKey_version(currentKeyEntity.getKey_version()!=null ? currentKeyEntity.getKey_version() + 1 : 1);
        newKeyEntity.setNetwork_name(scheme);
        newKeyEntity.setKey_value(newKey.getBytes());
        newKeyEntity.setKey_check_value(kcv);
        newKeyEntity.setKey_value_tr31(newKey.getBytes());
        newKeyEntity.setKey_check_value_tr31(kcv);
        newKeyEntity.setUpdated_at(LocalDateTime.now());
        newKeyEntity.setCreated_at(LocalDateTime.now());
        newKeyEntity.setCreated_by("SYSTEM-DKE");
        newKeyEntity.setValid_from(LocalDateTime.now());
        newKeyEntity.setValid_to(LocalDateTime.now().plusYears(10));
        newKeyEntity.setIs_tr31(true);
        keyManagementRepository.save(newKeyEntity);
        
        // Refresh cache
        refreshKeyCache();
        return true;
    }

    public boolean rotateSessionNetworkKey(String keyType,
                                           String scheme,
                                           String siteId,
                                           String listenerId,
                                           String connectionId,
                                           String newKey,
                                           String kcv) {
        return networkSessionKeyService.rotateSessionKey(
                keyType,
                scheme,
                siteId,
                listenerId,
                connectionId,
                newKey,
                kcv,
                "SYSTEM-DKE"
        );
    }
    
    /**
     * Refresh the key management cache
     * Clears the existing cache and reloads all active keys
     */
    @CacheEvict(value = "keyManagementCache", allEntries = true)
    public void refreshKeyCache() {
        logger.info("Refreshing key management cache");
        try {
            // Clear the cache
            var cache = cacheManager.getCache("keyManagementCache");
            if (cache != null) {
                cache.clear();
                logger.debug("Cleared keyManagementCache");
            }
            
            // Reload all active keys into cache by calling the loadAllActiveKeys method
            // This will repopulate the cache with the latest data from the database
            loadAllActiveKeys();
            logger.info("Key management cache refreshed successfully");
            
        } catch (Exception e) {
            logger.error("Error refreshing key management cache", e);
            throw new RuntimeException("Failed to refresh key management cache", e);
        }
    }
    
    private String bytesToHex(byte[] bytes) {
        StringBuilder result = new StringBuilder();
        for (byte b : bytes) {
            result.append(String.format("%02X", b));
        }
        return result.toString();
    }
    
    private String maskPan(String pan) {
        if (pan == null || pan.length() < 8) {
            return "****";
        }
        return pan.substring(0, 4) + "****" + pan.substring(pan.length() - 4);
    }
