package com.logicpoint.iso8583.service;

import com.logicpoint.iso8583.emv.CardVerificationResult;
import com.logicpoint.iso8583.emv.ChipData;
import com.logicpoint.iso8583.emv.IssuerApplicationData;
import com.logicpoint.iso8583.util.Scheme;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.stereotype.Service;

import java.util.List;
import java.util.ArrayList;
import java.util.Locale;

/**
 * Service for Card Verification Result (CVR) validation and processing.
 * CVR is extracted from Issuer Application Data (IAD, tag 9F10).
 * 
 * @author Logic Point Systems
 * @version 1.0.0
 */
@Service
public class CvrValidationService {

    private static final Logger logger = LoggerFactory.getLogger(CvrValidationService.class);
    
    private static final String TAG_9F10 = "9F10"; // Issuer Application Data

    /**
     * Validates CVR from chip data
     * 
     * @param chipData EMV chip data containing IAD
     * @param scheme Payment scheme (RUPAY, VISA, MASTERCARD)
     * @return CvrValidationResult
     */
    public CvrValidationResult validateCvr(ChipData chipData, Scheme scheme) {
        logger.debug("Starting CVR validation for scheme: {}", scheme);
        
        if (chipData == null || chipData.isEmpty()) {
            logger.warn("Chip data is null or empty, CVR validation failed");
            return CvrValidationResult.failed("No chip data available for CVR validation");
        }

        // Extract IAD from tag 9F10
        String iadHex = chipData.getTagValue(TAG_9F10);
        if (iadHex == null || iadHex.isEmpty()) {
            logger.warn("IAD (9F10) not found in chip data");
            return CvrValidationResult.failed("IAD not found in chip data",List.of("CVR_VALIDATION_MANDATORY_MISSING"));
        }

        try {
            // Parse IAD to get format and extract CVR
            IssuerApplicationData iad = new IssuerApplicationData(iadHex, scheme);
            String cvrHex = iad.getCardVerificationResults();
            
            if (cvrHex == null || cvrHex.isEmpty()) {
                logger.warn("CVR not found in IAD");
                return CvrValidationResult.failed("CVR not found in IAD",List.of("CVR_VALIDATION_MANDATORY_MISSING"));
            }

            // Parse CVR
            CardVerificationResult cvr = new CardVerificationResult(cvrHex, iad.getFormat());
            
            // Perform comprehensive validation
            return performCvrValidation(cvr);
            
        } catch (IllegalArgumentException e) {
            logger.error("Invalid IAD or CVR format: {}", e.getMessage(), e);
            return CvrValidationResult.failed("Invalid IAD or CVR format: " + e.getMessage());
        } catch (Exception e) {
            logger.error("Error parsing CVR: {}", e.getMessage(), e);
            return CvrValidationResult.failed("Error parsing CVR: " + e.getMessage());
        }
    }

    /**
     * Performs comprehensive CVR validation
     */
    private CvrValidationResult performCvrValidation(CardVerificationResult cvr) {
        List<String> errors = new ArrayList<>();
        List<String> warnings = new ArrayList<>();

        logger.debug("CVR Format: {}, Value: {}", cvr.getIadFormat(), cvr.getHexCvr());

        // Critical failures - transaction must be declined
        if (cvr.pinTryLimitExceeded()) {
            logger.error("PIN try limit exceeded according to CVR");
            errors.add("PIN_TRY_LIMIT_EXCEEDED: PIN try limit exceeded");
            return CvrValidationResult.failed("PIN try limit exceeded", errors);
        }

        if (cvr.issuerAuthenticationFailed()) {
            logger.error("Issuer authentication failed according to CVR");
            errors.add("ISSUER_AUTH_FAILED: Issuer authentication failed");
            return CvrValidationResult.failed("Issuer authentication failed", errors);
        }

        // Offline PIN verification failed
        if (cvr.offlinePinVerificationFailed() && cvr.offlinePinVerificationPerformed()) {
            logger.error("Offline PIN verification failed");
            errors.add("OFFLINE_PIN_FAILED: Offline PIN verification failed");
            return CvrValidationResult.failed("Offline PIN verification failed", errors);
        }

        // Check PIN try counter
        int pinTryCounter = cvr.getPinTryCounter();
        if (pinTryCounter >= 0 && pinTryCounter < 3) {
            warnings.add(String.format("LOW_PIN_TRIES: Only %d PIN tries remaining", pinTryCounter));
        }

        // Offline limits exceeded - require online authorization
        if (cvr.upperOfflineTransactionCountLimitExceeded()) {
            warnings.add("UPPER_OFFLINE_COUNT_EXCEEDED: Upper offline transaction count limit exceeded - online auth required");
        }

        if (cvr.upperCumulativeOfflineAmountLimitExceeded()) {
            warnings.add("UPPER_OFFLINE_AMOUNT_EXCEEDED: Upper cumulative offline amount limit exceeded - online auth required");
        }

        if (cvr.lowerOfflineTransactionCountLimitExceeded()) {
            warnings.add("LOWER_OFFLINE_COUNT_EXCEEDED: Lower offline transaction count limit exceeded");
        }

        if (cvr.lowerCumulativeOfflineAmountLimitExceeded()) {
            warnings.add("LOWER_OFFLINE_AMOUNT_EXCEEDED: Lower cumulative offline amount limit exceeded");
        }

        // Last online transaction not completed
        if (cvr.lastOnlineTransactionNotCompleted()) {
            warnings.add("LAST_ONLINE_TXN_INCOMPLETE: Last online transaction was not completed");
        }

        // Issuer script processing failed
        if (cvr.issuerScriptProcessingFailed()) {
            warnings.add("ISSUER_SCRIPT_FAILED: Issuer script processing failed");
        }

        // Go online on next transaction
        if (cvr.goOnlineOnNextTransactionSet()) {
            warnings.add("GO_ONLINE_NEXT_TXN: Card requires online authorization on next transaction");
        }

        // Unable to go online
        if (cvr.unableToGoOnline()) {
            warnings.add("UNABLE_TO_GO_ONLINE: Card unable to go online");
        }

        // Offline data authentication failed on previous transaction
        if (cvr.offlineDataAuthFailedOnPreviousTransaction()) {
            warnings.add("ODA_FAILED_PREVIOUS: Offline data authentication failed on previous transaction");
        }

        boolean requiresOnline = cvr.requiresOnlineAuthorization();

        if (!errors.isEmpty()) {
            logger.warn("CVR validation failed with errors: {}", errors);
            return CvrValidationResult.failed("CVR validation failed", errors);
        }

        if (!warnings.isEmpty()) {
            logger.info("CVR validation passed with warnings: {}", warnings);
            return CvrValidationResult.successfulWithWarnings(
                "CVR validation passed but requires online authorization", warnings, requiresOnline);
        }

        logger.debug("CVR validation passed successfully");
        return CvrValidationResult.successful("CVR validation passed", requiresOnline);
    }

    /**
     * Extracts and parses CVR from IAD
     */
    public CardVerificationResult parseCvrFromIad(String iadHex, String scheme) {
        Scheme parsedScheme = Scheme.valueOf(scheme.trim().toUpperCase(Locale.ROOT));
        IssuerApplicationData iad = new IssuerApplicationData(iadHex, parsedScheme);
        String cvrHex = iad.getCardVerificationResults();
        return new CardVerificationResult(cvrHex, iad.getFormat());
    }

    /**
     * Result class for CVR validation
     */
    public static class CvrValidationResult {
        private final boolean valid;
        private final String message;
        private final List<String> errors;
        private final List<String> warnings;
        private final boolean requiresOnlineAuthorization;

        private CvrValidationResult(boolean valid, String message, List<String> errors, 
                                   List<String> warnings, boolean requiresOnline) {
            this.valid = valid;
            this.message = message;
            this.errors = errors != null ? new ArrayList<>(errors) : new ArrayList<>();
            this.warnings = warnings != null ? new ArrayList<>(warnings) : new ArrayList<>();
            this.requiresOnlineAuthorization = requiresOnline;
        }

        public static CvrValidationResult successful(String message) {
            return new CvrValidationResult(true, message, null, null, false);
        }

        public static CvrValidationResult successful(String message, boolean requiresOnline) {
            return new CvrValidationResult(true, message, null, null, requiresOnline);
        }

        public static CvrValidationResult successfulWithWarnings(String message, 
                                                                 List<String> warnings, 
                                                                 boolean requiresOnline) {
            return new CvrValidationResult(true, message, null, warnings, requiresOnline);
        }

        public static CvrValidationResult failed(String message) {
            return new CvrValidationResult(false, message, null, null, false);
        }

        public static CvrValidationResult failed(String message, List<String> errors) {
            return new CvrValidationResult(false, message, errors, null, false);
        }

        // Getters
        public boolean isValid() { return valid; }
        public String getMessage() { return message; }
        public List<String> getErrors() { return new ArrayList<>(errors); }
        public List<String> getWarnings() { return new ArrayList<>(warnings); }
        public boolean requiresOnlineAuthorization() { return requiresOnlineAuthorization; }

        @Override
        public String toString() {
            return String.format("CvrValidationResult{valid=%s, message='%s', errors=%s, warnings=%s, requiresOnline=%s}",
                    valid, message, errors, warnings, requiresOnlineAuthorization);
        }
    }
}
