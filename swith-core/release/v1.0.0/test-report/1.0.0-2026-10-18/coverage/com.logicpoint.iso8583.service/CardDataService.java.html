package com.logicpoint.iso8583.service;

import com.logicpoint.iso8583.context.RequestContext;
import com.logicpoint.iso8583.entity.CardData;
import com.logicpoint.iso8583.repository.CardDataRepository;
import com.logicpoint.iso8583.validation.orchestrator.CardDataContext;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

/**
 * Service for fetching and managing card data.
 * Provides centralized access to card information with caching support.
 * 
 * @author LogicPoint Systems
 * @version 1.0
 * @since 2025-01-27
 */
@Service
public class CardDataService {
    @Autowired
    private com.logicpoint.iso8583.cmsintegration.CmsAuthorizationLookup cmsLookup;
    
    private static final Logger logger = LoggerFactory.getLogger(CardDataService.class);
    
    private final CardDataRepository cardDataRepository;
    private final CardDataEncryptionService cardDataEncryptionService;
    private final CacheInitializationService cacheInitializationService;
    
    @Autowired
    public CardDataService(CardDataRepository cardDataRepository,
                           CardDataEncryptionService cardDataEncryptionService,
                           CacheInitializationService cacheInitializationService) {
        this.cardDataRepository = cardDataRepository;
        this.cardDataEncryptionService = cardDataEncryptionService;
        this.cacheInitializationService = cacheInitializationService;
    }
    
    /**
     * Fetches card data by PAN (Primary Account Number).
     * Uses Spring Cache for performance optimization.
     * 
     * @param pan The card number (PAN)
     * @return CardDataContext containing the card data or error information
     */
    //@Cacheable(value = "cardData", key = "#pan", unless = "#result == null")
    public CardDataContext fetchCardData(String pan) {
        try {
            logger.debug("Fetching card data for PAN: {}", maskPan(pan));
            
            if (pan == null || pan.trim().isEmpty()) {
                logger.warn("Invalid PAN provided: {}", pan);
                return CardDataContext.failure("INVALID_PAN: PAN is required");
            }
            
            Integer institutionId = resolveInstitutionId(pan);
            CardData cardData = findCardData(pan, institutionId);
            
            if (cardData == null) {
                logger.warn("Card not found in database for PAN: {}", maskPan(pan));
                return CardDataContext.failure("CARD_NOT_FOUND: Card not found in database");
            }

            cardDataEncryptionService.hydrateDecryptedFields(cardData, institutionId);
            
            logger.debug("Successfully fetched card data for PAN: {}, Status: {}", 
                        maskPan(pan), cardData.getStatus());
            
            return CardDataContext.success(cardData);
            
        } catch (Exception e) {
            logger.error("Error fetching card data for PAN: {}: {}", maskPan(pan), e.getMessage(), e);
            return CardDataContext.failure("DATABASE_ERROR: " + e.getMessage());
        }
    }

    public CardDataContext fetchCardData(RequestContext requestContext) {
        try {
            if (requestContext == null) {
                return CardDataContext.failure("INVALID_REQUEST_CONTEXT: Request context is required");
            }

            String pan = requestContext.getCardNumber();
            logger.debug("Fetching card data using protected PAN lookup for PAN: {}", maskPan(pan));

            if (pan == null || pan.trim().isEmpty()) {
                logger.warn("Invalid PAN provided in request context");
                return CardDataContext.failure("INVALID_PAN: PAN is required");
            }

            Integer institutionId = parseInstitutionId(requestContext.getInstitutionId());
            String panLookupHmac = requestContext.getPanHash();
            if ((panLookupHmac == null || panLookupHmac.isBlank()) && institutionId != null) {
                panLookupHmac = cardDataEncryptionService.computePanLookupHmac(pan, institutionId);
                requestContext.setPanHash(panLookupHmac);
            }

            if (panLookupHmac == null || panLookupHmac.isBlank()) {
                logger.warn("PAN lookup HMAC could not be resolved for PAN: {}", maskPan(pan));
                return CardDataContext.failure("CARD_LOOKUP_UNAVAILABLE: PAN lookup hash is not available");
            }

            CardData managed = cmsLookup.lookup(requestContext, panLookupHmac, institutionId);
            if (managed != null) return CardDataContext.success(managed);
            CardData cardData = cardDataRepository.findByPanLookupHmac(panLookupHmac);
            if (cardData == null) {
                logger.warn("Card not found in database for PAN: {}", maskPan(pan));
                return CardDataContext.failure("CARD_NOT_FOUND: Card not found in database");
            }

            cardDataEncryptionService.hydrateDecryptedFields(cardData, institutionId);
            logger.debug("Successfully fetched card data for PAN: {}, Status: {}", maskPan(pan), cardData.getStatus());
            return CardDataContext.success(cardData);
        } catch (Exception e) {
            requestContext.setInternalResponseCode(com.logicpoint.iso8583.responsecode.InternalResCode.RESTRICTED_CARD);
            logger.warn("Protected card lookup rejected: {}", e.getClass().getSimpleName());
            return CardDataContext.failure("CARD_RESTRICTED: Card lookup or authorization policy failed");
        }
    }
    
    /**
     * Fetches card data by PAN without caching (for real-time data).
     * 
     * @param pan The card number (PAN)
     * @return CardDataContext containing the card data or error information
     */
    public CardDataContext fetchCardDataRealTime(String pan) {
        try {
            logger.debug("Fetching real-time card data for PAN: {}", maskPan(pan));
            
            if (pan == null || pan.trim().isEmpty()) {
                logger.warn("Invalid PAN provided: {}", pan);
                return CardDataContext.failure("INVALID_PAN: PAN is required");
            }
            
            Integer institutionId = resolveInstitutionId(pan);
            CardData cardData = findCardData(pan, institutionId);
            
            if (cardData == null) {
                logger.warn("Card not found in database for PAN: {}", maskPan(pan));
                return CardDataContext.failure("CARD_NOT_FOUND: Card not found in database");
            }

            cardDataEncryptionService.hydrateDecryptedFields(cardData, institutionId);
            
            logger.debug("Successfully fetched real-time card data for PAN: {}, Status: {}", 
                        maskPan(pan), cardData.getStatus());
            
            return CardDataContext.success(cardData);
            
        } catch (Exception e) {
            logger.error("Error fetching real-time card data for PAN: {}: {}", maskPan(pan), e.getMessage(), e);
            return CardDataContext.failure("DATABASE_ERROR: " + e.getMessage());
        }
    }
    
    /**
     * Validates if the fetched card data is suitable for transaction processing.
     * 
     * @param cardContext The card data context to validate
     * @return true if card data is valid for transactions, false otherwise
     */
    public boolean isCardDataValidForTransaction(CardDataContext cardContext) {
        if (!cardContext.isValid()) {
            logger.warn("Card data context is invalid: {}", cardContext.getFetchError());
            return false;
        }
        
        CardData cardData = cardContext.getCardData();
        
        // Check if card status is active
        if (!"ACTIVE".equalsIgnoreCase(cardData.getStatus())) {
            logger.warn("Card is not active. Status: {}", cardData.getStatus());
            return false;
        }
        
        // Check if card is not expired
        if (isCardExpired(cardData.getExpiryDate())) {
            logger.warn("Card has expired. Expiry date: {}", cardData.getExpiryDate());
            return false;
        }
        
        // Check if PIN is not blocked - using reflection to access the field
        try {
            java.lang.reflect.Field pinBlockedField = CardData.class.getDeclaredField("pin_blocked");
            pinBlockedField.setAccessible(true);
            Boolean pinBlocked = (Boolean) pinBlockedField.get(cardData);
            
            if (Boolean.TRUE.equals(pinBlocked)) {
                logger.warn("Card PIN is blocked");
                return false;
            }
        } catch (Exception e) {
            logger.warn("Could not check PIN blocked status: {}", e.getMessage());
        }
        
        logger.debug("Card data validation successful for transaction processing");
        return true;
    }
    
    /**
     * Checks if a card has expired.
     * 
     * @param expiryDate The expiry date in MMYY format
     * @return true if expired, false otherwise
     */
    private boolean isCardExpired(String expiryDate) {
        try {
            if (expiryDate == null || expiryDate.length() != 4) {
                return true; // Consider invalid format as expired
            }
            
            String year = expiryDate.substring(0, 2);
            String month = expiryDate.substring(2);
            
            int expMonth = Integer.parseInt(month);
            int expYear = Integer.parseInt(year);
            
            if (expMonth < 1 || expMonth > 12) {
                return true; // Invalid month
            }
            
            // Get current date
            java.time.LocalDate currentDate = java.time.LocalDate.now();
            int currentYear = currentDate.getYear() % 100; // Last 2 digits
            int currentMonth = currentDate.getMonthValue();
            
            // Compare year first, then month
            if (expYear < currentYear || (expYear == currentYear && expMonth < currentMonth)) {
                return true;
            }
            
            return false;
            
        } catch (Exception e) {
            logger.error("Error checking card expiry: {}", e.getMessage());
            return true; // Consider as expired if error occurs
        }
    }
    
    /**
     * Masks PAN for logging purposes.
     * 
     * @param pan The PAN to mask
     * @return Masked PAN
     */
    private String maskPan(String pan) {
        if (pan == null || pan.length() < 8) {
            return "****";
        }
        return pan.substring(0, 4) + "****" + pan.substring(pan.length() - 4);
    }

    private CardData findCardData(String pan, Integer institutionId) {
        String lookup = cardDataEncryptionService.computePanLookupHmac(pan, institutionId);
        if (lookup != null && cmsLookup.isManaged(lookup)) {
            throw new IllegalStateException("CMS cards require transaction evidence lookup");
        }
        CardData cardData = cardDataRepository.findByPan(pan);
        if (cardData != null) {
            return cardData;
        }

        String panLookupHmac = cardDataEncryptionService.computePanLookupHmac(pan, institutionId);
        if (panLookupHmac == null) {
            return null;
        }

        return cardDataRepository.findByPanLookupHmac(panLookupHmac);
    }

    private Integer resolveInstitutionId(String pan) {
        if (pan == null || pan.length() < 6) {
            return null;
        }
        CacheInitializationService.BinInstitutionData binData = cacheInitializationService
                .loadBinInstitutionCache()
                .get(pan.substring(0, 6));
        if (binData == null || binData.getBinDetails() == null) {
            return null;
        }
        return binData.getBinDetails().getInstitution_id();
    }

    private Integer parseInstitutionId(String institutionId) {
        if (institutionId == null || institutionId.isBlank()) {
            return null;
        }
        try {
            return Integer.valueOf(institutionId);
        } catch (NumberFormatException ex) {
            return null;
        }
    }

    /**
     * Updates the card data in the data store.
     * This method should be called when card attributes (e.g., PIN retry count, block status, expiry, etc.)
     * need to be updated after a transaction or administrative action.
     *
     * @param cardData The CardData object containing updated card information
     * @return true if update was successful, false otherwise
     */
    public boolean updateCardData(CardData cardData) {
        try {
            if (cardData == null) {
                logger.warn("Cannot update card data: CardData is null");
                return false;
            }
            // Example: If using Spring Data JPA repository
            // cardDataRepository.save(cardData);

            // If using a custom DAO or another persistence mechanism, implement accordingly.
            // For demonstration, assume a repository exists:
            if (cardDataRepository != null) {
                if (cmsLookup.isManaged(cardData.getAuth_id())) {
                    cmsLookup.updatePinCounters(cardData);
                    return true;
                }
                cardDataEncryptionService.prepareForPersistence(cardData, resolveInstitutionId(cardData.getPan()));
                cardDataRepository.save(cardData);
                logger.info("Card data updated successfully for PAN: {}", maskPan(cardData.getPan()));
                return true;
            } else {
                logger.error("CardDataRepository is not initialized. Cannot update card data for PAN: {}", maskPan(cardData.getPan()));
                return false;
            }
        } catch (Exception e) {
            logger.error("Error updating card data for PAN {}: {}", 
                cardData != null ? maskPan(cardData.getPan()) : "null", e.getMessage());
            return false;
        }
    }
}
