package com.logicpoint.iso8583.service;

import com.logicpoint.iso8583.context.RequestContext;

import com.logicpoint.iso8583.entity.Transaction;

import com.logicpoint.iso8583.responsecode.InternalResCode;
import com.logicpoint.iso8583.validation.TransactionType;
import com.logicpoint.iso8583.validation.ValidationResult;
import java.time.LocalDateTime;
import java.time.ZoneId;

//import org.hibernate.engine.jdbc.env.internal.LobCreationLogging_.logger;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Service;

@Service
public class PrepaidLimitValidationService {

    private static final Logger log = LoggerFactory.getLogger(PrepaidLimitValidationService.class);
    private static final String ACTIVE = "ACTIVE";

    private final TransactionLimitsRepository limitsRepository;
    private final TransactionRepository transactionRepository;
    private final CacheInitializationService cacheInitializationService;
    private final boolean enabled;
    private final boolean cashWithdrawalEnabled;
    private final int cashWithdrawalCount;
    private final long cashWithdrawalWindowMinutes;
    private final ZoneId periodZone;

    public PrepaidLimitValidationService(
            TransactionLimitsRepository limitsRepository,
            TransactionRepository transactionRepository,
            CacheInitializationService cacheInitializationService,
            @Value("${prepaid-limits.enabled:true}") boolean enabled,
            @Value("${prepaid-limits.cash-withdrawal.enabled:true}") boolean cashWithdrawalEnabled,
            @Value("${prepaid-limits.cash-withdrawal.count:5}") int cashWithdrawalCount,
            @Value("${prepaid-limits.cash-withdrawal.window-minutes:60}") long cashWithdrawalWindowMinutes,
            @Value("${prepaid-limits.time-zone:Asia/Kolkata}") String timeZone) {
        this.limitsRepository = limitsRepository;
        this.transactionRepository = transactionRepository;
        this.cacheInitializationService = cacheInitializationService;
        this.enabled = enabled;
        this.cashWithdrawalEnabled = cashWithdrawalEnabled;
        this.cashWithdrawalCount = cashWithdrawalCount;
        this.cashWithdrawalWindowMinutes = cashWithdrawalWindowMinutes;
        this.periodZone = ZoneId.of(timeZone);
    }

    /**
     * Temporary single-instance limit enforcement for certification.
     */
    public synchronized ValidationResult validate(
            RequestContext requestContext,
            CardData cardData) {
        if (!enabled
                || cardData == null
                || !Character.valueOf('P').equals(cardData.getCard_type())
                || requestContext.getTransactionType() != TransactionType.CASH_WITHDRAWAL) {
            return ValidationResult.builder().build();
        }
        if (!cashWithdrawalEnabled) {
            log.info("Prepaid cash withdrawal limit validation disabled by configuration");
            return ValidationResult.builder().build();
        }

        if (cashWithdrawalCount <= 0 || cashWithdrawalWindowMinutes <= 0) {
            log.warn("Prepaid cash withdrawal count/window configuration is invalid. count={}, windowMinutes={}",
                    cashWithdrawalCount, cashWithdrawalWindowMinutes);
            return ValidationResult.builder().build();
        }

        if (requestContext.getPanHash() == null || requestContext.getPanHash().isBlank()) {
            requestContext.setInternalResponseCode(InternalResCode.SYSTEM_ERROR);
            return ValidationResult.builder()
                    .addError("PREPAID_LIMIT_CHECK_UNAVAILABLE: PAN lookup hash is missing")
                    .build();
        }

        // Transaction.createdAt is persisted with LocalDateTime.now(), so use the
        // same JVM clock here. A rolling duration does not require a calendar zone.
        LocalDateTime periodEnd = LocalDateTime.now();
        LocalDateTime periodStart = periodEnd.minusMinutes(cashWithdrawalWindowMinutes);
        long successfulCount = transactionRepository.countSuccessfulCashWithdrawalsInWindow(
                requestContext.getPanHash(),
                Transaction.Status.SUCCESS.name(),
                periodStart,
                periodEnd);
        log.debug("Prepaid cash withdrawal count in last {} minutes: {} / {}",
                cashWithdrawalWindowMinutes, successfulCount, cashWithdrawalCount);

        if (successfulCount >= cashWithdrawalCount) {
            return decline(requestContext, "PREPAID_MONTHLY_LIMIT_EXCEEDED");
        }
        return ValidationResult.builder().build();
    }

    private ValidationResult decline(RequestContext requestContext, String reason) {
        requestContext.setInternalResponseCode(InternalResCode.EXCEEDS_LIMIT);
        return ValidationResult.builder().addError(reason).build();
    }
}
