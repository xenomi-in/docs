package com.logicpoint.iso8583.service;

import com.logicpoint.iso8583.context.RequestContext;
import io.micrometer.core.instrument.Counter;
import io.micrometer.core.instrument.Gauge;
import io.micrometer.core.instrument.MeterRegistry;
import org.springframework.stereotype.Component;

import java.time.Instant;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.ConcurrentMap;
import java.util.concurrent.atomic.AtomicInteger;
import java.util.concurrent.atomic.AtomicLong;

/** Low-overhead, in-memory metrics recorded once per final transaction outcome. */
@Component
public class TransactionOutcomeMetrics {

    private static final String ENDPOINT_SWITCH = "switch";
    private final MeterRegistry registry;
    private final ConcurrentMap<MetricKey, Counter> counters = new ConcurrentHashMap<>();
    private final AtomicInteger inFlight = new AtomicInteger();
    private final AtomicLong lastTransactionEpochSeconds = new AtomicLong();

    public TransactionOutcomeMetrics(MeterRegistry registry) {
        this.registry = registry;
        Gauge.builder("switch_transactions_in_flight", inFlight, AtomicInteger::get)
                .description("Transactions currently being processed")
                .register(registry);
        Gauge.builder("switch_last_transaction_timestamp_seconds",
                        lastTransactionEpochSeconds, AtomicLong::get)
                .description("Unix timestamp of the last completed transaction")
                .register(registry);
    }

    public void started() {
        inFlight.incrementAndGet();
    }

    public void completed(RequestContext context, String result, String responseCode) {
        MetricKey key = new MetricKey(
                normalizeResult(result),
                normalizeChannel(context),
                normalizeTransactionType(context),
                normalizeNetwork(context),
                ENDPOINT_SWITCH,
                normalizeResponseCode(responseCode));
        counters.computeIfAbsent(key, this::registerCounter).increment();
        lastTransactionEpochSeconds.set(Instant.now().getEpochSecond());
    }

    public void finished() {
        inFlight.updateAndGet(current -> Math.max(0, current - 1));
    }

    private Counter registerCounter(MetricKey key) {
        return Counter.builder("switch_transactions_total")
                .description("Final ISO 8583 transaction outcomes")
                .tags("result", key.result,
                        "channel", key.channel,
                        "transaction_type", key.transactionType,
                        "network", key.network,
                        "endpoint", key.endpoint,
                        "response_code", key.responseCode)
                .register(registry);
    }

    private String normalizeResult(String result) {
        return switch (result == null ? "" : result.toLowerCase()) {
            case "success", "declined", "timeout", "error", "rejected" -> result.toLowerCase();
            default -> "error";
        };
    }

    private String normalizeResponseCode(String responseCode) {
        return responseCode != null && responseCode.matches("[A-Za-z0-9]{2,3}")
                ? responseCode.toUpperCase()
                : "none";
    }

    private String normalizeTransactionType(RequestContext context) {
        var type = context == null ? null : context.getTransactionType();
        return type == null ? "unknown" : type.name().toLowerCase();
    }

    private String normalizeNetwork(RequestContext context) {
        return context == null || context.getScheme() == null
                ? "unknown"
                : context.getScheme().name().toLowerCase();
    }

    private String normalizeChannel(RequestContext context) {
        return context == null || context.getChannel() == null
                ? "unknown"
                : context.getChannel().name().toLowerCase();
    }

    private record MetricKey(String result, String channel, String transactionType,
                             String network, String endpoint, String responseCode) { }
}
