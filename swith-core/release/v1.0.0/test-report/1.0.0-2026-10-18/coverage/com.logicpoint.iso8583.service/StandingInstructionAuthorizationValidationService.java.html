package com.logicpoint.iso8583.service;

import com.logicpoint.iso8583.config.StandingInstructionAuthorizationProperties;
import com.logicpoint.iso8583.context.RequestContext;
import com.logicpoint.iso8583.util.Scheme;
import com.logicpoint.iso8583.validation.TransactionType;
import java.math.BigDecimal;
import java.time.LocalDate;
import java.time.format.DateTimeFormatter;
import java.time.format.DateTimeParseException;
import java.time.format.ResolverStyle;
import java.util.ArrayList;
import java.util.List;
import java.util.Map;
import org.springframework.stereotype.Service;

/**
 * Validates standing-instruction mandate data carried in ISO8583 authorization TLV fields.
 */
@Service
public class StandingInstructionAuthorizationValidationService {

    public static final String ATTR_SI_REQUEST_TYPE = "si_request_type";
    public static final String ATTR_SI_MANDATE_ID = "si_mandate_id";

    private static final DateTimeFormatter SI_DATE_FORMAT =
            DateTimeFormatter.ofPattern("ddMMuuuu").withResolverStyle(ResolverStyle.STRICT);

    private final StandingInstructionAuthorizationProperties properties;
    private final StandingInstructionMandateService mandateService;

    public StandingInstructionAuthorizationValidationService(StandingInstructionAuthorizationProperties properties,
                                                            StandingInstructionMandateService mandateService) {
        this.properties = properties;
        this.mandateService = mandateService;
    }

    public List<String> validate(RequestContext requestContext, Scheme scheme, TransactionType transactionType) {
        List<String> errors = new ArrayList<>();
        if (!properties.isEnabled() || requestContext == null || scheme == null || transactionType == null) {
            return errors;
        }

        StandingInstructionAuthorizationProperties.SchemeRule rule = properties.getRule(scheme.name());
        if (rule == null || !rule.isEnabled() || !isStandingInstruction(transactionType)) {
            return errors;
        }

        Map<String, String> tags;
        try {
            tags = requestContext.getParsedTlv(rule.getDataField());
        } catch (Exception ex) {
            errors.add("INVALID_SI_DATA_FORMAT: Unable to parse DE" + rule.getDataField()
                    + " for standing-instruction data");
            return errors;
        }

        String mandateId = extractMandateId(requestContext, tags, rule);
        ParsedSiData siData = validateSiData(requestContext, tags, rule, mandateId, errors);
        if (errors.isEmpty() && siData != null) {
            errors.addAll(mandateService.apply(requestContext, scheme, transactionType, siData));
        }
        return errors;
    }

    private String extractMandateId(RequestContext requestContext,
                                    Map<String, String> tags,
                                    StandingInstructionAuthorizationProperties.SchemeRule rule) {
        String mandateId = normalize(tags.get(rule.getMandateIdTag()));
        requestContext.setAttribute(ATTR_SI_MANDATE_ID, mandateId);
        return mandateId;
    }

    private ParsedSiData validateSiData(RequestContext requestContext,
                                        Map<String, String> tags,
                                        StandingInstructionAuthorizationProperties.SchemeRule rule,
                                        String mandateId,
                                        List<String> errors) {
        String siData = tags.get(rule.getSiDataTag());
        if (isBlank(siData)) {
            return new ParsedSiData(mandateId, null, null, null, null, null, null, null, null, null);
        }

        if (siData.length() < 49) {
            errors.add("INVALID_SI_DATA: DE" + rule.getDataField() + " tag " + rule.getSiDataTag()
                    + " is too short; expected at least 49 characters");
            return null;
        }

        String requestType = siData.substring(0, 2);
        String numberOfInstallments = siData.substring(2, 5);
        String preferredDay = siData.substring(5, 7);
        String minAmount = siData.substring(7, 19);
        String maxAmount = siData.substring(19, 31);
        String startDate = siData.substring(31, 39);
        String endDate = siData.substring(39, 47);
        String notificationMode = siData.substring(47, 48);
        String frequencyIndicator = siData.substring(48, 49);

        requestContext.setAttribute(ATTR_SI_REQUEST_TYPE, requestType);
        validateNumeric("INVALID_SI_DATA_NO_OF_INSTALLMENTS",
                "Number of instalments must be 3 digits", numberOfInstallments, "\\d{3}", errors);
        validatePreferredDay(preferredDay, errors);
        AmountRange amountRange = validateAmountRange(minAmount, maxAmount, errors);
        DateRange dateRange = validateDateRange(startDate, endDate, errors);
        validateNumeric("INVALID_SI_DATA_NOTIFICATION_MODE",
                "Notification mode must be 0, 1, or 2", notificationMode, "[012]", errors);
        validateNumeric("INVALID_SI_DATA_FREQUENCY_INDICATOR",
                "Frequency indicator must be 0-6", frequencyIndicator, "[0-6]", errors);
        return new ParsedSiData(
                mandateId,
                requestType,
                numberOfInstallments,
                preferredDay,
                amountRange.minAmount(),
                amountRange.maxAmount(),
                dateRange.startDate(),
                dateRange.endDate(),
                notificationMode,
                frequencyIndicator);
    }

    private void validatePreferredDay(String preferredDay, List<String> errors) {
        if (!preferredDay.matches("\\d{2}")) {
            errors.add("INVALID_SI_DATA_PREFERRED_DAY: Preferred day must be 2 digits");
            return;
        }

        int day = Integer.parseInt(preferredDay);
        if (day < 1 || day > 31) {
            errors.add("INVALID_SI_DATA_PREFERRED_DAY: Preferred day must be between 01 and 31");
        }
    }

    private AmountRange validateAmountRange(String minAmount, String maxAmount, List<String> errors) {
        if (!minAmount.matches("\\d{12}")) {
            errors.add("INVALID_SI_DATA_MIN_AMOUNT: Minimum amount must be 12 digits");
            return new AmountRange(null, null);
        }
        if (!maxAmount.matches("\\d{12}")) {
            errors.add("INVALID_SI_DATA_MAX_AMOUNT: Maximum amount must be 12 digits");
            return new AmountRange(null, null);
        }

        BigDecimal min = new BigDecimal(minAmount).movePointLeft(2);
        BigDecimal max = new BigDecimal(maxAmount).movePointLeft(2);
        if (min.compareTo(max) > 0) {
            errors.add("INVALID_SI_DATA_AMOUNT_RANGE: Minimum amount cannot exceed maximum amount");
        }
        return new AmountRange(min, max);
    }

    private DateRange validateDateRange(String startDate, String endDate, List<String> errors) {
        LocalDate parsedStart = parseDate("INVALID_SI_DATA_START_DATE", "Start date", startDate, errors);
        LocalDate parsedEnd = parseDate("INVALID_SI_DATA_END_DATE", "End date", endDate, errors);
        if (parsedStart != null && parsedEnd != null && parsedStart.isAfter(parsedEnd)) {
            errors.add("INVALID_SI_DATA_DATE_RANGE: Start date cannot be after end date");
        }
        return new DateRange(parsedStart, parsedEnd);
    }

    private LocalDate parseDate(String code, String label, String value, List<String> errors) {
        if (isBlank(value) || "00000000".equals(value)) {
            return null;
        }
        if (!value.matches("\\d{8}")) {
            errors.add(code + ": " + label + " must be 8 digits in DDMMYYYY format");
            return null;
        }
        try {
            return LocalDate.parse(value, SI_DATE_FORMAT);
        } catch (DateTimeParseException ex) {
            errors.add(code + ": " + label + " must be a valid DDMMYYYY date");
            return null;
        }
    }

    private void validateNumeric(String code, String message, String value, String pattern, List<String> errors) {
        if (!value.matches(pattern)) {
            errors.add(code + ": " + message);
        }
    }

    private String normalize(String value) {
        return isBlank(value) ? null : value.trim().toUpperCase();
    }

    private boolean isBlank(String value) {
        return value == null || value.trim().isEmpty();
    }

    private boolean isStandingInstruction(TransactionType transactionType) {
        return switch (transactionType) {
            case SI_REGISTRATION, SI_EDIT, SI_TRANSACTION, SI_DEREGISTRATION,
                    SI_REVOKEALL, SI_PAUSE, SI_START, SI_PDN -> true;
            default -> false;
        };
    }

    public record ParsedSiData(
            String mandateId,
            String requestType,
            String numberOfInstallments,
            String preferredDay,
            BigDecimal minAmount,
            BigDecimal maxAmount,
            LocalDate startDate,
            LocalDate endDate,
            String notificationMode,
            String frequencyIndicator) {
    }

    private record AmountRange(BigDecimal minAmount, BigDecimal maxAmount) {
    }

    private record DateRange(LocalDate startDate, LocalDate endDate) {
    }
}
