package com.logicpoint.iso8583.service;

import com.logicpoint.iso8583.context.RequestContext;
import java.util.Map;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Service;

import com.logicpoint.iso8583.entity.Transaction;

@Service
public class TransactionChdEncryptionService {

    private static final Logger logger = LoggerFactory.getLogger(TransactionChdEncryptionService.class);

    private final AesGcmCryptoService aesGcmCryptoService;
    private final ChdKeyCacheService chdKeyCacheService;

    @Value("${chd.encryption.enabled:false}")
    private boolean encryptionEnabled;

    @Value("${chd.encryption.dual-write-plaintext:true}")
    private boolean dualWritePlaintext;

    public TransactionChdEncryptionService(AesGcmCryptoService aesGcmCryptoService,
                                           ChdKeyCacheService chdKeyCacheService) {
        this.aesGcmCryptoService = aesGcmCryptoService;
        this.chdKeyCacheService = chdKeyCacheService;
    }

    public void applyOnWrite(Transaction transaction, RequestContext requestContext) {
        if (transaction == null || requestContext == null || requestContext.getFields() == null) {
            return;
        }

        Integer institutionId = parseInstitutionId(requestContext.getInstitutionId());
        if (institutionId == null) {
            logger.warn("Institution id could not be resolved from PAN BIN; encrypted PAN will not be stored");
            return;
        }
        transaction.setInstitutionId(institutionId);

        Map<Integer, String> fields = requestContext.getFields();
        if (!encryptionEnabled) {
            logger.debug("Transaction CHD encryption is disabled; skipping encrypted PAN persistence");
            return;
        }

        Integer transactionChdKeyVersion = chdKeyCacheService.getActiveKeyVersion(institutionId, ChdKeyCacheService.TXN_KEY_TYPE);
        byte[] transactionKey = chdKeyCacheService.getActiveKey(institutionId, ChdKeyCacheService.TXN_KEY_TYPE);
        if (transactionChdKeyVersion == null || transactionKey == null) {
            logger.warn("Transaction CHD encryption key is not available for institution {}; encrypted PAN will not be stored", institutionId);
            return;
        }

        transaction.setTxnChdKeyVersion(transactionChdKeyVersion);

        String pan = fields.get(2);
        if (pan != null && !pan.isBlank()) {
            transaction.setDe2PanMasked(maskPan(pan));
            EncryptedFieldValue encryptedPan = encryptTransactionValue(
                pan,
                transactionKey,
                transactionChdKeyVersion,
                buildAad(transaction, "de2"));
            if (encryptedPan == null) {
                throw new IllegalStateException("Transaction PAN encryption could not be completed");
            }
            transaction.setDe2PanCt(encryptedPan.getCiphertext());
            transaction.setDe2PanIv(encryptedPan.getIv());
            transaction.setDe2PanTag(encryptedPan.getTag());
            transaction.setPanLookupHmac(resolvePanHash(requestContext, pan, institutionId));
            if (!dualWritePlaintext) {
                transaction.setDe2Pan(null);
            }
        } else {
            transaction.setPanLookupHmac(null);
        }
        transaction.setDe35Track2(null);
        transaction.setDe45Track1(null);
    }

    public void copyEncryptedPan(Transaction source, Transaction target) {
        if (source == null || target == null) {
            return;
        }
        target.setInstitutionId(source.getInstitutionId());
        target.setTxnChdKeyVersion(source.getTxnChdKeyVersion());
        target.setPanLookupHmac(source.getPanLookupHmac());
        target.setDe2PanCt(source.getDe2PanCt());
        target.setDe2PanIv(source.getDe2PanIv());
        target.setDe2PanTag(source.getDe2PanTag());
    }

    private String buildAad(Transaction transaction, String fieldName) {
        return "transaction|" + (transaction.getTransactionId() == null ? "new" : transaction.getTransactionId()) + "|" + fieldName;
    }

    private String maskPan(String pan) {
        if (pan == null || pan.length() < 10) {
            return pan;
        }
        return pan.substring(0, 6) + "******" + pan.substring(pan.length() - 4);
    }

    private String resolvePanHash(RequestContext requestContext, String pan, Integer institutionId) {
        if (requestContext.getPanHash() != null && !requestContext.getPanHash().isBlank()) {
            return requestContext.getPanHash();
        }
        byte[] panLookupKey = chdKeyCacheService.getActiveKey(institutionId, ChdKeyCacheService.PAN_HMAC_KEY_TYPE);
        if (panLookupKey != null) {
            String panHash = aesGcmCryptoService.hmacHex(pan, panLookupKey);
            requestContext.setPanHash(panHash);
            return panHash;
        }
        logger.warn("PAN lookup HMAC key is not available for institution {}; pan_lookup_hmac will remain null", institutionId);
        return null;
    }

    private EncryptedFieldValue encryptTransactionValue(String value, byte[] transactionKey, Integer keyVersion, String aad) {
        return aesGcmCryptoService.encrypt(value, transactionKey, keyVersion, aad);
    }

    private Integer parseInstitutionId(String institutionId) {
        if (institutionId == null || institutionId.isBlank()) {
            return null;
        }
        try {
            return Integer.valueOf(institutionId);
        } catch (NumberFormatException ex) {
            return null;
        }
    }
}
