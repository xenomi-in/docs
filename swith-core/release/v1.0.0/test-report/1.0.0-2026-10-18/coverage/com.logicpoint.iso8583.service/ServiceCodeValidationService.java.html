package com.logicpoint.iso8583.service;

import com.logicpoint.iso8583.entity.ServiceCode;
import com.logicpoint.iso8583.repository.ServiceCodeRepository;
import com.logicpoint.iso8583.util.Scheme;
import com.logicpoint.iso8583.validation.ValidationResult;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.cache.annotation.Cacheable;
import org.springframework.stereotype.Service;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import com.logicpoint.iso8583.validation.TransactionType;

import java.util.Arrays;
import java.util.List;
import java.util.Map;
import java.util.ArrayList;
import java.util.HashMap;
import java.util.Optional;

/**
 * @author LogicPoint Systems
 * Service for validating service codes based on payment schemes.
 * Service codes determine card capabilities and transaction restrictions.
 */
@Service
public class ServiceCodeValidationService {
    
    private static final Logger log = LoggerFactory.getLogger(ServiceCodeValidationService.class);
    
    private final ServiceCodeRepository serviceCodeRepository;
    //private final ResponseCodeMapper responseCodeMapper;
    
    /**
     * Constructs the service with injected repository.
     *
     * @param serviceCodeRepository Service code repository
     */
    @Autowired
    public ServiceCodeValidationService(
            ServiceCodeRepository serviceCodeRepository
            //ResponseCodeMapper responseCodeMapper
            ) {
        this.serviceCodeRepository = serviceCodeRepository;
        //this.responseCodeMapper = responseCodeMapper;
    }
    
    /**
     * Validates service code for a transaction.
     * 
     * @param serviceCode The 3-digit service code
     * @param scheme The payment scheme
     * @param transactionType The transaction type
     * @param transactionData The transaction data
     * @return ValidationResult with validation status
     */
    @Cacheable(value = "serviceCodeCache", key = "#serviceCode + '_' + #scheme.name() + '_' + #transactionType.name()")
    public ValidationResult validateServiceCode(
            String serviceCode,
            Scheme scheme,
            TransactionType transactionType,
            Map<Integer, String> transactionData) {
        
        log.debug("Validating service code: {} for scheme: {}, transactionType: {}", 
            serviceCode, scheme, transactionType);
        
        // Basic validation
        if (serviceCode == null || serviceCode.trim().isEmpty()) {
            log.warn("Service code is null or empty");
            return createFailureResult("INVALID_SERVICE_CODE: Service code is required", scheme, transactionType);
        }
        
        if (!serviceCode.matches("\\d{3}")) {
            log.warn("Invalid service code format: {}", serviceCode);
            return createFailureResult("INVALID_SERVICE_CODE: Service code must be 3 digits", scheme, transactionType);
        }
        
        Optional<ServiceCode> serviceCodeConfig = findServiceCodeConfig(serviceCode, scheme);
        
        if (serviceCodeConfig.isEmpty()) {
            log.warn("Service code not found: {} for scheme: {}", serviceCode, scheme);
            return createFailureResult("INVALID_SERVICE_CODE: Service code not supported", scheme, transactionType);
        }
        
        ServiceCode config = serviceCodeConfig.get();
        
        // Validate service code capabilities
        ValidationResult capabilityResult = validateServiceCodeCapabilities(config, transactionData, transactionType);
        if (!capabilityResult.isValid()) {
            return capabilityResult;
        }
        
        // Validate transaction restrictions
        ValidationResult restrictionResult = validateTransactionRestrictions(config, transactionType, transactionData);
        if (!restrictionResult.isValid()) {
            return restrictionResult;
        }
        
        log.debug("Service code validation successful: {}", serviceCode);
        return new ValidationResult(new ArrayList<>(), new ArrayList<>());
    }

    private Optional<ServiceCode> findServiceCodeConfig(String serviceCode, Scheme scheme) {
        Optional<ServiceCode> configuredServiceCode = serviceCodeRepository.findByServiceCodeAndScheme(
            serviceCode, scheme.name());
        if (configuredServiceCode.isPresent()) {
            return configuredServiceCode;
        }

        if (scheme == Scheme.RUPAY) {
            return deriveStandardRuPayServiceCode(serviceCode);
        }

        return Optional.empty();
    }

    private Optional<ServiceCode> deriveStandardRuPayServiceCode(String serviceCode) {
        char interchange = serviceCode.charAt(0);
        char authorizationProcessing = serviceCode.charAt(1);
        char rangeOfServices = serviceCode.charAt(2);

        if (!isKnownInterchangeQualifier(interchange)
                || !isKnownAuthorizationProcessing(authorizationProcessing)
                || !isKnownRangeOfServices(rangeOfServices)) {
            return Optional.empty();
        }

        ServiceCode config = new ServiceCode(serviceCode, Scheme.RUPAY.name());
        config.setInterchangeQualifier(String.valueOf(interchange));
        config.setAuthorizationProcessing(String.valueOf(authorizationProcessing));
        config.setRangeOfServices(String.valueOf(rangeOfServices));
        config.setDescription("Derived standard RuPay service code");
        config.setIsInternational(interchange == '1' || interchange == '2');
        config.setIsChip(interchange == '2' || interchange == '6');
        config.setIsMagneticStripe(interchange != '6');
        config.setIsContactless(false);
        config.setIsOnlinePin(rangeOfServices == '0' || rangeOfServices == '3'
                || rangeOfServices == '5' || rangeOfServices == '6');
        config.setIsOfflinePin(false);
        config.setIsSignature(rangeOfServices == '1' || rangeOfServices == '2'
                || rangeOfServices == '6' || rangeOfServices == '7');
        config.setIsNoCvm(false);
        config.setIsActive(true);
        return Optional.of(config);
    }

    private boolean isKnownInterchangeQualifier(char digit) {
        return digit == '1' || digit == '2' || digit == '5'
                || digit == '6' || digit == '7' || digit == '9';
    }

    private boolean isKnownAuthorizationProcessing(char digit) {
        return digit == '0' || digit == '2' || digit == '4' || digit == '6';
    }

    private boolean isKnownRangeOfServices(char digit) {
        return digit >= '0' && digit <= '7';
    }
    
    /**
     * Validates service code capabilities against transaction requirements.
     *
     * @param config Service code configuration
     * @param transactionData Transaction data
     * @param transactionType Transaction type
     * @return Validation result
     */
    private ValidationResult validateServiceCodeCapabilities(ServiceCode config, Map<Integer, String> transactionData,TransactionType transactionType) {
        String posEntryMode = transactionData.get(22); // POS Entry Mode
        String posConditionCode = transactionData.get(25); // POS Condition Code
        
        // Check contactless capability
        if (isContactlessTransaction(posEntryMode, posConditionCode) && !config.getIsContactless()) {
            log.warn("Contactless transaction attempted with non-contactless card: {}", config.getServiceCode());
            return createFailureResult("INVALID_SERVICE_CODE: Card does not support contactless", 
                Scheme.valueOf(config.getScheme()), transactionType);
        }
        
        // Check chip capability
        if (isChipTransaction(posEntryMode) && !config.getIsChip()) {
            log.warn("Chip transaction attempted with non-chip card: {}", config.getServiceCode());
            return createFailureResult("INVALID_SERVICE_CODE: Card does not support chip", 
                Scheme.valueOf(config.getScheme()), transactionType);
        }
        
        // Check magnetic stripe capability
        if (isMagneticStripeTransaction(posEntryMode) && !config.getIsMagneticStripe()) {
            log.warn("Magnetic stripe transaction attempted with non-magnetic stripe card: {}", config.getServiceCode());
            return createFailureResult("INVALID_SERVICE_CODE: Card does not support magnetic stripe", 
                Scheme.valueOf(config.getScheme()), transactionType);
        }
        
        return new ValidationResult(new ArrayList<>(), new ArrayList<>());
    }
    
    /**
     * Validates transaction restrictions based on service code.
     *
     * @param config Service code configuration
     * @param transactionType Transaction type
     * @param transactionData Transaction data
     * @return Validation result
     */
    private ValidationResult validateTransactionRestrictions(ServiceCode config, TransactionType transactionType, 
                                                        Map<Integer, String> transactionData) {
        
        // Check international transaction restrictions
        if (isInternationalTransaction(transactionData) && !config.getIsInternational()) {
            log.warn("International transaction attempted with domestic-only card: {}", config.getServiceCode());
            return createFailureResult("INVALID_SERVICE_CODE: Card does not support international transactions", 
                Scheme.valueOf(config.getScheme()), transactionType);
        }
        
        // Check CVM (Cardholder Verification Method) requirements
        ValidationResult cvmResult = validateCvmRequirements(config, transactionData);
        if (!cvmResult.isValid()) {
            return cvmResult;
        }
        
        return new ValidationResult(new ArrayList<>(), new ArrayList<>());
    }
    
    /**
     * Validates CVM requirements based on service code.
     *
     * @param config Service code configuration
     * @param transactionData Transaction data
     * @return Validation result
     */
    private ValidationResult validateCvmRequirements(ServiceCode config, Map<Integer, String> transactionData) {
        String posEntryMode = transactionData.get(22);
        String posConditionCode = transactionData.get(25);
        
        // Check if PIN is required but not supported
        if (isPinRequired(posEntryMode, posConditionCode)) {
            if (!config.getIsOnlinePin() && !config.getIsOfflinePin()) {
                log.warn("PIN required but card does not support PIN: {}", config.getServiceCode());
                return createFailureResult("INVALID_SERVICE_CODE: Card does not support PIN", 
                    Scheme.valueOf(config.getScheme()), TransactionType.PURCHASE);
            }
        }
        
        // Check if signature is required but not supported
        if (isSignatureRequired(posEntryMode, posConditionCode)) {
            if (!config.getIsSignature()) {
                log.warn("Signature required but card does not support signature: {}", config.getServiceCode());
                return createFailureResult("INVALID_SERVICE_CODE: Card does not support signature", 
                    Scheme.valueOf(config.getScheme()), TransactionType.PURCHASE);
            }
        }
        
        return new ValidationResult(new ArrayList<>(), new ArrayList<>());
    }
    
    /**
     * Determines if transaction is contactless.
     *
     * @param posEntryMode POS entry mode
     * @param posConditionCode POS condition code
     * @return true if contactless
     */
    private boolean isContactlessTransaction(String posEntryMode, String posConditionCode) {
        if (posEntryMode == null) return false;
        
        // Contactless entry modes: 07, 91
        return posEntryMode.equals("07") || posEntryMode.equals("91") || 
               (posEntryMode.length() >= 2 && posEntryMode.substring(0, 2).equals("07"));
    }
    
    /**
     * Determines if transaction is chip-based.
     *
     * @param posEntryMode POS entry mode
     * @return true if chip-based
     */
    private boolean isChipTransaction(String posEntryMode) {
        if (posEntryMode == null) return false;
        
        // Chip entry modes: 05, 07, 95
        return posEntryMode.equals("05") || posEntryMode.equals("07") || posEntryMode.equals("95") ||
               (posEntryMode.length() >= 2 && 
                (posEntryMode.substring(0, 2).equals("05") || posEntryMode.substring(0, 2).equals("07")));
    }
    
    /**
     * Determines if transaction is magnetic stripe-based.
     *
     * @param posEntryMode POS entry mode
     * @return true if magnetic stripe
     */
    private boolean isMagneticStripeTransaction(String posEntryMode) {
        if (posEntryMode == null) return false;
        
        // Magnetic stripe entry modes: 01, 02
        return posEntryMode.equals("01") || posEntryMode.equals("02") ||
               (posEntryMode.length() >= 2 && 
                (posEntryMode.substring(0, 2).equals("01") || posEntryMode.substring(0, 2).equals("02")));
    }
    
    /**
     * Determines if transaction is international.
     *
     * @param transactionData Transaction data
     * @return true if international
     */
    private boolean isInternationalTransaction(Map<Integer, String> transactionData) {
        String acquiringCountry = transactionData.get(19); // Acquiring Country Code
        String panCountry = transactionData.get(20); // PAN Extended Country Code
        
        // Simple logic - in real implementation, you'd check against card's country
        return acquiringCountry != null && panCountry != null && !acquiringCountry.equals(panCountry);
    }
    
    /**
     * Determines if PIN is required.
     *
     * @param posEntryMode POS entry mode
     * @param posConditionCode POS condition code
     * @return true if PIN required
     */
    private boolean isPinRequired(String posEntryMode, String posConditionCode) {
        // PIN required for ATM transactions (01) and some POS transactions
        return "01".equals(posEntryMode) || "02".equals(posEntryMode);
    }
    
    /**
     * Determines if signature is required.
     *
     * @param posEntryMode POS entry mode
     * @param posConditionCode POS condition code
     * @return true if signature required
     */
    private boolean isSignatureRequired(String posEntryMode, String posConditionCode) {
        // Signature typically required for manual entry and some POS transactions
        return "01".equals(posEntryMode) || "03".equals(posEntryMode);
    }
    
    /**
     * Creates failure result with scheme-specific response code.
     *
     * @param error Error message
     * @param scheme Payment scheme
     * @param transactionType Transaction type
     * @return Failure validation result
     */
    private ValidationResult createFailureResult(String error, Scheme scheme, TransactionType transactionType) {
        //ResponseCodeMapper.ResponseCodeMapping mapping = responseCodeMapper.mapErrorToResponseCode(
        //    error, scheme, transactionType);
        
        return new ValidationResult(new ArrayList<>(Arrays.asList(error)), new ArrayList<>());
    }
    
    /**
     * Gets service code information.
     *
     * @param serviceCode Service code
     * @param scheme Scheme
     * @return Optional service code info
     */
    @Cacheable(value = "serviceCodeCache", key = "#serviceCode + '_' + #scheme")
    public Optional<ServiceCode> getServiceCodeInfo(String serviceCode, String scheme) {
        return serviceCodeRepository.findByServiceCodeAndScheme(serviceCode, scheme);
    }
    
    /**
     * Gets all service codes for a scheme.
     *
     * @param scheme Scheme
     * @return List of service codes
     */
    public List<ServiceCode> getServiceCodesByScheme(String scheme) {
        return serviceCodeRepository.findBySchemeAndIsActiveTrue(scheme);
    }
    
    /**
     * Gets contactless service codes for a scheme.
     *
     * @param scheme Scheme
     * @return List of contactless service codes
     */
    public List<ServiceCode> getContactlessServiceCodes(String scheme) {
        return serviceCodeRepository.findContactlessByScheme(scheme);
    }
    
    /**
     * Gets chip service codes for a scheme.
     *
     * @param scheme Scheme
     * @return List of chip service codes
     */
    public List<ServiceCode> getChipServiceCodes(String scheme) {
        return serviceCodeRepository.findChipByScheme(scheme);
    }
}
