    @PostConstruct

        if (rawHsmPayloadLoggingEnabled) {
            boolean production = environment != null
                    && Arrays.stream(environment.getActiveProfiles())
                            .anyMatch(profile -> "prod".equalsIgnoreCase(profile)
                                    || "production".equalsIgnoreCase(profile));

                        "Raw HSM payload logging cannot be enabled with a production profile");
            }
            logger.warn("Raw HSM payload logging is enabled. HSM commands and responses may contain CHD, SAD, or cryptographic keys");

    /*
     * This command translates a Visa working key from encryption under a key encryption key (KEK) / ZMK to encryption under the MFK,
     */
    public String TranslateWorkingKey(String workingKey, String keyType,boolean  isTR31Format, String scheme) {
        String networkName = normalizeNetworkName(scheme);
        String kek = keyManagementService.getNetworkKey(networkName, "KEK");
        logger.debug("KEK in TranslateWorkingKey resolved for network: {}, available: {}", networkName, kek != null);
        if (kek == null || kek.isBlank()) {
            logger.error("No active network KEK found for scheme: {}", networkName);
            return "01;No active network KEK found for scheme: " + networkName;
        }

        String modifier = "0";
        FuturexHsmCmd futurexHsmCmd = new FuturexHsmCmd("TVWK");
        try {
            switch (keyType) {
                case "ZPK":
                    modifier = "1";
                    break;
                case "AES":
                    break;
                default:
            }
            //futurexHsmCmd.addField("AO", "TVWK"); // command indicator
            futurexHsmCmd.addField("AS", modifier); // modifier
            futurexHsmCmd.addField("FD", workingKey); // working key (encrypted under KEK / ZMK)
            futurexHsmCmd.addField("AP", kek); // key encryption key (KEK / ZMK)
            if(isTR31Format) futurexHsmCmd.addField("OF", "T");
            else futurexHsmCmd.addField("OF", "C"); // output key fromat C: Cryptogram , T: TR-31 , A : AKB format key block
            String command = futurexHsmCmd.build();
            String response = hsmClient.sendMessage(command);
            FuturexHsmCmd responseCmd = FuturexHsmCmd.parse(response);
            if (responseCmd.isSuccess()) {
                String keyUnderMFK = responseCmd.getField("BG");
                String KCV = responseCmd.getField("AE");
                return "00;"+keyUnderMFK + ";" + KCV;

            } else {
                return "01;"+responseCmd.getErroMsg();
            }
        } catch (Exception e) {
            logger.error("Error building command: {}", e.getMessage());
            return null;
        }

    }

    private String normalizeNetworkName(String scheme) {
        return scheme == null || scheme.isBlank() ? "UNKNOWN" : scheme.trim().toUpperCase();
    }

    /*
    * This command generates a working key encrypted under the MFK modifier of the key type. 
     */
    public String generateWorkingKey(String keyType) {
        try {
            FuturexHsmCmd futurexHsmCmd = new FuturexHsmCmd("GWKS");
            // futurexHsmCmd.addField("AO", "GWKS"); // command indicator

            futurexHsmCmd.addField("FS", "2"); // 3DES keyType  1 - single DES key , 2 - Double length DES key , 3 - Triple Length DES key

            switch (keyType) {
                case "PVK":
                    futurexHsmCmd.addField("AS", "4"); // modifier  4 - PIN verfication Key
                    break;
                case "DKE":
                    futurexHsmCmd.addField("AS", "2"); // modifier  2 - Data Encryption Key
                    break;
                case "PGK":
                    futurexHsmCmd.addField("AS", "9"); // modifier 9 - PIN Generation Key
                    break;
                case "CVK":
                    futurexHsmCmd.addField("AS", "4"); // modifier 4 - CVK
                    break;
            }
            String command = futurexHsmCmd.build();
            String response = hsmClient.sendMessage(command);
            FuturexHsmCmd responseCmd = FuturexHsmCmd.parse(response);
            if (responseCmd.isSuccess()) {
                String workingKey = responseCmd.getField("BG");
                String KCV = responseCmd.getField("AE");
                return workingKey + ";" + KCV;
            } else {
                return responseCmd.getErroMsg();
            }
        } catch (Exception e) {
            logger.error("Error building command: {}", e.getMessage());
            return null;
        }
    }

    public HsmWrappedKeyResult wrapClearChdKey(byte[] clearKey, Integer institutionId, String wrappingKeyType, boolean isTr31Format) {
        if (clearKey == null || clearKey.length == 0) {
            return null;
        }

        String wrappingKey = getWrappingKeyForInstitution(institutionId, wrappingKeyType);
        if (wrappingKey == null || wrappingKey.isBlank()) {
            logger.error("No institution {} wrapping key available for institution {}", wrappingKeyType, institutionId);
            return null;
        }

        try {
            String wrappedKey = encryptData(bytesToHex(clearKey), wrappingKey, "AES");
            if (wrappedKey == null || wrappedKey.isBlank()) {
                return null;
            }

            String kcv = generateKeyCheckValue(clearKey);
            return new HsmWrappedKeyResult(wrappedKey, kcv, null);
        } finally {
            Arrays.fill(clearKey, (byte) 0);
        }
    }

    public byte[] unwrapChdKey(byte[] wrappedKey, Integer institutionId, String wrappingKeyType, boolean isTr31Format) {
        if (wrappedKey == null || wrappedKey.length == 0) {
            return null;
        }

        String wrappingKey = getWrappingKeyForInstitution(institutionId, wrappingKeyType);
        if (wrappingKey == null || wrappingKey.isBlank()) {
            logger.error("No institution {} wrapping key available for institution {}", wrappingKeyType, institutionId);
            return null;
        }

        String wrappedKeyValue = new String(wrappedKey, StandardCharsets.US_ASCII).trim();
        String decryptedHex = decryptData(wrappedKeyValue, wrappingKey, "AES");
        if (decryptedHex == null || decryptedHex.isBlank()) {
            return null;
        }
        return hexToBytes(decryptedHex);
    }

    public String generateOffsetWithPinBlock(String pinBlock, String pan, String pinMethod) {
        try {
            String pvk = "7C2F09C04E63558BF1D80E171EEC0A58";
            String pek = "B41AF6C89B8509CA2A8DDA27F18F86AB";
            String decimalizationTable = "0123456789012345";
            String pgk = "935258018D5D77017F59E5585CEFDB3B";  // pin generation key
            FuturexHsmCmd futurexHsmCmd = new FuturexHsmCmd("GOFF");
            // Extract the rightmost 11 digits of the PAN as per ISO 9564 standard
            // Remove the check digit (last digit)
            String withoutCheckDigit = pan.substring(0, pan.length() - 1);
            // Take the last 11 digits
            String partialPan = withoutCheckDigit.substring(withoutCheckDigit.length() - 11);
            String additionalPinBlockData = withoutCheckDigit.substring(withoutCheckDigit.length() - 12);
            if (pinMethod.equals("VISA")) {// VISA 
                futurexHsmCmd.addField("BF", "3");
                futurexHsmCmd.addField("BY", "0"); // PVKI
                futurexHsmCmd.addField("CA", pvk); // PIN verfication Key
                futurexHsmCmd.addField("BZ", partialPan); // partial PAN
            } else if (pinMethod.equals("IBM3624")) {// IBM3624
                futurexHsmCmd.addField("BJ", "1");   // Major Key = 1( MFK) 
                futurexHsmCmd.addField("BF", "2");   // PIN generation method = 2( IBM3624)
                futurexHsmCmd.addField("AW", "1");   // PinBlock format = 1(ISO 9564-1 Format 0)
                futurexHsmCmd.addField("BS", decimalizationTable);   // decimalization table 
                futurexHsmCmd.addField("AT", "F"); // padding character
                futurexHsmCmd.addField("BE", pan); // PAN value (card number)
                futurexHsmCmd.addField("CO", pgk);   // pin generation key 
            } else if (pinMethod.equals("ISO9564")) {
                futurexHsmCmd.addField("BF", "1"); // ISO9564

            } else {
                return "Invalid pin method";
            }
            futurexHsmCmd.addField("AL", pinBlock); // pin block
            futurexHsmCmd.addField("AX", pek); //   pin encryption key PEK
            futurexHsmCmd.addField("AK", additionalPinBlockData); // additional Pin block data

            String command = futurexHsmCmd.build();
            String response = hsmClient.sendMessage(command);
            FuturexHsmCmd responseCmd = FuturexHsmCmd.parse(response);
            if (responseCmd.isSuccess()) {
                String offset = null;
                if (pinMethod.equals("VISA")) {
                    offset = responseCmd.getField("BX");
                } else if (pinMethod.equals("IBM3624")) {
                    offset = responseCmd.getField("AY");
                }
                return offset;
            } else {
                return responseCmd.getErroMsg();
            }
        } catch (Exception e) {
            logger.error("Error building command: {}", e.getMessage());
            return null;
        }
    }

    /**
     * Sends a message asynchronously to the HSM.
     *
     * @param message The message to send
     * @return CompletableFuture with the response
     */
    public CompletableFuture<String> sendMessageAsync(String message) {
        logRawHsmPayload("outbound-async", message);
        return hsmClient.sendMessageAsync(message)
                .whenComplete((response, throwable) -> {
                    if (throwable == null) {
                        logRawHsmPayload("inbound-async", response);
                    }
                });
    }

    /**
     * Sends a message synchronously to the HSM.
     *
     * @param message The message to send
     * @return Response from HSM
     * @throws Exception if communication fails
     */
    public String sendMessage(String message) throws Exception {
        logRawHsmPayload("outbound-sync", message);
        String response = hsmClient.sendMessage(message);
        logRawHsmPayload("inbound-sync", response);
        return response;
    }

    /**
     * Sends multiple messages concurrently to demonstrate connection pooling.
     *
     * @param messages Array of messages to send
     * @return Array of CompletableFuture responses
     */
    public CompletableFuture<String>[] sendMessagesConcurrently(String... messages) {
        @SuppressWarnings("unchecked")
        CompletableFuture<String>[] futures = new CompletableFuture[messages.length];

        for (int i = 0; i < messages.length; i++) {
            final int messageIndex = i;
            futures[i] = sendMessageAsync(messages[i])
                    .thenApply(response -> {
                        logger.debug("Received HSM response for concurrent message index={}", messageIndex);
                        return response;
                    })
                    .exceptionally(throwable -> {
                        logger.error("Error sending message {}: {}", messageIndex, throwable.getMessage());
                        return null;
                    });
        }

        return futures;
    }

    }

               KT.append((iccData.getTagValue("9F10")).substring(4)); // IAD 
               
           }else if (CVN.equals("04") || CVN.equals("01") || CVN.equals("02")) {   // for CVN =04/02/01 for JCB card
                KT.append((iccData.getTagValue("9F10")).substring(6)); // IAD 
           
            }else KT.append(iccData.getTagValue("9F10")); // IAD 
            futurexHsmCmd.addField("KT", KT.toString()); // Key usage data
            futurexHsmCmd.addField("BJ", ARC); // Approval response code

            String response = hsmClient.sendMessage(futurexHsmCmd.build());
            if (response != null) {
                FuturexHsmCmd responseCmd = FuturexHsmCmd.parse(response);
                if (responseCmd.isSuccess()) {

                    String verifyResult = responseCmd.getField("BB");
                    String arpc = responseCmd.getField("BG") + ARC;
                    if (CVN.equals("B5")) {

                    }else if(CVN.equals("04")) {
                        arpc = responseCmd.getField("BG") +CSU + ipd;
                   
                    }else if (CVN.equals("05") || CVN.equals("06")) {
                        arpc = responseCmd.getField("BG") + ARC;
                    }
                    //verifyResult = "Y";
                    //arpc ="2BE58B2A80000000"+ARC;
                    return verifyResult + ";" + arpc;
                } else {
                    return "F" + responseCmd.getErroMsg();
                }
            } else {
                return null;
            }

        } catch (Exception e) {
            logger.error("Error verifying ARQC and generating ARPC: {}", e.getMessage());
            return null;
        }
    }


   
    /**
     * Generates CVV (Card Verification Value) for a card.
     *
     * @param pan Primary Account Number
     * @param expiryDate Card expiry date (MMYY format)
     * @param serviceCode Service code (e.g., "201" for ATM, "101" for POS)
     * @param cvvKey CVV generation key
     * @return Generated CVV or null if failed
     */
    public String generateCvv(String pan, String expiryDate, String serviceCode, String cvvKey) {
        try {

            // CMD: [AOGCVC;AV8173193120877474;CAB9381732EB907253611FFD54159116BD;FA3012;FB620;FT1]
            FuturexHsmCmd futurexHsmCmd = new FuturexHsmCmd("GCVV");
            //cvvKey = "C6CA6CCD5C517A0F0511740BB7ED1860";
            cvvKey = getWorkingKeyForCard(pan, "CVK");
            logger.debug("CVK resolved for CVV generation: {}", cvvKey != null && !cvvKey.isBlank());
            // Add required fields for CVV generation
            futurexHsmCmd.addField("AV", pan); // PAN
            futurexHsmCmd.addField("FA", expiryDate); // Expiry date
            futurexHsmCmd.addField("FB", serviceCode); // Service code
            futurexHsmCmd.addField("CA", cvvKey); // CVV key
            futurexHsmCmd.addField("FT", "1"); // CVV type

            String command = futurexHsmCmd.build();
            String response = hsmClient.sendMessage(command);
            if (response != null) {
                FuturexHsmCmd responseCmd = FuturexHsmCmd.parse(response);

                if (responseCmd.isSuccess()) {
                    String cvv = responseCmd.getField("FC"); // Generated CVV
                    logger.debug("CVV generated successfully for PAN: {}", maskPan(pan));
                    return cvv;
                } else {
                    logger.error("CVV generation failed: {}", responseCmd.getErroMsg());
                    return null;
                }
            } else {
                return null;
            }

        } catch (Exception e) {
            logger.error("Error generating CVV: {}", e.getMessage());
            return null;
        }
    }

    /**
     * Verifies CVV (Card Verification Value) for a card.
     *
     * @param pan Primary Account Number
     * @param expiryDate Card expiry date (MMYY format)
     * @param serviceCode Service code
     * @param cvvToVerify CVV value to verify
     * @param cvvKey CVV verification key
     * @return true if CVV is valid, false otherwise
     */
    public int verifyCvv(String pan, String expiryDate, String serviceCode, String cvvToVerify, String cvvKey,String cvv, int cvvType) {
        try {
            FuturexHsmCmd futurexHsmCmd = new FuturexHsmCmd("VCVV");
            //cvvKey = "C6CA6CCD5C517A0F0511740BB7ED1860";
            // Add required fields for CVV generation
            String cvvKey1 = getWorkingKeyForCard(pan, "CVK");
            logger.debug("CVK resolved for CVV verification: {}", cvvKey1 != null && !cvvKey1.isBlank());
            futurexHsmCmd.addField("AV", pan); // PAN
            futurexHsmCmd.addField("FA", expiryDate); // Expiry date
            futurexHsmCmd.addField("FB", serviceCode); // Service code
            futurexHsmCmd.addField("CA", cvvKey1); // CVV key
            if(cvvType == 0){
                futurexHsmCmd.addField("FC", cvv); // CVV value read from track
            }else{
                futurexHsmCmd.addField("FT", cvv); // CVV value embossed on card
            }

            String command = futurexHsmCmd.build();
            String response = hsmClient.sendMessage(command);
            if (response != null) {
                FuturexHsmCmd responseCmd = FuturexHsmCmd.parse(response);

                if (responseCmd.isSuccess()) {
                    String verficationResult = responseCmd.getField("BB"); // Generated CVV
                    if(verficationResult.equals("Y")){
                        logger.debug("CVV Verification success for PAN: {}", verficationResult);
                         return 0;
                    } else{
                        logger.error("CVV Verification failed: {}", verficationResult);
                        return 1;
                    } 
                } else {
                    logger.error("CVV verification failed: {}", responseCmd.getErroMsg());
                    return 2;
                }
            } else {
                logger.error("CVV verification failed(No response from HSM): {}", response);
                return 2;
            }

        } catch (Exception e) {
            logger.error("Error verifying CVV: {}", e.getMessage());
            return 2;
        }
    }
    /**
     * Generates a MAC (Message Authentication Code) for the given input string.
     *
     * @param data The input data to generate the MAC for
     * @param scheme The scheme of the card
     * @return The generated MAC string, or null if failure
     */
    public String generateMac(String data, String scheme) {
        try {
            FuturexHsmCmd futurexHsmCmd = new FuturexHsmCmd("GMAC");
            // Use ZAK (MAC Key) for MAC generation. For demo, using RuPay default BIN, but you may update as needed.
            String key = getWorkingKeyForCard("8000000000000000", "ZAK");
            if (key == null || key.isEmpty()) {
                logger.error("MAC key is not found for scheme: {}", scheme);
                return null;
            }
            // Macro fields as per HSM requirements (may adjust for actual integration)
            futurexHsmCmd.addField("BN", "0"); // MAC block number if needed
            futurexHsmCmd.addField("AR", key); // MAC Key
            futurexHsmCmd.addField("BO", data); // Data for MAC
            futurexHsmCmd.addField("AD", "8"); // 8 -> 16 digit Hexadecimal encoded ASCII data
            futurexHsmCmd.addField("BF", "1"); // ISO9797 Algorithm 3 (see verifyMac for rationale)
            
            String command = futurexHsmCmd.build();
            String response = hsmClient.sendMessage(command);

            if (response != null) {
                FuturexHsmCmd responseCmd = FuturexHsmCmd.parse(response);
                if (responseCmd.isSuccess()) {
                    String macValue = responseCmd.getField("BP"); // MAC value
                    logger.debug("Generated MAC: {} for data: {}", macValue, data);
                    return macValue;
                } else {
                    logger.error("MAC generation failed: {}", responseCmd.getErroMsg());
                    return null;
                }
            } else {
                logger.error("MAC generation failed (No response from HSM): {}", data);
                return null;
            }
        } catch (Exception e) {
            logger.error("Error generating MAC: {}", e.getMessage());
            return null;
        }
    }
    /**
     * Verifies the MAC (Message Authentication Code) for the given input string.
     *
     * @param data The input data to verify the MAC for
     * @param mac The MAC value to verify against
     * @param scheme The scheme of the card
     * @return true if the MAC is valid, false otherwise
     */
    public boolean verifyMac(String data, String mac, String scheme) {
        try {
            FuturexHsmCmd futurexHsmCmd = new FuturexHsmCmd("VMAC");
            String zak = keyManagementService.getCurrentSessionNetworkKey(scheme, "ZAK");
            logger.debug("Session ZAK in verifyMac resolved for scheme: {}, available: {}", scheme, zak != null);
            if (zak == null || zak.isBlank()) {
                logger.error("Session-scoped ZAK is not found for scheme: {}", scheme);
                return false;
            }
            // Add required fields for MAC verification
            futurexHsmCmd.addField("BN", "0"); 
            futurexHsmCmd.addField("AR", zak); // MAC Key 
            futurexHsmCmd.addField("BO", data); // data to verify the MAC
            futurexHsmCmd.addField("BP", mac); // MAC to verify
            futurexHsmCmd.addField("AD", "8"); // 8 -> 16 digit Hexadecimal encoded ASCII data
            futurexHsmCmd.addField("BF", "1"); /*  0: Standard/FIPS113. The entire length of the key is used for all data blocks. (default)
                                                                1: ISO9797 Algorithm 3. Key part 1 is used up until the last block (single DES), the entire key length is then used for the last block. */

            String command = futurexHsmCmd.build();
            String response = hsmClient.sendMessage(command);

            if (response != null) {
                FuturexHsmCmd responseCmd = FuturexHsmCmd.parse(response);

                if (responseCmd.isSuccess()) {
                    String verificationResult = responseCmd.getField("BB");
                    if ("Y".equalsIgnoreCase(verificationResult)) {
                        logger.debug("MAC verification succeeded for data: {}", data);
                        return true;
                    } else {
                        logger.error("MAC verification failed: {}", verificationResult);
                        return false;
                    }
                } else {
                    logger.error("MAC verification failed: {}", responseCmd.getErroMsg());
                    return false;
                }
            } else {
                logger.error("MAC verification failed (No response from HSM): {}", data);
                return false;
            }
        } catch (Exception e) {
            logger.error("Error verifying MAC: {}", e.getMessage());
            return false;
        }
    }
    /**
     * Encrypts data using a specified encryption key.
     *
     * @param data Data to encrypt (hex string)
     * @param encryptionKey Encryption key (hex string)
     * @param algorithm Encryption algorithm (e.g., "DES", "3DES", "AES")
     * @return Encrypted data or null if failed
     */
    public String encryptData(String data, String encryptionKey, String algorithm) {
        try {
            FuturexHsmCmd futurexHsmCmd = new FuturexHsmCmd("CBCE");
            String IV = "0000000000000000";
            // Add required fields for encryption
            futurexHsmCmd.addField("GG", encryptionKey); // encryption key under HSM control
            futurexHsmCmd.addField("GH", IV); // IV value
            futurexHsmCmd.addField("BO", data); // Data to be encrypted
            futurexHsmCmd.addField("ZA", "1"); // clear IV

            String command = futurexHsmCmd.build();
            String response = hsmClient.sendMessage(command);
            FuturexHsmCmd responseCmd = FuturexHsmCmd.parse(response);

            if (responseCmd.isSuccess()) {
                String encryptedData = responseCmd.getField("BP"); // Encrypted data
                logger.debug("Data encrypted successfully using algorithm: {}", algorithm);
                return encryptedData;
            } else {
                logger.error("Data encryption failed: {}", responseCmd.getErroMsg());
                return null;
            }

        } catch (Exception e) {
            logger.error("Error encrypting data: {}", e.getMessage());
            return null;
        }
    }

    /**
     * Decrypts data using a specified decryption key.
     *
     * @param encryptedData Encrypted data to decrypt (hex string)
     * @param decryptionKey Decryption key (hex string)
     * @param algorithm Decryption algorithm (e.g., "DES", "3DES", "AES")

        }
    }

    /**
     * Masks PAN for logging purposes.
     *
     * @param pan PAN to mask
     * @return Masked PAN
     */
    private String maskPan(String pan) {
        if (pan == null || pan.length() < 8) {
            return "****";
        }
        return pan.substring(0, 4) + "****" + pan.substring(pan.length() - 4);
    }

    private String getWorkingKeyForCard(String cardNumber, String keyType) {

    private String generateKeyCheckValue(byte[] clearKey) {
        byte[] preview = Arrays.copyOf(clearKey, Math.min(clearKey.length, 3));
        return bytesToHex(preview);
    }

    private String bytesToHex(byte[] data) {
        StringBuilder builder = new StringBuilder(data.length * 2);
        for (byte b : data) {
            builder.append(String.format("%02X", b));
        }
        return builder.toString();
    }

    private byte[] hexToBytes(String hex) {
        if (hex == null || hex.isBlank()) {
            return new byte[0];
        }
        byte[] result = new byte[hex.length() / 2];
        for (int i = 0; i < hex.length(); i += 2) {
            result[i / 2] = (byte) Integer.parseInt(hex.substring(i, i + 2), 16);
        }
        return result;
    }

    public void demonstrateConcurrentOperations() {

        });
    }
