package com.logicpoint.iso8583.service;

import com.logicpoint.iso8583.config.InstallmentAuthorizationProperties;
import com.logicpoint.iso8583.context.RequestContext;
import com.logicpoint.iso8583.util.Scheme;
import com.logicpoint.iso8583.validation.TransactionType;
import java.math.BigDecimal;
import java.util.ArrayList;
import java.util.List;
import java.util.Map;
import org.springframework.stereotype.Service;

/**
 * Validates installment/EMI data carried in ISO8583 authorization TLV fields.
 */
@Service
public class InstallmentAuthorizationValidationService {

    public static final String ATTR_EMI_TOTAL_AMOUNT = "emi_total_amount";
    public static final String ATTR_EMI_AMOUNT = "emi_amount";
    public static final String ATTR_EMI_INSTALLMENTS = "emi_installments";
    public static final String ATTR_EMI_INTEREST_RATE = "emi_interest_rate";

    private final InstallmentAuthorizationProperties properties;

    public InstallmentAuthorizationValidationService(InstallmentAuthorizationProperties properties) {
        this.properties = properties;
    }

    public List<String> validate(RequestContext requestContext, Scheme scheme, TransactionType transactionType) {
        List<String> errors = new ArrayList<>();
        if (!properties.isEnabled() || requestContext == null || scheme == null || transactionType == null) {
            return errors;
        }

        InstallmentAuthorizationProperties.SchemeRule rule = properties.getRule(scheme.name());
        if (rule == null || !rule.isEnabled() || !rule.getTransactionTypes().contains(transactionType)) {
            return errors;
        }

        Map<String, String> tags;
        try {
            tags = requestContext.getParsedTlv(rule.getDataField());
        } catch (Exception ex) {
            errors.add("INVALID_EMI_DATA_FORMAT: Unable to parse DE" + rule.getDataField()
                    + " for installment data");
            return errors;
        }

        String emiData = tags.get(rule.getEmiDataTag());
        if (isBlank(emiData)) {
            if (rule.isRequired()) {
                errors.add("MISSING_EMI_DATA: DE" + rule.getDataField() + " tag "
                        + rule.getEmiDataTag() + " is required for " + scheme + " " + transactionType);
            }
            return errors;
        }

        if (emiData.length() < rule.getMinLength()) {
            errors.add("INVALID_EMI_DATA: DE" + rule.getDataField() + " tag " + rule.getEmiDataTag()
                    + " is too short; expected at least " + rule.getMinLength() + " characters");
            return errors;
        }

        String totalAmount = emiData.substring(0, 12);
        String emiAmount = emiData.substring(12, 24);
        String installments = emiData.substring(24, 26);
        String interestRate = emiData.substring(26, 31);

        validateAmount("INVALID_EMI_TOTAL_AMOUNT", "Total amount", totalAmount, errors);
        validateAmount("INVALID_EMI_AMOUNT", "EMI amount", emiAmount, errors);
        validateInstallments(installments, errors);
        validateInterestRate(interestRate, errors);

        requestContext.setAttribute(ATTR_EMI_TOTAL_AMOUNT, totalAmount);
        requestContext.setAttribute(ATTR_EMI_AMOUNT, emiAmount);
        requestContext.setAttribute(ATTR_EMI_INSTALLMENTS, installments);
        requestContext.setAttribute(ATTR_EMI_INTEREST_RATE, interestRate);
        return errors;
    }

    private void validateAmount(String code, String label, String value, List<String> errors) {
        if (!value.matches("\\d{12}")) {
            errors.add(code + ": " + label + " must be 12 digits");
            return;
        }
        if (new BigDecimal(value).compareTo(BigDecimal.ZERO) <= 0) {
            errors.add(code + ": " + label + " must be greater than zero");
        }
    }

    private void validateInstallments(String value, List<String> errors) {
        if (!value.matches("\\d{2}")) {
            errors.add("INVALID_EMI_INSTALLMENTS: Number of installments must be 2 digits");
            return;
        }
        if (Integer.parseInt(value) <= 0) {
            errors.add("INVALID_EMI_INSTALLMENTS: Number of installments must be greater than zero");
        }
    }

    private void validateInterestRate(String value, List<String> errors) {
        if (!value.matches("\\d{5}")) {
            errors.add("INVALID_EMI_INTEREST_RATE: Interest rate must be 5 digits");
        }
    }

    private boolean isBlank(String value) {
        return value == null || value.trim().isEmpty();
    }
}
