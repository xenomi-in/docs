package com.logicpoint.iso8583.service;

import com.logicpoint.iso8583.context.RequestContext;
import com.logicpoint.iso8583.entity.CanonicalResponseCode;
import com.logicpoint.iso8583.repository.CanonicalResponseCodeRepository;
import com.logicpoint.iso8583.responsecode.CanonicalResponseCodeCacheValue;
import com.logicpoint.iso8583.responsecode.InternalResCode;
import com.logicpoint.iso8583.responsecode.ResolvedResponseCode;
import com.logicpoint.iso8583.responsecode.SchemeResponseView;
import com.logicpoint.iso8583.util.Scheme;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.scheduling.annotation.Scheduled;
import org.springframework.stereotype.Service;

import java.util.HashMap;
import java.util.List;
import java.util.Map;

@Slf4j
@Service
@RequiredArgsConstructor
public class ResponseCodeCacheService {

    private final CanonicalResponseCodeRepository repository;

    private volatile Map<String, CanonicalResponseCodeCacheValue> byInternalCode = Map.of();
    private volatile Map<String, Map<String, SchemeResponseView>> bySchemeThenInternalCode = Map.of();

    @Scheduled(fixedDelayString = "${response.code.cache.refresh.ms:300000}")
    public void refreshCache() {
        List<CanonicalResponseCode> rows = repository.findByActiveTrue();

        Map<String, CanonicalResponseCodeCacheValue> internalCache = new HashMap<>();
        Map<String, Map<String, SchemeResponseView>> schemeCache = new HashMap<>();
        schemeCache.put("RUPAY", new HashMap<>());
        schemeCache.put("VISA", new HashMap<>());
        schemeCache.put("MASTERCARD", new HashMap<>());

        for (CanonicalResponseCode row : rows) {
            CanonicalResponseCodeCacheValue value = CanonicalResponseCodeCacheValue.builder()
                    .internalCode(row.getInternalCode())
                    .canonicalDescription(row.getCanonicalDescription())
                    .isoDefaultResponseCode(row.getIsoDefaultResponseCode())
                    .rupayResponseCode(row.getRupayResponseCode())
                    .rupayDescription(row.getRupayDescription())
                    .visaResponseCode(row.getVisaResponseCode())
                    .visaDescription(row.getVisaDescription())
                    .mastercardResponseCode(row.getMastercardResponseCode())
                    .mastercardDescription(row.getMastercardDescription())
                    .category(row.getCategory())
                    .declineType(row.getDeclineType())
                    .retryable(row.isRetryable())
                    .reversalRequired(row.isReversalRequired())
                    .adviceAction(row.getAdviceAction())
                    .build();

            internalCache.put(row.getInternalCode(), value);

            schemeCache.get("RUPAY").put(row.getInternalCode(),
                    new SchemeResponseView("RUPAY", fallback(row.getRupayResponseCode(), row.getIsoDefaultResponseCode()), fallback(row.getRupayDescription(), row.getCanonicalDescription())));
            schemeCache.get("VISA").put(row.getInternalCode(),
                    new SchemeResponseView("VISA", fallback(row.getVisaResponseCode(), row.getIsoDefaultResponseCode()), fallback(row.getVisaDescription(), row.getCanonicalDescription())));
            schemeCache.get("MASTERCARD").put(row.getInternalCode(),
                    new SchemeResponseView("MASTERCARD", fallback(row.getMastercardResponseCode(), row.getIsoDefaultResponseCode()), fallback(row.getMastercardDescription(), row.getCanonicalDescription())));
        }

        Map<String, Map<String, SchemeResponseView>> immutableSchemeCache = new HashMap<>();
        schemeCache.forEach((scheme, mapping) -> immutableSchemeCache.put(scheme, Map.copyOf(mapping)));

        this.byInternalCode = Map.copyOf(internalCache);
        this.bySchemeThenInternalCode = Map.copyOf(immutableSchemeCache);
        log.info("Loaded {} canonical response codes into cache", byInternalCode.size());
    }

    public CanonicalResponseCodeCacheValue getByInternalCode(String internalCode) {
        return byInternalCode.get(internalCode);
    }

    public SchemeResponseView getBySchemeAndInternalCode(String scheme, String internalCode) {
        Map<String, SchemeResponseView> schemeMappings = bySchemeThenInternalCode.get(normalizeScheme(scheme));
        if (schemeMappings == null) {
            return null;
        }
        return schemeMappings.get(internalCode);
    }

    public ResolvedResponseCode resolve(String internalCode, Scheme scheme) {
        CanonicalResponseCodeCacheValue row = byInternalCode.get(internalCode);
        if (row == null) {
            row = byInternalCode.get("SYSTEM_ERROR");
        }

        if (row == null) {
            return new ResolvedResponseCode(
                    internalCode,
                    scheme != null ? scheme.name() : "DEFAULT",
                    "96",
                    "System malfunction",
                    "ERROR",
                    "TECHNICAL",
                    true,
                    true,
                    "SEND_REVERSAL"
            );
        }

        String schemeName = scheme != null ? scheme.name() : "DEFAULT";
        SchemeResponseView schemeView = getBySchemeAndInternalCode(schemeName, row.getInternalCode());

        return new ResolvedResponseCode(
                row.getInternalCode(),
                schemeName,
                schemeView != null ? schemeView.getResponseCode() : row.getIsoDefaultResponseCode(),
                schemeView != null ? schemeView.getDescription() : row.getCanonicalDescription(),
                row.getCategory(),
                row.getDeclineType(),
                row.isRetryable(),
                row.isReversalRequired(),
                row.getAdviceAction()
        );
    }

    public ResolvedResponseCode resolve(InternalResCode internalCode, Scheme scheme) {
        return resolve(internalCode != null ? internalCode.name() : InternalResCode.SYSTEM_ERROR.name(), scheme);
    }

    public ResolvedResponseCode applyToDe39(Map<Integer, String> fields, InternalResCode internalCode, Scheme scheme) {
        ResolvedResponseCode resolved = resolve(internalCode, scheme);
        fields.put(39, resolved.getResponseCode());
        log.info("Applied response code. internalCode={}, scheme={}, de39={}, description={}",
                resolved.getInternalCode(),
                resolved.getScheme(),
                resolved.getResponseCode(),
                resolved.getDescription());
        return resolved;
    }

    public ResolvedResponseCode applyToDe39(RequestContext requestContext) {
        return applyToDe39(
                requestContext.getFields(),
                requestContext.getInternalResponseCode() != null ? requestContext.getInternalResponseCode() : InternalResCode.SYSTEM_ERROR,
                requestContext.getScheme()
        );
    }

    private String fallback(String value, String fallback) {
        return value == null || value.isBlank() ? fallback : value;
    }

    private String normalizeScheme(String scheme) {
        return scheme == null ? "DEFAULT" : scheme.toUpperCase();
    }
}
