package com.logicpoint.iso8583.service;

import com.logicpoint.iso8583.context.RequestContext;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;

import org.springframework.boot.autoconfigure.AutoConfigureOrder;
import org.springframework.scheduling.annotation.Async;
import org.springframework.stereotype.Service;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

import java.util.Map;
import java.util.concurrent.CompletableFuture;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.Executors;
import java.util.concurrent.ScheduledExecutorService;
import java.util.concurrent.TimeUnit;

import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.beans.factory.annotation.Value;

import jakarta.annotation.PreDestroy;

@Slf4j
@Service
//@RequiredArgsConstructor
public class AutomaticReversalService {
    
    private static final Logger log = LoggerFactory.getLogger(AutomaticReversalService.class);  
    private final AccountingApiService accountingApiService;
    private final ReversalStorageService reversalStorageService;
    @Value("${reversal.active-context-ttl-ms:900000}")
    private long activeContextTtlMs;
    
    @Autowired
    public AutomaticReversalService(AccountingApiService accountingApiService, ReversalStorageService reversalStorageService) {
        this.accountingApiService = accountingApiService;
        this.reversalStorageService = reversalStorageService;
    }
    // Lightweight in-memory tracking - auto-cleanup
    private final Map<String, TransactionContext> activeTransactions = new ConcurrentHashMap<>();
    
    // Background cleanup thread
    private final ScheduledExecutorService cleanupExecutor = Executors.newSingleThreadScheduledExecutor(
        r -> {
            Thread t = new Thread(r, "reversal-cleanup");
            t.setDaemon(true);
            return t;
        });
    
    {
        // Cleanup completed transactions every 5 minutes
        cleanupExecutor.scheduleWithFixedDelay(
            this::cleanupCompletedTransactions,
            5, 5, TimeUnit.MINUTES
        );
    }
    
    /**
     * Fast registration - minimal overhead
     */
    public TransactionContext registerTransaction(RequestContext requestContext) {
        String uniqueTransactionId = requestContext.getUniqueTransactionId();
        TransactionContext context = new TransactionContext(
                uniqueTransactionId,
                uniqueTransactionId,
                requestContext.getScheme() != null ? requestContext.getScheme().name() : null,
                requestContext.getFields());
        activeTransactions.put(uniqueTransactionId, context);
        return context;
    }
    
    /**
     * Fast checkpoint - atomic operation
     */
    public void markAccountingApproved(String uniqueTransactionId, String accountingTxnId) {
        TransactionContext context = activeTransactions.get(uniqueTransactionId);
        if (context != null) {
            context.markAccountingApproved(accountingTxnId);
        }
    }
    
    /**
     * Fast completion - removes from tracking
     */
    public void markResponseSent(String uniqueTransactionId) {
        TransactionContext context = activeTransactions.remove(uniqueTransactionId);
        if (context != null) {
            context.markResponseSent();
        }
    }
    
    /**
     * Async failure handling - zero impact on main thread
     */
    @Async("reversalExecutor")
    public CompletableFuture<Void> handlePostApprovalFailure(
            String uniqueTransactionId, 
            String failurePoint, 
            String failureReason, 
            Exception exception) {
        
        TransactionContext context = activeTransactions.get(uniqueTransactionId);
        if (context == null || !context.requiresReversal()) {
            return CompletableFuture.completedFuture(null);
        }
        
        // Async reversal initiation
        initiateReversalAsync(context, failurePoint, failureReason);
        return CompletableFuture.completedFuture(null);
    }
    
    /**
     * Async reversal - completely non-blocking
     */
    @Async("reversalExecutor")
    private void initiateReversalAsync(TransactionContext context, 
                                      String failurePoint, 
                                      String failureReason) {
        // This runs in separate thread - zero impact on transaction processing
        try {
            if (context.getReversalTriggered().get()) {
                return; // Already triggered
            }
            context.getReversalTriggered().set(true);
            
            // Create and save reversal (async file write if DB down)
           // ReversalRequest request = createReversalRequest(context, failurePoint, failureReason);
            //reversalStorageService.saveReversalRequest(request);
            
            // Execute reversal async
           // executeReversalAsync(request);
            
        } catch (Exception e) {
            log.error("Failed to initiate reversal", e);
        }
    }
    
    // ... rest of methods remain same but use async patterns
    
    /**
     * Cleanup completed transactions to prevent memory leak
     */
    private void cleanupCompletedTransactions() {
        activeTransactions.entrySet().removeIf(entry -> {
            TransactionContext ctx = entry.getValue();
            boolean shouldRemove = ctx.getResponseSent().get() || 
                                 ctx.getReversalTriggered().get() ||
                                 isStale(ctx);
            if (shouldRemove) {
                log.debug("Cleaning up transaction context: {}", entry.getKey());
            }
            return shouldRemove;
        });
    }

    public void retryPendingReversals() {
        log.debug("retryPendingReversals called - no retry logic implemented yet");
        // TODO: implement DB-based retry flow for pending reversal requests
    }

    @PreDestroy
    public void shutdownCleanupExecutor() {
        cleanupExecutor.shutdownNow();
    }

    private boolean isStale(TransactionContext context) {
        return context.getAgeMillis() > activeContextTtlMs || context.getIdleMillis() > activeContextTtlMs;
    }
    
}
