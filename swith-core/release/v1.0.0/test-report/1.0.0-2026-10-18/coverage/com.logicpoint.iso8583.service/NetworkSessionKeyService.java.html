package com.logicpoint.iso8583.service;

import com.logicpoint.iso8583.entity.NetworkSessionKey;
import com.logicpoint.iso8583.repository.NetworkSessionKeyRepository;
import jakarta.transaction.Transactional;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.cache.Cache;
import org.springframework.cache.CacheManager;
import org.springframework.stereotype.Service;

import java.nio.charset.StandardCharsets;
import java.time.LocalDateTime;
import java.util.List;
import java.util.Optional;

@Service
public class NetworkSessionKeyService {

    private static final Logger log = LoggerFactory.getLogger(NetworkSessionKeyService.class);

    private final NetworkSessionKeyRepository repository;
    private final CacheManager cacheManager;

    public NetworkSessionKeyService(NetworkSessionKeyRepository repository, CacheManager cacheManager) {
        this.repository = repository;
        this.cacheManager = cacheManager;
    }

    public String getActiveKeyValue(String siteId, String listenerId, String connectionId, String scheme, String keyType) {
        if (isBlank(siteId) || isBlank(listenerId) || isBlank(connectionId) || isBlank(scheme) || isBlank(keyType)) {
            return null;
        }

        String cacheKey = buildCacheKey(siteId, listenerId, connectionId, scheme, keyType);
        Cache cache = cacheManager.getCache("networkSessionKeyCache");
        if (cache != null) {
            String cached = cache.get(cacheKey, String.class);
            if (cached != null) {
                return cached;
            }
        }

        Optional<NetworkSessionKey> key = repository.findCurrentActiveKey(siteId, listenerId, connectionId, scheme, keyType);
        if (key.isEmpty()) {
            return null;
        }

        String resolved = resolveKeyValue(key.get());

        if (resolved != null && cache != null) {
            cache.put(cacheKey, resolved);
        }

        return resolved;
    }

    public String getActiveKeyValue(String siteId, String listenerId, String scheme, String keyType) {
        if (isBlank(siteId) || isBlank(listenerId) || isBlank(scheme) || isBlank(keyType)) {
            return null;
        }

        String cacheKey = buildListenerCacheKey(siteId, listenerId, scheme, keyType);
        Cache cache = cacheManager.getCache("networkSessionKeyCache");
        if (cache != null) {
            String cached = cache.get(cacheKey, String.class);
            if (cached != null) {
                return cached;
            }
        }

        Optional<NetworkSessionKey> key = repository.findCurrentActiveKeyForListener(siteId, listenerId, scheme, keyType);
        if (key.isEmpty()) {
            return null;
        }

        String resolved = resolveKeyValue(key.get());

        if (resolved != null && cache != null) {
            cache.put(cacheKey, resolved);
        }

        return resolved;
    }

    @Transactional
    public boolean rotateSessionKey(String keyType,
                                    String scheme,
                                    String siteId,
                                    String listenerId,
                                    String connectionId,
                                    String newKey,
                                    String kcv,
                                    String createdBy) {
        if (isBlank(keyType) || isBlank(scheme) || isBlank(siteId) || isBlank(listenerId) || isBlank(connectionId) || isBlank(newKey)) {
            return false;
        }

        List<NetworkSessionKey> existing = repository.findKeyHistory(siteId, listenerId, connectionId, scheme, keyType);
        long nextVersion = 1L;
        LocalDateTime now = LocalDateTime.now();

        for (NetworkSessionKey key : existing) {
            if (key.getKeyVersion() != null && key.getKeyVersion() >= nextVersion) {
                nextVersion = key.getKeyVersion() + 1;
            }
            if ("ACTIVE".equalsIgnoreCase(key.getSessionStatus()) && "A".equalsIgnoreCase(key.getKeyStatus())) {
                key.setSessionStatus("INACTIVE");
                key.setKeyStatus("I");
                key.setUpdatedAt(now);
                repository.save(key);
            }
        }

        NetworkSessionKey sessionKey = new NetworkSessionKey();
        sessionKey.setSiteId(siteId);
        sessionKey.setListenerId(listenerId);
        sessionKey.setConnectionId(connectionId);
        sessionKey.setScheme(scheme);
        sessionKey.setKeyType(keyType);
        sessionKey.setKeyValue(newKey.getBytes(StandardCharsets.US_ASCII));
        sessionKey.setKeyValueTr31(newKey.getBytes(StandardCharsets.US_ASCII));
        sessionKey.setKeyCheckValue(kcv);
        sessionKey.setKeyCheckValueTr31(kcv);
        sessionKey.setKeyStatus("A");
        sessionKey.setSessionStatus("ACTIVE");
        sessionKey.setKeyVersion(nextVersion);
        sessionKey.setActivatedAt(now);
        sessionKey.setCreatedAt(now);
        sessionKey.setUpdatedAt(now);
        sessionKey.setCreatedBy(createdBy);
        repository.save(sessionKey);
        putCacheValue(siteId, listenerId, connectionId, scheme, keyType, newKey);
        log.info("Rotated session key for site={}, listener={}, connection={}, scheme={}, keyType={}",
                siteId, listenerId, connectionId, scheme, keyType);
        return true;
    }

    public void evictSessionKey(String siteId, String listenerId, String connectionId, String scheme, String keyType) {
        Cache cache = cacheManager.getCache("networkSessionKeyCache");
        if (cache != null) {
            cache.evict(buildCacheKey(siteId, listenerId, connectionId, scheme, keyType));
        }
    }

    private void putCacheValue(String siteId, String listenerId, String connectionId, String scheme, String keyType, String keyValue) {
        Cache cache = cacheManager.getCache("networkSessionKeyCache");
        if (cache != null && keyValue != null) {
            cache.put(buildCacheKey(siteId, listenerId, connectionId, scheme, keyType), keyValue);
        }
    }

    private void touchLastUsedAt(NetworkSessionKey sessionKey) {
        LocalDateTime now = LocalDateTime.now();
        LocalDateTime lastUsedAt = sessionKey.getLastUsedAt();
        if (lastUsedAt == null || lastUsedAt.isBefore(now.minusMinutes(1))) {
            sessionKey.setLastUsedAt(now);
            repository.save(sessionKey);
        }
    }

    private String resolveKeyValue(NetworkSessionKey sessionKey) {
        touchLastUsedAt(sessionKey);
        if (sessionKey.getKeyValueTr31() != null && sessionKey.getKeyValueTr31().length > 0) {
            return new String(sessionKey.getKeyValueTr31(), StandardCharsets.US_ASCII);
        }
        if (sessionKey.getKeyValue() != null && sessionKey.getKeyValue().length > 0) {
            return new String(sessionKey.getKeyValue(), StandardCharsets.US_ASCII);
        }
        return null;
    }

    private String buildCacheKey(String siteId, String listenerId, String connectionId, String scheme, String keyType) {
        return String.join("::", siteId, listenerId, connectionId, scheme, keyType);
    }

    private String buildListenerCacheKey(String siteId, String listenerId, String scheme, String keyType) {
        return String.join("::", siteId, listenerId, scheme, keyType);
    }

    private boolean isBlank(String value) {
        return value == null || value.isBlank();
    }
}
