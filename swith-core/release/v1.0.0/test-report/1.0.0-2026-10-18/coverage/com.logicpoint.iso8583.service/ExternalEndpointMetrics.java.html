package com.logicpoint.iso8583.service;

import com.logicpoint.iso8583.config.AccountingApiProperties;
import com.logicpoint.iso8583.hsm.SimpleNettyHsmClient;
import io.micrometer.core.instrument.Gauge;
import io.micrometer.core.instrument.MeterRegistry;
import jakarta.annotation.PostConstruct;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.scheduling.annotation.Scheduled;
import org.springframework.stereotype.Component;

import java.net.HttpURLConnection;
import java.net.URI;
import java.time.Instant;
import java.util.concurrent.atomic.AtomicInteger;
import java.util.concurrent.atomic.AtomicLong;

/** Prometheus gauges for downstream availability and the custom Netty HSM pool. */
@Component
public class ExternalEndpointMetrics {

    private static final Logger log = LoggerFactory.getLogger(ExternalEndpointMetrics.class);

    private final HsmKeepAliveServiceInterface hsmHealth;
    private final SimpleNettyHsmClient hsmClient;
    private final AccountingApiProperties accounting;
    private final MeterRegistry registry;
    private final AtomicInteger accountingUp = new AtomicInteger(0);
    private final AtomicLong accountingProbeDurationMs = new AtomicLong(0);
    private final AtomicLong accountingLastCheckEpochSeconds = new AtomicLong(0);
    private final AtomicLong accountingLastSuccessEpochSeconds = new AtomicLong(0);
    private final AtomicInteger accountingHttpStatus = new AtomicInteger(0);
    private final AtomicInteger hsmPreviousUp = new AtomicInteger(0);
    private final AtomicInteger accountingPreviousUp = new AtomicInteger(0);
    private final AtomicLong hsmConnectedSinceEpochSeconds = new AtomicLong(0);
    private final AtomicLong accountingConnectedSinceEpochSeconds = new AtomicLong(0);

    @Value("${accounting.api.enabled:false}")
    private boolean accountingEnabled;

    @Value("${hsm.futurex.host:unknown}")
    private String hsmHost;

    @Value("${hsm.futurex.port:4121}")
    private int hsmPort;

    @Value("${hsm.futurex.ssl-enabled:true}")
    private boolean hsmSslEnabled;

    @Value("${monitoring.external-endpoints.accounting.connect-timeout-ms:2000}")
    private int accountingProbeTimeoutMs;

    @Value("${monitoring.external-endpoints.accounting.read-timeout-ms:3000}")
    private int accountingProbeReadTimeoutMs;

    @Value("${monitoring.external-endpoints.accounting.path:}")
    private String accountingHealthPath;

    public ExternalEndpointMetrics(HsmKeepAliveServiceInterface hsmHealth,
                                   SimpleNettyHsmClient hsmClient,
                                   AccountingApiProperties accounting,
                                   MeterRegistry registry) {
        this.hsmHealth = hsmHealth;
        this.hsmClient = hsmClient;
        this.accounting = accounting;
        this.registry = registry;
    }

    @PostConstruct
    void bindMetrics() {
        URI accountingUri = buildAccountingProbeUri();
        String accountingPort = Integer.toString(resolvePort(accountingUri));
        String hsmAddress = hsmHost + ":" + hsmPort;
        String accountingAddress = accountingUri.getHost() + ":" + accountingPort;

        Gauge.builder("switch_external_endpoint_up", hsmHealth,
                        health -> health.getHealthyConnectionCount() > 0 ? 1 : 0)
                .description("Whether the external endpoint is reachable (1) or unavailable (0)")
                .tag("endpoint", "hsm")
                .tag("address", hsmAddress)
                .register(registry);
        Gauge.builder("switch_external_endpoint_up", accountingUp, AtomicInteger::get)
                .description("Whether the external endpoint is reachable (1) or unavailable (0)")
                .tag("endpoint", "accounting")
                .tag("address", accountingAddress)
                .register(registry);
        Gauge.builder("switch_external_endpoint_configured", this,
                        ignored -> accountingEnabled ? 1 : 0)
                .tag("endpoint", "accounting")
                .register(registry);

        Gauge.builder("switch_external_endpoint_info", this, ignored -> 1)
                .description("Configured external endpoint details")
                .tags("endpoint", "hsm",
                        "scheme", hsmSslEnabled ? "tls" : "tcp",
                        "host", hsmHost,
                        "port", Integer.toString(hsmPort),
                        "path", "")
                .register(registry);
        Gauge.builder("switch_external_endpoint_info", this, ignored -> 1)
                .description("Configured external endpoint details")
                .tags("endpoint", "accounting",
                        "scheme", nullToEmpty(accountingUri.getScheme()),
                        "host", nullToEmpty(accountingUri.getHost()),
                        "port", accountingPort,
                        "path", nullToEmpty(accountingUri.getPath()))
                .register(registry);

        bindHsmPoolGauge("active", "Connections currently borrowed from the HSM pool",
                SimpleNettyHsmClient::getBorrowedConnectionCount);
        bindHsmPoolGauge("idle", "Idle connections in the HSM pool",
                SimpleNettyHsmClient::getIdleConnectionCount);
        bindHsmPoolGauge("total", "Total open HSM connections",
                SimpleNettyHsmClient::getTotalConnectionCount);
        bindHsmPoolGauge("max", "Maximum HSM pool size",
                SimpleNettyHsmClient::getMaxConnectionCount);
        bindHsmPoolGauge("pending", "Callers waiting for an HSM connection",
                SimpleNettyHsmClient::getPendingConnectionCount);

        Gauge.builder("switch_external_endpoint_probe_duration_milliseconds",
                        accountingProbeDurationMs, AtomicLong::get)
                .tag("endpoint", "accounting").register(registry);
        Gauge.builder("switch_external_endpoint_last_check_timestamp_seconds",
                        hsmHealth, health -> health.getLastActivityTime() > 0
                                ? health.getLastActivityTime() / 1000.0
                                : 0)
                .description("Unix timestamp when endpoint connectivity was last checked")
                .tag("endpoint", "hsm")
                .tag("address", hsmAddress)
                .register(registry);
        Gauge.builder("switch_external_endpoint_last_check_timestamp_seconds",
                        accountingLastCheckEpochSeconds, AtomicLong::get)
                .description("Unix timestamp when endpoint connectivity was last checked")
                .tag("endpoint", "accounting")
                .tag("address", accountingAddress)
                .register(registry);
        Gauge.builder("switch_external_endpoint_last_success_timestamp_seconds",
                        accountingLastSuccessEpochSeconds, AtomicLong::get)
                .tag("endpoint", "accounting").register(registry);
        Gauge.builder("switch_external_endpoint_connected_since_timestamp_seconds",
                        hsmConnectedSinceEpochSeconds, AtomicLong::get)
                .description("Unix timestamp when the current endpoint connection became available; zero while down")
                .tag("endpoint", "hsm")
                .tag("address", hsmAddress)
                .register(registry);
        Gauge.builder("switch_external_endpoint_connected_since_timestamp_seconds",
                        accountingConnectedSinceEpochSeconds, AtomicLong::get)
                .description("Unix timestamp when the current endpoint connection became available; zero while down")
                .tag("endpoint", "accounting")
                .tag("address", accountingAddress)
                .register(registry);
        Gauge.builder("switch_external_endpoint_http_status", accountingHttpStatus, AtomicInteger::get)
                .description("HTTP status returned by the last Accounting API probe; zero means no response")
                .tag("endpoint", "accounting").register(registry);
    }

    @Scheduled(fixedDelayString = "${monitoring.external-endpoints.hsm.connection-state-interval-ms:1000}",
               initialDelayString = "${monitoring.external-endpoints.hsm.connection-state-initial-delay-ms:1000}")
    void trackHsmConnectionState() {
        updateConnectedSince(hsmPreviousUp, hsmConnectedSinceEpochSeconds,
                hsmHealth.getHealthyConnectionCount() > 0);
    }

    private void bindHsmPoolGauge(String state, String description,
                                  java.util.function.ToDoubleFunction<SimpleNettyHsmClient> value) {
        Gauge.builder("switch_hsm_connections", hsmClient, value)
                .description(description)
                .tag("state", state)
                .register(registry);
    }

    /** HTTP reachability probe; it never invokes a transactional API operation. */
    @Scheduled(fixedDelayString = "${monitoring.external-endpoints.accounting.interval-ms:30000}",
               initialDelayString = "${monitoring.external-endpoints.accounting.initial-delay-ms:5000}")
    void probeAccounting() {
        if (!accountingEnabled) {
            accountingUp.set(0);
            accountingHttpStatus.set(0);
            accountingLastCheckEpochSeconds.set(Instant.now().getEpochSecond());
            updateConnectedSince(accountingPreviousUp, accountingConnectedSinceEpochSeconds, false);
            return;
        }

        long started = System.nanoTime();
        HttpURLConnection connection = null;
        try {
            URI uri = buildAccountingProbeUri();
            connection = (HttpURLConnection) uri.toURL().openConnection();
            connection.setRequestMethod("GET");
            connection.setConnectTimeout(accountingProbeTimeoutMs);
            connection.setReadTimeout(accountingProbeReadTimeoutMs);
            connection.setInstanceFollowRedirects(false);
            connection.setRequestProperty("Accept", "application/json");

            int status = connection.getResponseCode();
            accountingHttpStatus.set(status);
            // Any non-server response proves the HTTP API itself was reached. Authentication
            // failures and a missing optional health path are therefore still reachable.
            boolean reachable = status >= 200 && status < 500;
            accountingUp.set(reachable ? 1 : 0);
            updateConnectedSince(accountingPreviousUp, accountingConnectedSinceEpochSeconds, reachable);
            if (reachable) {
                accountingLastSuccessEpochSeconds.set(Instant.now().getEpochSecond());
            }
        } catch (Exception ex) {
            accountingUp.set(0);
            accountingHttpStatus.set(0);
            updateConnectedSince(accountingPreviousUp, accountingConnectedSinceEpochSeconds, false);
            log.warn("Accounting API HTTP probe failed: {}", ex.getMessage());
        } finally {
            if (connection != null) {
                connection.disconnect();
            }
            accountingLastCheckEpochSeconds.set(Instant.now().getEpochSecond());
            accountingProbeDurationMs.set((System.nanoTime() - started) / 1_000_000);
        }
    }

    private void updateConnectedSince(AtomicInteger previousState,
                                      AtomicLong connectedSinceEpochSeconds,
                                      boolean connected) {
        int current = connected ? 1 : 0;
        int previous = previousState.getAndSet(current);
        if (current == 0) {
            connectedSinceEpochSeconds.set(0);
        } else if (previous == 0) {
            connectedSinceEpochSeconds.set(Instant.now().getEpochSecond());
        }
    }

    private URI buildAccountingProbeUri() {
        String baseUrl = accounting.getBaseUrl().replaceAll("/+$", "");
        if (accountingHealthPath == null || accountingHealthPath.isBlank()) {
            return URI.create(baseUrl);
        }
        String path = accountingHealthPath.startsWith("/")
                ? accountingHealthPath
                : "/" + accountingHealthPath;
        return URI.create(baseUrl + path);
    }

    private int resolvePort(URI uri) {
        if (uri.getPort() >= 0) {
            return uri.getPort();
        }
        return "https".equalsIgnoreCase(uri.getScheme()) ? 443 : 80;
    }

    private String nullToEmpty(String value) {
        return value == null ? "" : value;
    }
}
