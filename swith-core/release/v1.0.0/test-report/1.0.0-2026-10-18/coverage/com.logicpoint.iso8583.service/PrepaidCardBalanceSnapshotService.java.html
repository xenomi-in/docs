package com.logicpoint.iso8583.service;

import com.logicpoint.iso8583.entity.PrepaidCardBalanceSnapshot;
import com.logicpoint.iso8583.repository.PrepaidCardBalanceSnapshotRepository;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.math.BigDecimal;
import java.math.RoundingMode;
import java.time.LocalDate;
import java.time.ZoneId;
import java.util.Optional;

@Service
public class PrepaidCardBalanceSnapshotService {

    private final PrepaidCardBalanceSnapshotRepository repository;
    private final ZoneId businessZone;
    private final BigDecimal openingLedgerOffset;

    public PrepaidCardBalanceSnapshotService(
            PrepaidCardBalanceSnapshotRepository repository,
            @Value("${prepaid-balance-snapshot.time-zone:Asia/Kolkata}") String timeZone,
            @Value("${prepaid-balance-snapshot.opening-ledger-offset:1.00}") BigDecimal openingLedgerOffset) {
        this.repository = repository;
        this.businessZone = ZoneId.of(timeZone);
        this.openingLedgerOffset = openingLedgerOffset;
    }

    @Transactional
    public PrepaidCardBalanceSnapshot updateFromAccountingAvailable(
            String panLookupHmac,
            Integer cardId,
            String currencyCode,
            BigDecimal accountingAvailableBalance,
            BigDecimal accountingLedgerBalance,
            BigDecimal transactionAmount,
            boolean debit) {
        LocalDate businessDate = LocalDate.now(businessZone);
        BigDecimal availableBalance = normalize(accountingAvailableBalance);
        String currency = currencyCode == null || currencyCode.isBlank() ? "356" : currencyCode;

        Optional<PrepaidCardBalanceSnapshot> existing = repository
                .findByPanLookupHmacAndBusinessDateAndCurrencyCode(panLookupHmac, businessDate, currency);
        PrepaidCardBalanceSnapshot snapshot = existing
                .orElseGet(() -> createOpeningSnapshot(
                        panLookupHmac,
                        cardId,
                        businessDate,
                        currency,
                        availableBalance,
                        accountingLedgerBalance));

        snapshot.setCardId(cardId);
        if (existing.isPresent()) {
            snapshot.setAvailableBalance(applyTransactionAmount(
                    snapshot.getAvailableBalance(),
                    transactionAmount,
                    debit));
        }
        return repository.save(snapshot);
    }

    public PrepaidCardBalanceSnapshot updateFromAccountingAvailable(
            String panLookupHmac,
            Integer cardId,
            String currencyCode,
            BigDecimal accountingAvailableBalance,
            BigDecimal accountingLedgerBalance) {
        return updateFromAccountingAvailable(
                panLookupHmac,
                cardId,
                currencyCode,
                accountingAvailableBalance,
                accountingLedgerBalance,
                BigDecimal.ZERO,
                true);
    }

    private PrepaidCardBalanceSnapshot createOpeningSnapshot(
            String panLookupHmac,
            Integer cardId,
            LocalDate businessDate,
            String currencyCode,
            BigDecimal availableBalance,
            BigDecimal accountingLedgerBalance) {
        PrepaidCardBalanceSnapshot snapshot = new PrepaidCardBalanceSnapshot();
        snapshot.setPanLookupHmac(panLookupHmac);
        snapshot.setCardId(cardId);
        snapshot.setBusinessDate(businessDate);
        snapshot.setCurrencyCode(currencyCode);
        snapshot.setLedgerBalance(resolveOpeningLedger(
                panLookupHmac,
                currencyCode,
                businessDate,
                availableBalance,
                accountingLedgerBalance));
        snapshot.setAvailableBalance(availableBalance);
        return snapshot;
    }

    private BigDecimal resolveOpeningLedger(
            String panLookupHmac,
            String currencyCode,
            LocalDate businessDate,
            BigDecimal availableBalance,
            BigDecimal accountingLedgerBalance) {
        if (accountingLedgerBalance != null) {
            return normalize(accountingLedgerBalance);
        }
        return repository.findFirstByPanLookupHmacAndCurrencyCodeAndBusinessDateBeforeOrderByBusinessDateDesc(
                        panLookupHmac,
                        currencyCode,
                        businessDate)
                .map(PrepaidCardBalanceSnapshot::getAvailableBalance)
                .map(this::normalize)
                .orElseGet(() -> normalize(availableBalance.add(openingLedgerOffset)));
    }

    private BigDecimal applyTransactionAmount(BigDecimal currentAvailable, BigDecimal transactionAmount, boolean debit) {
        BigDecimal amount = normalize(transactionAmount);
        BigDecimal available = normalize(currentAvailable);
        return debit ? normalize(available.subtract(amount)) : normalize(available.add(amount));
    }

    private BigDecimal normalize(BigDecimal amount) {
        return (amount == null ? BigDecimal.ZERO : amount).setScale(2, RoundingMode.DOWN);
    }
}
