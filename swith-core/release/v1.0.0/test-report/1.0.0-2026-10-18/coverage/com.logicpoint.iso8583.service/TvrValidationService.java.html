package com.logicpoint.iso8583.service;

import com.logicpoint.iso8583.emv.TerminalVerificationResults;
import org.springframework.stereotype.Service;
import java.util.List;
import java.util.ArrayList;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

@Service
public class TvrValidationService {
    
    /**
     * Validates TVR and determines if transaction should proceed
     * @param tvr Terminal Verification Results
     * @return TvrValidationResult
     */
    private static final Logger log = LoggerFactory.getLogger(TvrValidationService.class);
    public boolean validateTvr(TerminalVerificationResults tvr) {
        return validateTvr(tvr, false, true, true);
    }

    public boolean validateTvr(TerminalVerificationResults tvr,
                               boolean declineOnSdaFailure,
                               boolean declineOnDdaFailure,
                               boolean declineOnCdaFailure) {
        List<String> errors = new ArrayList<>();
        
        // Critical failures - transaction must be declined
        if (tvr.panInHotlist()) {
            log.error("PAN in hotlist - card blocked");
            return false;
        }
        
        if (tvr.expiredApplication()) {
            log.error("Application expired");
            return false;
        }
        
        if (tvr.iccDataMissing()) {
            log.error("ICC data missing");
            return false;
        }

        if (declineOnSdaFailure && tvr.sdaFailed()) {
            log.error("Static data authentication failed");
            return false;
        }
        
        // Authentication failures
        if (declineOnDdaFailure && tvr.ddaFailed()) {
            log.error("Dynamic data authentication failed");
            return false;
        }

        if (declineOnCdaFailure && tvr.cdaFailed()) {
            log.error("Combined data authentication failed");
            return false;
        }
        
        // CVM failures (from TVR Byte 3)
        if (tvr.cardholderVerificationNotSuccessful()) {
            log.error("Cardholder verification not successful");
            return false;
        }
        
        if (tvr.pinTryLimitExceeded()) {
            log.error("PIN try limit exceeded");
            return false;
        }
        
        // Application issues
        if (tvr.applicationNotEffective()) {
            errors.add("Application not yet effective");
        }
        
        if (tvr.serviceNotAllowedForCardProduct()) {
            errors.add("Service not allowed for this card product");
        }
        
        // Issuer authentication failures
        if (tvr.issuerAuthenticationFailed()) {
            errors.add("Issuer authentication failed");
        }
        
        if (!errors.isEmpty()) {
            log.error("TVR validation failed: " + String.join(", ", errors));
            return false;
        }
        
        // Warnings - transaction may proceed with conditions
        List<String> warnings = new ArrayList<>();
        if (tvr.newCard()) {
            warnings.add("New card detected");
        }
        
        if (tvr.transactionExceedsFloorLimit()) {
            warnings.add("Transaction exceeds floor limit - online auth required");
        }
        
        log.info("TVR validation passed");
        return true;
    }
    
    /**
     * Checks if TVR requires online authorization
     */
    public boolean requiresOnlineAuthorization(TerminalVerificationResults tvr) {
        return tvr.transactionExceedsFloorLimit() ||
               tvr.lowerConsecutiveOfflineLimitExceeded() ||
               tvr.upperConsecutiveOfflineLimitExceeded() ||
               tvr.transactionSelectedRandomlyOnlineProcessing() ||
               tvr.merchantForcedTransactionOnline() ||
               tvr.cardholderVerificationNotSuccessful() ||
               tvr.pinTryLimitExceeded();
    }
    
    
    public static void main(String[] args) {
        TerminalVerificationResults tvr = new TerminalVerificationResults("0000048000");
        System.out.println("TVR: " + tvr.toString());
        System.out.println("TVR validation: " + new TvrValidationService().validateTvr(tvr));
    }
}
