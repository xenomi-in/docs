package com.logicpoint.iso8583.service;

import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Service;

import com.logicpoint.iso8583.entity.CardData;

@Service
public class CardDataEncryptionService {

    /** CMS onboarding must never silently fall back to plaintext persistence. */
    public void prepareProtectedForPersistence(CardData cardData, Integer institutionId) {
        if (!encryptionEnabled || dualWritePlaintext || cardData.getAuth_id() == null) {
            throw new IllegalStateException("CMS registration requires encryption, stable identity and plaintext dual-write disabled");
        }
        prepareForPersistence(cardData, institutionId);
        if (cardData.getPan_lookup_hmac() == null || cardData.getPan() != null) {
            throw new IllegalStateException("Protected PAN lookup could not be created");
        }
    }

    private final AesGcmCryptoService aesGcmCryptoService;
    private final ChdKeyCacheService chdKeyCacheService;

    @Value("${chd.encryption.enabled:false}")
    private boolean encryptionEnabled;

    @Value("${chd.encryption.dual-write-plaintext:true}")
    private boolean dualWritePlaintext;

    public CardDataEncryptionService(AesGcmCryptoService aesGcmCryptoService,
                                     ChdKeyCacheService chdKeyCacheService) {
        this.aesGcmCryptoService = aesGcmCryptoService;
        this.chdKeyCacheService = chdKeyCacheService;
    }

    public void prepareForPersistence(CardData cardData) {
        prepareForPersistence(cardData, null);
    }

    public void prepareForPersistence(CardData cardData, Integer institutionId) {
        if (cardData == null) {
            return;
        }

        if (cardData.getPan() != null) {
            cardData.setMasked_pan(maskPan(cardData.getPan()));
        }
        cardData.setCvv(null);

        if (!encryptionEnabled) {
            return;
        }

        if (cardData.getPan() != null && !cardData.getPan().isBlank()) {
            EncryptedFieldValue encryptedPan = encryptCardValue(cardData.getPan(), institutionId, buildAad(cardData, "pan"));
            if (encryptedPan == null) {
                throw new IllegalStateException("PAN encryption could not be completed");
            }
            cardData.setPan_ct(encryptedPan.getCiphertext());
            cardData.setPan_iv(encryptedPan.getIv());
            cardData.setPan_tag(encryptedPan.getTag());
            cardData.setPan_key_version(encryptedPan.getKeyVersion());
            cardData.setPan_lookup_hmac(computePanLookupHmac(cardData.getPan(), institutionId));
            if (!dualWritePlaintext) {
                cardData.setPan(null);
            }
        }

        if (cardData.getExpiryDate() != null && !cardData.getExpiryDate().isBlank()) {
            EncryptedFieldValue encryptedExpiry = encryptCardValue(cardData.getExpiryDate(), institutionId, buildAad(cardData, "expiry"));
            if (encryptedExpiry == null) {
                throw new IllegalStateException("Expiry encryption could not be completed");
            }
            cardData.setExpiry_ct(encryptedExpiry.getCiphertext());
            cardData.setExpiry_iv(encryptedExpiry.getIv());
            cardData.setExpiry_tag(encryptedExpiry.getTag());
            cardData.setExpiry_key_version(encryptedExpiry.getKeyVersion());
            if (!dualWritePlaintext) {
                cardData.setExpiryDate(null);
            }
        }

        if (cardData.getPvv_offest() != null && !cardData.getPvv_offest().isBlank()) {
            EncryptedFieldValue encryptedPvv = encryptCardValue(cardData.getPvv_offest(), institutionId, buildAad(cardData, "pvv"));
            if (encryptedPvv == null) {
                throw new IllegalStateException("PVV encryption could not be completed");
            }
            cardData.setPvv_ct(encryptedPvv.getCiphertext());
            cardData.setPvv_iv(encryptedPvv.getIv());
            cardData.setPvv_tag(encryptedPvv.getTag());
            cardData.setPvv_key_version(encryptedPvv.getKeyVersion());
            if (!dualWritePlaintext) {
                cardData.setPvv_offest(null);
            }
        }

        if (cardData.getOld_pvv_offest() != null && !cardData.getOld_pvv_offest().isBlank()) {
            EncryptedFieldValue encryptedOldPvv = encryptCardValue(cardData.getOld_pvv_offest(), institutionId, buildAad(cardData, "old_pvv"));
            if (encryptedOldPvv == null) {
                throw new IllegalStateException("Old PVV encryption could not be completed");
            }
            cardData.setOld_pvv_ct(encryptedOldPvv.getCiphertext());
            cardData.setOld_pvv_iv(encryptedOldPvv.getIv());
            cardData.setOld_pvv_tag(encryptedOldPvv.getTag());
            cardData.setOld_pvv_key_version(encryptedOldPvv.getKeyVersion());
            if (!dualWritePlaintext) {
                cardData.setOld_pvv_offest(null);
            }
        }
    }

    public void hydrateDecryptedFields(CardData cardData) {
        hydrateDecryptedFields(cardData, null);
    }

    public void hydrateDecryptedFields(CardData cardData, Integer institutionId) {
        if (cardData == null || !encryptionEnabled) {
            return;
        }

        if ((cardData.getPan() == null || cardData.getPan().isBlank())
                && cardData.getPan_ct() != null && cardData.getPan_iv() != null
                && cardData.getPan_tag() != null && cardData.getPan_key_version() != null) {
            String decryptedPan = decryptCardValue(cardData.getPan_ct(), cardData.getPan_iv(), cardData.getPan_tag(), institutionId, cardData.getPan_key_version(), buildAad(cardData, "pan"));
            if (decryptedPan != null) {
                cardData.setPan(decryptedPan);
            }
        }

        if ((cardData.getExpiryDate() == null || cardData.getExpiryDate().isBlank())
                && cardData.getExpiry_ct() != null && cardData.getExpiry_iv() != null
                && cardData.getExpiry_tag() != null && cardData.getExpiry_key_version() != null) {
            String decryptedExpiry = decryptCardValue(cardData.getExpiry_ct(), cardData.getExpiry_iv(), cardData.getExpiry_tag(), institutionId, cardData.getExpiry_key_version(), buildAad(cardData, "expiry"));
            if (decryptedExpiry != null) {
                cardData.setExpiryDate(decryptedExpiry);
            }
        }

        if ((cardData.getPvv_offest() == null || cardData.getPvv_offest().isBlank())
                && cardData.getPvv_ct() != null && cardData.getPvv_iv() != null
                && cardData.getPvv_tag() != null && cardData.getPvv_key_version() != null) {
            String decryptedPvv = decryptCardValue(cardData.getPvv_ct(), cardData.getPvv_iv(), cardData.getPvv_tag(), institutionId, cardData.getPvv_key_version(), buildAad(cardData, "pvv"));
            if (decryptedPvv != null) {
                cardData.setPvv_offest(decryptedPvv);
            }
        }

        if ((cardData.getOld_pvv_offest() == null || cardData.getOld_pvv_offest().isBlank())
                && cardData.getOld_pvv_ct() != null && cardData.getOld_pvv_iv() != null
                && cardData.getOld_pvv_tag() != null && cardData.getOld_pvv_key_version() != null) {
            String decryptedOldPvv = decryptCardValue(cardData.getOld_pvv_ct(), cardData.getOld_pvv_iv(), cardData.getOld_pvv_tag(), institutionId, cardData.getOld_pvv_key_version(), buildAad(cardData, "old_pvv"));
            if (decryptedOldPvv != null) {
                cardData.setOld_pvv_offest(decryptedOldPvv);
            }
        }
    }

    public String computePanLookupHmac(String pan) {
        return computePanLookupHmac(pan, null);
    }

    public String computePanLookupHmac(String pan, Integer institutionId) {
        if (pan == null || !encryptionEnabled) {
            return null;
        }

        if (institutionId == null) {
            if (!chdKeyCacheService.isPanLookupReady()) {
                return null;
            }
            return aesGcmCryptoService.hmacHex(pan, chdKeyCacheService.getActivePanLookupKey());
        }

        byte[] panLookupKey = chdKeyCacheService.getActiveKey(institutionId, ChdKeyCacheService.PAN_HMAC_KEY_TYPE);
        if (panLookupKey == null) {
            return null;
        }
        return aesGcmCryptoService.hmacHex(pan, panLookupKey);
    }

    public EncryptedFieldValue encryptPan(String pan, Integer institutionId, String aad) {
        if (pan == null || pan.isBlank()) {
            return null;
        }
        return encryptCardValue(pan, institutionId, aad);
    }

    public String maskPanForDisplay(String pan) {
        return maskPan(pan);
    }

    private EncryptedFieldValue encryptCardValue(String value, Integer institutionId, String aad) {
        byte[] key;
        Integer keyVersion;

        if (institutionId == null) {
            key = chdKeyCacheService.getActiveCardKey();
            keyVersion = chdKeyCacheService.getActiveCardKeyVersion();
            if (key == null || keyVersion == null) {
                return null;
            }
        } else {
            key = chdKeyCacheService.getActiveKey(institutionId, ChdKeyCacheService.CARD_KEY_TYPE);
            keyVersion = chdKeyCacheService.getActiveKeyVersion(institutionId, ChdKeyCacheService.CARD_KEY_TYPE);
            if (key == null || keyVersion == null) {
                return null;
            }
        }
        return aesGcmCryptoService.encrypt(value, key, keyVersion, aad);
    }

    private String decryptCardValue(byte[] ciphertext, byte[] iv, byte[] tag, Integer institutionId, Integer keyVersion, String aad) {
        byte[] key = institutionId == null
                ? chdKeyCacheService.getKeyByTypeAndVersion(ChdKeyCacheService.CARD_KEY_TYPE, keyVersion)
                : chdKeyCacheService.getKeyByInstitutionAndVersion(institutionId, ChdKeyCacheService.CARD_KEY_TYPE, keyVersion);
        if (key == null) {
            return null;
        }
        return aesGcmCryptoService.decrypt(ciphertext, iv, tag, key, aad);
    }

    private String buildAad(CardData cardData, String fieldName) {
        return "card_data|" + (cardData.getAuth_id() == null ? "new" : cardData.getAuth_id()) + "|" + fieldName;
    }

    private String maskPan(String pan) {
        if (pan == null || pan.length() < 10) {
            return pan;
        }
        return pan.substring(0, 6) + "******" + pan.substring(pan.length() - 4);
    }
}
