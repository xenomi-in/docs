package com.logicpoint.iso8583.service;

import com.logicpoint.iso8583.context.RequestContext;
import com.logicpoint.iso8583.entity.StandingInstructionMandate;
import com.logicpoint.iso8583.entity.StandingInstructionMandate.MandateStatus;
import com.logicpoint.iso8583.repository.StandingInstructionMandateRepository;
import com.logicpoint.iso8583.responsecode.InternalResCode;
import com.logicpoint.iso8583.util.Scheme;
import com.logicpoint.iso8583.validation.TransactionType;
import java.math.BigDecimal;
import java.time.LocalDate;
import java.time.LocalDateTime;
import java.util.ArrayList;
import java.util.List;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

/**
 * Applies persisted standing-instruction mandate state transitions and follow-through checks.
 */
@Service
public class StandingInstructionMandateService {

    public static final String ATTR_SI_MANDATE_STATUS = "si_mandate_status";

    private final StandingInstructionMandateRepository mandateRepository;
    private final StandingInstructionHubClient hubClient;

    public StandingInstructionMandateService(StandingInstructionMandateRepository mandateRepository,
                                             StandingInstructionHubClient hubClient) {
        this.mandateRepository = mandateRepository;
        this.hubClient = hubClient;
    }

    @Transactional
    public List<String> apply(RequestContext requestContext,
                              Scheme scheme,
                              TransactionType transactionType,
                              StandingInstructionAuthorizationValidationService.ParsedSiData siData) {
        List<String> errors = new ArrayList<>();
        String schemeName = scheme.name();
        String mandateId = normalize(siData.mandateId());

        switch (transactionType) {
            case SI_REGISTRATION -> createMandate(requestContext, schemeName, mandateId, siData, errors);
            case SI_EDIT -> updateMandate(requestContext, schemeName, mandateId, siData, errors);
            case SI_TRANSACTION -> validateMandateUse(requestContext, schemeName, mandateId, true, errors);
            case SI_DEREGISTRATION ->
                    transitionMandate(requestContext, schemeName, mandateId, MandateStatus.REVOKED, errors);
            case SI_REVOKEALL -> revokeAll(requestContext, schemeName, errors);
            case SI_PAUSE -> transitionMandate(requestContext, schemeName, mandateId, MandateStatus.PAUSED, errors);
            case SI_START -> transitionMandate(requestContext, schemeName, mandateId, MandateStatus.ACTIVE, errors);
            case SI_PDN -> validateMandateUse(requestContext, schemeName, mandateId, false, errors);
            default -> errors.add("UNSUPPORTED_SI_TRANSACTION_TYPE: " + transactionType);
        }

        return errors;
    }

    private void createMandate(RequestContext requestContext,
                               String scheme,
                               String mandateId,
                               StandingInstructionAuthorizationValidationService.ParsedSiData siData,
                               List<String> errors) {
        if (isBlank(mandateId)) {
            addDecline(requestContext, InternalResCode.INVALID_TRANSACTION,
                    "MISSING_SI_MANDATE_ID: DE48 tag 086 is required to create SI mandate", errors);
            return;
        }

        StandingInstructionMandate mandate = mandateRepository.lockBySchemeAndMandateId(scheme, mandateId)
                .orElseGet(StandingInstructionMandate::new);
        if (mandate.getId() != null && mandate.getStatus() == MandateStatus.ACTIVE) {
            addDecline(requestContext, InternalResCode.INVALID_TRANSACTION,
                    "DUPLICATE_SI_MANDATE: Active mandate already exists for ID " + mandateId, errors);
            return;
        }

        mandate.setScheme(scheme);
        mandate.setMandateId(mandateId);
        mandate.setStatus(MandateStatus.ACTIVE);
        mandate.setRevokedAt(null);
        mandate.setPausedAt(null);
        copyMandateData(mandate, requestContext, siData);
        StandingInstructionMandate saved = mandateRepository.save(mandate);
        hubClient.mandateCreated(saved);
        requestContext.setAttribute(ATTR_SI_MANDATE_STATUS, mandate.getStatus().name());
    }

    private void updateMandate(RequestContext requestContext,
                               String scheme,
                               String mandateId,
                               StandingInstructionAuthorizationValidationService.ParsedSiData siData,
                               List<String> errors) {
        StandingInstructionMandate mandate = lockMandate(requestContext, scheme, mandateId, errors);
        if (mandate == null || !ensureActive(requestContext, mandate, errors)) {
            return;
        }

        copyMandateData(mandate, requestContext, siData);
        StandingInstructionMandate saved = mandateRepository.save(mandate);
        hubClient.mandateUpdated(saved);
        requestContext.setAttribute(ATTR_SI_MANDATE_STATUS, mandate.getStatus().name());
    }

    private void validateMandateUse(RequestContext requestContext,
                                    String scheme,
                                    String mandateId,
                                    boolean recordDebit,
                                    List<String> errors) {
        StandingInstructionMandate mandate = lockMandate(requestContext, scheme, mandateId, errors);
        if (mandate == null || !ensureActive(requestContext, mandate, errors)) {
            return;
        }

        BigDecimal txnAmount = resolveTransactionAmount(requestContext);
        if (txnAmount != null) {
            if (mandate.getMinAmount() != null && txnAmount.compareTo(mandate.getMinAmount()) < 0) {
                addDecline(requestContext, InternalResCode.INVALID_AMOUNT,
                        "SI_AMOUNT_BELOW_MINIMUM: Debit amount is below mandate minimum", errors);
                return;
            }
            if (mandate.getMaxAmount() != null && txnAmount.compareTo(mandate.getMaxAmount()) > 0) {
                addDecline(requestContext, InternalResCode.EXCEEDS_LIMIT,
                        "SI_AMOUNT_ABOVE_MAXIMUM: Debit amount exceeds mandate maximum", errors);
                return;
            }
        }

        LocalDate today = LocalDate.now();
        if (mandate.getStartDate() != null && today.isBefore(mandate.getStartDate())) {
            addDecline(requestContext, InternalResCode.INVALID_TRANSACTION,
                    "SI_MANDATE_NOT_STARTED: Debit is before mandate start date", errors);
            return;
        }
        if (mandate.getEndDate() != null && today.isAfter(mandate.getEndDate())) {
            addDecline(requestContext, InternalResCode.EXPIRED_CARD,
                    "SI_MANDATE_EXPIRED: Debit is after mandate end date", errors);
            return;
        }

        if (recordDebit) {
            mandate.setLastDebitAt(LocalDateTime.now());
            mandate.setLastRrn(requestContext.getFields().get(37));
            mandate.setLastStan(requestContext.getFields().get(11));
            mandate.setLastTransactionId(requestContext.getUniqueTransactionId());
            mandateRepository.save(mandate);
        }
        requestContext.setAttribute(ATTR_SI_MANDATE_STATUS, mandate.getStatus().name());
    }

    private void transitionMandate(RequestContext requestContext,
                                   String scheme,
                                   String mandateId,
                                   MandateStatus targetStatus,
                                   List<String> errors) {
        StandingInstructionMandate mandate = lockMandate(requestContext, scheme, mandateId, errors);
        if (mandate == null) {
            return;
        }

        if (targetStatus == MandateStatus.ACTIVE && mandate.getStatus() != MandateStatus.PAUSED) {
            addDecline(requestContext, InternalResCode.INVALID_TRANSACTION,
                    "SI_MANDATE_CANNOT_START: Only a paused mandate can be started", errors);
            return;
        }
        if (targetStatus == MandateStatus.PAUSED && !ensureActive(requestContext, mandate, errors)) {
            return;
        }

        mandate.setStatus(targetStatus);
        if (targetStatus == MandateStatus.REVOKED) {
            mandate.setRevokedAt(LocalDateTime.now());
        } else if (targetStatus == MandateStatus.PAUSED) {
            mandate.setPausedAt(LocalDateTime.now());
        } else if (targetStatus == MandateStatus.ACTIVE) {
            mandate.setPausedAt(null);
        }
        mandate.setLastRrn(requestContext.getFields().get(37));
        mandate.setLastStan(requestContext.getFields().get(11));
        mandate.setLastTransactionId(requestContext.getUniqueTransactionId());
        StandingInstructionMandate saved = mandateRepository.save(mandate);
        if (targetStatus == MandateStatus.REVOKED) {
            hubClient.mandateRevoked(saved);
        } else if (targetStatus == MandateStatus.PAUSED) {
            hubClient.mandatePaused(saved);
        } else if (targetStatus == MandateStatus.ACTIVE) {
            hubClient.mandateUpdated(saved);
        }
        requestContext.setAttribute(ATTR_SI_MANDATE_STATUS, mandate.getStatus().name());
    }

    private void revokeAll(RequestContext requestContext, String scheme, List<String> errors) {
        String panLookupHmac = requestContext.getPanHash();
        if (isBlank(panLookupHmac)) {
            addDecline(requestContext, InternalResCode.SYSTEM_ERROR,
                    "SI_CANCEL_ALL_UNAVAILABLE: PAN lookup hash is required for cancel-all", errors);
            return;
        }

        List<StandingInstructionMandate> mandates = mandateRepository.lockBySchemeAndPanLookupHmacAndStatusIn(
                scheme,
                panLookupHmac,
                List.of(MandateStatus.ACTIVE, MandateStatus.PAUSED));
        if (mandates.isEmpty()) {
            addDecline(requestContext, InternalResCode.INVALID_TRANSACTION,
                    "SI_CANCEL_ALL_NOT_FOUND: No active or paused mandates found for card", errors);
            return;
        }

        LocalDateTime now = LocalDateTime.now();
        for (StandingInstructionMandate mandate : mandates) {
            mandate.setStatus(MandateStatus.REVOKED);
            mandate.setRevokedAt(now);
            mandate.setLastRrn(requestContext.getFields().get(37));
            mandate.setLastStan(requestContext.getFields().get(11));
            mandate.setLastTransactionId(requestContext.getUniqueTransactionId());
        }
        mandateRepository.saveAll(mandates);
        for (StandingInstructionMandate mandate : mandates) {
            hubClient.mandateRevoked(mandate);
        }
    }

    private StandingInstructionMandate lockMandate(RequestContext requestContext,
                                                   String scheme,
                                                   String mandateId,
                                                   List<String> errors) {
        if (isBlank(mandateId)) {
            addDecline(requestContext, InternalResCode.INVALID_TRANSACTION,
                    "MISSING_SI_MANDATE_ID: DE48 tag 086 is required for SI mandate lookup", errors);
            return null;
        }

        StandingInstructionMandate mandate = mandateRepository.lockBySchemeAndMandateId(scheme, mandateId).orElse(null);
        if (mandate == null) {
            addDecline(requestContext, InternalResCode.INVALID_TRANSACTION,
                    "SI_MANDATE_NOT_FOUND: No mandate found for ID " + mandateId, errors);
            return null;
        }
        requestContext.setAttribute(ATTR_SI_MANDATE_STATUS, mandate.getStatus().name());
        return mandate;
    }

    private boolean ensureActive(RequestContext requestContext, StandingInstructionMandate mandate, List<String> errors) {
        if (mandate.getStatus() == MandateStatus.ACTIVE) {
            return true;
        }

        InternalResCode responseCode = mandate.getStatus() == MandateStatus.REVOKED
                ? InternalResCode.DO_NOT_HONOR
                : InternalResCode.RESTRICTED_CARD;
        addDecline(requestContext, responseCode,
                "SI_MANDATE_NOT_ACTIVE: Mandate " + mandate.getMandateId()
                        + " is " + mandate.getStatus(), errors);
        return false;
    }

    private void copyMandateData(StandingInstructionMandate mandate,
                                 RequestContext requestContext,
                                 StandingInstructionAuthorizationValidationService.ParsedSiData siData) {
        mandate.setPanLookupHmac(requestContext.getPanHash());
        mandate.setInstitutionId(requestContext.getInstitutionId());
        mandate.setCardAcceptorId(requestContext.getFields().get(42));
        mandate.setTerminalId(requestContext.getFields().get(41));
        mandate.setCurrencyCode(requestContext.getFields().get(49));
        mandate.setSiRequestType(siData.requestType());
        mandate.setNumberOfInstallments(siData.numberOfInstallments());
        mandate.setPreferredDay(siData.preferredDay());
        mandate.setMinAmount(siData.minAmount());
        mandate.setMaxAmount(siData.maxAmount());
        mandate.setStartDate(siData.startDate());
        mandate.setEndDate(siData.endDate());
        mandate.setNotificationMode(siData.notificationMode());
        mandate.setFrequencyIndicator(siData.frequencyIndicator());
        mandate.setLastRrn(requestContext.getFields().get(37));
        mandate.setLastStan(requestContext.getFields().get(11));
        mandate.setLastTransactionId(requestContext.getUniqueTransactionId());
    }

    private BigDecimal resolveTransactionAmount(RequestContext requestContext) {
        String amount = requestContext.getFields().get(4);
        if (isBlank(amount) || !amount.matches("\\d+")) {
            return null;
        }
        return new BigDecimal(amount).movePointLeft(2);
    }

    private void addDecline(RequestContext requestContext,
                            InternalResCode responseCode,
                            String error,
                            List<String> errors) {
        requestContext.setInternalResponseCode(responseCode);
        errors.add(error);
    }

    private String normalize(String value) {
        return isBlank(value) ? null : value.trim().toUpperCase();
    }

    private boolean isBlank(String value) {
        return value == null || value.trim().isEmpty();
    }
}
