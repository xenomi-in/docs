package com.logicpoint.iso8583.service;

import com.logicpoint.iso8583.context.RequestContext;
import com.logicpoint.iso8583.entity.Transaction;
import com.logicpoint.iso8583.repository.TransactionRepository;
import com.logicpoint.iso8583.responsecode.InternalResCode;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.stereotype.Service;

import java.util.List;

@Service
public class DuplicateCheckService {

    private static final Logger log = LoggerFactory.getLogger(DuplicateCheckService.class);

    private final TransactionRepository transactionRepository;
    private final ResponseAssembler responseAssembler;

    public DuplicateCheckService(TransactionRepository transactionRepository, ResponseAssembler responseAssembler) {
        this.transactionRepository = transactionRepository;
        this.responseAssembler = responseAssembler;
    }

    public boolean isDuplicate(RequestContext requestContext) {
        String terminalId = requestContext.getFields().get(41);
        String rrn = requestContext.getFields().get(37);
        String transmissionTime = requestContext.getFields().get(7);

        if (terminalId == null || rrn == null || transmissionTime == null) {
            log.warn("Missing required fields for duplicate check - Terminal: {}, RRN: {}, Time: {}", terminalId, rrn, transmissionTime);
            return false;
        }

        List<Transaction> existingTransaction = transactionRepository.findByRrnAndTID(
                terminalId,
                rrn,
                transmissionTime,
                requestContext.getMti()
        );

        if (existingTransaction.isEmpty()) {
            return false;
        }

        log.warn("Duplicate transaction detected - MTI: {}, Time: {}, Terminal: {}, RRN: {}",
                requestContext.getMti(), transmissionTime, terminalId, rrn);
        responseAssembler.setResponseMti(requestContext);
        responseAssembler.applyResponseCode(requestContext, InternalResCode.DUPLICATE_TRANSACTION);
        return true;
    }
}
