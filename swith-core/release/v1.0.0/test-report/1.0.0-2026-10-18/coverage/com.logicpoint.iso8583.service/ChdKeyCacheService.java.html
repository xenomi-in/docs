package com.logicpoint.iso8583.service;

import java.time.LocalDateTime;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Service;

import com.logicpoint.iso8583.entity.KeyManagement;
import com.logicpoint.iso8583.repository.KeyManagementRepository;

import jakarta.annotation.PostConstruct;

@Service
public class ChdKeyCacheService {

    public static final String CARD_KEY_TYPE = "CHD_CARD_AES";
    public static final String TXN_KEY_TYPE = "CHD_TXN_AES";
    public static final String PAN_HMAC_KEY_TYPE = "CHD_PAN_HMAC";

    private static final Logger logger = LoggerFactory.getLogger(ChdKeyCacheService.class);

    private final KeyManagementRepository keyManagementRepository;
    private final NettyHsmService nettyHsmService;

    @Value("${chd.encryption.enabled:false}")
    private boolean encryptionEnabled;

    @Value("${chd.encryption.institution-id:1}")
    private Integer defaultInstitutionId;

    @Value("${chd.encryption.wrapping-key-type:DEK}")
    private String wrappingKeyType;

    private volatile Map<Integer, Map<String, Map<Integer, byte[]>>> keysByInstitutionTypeVersion = Map.of();
    private volatile Map<Integer, Map<String, Integer>> activeVersionsByInstitutionType = Map.of();

    public ChdKeyCacheService(KeyManagementRepository keyManagementRepository, NettyHsmService nettyHsmService) {
        this.keyManagementRepository = keyManagementRepository;
        this.nettyHsmService = nettyHsmService;
    }

    @PostConstruct
    public void loadKeysOnStartup() {
        if (!encryptionEnabled) {
            logger.info("CHD encryption is disabled; skipping CHD key cache load");
            return;
        }
        refresh();
    }

    public void refresh() {
        List<String> managedKeyTypes = List.of(CARD_KEY_TYPE, TXN_KEY_TYPE, PAN_HMAC_KEY_TYPE);
        List<KeyManagement> activeKeys = keyManagementRepository.findActiveInstitutionKeysByKeyTypes(managedKeyTypes, LocalDateTime.now());
        Map<Integer, Map<String, Map<Integer, byte[]>>> resolvedKeys = new HashMap<>();
        Map<Integer, Map<String, Integer>> resolvedVersions = new HashMap<>();

        for (KeyManagement key : activeKeys) {
            byte[] clearKey = unwrapKeyMaterial(key);
            if (clearKey == null) {
                logger.warn("Skipping CHD key {} version {} for institution {} because unwrap failed",
                        key.getKey_type(), key.getKey_version(), key.getInstitution_id());
                continue;
            }

            resolvedKeys
                    .computeIfAbsent(key.getInstitution_id(), ignored -> new HashMap<>())
                    .computeIfAbsent(key.getKey_type(), ignored -> new HashMap<>())
                    .put(key.getKey_version(), clearKey);

            if (Boolean.TRUE.equals(key.getIs_current())) {
                resolvedVersions
                        .computeIfAbsent(key.getInstitution_id(), ignored -> new HashMap<>())
                        .put(key.getKey_type(), key.getKey_version());
            }
        }

        keysByInstitutionTypeVersion = deepCopyKeyCache(resolvedKeys);
        activeVersionsByInstitutionType = deepCopyVersionCache(resolvedVersions);
    }

    public boolean isEncryptionReady() {
        return encryptionEnabled
                && getActiveKey(defaultInstitutionId, CARD_KEY_TYPE) != null
                && getActiveKey(defaultInstitutionId, TXN_KEY_TYPE) != null
                && getActiveKey(defaultInstitutionId, PAN_HMAC_KEY_TYPE) != null;
    }

    public boolean isTransactionEncryptionReady() {
        return encryptionEnabled
                && getActiveKey(defaultInstitutionId, TXN_KEY_TYPE) != null;
    }

    public boolean isPanLookupReady() {
        return encryptionEnabled
                && getActiveKey(defaultInstitutionId, PAN_HMAC_KEY_TYPE) != null;
    }

    public byte[] getActiveCardKey() {
        return getActiveKey(defaultInstitutionId, CARD_KEY_TYPE);
    }

    public byte[] getActiveTransactionKey() {
        return getActiveKey(defaultInstitutionId, TXN_KEY_TYPE);
    }

    public byte[] getActivePanLookupKey() {
        return getActiveKey(defaultInstitutionId, PAN_HMAC_KEY_TYPE);
    }

    public Integer getActiveCardKeyVersion() {
        return getActiveKeyVersion(defaultInstitutionId, CARD_KEY_TYPE);
    }

    public Integer getActiveTransactionKeyVersion() {
        return getActiveKeyVersion(defaultInstitutionId, TXN_KEY_TYPE);
    }

    public Map<String, Object> getStatus(Integer institutionId) {
        Integer targetInstitutionId = institutionId != null ? institutionId : defaultInstitutionId;

        Map<String, Object> status = new HashMap<>();
        status.put("chdEncryptionEnabled", encryptionEnabled);
        status.put("defaultInstitutionId", defaultInstitutionId);
        status.put("institutionId", targetInstitutionId);
        status.put("wrappingKeyType", wrappingKeyType);

        Integer cardKeyVersion = getActiveKeyVersion(targetInstitutionId, CARD_KEY_TYPE);
        Integer txnKeyVersion = getActiveKeyVersion(targetInstitutionId, TXN_KEY_TYPE);
        Integer panHmacKeyVersion = getActiveKeyVersion(targetInstitutionId, PAN_HMAC_KEY_TYPE);

        status.put("cardEncryptionReady", encryptionEnabled && getActiveKey(targetInstitutionId, CARD_KEY_TYPE) != null);
        status.put("transactionEncryptionReady", encryptionEnabled && getActiveKey(targetInstitutionId, TXN_KEY_TYPE) != null);
        status.put("panLookupReady", encryptionEnabled && getActiveKey(targetInstitutionId, PAN_HMAC_KEY_TYPE) != null);
        status.put("cardKeyVersion", cardKeyVersion);
        status.put("transactionKeyVersion", txnKeyVersion);
        status.put("panLookupKeyVersion", panHmacKeyVersion);

        Map<String, Integer> versions = new HashMap<>();
        if (cardKeyVersion != null) {
            versions.put(CARD_KEY_TYPE, cardKeyVersion);
        }
        if (txnKeyVersion != null) {
            versions.put(TXN_KEY_TYPE, txnKeyVersion);
        }
        if (panHmacKeyVersion != null) {
            versions.put(PAN_HMAC_KEY_TYPE, panHmacKeyVersion);
        }
        status.put("activeKeyVersions", versions);
        return status;
    }

    public byte[] getKeyByTypeAndVersion(String keyType, Integer keyVersion) {
        return getKeyByInstitutionAndVersion(defaultInstitutionId, keyType, keyVersion);
    }

    public byte[] getActiveKey(Integer institutionId, String keyType) {
        Integer keyVersion = getActiveKeyVersion(institutionId, keyType);
        return getKeyByInstitutionAndVersion(institutionId, keyType, keyVersion);
    }

    public Integer getActiveKeyVersion(Integer institutionId, String keyType) {
        Map<String, Integer> versionsByType = activeVersionsByInstitutionType.get(institutionId);
        return versionsByType == null ? null : versionsByType.get(keyType);
    }

    public byte[] getKeyByInstitutionAndVersion(Integer institutionId, String keyType, Integer keyVersion) {
        if (institutionId == null || keyType == null || keyVersion == null) {
            return null;
        }

        Map<String, Map<Integer, byte[]>> byType = keysByInstitutionTypeVersion.get(institutionId);
        if (byType == null) {
            return null;
        }
        Map<Integer, byte[]> byVersion = byType.get(keyType);
        return byVersion == null ? null : byVersion.get(keyVersion);
    }

    private byte[] unwrapKeyMaterial(KeyManagement key) {
        if (key == null || key.getKey_value() == null || key.getKey_value().length == 0) {
            return null;
        }

        String wrappingKeyRef = key.getWrapping_key_ref();
        Integer wrappingInstitutionId = extractInstitutionIdFromWrappingKeyRef(key.getInstitution_id(), wrappingKeyRef);
        byte[] clearKey = nettyHsmService.unwrapChdKey(
                key.getKey_value(),
                wrappingInstitutionId,
                wrappingKeyType,
                Boolean.TRUE.equals(key.getIs_tr31()));
        if (clearKey == null || clearKey.length == 0) {
            logger.warn("Unable to unwrap CHD key {} version {} for institution {}",
                    key.getKey_type(), key.getKey_version(), key.getInstitution_id());
        }
        return clearKey;
    }

    private Integer extractInstitutionIdFromWrappingKeyRef(Integer fallbackInstitutionId, String wrappingKeyRef) {
        if (wrappingKeyRef == null || wrappingKeyRef.isBlank()) {
            return fallbackInstitutionId;
        }
        String[] parts = wrappingKeyRef.split(":", 2);
        if (parts.length == 2 && !parts[1].isBlank()) {
            try {
                return Integer.valueOf(parts[1]);
            } catch (NumberFormatException ex) {
                logger.warn("Invalid wrapping key reference {}; falling back to institution {}", wrappingKeyRef, fallbackInstitutionId);
            }
        }
        return fallbackInstitutionId;
    }

    private Map<Integer, Map<String, Map<Integer, byte[]>>> deepCopyKeyCache(Map<Integer, Map<String, Map<Integer, byte[]>>> source) {
        Map<Integer, Map<String, Map<Integer, byte[]>>> copy = new HashMap<>();
        for (Map.Entry<Integer, Map<String, Map<Integer, byte[]>>> institutionEntry : source.entrySet()) {
            Map<String, Map<Integer, byte[]>> typeCopy = new HashMap<>();
            for (Map.Entry<String, Map<Integer, byte[]>> typeEntry : institutionEntry.getValue().entrySet()) {
                Map<Integer, byte[]> versionCopy = new HashMap<>();
                for (Map.Entry<Integer, byte[]> versionEntry : typeEntry.getValue().entrySet()) {
                    byte[] value = versionEntry.getValue();
                    byte[] clone = value == null ? null : value.clone();
                    versionCopy.put(versionEntry.getKey(), clone);
                }
                typeCopy.put(typeEntry.getKey(), Map.copyOf(versionCopy));
            }
            copy.put(institutionEntry.getKey(), Map.copyOf(typeCopy));
        }
        return Map.copyOf(copy);
    }

    private Map<Integer, Map<String, Integer>> deepCopyVersionCache(Map<Integer, Map<String, Integer>> source) {
        Map<Integer, Map<String, Integer>> copy = new HashMap<>();
        for (Map.Entry<Integer, Map<String, Integer>> institutionEntry : source.entrySet()) {
            copy.put(institutionEntry.getKey(), Map.copyOf(new HashMap<>(institutionEntry.getValue())));
        }
        return Map.copyOf(copy);
    }
}
