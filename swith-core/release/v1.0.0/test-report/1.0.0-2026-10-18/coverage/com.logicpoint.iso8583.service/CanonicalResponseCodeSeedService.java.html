    @PostConstruct
    public void seedIfRequired() {
        List<CanonicalResponseCode> seeds = buildSeeds();
        int inserted = 0;
        int skipped = 0;

        for (CanonicalResponseCode seed : seeds) {
            CanonicalResponseCode existing = repository.findById(seed.getInternalCode()).orElse(null);
            if (existing == null) {
                repository.save(seed);
                inserted++;
            } else {
                skipped++;
            }
        }

        log.info("Canonical response code seed complete. inserted={}, skippedExisting={}", inserted, skipped);
        responseCodeCacheService.refreshCache();
    }

    private CanonicalResponseCode seed(
            String internalCode,
            String canonicalDescription,
            String isoDefaultResponseCode,
            String rupayResponseCode,
            String rupayDescription,
            String visaResponseCode,
            String visaDescription,
            String mastercardResponseCode,
            String mastercardDescription,
            String category,
            String declineType,
            boolean retryable,
            boolean reversalRequired,
            String adviceAction) {
        CanonicalResponseCode row = new CanonicalResponseCode();
        LocalDateTime now = LocalDateTime.now();
        row.setInternalCode(internalCode);
        row.setCanonicalDescription(canonicalDescription);
        row.setIsoDefaultResponseCode(isoDefaultResponseCode);
        row.setRupayResponseCode(rupayResponseCode);
        row.setRupayDescription(rupayDescription);
        row.setVisaResponseCode(visaResponseCode);
        row.setVisaDescription(visaDescription);
        row.setMastercardResponseCode(mastercardResponseCode);
        row.setMastercardDescription(mastercardDescription);
        row.setCategory(category);
        row.setDeclineType(declineType);
        row.setRetryable(retryable);
        row.setReversalRequired(reversalRequired);
        row.setAdviceAction(adviceAction);
        row.setActive(true);
        row.setVersion(1);
        row.setCreatedAt(now);
        row.setUpdatedAt(now);
        return row;
    }
