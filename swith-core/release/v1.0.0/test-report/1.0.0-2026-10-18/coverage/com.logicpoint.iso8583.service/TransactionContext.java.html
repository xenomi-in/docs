package com.logicpoint.iso8583.service;

import lombok.Data;
import java.time.LocalDateTime;
import java.util.Map;
import java.util.concurrent.atomic.AtomicBoolean;
import java.util.concurrent.atomic.AtomicReference;

/**
 * Lightweight transaction context - optimized for performance
 */
@Data
public class TransactionContext {
    // Use primitives and atomic references for minimal overhead
    private final String uniqueTransactionId;
    private final AtomicReference<String> accountingTxnId = new AtomicReference<>();
    private final AtomicBoolean accountingApproved = new AtomicBoolean(false);
    private final AtomicBoolean responseSent = new AtomicBoolean(false);
    private final AtomicBoolean reversalTriggered = new AtomicBoolean(false);
    private final long createdAtEpochMillis;
    private final AtomicReference<Long> lastTouchedEpochMillis = new AtomicReference<>();
    
    // Store only essential data - not full field map
    private final String transactionId;
    private final String originalRrn;
    private final String originalStan;
    private final String originalAmount;
    private final String originalMti;
    private final String currency;
    private final String network;
    
    // Lazy load full fields only when reversal needed
    private transient Map<Integer, String> fullFields;
    
    public TransactionContext(String uniqueTransactionId, String transactionId, String network, Map<Integer, String> fields) {
        this.uniqueTransactionId = uniqueTransactionId;
        this.transactionId = transactionId;
        this.originalRrn = fields.get(37);
        this.originalStan = fields.get(11);
        this.originalAmount = fields.get(4);
        this.originalMti = fields.get(0);
        this.currency = fields.get(49);
        this.network = network;
        this.createdAtEpochMillis = System.currentTimeMillis();
        this.lastTouchedEpochMillis.set(createdAtEpochMillis);
        // Store full fields only if reversal might be needed
        this.fullFields = null; // Lazy initialization
    }
    
    public void setFullFieldsForReversal(Map<Integer, String> fields) {
        this.fullFields = new java.util.HashMap<>(fields);
        touch();
    }
    
    public void markAccountingApproved(String accountingTxnId) {
        this.accountingTxnId.set(accountingTxnId);
        this.accountingApproved.set(true);
        touch();
    }
    
    public void markResponseSent() {
        this.responseSent.set(true);
        touch();
    }
    
    public boolean requiresReversal() {
        return accountingApproved.get() && !responseSent.get() && !reversalTriggered.get();
    }
    
    public String getAccountingTransactionId() {
        return accountingTxnId.get();
    }

    public long getAgeMillis() {
        return System.currentTimeMillis() - createdAtEpochMillis;
    }

    public long getIdleMillis() {
        Long lastTouched = lastTouchedEpochMillis.get();
        return lastTouched == null ? getAgeMillis() : System.currentTimeMillis() - lastTouched;
    }

    public void touch() {
        lastTouchedEpochMillis.set(System.currentTimeMillis());
    }
}
