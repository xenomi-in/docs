package com.logicpoint.iso8583.service;

import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.cache.CacheManager;
import org.springframework.stereotype.Service;
import org.springframework.context.ApplicationEventPublisher;
import org.springframework.context.event.EventListener;

/**
 * Service for handling cache updates when data changes in the database.
 * Provides methods to refresh specific caches and handles cache invalidation events.
 */
@Service
public class CacheUpdateService {

    @Autowired
    private CacheManager cacheManager;

    @Autowired
    private CacheInitializationService cacheInitializationService;

    @Autowired
    private ApplicationEventPublisher eventPublisher;

    /**
     * Refresh BIN and Institution cache
     */
    public void refreshBinInstitutionCache() {
        cacheManager.getCache("binInstitutionCache").clear();
        cacheInitializationService.loadBinInstitutionCache();
    }

    /**
     * Refresh Card Data cache
     */
    public void refreshCardDataCache() {
        cacheManager.getCache("cardDataCache").clear();
        cacheInitializationService.loadCardDataCache();
    }

    /**
     * Refresh all caches
     */
    public void refreshAllCaches() {
        cacheManager.getCache("binInstitutionCache").clear();
        cacheManager.getCache("cardDataCache").clear();
        cacheManager.getCache("institutionValidation").clear();
        cacheManager.getCache("cardValidation").clear();
        cacheManager.getCache("serviceCodeCache").clear();
        
        cacheInitializationService.loadBinInstitutionCache();
       // cacheInitializationService.loadCardDataCache();
    }

    /**
     * Publish cache update event
     */
    public void publishCacheUpdateEvent(String cacheName) {
        eventPublisher.publishEvent(new CacheUpdateEvent(cacheName));
    }

    /**
     * Event listener for cache updates
     */
    @EventListener
    public void handleCacheUpdateEvent(CacheUpdateEvent event) {
        switch (event.getCacheName()) {
            case "binInstitutionCache":
                refreshBinInstitutionCache();
                break;
           
            case "all":
                refreshAllCaches();
                break;
        }
    }

    /**
     * Cache update event class
     */
    public static class CacheUpdateEvent {
        private final String cacheName;

        public CacheUpdateEvent(String cacheName) {
            this.cacheName = cacheName;
        }

        public String getCacheName() {
            return cacheName;
        }
    }
}
