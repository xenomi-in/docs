package com.logicpoint.iso8583.service;

import com.logicpoint.iso8583.context.RequestContext;
import com.logicpoint.iso8583.emv.CardholderVerificationMethodResults;
import com.logicpoint.iso8583.emv.ChipData;
import com.logicpoint.iso8583.emv.ChipDataParser;
import com.logicpoint.iso8583.emv.TerminalVerificationResults;
import com.logicpoint.iso8583.entity.BinDetails;
import com.logicpoint.iso8583.mcc.MccRuleEngine;
import com.logicpoint.iso8583.responsecode.InternalResCode;
import com.logicpoint.iso8583.util.Scheme;
import com.logicpoint.iso8583.validation.SchemeValidatorService;
import com.logicpoint.iso8583.validation.TransactionIdentificationRule;
import com.logicpoint.iso8583.validation.TransactionType;
import com.logicpoint.iso8583.validation.ValidationResult;
import com.logicpoint.iso8583.validation.orchestrator.TransactionValidationOrchestrator;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Service;

import java.io.IOException;
import java.util.List;
import java.util.Map;
import java.util.Optional;
import java.util.stream.Collectors;
import java.util.ArrayList;

@Service
public class FinancialRequestProcessor {

    private static final Logger log = LoggerFactory.getLogger(FinancialRequestProcessor.class);

    private final SchemeValidatorService validator;
    private final TransactionValidationOrchestrator validationOrchestrator;
    private final ResponseAssembler responseAssembler;
    private final MccRuleEngine mccRuleEngine;
    private final CacheInitializationService cacheInitializationService;

    public FinancialRequestProcessor(
            TransactionValidationOrchestrator validationOrchestrator,
            ResponseAssembler responseAssembler,
            MccRuleEngine mccRuleEngine,
            CacheInitializationService cacheInitializationService,
            @Value("${scheme.validation.rules-dir:validation_rules}") String validationRulesDir
    ) throws IOException {
        this.validator = new SchemeValidatorService(validationRulesDir);
        this.validationOrchestrator = validationOrchestrator;
        this.responseAssembler = responseAssembler;
        this.mccRuleEngine = mccRuleEngine;
        this.cacheInitializationService = cacheInitializationService;
    }

    public Map<Integer, String> process(RequestContext requestContext) {
        Map<Integer, String> fields = requestContext.getFields();
        responseAssembler.setResponseMti(requestContext);

        if (requestContext.getScheme() != Scheme.RUPAY) {
            responseAssembler.applyResponseCode(requestContext, InternalResCode.DO_NOT_HONOR);
            return fields;
        }

        ValidationOutcome validationOutcome = validateTransactionType(requestContext);
        log.info("Processing financial request for transaction type: {}", validationOutcome.transactionType());

        if (validationOutcome.transactionType() == TransactionType.UNKNOWN) {
            responseAssembler.applyResponseCode(requestContext, InternalResCode.COMPLIANCE_VIOLATION);
            return fields;
        }

        requestContext.setTransactionType(validationOutcome.transactionType());
        requestContext.setAttribute("matchedTransactionType", validationOutcome.transactionType().name());

        if (!validationOutcome.valid()) {
            log.warn("Transaction validation failed. Missing fields: {}", validationOutcome.missingFields());
            responseAssembler.applyResponseCode(requestContext, InternalResCode.FORMAT_ERROR);
            omitPinChangeNewPinBlock(requestContext);
            return fields;
        }

        if (!mccRuleEngine.validate(requestContext)) {
            omitPinChangeNewPinBlock(requestContext);
            return fields;
        }

        ValidationResult validationResult = validationOrchestrator.validateTransaction(requestContext);
        log.info("Validation Result: {}", validationResult.getErrors());

        if (!validationResult.isValid()) {
            responseAssembler.ensureResponseCode(requestContext);
            omitPinChangeNewPinBlock(requestContext);
            return fields;
        }

        responseAssembler.applyValidationUpdates(requestContext, validationResult);
        responseAssembler.applyFinancialApproval(requestContext);
        omitPinChangeNewPinBlock(requestContext);
        log.info("Transaction validation successful");
        return fields;
    }

    public TransactionType identifyTransactionType(RequestContext requestContext) {
        Map<String, String> convertedFields = requestContext.getFields().entrySet().stream()
                .collect(Collectors.toMap(entry -> String.valueOf(entry.getKey()), Map.Entry::getValue));
        return validator.identify(requestContext.getScheme(), convertedFields)
                .map(rule -> rule.transactionType)
                .orElse(TransactionType.UNKNOWN);
    }

    private void omitPinChangeNewPinBlock(RequestContext requestContext) {
        if (requestContext.getTransactionType() == TransactionType.PIN_CHANGE) {
            responseAssembler.omitDe120Tag(requestContext, "004");
        }
    }

    private ValidationOutcome validateTransactionType(RequestContext requestContext) {
        Map<Integer, String> fields = requestContext.getFields();
        Map<String, String> convertedFields = fields.entrySet().stream()
                .collect(Collectors.toMap(entry -> String.valueOf(entry.getKey()), Map.Entry::getValue));
        Optional<TransactionIdentificationRule> rule = validator.identify(requestContext.getScheme(), convertedFields);

        if (rule.isPresent()) {
            List<String> missing = new ArrayList<>(rule.get().validateFields(convertedFields));
            removeOfflinePinMissingDe52IfAllowed(requestContext, missing);
            log.info("Mandatory Fields are Missign");
            return new ValidationOutcome(rule.get().transactionType, missing);
        }

        return new ValidationOutcome(TransactionType.UNKNOWN, List.of("UNKNOWN_TRANSACTION_TYPE"));
    }

    private void removeOfflinePinMissingDe52IfAllowed(RequestContext requestContext, List<String> missingFields) {
        if (missingFields == null || !missingFields.contains("52")) {
            return;
        }
        if (!isSuccessfulOfflinePin(requestContext)) {
            return;
        }
        if (!isOfflinePinAllowedForBin(requestContext.getBin())) {
            log.warn("Offline PIN selected but BIN configuration does not allow offline PIN. bin={}", requestContext.getBin());
            return;
        }

        missingFields.remove("52");
        log.debug("Allowing missing DE52 because DE55 tag 9F34 indicates offline PIN for BIN {}", requestContext.getBin());
    }

    private boolean isSuccessfulOfflinePin(RequestContext requestContext) {
        String de55 = requestContext.getFields().get(55);
        if (de55 == null || de55.isBlank()) {
            return false;
        }

        try {
            ChipData chipData = requestContext.getParsedChipData(55);
            String cvmHex = chipData != null ? chipData.getTagValue("9F34") : null;
            if (cvmHex == null || cvmHex.isBlank()) {
                chipData = new ChipDataParser().parseChipData(de55);
                cvmHex = chipData != null ? chipData.getTagValue("9F34") : null;
            }
            if (cvmHex == null || cvmHex.isBlank()) {
                return false;
            }

            CardholderVerificationMethodResults cvm = new CardholderVerificationMethodResults(cvmHex);
            if (!cvm.isOfflinePinVerificationPerformed() || !cvm.isCvmSuccessful()) {
                return false;
            }

            String tvrHex = chipData != null ? chipData.getTagValue("95") : null;
            if (tvrHex != null && !tvrHex.isBlank()) {
                TerminalVerificationResults tvr = new TerminalVerificationResults(tvrHex);
                if (tvr.cardholderVerificationNotSuccessful()
                        || tvr.pinTryLimitExceeded()
                        || tvr.pinRequiredButNotEntered()) {
                    return false;
                }
            }

            return true;
        } catch (Exception ex) {
            log.warn("Unable to determine offline PIN from DE55 for mandatory DE52 validation: {}", ex.getMessage());
            return false;
        }
    }

    private boolean isOfflinePinAllowedForBin(String bin) {
        if (bin == null || bin.isBlank()) {
            return true;
        }
        if (cacheInitializationService == null) {
            return true;
        }

        CacheInitializationService.BinInstitutionData binData = cacheInitializationService
                .loadBinInstitutionCache()
                .get(bin);
        BinDetails binDetails = binData == null ? null : binData.getBinDetails();
        return binDetails == null
                || binDetails.getAllow_offline_pin() == null
                || binDetails.getAllow_offline_pin();
    }

    private record ValidationOutcome(TransactionType transactionType, List<String> missingFields) {
        boolean valid() {
            return missingFields == null || missingFields.isEmpty();
        }
    }
}
