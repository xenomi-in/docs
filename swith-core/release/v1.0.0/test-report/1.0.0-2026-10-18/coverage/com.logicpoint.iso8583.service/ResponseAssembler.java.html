package com.logicpoint.iso8583.service;

import com.logicpoint.iso8583.context.RequestContext;
import com.logicpoint.iso8583.responsecode.InternalResCode;
import com.logicpoint.iso8583.util.AuthCodeGenerator;
import com.logicpoint.iso8583.validation.ValidationResult;
import com.logicpoint.iso8583.validation.TlvUtils;
import org.springframework.stereotype.Service;

import java.util.LinkedHashMap;
import java.util.Map;

@Service
public class ResponseAssembler {

    private final ResponseCodeCacheService responseCodeCacheService;

    public ResponseAssembler(ResponseCodeCacheService responseCodeCacheService) {
        this.responseCodeCacheService = responseCodeCacheService;
    }

    public void setResponseMti(RequestContext requestContext) {
        Map<Integer, String> fields = requestContext.getFields();
        String requestMti = requestContext.getMti();
        if (requestMti == null || requestMti.length() != 4) {
            return;
        }

        if (requestMti.endsWith("100") || requestMti.endsWith("200") || requestMti.endsWith("400")) {
            fields.put(0, requestMti.substring(0, 2) + "10");
        } else if (requestMti.endsWith("420")) {
            fields.put(0, requestMti.substring(0, 2) + "30");
        }
    }

    public void applyResponseCode(RequestContext requestContext, InternalResCode internalResCode) {
        requestContext.setInternalResponseCode(internalResCode);
        responseCodeCacheService.applyToDe39(requestContext);
    }

    public void ensureResponseCode(RequestContext requestContext) {
        if (requestContext.getInternalResponseCode() == null) {
            requestContext.setInternalResponseCode(InternalResCode.SYSTEM_ERROR);
        }
        responseCodeCacheService.applyToDe39(requestContext);
    }

    public void applyValidationUpdates(RequestContext requestContext, ValidationResult validationResult) {
        if (validationResult == null || !validationResult.hasUpdatedDataElements()) {
            return;
        }

        Object updatedDe55 = validationResult.getUpdatedDataElements().get("DE55");
        if (updatedDe55 instanceof String de55) {
            requestContext.getFields().put(55, de55);
        }

        Object updatedDe102 = validationResult.getUpdatedDataElements().get("DE102");
        if (updatedDe102 instanceof String de102 && !de102.isBlank()) {
            requestContext.getFields().put(102, de102);
        }
    }

    public void applyFinancialApproval(RequestContext requestContext) {
        requestContext.getFields().put(38, AuthCodeGenerator.generateAuthCode());
        applyResponseCode(requestContext, InternalResCode.APPROVED);
    }

    public void omitDe120Tag(RequestContext requestContext, String tag) {
        if (requestContext == null || tag == null || tag.isBlank()) {
            return;
        }

        Map<Integer, String> fields = requestContext.getFields();
        String de120 = fields.get(120);
        if (de120 == null || de120.isBlank()) {
            return;
        }

        Map<String, String> tags = new LinkedHashMap<>(TlvUtils.parseTlv(de120));
        if (tags.remove(tag) == null) {
            return;
        }
        if (tags.isEmpty()) {
            fields.remove(120);
            return;
        }

        StringBuilder rebuilt = new StringBuilder();
        tags.forEach((tlvTag, value) -> rebuilt
                .append(tlvTag)
                .append(String.format("%03d", value.length()))
                .append(value));
        fields.put(120, rebuilt.toString());
    }
}
