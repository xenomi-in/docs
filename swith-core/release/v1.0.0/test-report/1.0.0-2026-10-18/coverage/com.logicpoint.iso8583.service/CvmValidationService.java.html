package com.logicpoint.iso8583.service;

import com.logicpoint.iso8583.emv.CardholderVerificationMethodResults;
import com.logicpoint.iso8583.emv.ChipData;
import com.logicpoint.iso8583.util.CardVerificationMethod;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Service;

import java.math.BigDecimal;
import java.util.LinkedHashMap;
import java.util.Map;

/**
 * Service for Cardholder Verification Method (CVM) validation and processing.
 * Provides business logic for CVM validation based on EMV specifications and
 * payment scheme requirements.
 * 
 * @author Logic Point Systems
 * @version 1.0.0
 */
@Service
public class CvmValidationService {

    private static final Logger logger = LoggerFactory.getLogger(CvmValidationService.class);
    
    private static final String TAG_9F34 = "9F34"; // Cardholder Verification Method Results
    
    @Value("${scheme.rupay.cvm.contactless-cvm-limit:5000}")
    private BigDecimal contactlessCvmLimit = new BigDecimal("5000");

    @Value("${scheme.rupay.cvm.contactless-no-cvm-limit:2500}")
    private BigDecimal contactlessNoCvmLimit = new BigDecimal("2500");
    
    /**
     * Validates CVM for a given transaction based on EMV data and transaction context
     * 
     * @param chipData EMV chip data containing CVM information
     * @param transactionAmount Transaction amount in cents
     * @param isContactless Whether this is a contactless transaction
     * @param terminalCapabilities Terminal capabilities for CVM support
     * @return CvmValidationResult containing validation details
     */
    public CvmValidationResult validateCvm(ChipData chipData, BigDecimal transactionAmount, 
                                         boolean isContactless, String terminalCapabilities) {
        return validateCvm(chipData, transactionAmount, isContactless, terminalCapabilities, false);
    }

    public CvmValidationResult validateCvm(ChipData chipData, BigDecimal transactionAmount,
                                         boolean isContactless, String terminalCapabilities,
                                         boolean pinBlockPresent) {
        return validateCvm(chipData, transactionAmount, isContactless, terminalCapabilities, pinBlockPresent, true);
    }

    public CvmValidationResult validateCvm(ChipData chipData, BigDecimal transactionAmount,
                                         boolean isContactless, String terminalCapabilities,
                                         boolean pinBlockPresent,
                                         boolean offlinePinAllowed) {
        transactionAmount= transactionAmount.divide(new BigDecimal(100));
        logger.debug("Starting CVM validation for transaction amount: {}, contactless: {}", 
                    transactionAmount, isContactless);
        
        if (chipData == null || chipData.isEmpty()) {
            logger.warn("Chip data is null or empty, CVM validation failed");
            return CvmValidationResult.failed("No chip data available for CVM validation");
        }
        
        // Extract CVM Results from tag 9F34
        String cvmHex = chipData.getTagValue(TAG_9F34);
        if (cvmHex == null || cvmHex.isEmpty()) {
            logger.warn("CVM Results (9F34) not found in chip data");
            return CvmValidationResult.failed("CVM Results not found in chip data");
        }
        
        try {
            CardholderVerificationMethodResults cvmResults = new CardholderVerificationMethodResults(cvmHex);
            
            // Perform comprehensive CVM validation
            return performCvmValidation(cvmResults, transactionAmount, isContactless, terminalCapabilities, pinBlockPresent, offlinePinAllowed);
            
        } catch (IllegalArgumentException e) {
            logger.error("Invalid CVM Results format: {}", cvmHex, e);
            return CvmValidationResult.failed("Invalid CVM Results format: " + e.getMessage());
        }
    }
    
    /**
     * Performs comprehensive CVM validation based on EMV specifications
     */
    private CvmValidationResult performCvmValidation(CardholderVerificationMethodResults cvmResults, 
                                                   BigDecimal transactionAmount, boolean isContactless, 
                                                   String terminalCapabilities,
                                                   boolean pinBlockPresent,
                                                   boolean offlinePinAllowed) {
        
        logger.debug("CVM Code: {}, Condition Code: {}, CVM Result: {}",
                    cvmResults.getCvmCodeDescription(),
                    cvmResults.getConditionCodeDescription(),
                    cvmResults.getCvmResultsDescription());

        if (cvmResults.isOnlinePinVerificationPerformed() && cvmResults.isCvmResultUnknown()) {
            logger.debug("Allowing unknown CVM result for online PIN verification; PIN is validated separately");
        }

        if (cvmResults.isOnlinePinVerificationPerformed() && !pinBlockPresent) {
            return CvmValidationResult.failed("Online PIN selected but DE52 PIN data is missing");
        }

        if (cvmResults.isOfflinePinVerificationPerformed()) {
            if (!offlinePinAllowed) {
                return CvmValidationResult.failed("Offline PIN is not allowed for this BIN");
            }
            if (!cvmResults.isCvmSuccessful()) {
                return CvmValidationResult.failed("Offline PIN selected but CVM result is not successful");
            }
            logger.debug("Offline PIN selected in 9F34; accepting CVM result without issuer PIN re-verification");
        }
        
        // Check if CVM is valid for transaction processing
        if (!cvmResults.isValidForTransaction()) {
            String reason = String.format("CVM validation failed: %s - %s", 
                                        cvmResults.getCvmCodeDescription(), 
                                        cvmResults.getCvmResultsDescription());
            logger.warn(reason);
            return CvmValidationResult.failed(reason);
        }
        
        // Validate CVM based on transaction context
        CvmValidationResult result = validateCvmByContext(cvmResults, transactionAmount, isContactless, pinBlockPresent);
        
        if (result.isValid()) {
            logger.debug("CVM validation successful: {}", result.getValidationMessage());
        } else {
            logger.warn("CVM validation failed: {}", result.getValidationMessage());
        }
        
        return result;
    }
    
    /**
     * Validates CVM based on transaction context (amount, contactless, etc.)
     */
    private CvmValidationResult validateCvmByContext(CardholderVerificationMethodResults cvmResults, 
                                                   BigDecimal transactionAmount, boolean isContactless,
                                                   boolean pinBlockPresent) {
        
        // For contactless transactions, check CVM limits
        if (isContactless) {
            return validateContactlessCvm(cvmResults, transactionAmount, pinBlockPresent);
        }
        
        // For contact transactions, validate based on CVM performed
        return validateContactCvm(cvmResults);
    }
    
    /**
     * Validates CVM for contactless transactions
     */
    private CvmValidationResult validateContactlessCvm(CardholderVerificationMethodResults cvmResults, 
                                                     BigDecimal transactionAmount,
                                                     boolean pinBlockPresent) {
        
        // Check if this is a contactless no-CVM transaction
        if (cvmResults.isContactlessNoCvm() || cvmResults.isNoCvmPerformed()) {
            // For contactless no-CVM, check if amount is within limit
            if (transactionAmount.compareTo(contactlessNoCvmLimit) > 0) {
                String reason = String.format("Contactless no-CVM transaction exceeds limit. Amount: %s, Limit: %s", 
                                            transactionAmount, contactlessNoCvmLimit);
                return CvmValidationResult.failed(reason);
            }
            return CvmValidationResult.successful("Contactless no-CVM transaction within limit");
        }

        if (cvmResults.isOfflinePinVerificationPerformed()) {
            return CvmValidationResult.successful("Contactless offline PIN accepted; PIN verified by ICC/terminal");
        }

        if (pinBlockPresent && cvmResults.isPinVerificationPerformed()) {
            return CvmValidationResult.successful("Contactless PIN CVM accepted for issuer PIN verification");
        }
        
        // For contactless with CVM, check if amount is within CVM limit
        if (transactionAmount.compareTo(contactlessCvmLimit) > 0) {
            String reason = String.format("Contactless transaction exceeds CVM limit. Amount: %s, Limit: %s", 
                                        transactionAmount, contactlessCvmLimit);
            return CvmValidationResult.failed(reason);
        }
        
        // Validate the CVM performed
        if (cvmResults.isCvmSuccessful()) {
            return CvmValidationResult.successful("Contactless CVM successful");
        }
        
        return CvmValidationResult.failed("Contactless CVM validation failed");
    }
    
    /**
     * Validates CVM for contact transactions
     */
    private CvmValidationResult validateContactCvm(CardholderVerificationMethodResults cvmResults) {
        
        // Check if CVM was successful
        if (cvmResults.isCvmSuccessful()) {
            return CvmValidationResult.successful("Contact CVM successful");
        }

        // Some terminals populate 9F34 with online PIN as the performed CVM but
        // leave the result byte as 00. In issuer flows we verify the PIN
        // separately via HSM, so this should not be treated as a decline here.
        if (cvmResults.isOnlinePinVerificationPerformed() && cvmResults.isCvmResultUnknown()) {
            return CvmValidationResult.successful("Contact online PIN accepted with unknown CVM result");
        }

        if (cvmResults.isOfflinePinVerificationPerformed()) {
            return CvmValidationResult.successful("Contact offline PIN accepted; PIN verified by ICC/terminal");
        }
        
        // Check if no CVM was required
        if (cvmResults.isNoCvmRequired()) {
            return CvmValidationResult.successful("No CVM required for contact transaction");
        }
        
        // Check if CVM is not applicable
        if (cvmResults.isCvmNotApplicable()) {
            return CvmValidationResult.successful("CVM not applicable for this transaction");
        }
        
        return CvmValidationResult.failed("Contact CVM validation failed");
    }
    
    /**
     * Determines the appropriate CVM method based on transaction context
     * 
     * @param transactionAmount Transaction amount in cents
     * @param isContactless Whether this is a contactless transaction
     * @param terminalCapabilities Terminal capabilities
     * @return Recommended CVM method
     */
    public CardVerificationMethod determineRecommendedCvm(BigDecimal transactionAmount, 
                                                         boolean isContactless, 
                                                         String terminalCapabilities) {
        
        if (isContactless) {
            // For contactless transactions
            if (transactionAmount.compareTo(contactlessNoCvmLimit) <= 0) {
                return CardVerificationMethod.NO_CVM_CONTACTLESS;
            } else if (transactionAmount.compareTo(contactlessCvmLimit) <= 0) {
                return CardVerificationMethod.OFFLINE_PIN;
            } else {
                return CardVerificationMethod.ONLINE_PIN;
            }
        } else {
            // For contact transactions
            if (transactionAmount.compareTo(new BigDecimal("10000")) <= 0) { // $100.00
                return CardVerificationMethod.OFFLINE_PIN;
            } else {
                return CardVerificationMethod.ONLINE_PIN;
            }
        }
    }
    
    /**
     * Checks if terminal supports the required CVM method
     * 
     * @param cvmMethod Required CVM method
     * @param terminalCapabilities Terminal capabilities hex string
     * @return true if terminal supports the CVM method
     */
    public boolean isCvmSupportedByTerminal(CardVerificationMethod cvmMethod, String terminalCapabilities) {
        
        if (terminalCapabilities == null || terminalCapabilities.length() < 6) {
            logger.warn("Invalid terminal capabilities format");
            return false;
        }
        
        try {
            // Parse terminal capabilities (3 bytes)
            int byte1 = Integer.parseInt(terminalCapabilities.substring(0, 2), 16);
            int byte2 = Integer.parseInt(terminalCapabilities.substring(2, 4), 16);
            
            return switch (cvmMethod) {
                case ONLINE_PIN -> (byte1 & 0x80) != 0; // Check if terminal supports online PIN (byte 1, bit 8)
                case OFFLINE_PIN -> (byte1 & 0x40) != 0; // Check if terminal supports offline PIN (byte 1, bit 7)
                case SIGNATURE -> (byte1 & 0x20) != 0; // Check if terminal supports signature (byte 1, bit 6)
                case NO_CVM, NO_CVM_CONTACTLESS -> true; // All terminals support no CVM
                case BIOMETRIC, BIOMETRIC_PIN, BIOMETRIC_SIGNATURE -> (byte2 & 0x80) != 0; // Check if terminal supports biometric (byte 2, bit 8)
                default -> false;
            };
            
        } catch (NumberFormatException e) {
            logger.error("Error parsing terminal capabilities: {}", terminalCapabilities, e);
            return false;
        }
    }
    
    /**
     * Gets CVM validation statistics for monitoring
     * 
     * @param chipData EMV chip data
     * @return Map containing CVM statistics
     */
    public Map<String, Object> getCvmStatistics(ChipData chipData) {
        if (chipData == null || !chipData.hasTag(TAG_9F34)) {
            return Map.of("error", "No CVM data available");
        }
        
        try {
            String cvmHex = chipData.getTagValue(TAG_9F34);
            CardholderVerificationMethodResults cvmResults = new CardholderVerificationMethodResults(cvmHex);
            
            Map<String, Object> statistics = new LinkedHashMap<>();
            statistics.put("cvmCode", cvmResults.getCvmCode());
            statistics.put("cvmCodeDescription", cvmResults.getCvmCodeDescription());
            statistics.put("conditionCode", cvmResults.getConditionCode());
            statistics.put("conditionCodeDescription", cvmResults.getConditionCodeDescription());
            statistics.put("cvmResult", cvmResults.getCvmResults());
            statistics.put("cvmResultDescription", cvmResults.getCvmResultsDescription());
            statistics.put("isValidForTransaction", cvmResults.isValidForTransaction());
            statistics.put("isPinVerificationPerformed", cvmResults.isPinVerificationPerformed());
            statistics.put("isSignatureVerificationPerformed", cvmResults.isSignatureVerificationPerformed());
            statistics.put("isBiometricVerificationPerformed", cvmResults.isBiometricVerificationPerformed());
            statistics.put("isNoCvmRequired", cvmResults.isNoCvmRequired());
            statistics.put("isContactlessNoCvm", cvmResults.isContactlessNoCvm());
            return statistics;
            
        } catch (Exception e) {
            logger.error("Error generating CVM statistics", e);
            return Map.of("error", "Error parsing CVM data: " + e.getMessage());
        }
    }
    
    /**
     * Result class for CVM validation
     */
    public static class CvmValidationResult {
        private final boolean valid;
        private final String validationMessage;
        private final String cvmCode;
        private final String cvmResult;
        
        private CvmValidationResult(boolean valid, String validationMessage, String cvmCode, String cvmResult) {
            this.valid = valid;
            this.validationMessage = validationMessage;
            this.cvmCode = cvmCode;
            this.cvmResult = cvmResult;
        }
        
        public static CvmValidationResult successful(String message) {
            return new CvmValidationResult(true, message, null, null);
        }
        
        public static CvmValidationResult failed(String message) {
            return new CvmValidationResult(false, message, null, null);
        }
        
        public static CvmValidationResult successful(String message, String cvmCode, String cvmResult) {
            return new CvmValidationResult(true, message, cvmCode, cvmResult);
        }
        
        public static CvmValidationResult failed(String message, String cvmCode, String cvmResult) {
            return new CvmValidationResult(false, message, cvmCode, cvmResult);
        }
        
        public boolean isValid() { return valid; }
        public String getValidationMessage() { return validationMessage; }
        public String getCvmCode() { return cvmCode; }
        public String getCvmResult() { return cvmResult; }
        
        @Override
        public String toString() {
            return String.format("CvmValidationResult{valid=%s, message='%s'}", valid, validationMessage);
        }
    }
