package com.logicpoint.iso8583.validation.factory;

import com.logicpoint.iso8583.validation.transaction.TransactionTypeValidator;
import com.logicpoint.iso8583.validation.transaction.impl.NcmcTxnValidation;
import com.logicpoint.iso8583.validation.transaction.impl.PurchaseTransactionValidator;
import com.logicpoint.iso8583.validation.transaction.impl.EcommerceTransactionValidator;
import com.logicpoint.iso8583.validation.transaction.impl.AtmFinancialTransactionValidator;
import com.logicpoint.iso8583.validation.transaction.impl.AtmServiceTransactionValidator;
import com.logicpoint.iso8583.validation.TransactionType;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Component;

import java.util.Map;
import java.util.HashMap;

/**
 * Factory class for creating appropriate transaction validators based on transaction type.
 * Follows the Factory pattern to encapsulate validator creation logic.
 */
@Component
public class TransactionValidatorFactory {
    
    private final Map<TransactionType, TransactionTypeValidator> validators;
    
    @Autowired
    public TransactionValidatorFactory(
            PurchaseTransactionValidator purchaseValidator,
            EcommerceTransactionValidator ecommerceValidator,
            AtmFinancialTransactionValidator atmFinancialValidator,
            NcmcTxnValidation ncmcValidator,
            AtmServiceTransactionValidator atmServiceTransactionValidator) {

        this.validators = new HashMap<>();
        this.validators.put(TransactionType.PURCHASE, purchaseValidator);
        this.validators.put(TransactionType.PURCHASE_CONTACTLESS, purchaseValidator);
        this.validators.put(TransactionType.PURCHASE_WITHCASHBACK, purchaseValidator);
        this.validators.put(TransactionType.PURCHASE_TRANSIT, purchaseValidator);
        this.validators.put(TransactionType.PURCHASE_FALLBACK, purchaseValidator);
        this.validators.put(TransactionType.ECOMMERCE, ecommerceValidator);
        this.validators.put(TransactionType.MOTO, ecommerceValidator);
        this.validators.put(TransactionType.IVR_PURCHASE, ecommerceValidator);
        this.validators.put(TransactionType.CASH_WITHDRAWAL, atmFinancialValidator);
        this.validators.put(TransactionType.CASH_WITHDRAWAL_FALLBACK, atmFinancialValidator);

        this.validators.put(TransactionType.CASH_AT_POS, atmFinancialValidator);
        this.validators.put(TransactionType.PREAUTH, purchaseValidator);
        this.validators.put(TransactionType.SI_REGISTRATION, ecommerceValidator);
        this.validators.put(TransactionType.SI_TRANSACTION, ecommerceValidator);
        this.validators.put(TransactionType.SI_DEREGISTRATION, ecommerceValidator);
        this.validators.put(TransactionType.SI_EDIT, ecommerceValidator);
        this.validators.put(TransactionType.SI_REVOKEALL, ecommerceValidator);
        this.validators.put(TransactionType.SI_PAUSE, ecommerceValidator);
        this.validators.put(TransactionType.SI_START, ecommerceValidator);
        this.validators.put(TransactionType.SI_PDN, ecommerceValidator);

        this.validators.put(TransactionType.EMI_PURCHASE, ecommerceValidator);
        this.validators.put(TransactionType.CARD_EMI_PURCHASE, ecommerceValidator);
        this.validators.put(TransactionType.SERVICE_CREATION, ncmcValidator);
        this.validators.put(TransactionType.SERVICE_CREATION_CONTACTLESS, ncmcValidator);
        this.validators.put(TransactionType.MONEY_LOAD, ncmcValidator);
        this.validators.put(TransactionType.ML_BALANCEUPDATE, ncmcValidator);
        this.validators.put(TransactionType.BALANCE_ENQUIRY, atmServiceTransactionValidator);
        this.validators.put(TransactionType.BALANCE_ENQUIRY_CL, atmServiceTransactionValidator);
        this.validators.put(TransactionType.PIN_CHANGE, atmServiceTransactionValidator);
        this.validators.put(TransactionType.MINI_STATEMENT, atmServiceTransactionValidator);
        

        
        // Add more validators as needed
        // this.validators.put(TransactionType.REFUND, refundValidator);
        // this.validators.put(TransactionType.BALANCE_INQUIRY, balanceInquiryValidator);
    }
    
    /**
     * Gets the appropriate validator for the given transaction type.
     * 
     * @param transactionType The transaction type
     * @return The appropriate validator, or null if not found
     */
    public TransactionTypeValidator getValidator(TransactionType transactionType) {
        return validators.get(transactionType);
    }
    
    /**
     * Checks if a validator exists for the given transaction type.
     * 
     * @param transactionType The transaction type
     * @return true if a validator exists, false otherwise
     */
    public boolean hasValidator(TransactionType transactionType) {
        return validators.containsKey(transactionType);
    }
    
    /**
     * Gets all available transaction types that have validators.
     * 
     * @return Set of transaction types with validators
     */
    public java.util.Set<TransactionType> getSupportedTransactionTypes() {
        return validators.keySet();
    }
}
