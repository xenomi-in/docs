    /**
     * Executes validations in optimized order with early termination.
     *
     * @param cardNumber Card number
     * @param bin BIN
     * @param transactionType Transaction type
     * @param amount Amount
     * @param merchantId Merchant ID
     * @param transactionData Transaction data
     * @param resultBuilder Result builder
     * @return Validation result
     */
    private ValidationResult executeValidationsInOrder(
            RequestContext requestContext,
            ValidationResult.Builder resultBuilder) {
        TransactionType transactionType = requestContext.getTransactionType();
        Map<Integer, String> transactionData = requestContext.getFields();
        String cardNumber = requestContext.getCardNumber();
        String bin = requestContext.getBin();

        // logger.debug("Starting optimized validation flow for card: {}", maskCardNumber(cardNumber));
        // 1. BIN validation (fastest, no DB calls if cached)
        ValidationResult binResult = validateBin(bin);
        if (!binResult.isValid()) {
            logger.error("BIN validation failed: {}", binResult.getErrors());
            return resultBuilder.addErrors(binResult.getErrors()).build();
        }
        resultBuilder.addWarnings(binResult.getWarnings());

        // 2. Transaction type validation
        ValidationResult transactionResult = validateTransactionType(requestContext);
        if (!transactionResult.isValid()) {
            return resultBuilder.addErrors(transactionResult.getErrors()).build();
        }
        resultBuilder.addWarnings(transactionResult.getWarnings());

        ValidationResult nepalBhutanCurrencyResult = validateNepalBhutanCurrency(requestContext);
        if (!nepalBhutanCurrencyResult.isValid()) {
            return nepalBhutanCurrencyResult;
        }

        // 3. Fetch card data ONCE for all subsequent validations
        logger.debug("Fetching card data once for all validation levels");
        CardDataContext cardContext = cardDataService.fetchCardData(requestContext);
        if (!cardContext.isValid()) {
            logger.error("Failed to fetch card data: {}", cardContext.getFetchError());
            return resultBuilder.addError("CARD_DATA_FETCH_FAILED: " + cardContext.getFetchError()).build();
        }

        // 4. Basic card validation (using fetched data)
        ValidationResult cardResult = validateCardWithContext(cardNumber, cardContext);
        if (!cardResult.isValid()) {
            return resultBuilder.addErrors(cardResult.getErrors()).build();
        }
        if (transactionType == TransactionType.PREAUTH && cardContext.getCardData().getCard_type() == Character.valueOf('P')) {
            return resultBuilder.addError("INVALID_TRANSACTION_TYPE: Preauth is not supported for prepaid cards").build();
        }

        ValidationResult cashAtPosCardTypeResult = validateCashAtPosCardType(
                requestContext, cardContext, transactionType);
        if (!cashAtPosCardTypeResult.isValid()) {
            return cashAtPosCardTypeResult;
        }

        resultBuilder.addWarnings(cardResult.getWarnings());
         addRuPayCreditAccountId(requestContext, cardContext, transactionType, resultBuilder);



        // 5. Service code validation
        ValidationResult serviceCodeResult = validateServiceCode(transactionData, transactionType);
        if (!serviceCodeResult.isValid()) {
            return serviceCodeResult; // Early termination
        }
        resultBuilder.addWarnings(serviceCodeResult.getWarnings());

        // 6. Card expiry validation (using fetched data)
        ValidationResult expiryResult = validateCardExpiryWithContext(transactionData, cardContext);
        if (!expiryResult.isValid()) {
            if (expiryResult.getErrors().contains("EXPIRED_CARD: Card has expired")) {
                requestContext.setInternalResponseCode(InternalResCode.EXPIRED_CARD_CAPTURE);
            }
            return resultBuilder.addErrors(expiryResult.getErrors()).build();
        }
        resultBuilder.addWarnings(expiryResult.getWarnings());

        // 6. PIN Retry Count validation
        // Check PIN try count against maximum allowed attempts
        Integer pinTryCount = cardContext.getPinRetryCount();
        Integer maxPinTryCount = cardContext.getMaxPinRetry();

        if (pinTryCount != null && maxPinTryCount != null) {
            if (pinTryCount >= maxPinTryCount) {
                ValidationResult pinRetryResult = new ValidationResult(List.of("PIN_TRY_EXCEEDED: PIN try count has reached or exceeded maximum allowed"), List.of(""));
                logger.warn("PIN try count ({}) has reached or exceeded maximum allowed ({}) ", pinTryCount, maxPinTryCount);
                requestContext.setInternalResponseCode(InternalResCode.PIN_TRY_EXCEEDED);
                return pinRetryResult;
            } else {
                logger.debug("PIN try count is valid for card: {}", maskCardNumber(cardNumber));
            }
        }

        // 7. Card authentications (PIN, CVV, ARQC, ARPC) using fetched data
        ValidationResult cardAuthenticationResult = validateCardAuthenticationWithContext(requestContext, cardContext, transactionType);
        if (!cardAuthenticationResult.isValid()) {
            if (requestContext.getInternalResponseCode() == null) {
                requestContext.setInternalResponseCode(InternalResCode.SYSTEM_ERROR);
            }
            return cardAuthenticationResult;
        }
        resultBuilder.addWarnings(cardAuthenticationResult.getWarnings());

        //omit DE in response 
        // Collect all updated data elements from validation results
        if (cardAuthenticationResult.hasUpdatedDataElements()) {
            resultBuilder.addUpdatedDataElements(cardAuthenticationResult.getUpdatedDataElements());
        }

        if (transactionType == TransactionType.MINI_STATEMENT) {
            // Temporary fix: Mini Statement is not supported by Accounting API yet.
            // Remove this local response generation once Accounting adds support.
            logger.info("Bypassing Accounting API for Mini Statement temporary local response");
            transactionData.put(120, buildMiniStatementDe120(transactionData.get(120)));
            if (isPrepaidAtmSource(requestContext, cardContext)) {
                transactionData.put(54, packPrepaidAtmBalanceDe54(
                        BigDecimal.valueOf(9476.51),
                        BigDecimal.valueOf(9475.51)));
            }
            return resultBuilder.build();
        }

        // Prepaid limit validation
        if (prepaidLimitValidationService != null && cardContext.getCardData().getCard_type() == 'P') {
            logger.info("Calling Prepaid Limit validation ");
            ValidationResult prepaidLimitResult = prepaidLimitValidationService.validate(
                    requestContext, cardContext.getCardData());
            if (!prepaidLimitResult.isValid()) {
                return prepaidLimitResult;
            }
        }

        // 8. Call Accounting API for final authorization
        if (transactionType == TransactionType.PIN_CHANGE) { // PIN change is not a financial transaction, so we don't need to call the accounting API
            logger.debug("Accounting API is disabled for PIN change, skipping authorization service call");
            ValidationResult pinChangeResult;
            if(cmsPinTransactions!=null && cmsPinTransactions.managed(cardContext.getCardData().getAuth_id())) {
                try {
                    var receipt=cmsPinTransactions.change(cardContext.getCardData().getAuth_id(),
                            requestContext.getUniqueTransactionId(),transactionData.get(52),
                            com.logicpoint.iso8583.validation.TlvUtils.parseTlv(transactionData.get(120)).get("004"));
                    var pinResult=new ValidationResult.Builder();
                    if(!"COMPLETED".equals(receipt.path("status").asText())) {
                        pinResult.addError("PIN change rejected");
                        requestContext.setInternalResponseCode(InternalResCode.INVALID_PIN);
                    }
                    pinChangeResult=pinResult.build();
                } catch(Exception e) {
                    requestContext.setInternalResponseCode(InternalResCode.SYSTEM_ERROR);
                    return new ValidationResult.Builder().addError("PIN change unavailable").build();
                }
            } else pinChangeResult = pinChangeService.processPinChange(transactionData, cardContext);
            if (!pinChangeResult.isValid()) {
                return pinChangeResult;
            }
            resultBuilder.addWarnings(pinChangeResult.getWarnings());
            return resultBuilder.build();
        } else if (accountingApiEnabled) {
            ValidationResult accountingResult = callAccountingApiForAuthorization(
                    requestContext, cardContext);
            if (!accountingResult.isValid()) {
                return accountingResult;
            }
            resultBuilder.addWarnings(accountingResult.getWarnings());
        } else {
            logger.debug("Accounting API is disabled, skipping authorization service call");
        }

        // Business rule validation (if needed)
        // ValidationResult businessResult = validateBusinessRules(cardNumber, bin, transactionType, amount, merchantId);
        // if (!businessResult.isValid()) {
        //     return resultBuilder.addErrors(businessResult.getErrors()).build();
        // }
        // resultBuilder.addWarnings(businessResult.getWarnings());
        logger.debug("All validation levels completed successfully for card: {}", maskCardNumber(cardNumber));
        return resultBuilder.build();
    }

    ValidationResult validateNepalBhutanCurrency(RequestContext requestContext) {
        if (requestContext == null || isCreditTransaction(requestContext)) {
            return new ValidationResult(List.of(), List.of());
        }

        Map<Integer, String> fields = requestContext.getFields();
        String acquirerCountry = normalize(fields.get(19));
        String merchantCountry = merchantCountry(fields.get(43));
        boolean isNepalOrBhutan = "524".equals(acquirerCountry)
                || "064".equals(acquirerCountry)
                || "NP".equals(merchantCountry)
                || "BT".equals(merchantCountry);
        if (!isNepalOrBhutan) {
            return new ValidationResult(List.of(), List.of());
        }

        String transactionCurrency = normalize(fields.get(49));
        if (transactionCurrency == null || "356".equals(transactionCurrency)) {
            return new ValidationResult(List.of(), List.of());
        }

        requestContext.setAttribute("complianceRule", "FEMA_NP_BT_FOREIGN_CURRENCY");
        requestContext.setInternalResponseCode(InternalResCode.INVALID_TRANSACTION);
        logger.warn("Declining transaction in Nepal or Bhutan with non-INR currency. "
                        + "transactionType={}, acquirerCountry={}, merchantCountry={}, currency={}",
                requestContext.getTransactionType(), acquirerCountry, merchantCountry, transactionCurrency);
        return new ValidationResult(
                List.of("INVALID_TRANSACTION: Foreign-currency transactions are not permitted in Nepal or Bhutan"),
                List.of());
    }

    private boolean isCreditTransaction(RequestContext requestContext) {
        TransactionType transactionType = requestContext.getTransactionType();
        if (transactionType == TransactionType.REFUND
                || transactionType == TransactionType.REVERSAL
                || transactionType == TransactionType.OCT
                || transactionType == TransactionType.CASH_DEPOSIT
                || transactionType == TransactionType.MONEY_LOAD
                || transactionType == TransactionType.ML_BALANCEUPDATE) {
            return true;
        }

        String processingCode = normalize(requestContext.getFields().get(3));
        return processingCode != null && processingCode.startsWith("20");
    }

    private String merchantCountry(String de43) {
        if (de43 == null || de43.length() != 40) {
            return null;
        }
        try {
            return normalize(CardAcceptorLocation.parse(de43).getTerminalCountryCode());
        } catch (IllegalArgumentException ex) {
            return null;
        }
    }

    private String normalize(String value) {
        if (value == null || value.isBlank()) {
            return null;
        }
        return value.trim().toUpperCase();
    }

    ValidationResult validateCashAtPosCardType(
            RequestContext requestContext,
            CardDataContext cardContext,
            TransactionType transactionType) {
        if (transactionType != TransactionType.CASH_AT_POS
                && transactionType != TransactionType.CASH_AT_POS_CL) {
            return new ValidationResult(List.of(), List.of());
        }

        Character cardType = cardContext != null && cardContext.isValid()
                ? cardContext.getCardData().getCard_type()
                : null;
        if (!Character.valueOf('C').equals(cardType)) {
            return new ValidationResult(List.of(), List.of());
        }

        requestContext.setInternalResponseCode(InternalResCode.INVALID_TRANSACTION);
        logger.warn("Declining {} because cash withdrawal at POS is not permitted for credit cards",
                transactionType);
        return new ValidationResult(
                List.of("INVALID_TRANSACTION: Cash withdrawal at POS is not permitted for credit cards"),
                List.of());

    private ValidationResult validateCardWithContext(String cardNumber, CardDataContext cardContext) {

    /**
     * Validates card expiry using pre-fetched card data context. This method
     * avoids additional database calls.
     *
     * @param transactionData Transaction data
     * @param cardContext Pre-fetched card data context
     * @return Validation result
     */
    private ValidationResult validateCardExpiryWithContext(Map<Integer, String> transactionData, CardDataContext cardContext) {
        String cardExp = transactionData.get(14); // DE14 for expiry date
        if (cardExp == null) {
            String track2 = transactionData.get(35);
            if (track2 != null && track2.length() >= 3) {
                String[] parts = track2.split("=");
                if (parts.length >= 2) {
                    cardExp = parts[1].substring(0, 4);
                    logger.debug("Card expiry date found in track2: {}", cardExp);
                }
            }
        }

        if (cardExp == null) {
            logger.warn("No expiry date found in transaction data");
            return ValidationResult.builder()
                    .addError("MISSING_EXPIRY_DATE: Card expiry date is required")
                    .build();
        }

        // Validate expiry date format
        if (!cardExp.matches("\\d{4}")) {
            return ValidationResult.builder()
                    .addError("INVALID_EXPIRY_FORMAT: Expiry date must be in MMYY format")
                    .build();
        }

        // Compare with pre-fetched card data
        String storedExpiry = cardContext.getExpiryDate();
        if (storedExpiry == null || !storedExpiry.equals(cardExp)) {
            logger.warn("Expiry date mismatch. Transaction: {}, Stored: {}", cardExp, storedExpiry);
            return ValidationResult.builder()
                    .addError("EXPIRED_CARD: Card has expired")
                    .build();
        }

        logger.debug("Card expiry validation successful using pre-fetched data: {}", cardExp);
        return ValidationResult.builder().build();
    }

    /**
     * Extracts service code from transaction data.
     *
     * @param transactionData Transaction data
     * @return Service code or null
     */
    private String extractServiceCode(Map<Integer, String> transactionData) {
        // Try Track 2 first (Field 35)
        String track2 = transactionData.get(35);
        if (track2 != null && track2.length() >= 3) {
            String[] parts = track2.split("=");
            if (parts.length >= 2) {
                return parts[1].substring(4, 7);
            }
        }

        return null;
    }

    /**
     * Masks card number for logging.
     *
     * @param cardNumber Card number
     * @return Masked card number
     */
    private String maskCardNumber(String cardNumber) {
        if (cardNumber == null || cardNumber.length() < 8) {
            return "****";
        }
        return cardNumber.substring(0, 4) + "****" + cardNumber.substring(cardNumber.length() - 4);
    }

    /**
     * Performs quick validation (card and BIN only).

    public ValidationResult quickValidate(String cardNumber, String bin) {
        logger.debug("Performing quick validation for card: {}, BIN: {}",
                maskCardNumber(cardNumber), bin);

        ValidationResult.Builder resultBuilder = ValidationResult.builder();

        // Card validation
        ValidationResult cardResult = validateCard(cardNumber);
        if (!cardResult.isValid()) {
            resultBuilder.addErrors(cardResult.getErrors());
        }
        resultBuilder.addWarnings(cardResult.getWarnings());

        // BIN validation
        ValidationResult binResult = validateBin(bin);
        if (!binResult.isValid()) {
            resultBuilder.addErrors(binResult.getErrors());
        }
        resultBuilder.addWarnings(binResult.getWarnings());

        return resultBuilder.build();
    }

    /**
     * Validates card authentication using pre-fetched card data context. This
     * method avoids additional database calls and uses cached card data.
     *
     * @param transactionData Transaction data
     * @param cardContext Pre-fetched card data context
     * @return Validation result
     */
    private ValidationResult validateCardAuthenticationWithContext(RequestContext requestContext, CardDataContext cardContext, TransactionType transactionType) {
        Map<Integer, String> transactionData = requestContext.getFields();
        logger.debug("Validating card authentication with context for card: {}",
                maskCardNumber(transactionData.get(2)));
        try {
            ValidationResult.Builder resultBuilder = ValidationResult.builder();
            String emvData = transactionData.get(55); // DE55 - ICC System Related Data
            ChipData chipData = null;

            // 1. CVM validation is local EMV parsing/business logic. Run it before
            // CVV/PIN so invalid CVM data can fail without HSM calls.
            if (emvData != null && !emvData.trim().isEmpty()) {
                chipData = requestContext.getParsedChipData(55);
                preserveOriginalDe55(requestContext, emvData, chipData);
                ValidationResult cvmResult = validateCvmWithContext(requestContext, chipData, transactionType);
                if (!cvmResult.isValid()) {
                    resultBuilder.addErrors(cvmResult.getErrors());
                    resultBuilder.addWarnings(cvmResult.getWarnings());
                    return resultBuilder.build();
                }
                resultBuilder.addWarnings(cvmResult.getWarnings());
            } else {
                logger.debug("No EMV data found, skipping CVM validation");
            }

            // 2. CVV/iCVV/CVV2 validation - selected from transaction type and payload.
            CvvVerificationRequest cvvRequest = resolveCvvVerificationRequest(requestContext, transactionType);
            if (cvvRequest != null) {
                if (!cvvRequest.hasSecurityCode()) {
                    resultBuilder.addError("CVV_VALIDATION_FAILED: Missing " + cvvRequest.type + " value");
                    String updatedDe48 = updateDE48(transactionData.get(48), "N", cvvRequest);
                    transactionData.put(48, updatedDe48);
                    requestContext.setInternalResponseCode(InternalResCode.CVV_FAILURE);
                    return resultBuilder.build();
                }

                logger.debug("Performing {} validation with pre-fetched data", cvvRequest.type);
                int cvvValid = validateCvvWithContext(transactionData, cardContext, cvvRequest);
                if (cvvValid == 1) {
                    resultBuilder.addError("CVV_VALIDATION_FAILED: " + cvvRequest.type + " validation failed");
                    String updatedDe48 = updateDE48(transactionData.get(48), "N", cvvRequest);
                    transactionData.put(48, updatedDe48);
                    requestContext.setInternalResponseCode(InternalResCode.CVV_FAILURE);
                    return resultBuilder.build();
                } else if (cvvValid == 0) {
                    String updatedDe48 = updateDE48(transactionData.get(48), "M", cvvRequest);
                    transactionData.put(48, updatedDe48);
                } else if (cvvValid == 2) {
                    resultBuilder.addError("CVV_VALIDATION_FAILED: System error");
                    requestContext.setInternalResponseCode(InternalResCode.SYSTEM_ERROR);
                    return resultBuilder.build();
                }
                logger.debug("{} validation successful", cvvRequest.type);
            }

            // 3. PIN Verification (if PIN block is present) - using pre-fetched data
            String pinBlock = transactionData.get(52); // DE52 - PIN Block
            if (transactionType==TransactionType.PIN_CHANGE && cmsPinTransactions!=null
                    && cmsPinTransactions.managed(cardContext.getCardData().getAuth_id())) {
                // Atomic change verifies the old PIN under the family lock after
                // replay lookup. Do not reject a successful retry using the old PIN.
                logger.debug("PIN verification deferred to atomic family PIN change");
            } else if (pinBlock != null && !pinBlock.trim().isEmpty()) {
                logger.debug("Performing PIN verification with pre-fetched data");
                InternalResCode pinValidationResult = validatePinWithContext(requestContext, cardContext);
                if (pinValidationResult != InternalResCode.APPROVED) {
                    resultBuilder.addError("PIN_VERIFICATION_FAILED: PIN verification failed");
                    requestContext.setInternalResponseCode(pinValidationResult);
                    return resultBuilder.build();
                }
                requestContext.setAttribute("issuerPinVerified", true);
                logger.debug("PIN verification successful");
            } else {
                logger.debug("No PIN block found, skipping PIN verification");
            }

            // 4. Remaining EMV validations and ARQC/ARPC generation.
            if (chipData != null) {
                ValidationResult cvrResult = validateCvrWithContext(requestContext, chipData);
                if (!cvrResult.isValid()) {
                    resultBuilder.addErrors(cvrResult.getErrors());
                    resultBuilder.addWarnings(cvrResult.getWarnings());
                    return resultBuilder.build();
                }
                resultBuilder.addWarnings(cvrResult.getWarnings());

                ValidationResult tvrResult = validateTvrWithContext(requestContext, chipData);
                if (!tvrResult.isValid()) {
                    resultBuilder.addErrors(tvrResult.getErrors());
                    resultBuilder.addWarnings(tvrResult.getWarnings());
                    return resultBuilder.build();
                }
                resultBuilder.addWarnings(tvrResult.getWarnings());

                logger.debug("Performing EMV ARQC verification and ARPC generation with pre-fetched data");
                ValidationResult emvResult = validateEmvCryptogramsWithContext(requestContext, cardContext, transactionType);
                if (!emvResult.isValid()) {
                    resultBuilder.addErrors(emvResult.getErrors());
                    if (requestContext.getInternalResponseCode() == null) {
                        requestContext.setInternalResponseCode(InternalResCode.ARQC_FAIL);
                    }
                    return resultBuilder.build();
                }
                resultBuilder.addWarnings(emvResult.getWarnings());
                logger.debug("EMV cryptogram validation successful");
            } else {
                logger.debug("No EMV data found, skipping ARQC/ARPC validation");
            }

            return resultBuilder.build();

        } catch (Exception e) {
            logger.error("Error during card authentication validation with context: {}", e.getMessage());
            requestContext.setInternalResponseCode(InternalResCode.SYSTEM_ERROR);
            return ValidationResult.builder().addError("CARD_AUTHENTICATION_ERROR: " + e.getMessage()).build();
        }
    }

    private ValidationResult validateCvmWithContext(RequestContext requestContext, ChipData chipData, TransactionType transactionType) {
        if (chipData == null || chipData.isEmpty()) {
            return ValidationResult.builder()
                    .addError("MISSING_EMV_DATA: Chip data unavailable for CVM validation")
                    .build();
        }

        BigDecimal transactionAmount = extractTransactionAmountInMinorUnits(requestContext.getFields());
        boolean isContactless = isContactlessTransaction(transactionType)
                || isContactlessPosEntryMode(requestContext.getFields().get(22));
        String terminalCapabilities = chipData.getTagValue("9F33");
        BinDetails binDetails = resolveBinDetails(requestContext.getBin());
        boolean offlinePinAllowed = isOfflinePinAllowed(binDetails);

        CvmValidationService.CvmValidationResult cvmResult = cvmValidationService.validateCvm(
                chipData,
                transactionAmount,
                isContactless,
                terminalCapabilities,
                hasPinBlock(requestContext.getFields()),
                offlinePinAllowed
        );

        if (!cvmResult.isValid()) {
            requestContext.setInternalResponseCode(resolveCvmFailureCode(requestContext, chipData));
            return ValidationResult.builder()
                    .addError("CVM_VALIDATION_FAILED: " + cvmResult.getValidationMessage())
                    .build();
        }

        return ValidationResult.builder().build();
    }

    private boolean hasPinBlock(Map<Integer, String> transactionData) {
        String pinBlock = transactionData.get(52);
        return pinBlock != null && !pinBlock.trim().isEmpty();
    }

    private ValidationResult validateCvrWithContext(RequestContext requestContext, ChipData chipData) {
        if (chipData == null || chipData.isEmpty()) {
            return ValidationResult.builder()
                    .addError("MISSING_EMV_DATA: Chip data unavailable for CVR validation")
                    .build();
        }

       
        CvrValidationService.CvrValidationResult cvrResult = cvrValidationService.validateCvr(chipData, requestContext.getScheme());

        if (!cvrResult.isValid()) {
            requestContext.setInternalResponseCode(resolveCvrFailureCode(cvrResult));
            ValidationResult.Builder resultBuilder = ValidationResult.builder();
            if (cvrResult.getErrors() != null && !cvrResult.getErrors().isEmpty()) {
                resultBuilder.addErrors(cvrResult.getErrors());
            } else {
                resultBuilder.addError("CVR_VALIDATION_FAILED: " + cvrResult.getMessage());
            }
            return resultBuilder.build();
        }

        ValidationResult.Builder resultBuilder = ValidationResult.builder();
        if (cvrResult.getWarnings() != null && !cvrResult.getWarnings().isEmpty()) {
            resultBuilder.addWarnings(cvrResult.getWarnings());
        }
        return resultBuilder.build();
    }

    private ValidationResult validateTvrWithContext(RequestContext requestContext, ChipData chipData) {
        if (chipData == null || chipData.isEmpty()) {
            return ValidationResult.builder()
                    .addError("MISSING_EMV_DATA: Chip data unavailable for TVR validation")
                    .build();
        }

        String tvrValue = chipData.getTagValue("95");
        if (tvrValue == null || tvrValue.isBlank()) {
            requestContext.setInternalResponseCode(InternalResCode.FORMAT_ERROR);
            return ValidationResult.builder()
                    .addError("TVR_VALIDATION_FAILED: TVR tag 95 missing")
                    .build();
        }

        TerminalVerificationResults tvr = new TerminalVerificationResults(tvrValue);
        BinDetails binDetails = resolveBinDetails(requestContext.getBin());
        boolean declineOnSdaFailure = shouldDeclineOnSdaFailure(binDetails);
        boolean declineOnDdaFailure = shouldDeclineOnDdaFailure(binDetails);
        boolean declineOnCdaFailure = shouldDeclineOnCdaFailure(binDetails);

        boolean valid = tvrValidationService.validateTvr(
                tvr,
                declineOnSdaFailure,
                declineOnDdaFailure,
                declineOnCdaFailure);
        if (!valid) {
            InternalResCode internalResponseCode = resolveTvrFailureCode(tvr);
            logger.warn("TVR validation failed. resolvedInternalResponseCode={}", internalResponseCode);
            requestContext.setInternalResponseCode(internalResponseCode);
            return ValidationResult.builder()
                    .addError("TVR_VALIDATION_FAILED: TVR validation failed")
                    .build();
        }

        ValidationResult.Builder resultBuilder = ValidationResult.builder();
        if (!declineOnSdaFailure && tvr.sdaFailed()) {
            resultBuilder.addWarning("TVR_SDA_FAILED_IGNORED: SDA failed but BIN configuration allows transaction to proceed");
        }
        if (!declineOnDdaFailure && tvr.ddaFailed()) {
            resultBuilder.addWarning("TVR_DDA_FAILED_IGNORED: DDA failed but BIN configuration allows transaction to proceed");
        }
        if (!declineOnCdaFailure && tvr.cdaFailed()) {
            resultBuilder.addWarning("TVR_CDA_FAILED_IGNORED: CDA failed but BIN configuration allows transaction to proceed");
        }

        return resultBuilder.build();
    }

    /**
     * Validates CVV using pre-fetched card data context. This method avoids
     * additional database calls.
     *
     * @param transactionData Transaction data
     * @param cardContext Pre-fetched card data context
     * @return true if CVV is valid, false otherwise
     */
    private int validateCvvWithContext(Map<Integer, String> transactionData, CardDataContext cardContext, CvvVerificationRequest cvvRequest) {
        try {

            String pan = transactionData.get(2); // DE2 - Primary Account Number
            String expiryDate = cvvRequest.expiryDate != null ? cvvRequest.expiryDate : extractExpiryDate(transactionData);

            // Validate expiry date against pre-fetched data
            String storedExpiry = cardContext.getExpiryDate();
            if (storedExpiry == null || !storedExpiry.equals(expiryDate)) {
                logger.warn("Expiry date mismatch for CVV validation. Transaction: {}, Stored: {}",
                        expiryDate, storedExpiry);
                return 1;
            }

            return switch (cvvRequest.type) {
                case ICVV -> nettyHsmService.verifyIcvv(pan, expiryDate, cvvRequest.securityCode);
                case CVV2 -> nettyHsmService.verifyCvv2(pan, expiryDate, cvvRequest.securityCode);
                case MAGSTRIPE -> nettyHsmService.verifyMagstripeCvv(
                        pan, expiryDate, cvvRequest.serviceCode, cvvRequest.securityCode);
            };

        } catch (Exception e) {
            logger.error("Error during CVV validation with context: {}", e.getMessage());
            return 2;
        }
    }

    private CvvVerificationRequest resolveCvvVerificationRequest(RequestContext requestContext, TransactionType transactionType) {
        Map<Integer, String> transactionData = requestContext.getFields();
        String expiryDate = extractExpiryDate(transactionData);

        if (isCardNotPresentTransaction(transactionType)) {
            String cvv2 = extractCvv2(requestContext);
            return new CvvVerificationRequest(CvvVerificationType.CVV2, cvv2, CVV2_SERVICE_CODE, expiryDate);
        }

        if (isEmvTransaction(transactionData, transactionType)) {
            TrackSecurityData emvTrackData = extractEmvTrackSecurityData(requestContext);
            String icvv = emvTrackData != null ? emvTrackData.securityCode : null;
            String emvExpiry = emvTrackData != null && emvTrackData.expiryDate != null
                    ? emvTrackData.expiryDate
                    : expiryDate;
            return new CvvVerificationRequest(CvvVerificationType.ICVV, icvv, ICVV_SERVICE_CODE, emvExpiry);
        }

        TrackSecurityData trackSecurityData = extractTrackSecurityData(transactionData.get(35));
        if (trackSecurityData == null) {
            return null;
        }
        return new CvvVerificationRequest(
                CvvVerificationType.MAGSTRIPE,
                trackSecurityData.securityCode,
                trackSecurityData.serviceCode,
                trackSecurityData.expiryDate);
    }

                || transactionType == TransactionType.SI_REGISTRATION
                || transactionType == TransactionType.SI_EDIT
                || transactionType == TransactionType.SI_TRANSACTION
                || transactionType == TransactionType.SI_DEREGISTRATION
                || transactionType == TransactionType.SI_PDN;

        }

    private TrackSecurityData extractEmvTrackSecurityData(RequestContext requestContext) {
        ChipData chipData = requestContext.getParsedChipData(55);
        if (chipData != null) {
            String track2Equivalent = chipData.getTagValue("9F5B");
            if (track2Equivalent == null || track2Equivalent.isBlank()) {
                track2Equivalent = chipData.getTagValue("57");
            }
            TrackSecurityData trackSecurityData = extractTrackSecurityData(track2Equivalent);
            if (trackSecurityData != null) {
                return trackSecurityData;
            }
        }
        return extractTrackSecurityData(requestContext.getFields().get(35));
    }

    private TrackSecurityData extractTrackSecurityData(String trackData) {
        if (trackData == null || trackData.isBlank()) {
            return null;
        }

        String normalizedTrackData = trackData.trim().replace('d', 'D');
        int separatorIndex = normalizedTrackData.indexOf('=');
        if (separatorIndex < 0) {
            separatorIndex = normalizedTrackData.indexOf('D');
        }
        if (separatorIndex < 0 || normalizedTrackData.length() <= separatorIndex + 7) {
            return null;
        }

        String rightSide = normalizedTrackData.substring(separatorIndex + 1);
        if (rightSide.endsWith("F")) {
            rightSide = rightSide.substring(0, rightSide.length() - 1);
        }
        String expiryDate = rightSide.substring(0, 4);
        String serviceCode = rightSide.substring(4, 7);
        //String securityCode = rightSide.length() >= 18 ? rightSide.substring(7, 10) : null;
        String securityCode = rightSide.length() >= 18 ? rightSide.substring(15, 18) : null;
        return new TrackSecurityData(expiryDate, serviceCode, normalizeSecurityCode(securityCode));
    }

    private String extractCvv2(RequestContext requestContext) {
        Map<String, String> tlvMap = requestContext.getParsedTlv(48);
        for (String tag : DIRECT_CVV2_TLV_TAGS) {
            String cvv2 = normalizeSecurityCode(tlvMap.get(tag));
            if (cvv2 != null) {
                return cvv2;
            }
        }
        return extractCvv2FromCavv(tlvMap.get("100"));
    }

    private String extractCvv2FromCavv(String cavv) {
        if (cavv == null || cavv.isBlank()) {
            return null;
        }
        try {
            String cavvDecoded = new String(Base64.getDecoder().decode(cavv));
            String paddedCavv = String.format("%-40s", cavvDecoded);
            return normalizeSecurityCode(paddedCavv.substring(6, 10));
        } catch (Exception ex) {
            logger.warn("Unable to extract CVV2 from CAVV tag 100: {}", ex.getMessage());
            return null;
        }
    }

    private String normalizeSecurityCode(String securityCode) {
        if (securityCode == null) {
            return null;
        }
        String normalized = securityCode.trim();
        return normalized.matches("\\d{3,4}") ? normalized : null;
    }

            return resultBuilder.addError("EMV_CRYPTOGRAM_ERROR: " + e.getMessage()).build();

    /**
     * Validates ARQC using NettyHsmService.
     *
     * @param transactionData Transaction data
     * @param scheme Card scheme
     * @param emvMethod EMV method
     * @return true if ARQC is valid, false otherwise
     */
    private InternalResCode validateArqcWithHsm(RequestContext requestContext, Scheme scheme, String emvMethod, TransactionType transactionType) {
        try {
            Map<Integer, String> transactionData = requestContext.getFields();
            String emvData = transactionData.get(55);
            String pan = transactionData.get(2);

            if (emvData == null || pan == null) {
                logger.warn("Missing EMV data or PAN for ARQC verification");
                requestContext.setInternalResponseCode(InternalResCode.FORMAT_ERROR);
                return InternalResCode.FORMAT_ERROR;
            }

            // Parse EMV data to extract required fields
            // This is a simplified implementation - actual ARQC verification
            // depends on your HSM model and EMV implementation
            // boolean arqcValid = cardAuthenticationService.verifyArqc(transactionData, scheme, emvMethod);
            ChipData chipData = requestContext.getParsedChipData(55);
            if (!new ChipDataParser().checkMandatoryTags(chipData, scheme.name(), false, transactionType)) {
                logger.warn("Mandatory tags are missing for EMV data");
                removeResponseDe55(transactionData);
                requestContext.setInternalResponseCode(InternalResCode.FORMAT_ERROR);
                return InternalResCode.FORMAT_ERROR;
            }
            String cardSequenceNumber = transactionData.get(23);
            String sequenceNumber = (cardSequenceNumber != null && cardSequenceNumber.length() > 1)
                    ? cardSequenceNumber.substring(1)
                    : "0";
            String arpc = nettyHsmService.verifyArqcAndGenerateArpc(chipData, scheme, emvMethod, pan, sequenceNumber, transactionType, false, "000000000000", 'A');
            String[] arpcParts = arpc.split(";");
            String arpcResult = arpcParts[0];
            String arpcValue = arpcParts[1];
            if (arpcResult.equals("Y")) {
                logger.debug("ARQC verification successful via HSM for scheme: {}, method: {}", scheme, emvMethod);
                transactionData.put(55, buildDe55WithOnlyTag91(arpcValue));
                return InternalResCode.APPROVED;
            } else {
                logger.warn("ARQC verification failed via HSM for scheme: {}, method: {}", scheme, emvMethod);
                return InternalResCode.ARQC_FAIL;
            }

        } catch (Exception e) {
            logger.error("Error during HSM ARQC verification: {}", e.getMessage());
            if (requestContext.getInternalResponseCode() == null) {
                requestContext.setInternalResponseCode(InternalResCode.SYSTEM_ERROR);
            }
            return InternalResCode.SYSTEM_ERROR;
        }
    }

        String expiryDate = transactionData.get(14);

                return parts[1].substring(0, 4);
            }
        }

        return null;
    }

    /**

     * @param transactionType Transaction type

     * @param merchantId Merchant ID
     * @param transactionData Transaction data
     * @return Validation result
     */
    private ValidationResult callAccountingApiForAuthorization(
            RequestContext requestContext,
            CardDataContext cardContext) {
        TransactionType transactionType = requestContext.getTransactionType();
        Map<Integer, String> transactionData = requestContext.getFields();
        String cardNumber = requestContext.getCardNumber();
        String bin = requestContext.getBin();
        Double amount = requestContext.getAmount();

        logger.debug("Calling Accounting API for authorization for card: {}, amount: {}",
                maskCardNumber(cardNumber), amount);

        try {
            // Convert amount to BigDecimal
            BigDecimal transactionAmount = BigDecimal.valueOf(amount);

            // Determine currency (default to USD, can be enhanced based on transaction data)
            String currency = "USD";
            if (transactionData.containsKey(49)) {
                String currencyCode = transactionData.get(49);
                // Map ISO 4217 currency codes
                switch (currencyCode) {
                    case "356" ->
                        currency = "INR";
                    case "840" ->
                        currency = "USD";
                    case "978" ->
                        currency = "EUR";
                    default ->
                        currency = "USD";
                }
            }

            // Determine network based on BIN
            String network = resolveScheme(requestContext, cardNumber);

            // Get MTI from transaction data or use default
            String mti = transactionData.getOrDefault(0, "0200");

            // Call Accounting API
            AccountingAuthorizationResponse authResponse = accountingApiService.authorizeTransaction(
                    requestContext, transactionAmount, currency, mti, network, transactionType.name());

            if (authResponse != null && "00".equals(authResponse.getResponseCode())) {
                logger.info("Accounting API authorization successful. Transaction ID: {}, Auth Code: {}",
                        authResponse.getTransactionId(), authResponse.getAuthorizationCode());
                // CRITICAL: Store accounting transaction ID for reversal tracking
                String uniqueTxnId = requestContext.getUniqueTransactionId();
                if (uniqueTxnId != null) {
                    // Store accounting txn ID in a field that will be available later
                    transactionData.put(901, authResponse.getTransactionId());
                    
                    // Mark accounting approved in context
                    // Note: You'll need to inject AutomaticReversalService here
                    // automaticReversalService.markAccountingApproved(uniqueTxnId, authResponse.getTransactionId());
                }
                if (transactionType == TransactionType.BALANCE_ENQUIRY
                        || transactionType == TransactionType.BALANCE_ENQUIRY_CL
                        || transactionType == TransactionType.CASH_WITHDRAWAL) {
                    BigDecimal balanceAvailable = resolveAvailableBalance(authResponse);
                    BigDecimal ledgerBalance = resolveLedgerBalance(authResponse, balanceAvailable);
                    if (isPrepaidAtmSource(requestContext, cardContext)) {
                        PrepaidCardBalanceSnapshot snapshot = prepaidCardBalanceSnapshotService.updateFromAccountingAvailable(
                                requestContext.getPanHash(),
                                cardContext.getCardData().getCard_id(),
                                transactionData.getOrDefault(49, "356"),
                                balanceAvailable,
                                resolveAccountingLedgerBalance(authResponse),
                                transactionAmount,
                                isDebitBalanceUpdate(transactionType));
                        ledgerBalance = snapshot.getLedgerBalance();
                        balanceAvailable = snapshot.getAvailableBalance();
                    }
                    String de54 = packResponseBalanceDe54(
                            requestContext, transactionType, cardContext, ledgerBalance, balanceAvailable);
                    logger.debug("{} DE54 ledger: {}, available: {}, packed length: {}",
                            transactionType,
                            ledgerBalance, balanceAvailable, de54.length());
                    transactionData.put(54, de54);
                }
                if (transactionType == TransactionType.PURCHASE || transactionType == TransactionType.PURCHASE_CONTACTLESS || transactionType == TransactionType.PURCHASE_WITHCASHBACK || transactionType == TransactionType.CASH_AT_POS || transactionType == TransactionType.CASH_AT_POS_CL 
                    || transactionType == TransactionType.CASH_WITHDRAWAL
                    || transactionType == TransactionType.BALANCE_ENQUIRY
                    || transactionType == TransactionType.MINI_STATEMENT
                    || transactionType == TransactionType.PIN_CHANGE
                    || transactionType == TransactionType.MONEY_LOAD){
                    if (authResponse.getNcmcData() != null) {
                        BigDecimal updateAmount = authResponse.getNcmcData().getUpdateAmount().setScale(2);
                        //--------------------------------- Temporary Global Balance Update  , shoul be removed after the real accounting api integration---------------------------------
                        String tempGbsMode = tempGbs == null ? "no-update" : tempGbs.trim().toLowerCase();
                        if (tempGbsMode.equals("credit")) {
                            if (transactionType == TransactionType.MONEY_LOAD) {
                                updateAmount = updateAmount.add(BigDecimal.valueOf(100));
                            } else {
                                updateAmount = BigDecimal.valueOf(50);
                            }
                            updateAmount = updateAmount.multiply(BigDecimal.valueOf(1));
                        } else if (tempGbsMode.equals("credit-less")) {
                            logger.info( " in credit-less : {}",updateAmount);
                            if (transactionType == TransactionType.MONEY_LOAD) {
                                updateAmount = updateAmount.subtract(BigDecimal.valueOf(50));
                            } else {
                                updateAmount = BigDecimal.valueOf(20);
                            }
                            logger.info( " in credit-less : {}",updateAmount);
                            //updateAmount = updateAmount.multiply(BigDecimal.valueOf(1));
                        } else if (tempGbsMode.equals("credit-more")) {
                            if (transactionType == TransactionType.MONEY_LOAD) {
                                updateAmount = updateAmount.add(BigDecimal.valueOf(99 ));
                            } else {
                                updateAmount = BigDecimal.valueOf(100);
                            }
                            //updateAmount = updateAmount.multiply(BigDecimal.valueOf(1));
                        } else if (tempGbsMode.equals("debit")) {
                            if (transactionType != TransactionType.MONEY_LOAD) {
                                updateAmount = BigDecimal.valueOf(20);
                            }
                            updateAmount = updateAmount.multiply(BigDecimal.valueOf(-1));
                        } else {
                            updateAmount = updateAmount.multiply(BigDecimal.valueOf(0));
                        }
                        //--------------------------------- Temporary Global Balance Update  , should be removed after the real accounting api integration---------------------------------
                        char balanceType = 'A';
                        if (updateAmount.compareTo(BigDecimal.ZERO) < 0) {
                            balanceType = 'D';
                        } else if (updateAmount.compareTo(BigDecimal.ZERO) > 0) {
                            balanceType = 'C';
                        }
                        updateAmount = updateAmount.compareTo(BigDecimal.ZERO) < 0 ? updateAmount.multiply(BigDecimal.valueOf(-1)) : updateAmount;
                        //String emvData = transactionData.get(55);
                        String pan = transactionData.get(2);
                        Scheme scheme = resolveSchemeEnum(requestContext, pan);
                        String emvMethod = determineEmvMethod(scheme.name());
                        ChipData chipData = getOriginalChipData(requestContext);
                        if (!new ChipDataParser().checkMandatoryTags(chipData, scheme.name(), false, transactionType)) {
                            logger.warn("Mandatory tags are missing for EMV data");
                            removeResponseDe55(transactionData);
                            requestContext.setInternalResponseCode(InternalResCode.FORMAT_ERROR);
                            return ValidationResult.builder()
                                    .addError("EMV_TAG_MISSING: Mandatory tags are missing for EMV data")
                                    .build();
                        }

                      // long balanceAmountMinorUnits = updateAmount.movePointRight(2).setScale(0, RoundingMode.DOWN).longValue();
                       // String balanceAmt = String.format("%012d", balanceAmountMinorUnits);
                        String balanceAmt = String.format("%012d", updateAmount.movePointRight(2).longValue());

                        logger.info(" Before ARPC regeneration : {}",balanceAmt);
                        String arpc = nettyHsmService.verifyArqcAndGenerateArpc(chipData, scheme, emvMethod, pan, transactionData.get(23).substring(1), transactionType, true, balanceAmt, balanceType);
                        logger.debug("ARPC generated via HSM {}", arpc);
                        String[] arpcParts = arpc.split(";");
                        //String arpcResult = arpcParts[0];
                        String arpcValue = arpcParts[1];
                        if (!arpcValue.trim().isEmpty()) {
                            logger.debug("ARQC verification via HSM for scheme: {}, method: {}", scheme, emvMethod);
                            transactionData.put(55, buildDe55WithOnlyTag91(arpcValue));
                            logger.debug("ARPC verification via HSM for scheme: {}, method: {}", scheme, emvMethod);
                            //return ResCode.ARQC_VERIFIED;
                        } else {
                            logger.warn("No ARQC verification via HSM for scheme: {}, method: {}", scheme, emvMethod);
                            //return ResCode.ARQC_FAIL;
                        }
	                    } else {
	                        logger.warn("No NCMC Data in response from accounting API processing as normal txn");
	                        //return ResCode.NCMC_DATA_MISSING;
	                        if (transactionData.containsKey(55)) {
	                            ChipData chipData = requestContext.getParsedChipData(55);
	                            String tag91 = chipData.getTagValue("91");
	                            updateDe55WithOnlyTag91(transactionData, tag91);
	                        }
	                    }
	                } else { // this else block to process other tans purchase, cash at pos, cash at pos contactless, balance enquiry, balance enquiry contactless, money load, etc.
	                    logger.debug("Processing as normal txn with out ARPC regeneration, updating DE55 with ARPC from previous transaction");
	                    if (transactionData.containsKey(55)) {
	                        ChipData chipData = requestContext.getParsedChipData(55);
	                        String tag91 = chipData.getTagValue("91");
	                        updateDe55WithOnlyTag91(transactionData, tag91);
	                    }
	                }

                // Add authorization details to the result
                ValidationResult.Builder resultBuilder = ValidationResult.builder();
                resultBuilder.addUpdatedDataElement("AUTH_CODE", authResponse.getAuthorizationCode());
                resultBuilder.addUpdatedDataElement("TRANSACTION_ID", authResponse.getTransactionId());
                resultBuilder.addUpdatedDataElement("AUTHORIZED_AMOUNT", authResponse.getAuthorizedAmount().toString());

                return resultBuilder.build();
            } else {
                return buildAccountingDeclineResult(requestContext, authResponse);
            }

        } catch (Exception e) {
            logger.error("Error calling Accounting API for authorization: {}", e.getMessage(), e);
            requestContext.setInternalResponseCode(InternalResCode.SYSTEM_ERROR);
            return ValidationResult.builder()
                    .addError("AUTHORIZATION_API_ERROR: " + e.getMessage())
                    .build();
        }
    }

        return (transactionType == TransactionType.BALANCE_ENQUIRY

                || transactionType == TransactionType.CASH_WITHDRAWAL)

                && Character.valueOf('P').equals(cardContext.getCardData().getCard_type());

    private boolean isPrepaidAtmSource(RequestContext requestContext, CardDataContext cardContext) {
        if (cardContext == null
                || cardContext.getCardData() == null
                || !Character.valueOf('P').equals(cardContext.getCardData().getCard_type())) {
            return false;
        }
        return isAtmSource(requestContext);
    }

    private boolean isAtmSource(RequestContext requestContext) {
        if (requestContext == null) {
            return false;
        }
        String merchantType = requestContext.getFields().get(18);
        if ("6011".equals(merchantType)) {
            return true;
        }
        if (requestContext.getScheme() == Scheme.RUPAY) {
            try {
                String source = requestContext.getParsedTlv(48).get("051");
                return source != null && source.trim().toUpperCase().startsWith("ATM");
            } catch (Exception ex) {
                logger.debug("Unable to parse DE48 tag 051 for ATM source detection: {}", ex.getMessage());
            }
        }
        return false;
    }

    private ChipData getOriginalChipData(RequestContext requestContext) {
        Object originalChipData = requestContext.getAttribute(ORIGINAL_CHIP_DATA_ATTRIBUTE);
        if (originalChipData instanceof ChipData chipData) {
            return chipData;

        return requestContext.getParsedChipData(55);

    private InternalResCode mapExternalErrorCodeToInternal(String errorCode) {
        if (errorCode == null || errorCode.isBlank()) {
            return InternalResCode.DO_NOT_HONOR;

        return switch (errorCode) {
            case "51" -> InternalResCode.INSUFFICIENT_FUNDS;
            case "54" -> InternalResCode.EXPIRED_CARD;
            case "55" -> InternalResCode.INVALID_PIN;
            case "57" -> InternalResCode.REVERSAL_NOT_PERMITTED;
            case "61" -> InternalResCode.EXCEEDS_LIMIT;
            case "62" -> InternalResCode.RESTRICTED_CARD;
            case "63" -> InternalResCode.SECURITY_VIOLATION;

    public String updateDE48(String de48, String matcher, TransactionType transactionType) {

                        ? CvvVerificationType.CVV2

    private String updateDE48(String de48, String matcher, CvvVerificationRequest cvvRequest) {
        return updateDE48(de48, matcher, cvvRequest.type);
    }

    private String updateDE48(String de48, String matcher, CvvVerificationType cvvType) {
        // Update DE48 tag data if present and required
        // DE48 - Additional Data Private
        logger.debug("DE48 tag data: {}", de48);
        try {
            Map<String, String> tlvMap = new java.util.LinkedHashMap<>();
            if (de48 != null && !de48.trim().isEmpty()) {
                logger.debug("Processing DE48 tag data for update");
                tlvMap.putAll(com.logicpoint.iso8583.validation.TlvUtils.parseTlv(de48));
            }

            String resultTag = cvvType == CvvVerificationType.CVV2 ? "053" : "054";
            tlvMap.put(resultTag, matcher);
            tlvMap.remove("058");

            StringBuilder updatedDe48 = new StringBuilder();
            for (Map.Entry<String, String> entry : tlvMap.entrySet()) {
                String tag = entry.getKey();
                String value = entry.getValue();
                String length = String.format("%03d", value.length());
                updatedDe48.append(tag).append(length).append(value);
            }
            logger.info("DE48 tag {} updated in transaction data", updatedDe48);
            return updatedDe48.toString();
        } catch (Exception ex) {
            logger.warn("Failed to parse or update DE48 tag data: {}", ex.getMessage());
            return de48; // Return original if update fails
        }
    }
