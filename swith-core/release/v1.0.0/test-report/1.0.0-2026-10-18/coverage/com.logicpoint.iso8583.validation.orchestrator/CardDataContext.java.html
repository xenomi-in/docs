package com.logicpoint.iso8583.validation.orchestrator;

import com.logicpoint.iso8583.entity.CardData;
import lombok.Data;
import lombok.Builder;
import lombok.NoArgsConstructor;
import lombok.AllArgsConstructor;

/**
 * Context class to hold all card data fetched once from the database.
 * This prevents multiple database calls during transaction validation.
 * 
 * @author LogicPoint Systems
 * @version 1.0
 * @since 2025-01-27
 */
@Data
@Builder
@NoArgsConstructor
@AllArgsConstructor
public class CardDataContext {
    
    /**
     * The card data entity containing all card information
     */
    private CardData cardData;
    
    /**
     * Flag indicating if card data was successfully fetched
     */
    private boolean cardDataFetched;
    
    /**
     * Error message if card data fetching failed
     */
    private String fetchError;
    
    /**
     * Timestamp when card data was fetched (for cache invalidation)
     */
    private long fetchTimestamp;
    
    /**
     * Creates a successful card data context
     */
    public static CardDataContext success(CardData cardData) {
        return CardDataContext.builder()
                .cardData(cardData)
                .cardDataFetched(true)
                .fetchTimestamp(System.currentTimeMillis())
                .build();
    }
    
    /**
     * Creates a failed card data context
     */
    public static CardDataContext failure(String error) {
        return CardDataContext.builder()
                .cardDataFetched(false)
                .fetchError(error)
                .fetchTimestamp(System.currentTimeMillis())
                .build();
    }
    
    /**
     * Checks if card data is available and valid
     */
    public boolean isValid() {
        return cardDataFetched && cardData != null;
    }
    
    /**
     * Gets the card status safely
     */
    public String getCardStatus() {
        return isValid() ? cardData.getStatus() : null;
    }
    
    /**
     * Gets the expiry date safely
     */
    public String getExpiryDate() {
        return isValid() ? cardData.getExpiryDate() : null;
    }
    
    /**
     * Gets the PVV offset safely
     */
    public String getPvvOffset() {
        return isValid() ? cardData.getPvv_offest() : null;
    }
    
    /**
     * Gets the key reference safely
     */
    public String getKeyRef() {
        return isValid() ? cardData.getKey_ref() : null;
    }
    
    /**
     * Gets the CVV key reference safely
     */
    public String getCvvKeyRef() {
        return isValid() ? cardData.getCvv_key_ref() : null;
    }
    
    /**
     * Gets the EMV AIP safely
     */
    public String getEmvAip() {
        return isValid() ? cardData.getEmv_aip() : null;
    }

    public String getCvv(){
    return isValid() ? cardData.getCvv() : null;
    }
    
    /**
     * Gets the EMV CAPK ID safely
     */
    public String getEmvCapkId() {
        return isValid() ? cardData.getEmv_capk_id() : null;
    }
    
    /**
     * Gets the ARQC validation required flag safely
     */
    public Boolean isArqcValidationRequired() {
        return isValid() ? cardData.getArqc_validation_required() : null;
    }
    
    /**
     * Gets the ATC safely
     */
    public Integer getAtc() {
        return isValid() ? cardData.getAtc() : null;
    }
    
    /**
     * Gets the PIN retry count safely
     */
    public Integer getPinRetryCount() {
        return isValid() ? cardData.getPin_retry_count() : null;
    }
    
    /**
     * Gets the max PIN retry safely
     */
    public Integer getMaxPinRetry() {
        return isValid() ? cardData.getMax_pin_retry() : null;
    }
    
    /**
     * Gets the PIN blocked flag safely
     */
    public Boolean isPinBlocked() {
        return isValid() ? cardData.getPin_blocked() : null;
    }
}
