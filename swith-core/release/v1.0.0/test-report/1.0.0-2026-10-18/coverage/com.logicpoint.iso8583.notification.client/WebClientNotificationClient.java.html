package com.logicpoint.iso8583.notification.client;

import com.logicpoint.iso8583.notification.dto.NotificationPublishResponse;
import com.logicpoint.iso8583.notification.dto.NotificationRequest;
import com.logicpoint.iso8583.notification.exception.NonRetryableNotificationException;
import com.logicpoint.iso8583.notification.exception.RetryableNotificationException;
import io.github.resilience4j.circuitbreaker.annotation.CircuitBreaker;
import java.util.Objects;
import org.springframework.beans.factory.annotation.Qualifier;
import org.springframework.http.HttpHeaders;
import org.springframework.http.HttpStatus;
import org.springframework.retry.annotation.Backoff;
import org.springframework.retry.annotation.Retryable;
import org.springframework.stereotype.Component;
import org.springframework.web.reactive.function.client.WebClient;
import org.springframework.web.reactive.function.client.WebClientRequestException;
import org.springframework.web.reactive.function.client.WebClientResponseException;

@Component
public class WebClientNotificationClient implements NotificationClient {

    private static final String PUBLISH_PATH = "/notifications/publish";
    private static final String TRACE_ID_HEADER = "X-Trace-Id";
    private static final String EVENT_ID_HEADER = "X-Event-Id";

    private final WebClient webClient;

    public WebClientNotificationClient(@Qualifier("notificationWebClient") WebClient webClient) {
        this.webClient = webClient;
    }

    @Override
    @CircuitBreaker(name = "notificationService")
    @Retryable(
            retryFor = RetryableNotificationException.class,
            noRetryFor = NonRetryableNotificationException.class,
            maxAttemptsExpression = "${notification.retry.max-attempts:3}",
            backoff = @Backoff(delayExpression = "${notification.retry.backoff-millis:500}")
    )
    public NotificationPublishResponse publish(NotificationRequest request, String traceId) {
        try {
            return webClient.post()
                    .uri(PUBLISH_PATH)
                    .header(TRACE_ID_HEADER, traceId)
                    .bodyValue(request)
                    .exchangeToMono(response -> {
                        HttpStatus status = HttpStatus.valueOf(response.statusCode().value());
                        if (status == HttpStatus.ACCEPTED) {
                            HttpHeaders headers = response.headers().asHttpHeaders();
                            String responseTraceId = firstNonBlank(headers.getFirst(TRACE_ID_HEADER), traceId);
                            return response.releaseBody()
                                    .thenReturn(new NotificationPublishResponse(headers.getFirst(EVENT_ID_HEADER), responseTraceId));
                        }
                        if (status == HttpStatus.SERVICE_UNAVAILABLE) {
                            return response.bodyToMono(String.class)
                                    .defaultIfEmpty("")
                                    .map(body -> new RetryableNotificationException("Notification service unavailable: " + body))
                                    .flatMap(error -> reactor.core.publisher.Mono.error(error));
                        }
                        return response.bodyToMono(String.class)
                                .defaultIfEmpty("")
                                .map(body -> new NonRetryableNotificationException(
                                        "Notification publish rejected with HTTP " + status.value() + ": " + body))
                                .flatMap(error -> reactor.core.publisher.Mono.error(error));
                    })
                    .block();
        } catch (WebClientRequestException ex) {
            throw new RetryableNotificationException("Notification service request failed", ex);
        } catch (WebClientResponseException.ServiceUnavailable ex) {
            throw new RetryableNotificationException("Notification service unavailable", ex);
        } catch (WebClientResponseException.BadRequest ex) {
            throw new NonRetryableNotificationException("Notification request validation failed: " + ex.getResponseBodyAsString());
        }
    }

    private String firstNonBlank(String value, String fallback) {
        return value == null || value.isBlank() ? Objects.requireNonNull(fallback) : value;
    }
}
