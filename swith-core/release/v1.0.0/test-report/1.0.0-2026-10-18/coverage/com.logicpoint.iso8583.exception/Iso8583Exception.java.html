package com.logicpoint.iso8583.exception;
/*
 * This is the exception for the ISO 8583 messages.
 * It is used to handle the exceptions for the ISO 8583 messages.
 * It is also used to handle the exceptions for the ISO 8583 messages.
 * 
 * @Author: LogicPoint
 * @Version: 1.0
 * @Since: 1.0
 * @Date: 2024-01-01
 * @LastModified: 2024-01-01
 * @LastModifiedBy: LogicPoint
 * @LastModifiedVersion: 1.0
 */
public class Iso8583Exception extends RuntimeException {
    private final String errorCode;
    private final String errorMessage;

    public Iso8583Exception(String errorCode, String errorMessage) {
        super(errorMessage);
        this.errorCode = errorCode;
        this.errorMessage = errorMessage;
    }

    public Iso8583Exception(String errorCode, String errorMessage, Throwable cause) {
        super(errorMessage, cause);
        this.errorCode = errorCode;
        this.errorMessage = errorMessage;
    }

    public String getErrorCode() {
        return errorCode;
    }

    public String getErrorMessage() {
        return errorMessage;
    }
}
