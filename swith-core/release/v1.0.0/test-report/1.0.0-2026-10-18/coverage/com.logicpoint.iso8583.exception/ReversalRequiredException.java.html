package com.logicpoint.iso8583.exception;

import java.util.Map;
/**
 * Base exception for reversal-triggering scenarios
 */
public abstract class ReversalRequiredException extends Iso8583Exception {
    private final Map<Integer, String> transactionData;
    private final String accountingTransactionId;
    
    public ReversalRequiredException(String errorCode, String errorMessage, 
                                    Map<Integer, String> transactionData,
                                    String accountingTransactionId) {
        super(errorCode, errorMessage);
        this.transactionData = transactionData;
        this.accountingTransactionId = accountingTransactionId;
    }
    
    public Map<Integer, String> getTransactionData() {
        return transactionData;
    }
    
    public String getAccountingTransactionId() {
        return accountingTransactionId;
    }
}
