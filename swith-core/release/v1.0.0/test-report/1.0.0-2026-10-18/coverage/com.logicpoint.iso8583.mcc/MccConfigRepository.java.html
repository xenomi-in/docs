package com.logicpoint.iso8583.mcc;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.dao.DataAccessException;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.stereotype.Repository;

import java.util.List;

@Repository
class MccConfigRepository {
    private static final Logger log = LoggerFactory.getLogger(MccConfigRepository.class);

    private final JdbcTemplate jdbcTemplate;

    MccConfigRepository(JdbcTemplate jdbcTemplate) {
        this.jdbcTemplate = jdbcTemplate;
    }

    List<MccRow> loadMccRows() {
        return queryList("""
                SELECT mcc_code
                FROM mcc
                WHERE is_valid = TRUE
                """, (rs, rowNum) -> new MccRow(rs.getInt("mcc_code")), "mcc");
    }

    List<MccGroupMapRow> loadMccGroupRows() {
        return queryList("""
                SELECT mgm.mcc_code, mg.bit_index
                FROM mcc_group_map mgm
                JOIN mcc_group mg ON mg.group_id = mgm.group_id
                WHERE mg.bit_index BETWEEN 0 AND 62
                """, (rs, rowNum) -> new MccGroupMapRow(
                rs.getInt("mcc_code"),
                rs.getInt("bit_index")
        ), "mcc_group_map");
    }

    List<SchemeMccValidityRow> loadSchemeValidityRows() {
        return queryList("""
                SELECT scheme, mcc_code, is_valid
                FROM scheme_mcc
                WHERE effective_from <= CURRENT_DATE
                  AND (effective_to IS NULL OR effective_to >= CURRENT_DATE)
                """, (rs, rowNum) -> new SchemeMccValidityRow(
                rs.getString("scheme"),
                rs.getInt("mcc_code"),
                rs.getBoolean("is_valid")
        ), "scheme_mcc");
    }

    List<SchemeGroupOverrideRow> loadSchemeGroupOverrides() {
        return queryList("""
                SELECT ovr.scheme, ovr.mcc_code, grp.bit_index, ovr.override_action
                FROM scheme_mcc_group_override ovr
                JOIN mcc_group grp ON grp.group_id = ovr.group_id
                WHERE ovr.effective_from <= CURRENT_DATE
                  AND (ovr.effective_to IS NULL OR ovr.effective_to >= CURRENT_DATE)
                  AND grp.bit_index BETWEEN 0 AND 62
                """, (rs, rowNum) -> new SchemeGroupOverrideRow(
                rs.getString("scheme"),
                rs.getInt("mcc_code"),
                rs.getInt("bit_index"),
                rs.getInt("override_action")
        ), "scheme_mcc_group_override");
    }

    List<MccRuleRow> loadRuleRows() {
        return queryList("""
                SELECT r.rule_id, r.institution_id, r.scheme, r.priority, r.rule_type,
                       r.mcc_code, grp.bit_index AS group_bit_index,
                       action, max_amount_minor, response_code
                FROM institution_mcc_rule r
                LEFT JOIN mcc_group grp ON grp.group_id = r.group_id
                WHERE r.active = TRUE
                  AND r.effective_from <= CURRENT_DATE
                  AND (r.effective_to IS NULL OR r.effective_to >= CURRENT_DATE)
                ORDER BY r.institution_id, r.scheme, r.priority, r.rule_id
                """, (rs, rowNum) -> new MccRuleRow(
                rs.getInt("rule_id"),
                rs.getInt("institution_id"),
                rs.getString("scheme"),
                rs.getInt("priority"),
                rs.getInt("rule_type"),
                rs.getObject("mcc_code", Integer.class),
                rs.getObject("group_bit_index", Integer.class),
                rs.getInt("action"),
                rs.getObject("max_amount_minor", Long.class),
                rs.getString("response_code")
        ), "institution_mcc_rule");
    }

    private <T> List<T> queryList(String sql, org.springframework.jdbc.core.RowMapper<T> mapper, String source) {
        try {
            return jdbcTemplate.query(sql, mapper);
        } catch (DataAccessException ex) {
            log.warn("MCC config table/query unavailable: {}", source);
            return List.of();
        }
    }

    record MccRow(int mccCode) {
    }

    record MccGroupMapRow(int mccCode, int bitIndex) {
    }

    record SchemeMccValidityRow(String scheme, int mccCode, boolean valid) {
    }

    record SchemeGroupOverrideRow(String scheme, int mccCode, int bitIndex, int overrideAction) {
    }

    record MccRuleRow(
            int ruleId,
            int institutionId,
            String scheme,
            int priority,
            int ruleType,
            Integer mccCode,
            Integer groupBitIndex,
            int action,
            Long maxAmountMinor,
            String responseCode
    ) {
    }
}
