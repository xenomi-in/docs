package com.logicpoint.iso8583.mcc;

import com.logicpoint.iso8583.context.RequestContext;
import com.logicpoint.iso8583.responsecode.InternalResCode;
import com.logicpoint.iso8583.service.ResponseAssembler;
import org.springframework.stereotype.Service;

@Service
public class MccRuleEngine {
    private static final String ATTR_MCC = "mcc";
    private static final String ATTR_MCC_GROUP_MASK = "mccGroupMask";
    private static final String ATTR_MCC_RULE_ID = "mccRuleId";

    private final MccConfigService configService;
    private final ResponseAssembler responseAssembler;
    private final ThreadLocal<MccEvaluationContext> evalContext =
            ThreadLocal.withInitial(MccEvaluationContext::new);

    public MccRuleEngine(MccConfigService configService, ResponseAssembler responseAssembler) {
        this.configService = configService;
        this.responseAssembler = responseAssembler;
    }

    public boolean validate(RequestContext requestContext) {
        MccEvaluationContext ctx = evalContext.get();
        ctx.reset();

        String rawMcc = requestContext.getFields().get(18);
        if (rawMcc == null || rawMcc.isBlank()) {
            decline(requestContext, InternalResCode.MISSING_MCC);
            return false;
        }

        int mcc = parseMcc(rawMcc);
        if (mcc < 0) {
            decline(requestContext, InternalResCode.INVALID_MCC);
            return false;
        }

        int institutionId = parseInstitutionId(requestContext.getInstitutionId());
        if (institutionId <= 0) {
            decline(requestContext, InternalResCode.NO_SUCH_ISSUER);
            return false;
        }

        int schemeSlot = MccSchemeSlots.slot(requestContext.getScheme());
        MccRuntimeConfig config = configService.current();
        if (!config.loaded()) {
            decline(requestContext, InternalResCode.SYSTEM_ERROR);
            return false;
        }
        MccTable table = config.table();
        if (!table.isValid(schemeSlot, mcc)) {
            decline(requestContext, InternalResCode.INVALID_MCC);
            return false;
        }

        ctx.institutionId(institutionId);
        ctx.schemeSlot(schemeSlot);
        ctx.mcc(mcc);
        ctx.amountMinor(parseAmountMinor(requestContext.getFields().get(4)));
        ctx.groupMask(table.groupMask(schemeSlot, mcc));

        requestContext.setAttribute(ATTR_MCC, mcc);
        requestContext.setAttribute(ATTR_MCC_GROUP_MASK, ctx.groupMask());

        MccDecision decision = config.rules(institutionId, schemeSlot).evaluate(ctx);
        if (decision == MccDecision.CONTINUE || decision == MccDecision.ALLOW) {
            return true;
        }

        requestContext.setAttribute(ATTR_MCC_RULE_ID, ctx.matchedRuleId());
        InternalResCode responseCode = internalCode(ctx.matchedReasonCode());
        decline(requestContext, responseCode);
        return false;
    }

    private void decline(RequestContext requestContext, InternalResCode responseCode) {
        responseAssembler.applyResponseCode(requestContext, responseCode);
    }

    private int parseMcc(String rawMcc) {
        if (rawMcc.length() != 4) {
            return -1;
        }
        int d0 = rawMcc.charAt(0) - '0';
        int d1 = rawMcc.charAt(1) - '0';
        int d2 = rawMcc.charAt(2) - '0';
        int d3 = rawMcc.charAt(3) - '0';
        if ((d0 | d1 | d2 | d3) < 0 || d0 > 9 || d1 > 9 || d2 > 9 || d3 > 9) {
            return -1;
        }
        return d0 * 1000 + d1 * 100 + d2 * 10 + d3;
    }

    private long parseAmountMinor(String rawAmount) {
        if (rawAmount == null || rawAmount.isBlank()) {
            return 0L;
        }
        long value = 0L;
        int len = rawAmount.length();
        for (int i = 0; i < len; i++) {
            int digit = rawAmount.charAt(i) - '0';
            if (digit < 0 || digit > 9) {
                return 0L;
            }
            value = value * 10L + digit;
        }
        return value;
    }

    private int parseInstitutionId(String institutionId) {
        if (institutionId == null || institutionId.isBlank()) {
            return 0;
        }
        int value = 0;
        for (int i = 0; i < institutionId.length(); i++) {
            int digit = institutionId.charAt(i) - '0';
            if (digit < 0 || digit > 9) {
                return 0;
            }
            value = value * 10 + digit;
        }
        return value;
    }

    private InternalResCode internalCode(int ordinal) {
        InternalResCode[] values = InternalResCode.values();
        if (ordinal < 0 || ordinal >= values.length) {
            return InternalResCode.MCC_NOT_ALLOWED;
        }
        return values[ordinal];
    }
}
