package com.logicpoint.iso8583.mcc;

final class MccTable {
    static final MccTable EMPTY = new MccTable(
            new long[(MccConstants.MCC_LIMIT + 63) >>> 6],
            new long[MccConstants.SCHEME_SLOT_COUNT][(MccConstants.MCC_LIMIT + 63) >>> 6],
            new long[MccConstants.SCHEME_SLOT_COUNT][(MccConstants.MCC_LIMIT + 63) >>> 6],
            new long[MccConstants.MCC_LIMIT],
            new long[MccConstants.SCHEME_SLOT_COUNT][MccConstants.MCC_LIMIT]
    );

    private final long[] canonicalValidBits;
    private final long[][] schemeValidBits;
    private final long[][] schemeInvalidBits;
    private final long[] canonicalGroupMaskByMcc;
    private final long[][] schemeGroupMaskByMcc;

    MccTable(
            long[] canonicalValidBits,
            long[][] schemeValidBits,
            long[][] schemeInvalidBits,
            long[] canonicalGroupMaskByMcc,
            long[][] schemeGroupMaskByMcc
    ) {
        this.canonicalValidBits = canonicalValidBits;
        this.schemeValidBits = schemeValidBits;
        this.schemeInvalidBits = schemeInvalidBits;
        this.canonicalGroupMaskByMcc = canonicalGroupMaskByMcc;
        this.schemeGroupMaskByMcc = schemeGroupMaskByMcc;
    }

    boolean isValid(int schemeSlot, int mcc) {
        if (mcc < 0 || mcc >= MccConstants.MCC_LIMIT) {
            return false;
        }
        if (schemeSlot > 0 && schemeSlot < schemeValidBits.length) {
            if (bitSet(schemeInvalidBits[schemeSlot], mcc)) {
                return false;
            }
            long[] schemeBits = schemeValidBits[schemeSlot];
            if (bitSet(schemeBits, mcc)) {
                return true;
            }
        }
        return bitSet(canonicalValidBits, mcc);
    }

    long groupMask(int schemeSlot, int mcc) {
        if (mcc < 0 || mcc >= MccConstants.MCC_LIMIT) {
            return 0L;
        }
        if (schemeSlot > 0 && schemeSlot < schemeGroupMaskByMcc.length) {
            long schemeMask = schemeGroupMaskByMcc[schemeSlot][mcc];
            if (schemeMask != 0L) {
                return schemeMask;
            }
        }
        return canonicalGroupMaskByMcc[mcc];
    }

    static void setBit(long[] bits, int value) {
        bits[value >>> 6] |= 1L << (value & 63);
    }

    private static boolean bitSet(long[] bits, int value) {
        return (bits[value >>> 6] & (1L << (value & 63))) != 0L;
    }
}
