package com.logicpoint.iso8583.mcc;

import com.logicpoint.iso8583.util.Scheme;

final class MccSchemeSlots {
    private MccSchemeSlots() {
    }

    static int slot(Scheme scheme) {
        if (scheme == null) {
            return 0;
        }
        return switch (scheme) {
            case VISA -> 1;
            case MASTERCARD -> 2;
            case RUPAY -> 3;
            case AMEX -> 4;
            case DISCOVER -> 5;
            case UNIONPAY -> 6;
            case DINERS -> 7;
        };
    }

    static int slot(String scheme) {
        if (scheme == null || scheme.isBlank()) {
            return 0;
        }
        try {
            return slot(Scheme.valueOf(scheme.trim().toUpperCase()));
        } catch (IllegalArgumentException ex) {
            return 0;
        }
    }
}
