package com.logicpoint.iso8583.mcc;

final class MccCompiledRules {
    static final MccCompiledRules EMPTY = new MccCompiledRules(
            new int[0],
            new int[0],
            new long[0],
            new long[0],
            new int[0],
            new int[0],
            new int[0]
    );

    private final int[] ruleIds;
    private final int[] opcodes;
    private final long[] longArg0;
    private final long[] longArg1;
    private final int[] action;
    private final int[] reasonCodes;
    private final int[] exactMcc;

    MccCompiledRules(
            int[] ruleIds,
            int[] opcodes,
            long[] longArg0,
            long[] longArg1,
            int[] action,
            int[] reasonCodes,
            int[] exactMcc
    ) {
        this.ruleIds = ruleIds;
        this.opcodes = opcodes;
        this.longArg0 = longArg0;
        this.longArg1 = longArg1;
        this.action = action;
        this.reasonCodes = reasonCodes;
        this.exactMcc = exactMcc;
    }

    MccDecision evaluate(MccEvaluationContext ctx) {
        int[] localOps = opcodes;
        for (int i = 0; i < localOps.length; i++) {
            if (!matches(i, localOps[i], ctx)) {
                continue;
            }
            ctx.matchedRuleId(ruleIds[i]);
            ctx.matchedReasonCode(reasonCodes[i]);
            return toDecision(action[i]);
        }
        return MccDecision.CONTINUE;
    }

    private boolean matches(int index, int opcode, MccEvaluationContext ctx) {
        return switch (opcode) {
            case MccConstants.RULE_EXACT_MCC -> ctx.mcc() == exactMcc[index];
            case MccConstants.RULE_MCC_GROUP -> (ctx.groupMask() & longArg0[index]) != 0L;
            case MccConstants.RULE_GROUP_AMOUNT_LIMIT ->
                    (ctx.groupMask() & longArg0[index]) != 0L && ctx.amountMinor() > longArg1[index];
            default -> false;
        };
    }

    private MccDecision toDecision(int ruleAction) {
        return switch (ruleAction) {
            case MccConstants.ACTION_ALLOW -> MccDecision.ALLOW;
            case MccConstants.ACTION_DECLINE -> MccDecision.DECLINE;
            case MccConstants.ACTION_STEP_UP -> MccDecision.STEP_UP;
            default -> MccDecision.CONTINUE;
        };
    }
}
