package com.logicpoint.iso8583.mcc;

import com.logicpoint.iso8583.responsecode.InternalResCode;
import jakarta.annotation.PostConstruct;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.scheduling.annotation.Scheduled;
import org.springframework.stereotype.Service;

import java.util.ArrayList;
import java.util.Comparator;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

@Service
class MccConfigService {
    private static final Logger log = LoggerFactory.getLogger(MccConfigService.class);
    private static final int ADD_GROUP = 1;
    private static final int REMOVE_GROUP = 2;

    private final MccConfigRepository repository;
    private volatile MccRuntimeConfig current = MccRuntimeConfig.EMPTY;

    MccConfigService(MccConfigRepository repository) {
        this.repository = repository;
    }

    @PostConstruct
    public void loadAtStartup() {
        refresh();
    }

    @Scheduled(fixedDelayString = "${mcc.cache.refresh.ms:300000}")
    public void refresh() {
        long startNanos = System.nanoTime();
        MccRuntimeConfig next = buildConfig();
        current = next;
        log.info("Loaded MCC runtime config in {} ms", (System.nanoTime() - startNanos) / 1_000_000L);
    }

    MccRuntimeConfig current() {
        return current;
    }

    private MccRuntimeConfig buildConfig() {
        List<MccConfigRepository.MccRow> mccRows = repository.loadMccRows();
        if (mccRows.isEmpty()) {
            log.warn("No MCC master data loaded; MCC validation will return SYSTEM_ERROR");
            return MccRuntimeConfig.EMPTY;
        }
        MccTable table = buildMccTable(mccRows);
        Map<Long, MccCompiledRules> rules = buildRules(table);
        return new MccRuntimeConfig(table, Map.copyOf(rules), true);
    }

    private MccTable buildMccTable(List<MccConfigRepository.MccRow> mccRows) {
        long[] canonicalValidBits = new long[(MccConstants.MCC_LIMIT + 63) >>> 6];
        long[][] schemeValidBits = new long[MccConstants.SCHEME_SLOT_COUNT][(MccConstants.MCC_LIMIT + 63) >>> 6];
        long[][] schemeInvalidBits = new long[MccConstants.SCHEME_SLOT_COUNT][(MccConstants.MCC_LIMIT + 63) >>> 6];
        long[] canonicalGroupMaskByMcc = new long[MccConstants.MCC_LIMIT];
        long[][] schemeGroupMaskByMcc = new long[MccConstants.SCHEME_SLOT_COUNT][MccConstants.MCC_LIMIT];

        for (MccConfigRepository.MccRow row : mccRows) {
            if (validMcc(row.mccCode())) {
                MccTable.setBit(canonicalValidBits, row.mccCode());
            }
        }

        for (MccConfigRepository.MccGroupMapRow row : repository.loadMccGroupRows()) {
            if (validMcc(row.mccCode()) && row.bitIndex() >= 0 && row.bitIndex() < MccConstants.GROUP_LIMIT) {
                canonicalGroupMaskByMcc[row.mccCode()] |= 1L << row.bitIndex();
            }
        }

        for (MccConfigRepository.SchemeMccValidityRow row : repository.loadSchemeValidityRows()) {
            int schemeSlot = MccSchemeSlots.slot(row.scheme());
            if (schemeSlot > 0 && validMcc(row.mccCode())) {
                if (row.valid()) {
                    MccTable.setBit(schemeValidBits[schemeSlot], row.mccCode());
                } else {
                    MccTable.setBit(schemeInvalidBits[schemeSlot], row.mccCode());
                }
            }
        }

        for (MccConfigRepository.SchemeGroupOverrideRow row : repository.loadSchemeGroupOverrides()) {
            int schemeSlot = MccSchemeSlots.slot(row.scheme());
            if (schemeSlot <= 0 || !validMcc(row.mccCode()) || row.bitIndex() < 0 || row.bitIndex() >= MccConstants.GROUP_LIMIT) {
                continue;
            }
            long existing = schemeGroupMaskByMcc[schemeSlot][row.mccCode()];
            if (existing == 0L) {
                existing = canonicalGroupMaskByMcc[row.mccCode()];
            }
            long bit = 1L << row.bitIndex();
            if (row.overrideAction() == ADD_GROUP) {
                schemeGroupMaskByMcc[schemeSlot][row.mccCode()] = existing | bit;
            } else if (row.overrideAction() == REMOVE_GROUP) {
                schemeGroupMaskByMcc[schemeSlot][row.mccCode()] = existing & ~bit;
            }
        }

        return new MccTable(canonicalValidBits, schemeValidBits, schemeInvalidBits, canonicalGroupMaskByMcc, schemeGroupMaskByMcc);
    }

    private Map<Long, MccCompiledRules> buildRules(MccTable table) {
        List<MccConfigRepository.MccRuleRow> rows = new ArrayList<>(repository.loadRuleRows());
        rows.sort(Comparator
                .comparingInt(MccConfigRepository.MccRuleRow::institutionId)
                .thenComparingInt(row -> MccSchemeSlots.slot(row.scheme()))
                .thenComparingInt(MccConfigRepository.MccRuleRow::priority)
                .thenComparingInt(MccConfigRepository.MccRuleRow::ruleId));

        Map<Long, List<MccConfigRepository.MccRuleRow>> grouped = new HashMap<>();
        for (MccConfigRepository.MccRuleRow row : rows) {
            int schemeSlot = MccSchemeSlots.slot(row.scheme());
            if (row.institutionId() <= 0 || schemeSlot <= 0 || !validAction(row.action())) {
                continue;
            }
            grouped.computeIfAbsent(MccRuntimeConfig.ruleKey(row.institutionId(), schemeSlot), ignored -> new ArrayList<>()).add(row);
        }

        Map<Long, MccCompiledRules> compiled = new HashMap<>(grouped.size());
        for (Map.Entry<Long, List<MccConfigRepository.MccRuleRow>> entry : grouped.entrySet()) {
            compiled.put(entry.getKey(), compileRules(entry.getValue(), table));
        }
        return compiled;
    }

    private MccCompiledRules compileRules(List<MccConfigRepository.MccRuleRow> rows, MccTable table) {
        int size = rows.size();
        int[] ruleIds = new int[size];
        int[] opcodes = new int[size];
        long[] longArg0 = new long[size];
        long[] longArg1 = new long[size];
        int[] actions = new int[size];
        int[] reasonCodes = new int[size];
        int[] exactMcc = new int[size];

        int count = 0;
        for (MccConfigRepository.MccRuleRow row : rows) {
            int opcode = row.ruleType();
            if (opcode == MccConstants.RULE_EXACT_MCC) {
                if (row.mccCode() == null || !validMcc(row.mccCode())) {
                    continue;
                }
                exactMcc[count] = row.mccCode();
            } else if (opcode == MccConstants.RULE_MCC_GROUP || opcode == MccConstants.RULE_GROUP_AMOUNT_LIMIT) {
                if (row.groupBitIndex() == null || row.groupBitIndex() < 0 || row.groupBitIndex() >= MccConstants.GROUP_LIMIT) {
                    continue;
                }
                longArg0[count] = 1L << row.groupBitIndex();
                if (opcode == MccConstants.RULE_GROUP_AMOUNT_LIMIT) {
                    longArg1[count] = row.maxAmountMinor() == null ? Long.MAX_VALUE : row.maxAmountMinor();
                }
            } else {
                continue;
            }

            ruleIds[count] = row.ruleId();
            opcodes[count] = opcode;
            actions[count] = row.action();
            reasonCodes[count] = responseOrdinal(row.responseCode(), row.action());
            count++;
        }

        if (count == size) {
            return new MccCompiledRules(ruleIds, opcodes, longArg0, longArg1, actions, reasonCodes, exactMcc);
        }
        return new MccCompiledRules(
                copy(ruleIds, count),
                copy(opcodes, count),
                copy(longArg0, count),
                copy(longArg1, count),
                copy(actions, count),
                copy(reasonCodes, count),
                copy(exactMcc, count)
        );
    }

    private int responseOrdinal(String configuredCode, int action) {
        if (configuredCode != null && !configuredCode.isBlank()) {
            try {
                return InternalResCode.valueOf(configuredCode.trim().toUpperCase()).ordinal();
            } catch (IllegalArgumentException ignored) {
                log.warn("Ignoring unknown MCC response_code {}", configuredCode);
            }
        }
        InternalResCode fallback = action == MccConstants.ACTION_STEP_UP
                ? InternalResCode.MCC_STEP_UP_REQUIRED
                : InternalResCode.MCC_NOT_ALLOWED;
        return fallback.ordinal();
    }

    private boolean validAction(int action) {
        return action == MccConstants.ACTION_ALLOW
                || action == MccConstants.ACTION_DECLINE
                || action == MccConstants.ACTION_STEP_UP;
    }

    private boolean validMcc(int mcc) {
        return mcc >= 0 && mcc < MccConstants.MCC_LIMIT;
    }

    private int[] copy(int[] source, int count) {
        int[] target = new int[count];
        System.arraycopy(source, 0, target, 0, count);
        return target;
    }

    private long[] copy(long[] source, int count) {
        long[] target = new long[count];
        System.arraycopy(source, 0, target, 0, count);
        return target;
    }
}
