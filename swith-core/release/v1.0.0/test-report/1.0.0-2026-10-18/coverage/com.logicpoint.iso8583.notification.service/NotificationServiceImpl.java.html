package com.logicpoint.iso8583.notification.service;

import tools.jackson.core.JacksonException;
import tools.jackson.databind.json.JsonMapper;
import com.logicpoint.iso8583.config.NotificationProperties;
import com.logicpoint.iso8583.context.RequestContext;
import com.logicpoint.iso8583.notification.client.NotificationClient;
import com.logicpoint.iso8583.notification.dto.DeclinedTransactionPayload;
import com.logicpoint.iso8583.notification.dto.NotificationPublishResponse;
import com.logicpoint.iso8583.notification.dto.NotificationRequest;
import com.logicpoint.iso8583.notification.mapper.DeclinedTransactionNotificationMapper;
import java.util.Map;
import java.util.UUID;
import java.util.concurrent.CompletableFuture;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.slf4j.MDC;
import org.springframework.scheduling.annotation.Async;
import org.springframework.stereotype.Service;

@Service
public class NotificationServiceImpl implements NotificationService {

    private static final Logger log = LoggerFactory.getLogger(NotificationServiceImpl.class);
    private static final String APPROVED_RESPONSE_CODE = "00";
    private static final String TRACE_ID_MDC_KEY = "traceId";
    private static final String TRANSACTION_ID_MDC_KEY = "txnId";

    private final NotificationProperties properties;
    private final DeclinedTransactionNotificationMapper mapper;
    private final NotificationClient notificationClient;
    private final JsonMapper objectMapper;

    public NotificationServiceImpl(
            NotificationProperties properties,
            DeclinedTransactionNotificationMapper mapper,
            NotificationClient notificationClient,
            JsonMapper objectMapper
    ) {
        this.properties = properties;
        this.mapper = mapper;
        this.notificationClient = notificationClient;
        this.objectMapper = objectMapper;
    }

    @Override
    @Async("notificationExecutor")
    public CompletableFuture<Void> publishDeclinedAuthorization(RequestContext requestContext, Map<Integer, String> finalResponseFields) {
        if (!properties.isEnabled()) {
            return CompletableFuture.completedFuture(null);
        }
        String responseCode = finalResponseFields.get(39);
        if (responseCode == null || APPROVED_RESPONSE_CODE.equals(responseCode)) {
            return CompletableFuture.completedFuture(null);
        }

        String traceId = resolveTraceId();
        try (MDC.MDCCloseable ignoredTxn = MDC.putCloseable(TRANSACTION_ID_MDC_KEY, requestContext.getUniqueTransactionId());
             MDC.MDCCloseable ignoredTrace = MDC.putCloseable(TRACE_ID_MDC_KEY, traceId)) {
            NotificationRequest request = mapper.toNotificationRequest(requestContext, finalResponseFields, traceId);
            log.info(
                    "Publishing declined authorization notification transactionId={} traceId={} request={}",
                    requestContext.getUniqueTransactionId(),
                    traceId,
                    toJson(maskCardData(request))
            );
            NotificationPublishResponse response = notificationClient.publish(request, traceId);
            log.info(
                    "Declined authorization notification accepted transactionId={} traceId={} eventId={} responseCode={}",
                    requestContext.getUniqueTransactionId(),
                    response.traceId(),
                    response.eventId(),
                    responseCode
            );
        } catch (Exception ex) {
            log.warn(
                    "Declined authorization notification failed transactionId={} traceId={} responseCode={} error={}",
                    requestContext.getUniqueTransactionId(),
                    traceId,
                    responseCode,
                    ex.getMessage(),
                    ex
            );
        }
        return CompletableFuture.completedFuture(null);
    }

    private String resolveTraceId() {
        String traceId = MDC.get(TRACE_ID_MDC_KEY);
        return traceId == null || traceId.isBlank() ? UUID.randomUUID().toString() : traceId;
    }

    private String toJson(Object value) {
        try {
            return objectMapper.writeValueAsString(value);
        } catch (JacksonException ex) {
            return String.valueOf(value);
        }
    }

    private NotificationRequest maskCardData(NotificationRequest request) {
        DeclinedTransactionPayload payload = request.payload();
        DeclinedTransactionPayload maskedPayload = new DeclinedTransactionPayload(
                payload.transactionId(),
                payload.stan(),
                payload.rrn(),
                maskPan(payload.pan()),
                payload.accountReference(),
                payload.processingCode(),
                payload.amount(),
                payload.currency(),
                payload.merchantId(),
                payload.terminalId(),
                payload.channel(),
                payload.responseCode(),
                payload.declineReason(),
                payload.timestamp(),
                payload.switchReference(),
                payload.issuerId(),
                payload.acquirerId()
        );
        return new NotificationRequest(
                request.eventType(),
                maskedPayload,
                request.eventId(),
                request.tenantId(),
                request.customerId(),
                request.aggregateType(),
                request.aggregateId(),
                request.eventSource(),
                request.correlationId(),
                request.channels(),
                request.priority()
        );
    }

    private String maskPan(String pan) {
        if (pan == null || pan.isBlank()) {
            return pan;
        }
        String digitsOnly = pan.replaceAll("\\D", "");
        if (digitsOnly.length() < 10) {
            return "******";
        }
        return digitsOnly.substring(0, 6) + "******" + digitsOnly.substring(digitsOnly.length() - 4);
    }
}
