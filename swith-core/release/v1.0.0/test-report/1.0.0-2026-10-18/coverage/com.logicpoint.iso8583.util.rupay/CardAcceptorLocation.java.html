package com.logicpoint.iso8583.util.rupay;

    /**
     * @author logic point systems
     * @version 1.0.0
     * @since 2025-09-16
     * @description Utility methods for parsing the Card Acceptor Location (DE43) field.
     * Format:
     *  01-23: Terminal Owner Name (right padded with spaces)
     *  24-36: Terminal City (right padded with spaces)  
     *  37-38: Terminal State Code
     *  39-40: Terminal Country Code
     */
   
public class CardAcceptorLocation {
        private final String terminalOwnerName;
        private final String terminalCity;
        private final String terminalStateCode;
        private final String terminalCountryCode;

        public CardAcceptorLocation(String terminalOwnerName, String terminalCity, String terminalStateCode, String terminalCountryCode) {
            this.terminalOwnerName = terminalOwnerName;
            this.terminalCity = terminalCity;
            this.terminalStateCode = terminalStateCode;
            this.terminalCountryCode = terminalCountryCode;
        }

        public String getTerminalOwnerName() {
            return terminalOwnerName;
        }

        public String getTerminalCity() {
            return terminalCity;
        }

        public String getTerminalStateCode() {
            return terminalStateCode;
        }

        public String getTerminalCountryCode() {
            return terminalCountryCode;
        }

        @Override
        public String toString() {
            return "CardAcceptorLocation{" +
                    "terminalOwnerName='" + terminalOwnerName + '\'' +
                    ", terminalCity='" + terminalCity + '\'' +
                    ", terminalStateCode='" + terminalStateCode + '\'' +
                    ", terminalCountryCode='" + terminalCountryCode + '\'' +
                    '}';
        }
    /**
     * Parses the Card Acceptor Location (DE43) field according to the specification.
     * @param de43Value the 40-character DE43 string
     * @return CardAcceptorLocation object with parsed fields
     * @throws IllegalArgumentException if input is null or not 40 characters
     */
    public static CardAcceptorLocation parse(String de43Value) {
        if (de43Value == null || de43Value.length() != 40) {
            throw new IllegalArgumentException("DE43 value must be exactly 40 characters");
        }
        String terminalOwnerName = de43Value.substring(0, 23).trim();
        String terminalCity = de43Value.substring(23, 36).trim();
        String terminalStateCode = de43Value.substring(36, 38);
        String terminalCountryCode = de43Value.substring(38, 40);
        return new CardAcceptorLocation(terminalOwnerName, terminalCity, terminalStateCode, terminalCountryCode);
    }
}
