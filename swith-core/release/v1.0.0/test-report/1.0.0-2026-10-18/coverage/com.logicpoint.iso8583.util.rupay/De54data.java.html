package com.logicpoint.iso8583.util.rupay;

import java.math.BigDecimal;
import java.math.RoundingMode;

public class De54data {
    /**
     * Represents the parsed DE54 data for RuPay as per the specified format.
     */
    public static class RuPayDe54Data {
        private String accountType;      // 2 digits
        private String amountType;       // 2 digits
        private String currencyCode;     // 3 digits
        private char amountSign;         // 1 char: 'C' or 'D'
        private long amount;             // 12 digits, in minor units (e.g., paise)
        private String raw;              // Original DE54 string

        public RuPayDe54Data(String accountType, String amountType, String currencyCode, char amountSign, long amount, String raw) {
            this.accountType = accountType;
            this.amountType = amountType;
            this.currencyCode = currencyCode;
            this.amountSign = amountSign;
            this.amount = amount;
            this.raw = raw;
        }

        public String getAccountType() {
            return accountType;
        }

        public String getAmountType() {
            return amountType;
        }

        public String getCurrencyCode() {
            return currencyCode;
        }

        public char getAmountSign() {
            return amountSign;
        }

        public long getAmount() {
            return amount;
        }

        public String getRaw() {
            return raw;
        }

        @Override
        public String toString() {
            return "RuPayDe54Data{" +
                    "accountType='" + accountType + '\'' +
                    ", amountType='" + amountType + '\'' +
                    ", currencyCode='" + currencyCode + '\'' +
                    ", amountSign=" + amountSign +
                    ", amount=" + amount +
                    ", raw='" + raw + '\'' +
                    '}';
        }
    }

    /**
     * Parses a DE54 string as per RuPay format.
     * @param de54 the DE54 string (should be 20 characters)
     * @return RuPayDe54Data object if valid, null otherwise
     */
    public static RuPayDe54Data parseDe54(String de54) {
        if (de54 == null || de54.length() != 20) {
            return null;
        }
        try {
            String accountType = de54.substring(0, 2);
            String amountType = de54.substring(2, 4);
            String currencyCode = de54.substring(4, 7);
            char amountSign = de54.charAt(7);
            String amountStr = de54.substring(8, 20);

            // Validate sign
            if (amountSign != 'C' && amountSign != 'D') {
                return null;
            }

            // Amount is 12 digits, left-padded with zeros
            long amount = Long.parseLong(amountStr);

            return new RuPayDe54Data(accountType, amountType, currencyCode, amountSign, amount, de54);
        } catch (Exception e) {
            System.out.println("Error parsing DE54: " + e.getMessage());
            return null;
        }
    }

    /**
     * Packs the given RuPayDe54Data into a DE54 string as per RuPay format.
     * @param data RuPayDe54Data object
     * @return DE54 string (20 characters) or null if invalid
     */
    public static String packDe54(RuPayDe54Data data) {
        if (data == null) {
            return null;
        }
        StringBuilder sb = new StringBuilder(20);
        sb.append(padLeft(data.getAccountType(), 2, '0'));
        sb.append(padLeft(data.getAmountType(), 2, '0'));
        sb.append(padLeft(data.getCurrencyCode(), 3, '0'));
        sb.append(data.getAmountSign());
        sb.append(padLeft(Long.toString(data.getAmount()), 12, '0'));
        return sb.toString();
    }

    public static String packBalanceEnquiryDe54(BigDecimal ledgerBalance, BigDecimal availableBalance) {
        return packBalanceResponseDe54("30", "01", "02", ledgerBalance, availableBalance);
    }

    public static String packBalanceResponseDe54(
            String accountType,
            String ledgerAmountType,
            String availableAmountType,
            BigDecimal ledgerBalance,
            BigDecimal availableBalance) {
        RuPayDe54Data ledger = toBalanceAmount(accountType, ledgerAmountType, ledgerBalance);
        RuPayDe54Data available = toBalanceAmount(accountType, availableAmountType, availableBalance);
        return packDe54(ledger) + packDe54(available);
    }

    private static RuPayDe54Data toBalanceAmount(String accountType, String amountType, BigDecimal balance) {
        BigDecimal safeBalance = balance == null ? BigDecimal.ZERO : balance;
        char amountSign = safeBalance.compareTo(BigDecimal.ZERO) < 0 ? 'D' : 'C';
        long minorUnits = safeBalance.abs()
                .multiply(BigDecimal.valueOf(100))
                .setScale(0, RoundingMode.DOWN)
                .longValue();
        return new RuPayDe54Data(accountType, amountType, "356", amountSign, minorUnits, "");
    }

    /**
     * Utility method to left-pad a string with a given character.
     * @param value the string to pad
     * @param length the desired length
     * @param padChar the character to pad with
     * @return padded string
     */
    private static String padLeft(String value, int length, char padChar) {
        if (value == null) value = "";
        if (value.length() >= length) return value.substring(0, length);
        StringBuilder sb = new StringBuilder(length);
        for (int i = value.length(); i < length; i++) {
            sb.append(padChar);
        }
        sb.append(value);
        return sb.toString();
    }
    public static void main(String[] args) {
        String de54 = "9090356D000000000100";
       // 9090356D000000000100
        RuPayDe54Data de54data = parseDe54(de54);
        System.out.println(de54data);
    }
}
