/**
 * Parser for RuPay DE61 (POS Data) field.

        // Subfield 6: Card Present

    /**
     * Parsed DE61 data structure
     */
    public static class RuPayDE61Data {
        private final String rawData;
        private final Map<RuPayDE61Subfields, String> subfieldValues;
        private final List<String> errors;
        private final List<String> warnings;
        
        public RuPayDE61Data(String rawData, Map<RuPayDE61Subfields, String> subfieldValues, 
                           List<String> errors, List<String> warnings) {
            this.rawData = rawData;
            this.subfieldValues = subfieldValues;
            this.errors = errors;
            this.warnings = warnings;
        }
        
        public String getRawData() { return rawData; }
        public Map<RuPayDE61Subfields, String> getSubfieldValues() { return subfieldValues; }
        public List<String> getErrors() { return errors; }
        public List<String> getWarnings() { return warnings; }
        public boolean hasErrors() { return !errors.isEmpty(); }
        public boolean hasWarnings() { return !warnings.isEmpty(); }
        
        public String getSubfieldValue(RuPayDE61Subfields subfield) {
            return subfieldValues.get(subfield);
        }
        
        public CardDataInputCapability getCardDataInputCapability() {
            String value = getSubfieldValue(RuPayDE61Subfields.CARD_DATA_INPUT_CAPABILITY);
            return CardDataInputCapability.fromValue(value);
        }
        
        public CardholderAuthCapability getCardholderAuthCapability() {
            String value = getSubfieldValue(RuPayDE61Subfields.CARDHOLDER_AUTH_CAPABILITY);
            return CardholderAuthCapability.fromValue(value);
        }
        
        public CardCaptureCapability getCardCaptureCapability() {
            String value = getSubfieldValue(RuPayDE61Subfields.CARD_CAPTURE_CAPABILITY);
            return CardCaptureCapability.fromValue(value);
        }
        
        public TerminalOperatingEnvironment getTerminalOperatingEnvironment() {
            String value = getSubfieldValue(RuPayDE61Subfields.TERMINAL_OPERATING_ENVIRONMENT);
            return TerminalOperatingEnvironment.fromValue(value);
        }
    }
    
    /**
     * Parse RuPay DE61 field
     */
    public static RuPayDE61Data parseDE61(String de61Data) {
        Map<RuPayDE61Subfields, String> subfieldValues = new HashMap<>();
        List<String> errors = new ArrayList<>();
        List<String> warnings = new ArrayList<>();
        
        if (de61Data == null || de61Data.trim().isEmpty()) {
            errors.add("DE61 data is null or empty");
            return new RuPayDE61Data(de61Data, subfieldValues, errors, warnings);
        }
        
        // Validate length - DE61 should be at least 12 digits for all subfields
        if (de61Data.length() < 12) {
            errors.add("DE61 data too short: expected at least 12 digits, got " + de61Data.length());
            return new RuPayDE61Data(de61Data, subfieldValues, errors, warnings);
        }
        
        try {
            // Parse each subfield by position (each is 1 digit)
            for (RuPayDE61Subfields subfield : RuPayDE61Subfields.values()) {
                if (subfield.getPosition() < de61Data.length()) {
                    String value = String.valueOf(de61Data.charAt(subfield.getPosition()));
                    subfieldValues.put(subfield, value);
                    
                    // Validate subfield value
                    validateSubfieldValue(subfield, value, errors, warnings);
                } else {
                    warnings.add("Subfield " + subfield.getDescription() + " not present in data");
                }
            }
            
            // Additional business validations
            validateBusinessRules(subfieldValues, errors, warnings);
            
        } catch (Exception e) {
            logger.error("Error parsing DE61 data: {}", de61Data, e);
            errors.add("Failed to parse DE61 data: " + e.getMessage());
        }
        
        return new RuPayDE61Data(de61Data, subfieldValues, errors, warnings);
    }
    
    /**
     * Validate individual subfield value
     */
    private static void validateSubfieldValue(RuPayDE61Subfields subfield, String value, 
                                            List<String> errors, List<String> warnings) {
        
        if (!value.matches("\\d")) {
            errors.add("Invalid " + subfield.getDescription() + ": must be a single digit, got '" + value + "'");
            return;
        }
        
        switch (subfield) {
            case CARD_DATA_INPUT_CAPABILITY:
                validateCardDataInputCapability(value, errors, warnings);
                break;
                
            case CARDHOLDER_AUTH_CAPABILITY:
                validateCardholderAuthCapability(value, errors, warnings);
                break;
                
            case CARD_CAPTURE_CAPABILITY:
                validateCardCaptureCapability(value, errors, warnings);
                break;
                
            case TERMINAL_OPERATING_ENVIRONMENT:
                validateTerminalOperatingEnvironment(value, errors, warnings);
                break;
        }
    }
    
    /**
     * Validate Card Data Input Capability
     */
    private static void validateCardDataInputCapability(String value, List<String> errors, List<String> warnings) {
        CardDataInputCapability capability = CardDataInputCapability.fromValue(value);
        if (capability == CardDataInputCapability.UNKNOWN) {
            warnings.add("Unknown Card Data Input Capability: " + value);
        }
        
        // Business rule validations
        if ("0".equals(value)) {
            warnings.add("Card Data Input Capability is 'Unknown' - may indicate data quality issue");
        }
    }
    
    /**
     * Validate Cardholder Authentication Capability
     */
    private static void validateCardholderAuthCapability(String value, List<String> errors, List<String> warnings) {
        CardholderAuthCapability capability = CardholderAuthCapability.fromValue(value);
        if (capability == CardholderAuthCapability.UNKNOWN) {
            warnings.add("Unknown Cardholder Authentication Capability: " + value);
        }
    }
    
    /**
     * Validate Card Capture Capability
     */
    private static void validateCardCaptureCapability(String value, List<String> errors, List<String> warnings) {
        CardCaptureCapability capability = CardCaptureCapability.fromValue(value);
        if (capability == CardCaptureCapability.UNKNOWN) {
            warnings.add("Unknown Card Capture Capability: " + value);
        }
    }
    
    /**
     * Validate Terminal Operating Environment
     */
    private static void validateTerminalOperatingEnvironment(String value, List<String> errors, List<String> warnings) {
        TerminalOperatingEnvironment env = TerminalOperatingEnvironment.fromValue(value);
        if (env == TerminalOperatingEnvironment.UNKNOWN) {
            warnings.add("Unknown Terminal Operating Environment: " + value);
        }
        
        // Business rule validations
        if ("6".equals(value)) {
            warnings.add("Terminal Operating Environment is 'No terminal used' - verify transaction type");
        }
    }
    
    /**
     * Validate business rules across subfields
     */
    private static void validateBusinessRules(Map<RuPayDE61Subfields, String> subfieldValues, 
                                            List<String> errors, List<String> warnings) {
        
        // Rule 1: If card data input capability is 0 (Unknown), other fields should be validated
        String cardDataInput = subfieldValues.get(RuPayDE61Subfields.CARD_DATA_INPUT_CAPABILITY);
        if ("0".equals(cardDataInput)) {
            warnings.add("Card Data Input Capability is 'Unknown' - verify transaction data quality");
        }
        
        // Rule 2: If terminal operating environment is 6 (No terminal used), validate other fields
        String terminalEnv = subfieldValues.get(RuPayDE61Subfields.TERMINAL_OPERATING_ENVIRONMENT);
        if ("6".equals(terminalEnv)) {
            warnings.add("Terminal Operating Environment is 'No terminal used' - verify for e-commerce transactions");
        }
        
        // Rule 3: Check for suspicious combinations
        String cardDataInputCap = subfieldValues.get(RuPayDE61Subfields.CARD_DATA_INPUT_CAPABILITY);
        String cardholderAuth = subfieldValues.get(RuPayDE61Subfields.CARDHOLDER_AUTH_CAPABILITY);
        
        if ("7".equals(cardDataInputCap) && "0".equals(cardholderAuth)) {
            warnings.add("Contactless transaction with unknown authentication method");
        }
        
        // Rule 4: Validate M-POS transactions
        if ("7".equals(terminalEnv) || "8".equals(terminalEnv)) {
            logger.info("M-POS transaction detected");
            // Additional M-POS specific validations can be added here
        }
    }
    
    /**
     * Create DE61 data from subfield values
     */
    public static String createDE61(Map<RuPayDE61Subfields, String> subfieldValues) {
        StringBuilder de61 = new StringBuilder();
        
        // Initialize with zeros for all 12 positions
        for (int i = 0; i < 12; i++) {
            de61.append("0");
        }
        
        // Set values for provided subfields
        for (Map.Entry<RuPayDE61Subfields, String> entry : subfieldValues.entrySet()) {
            RuPayDE61Subfields subfield = entry.getKey();
            String value = entry.getValue();
            
            if (value != null && value.length() == 1 && value.matches("\\d")) {
                de61.setCharAt(subfield.getPosition(), value.charAt(0));
            }
        }
        
        return de61.toString();
    }
