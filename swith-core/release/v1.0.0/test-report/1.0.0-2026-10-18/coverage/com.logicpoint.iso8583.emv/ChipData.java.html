package com.logicpoint.iso8583.emv;


import java.util.*;

/**
 * Container for parsed EMV chip data.
 * 
 * Author: @logicpoint  
 * Date: 2025-07-11
 * Version: 1.0.0
 * Description: This class is used to parse the EMV chip data.
 * 
 * 
 */
public class ChipData {
    private final Map<String, String> tags;
    private final Map<String, String> tagDescriptions;

    public ChipData() {
        this.tags = new HashMap<>();
        this.tagDescriptions = initializeTagDescriptions();
    }

    public ChipData(Map<String, String> tags) {
        this.tags = new HashMap<>(tags);
        this.tagDescriptions = initializeTagDescriptions();
    }

    /**
     * Gets value for a specific tag.
     */
    public String getTagValue(String tag) {
        return tags.get(tag);
    }

    public void removeTag(String tag) {
        tags.remove(tag);
    }
    public void removeAllTags() {
        tags.clear();
    }
    /**
     * Sets value for a specific tag.
     */
    public void setTagValue(String tag, String value) {
        tags.put(tag, value);
    }
    /**
     * Checks if tag exists.
     */
    public boolean hasTag(String tag) {
        return tags.containsKey(tag);
    }

    /**
     * Gets all tags.
     */
    public Set<String> getAllTags() {
        return new HashSet<>(tags.keySet());
    }

    /**
     * Builds ICC data.
     */
    public String buildIccData() {
        return new ChipDataParser().packTlv(tags);
    }

    /**
     * Gets tag description.
     */
    public String getTagDescription(String tag) {
        return tagDescriptions.getOrDefault(tag, "Unknown Tag");
    }

    /**
     * Checks if chip data is empty.
     */
    public boolean isEmpty() {
        return tags.isEmpty();
    }

    /**
     * Gets size of chip data.
     */
    public int size() {
        return tags.size();
    }

    /**
     * Gets all tag-value pairs.
     */
    public Map<String, String> getAllTagsAndValues() {
        return new HashMap<>(tags);
    }

    /**
     * Gets formatted chip data for logging.
     */
    public String getFormattedData() {
        if (tags.isEmpty()) {
            return "No EMV data";
        }

        StringBuilder sb = new StringBuilder();
        sb.append("EMV Data:\n");
        
        for (Map.Entry<String, String> entry : tags.entrySet()) {
            String tag = entry.getKey();
            String value = entry.getValue();
            String description = getTagDescription(tag);
            
            sb.append(String.format("  %s (%s): %s\n", tag, description, value));
        }
        
        return sb.toString();
    }

    /**
     * Initializes tag descriptions.
     */
    private Map<String, String> initializeTagDescriptions() {
        Map<String, String> descriptions = new HashMap<>();
        
        // Amount related
        descriptions.put("9F02", "Amount, Authorized (Numeric)");
        descriptions.put("9F03", "Amount, Other (Numeric)");
        descriptions.put("9F04", "Amount, Other (Binary)");
        
        // Country and currency
        descriptions.put("9F1A", "Terminal Country Code");
        descriptions.put("5F2A", "Transaction Currency Code");
        
        // Application data
        descriptions.put("82", "Application Interchange Profile");
        descriptions.put("9F36", "Application Transaction Counter");
        descriptions.put("9F37", "Unpredictable Number");
        descriptions.put("9F10", "Issuer Application Data");
        
        // Cryptogram data
        descriptions.put("9F26", "Application Cryptogram");
        descriptions.put("9F27", "Cryptogram Information Data");
        
        // Terminal data
        descriptions.put("9F33", "Terminal Capabilities");
        descriptions.put("9F34", "Cardholder Verification Method Results");
        descriptions.put("9F35", "Terminal Type");
        descriptions.put("9F1E", "Interface Device Serial Number");
        
        // Application identifiers
        descriptions.put("84", "Dedicated File Name");
        descriptions.put("9F08", "Application Version Number");
        descriptions.put("9F09", "Application Version Number");
        
        // Merchant data
        descriptions.put("9F15", "Merchant Category Code");
        descriptions.put("9F16", "Merchant Identifier");
        descriptions.put("9F4E", "Merchant Name and Location");
        
        // Transaction data
        descriptions.put("9F21", "Transaction Time");
        descriptions.put("9F20", "Transaction Date");
        descriptions.put("9F41", "Transaction Sequence Counter");
        
        // Application identifiers
        descriptions.put("9F42", "Application Identifier (Terminal)");
        descriptions.put("9F44", "Application Identifier (ICC)");
        
        // Public key data
        descriptions.put("9F46", "ICC Public Key Certificate");
        descriptions.put("9F47", "ICC Public Key Exponent");
        descriptions.put("9F48", "ICC Public Key Remainder");
        
        // Authentication data
        descriptions.put("9F49", "Dynamic Data Authentication Data Object List");
        descriptions.put("9F4A", "Static Data Authentication Tag List");
        descriptions.put("9F4B", "Signed Dynamic Application Data");
        
        // Log data
        descriptions.put("9F4D", "Log Entry");
        descriptions.put("9F4F", "Log Format");
        
        // ICC dynamic data
        descriptions.put("9F4C", "ICC Dynamic Number");
        
        // Cumulative amounts
        descriptions.put("9F54", "Cumulative Total Transaction Amount");
        descriptions.put("9F55", "Cumulative Total Transaction Amount Upper Limit");
        
        // Issuer data
        descriptions.put("9F56", "Issuer Counter");
        descriptions.put("9F57", "Issuer Action Code - Default");
        descriptions.put("9F58", "Issuer Action Code - Denial");
        descriptions.put("9F59", "Issuer Action Code - Online");
        
        // Account data
        descriptions.put("9F5A", "Application Primary Account Number");
        descriptions.put("9F5B", "Track 2 Equivalent Data");
        
        return descriptions;
    }

    @Override
    public String toString() {
        return getFormattedData();
    }
}
