package com.logicpoint.iso8583.emv;

import com.logicpoint.iso8583.util.ISOUtil;

import java.io.PrintStream;
import java.util.Objects;

/**
 * Cardholder Verification Method Results (CVM) parser.
 * Parses the 9F34 tag which contains information about the CVM performed during the transaction.
 * 
 * The CVM Results field is 3 bytes long and contains:
 * - Byte 1: CVM Code (low 6 bits indicate the CVM performed)
 * - Byte 2: Condition Code
 * - Byte 3: CVM Result
 * 
 * Based on EMV Book 3, Section A1.4.2
 * 
 * @author Logic Point Systems
 * @version 1.0.0
 */
public final class CardholderVerificationMethodResults {

    
    private final byte[] cvmResults;
    
    // CVM Code constants (Byte 1)
    public static final int CVM_CODE_FAILED = 0x00;
    public static final int CVM_CODE_PLAINTEXT_PIN_VERIFICATION_PERFORMED = 0x01;
    public static final int CVM_CODE_ENCRYPTED_PIN_VERIFICATION_PERFORMED = 0x02;
    public static final int CVM_CODE_PLAINTEXT_PIN_VERIFICATION_PERFORMED_ICC = 0x03;
    public static final int CVM_CODE_ENCRYPTED_PIN_VERIFICATION_PERFORMED_ICC = 0x04;
    public static final int CVM_CODE_SIGNATURE_VERIFICATION_PERFORMED = 0x05;
    public static final int CVM_CODE_NO_CVM_REQUIRED = 0x06;
    public static final int CVM_CODE_FAILED_CVM_REQUIRED = 0x07;
    public static final int CVM_CODE_SIGNATURE_VERIFICATION_PERFORMED_PAPER = 0x08;
    public static final int CVM_CODE_NO_CVM_REQUIRED_CONTACTLESS = 0x09;
    public static final int CVM_CODE_BIOMETRIC_VERIFICATION_PERFORMED = 0x0A;
    public static final int CVM_CODE_BIOMETRIC_AND_PIN_VERIFICATION_PERFORMED = 0x0B;
    public static final int CVM_CODE_BIOMETRIC_AND_SIGNATURE_VERIFICATION_PERFORMED = 0x0C;
    public static final int CVM_CODE_OTHER_CVM_PERFORMED = 0x1E;
    public static final int CVM_CODE_NO_CVM_PERFORMED = 0x1F;

    
    
    // CVM Results constants (Byte 3)
    public static final int CVM_RESULT_UNKNOWN = 0x00;
    public static final int CVM_RESULT_FAILED = 0x01;
    public static final int CVM_RESULT_SUCCESSFUL = 0x02;
    public static final int CVM_RESULT_NOT_PERFORMED = 0x03;
    public static final int CVM_RESULT_NOT_SUPPORTED = 0x04;
    public static final int CVM_RESULT_NOT_APPLICABLE = 0x05;

    public CardholderVerificationMethodResults(byte[] cvmResults) {
        Objects.requireNonNull(cvmResults);
        if (cvmResults.length != 3) {
            throw new IllegalArgumentException("CVM Results length must be 3 bytes, got: " + cvmResults.length);
        }
        this.cvmResults = cvmResults.clone();
    }

    public CardholderVerificationMethodResults(String hexCVM) {
        this(ISOUtil.hex2byte(hexCVM));
    }

    /**
     * Gets the CVM Code (Byte 1)
     * @return CVM Code as integer
     */
    public int getCvmCode() {
        return cvmResults[0] & 0x3F;
    }

    /**
     * Gets the Condition Code (Byte 2)
     * @return Condition Code as integer
     */
    public int getConditionCode() {
        return cvmResults[1] & 0xFF;
    }

    /**
     * Gets the CVM Results (Byte 3)
     * @return CVM Results as integer
     */
    public int getCvmResults() {
        return cvmResults[2] & 0xFF;
    }

    /**
     * Legacy accessor retained for compatibility.
     * 9F34 byte 2 is the condition code, not a reserved byte.
     *
     * @return Condition Code as integer
     */
    @Deprecated
    public int getReserved() {
        return getConditionCode();
    }

    /**
     * Checks if CVM was successful
     * @return true if CVM was successful
     */
    public boolean isCvmSuccessful() {
        return getCvmResults() == CVM_RESULT_SUCCESSFUL;
    }

    /**
     * Checks if CVM failed
     * @return true if CVM failed
     */
    public boolean isCvmFailed() {
        return getCvmResults() == CVM_RESULT_FAILED;
    }

    /**
     * Checks if CVM was not performed
     * @return true if CVM was not performed
     */
    public boolean isCvmNotPerformed() {
        return getCvmResults() == CVM_RESULT_NOT_PERFORMED;
    }

    /**
     * Checks if CVM is not supported
     * @return true if CVM is not supported
     */
    public boolean isCvmNotSupported() {
        return getCvmResults() == CVM_RESULT_NOT_SUPPORTED;
    }

    /**
     * Checks if CVM is not applicable
     * @return true if CVM is not applicable
     */
    public boolean isCvmNotApplicable() {
        return getCvmResults() == CVM_RESULT_NOT_APPLICABLE;
    }

    /**
     * Checks if the terminal reported an unknown CVM result.
     * Some terminals send 00 for online PIN flows, so callers may choose to
     * treat this as non-fatal in that specific scenario.
     *
     * @return true if the CVM result is unknown
     */
    public boolean isCvmResultUnknown() {
        return getCvmResults() == CVM_RESULT_UNKNOWN;
    }

    /**
     * Checks if PIN verification was performed
     * @return true if PIN verification was performed
     */
    public boolean isPinVerificationPerformed() {
        int code = getCvmCode();
        return code == CVM_CODE_PLAINTEXT_PIN_VERIFICATION_PERFORMED ||
               code == CVM_CODE_ENCRYPTED_PIN_VERIFICATION_PERFORMED ||
               code == CVM_CODE_PLAINTEXT_PIN_VERIFICATION_PERFORMED_ICC ||
               code == CVM_CODE_ENCRYPTED_PIN_VERIFICATION_PERFORMED_ICC ||
               code == CVM_CODE_BIOMETRIC_AND_PIN_VERIFICATION_PERFORMED;
    }

    /**
     * Checks if signature verification was performed
     * @return true if signature verification was performed
     */
    public boolean isSignatureVerificationPerformed() {
        int code = getCvmCode();
        return code == CVM_CODE_SIGNATURE_VERIFICATION_PERFORMED ||
               code == CVM_CODE_SIGNATURE_VERIFICATION_PERFORMED_PAPER ||
               code == CVM_CODE_BIOMETRIC_AND_SIGNATURE_VERIFICATION_PERFORMED;
    }

    /**
     * Checks if biometric verification was performed
     * @return true if biometric verification was performed
     */
    public boolean isBiometricVerificationPerformed() {
        int code = getCvmCode();
        return code == CVM_CODE_BIOMETRIC_VERIFICATION_PERFORMED ||
               code == CVM_CODE_BIOMETRIC_AND_PIN_VERIFICATION_PERFORMED ||
               code == CVM_CODE_BIOMETRIC_AND_SIGNATURE_VERIFICATION_PERFORMED;
    }

    /**
     * Checks if no CVM was required
     * @return true if no CVM was required
     */
    public boolean isNoCvmRequired() {
        int code = getCvmCode();
        return code == CVM_CODE_NO_CVM_REQUIRED ||
               code == CVM_CODE_NO_CVM_REQUIRED_CONTACTLESS;
    }

    /**
     * Checks if this is a contactless transaction with no CVM
     * @return true if contactless transaction with no CVM
     */
    public boolean isContactlessNoCvm() {
        return getCvmCode() == CVM_CODE_NO_CVM_REQUIRED_CONTACTLESS;
    }

    /**
     * Checks if the terminal reported that no CVM was performed.
     *
     * @return true if no CVM was performed
     */
    public boolean isNoCvmPerformed() {
        return getCvmCode() == CVM_CODE_NO_CVM_PERFORMED;
    }

    /**
     * Checks if CVM failed but was required
     * @return true if CVM failed but was required
     */
    public boolean isCvmFailedButRequired() {
        return getCvmCode() == CVM_CODE_FAILED_CVM_REQUIRED;
    }

    /**
     * Checks if offline PIN verification was performed
     * @return true if offline PIN verification was performed
     */
    public boolean isOfflinePinVerificationPerformed() {
        int code = getCvmCode();
        return code == CVM_CODE_PLAINTEXT_PIN_VERIFICATION_PERFORMED_ICC ||
               code == CVM_CODE_ENCRYPTED_PIN_VERIFICATION_PERFORMED_ICC;
    }

    /**
     * Checks if online PIN verification was performed
     * @return true if online PIN verification was performed
     */
    public boolean isOnlinePinVerificationPerformed() {
        int code = getCvmCode();
        return code == CVM_CODE_PLAINTEXT_PIN_VERIFICATION_PERFORMED ||
               code == CVM_CODE_ENCRYPTED_PIN_VERIFICATION_PERFORMED;
    }

    /**
     * Gets a human-readable description of the CVM Code
     * @return Description of the CVM Code
     */
    public String getCvmCodeDescription() {
        return switch (getCvmCode()) {
            case CVM_CODE_FAILED -> "CVM Failed";
            case CVM_CODE_PLAINTEXT_PIN_VERIFICATION_PERFORMED -> "Plaintext PIN Verification Performed";
            case CVM_CODE_ENCRYPTED_PIN_VERIFICATION_PERFORMED -> "Enciphered PIN verified online";
            case CVM_CODE_PLAINTEXT_PIN_VERIFICATION_PERFORMED_ICC -> "Plaintext PIN Verification Performed (ICC)";
            case CVM_CODE_ENCRYPTED_PIN_VERIFICATION_PERFORMED_ICC -> "Enciphered PIN verification performed by ICC";
            case CVM_CODE_SIGNATURE_VERIFICATION_PERFORMED -> "Signature Verification Performed";
            case CVM_CODE_NO_CVM_REQUIRED -> "No CVM Required";
            case CVM_CODE_FAILED_CVM_REQUIRED -> "Failed - CVM Required";
            case CVM_CODE_SIGNATURE_VERIFICATION_PERFORMED_PAPER -> "Signature Verification Performed (Paper)";
            case CVM_CODE_NO_CVM_REQUIRED_CONTACTLESS -> "No CVM Required (Contactless)";
            case CVM_CODE_BIOMETRIC_VERIFICATION_PERFORMED -> "Biometric Verification Performed";
            case CVM_CODE_BIOMETRIC_AND_PIN_VERIFICATION_PERFORMED -> "Biometric and PIN Verification Performed";
            case CVM_CODE_BIOMETRIC_AND_SIGNATURE_VERIFICATION_PERFORMED -> "Biometric and Signature Verification Performed";
            case CVM_CODE_OTHER_CVM_PERFORMED -> "Other CVM Performed";
            case CVM_CODE_NO_CVM_PERFORMED -> "No CVM Performed";
            default -> "Unknown CVM Code: " + getCvmCode();
        };
    }

    /**
     * Gets a human-readable description of the CVM Results
     * @return Description of the CVM Results
     */
    public String getCvmResultsDescription() {
        return switch (getCvmResults()) {
            case CVM_RESULT_UNKNOWN -> "Unknown";
            case CVM_RESULT_FAILED -> "Failed";
            case CVM_RESULT_SUCCESSFUL -> "Successful";
            case CVM_RESULT_NOT_PERFORMED -> "Not Performed";
            case CVM_RESULT_NOT_SUPPORTED -> "Not Supported";
            case CVM_RESULT_NOT_APPLICABLE -> "Not Applicable";
            default -> "Unknown CVM Result: " + getCvmResults();
        };
    }

    /**
     * Gets a human-readable description of the condition code (Byte 2).
     *
     * @return Condition Code description
     */
    public String getConditionCodeDescription() {
        return switch (getConditionCode()) {
            case 0x00 -> "Always";
            case 0x01 -> "If unattended cash";
            case 0x02 -> "If not unattended cash and not manual cash and not purchase with cashback";
            case 0x03 -> "If terminal supports the CVM";
            case 0x04 -> "If manual cash";
            case 0x05 -> "If purchase with cashback";
            case 0x06 -> "If transaction is in the application currency and is under X value";
            case 0x07 -> "If transaction is in the application currency and is over X value";
            case 0x08 -> "If transaction is in the application currency and is under Y value";
            case 0x09 -> "If transaction is in the application currency and is over Y value";
            default -> "Condition Code: " + String.format("%02X", getConditionCode());
        };
    }

    /**
     * Validates the CVM Results based on business rules
     * @return true if CVM is valid for transaction processing
     */
    public boolean isValidForTransaction() {
        // If CVM failed but was required, transaction should be declined
        if (isCvmFailedButRequired()) {
            return false;
        }
        
        // If CVM was performed and failed, transaction should be declined
        if (isCvmFailed()) {
            return false;
        }

        // Many terminals send 00 for online PIN flows while PIN is verified
        // separately by the issuer/HSM. Do not fail those transactions here.
        if (isOnlinePinVerificationPerformed() && isCvmResultUnknown()) {
            return true;
        }

        // Offline PIN is verified by the ICC/terminal and is represented in
        // 9F34/TVR. The issuer must not re-verify it using DE52.
        if (isOfflinePinVerificationPerformed() && isCvmSuccessful()) {
            return true;
        }

        // RuPay contactless certification may send 9F34=1F0300
        // (No CVM Performed / terminal supports CVM / Unknown result).
        // Treat it as eligible no-CVM data; the contactless amount limit is
        // enforced later by CvmValidationService.
        if (isNoCvmPerformed() && (isCvmResultUnknown() || isCvmNotPerformed())) {
            return true;
        }
        
        // If CVM is successful or not required, transaction can proceed
        return isCvmSuccessful() || isNoCvmRequired() || isCvmNotApplicable();
    }

    /**
     * Gets the raw CVM Results as byte array
     * @return CVM Results as byte array
     */
    public byte[] getRawCvmResults() {
        return cvmResults.clone();
    }

    /**
     * Gets the CVM Results as hex string
     * @return CVM Results as hex string
     */
    public String getHexCvmResults() {
        return ISOUtil.hexString(cvmResults);
    }

    /**
     * Dumps CVM Results information to a PrintStream
     * @param p PrintStream to dump to
     * @param indent Indentation string
     */
    public void dump(PrintStream p, String indent) {
        String inner = indent + "  ";
        p.printf("%s<cardholder-verification-method-results value='%s'>%n", indent, getHexCvmResults());
        p.printf("%sByte 1 (CVM Code): %s (%s)%n", inner, 
                String.format("%02X", getCvmCode()), getCvmCodeDescription());
        p.printf("%sByte 2 (Condition Code): %s (%s)%n", inner,
                String.format("%02X", getConditionCode()), getConditionCodeDescription());
        p.printf("%sByte 3 (CVM Result): %s (%s)%n", inner,
                String.format("%02X", getCvmResults()), getCvmResultsDescription());
        p.printf("%sValid for Transaction: %s%n", inner, isValidForTransaction());
        p.printf("%s</cardholder-verification-method-results>%n", indent);
    }

    @Override
    public String toString() {
        return String.format("CVMResults[Code=%s, Result=%s, Valid=%s]", 
                getCvmCodeDescription(), getCvmResultsDescription(), isValidForTransaction());
    }

    @Override
    public boolean equals(Object obj) {
        if (this == obj) return true;
        if (obj == null || getClass() != obj.getClass()) return false;
        CardholderVerificationMethodResults that = (CardholderVerificationMethodResults) obj;
        return java.util.Arrays.equals(cvmResults, that.cvmResults);
    }

    @Override
    public int hashCode() {
        return java.util.Arrays.hashCode(cvmResults);
    }

    /**
     * Main method for testing CVM parsing
     */
    public static void main(String[] args) {
        // Test with different CVM scenarios
        testCvmParsing("000000", "No CVM Required - Successful");
        testCvmParsing("010200", "Plaintext PIN - Successful");
        testCvmParsing("050100", "Signature - Failed");
        testCvmParsing("090000", "Contactless No CVM");
        testCvmParsing("070100", "Failed CVM Required");
        testCvmParsing("420300", "Enciphered PIN verified online");
        testCvmParsing("1F0300", "Enciphered PIN verified online");
    }

    private static void testCvmParsing(String hexCvm, String description) {
        System.out.println("\n=== " + description + " ===");
        try {
            CardholderVerificationMethodResults cvm = new CardholderVerificationMethodResults(hexCvm);
            cvm.dump(System.out, "");
        } catch (Exception e) {
            System.err.println("Error parsing CVM: " + e.getMessage());
        }
    }
}
