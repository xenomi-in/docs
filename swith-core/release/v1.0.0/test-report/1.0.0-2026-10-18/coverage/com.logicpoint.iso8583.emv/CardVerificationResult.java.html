package com.logicpoint.iso8583.emv;

import com.logicpoint.iso8583.util.ISOUtil;
import com.logicpoint.iso8583.emv.IssuerApplicationData.Format;
import java.io.PrintStream;
import java.util.Objects;

/**
 * Card Verification Result (CVR) parser.
 * Parses CVR from Issuer Application Data (IAD, tag 9F10).
 * 
 * CVR length varies by IAD format:
 * - EMV Format A: 5 bytes (10 hex chars)
 * - VIS format: 4 bytes (8 hex chars) or 5 bytes (10 hex chars)
 * - M/CHIP: 6 bytes (12 hex chars)
 * - RuPay EMV Format A: 8 bytes (16 hex chars)
 * 
 * Based on EMV 4.3 Book 3 specifications
 * 
 * @author Logic Point Systems
 * @version 1.0.0
 */
public final class CardVerificationResult {

    private final byte[] cvr;
    private final Format iadFormat;
    private final int cvrLength;

    /**
     * Constructor with hex string CVR
     * 
     * @param hexCVR Hexadecimal representation of CVR
     * @param iadFormat IAD format to determine CVR structure
     */
    public CardVerificationResult(String hexCVR, Format iadFormat) {
        Objects.requireNonNull(hexCVR, "CVR data cannot be null");
        Objects.requireNonNull(iadFormat, "IAD format cannot be null");
        
        this.iadFormat = iadFormat;
        this.cvr = ISOUtil.hex2byte(hexCVR);
        this.cvrLength = cvr.length;
        
        // Validate CVR length based on format
        validateCvrLength();
    }

    /**
     * Constructor with byte array CVR
     */
    public CardVerificationResult(byte[] cvr, Format iadFormat) {
        Objects.requireNonNull(cvr, "CVR data cannot be null");
        Objects.requireNonNull(iadFormat, "IAD format cannot be null");
        
        this.iadFormat = iadFormat;
        this.cvr = cvr.clone();
        this.cvrLength = cvr.length;
        
        validateCvrLength();
    }

    private void validateCvrLength() {
        switch (iadFormat) {
            case EMV_FORMAT_A:
            case RUPAY_EMV_FORMAT_A:
                if (cvrLength != 5 && cvrLength != 8) {
                    throw new IllegalArgumentException(
                        String.format("CVR length must be 5 or 8 bytes for %s format, got: %d", 
                                    iadFormat, cvrLength));
                }
                break;
            case VIS:
                if (cvrLength != 4 && cvrLength != 5) {
                    throw new IllegalArgumentException(
                        String.format("CVR length must be 4 or 5 bytes for %s format, got: %d", 
                                    iadFormat, cvrLength));
                }
                break;
            case M_CHIP:
                if (cvrLength != 6) {
                    throw new IllegalArgumentException(
                        String.format("CVR length must be 6 bytes for %s format, got: %d", 
                                    iadFormat, cvrLength));
                }
                break;
            default:
                // Accept any length for unknown/other formats
                break;
        }
    }

    private boolean isBitOn(byte value, int position) {
        return ((value >> (position - 1)) & 1) == 1;
    }

    // ========== Byte 1 Checks (Common across formats) ==========

    /**
     * Gets the first GENERATE AC cryptogram type (lower nibble of byte 1)
     * 0x1 = AAC, 0x2 = TC, 0x5 = ARQC
     */
    public int getFirstGenerateAcCryptogramType() {
        if (cvrLength < 1) return 0;
        return cvr[0] & 0x0F; // Lower nibble
    }

    /**
     * Gets the second GENERATE AC cryptogram type (upper nibble of byte 1)
     */
    public int getSecondGenerateAcCryptogramType() {
        if (cvrLength < 1) return 0;
        return (cvr[0] >> 4) & 0x0F; // Upper nibble
    }

    /**
     * Checks if CDA (Combined DDA/Application Cryptogram Generation) was performed
     */
    public boolean cdaPerformed() {
        if (cvrLength < 2) return false;
        return isBitOn(cvr[1], 8);
    }

    /**
     * Checks if offline DDA (Dynamic Data Authentication) was performed
     */
    public boolean offlineDdaPerformed() {
        if (cvrLength < 2) return false;
        return isBitOn(cvr[1], 7);
    }

    /**
     * Checks if issuer authentication was not performed
     */
    public boolean issuerAuthenticationNotPerformed() {
        if (cvrLength < 2) return false;
        return isBitOn(cvr[1], 6);
    }

    /**
     * Checks if issuer authentication failed
     */
    public boolean issuerAuthenticationFailed() {
        if (cvrLength < 2) return false;
        return isBitOn(cvr[1], 5);
    }

    // ========== Byte 2 Checks ==========

    /**
     * Gets the PIN Try Counter (lower nibble of byte 2)
     * Indicates how many PIN tries remain (0x0-0xF)
     */
    public int getPinTryCounter() {
        if (cvrLength < 2) return -1;
        return (cvr[1] >> 4) & 0x0F;
    }

    /**
     * Checks if offline PIN verification was performed
     */
    public boolean offlinePinVerificationPerformed() {
        if (cvrLength < 2) return false;
        return isBitOn(cvr[1], 4);
    }

    /**
     * Checks if offline PIN verification was performed but not successful
     */
    public boolean offlinePinVerificationFailed() {
        if (cvrLength < 2) return false;
        return isBitOn(cvr[1], 3);
    }

    /**
     * Checks if PIN try limit was exceeded
     */
    public boolean pinTryLimitExceeded() {
        if (cvrLength < 2) return false;
        return isBitOn(cvr[1], 2);
    }

    /**
     * Checks if last online transaction was not completed
     */
    public boolean lastOnlineTransactionNotCompleted() {
        if (cvrLength < 2) return false;
        return isBitOn(cvr[1], 1);
    }

    // ========== Byte 3 Checks ==========

    /**
     * Checks if lower offline transaction count limit was exceeded
     */
    public boolean lowerOfflineTransactionCountLimitExceeded() {
        if (cvrLength < 3) return false;
        return isBitOn(cvr[2], 8);
    }

    /**
     * Checks if upper offline transaction count limit was exceeded
     */
    public boolean upperOfflineTransactionCountLimitExceeded() {
        if (cvrLength < 3) return false;
        return isBitOn(cvr[2], 7);
    }

    /**
     * Checks if lower cumulative offline amount limit was exceeded
     */
    public boolean lowerCumulativeOfflineAmountLimitExceeded() {
        if (cvrLength < 3) return false;
        return isBitOn(cvr[2], 6);
    }

    /**
     * Checks if upper cumulative offline amount limit was exceeded
     */
    public boolean upperCumulativeOfflineAmountLimitExceeded() {
        if (cvrLength < 3) return false;
        return isBitOn(cvr[2], 5);
    }

    // ========== Byte 4 Checks ==========

    /**
     * Gets number of successfully processed issuer script commands
     * (bits 8-5 of byte 4)
     */
    public int getIssuerScriptCommandsProcessed() {
        if (cvrLength < 4) return 0;
        return (cvr[3] >> 4) & 0x0F;
    }

    /**
     * Checks if issuer script processing failed
     */
    public boolean issuerScriptProcessingFailed() {
        if (cvrLength < 4) return false;
        return isBitOn(cvr[3], 4);
    }

    /**
     * Checks if offline data authentication failed on previous transaction
     */
    public boolean offlineDataAuthFailedOnPreviousTransaction() {
        if (cvrLength < 4) return false;
        return isBitOn(cvr[3], 3);
    }

    /**
     * Checks if "Go Online on Next Transaction" flag was set
     */
    public boolean goOnlineOnNextTransactionSet() {
        if (cvrLength < 4) return false;
        return isBitOn(cvr[3], 2);
    }

    /**
     * Checks if unable to go online
     */
    public boolean unableToGoOnline() {
        if (cvrLength < 4) return false;
        return isBitOn(cvr[3], 1);
    }

    // ========== RuPay Specific Checks (Byte 6-8) ==========

    /**
     * Checks if this is a RuPay format CVR
     */
    public boolean isRuPayFormat() {
        return iadFormat == Format.RUPAY_EMV_FORMAT_A && cvrLength >= 8;
    }

    // Additional RuPay-specific bit checks can be added here for bytes 5-8

    // ========== Validation Methods ==========

    /**
     * Validates if CVR indicates transaction should proceed
     * @return true if transaction can proceed based on CVR
     */
    public boolean isValidForTransaction() {
        // Critical failures - must decline
        if (pinTryLimitExceeded()) {
            return false;
        }

        if (issuerAuthenticationFailed()) {
            return false;
        }

        // Offline PIN failed - should decline unless online auth is available
        if (offlinePinVerificationFailed() && offlinePinVerificationPerformed()) {
            return false;
        }

        // Offline limits exceeded - should force online authorization
        // These are warnings, not hard failures
        if (upperOfflineTransactionCountLimitExceeded() || 
            upperCumulativeOfflineAmountLimitExceeded()) {
            // Force online authorization but don't decline
        }

        // Issuer script failed - may require online authorization
        if (issuerScriptProcessingFailed()) {
            // May require online auth but not necessarily a decline
        }

        return true;
    }

    /**
     * Checks if CVR requires online authorization
     */
    public boolean requiresOnlineAuthorization() {
        return upperOfflineTransactionCountLimitExceeded() ||
               upperCumulativeOfflineAmountLimitExceeded() ||
               lowerOfflineTransactionCountLimitExceeded() ||
               lowerCumulativeOfflineAmountLimitExceeded() ||
               goOnlineOnNextTransactionSet() ||
               offlinePinVerificationFailed() ||
               issuerScriptProcessingFailed() ||
               lastOnlineTransactionNotCompleted() ||
               unableToGoOnline();
    }

    // ========== Utility Methods ==========

    /**
     * Gets the raw CVR as byte array
     */
    public byte[] getRawCvr() {
        return cvr.clone();
    }

    /**
     * Gets CVR as hex string
     */
    public String getHexCvr() {
        return ISOUtil.hexString(cvr);
    }

    /**
     * Gets IAD format
     */
    public Format getIadFormat() {
        return iadFormat;
    }

    /**
     * Gets CVR length in bytes
     */
    public int getCvrLength() {
        return cvrLength;
    }

    /**
     * Dumps CVR information to PrintStream
     */
    public void dump(PrintStream p, String indent) {
        String inner = indent + "  ";
        p.printf("%s<card-verification-result format='%s' value='%s'>%n", 
                indent, iadFormat, getHexCvr());
        
        p.printf("%sByte 1 - First GC: 0x%X, Second GC: 0x%X%n", 
                inner, getFirstGenerateAcCryptogramType(), getSecondGenerateAcCryptogramType());
        
        if (cvrLength >= 2) {
            p.printf("%sByte 2 - PIN Try Counter: %d, Offline PIN Performed: %s, Failed: %s%n",
                    inner, getPinTryCounter(), 
                    offlinePinVerificationPerformed(), offlinePinVerificationFailed());
        }
        
        if (cvrLength >= 3) {
            p.printf("%sByte 3 - Limits: Lower Count: %s, Upper Count: %s, Lower Amount: %s, Upper Amount: %s%n",
                    inner,
                    lowerOfflineTransactionCountLimitExceeded(),
                    upperOfflineTransactionCountLimitExceeded(),
                    lowerCumulativeOfflineAmountLimitExceeded(),
                    upperCumulativeOfflineAmountLimitExceeded());
        }
        
        p.printf("%sValid for Transaction: %s%n", inner, isValidForTransaction());
        p.printf("%sRequires Online Auth: %s%n", inner, requiresOnlineAuthorization());
        p.printf("%s</card-verification-result>%n", indent);
    }

    @Override
    public String toString() {
        return String.format("CVR[Format=%s, Value=%s, Valid=%s, RequiresOnline=%s]",
                iadFormat, getHexCvr(), isValidForTransaction(), requiresOnlineAuthorization());
    }
}
