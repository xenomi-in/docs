    // Pattern for hex string validation
    private static final Pattern HEX_PATTERN = Pattern.compile("^[0-9A-Fa-f]+$");

    /**
     * Parses EMV chip data from DE55 field.
     * 
     * @param emvData The EMV data string (hex format)
     * @return ChipData object containing parsed EMV tags
     */
    public ChipData parseChipData(String emvData) {
        if (emvData == null || emvData.trim().isEmpty()) {
            logger.warn("EMV data is null or empty");
            return new ChipData();
        }

        logger.debug("Parsing EMV data: {}", emvData);
        
        try {
            Map<String, String> tags = parseTLV(emvData);
            return new ChipData(tags);
        } catch (Exception e) {
            logger.error("Error parsing EMV data: {}", emvData, e);
            return new ChipData();
        }
    }

    /**
     * Parses TLV (Tag-Length-Value) format from hex string.
     * 
     * @param hexData The hex string containing TLV data
     * @return Map of tag to value
     */
    private Map<String, String> parseTLV(String hexData) {
        Map<String, String> tags = new HashMap<>();
        
        if (!isValidHex(hexData)) {
            logger.error("Invalid hex data: {}", hexData);
            return tags;
        }

        int offset = 0;
        while (offset < hexData.length()) {
            try {
                // Parse tag
                String tag = parseTag(hexData, offset);
                if (tag == null) break;
                
                offset += tag.length();
                
                // Parse length
                int length = parseLength(hexData, offset);
                offset += 2; // Length is 1 byte
                
                // Parse value
                if (offset + length * 2 <= hexData.length()) {
                    String value = hexData.substring(offset, offset + length * 2);
                    tags.put(tag, value);
                    offset += length * 2;
                } else {
                    logger.warn("Incomplete TLV data at offset {}", offset);
                    break;
                }
            } catch (Exception e) {
                logger.error("Error parsing TLV at offset {}: {}", offset, e.getMessage());
                break;
            }
        }

        return tags;
    }

    /**
     * Parses tag from hex string.
     */
    private String parseTag(String hexData, int offset) {
        if (offset >= hexData.length()) return null;
        
        // Handle 1-byte tags
        if (offset + 2 <= hexData.length()) {
            String tag = hexData.substring(offset, offset + 2);
            
            // Handle 2-byte tags (first byte has bit 5 set)
            if (isTwoByteTag(tag)) {
                if (offset + 4 <= hexData.length()) {
                    return hexData.substring(offset, offset + 4);
                }
            }
            
            return tag;
        }
        
        return null;
    }

    /**
     * Checks if tag is a 2-byte tag.
     */
    private boolean isTwoByteTag(String tag) {
        if (tag.length() != 2) return false;
        
        try {
            int tagValue = Integer.parseInt(tag, 16);
            return (tagValue & 0x1F) == 0x1F; // Bit 5 is set
        } catch (NumberFormatException e) {
            return false;
        }
    }

    /**
     * Parses length from hex string.
     */
    private int parseLength(String hexData, int offset) {
        if (offset + 2 > hexData.length()) {
            throw new IllegalArgumentException("Insufficient data for length field");
        }
        
        String lengthHex = hexData.substring(offset, offset + 2);
        return Integer.parseInt(lengthHex, 16);
    }

    /**
     * Validates if string is valid hex.
     */
    private boolean isValidHex(String hex) {
        return hex != null && HEX_PATTERN.matcher(hex).matches();
    }

    /**
     * Converts hex string to readable format.
     */
    public String hexToReadable(String hex) {
        if (hex == null || hex.isEmpty()) return "";
        
        StringBuilder result = new StringBuilder();
        for (int i = 0; i < hex.length(); i += 2) {
            if (i + 1 < hex.length()) {
                String hexPair = hex.substring(i, i + 2);
                try {
                    int decimal = Integer.parseInt(hexPair, 16);
                    if (decimal >= 32 && decimal <= 126) { // Printable ASCII
                        result.append((char) decimal);
                    } else {
                        result.append(".");
                    }
                } catch (NumberFormatException e) {
                    result.append(".");
                }
            }
        }
        return result.toString();
    }

    /**
     * Validates chip data integrity.
     */
    public boolean validateChipData(ChipData chipData) {
        if (chipData == null || chipData.isEmpty()) {
            return false;
        }

        // Check for required tags based on transaction type
        boolean hasApplicationCryptogram = chipData.hasTag(TAG_9F26);
        boolean hasCryptogramInfo = chipData.hasTag(TAG_9F27);
        boolean hasATC = chipData.hasTag(TAG_9F36);
        boolean hasUnpredictableNumber = chipData.hasTag(TAG_9F37);

        return hasApplicationCryptogram && hasCryptogramInfo && hasATC && hasUnpredictableNumber;
    }
    

    /**
     * Packs a map of tag-value pairs into a TLV (Tag-Length-Value) formatted string.
     * Each tag and value is encoded as: [TAG][LENGTH][VALUE]
     * - TAG: variable length (usually 2-4 hex chars)
     * - LENGTH: 2 bytes (hex) representing the length of VALUE in bytes
     * - VALUE: hex string
     *
     * Example: tag "9F02" with value "000000010000" (6 bytes) becomes "9F0206000000010000"
     *
     * @param tagValueMap Map of tag to value (hex string)
     * @return TLV formatted string
     */
    public String packTlv(Map<String, String> tagValueMap) {
        if (tagValueMap == null || tagValueMap.isEmpty()) {
            return "";
        }
        StringBuilder tlvBuilder = new StringBuilder();
        for (Map.Entry<String, String> entry : tagValueMap.entrySet()) {
            String tag = entry.getKey();
            String value = entry.getValue() != null ? entry.getValue() : "";
            int valueLength = value.length() / 2; // Each 2 hex chars = 1 byte

            // Tag: as is (variable length)
            tlvBuilder.append(tag);

            // Length: 2 bytes (4 hex chars), left padded with zeros if needed
            String lengthHex = String.format("%02X", valueLength);
            tlvBuilder.append(lengthHex);

            // Value: as is
            tlvBuilder.append(value);
        }
        return tlvBuilder.toString();
    }

     */
    public CardholderVerificationMethodResults extractCvmResults(ChipData chipData) {
        if (chipData == null || chipData.isEmpty()) {
            logger.warn("Chip data is null or empty, cannot extract CVM results");
            return null;
        }
        
        String cvmHex = chipData.getTagValue(TAG_9F34);
        if (cvmHex == null || cvmHex.isEmpty()) {
            logger.debug("CVM Results (9F34) not found in chip data");
            return null;
        }
        
        try {
            return new CardholderVerificationMethodResults(cvmHex);
        } catch (IllegalArgumentException e) {
            logger.error("Invalid CVM Results format in tag 9F34: {}", cvmHex, e);
            return null;
        }
    }
    
    /**
     * Validates CVM Results from chip data
     * 
     * @param chipData The chip data containing CVM information
     * @return true if CVM Results are valid and suitable for transaction processing
     */
    public boolean validateCvmResults(ChipData chipData) {
        CardholderVerificationMethodResults cvmResults = extractCvmResults(chipData);
        if (cvmResults == null) {
            return false;
        }
        
        return cvmResults.isValidForTransaction();
    }
    
    /**
     * Gets CVM information as a formatted string for logging/debugging
     * 
     * @param chipData The chip data containing CVM information
     * @return Formatted CVM information string
     */
    public String getCvmInfo(ChipData chipData) {
        CardholderVerificationMethodResults cvmResults = extractCvmResults(chipData);
        if (cvmResults == null) {
            return "No CVM Results available";
        }
        
        return String.format("CVM Code: %s (%s), Condition Code: %s (%s), Result: %s (%s), Valid: %s",
                           cvmResults.getCvmCode(),
                           cvmResults.getCvmCodeDescription(),
                           cvmResults.getConditionCode(),
                           cvmResults.getConditionCodeDescription(),
                           cvmResults.getCvmResults(),
                           cvmResults.getCvmResultsDescription(),
                           cvmResults.isValidForTransaction());
    }
    
    /**
     * Checks if specific CVM type was performed based on chip data
     * 
     * @param chipData The chip data containing CVM information
     * @param cvmType The type of CVM to check for
     * @return true if the specified CVM type was performed successfully
     */
    public boolean isCvmTypePerformed(ChipData chipData, String cvmType) {
        CardholderVerificationMethodResults cvmResults = extractCvmResults(chipData);
        if (cvmResults == null) {
            return false;
        }
        
        return switch (cvmType.toUpperCase()) {
            case "PIN" -> cvmResults.isPinVerificationPerformed() && cvmResults.isCvmSuccessful();
            case "SIGNATURE" -> cvmResults.isSignatureVerificationPerformed() && cvmResults.isCvmSuccessful();
            case "BIOMETRIC" -> cvmResults.isBiometricVerificationPerformed() && cvmResults.isCvmSuccessful();
            case "NO_CVM" -> cvmResults.isNoCvmRequired() && cvmResults.isCvmSuccessful();
            case "CONTACTLESS_NO_CVM" -> cvmResults.isContactlessNoCvm() && cvmResults.isCvmSuccessful();
            default -> {
                logger.warn("Unknown CVM type: {}", cvmType);
                yield false;
            }
        };
    }
    
    /**
     * Checks if offline PIN verification was performed
     * 
     * @param chipData The chip data containing CVM information
     * @return true if offline PIN verification was performed successfully
     */
    public boolean isOfflinePinPerformed(ChipData chipData) {
        CardholderVerificationMethodResults cvmResults = extractCvmResults(chipData);
        return cvmResults != null && cvmResults.isOfflinePinVerificationPerformed() && cvmResults.isCvmSuccessful();
    }
    
    /**
     * Checks if online PIN verification was performed
     * 
     * @param chipData The chip data containing CVM information
     * @return true if online PIN verification was performed successfully
     */
    public boolean isOnlinePinPerformed(ChipData chipData) {
        CardholderVerificationMethodResults cvmResults = extractCvmResults(chipData);
        return cvmResults != null && cvmResults.isOnlinePinVerificationPerformed() && cvmResults.isCvmSuccessful();
    }
    
    /**
     * Checks if this is a contactless transaction based on CVM data
     * 
     * @param chipData The chip data containing CVM information
     * @return true if this appears to be a contactless transaction
     */
    public boolean isContactlessTransaction(ChipData chipData) {
        CardholderVerificationMethodResults cvmResults = extractCvmResults(chipData);
        return cvmResults != null && cvmResults.isContactlessNoCvm();
    }
