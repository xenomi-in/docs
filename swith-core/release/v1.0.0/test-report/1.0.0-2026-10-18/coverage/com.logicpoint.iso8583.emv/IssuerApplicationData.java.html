package com.logicpoint.iso8583.emv;

 /**
  * Issuer Application Data parser (IAD, tag 0x9F10) with support for the following formats:
  * 
  * - VIS 1.5
  * - M/CHIP 4
  * - EMV Format A (as per the EMV 4.3 Book 3 spec)
  * - RuPay EMV Format A (as per the RuPay EMV 4.3 Book 3 spec)
  */
 public final class IssuerApplicationData {
 
     private final String iad;
     private String cvn;
     private String cvr;
     private String dki;
     private String idd;
     private String dac;
     private String counters;
     private CountersInfo countersInfo;
     private String lastOnlineATC;
     private Format format;
     private String cci;
     private IddInfo iddInfo;
 
     //private CryptogramSpec cryptogramSpec = null;
     /**
      * 
      * @param hexIAD Hexadecimal (String) representation of the IAD.
      */
     public IssuerApplicationData(String hexIAD,Scheme scheme) {
 
         Objects.requireNonNull(hexIAD, "IAD data cannot be null.");
 
         iad = hexIAD.trim();
 
         if (iad.length() < 14)
             throw new IllegalArgumentException("Invalid IAD length.");
 
         format = Format.UNKNOWN;
 
         int len = iad.length() / 2; // length in bytes

         if(scheme.equals(Scheme.RUPAY) && iad.startsWith("0105")){
            format = Format.RUPAY_EMV_FORMAT_A;
            unpackVIS(iad);
         }else if(scheme.equals(Scheme.RUPAY) && iad.startsWith("0FB5")){
            format = Format.RUPAY_VIS;
            unpackRuPayEMVFormatA(iad);
            countersInfo = parseRuPayFormatACounters();
            iddInfo = parseRuPayFormatAIdd();

         }else if (Arrays.asList(18, 20, 26, 28).contains(len)) {
             //Therefore, the length of the Issuer Application Data is 18 bytes and for 
             // M/Chip Advance it may be 18, 20, 26, or 28.
             unpackMCHIP(iad);
         } else if (len == 32 && iad.startsWith("0F") && iad.startsWith("0F", 32)) {
             // EMV_v4.3_Book_3
             // C7.2 Issuer Application Data for Format Code 'A'
             unpackEMVFormatA(iad);
         } else if ((len >= 7 && len <= 23) || len == 32) {
             // IAD Format 0/1/3 or IAD Format 2
             unpackVIS(iad);
         } else {
             unpackOther(iad);
         }
     }
 
     /**
      * 
      * @param iad Byte array representation of the IAD.
      */
     public IssuerApplicationData(byte[] iad,Scheme scheme) {
 
         this(ISOUtil.byte2hex(Objects.requireNonNull(iad, "IAD data cannot be null.")),scheme);
     }
 
     private void unpackMCHIP(String data) {
         
         int length = data.length();
         
         format = Format.M_CHIP;
         dki = data.substring(0, 2);
         cvn = data.substring(2, 4);
        // cryptogramSpec = new MChipCryptogram(cvn);
         cvr = data.substring(4, 16);
         dac = data.substring(16, 20);
 
         //Counters (Plaintext or Encrypted)/Accumulators
         if (length <= 40) { //20 bytes
             counters = data.substring(20, 36);
         } else {
             counters = data.substring(20, 52);
         }
 
         // Last Online ATC (Only for M/Chip Advance)
         if (length == 40) {
             lastOnlineATC = data.substring(36, 40);
         } else if (length == 56) {
             lastOnlineATC = data.substring(52, 56);
         }
     }
 
     private void unpackVIS(String iad) {
 
         format = Format.VIS;
         boolean format2 = iad.length() == 64;
 
         if (format2) {
             cvn = iad.substring(2, 4);
             dki = iad.substring(4, 6);
             cvr = iad.substring(6, 16);
             idd = iad.substring(16);
         }
         else {
             dki = iad.substring(2, 4);
             cvn = iad.substring(4, 6);
             cvr = iad.substring(6, 14);
 
             if (iad.length() > 14)
                 idd = iad.substring(13);
         }
        // cryptogramSpec = new VISACryptogram(cvn);
     }
 
     private void unpackEMVFormatA(String data) {
 
         format = Format.EMV_FORMAT_A;
         cvn = data.substring(2, 4);
         dki = data.substring(4, 6);
         cvr = data.substring(6, 16);
         idd = data.substring(34);
     }  

     private void unpackRuPayEMVFormatA(String data) {
 
        format = Format.EMV_FORMAT_A;
        cvn = data.substring(2, 4);
        dki = data.substring(4, 6);  // derivation key index
        cvr = data.substring(6, 22); // card verification results
        counters = data.substring(22, 50); // counters
        idd = data.substring(50); // issuer discretionary data
    } 

 
     private void unpackOther(String data) {
 
         format = Format.OTHER;
         dki = data.substring(2, 4);
         cvn = data.substring(4, 6);
 
         if (data.length() == 64) {            
             String bridge = dki;
             dki = cvn;
             cvn = bridge;            
         }
 
         int cvrLength = Integer.parseInt(data.substring(0, 2), 16);
         cvr = data.substring(8, 8 + cvrLength);
 
         if ((8 + 2 + cvrLength) < data.length())
             idd = data.substring(8 + 2 + cvrLength);        
     }

     public void dump(PrintStream p, String indent) {
         
         String inner = indent + "  ";
         
         p.printf("%s<iad format='%s' value='%s'>", indent, getFormat().toString(), iad);
 
        
         if (dki != null)
             p.printf("%n%sKey derivation index: '%s'", inner, getDerivationKeyIndex());
 
         if (cvn != null)
             p.printf("%n%sCryptogram version number: '%s'", inner, getCryptogramVersionNumber());
 
         if (cvr != null)
             p.printf("%n%sCard verification results: '%s'", inner, getCardVerificationResults());
 
         if (idd != null){
             p.printf("%n%sIssuer discretionary data: '%s'", inner, getIssuerDiscretionaryData()); 
             if (iddInfo != null){
                p.printf("%n%s  Service ID selected: '%s'", inner, iddInfo.getServiceIdSelected());
                p.printf("%n%s  Service management info: '%s'", inner, iddInfo.getServiceManagementInfo());
                p.printf("%n%s  Service compatibility version: '%s'", inner, iddInfo.getServiceCompatibilityVersion());
                p.printf("%n%s  Host compatibility version: '%s'", inner, iddInfo.getHostCompatibilityVersion());
                p.printf("%n%s  RFU: '%s'", inner, iddInfo.getRfu());
             }
         }
 
         if (dac != null)
             p.printf("%n%sDAC/ICC dynamic number: '%s'", inner, getDAC());            
 
         if (counters != null){
             p.printf("%n%sPlaintext/Encrypted counters: '%s'", inner, getCounters());
             if (countersInfo != null){
                p.printf("%n%s  Global balance: '%s'", inner, countersInfo.getGlobalBalance());
                p.printf("%n%s  Service balance current transaction: '%s'", inner, countersInfo.getServiceBalanceCurrentTransaction());
                p.printf("%n%s  RFU: '%s'", inner, countersInfo.getRfu());
             }
         }
         if (lastOnlineATC != null)
             p.printf("%n%sLast ATC Online: '%s'", inner, getLastOnlineATC());
         p.printf("%n%s</iad>%n", indent);  
     }
 
     public enum Format {
         UNKNOWN,
         OTHER,
         M_CHIP,
         EMV_FORMAT_A,
         VIS,
         RUPAY_EMV_FORMAT_A,
         RUPAY_VIS
     }

     /**
      * Parses the IDD field for RuPay EMV Format A as per provided specification.
      * Field offsets are zero-based, all values returned as hexadecimal strings.
      *
      * Bytes (positions are in bytes, each byte = 2 hex chars, index is 0-based):
      *  1-2   : Service Id selected in current transaction     (4 hex chars)
      *  3-4   : Service Management Info                       (4 hex chars)
      *  5     : Service Compatibility Version                 (2 hex chars)
      *  6     : Host Compatibility Version                    (2 hex chars)
      *  7     : RFU                                           (2 hex chars)
      *
      * @return an IddInfo object with the parsed fields, or null if idd is unavailable/invalid
      */
     public IddInfo parseRuPayFormatAIdd() {
         if (idd == null || idd.length() < 14) {
             // At least 7 bytes = 14 hex chars required
             return null;
         }
         String serviceIdSelected = idd.substring(0, 4);            // Bytes 1-2
         String serviceManagementInfo = idd.substring(4, 8);        // Bytes 3-4
         String serviceCompatibilityVersion = idd.substring(8, 10); // Byte 5
         String hostCompatibilityVersion = idd.substring(10, 12);   // Byte 6
         String rfu = idd.substring(12, 14);                        // Byte 7

         return new IddInfo(
             serviceIdSelected,
             serviceManagementInfo,
             serviceCompatibilityVersion,
             hostCompatibilityVersion,
             rfu
         );
     }

     /**
      * POJO to hold the parsed IDD fields for RuPay Format A

         public String getServiceCompatibilityVersion() {
             return serviceCompatibilityVersion;

             return hostCompatibilityVersion;
         }

         }

         @Override

                 '}';
         }
     }
     /**
      * Parses the counters field for RuPay EMV Format A as per specification.
      * Field offsets are zero-based. All values returned as hex strings.
      *
      * Bytes:
      *  1-6   : Global Balance            (12 hex chars)
      *  7-12  : Service Balance Current Transaction (12 hex chars)
      *  13-14 : RFU                       (4 hex chars)
      *
      * @return a CountersInfo object with parsed fields, or null if counters is unavailable/invalid
      */
     public CountersInfo parseRuPayFormatACounters() {
         if (counters == null || counters.length() < 28) {
             // Expecting at least 14 bytes = 28 hex chars
             return null;
         }
         String globalBalance = counters.substring(0, 12);    // Bytes 1-6
         String serviceBalance = counters.substring(12, 24);  // Bytes 7-12
         String rfu = counters.substring(24, 28);             // Bytes 13-14
         return new CountersInfo(globalBalance, serviceBalance, rfu);
     }

     /**
