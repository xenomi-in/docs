package com.logicpoint.iso8583.repository;

import com.logicpoint.iso8583.entity.NetworkSessionKey;
import org.springframework.data.jpa.repository.JpaRepository;
import org.springframework.data.jpa.repository.Query;
import org.springframework.data.repository.query.Param;
import org.springframework.stereotype.Repository;

import java.util.List;
import java.util.Optional;

@Repository
public interface NetworkSessionKeyRepository extends JpaRepository<NetworkSessionKey, Long> {

    @Query("""
        SELECT n FROM NetworkSessionKey n
        WHERE n.siteId = :siteId
          AND n.listenerId = :listenerId
          AND n.connectionId = :connectionId
          AND n.scheme = :scheme
          AND n.keyType = :keyType
          AND n.sessionStatus = 'ACTIVE'
          AND n.keyStatus = 'A'
        ORDER BY n.keyVersion DESC, n.updatedAt DESC
        """)
    List<NetworkSessionKey> findActiveKeys(
            @Param("siteId") String siteId,
            @Param("listenerId") String listenerId,
            @Param("connectionId") String connectionId,
            @Param("scheme") String scheme,
            @Param("keyType") String keyType
    );

    default Optional<NetworkSessionKey> findCurrentActiveKey(String siteId, String listenerId, String connectionId, String scheme, String keyType) {
        List<NetworkSessionKey> keys = findActiveKeys(siteId, listenerId, connectionId, scheme, keyType);
        return keys.isEmpty() ? Optional.empty() : Optional.of(keys.get(0));
    }

    @Query("""
        SELECT n FROM NetworkSessionKey n
        WHERE n.siteId = :siteId
          AND n.listenerId = :listenerId
          AND n.scheme = :scheme
          AND n.keyType = :keyType
          AND n.sessionStatus = 'ACTIVE'
          AND n.keyStatus = 'A'
        ORDER BY n.keyVersion DESC, n.updatedAt DESC
        """)
    List<NetworkSessionKey> findActiveKeysForListener(
            @Param("siteId") String siteId,
            @Param("listenerId") String listenerId,
            @Param("scheme") String scheme,
            @Param("keyType") String keyType
    );

    default Optional<NetworkSessionKey> findCurrentActiveKeyForListener(String siteId, String listenerId, String scheme, String keyType) {
        List<NetworkSessionKey> keys = findActiveKeysForListener(siteId, listenerId, scheme, keyType);
        return keys.isEmpty() ? Optional.empty() : Optional.of(keys.get(0));
    }

    @Query("""
        SELECT n FROM NetworkSessionKey n
        WHERE n.siteId = :siteId
          AND n.listenerId = :listenerId
          AND n.connectionId = :connectionId
          AND n.scheme = :scheme
          AND n.keyType = :keyType
        ORDER BY n.keyVersion DESC, n.updatedAt DESC
        """)
    List<NetworkSessionKey> findKeyHistory(
            @Param("siteId") String siteId,
            @Param("listenerId") String listenerId,
            @Param("connectionId") String connectionId,
            @Param("scheme") String scheme,
            @Param("keyType") String keyType
    );
}
