/**

 * @description Generic validator for Card Not Present (CNP) transactions including E-COMMERCE, IVR, and MOTO.
 * This validator handles scheme-specific validation rules for different payment networks.
 * 
 * Supported transaction types:
 * - ECOMMERCE: Online transactions
 * - IVR_PURCHASE: Interactive Voice Response transactions  
 * - MOTO: Mail Order/Telephone Order transactions
 * - EMI_PURCHASE: EMI purchase transactions
 * - CARD_EMI_PURCHASE: POS EMI purchase transactions
 * 
 * Scheme identification is resolved from RequestContext
 */
@Service
public class EcommerceTransactionValidator extends AbstractTransactionTypeValidator {
    
    private static final Logger log = LoggerFactory.getLogger(EcommerceTransactionValidator.class);
    private final InstallmentAuthorizationValidationService installmentAuthorizationValidationService;
    private final PosDataAuthorizationValidationService posDataAuthorizationValidationService;
    private final PrivateDataAuthorizationValidationService privateDataAuthorizationValidationService;
    private final StandingInstructionAuthorizationValidationService standingInstructionAuthorizationValidationService;
    private final ThreeDsAuthorizationValidationService threeDsAuthorizationValidationService;
    private final TokenAuthorizationValidationService tokenAuthorizationValidationService;

    public EcommerceTransactionValidator(InstallmentAuthorizationValidationService installmentAuthorizationValidationService,
                                         PosDataAuthorizationValidationService posDataAuthorizationValidationService,
                                         PrivateDataAuthorizationValidationService privateDataAuthorizationValidationService,
                                         StandingInstructionAuthorizationValidationService standingInstructionAuthorizationValidationService,
                                         ThreeDsAuthorizationValidationService threeDsAuthorizationValidationService,
                                         TokenAuthorizationValidationService tokenAuthorizationValidationService) {
        this.installmentAuthorizationValidationService = installmentAuthorizationValidationService;
        this.posDataAuthorizationValidationService = posDataAuthorizationValidationService;
        this.privateDataAuthorizationValidationService = privateDataAuthorizationValidationService;
        this.standingInstructionAuthorizationValidationService = standingInstructionAuthorizationValidationService;
        this.threeDsAuthorizationValidationService = threeDsAuthorizationValidationService;
        this.tokenAuthorizationValidationService = tokenAuthorizationValidationService;
    }
    
    // Transaction-specific amount limits
    private static final Map<TransactionType, Double> MAX_AMOUNTS = Map.of(
        TransactionType.ECOMMERCE, 5000.0,
        TransactionType.MOTO, 2500.0,
        TransactionType.IVR_PURCHASE, 1000.0
    );
    
    // Supported transaction types
    private static final Set<TransactionType> SUPPORTED_TYPES = EnumSet.of(
        TransactionType.ECOMMERCE, 
        TransactionType.MOTO, 
        TransactionType.IVR_PURCHASE,
        TransactionType.EMI_PURCHASE,
        TransactionType.CARD_EMI_PURCHASE,
        TransactionType.SI_REGISTRATION,
        TransactionType.SI_EDIT,
        TransactionType.SI_TRANSACTION,
        TransactionType.SI_DEREGISTRATION,
        TransactionType.SI_PAUSE,
        TransactionType.SI_START,
        TransactionType.SI_REVOKEALL,
        TransactionType.SI_PDN

    );
    
    @Override
    public ValidationResult validateTransactionType(TransactionType transactionType, Map<Integer, String> transactionData) {
        return validateTransactionType(transactionType, buildFallbackRequestContext(transactionData));
    }

    @Override
    public ValidationResult validateTransactionType(TransactionType transactionType, RequestContext requestContext) {
        Map<Integer, String> transactionData = requestContext != null ? requestContext.getFields() : null;
        List<String> errors = new ArrayList<>();
        
        // Validate transaction type is supported
        if (!SUPPORTED_TYPES.contains(transactionType)) {
            errors.add(String.format("INVALID_TRANSACTION_TYPE: Expected one of %s, got %s", 
                SUPPORTED_TYPES, transactionType));
            return new ValidationResult(errors, new ArrayList<>());
        }
        
        Scheme scheme = requestContext.getScheme() ;
        /*requestContext != null && requestContext.getScheme() != null
                ? requestContext.getScheme()
                : getSchemeFromTransactionData(transactionData);*/
        log.info("Validating {} transaction for scheme: {}", transactionType, scheme);
        
        // Checking for international transaction validation
        boolean isInternationalTransaction = isValidInternationalTransaction(requestContext);
        if (isInternationalTransaction && transactionType == TransactionType.EMI_PURCHASE && scheme == Scheme.RUPAY) {
            errors.add("INVALID_TRANSACTION: EMI not allowd for international transaction");
            return new ValidationResult(errors, new ArrayList<>());
        }else if (isInternationalTransaction && transactionType == TransactionType.CARD_EMI_PURCHASE && scheme == Scheme.RUPAY) {
            errors.add("INVALID_TRANSACTION: POS EMI not allowd for international transaction");
            return new ValidationResult(errors, new ArrayList<>());
        } else if(isInternationalTransaction ){
            log.info("International transaction: {}", transactionType);
        } else{
            log.info("Domestic transaction: {}", transactionType);
        }
        
        // Validate scheme-specific DE61 field
        validateSchemeBasedDe61(requestContext, scheme, transactionType, transactionData, errors);

        // Execute only the authorization-data validators applicable to this transaction type.
        validateAuthorizationData(requestContext, scheme, transactionType, errors);
        
        // Validate transaction-specific requirements
        validateTransactionSpecificRequirements(transactionType, requestContext, errors);
        
        log.debug("Validation completed for {} {} transaction with {} errors", scheme, transactionType, errors.size());
        
        return new ValidationResult(errors, new ArrayList<>());
    }

    private void validateAuthorizationData(RequestContext requestContext,
                                           Scheme scheme,
                                           TransactionType transactionType,
                                           List<String> errors) {
        switch (transactionType) {
            case ECOMMERCE, MOTO -> {
                validateTokenAuthorization(requestContext, scheme, transactionType, errors);
                validateThreeDsAuthorization(requestContext, scheme, transactionType, errors);
            }
            case IVR_PURCHASE -> {
                errors.addAll(privateDataAuthorizationValidationService.validate(
                        requestContext, scheme, transactionType));
                validateTokenAuthorization(requestContext, scheme, transactionType, errors);
                validateThreeDsAuthorization(requestContext, scheme, transactionType, errors);
            }
            case EMI_PURCHASE, CARD_EMI_PURCHASE -> {
                validateTokenAuthorization(requestContext, scheme, transactionType, errors);
                errors.addAll(installmentAuthorizationValidationService.validate(
                        requestContext, scheme, transactionType));
                validateThreeDsAuthorization(requestContext, scheme, transactionType, errors);
            }
            case SI_REGISTRATION, SI_EDIT, SI_TRANSACTION, SI_DEREGISTRATION -> {
                validateTokenAuthorization(requestContext, scheme, transactionType, errors);
                validateStandingInstruction(requestContext, scheme, transactionType, errors);
            }
            case SI_REVOKEALL, SI_PAUSE, SI_START, SI_PDN ->
                    validateStandingInstruction(requestContext, scheme, transactionType, errors);
            default -> log.debug("No authorization-data validators configured for {}", transactionType);
        }
    }

    private void validateTokenAuthorization(RequestContext requestContext,
                                            Scheme scheme,
                                            TransactionType transactionType,
                                            List<String> errors) {
        errors.addAll(tokenAuthorizationValidationService.validate(requestContext, scheme, transactionType));
    }

    private void validateThreeDsAuthorization(RequestContext requestContext,
                                              Scheme scheme,
                                              TransactionType transactionType,
                                              List<String> errors) {
        errors.addAll(threeDsAuthorizationValidationService.validate(requestContext, scheme, transactionType));
    }

    private void validateStandingInstruction(RequestContext requestContext,
                                             Scheme scheme,
                                             TransactionType transactionType,
                                             List<String> errors) {
        errors.addAll(standingInstructionAuthorizationValidationService.validate(
                requestContext, scheme, transactionType));
    }

    /**
     * Gets the scheme from transaction data field 999
     */
    private Scheme getSchemeFromTransactionData(Map<Integer, String> transactionData) {
        if (transactionData == null) {
            return Scheme.RUPAY;
        }
        return Scheme.RUPAY;
    }

    private RequestContext buildFallbackRequestContext(Map<Integer, String> transactionData) {
        return new RequestContext("", "", transactionData != null ? transactionData.get(32) : null, "", getSchemeFromTransactionData(transactionData), transactionData != null ? transactionData : new HashMap<>());
    }
    
    /**
     * Validates international transaction
     */
    private boolean isValidInternationalTransaction(RequestContext requestContext) {
        int isInternationalTransaction = InternationalTxnValidation.isInternationalTransaction(requestContext);
        return isInternationalTransaction > 1;
    }
    
    /**
     * Validates DE61 field based on scheme and transaction type
     */
    private void validateSchemeBasedDe61(RequestContext requestContext, Scheme scheme, TransactionType transactionType,
                                        Map<Integer, String> transactionData, List<String> errors) {
        errors.addAll(posDataAuthorizationValidationService.validate(requestContext, scheme, transactionType));
    }
    
    /**
     * Validates transaction-specific requirements
     */
    private void validateTransactionSpecificRequirements(TransactionType transactionType,
                                                         RequestContext requestContext,
                                                         List<String> errors) {
        switch (transactionType) {
            case ECOMMERCE:
                validateEcommerceSpecificRequirements(requestContext.getFields(), errors);
                break;
            case MOTO:
                validateMotoSpecificRequirements(requestContext.getFields(), errors);
                break;
            case IVR_PURCHASE:
                validateIvrSpecificRequirements(requestContext, errors);
                break;
            default:
                log.warn("No specific requirements defined for transaction type: {}", transactionType);
        }
    }

    /**
     * Validates CAVV (Cardholder Authentication Verification Value) for e-commerce transactions
     */
    private void validateCavvForEcommerce(RequestContext requestContext, List<String> errors) {
        Map<Integer, String> transactionData = requestContext.getFields();
        String de48 = transactionData.get(48);
        if (de48 == null || de48.isEmpty()) {
            errors.add("MISSING_DE48: DE48 is required for CAVV extraction in e-commerce transactions");
            return;
        }
        
        try {
            Map<String, String> tlvMap = requestContext.getParsedTlv(48);
            String cavv = tlvMap.get("100");
            if (cavv != null && !cavv.isEmpty()) {
                try {
                    // Decode Base64 CAVV value
                    byte[] decodedBytes = Base64.getDecoder().decode(cavv);
                    String cavvDecoded = new String(decodedBytes);

                    // Defensive: If decoded string is shorter than expected, pad with spaces
                    String paddedCavv = String.format("%-40s", cavvDecoded);

                    // Parse fields as per specification
                    String authStatus = paddedCavv.substring(0, 2); // Position 1-2
                    String otpStatus = paddedCavv.substring(2, 4);  // Position 3-4
                    String keyIndicator = paddedCavv.substring(4, 6); // Position 5-6
                    String cvv2 = paddedCavv.substring(6, 10); // Position 7-10
                    String authTranIdLast4 = paddedCavv.substring(10, 14); // Position 11-14
                    String authTranId = paddedCavv.substring(14, 30); // Position 15-30
                    String reservedZeros = paddedCavv.substring(30, 34); // Position 31-34
                    String dateYyddd = paddedCavv.substring(34, 40); // Position 35-40

                    // Validate authentication status
                    Set<String> validAuthStatus = Set.of("00", "05", "06", "07");
                    if (!validAuthStatus.contains(authStatus)) {
                        errors.add("INVALID_CAVV_AUTH_STATUS: Unexpected authentication status in CAVV: " + authStatus);
                    }

                    // Validate OTP status (optional, as per business rules)
                    if (!"00".equals(otpStatus)) {
                        log.info("CAVV OTP status is not '00' (OTP Authentication done): {}", otpStatus);
                    }

                    // Validate reserved zeros
                    if (!reservedZeros.trim().equals("0000")) {
                        errors.add("INVALID_CAVV_RESERVED: Reserved field in CAVV (positions 31-34) must be '0000'");
                    }

                    // Validate date format (YYDDD)
                    if (!dateYyddd.trim().matches("\\d{5,6}")) {
                        errors.add("INVALID_CAVV_DATE: Date in CAVV (positions 35-40) must be in YYDDD format, found: " + dateYyddd.trim());
                    }

                    log.debug("Parsed CAVV metadata - AuthStatus: {}, OTPStatus: {}, KeyIndicator: {}, Date: {}",
                            authStatus, otpStatus, keyIndicator, dateYyddd);

                } catch (IllegalArgumentException e) {
                    log.error("CAVV Base64 decoding failed: {}", e.getMessage(), e);
                    errors.add("INVALID_CAVV_FORMAT: Unable to decode Base64 CAVV value");
                } catch (Exception ex) {
                    log.error("CAVV parsing error: {}", ex.getMessage(), ex);
                    errors.add("INVALID_CAVV_FORMAT: Error parsing CAVV value");
                }
            }
            if (cavv == null || cavv.isEmpty()) {
                errors.add("MISSING_CAVV: CAVV (Tag 100) is required in DE48 for e-commerce transactions");
            } else {
                log.debug("CAVV validation successful for e-commerce transaction");
            }
        } catch (Exception e) {
            log.error("Failed to parse DE48 for CAVV extraction: {}", e.getMessage(), e);
            errors.add("INVALID_DE48_FORMAT: Unable to parse DE48 for CAVV extraction");
        }
    }
    
    /**
     * Validates merchant category code
     */
    private void validateMerchantCategoryCode(Map<Integer, String> transactionData, List<String> errors) {
        String mcc = transactionData.get(18);
        if (mcc == null || mcc.isEmpty()) {
            errors.add("MISSING_MCC: Merchant Category Code (DE18) is required for e-commerce transactions");
            return;
        }
        
        // Validate MCC format (4 digits)
        if (!mcc.matches("\\d{4}")) {
            errors.add("INVALID_MCC: Merchant Category Code must be 4 digits");
        }
        
        log.debug("Merchant Category Code validation: {}", mcc);
    }
