    @Override
    public ValidationResult validateTransactionType(TransactionType transactionType, RequestContext requestContext) {
        Map<Integer, String> transactionData = requestContext != null ? requestContext.getFields() : null;
        List<String> errors = new ArrayList<>();
        Scheme scheme = requestContext != null && requestContext.getScheme() != null
                ? requestContext.getScheme()
                : Scheme.RUPAY;
        
        Map<String, String> tlvData = requestContext != null ? requestContext.getParsedTlv(48) : Map.of();
        
        if(scheme == Scheme.RUPAY){
            validateMoneyLoadAfa(requestContext, transactionType, transactionData, errors);
            if (!errors.isEmpty()) {
                return new ValidationResult(errors, new ArrayList<>());
            }

            ChipData chipData = requestContext != null ? requestContext.getParsedChipData(55) : null;
            validateSchemeBasedDe61(requestContext, scheme, transactionType, transactionData, errors);
            if (chipData == null) {
                errors.add("MISSING_DE55: ICC data is required for NCMC validation");
                return new ValidationResult(errors, new ArrayList<>());
            }
            String iadValue = chipData.getTagValue("9F10");
            if (iadValue == null || iadValue.isBlank()) {
                errors.add("MISSING_DE55_TAG_9F10: Issuer Application Data is required for NCMC validation");
                return new ValidationResult(errors, new ArrayList<>());
            }

            IssuerApplicationData issuerApplicationData;
            try {
                issuerApplicationData = new IssuerApplicationData(iadValue, scheme);
            } catch (IllegalArgumentException ex) {
                errors.add("INVALID_DE55_TAG_9F10: Invalid Issuer Application Data for NCMC validation");
                return new ValidationResult(errors, new ArrayList<>());
            }
            ByteArrayOutputStream baos = new ByteArrayOutputStream();
            PrintStream ps = new PrintStream(baos);
            issuerApplicationData.dump(ps, "");
            log.debug("Issuer Application Data: {}", baos.toString());
            switch (transactionType) {
                case SERVICE_CREATION:                   
                case SERVICE_CREATION_CONTACTLESS:
                    validateServiceCreation(issuerApplicationData,chipData,tlvData,errors); 
                    break;
                case MONEY_LOAD:
                    validateMoneyLoad(issuerApplicationData,chipData,tlvData,errors);
                    break;
                case ML_BALANCEUPDATE:
                    validateMoneyLoadBalanceUpdate(issuerApplicationData,chipData,tlvData,errors);
                    break;
            }

            
        }
        return new ValidationResult(errors, new ArrayList<>());
    }

    private void validateMoneyLoadAfa(RequestContext requestContext,
                                      TransactionType transactionType,
                                      Map<Integer, String> transactionData,
                                      List<String> errors) {
        if (!afaEnabled || transactionType != TransactionType.MONEY_LOAD || transactionData == null) {
            return;
        }

        BigDecimal amount = parseAmount(transactionData.get(4));
        if (amount == null || amount.compareTo(afaLimit) <= 0 || hasPinData(transactionData)) {
            return;
        }

        if (requestContext != null) {
            requestContext.setInternalResponseCode(InternalResCode.INVALID_PIN);
        }
        errors.add(String.format("INVALID_PIN: NCMC Money Load amount %s exceeds AFA limit %s without PIN data",
                amount, afaLimit));
    }

    private BigDecimal parseAmount(String amountField) {
        if (amountField == null || amountField.isBlank()) {
            return null;
        }
        try {
            return new BigDecimal(amountField).movePointLeft(2);
        } catch (NumberFormatException ex) {
            return null;
        }
    }

    private boolean hasPinData(Map<Integer, String> transactionData) {
        String pinData = transactionData.get(52);
        return pinData != null && !pinData.isBlank();
    }

    private RequestContext buildFallbackRequestContext(Map<Integer, String> transactionData) {
        return new RequestContext("", "", transactionData != null ? transactionData.get(32) : null, "", Scheme.RUPAY, transactionData != null ? transactionData : new HashMap<>());
    }

    /**
     * Gets validation rules for scheme and transaction type
     */
    private De61ValidationRules getValidationRules(Scheme scheme, TransactionType transactionType) {
        Map<TransactionType, De61ValidationRules> schemeRules = SCHEME_RULES.get(scheme.name());
        return schemeRules != null ? schemeRules.get(transactionType) : null;
    }
      /**
     * Validates DE61 field based on scheme and transaction type
     */
    private void validateSchemeBasedDe61(RequestContext requestContext, Scheme scheme, TransactionType transactionType,
                                        Map<Integer, String> transactionData, List<String> errors) {
        String de61 = transactionData.get(61);
        if (de61 == null || de61.isEmpty()) {
            errors.add(String.format("MISSING_DE61: POS Data Code (DE61) is required for %s %s transactions", 
                scheme, transactionType));
            return;
        }
        
        try {
            RuPayDe61Parser.RuPayDE61Data de61data = requestContext != null
                    ? requestContext.getParsedDe61(61)
                    : RuPayDe61Parser.parseDE61(de61);
            if (de61data.hasErrors()) {
                errors.addAll(de61data.getErrors());
                return;
            }
            
            log.debug("Validating DE61 for scheme: {}, transaction type: {}", scheme, transactionType);
            
            // Get validation rules for the scheme and transaction type
            De61ValidationRules rules = getValidationRules(scheme, transactionType);
            if (rules != null) {
                validateDe61Subfields(de61data, rules, scheme, transactionType, errors);
            } else {
                log.warn("No validation rules found for scheme: {} and transaction type: {}", scheme, transactionType);
                // For unsupported schemes, use RuPay rules as fallback
                if (scheme != Scheme.RUPAY) {
                    De61ValidationRules rupayRules = getValidationRules(Scheme.RUPAY, transactionType);
                    if (rupayRules != null) {
                        log.info("Using RuPay rules as fallback for scheme: {}", scheme);
                        validateDe61Subfields(de61data, rupayRules, scheme, transactionType, errors);
                    }
                }
            }
            
            
        } catch (Exception e) {
            log.error("Error parsing DE61 for {} {} transaction: {}", scheme, transactionType, e.getMessage());
            errors.add(String.format("INVALID_DE61: Error parsing DE61 field for %s %s transaction", 
                scheme, transactionType));
        }
    }

     /**
     * Validates DE61 subfields based on rules
     */
    private void validateDe61Subfields(RuPayDe61Parser.RuPayDE61Data de61data, De61ValidationRules rules,
                                      Scheme scheme, TransactionType transactionType, List<String> errors) {
        
        // Validate SF1 (Card Data Input Capability) - only for E-commerce
        if (transactionType == TransactionType.ECOMMERCE && rules.getSf1Values() != null) {
            validateSubfield(de61data, RuPayDE61Subfields.CARD_DATA_INPUT_CAPABILITY, 
                rules.getSf1Values(), scheme, transactionType, errors);
        }
        
        // Validate SF5 (Cardholder Present)
        validateSubfield(de61data, RuPayDE61Subfields.CARDHOLDER_PRESENT, 
            rules.getSf5Values(), scheme, transactionType, errors);
        
        // Validate SF6 (Card Present)
        validateSubfield(de61data, RuPayDE61Subfields.CARD_PRESENT, 
            rules.getSf6Values(), scheme, transactionType, errors);
        
        // Validate SF7 (Card Data Input Mode)
        validateSubfield(de61data, RuPayDE61Subfields.CARD_DATA_INPUT_MODE, 
            rules.getSf7Values(), scheme, transactionType, errors);
        
        // Validate SF8 (Cardholder Authentication Method)
        validateSubfield(de61data, RuPayDE61Subfields.CARDHOLDER_AUTH_METHOD, 
            rules.getSf8Values(), scheme, transactionType, errors);
    }
    
    /**
     * Validates a specific subfield
     */
    private void validateSubfield(RuPayDe61Parser.RuPayDE61Data de61data, RuPayDE61Subfields subfield,
                                 Set<String> allowedValues, Scheme scheme, TransactionType transactionType, 
                                 List<String> errors) {
        if (allowedValues == null) return;
        
        String value = de61data.getSubfieldValue(subfield);
        String normalizedValue = subfield == RuPayDE61Subfields.CARDHOLDER_AUTH_METHOD ? 
            value.toUpperCase() : value;
        
        if (!allowedValues.contains(normalizedValue)) {
            String allowedValuesStr = allowedValues.stream().sorted().collect(Collectors.joining(","));
            String subfieldName = SUBFIELD_NAMES.get(subfield);
            errors.add(String.format("INVALID_DE61: %s Expected one of [%s] for %s %s transactions", 
                subfieldName, allowedValuesStr, scheme, transactionType));
        }
    }

 /**
     * DE61 validation rules container

    private static class De61ValidationRules {
        private final Set<String> sf1Values; // Card Data Input Capability
        private final Set<String> sf5Values; // Cardholder Present
        private final Set<String> sf6Values; // Card Present
        private final Set<String> sf7Values; // Card Data Input Mode
        private final Set<String> sf8Values; // Cardholder Authentication Method
        
        public De61ValidationRules(Set<String> sf1Values, Set<String> sf5Values, 
                                  Set<String> sf6Values, Set<String> sf7Values, Set<String> sf8Values) {
            this.sf1Values = sf1Values;
            this.sf5Values = sf5Values;
            this.sf6Values = sf6Values;
            this.sf7Values = sf7Values;
            this.sf8Values = sf8Values;
        }
        
        // Getters
        public Set<String> getSf1Values() { return sf1Values; }
        public Set<String> getSf5Values() { return sf5Values; }
        public Set<String> getSf6Values() { return sf6Values; }
        public Set<String> getSf7Values() { return sf7Values; }
        public Set<String> getSf8Values() { return sf8Values; }
    }
