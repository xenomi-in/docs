package com.logicpoint.iso8583.validation.transaction.impl;

import com.logicpoint.iso8583.validation.transaction.TransactionTypeValidator;
import com.logicpoint.iso8583.validation.ValidationResult;
import com.logicpoint.iso8583.validation.TransactionType;
import org.springframework.stereotype.Component;

import java.util.ArrayList;
import java.util.List;
import java.util.Map;
import java.util.HashMap;

/**
 * Abstract base class for transaction type validators with common functionality.
 */
@Component
public abstract class AbstractTransactionTypeValidator implements TransactionTypeValidator {
    
    protected static final Map<TransactionType, List<String>> REQUIRED_FIELDS = new HashMap<>();
    
    static {
        // Define required fields for each transaction type
        List<String> purchaseFields = new ArrayList<>();
        purchaseFields.add("PAN");
        purchaseFields.add("AMOUNT");
        purchaseFields.add("MERCHANT_ID");
        purchaseFields.add("EXPIRY_DATE");
        REQUIRED_FIELDS.put(TransactionType.PURCHASE, purchaseFields);
        
        List<String> ecommerceFields = new ArrayList<>();
        ecommerceFields.add("PAN");
        ecommerceFields.add("AMOUNT");
        ecommerceFields.add("MERCHANT_ID");
        ecommerceFields.add("EXPIRY_DATE");
        ecommerceFields.add("CVV");
        ecommerceFields.add("IP_ADDRESS");
        REQUIRED_FIELDS.put(TransactionType.ECOMMERCE, ecommerceFields);
        
        List<String> atmFields = new ArrayList<>();
        atmFields.add("PAN");
        atmFields.add("AMOUNT");
        atmFields.add("ATM_ID");
        atmFields.add("PIN");
        REQUIRED_FIELDS.put(TransactionType.CASH_WITHDRAWAL, atmFields);
        
        List<String> refundFields = new ArrayList<>();
        refundFields.add("ORIGINAL_TRANSACTION_ID");
        refundFields.add("AMOUNT");
        refundFields.add("MERCHANT_ID");
        REQUIRED_FIELDS.put(TransactionType.REFUND, refundFields);
    }
    
    @Override
    public ValidationResult validate(TransactionType transactionType) {
        return validateTransactionType(transactionType, (Map<Integer, String>) null);
    }
    
    @Override
    public ValidationResult validateAmountLimits(TransactionType transactionType, Double amount) {
        List<String> errors = new ArrayList<>();
        
        if (amount == null || amount <= 0) {
            errors.add("AMOUNT_LIMIT: Invalid transaction amount");
            return new ValidationResult(errors, new ArrayList<>());
        }
        
        // Check transaction type specific limits
      /*  double maxAmount = getMaxAmountForTransactionType(transactionType);
        if (amount > maxAmount) {
            errors.add("AMOUNT_LIMIT: Amount exceeds maximum limit for " + transactionType);
        }
        */
        return new ValidationResult(errors, new ArrayList<>());
    }
    
    @Override
    public ValidationResult validateVelocityLimits(TransactionType transactionType, String cardNumber) {
        List<String> errors = new ArrayList<>();
        
        // In a real implementation, this would check against a database
        // for velocity limits (e.g., max transactions per hour, day, etc.)
        if (isVelocityLimitExceeded(transactionType, cardNumber)) {
            errors.add("VELOCITY_LIMIT: Velocity limit exceeded for " + transactionType);
        }
        
        return new ValidationResult(errors, new ArrayList<>());
    }
    
    /**
     * Get the maximum amount allowed for a specific transaction type.
     */
    protected abstract double getMaxAmountForTransactionType(TransactionType transactionType);
    
    /**
     * Check if velocity limits are exceeded for the given card and transaction type.
     */
    protected abstract boolean isVelocityLimitExceeded(TransactionType transactionType, String cardNumber);
    
    /**
     * Get the list of required fields for a transaction type.
     */
    protected List<String> getRequiredFields(TransactionType transactionType) {
        return REQUIRED_FIELDS.getOrDefault(transactionType, new ArrayList<>());
    }
}
