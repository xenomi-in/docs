package com.logicpoint.iso8583.validation.transaction.impl;

import io.swagger.v3.oas.annotations.servers.Server;

import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Service;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

import com.logicpoint.iso8583.validation.ValidationResult;
import com.logicpoint.iso8583.validation.TransactionType;

import java.math.BigDecimal;
import java.util.Random;
import java.util.Map;
import java.util.List;
import java.util.ArrayList;
import java.util.Set;
import java.util.EnumSet;

import com.logicpoint.iso8583.util.rupay.De54data;

/* @author LogicPoint Systems
 * @version 1.0
 * @date 18-10-2025
 * @description Generic validator for ATM service transactions including BALANCE_INQUIRY, MINI_STATEMENT, PIN_CHANGE, STATEMENT_REQUEST etc...
 * This validator handles scheme-specific validation rules for different payment networks.
 *
 * Supported transaction types: - BALANCE_INQUIRY: Balance inquiry transactions
 * - MINI_STATEMENT: Mini statement transactions
 * - PIN_CHANGE: PIN change transactions
 * - STATEMENT_REQUEST: Statement request transaction
 * 
 *
 * Scheme identification: Field 999 contains the scheme name
 */
@Service
public class AtmServiceTransactionValidator extends AbstractTransactionTypeValidator {
    private static final Logger log = LoggerFactory.getLogger(AtmServiceTransactionValidator.class);

    @Value("${accounting.api.enabled:false}")
    boolean accountingApiEnabled;
    private static final Set<TransactionType> SUPPORTED_TYPES = EnumSet.of(
        TransactionType.BALANCE_ENQUIRY,
        TransactionType.MINI_STATEMENT,
        TransactionType.PIN_CHANGE,
        TransactionType.STATEMENT_REQUEST,
        TransactionType.BALANCE_ENQUIRY_CL
    );

    @Override
    public ValidationResult validateTransactionType(TransactionType transactionType, Map<Integer, String> transactionData) {
        List<String> errors = new ArrayList<>();
        if(!SUPPORTED_TYPES.contains(transactionType)) {
            errors.add("INVALID_TRANSACTION_TYPE: Expected one of " + SUPPORTED_TYPES + ", got " + transactionType);
            return new ValidationResult(errors, new ArrayList<>());
        }
        String txnAmount = transactionData.get(4);
        if(!txnAmount.equals("000000000000")) {
            errors.add("INVALID_TRANSACTION_AMOUNT: Transaction amount must be greater than zero");
            return new ValidationResult(errors, new ArrayList<>());
        }

        // adding De54 for balance enquiry this should be removied after accounting integration
        if(!accountingApiEnabled && (transactionType == TransactionType.BALANCE_ENQUIRY || transactionType == TransactionType.BALANCE_ENQUIRY_CL)){
            Random random = new Random();
            BigDecimal availableBalance = BigDecimal.valueOf(random.nextInt(10000));
            BigDecimal ledgerBalance = availableBalance.add(BigDecimal.ONE);
            String de54 = De54data.packBalanceEnquiryDe54(ledgerBalance, availableBalance);
            transactionData.put(54, de54);
        }
        return new ValidationResult(errors, new ArrayList<>());
    }

    @Override
    public ValidationResult validateAmountLimits(TransactionType transactionType, Double amount) {
        return null;
    }

    @Override
    public ValidationResult validateVelocityLimits(TransactionType transactionType, String cardNumber) {
        return null;
    }

    @Override
    public double getMaxAmountForTransactionType(TransactionType transactionType) {
        return 0;
    }

    @Override
    public boolean isVelocityLimitExceeded(TransactionType transactionType, String cardNumber) {
        return false;
    }

    @Override
    public String getValidatorType() {
        return "ATM_SERVICE";
    }
}
