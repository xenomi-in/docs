 * Generic validator for Cash withdrawal transactions including CASH_WITHDRAWAL
 * and CASH_AT_POS. This validator handles scheme-specific validation rules for
 * different payment networks.
 *
 * Supported transaction types: - CASH_WITHDRAWAL: ATM cash withdrawal

 *
 * Scheme identification is resolved from RequestContext

    @Override

        return validateTransactionType(transactionType, buildFallbackRequestContext(transactionData));

    @Override
    public ValidationResult validateTransactionType(TransactionType transactionType, RequestContext requestContext) {
        Map<Integer, String> transactionData = requestContext != null ? requestContext.getFields() : null;
        List<String> errors = new ArrayList<>();

        // Validate transaction type is supported
        if (!SUPPORTED_TYPES.contains(transactionType)) {
            errors.add(String.format("INVALID_TRANSACTION_TYPE: Expected one of %s, got %s",
                    SUPPORTED_TYPES, transactionType));
            return new ValidationResult(errors, new ArrayList<>());
        }
        //DE4: Transaction amount validation > zero
        Double amount = 0.0;
        String amountStr = transactionData.get(4); // DE4: Transaction amount (usually in smallest currency unit, e.g., paise)
        if (amountStr != null && !amountStr.isEmpty()) {
            try {

                amount = Double.parseDouble(amountStr) / 100.0;
                if (amount <= 0) {
                    errors.add("INVALID_AMOUNT: Transaction amount must be greater than zero");
                    return new ValidationResult(errors, new ArrayList<>());
                }
            } catch (NumberFormatException ex) {
                errors.add("INVALID_AMOUNT: Unable to parse transaction amount");
                return new ValidationResult(errors, new ArrayList<>());
            }
        } else {
            errors.add("MISSING_AMOUNT: Transaction amount (DE4) is required");
            return new ValidationResult(errors, new ArrayList<>());
        }

        if (transactionType == TransactionType.CASH_AT_POS_CL) {
            // Check if the transaction amount is greater than 5000.00
            try {
                // Assuming amount is in paise (INR), so 5000.00 INR = 500000 paise
                if (amount > 5000.00 && !transactionData.containsKey(52)) {
                    errors.add("INVALID_TRANSACTION: Transaction amount exceeds 5000.00 INR for contactless purchase with out PIN");
                    return new ValidationResult(errors, new ArrayList<>());
                }
            } catch (NumberFormatException ex) {
                errors.add("INVALID_AMOUNT: Unable to parse transaction amount");
                return new ValidationResult(errors, new ArrayList<>());
            }
        } 

        Scheme scheme = requestContext != null && requestContext.getScheme() != null
                ? requestContext.getScheme()
                : getSchemeFromTransactionData(transactionData);
        log.info("Validating {} transaction for scheme: {}", transactionType, scheme);

        // Check international transaction validation
        if (!isValidInternationalTransaction(transactionData)) {
            errors.add("INVALID_TRANSACTION: International transaction validation failed");
            return new ValidationResult(errors, new ArrayList<>());
        }

        if (isFallBackTxn(transactionData, scheme.name())) {
            log.info("FallBack transaction detected for {} {} transaction", scheme, transactionType);
            //return new ValidationResult(errors, new ArrayList<>());
        }

        // Validate MCC based on transaction type
        validateMerchantCategoryCode(transactionType, transactionData, errors);

        // Validate scheme-specific DE61 field
        boolean isvalid = validateSchemeBasedDe61(scheme, transactionType, transactionData, errors);
        if (!isvalid) {

            log.error("Validation failed for {} {} transaction", scheme, transactionType);
            return new ValidationResult(errors, new ArrayList<>());
        } else {
            log.info("De61 validation completed for {} {} transaction", scheme, transactionType);
        }

        // Validate transaction-specific requirements
        //validateTransactionSpecificRequirements(transactionType, transactionData, errors);
        log.debug("Validation completed for {} {} transaction with {} errors", scheme, transactionType, errors.size());

        return new ValidationResult(errors, new ArrayList<>());
    }

    /**
     * Gets the scheme from transaction data field 999
     */
    private Scheme getSchemeFromTransactionData(Map<Integer, String> transactionData) {
        if (transactionData == null) {
            return Scheme.RUPAY;
        }
        String schemeStr = transactionData.get(999);
        if (schemeStr != null && !schemeStr.isEmpty()) {
            try {
                return Scheme.valueOf(schemeStr.toUpperCase());
            } catch (IllegalArgumentException e) {
                log.warn("Invalid scheme in field 999: {}, defaulting to RUPAY", schemeStr);
            }
        } else {
            log.warn("No scheme found in field 999, defaulting to RUPAY");
        }
        return Scheme.RUPAY;
    }

    private RequestContext buildFallbackRequestContext(Map<Integer, String> transactionData) {
        return new RequestContext("", "", transactionData != null ? transactionData.get(32) : null, "", getSchemeFromTransactionData(transactionData), transactionData != null ? transactionData : new HashMap<>());
    }

    /**
     * Validates international transaction
     */
    private boolean isValidInternationalTransaction(Map<Integer, String> transactionData) {
        int isInternationalTransaction = InternationalTxnValidation.isInternationalTransaction(transactionData);
        return isInternationalTransaction != 0;
    }

    /**
     * Validates merchant category code based on transaction type
     */
    private void validateMerchantCategoryCode(TransactionType transactionType, Map<Integer, String> transactionData, List<String> errors) {
        String mcc = transactionData.get(18);
        if (mcc == null || mcc.isEmpty()) {
            errors.add("MISSING_MCC: Merchant Category Code (DE18) is required for cash withdrawal transactions");
            return;
        }

        Set<String> allowedMccs = MCC_RULES.get(transactionType);
        if (allowedMccs != null && !allowedMccs.contains(mcc)) {
            String allowedMccsStr = allowedMccs.stream().sorted().collect(Collectors.joining(","));
            errors.add(String.format("INVALID_MCC: Merchant Category Code must be one of [%s] for %s transactions, got %s",
                    allowedMccsStr, transactionType, mcc));
        }

        log.debug("MCC validation for {} transaction: {}", transactionType, mcc);
    }

    /**
     * Validates DE61 field based on scheme and transaction type
     */
    private boolean validateSchemeBasedDe61(Scheme scheme, TransactionType transactionType,
            Map<Integer, String> transactionData, List<String> errors) {
        boolean isvalid = false;
        String de61 = transactionData.get(61);
        if (de61 == null || de61.isEmpty()) {
            errors.add(String.format("MISSING_DE61: POS Data Code (DE61) is required for %s %s transactions",
                    scheme, transactionType));
            return false;
        }

        try {
            RuPayDe61Parser.RuPayDE61Data de61data = RuPayDe61Parser.parseDE61(de61);
            if (de61data.hasErrors()) {
                errors.addAll(de61data.getErrors());
                return false;
            }

            log.debug("Validating DE61 for scheme: {}, transaction type: {}", scheme, transactionType);

            // Get validation rules for the scheme and transaction type
            AtmTxnDe61ValidationRules rules = getValidationRules(scheme, transactionType);
            if (rules != null) {
                isvalid = validateDe61Subfields(de61data, rules, scheme, transactionType, errors);
                return isvalid;
            } else {
                log.warn("No validation rules found for scheme: {} and transaction type: {}", scheme, transactionType);
                // For unsupported schemes, use RuPay rules as fallback
                if (scheme != Scheme.RUPAY) {
                    AtmTxnDe61ValidationRules rupayRules = getValidationRules(Scheme.RUPAY, transactionType);
                    if (rupayRules != null) {
                        log.info("Using RuPay rules as fallback for scheme: {}", scheme);
                        isvalid = validateDe61Subfields(de61data, rupayRules, scheme, transactionType, errors);
                        return isvalid;
                    }
                }
            }
            return isvalid;
        } catch (Exception e) {
            log.error("Error parsing DE61 for {} {} transaction: {}", scheme, transactionType, e.getMessage());
            errors.add(String.format("INVALID_DE61: Error parsing DE61 field for %s %s transaction",
                    scheme, transactionType));
            return false;
        }
    }

    /**
     * Gets validation rules for scheme and transaction type
     */
    private AtmTxnDe61ValidationRules getValidationRules(Scheme scheme, TransactionType transactionType) {
        Map<TransactionType, AtmTxnDe61ValidationRules> schemeRules = SCHEME_RULES.get(scheme.name());
        return schemeRules != null ? schemeRules.get(transactionType) : null;
    }

    /**
     * Validates DE61 subfields based on rules
     */
    private boolean validateDe61Subfields(RuPayDe61Parser.RuPayDE61Data de61data, AtmTxnDe61ValidationRules rules,
            Scheme scheme, TransactionType transactionType, List<String> errors) {

        boolean isvalid;
        // Validate SF1 (Card Data Input Capability)
        isvalid = validateSubfield(de61data, RuPayDE61Subfields.CARD_DATA_INPUT_CAPABILITY,
                rules.getSf1Values(), scheme, transactionType, errors);

        // Validate SF5 (Cardholder Present) - must be specific value
        if (isvalid) {
            isvalid = validateSubfield(de61data, RuPayDE61Subfields.CARDHOLDER_PRESENT,
                    rules.getSf5Values(), scheme, transactionType, errors); 
        }else {
            return isvalid;
        }
        // Validate SF6 (Card Present)
        if (isvalid) {
            isvalid = validateSubfield(de61data, RuPayDE61Subfields.CARD_PRESENT,
                    rules.getSf6Values(), scheme, transactionType, errors); 
        }else {
            return isvalid;
        }
        // Validate SF7 (Card Data Input Mode)
        if (isvalid) {
            isvalid = validateSubfield(de61data, RuPayDE61Subfields.CARD_DATA_INPUT_MODE,
                    rules.getSf7Values(), scheme, transactionType, errors); 
        }else {
            return isvalid;
        }
        // Validate SF8 (Cardholder Authentication Method)
        if (isvalid) {
            isvalid = validateSubfield(de61data, RuPayDE61Subfields.CARDHOLDER_AUTH_METHOD,
                    rules.getSf8Values(), scheme, transactionType, errors);
        }
        return isvalid;
    }

    /**
     * Validates a specific subfield with allowed values
     */
    private boolean validateSubfield(RuPayDe61Parser.RuPayDE61Data de61data, RuPayDE61Subfields subfield,
            Set<String> allowedValues, Scheme scheme, TransactionType transactionType,
            List<String> errors) {
        if (allowedValues == null) {
            return false;
        }

        String value = de61data.getSubfieldValue(subfield);
        if (!allowedValues.contains(value)) {
            String allowedValuesStr = allowedValues.stream().sorted().collect(Collectors.joining(","));
            String subfieldName = SUBFIELD_NAMES.get(subfield);
            errors.add(String.format("INVALID_DE61: %s Expected one of [%s] for %s %s transactions",
                    subfieldName, allowedValuesStr, scheme, transactionType));
            return false;
        } else {
            return true;
        }

    }

    /**
     * Validates a specific subfield with exact value requirement
     */
    private boolean validateSpecificSubfield(RuPayDe61Parser.RuPayDE61Data de61data, RuPayDE61Subfields subfield,
            String expectedValue, Scheme scheme, TransactionType transactionType,
            List<String> errors) {
        if (expectedValue == null) {
            return false;
        }

        String value = de61data.getSubfieldValue(subfield);
        if (!expectedValue.equals(value)) {
            String subfieldName = SUBFIELD_NAMES.get(subfield);
            errors.add(String.format("INVALID_DE61: %s Expected '%s' for %s %s transactions, got '%s'",
                    subfieldName, expectedValue, scheme, transactionType, value));
            return false;
        } else {
            return true;
        }
    }

    /**
     * Validates transaction-specific requirements
     */
    private void validateTransactionSpecificRequirements(TransactionType transactionType,
            Map<Integer, String> transactionData,
            List<String> errors) {
        switch (transactionType) {
            case CASH_WITHDRAWAL:
                validateCashWithdrawalSpecificRequirements(transactionData, errors);
                break;
            case CASH_AT_POS:
                validateCashAtPosSpecificRequirements(transactionData, errors);
                break;
            default:
                log.warn("No specific requirements defined for transaction type: {}", transactionType);
        }
    }

    /**
     * Validates cash withdrawal specific requirements
     */
    private void validateCashWithdrawalSpecificRequirements(Map<Integer, String> transactionData, List<String> errors) {
        // Validate PIN block format
        validatePinBlockFormat(transactionData, errors);

        // Validate processing code
        validateProcessingCode(transactionData, "010000", errors);

        // Validate amount limits
        validateAmountLimits(transactionData, TransactionType.CASH_WITHDRAWAL, errors);

        log.debug("Cash withdrawal specific requirements validated");
    }

    /**
     * Validates cash at POS specific requirements
     */
    private void validateCashAtPosSpecificRequirements(Map<Integer, String> transactionData, List<String> errors) {
        // Validate PIN block format
        validatePinBlockFormat(transactionData, errors);

        // Validate processing code
        validateProcessingCode(transactionData, "010000", errors);

        // Validate amount limits
        validateAmountLimits(transactionData, TransactionType.CASH_AT_POS, errors);

        log.debug("Cash at POS specific requirements validated");
    }

    /**
     * Validates PIN block format
     */
    private void validatePinBlockFormat(Map<Integer, String> transactionData, List<String> errors) {
        String pinBlock = transactionData.get(52);
        if (pinBlock == null || pinBlock.isEmpty()) {
            return; // Already handled in required fields validation
        }

        // Validate PIN block format (should be 16 hex characters)
        if (!pinBlock.matches("[0-9A-Fa-f]{16}")) {
            errors.add("INVALID_PIN_FORMAT: PIN block must be 16 hexadecimal characters");
        }

        log.debug("PIN block format validation completed");
    }

    /**
     * Validates processing code
     */
    private void validateProcessingCode(Map<Integer, String> transactionData, String expectedCode, List<String> errors) {
        String processingCode = transactionData.get(3);
        if (processingCode == null || processingCode.isEmpty()) {
            return; // Already handled in required fields validation
        }

        if (!expectedCode.equals(processingCode)) {
            log.debug("Processing code validation: expected {}, got {}", expectedCode, processingCode);
            // Note: Not adding error as processing codes can vary
        }
    }

    /**
     * Validates amount limits based on transaction type
     */
    private void validateAmountLimits(Map<Integer, String> transactionData, TransactionType transactionType, List<String> errors) {
        String amountStr = transactionData.get(4);
        if (amountStr == null || amountStr.isEmpty()) {
            return; // Already handled in required fields validation
        }

        try {
            double amount = Double.parseDouble(amountStr) / 100.0; // Convert from paise to rupees
            log.debug("Cash withdrawal amount: {}", amount);

            double maxAmount = getMaxAmountForTransactionType(transactionType);
            if (amount > maxAmount) {
                errors.add(String.format("AMOUNT_EXCEEDED: Transaction amount %.2f exceeds maximum limit %.2f for %s",
                        amount, maxAmount, transactionType));
                return;
            }

            // Minimum amount validation
            if (amount <= 0) {
                errors.add("INVALID_AMOUNT: Transaction amount must be greater than zero");
                return;
            }

        } catch (NumberFormatException e) {
            errors.add("INVALID_AMOUNT: Unable to parse transaction amount");
        }
    }
