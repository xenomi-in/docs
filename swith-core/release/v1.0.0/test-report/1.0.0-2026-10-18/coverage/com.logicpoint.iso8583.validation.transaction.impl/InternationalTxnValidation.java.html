package com.logicpoint.iso8583.validation.transaction.impl;

import com.logicpoint.iso8583.context.RequestContext;
import com.logicpoint.iso8583.util.rupay.CardAcceptorLocation;
import com.logicpoint.iso8583.validation.TlvUtils;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

import java.util.Map;

/*
 * International transaction validation
 * @author LogicPoint Systems
 * @version 1.0
 */
public class InternationalTxnValidation {
    
    private static final Logger logger = LoggerFactory.getLogger(InternationalTxnValidation.class);
    private static final String BASE_COUNTRY_CODE = "IN";
    private static final String BASE_CURRENCY_CODE = "356";
    public static int isInternationalTransaction(RequestContext requestContext){
        if (requestContext == null) {
            return 0;
        }
        return isInternationalTransaction(requestContext.getFields(), requestContext.getParsedTlv(48));
    }

    public static int isInternationalTransaction(Map<Integer, String> transactionData){
        String de48 = transactionData != null ? transactionData.get(48) : null;
        Map<String, String> de48Map = de48 == null || de48.isBlank()
                ? Map.of()
                : TlvUtils.parseTlv(de48);
        return isInternationalTransaction(transactionData, de48Map);
    }

    private static int isInternationalTransaction(Map<Integer, String> transactionData, Map<String, String> de48Map){
    try{
        String acquiringCountryCode = transactionData.get(19); // Acquiring Country Code
        String terminalLocation = transactionData.get(43); // Terminal Location
        String txnCurrencyCode = transactionData.get(49); // Transaction Currency Code
        String txnBillingCurrencyCode = transactionData.get(51); // Billing Currency Code
        String dcc = de48Map.get("088");
        CardAcceptorLocation cardAcceptorLocation = CardAcceptorLocation.parse(terminalLocation);
        String terminalCountryCode = cardAcceptorLocation.getTerminalCountryCode();
        if(!terminalCountryCode.equals(BASE_COUNTRY_CODE)){
            logger.info("International transaction , Terminal country code is not IN");
            
            if(txnCurrencyCode.equals(txnBillingCurrencyCode)){
                logger.info("International transaction(Cross border) , With DCC  ");
                // check DE48 tag 88  validation need to check for dcc
                return 2; // Cross border transaction without DCC
            }else {
                // check DE48 tag 88 
               /*  if(dcc != null &&  dcc.equals("1")){
                    logger.info("International transaction(Cross border) , With out DCC ");
                    return 3; // Cross border transaction with DCC
                }else {*/
                    logger.info("International transaction(Cross border) ,With out DCC ");
                    return 3; // Cross border transaction with out DCC validation 
            }
        }else{
            logger.info("Domestic transaction , Terminal country code is {}", terminalCountryCode);
            return 1; // Domestic transaction
        }
    }catch(Exception e){
        logger.error("Error in isInternationalTransaction: {}", e.getMessage());
        return 0;
    }
    }
}
