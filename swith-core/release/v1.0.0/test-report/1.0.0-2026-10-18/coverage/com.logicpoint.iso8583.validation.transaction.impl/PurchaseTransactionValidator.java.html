package com.logicpoint.iso8583.validation.transaction.impl;

import com.logicpoint.iso8583.validation.ValidationResult;
import com.logicpoint.iso8583.validation.TransactionType;
import org.springframework.stereotype.Service;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import java.util.ArrayList;
import java.util.List;
import java.util.Map;
import com.logicpoint.iso8583.util.rupay.De54data;
import com.logicpoint.iso8583.util.rupay.RuPayDe61Parser;
import com.logicpoint.iso8583.util.rupay.RuPayDe61Parser.RuPayDE61Subfields;
/**
 * Validator for purchase transactions with specific business rules.
 */
@Service
public class PurchaseTransactionValidator extends AbstractTransactionTypeValidator {
    Logger log = LoggerFactory.getLogger(PurchaseTransactionValidator.class);
    private static final double MAX_PURCHASE_AMOUNT = 10000.0; // $10,000
    
    @Override
    public ValidationResult validateTransactionType(TransactionType transactionType, Map<Integer, String> transactionData) {
        List<String> errors = new ArrayList<>();
        
        if (transactionType == TransactionType.PURCHASE || transactionType == TransactionType.PURCHASE_CONTACTLESS || transactionType == TransactionType.PURCHASE_WITHCASHBACK || transactionType == TransactionType.PREAUTH || transactionType == TransactionType.PURCHASE_FALLBACK) {
            log.info("Purchase Transaction Validated - txn Type: {}", transactionType);
        }else{
            errors.add("INVALID_TRANSACTION_TYPE: Expected PURCHASE, PURCHASE_CONTACTLESS or PURCHASE_WITHCASHBACK transaction");
            return new ValidationResult(errors, new ArrayList<>());
        }
        if(transactionType == TransactionType.PREAUTH){
            log.info("Preauth Transaction Validated - txn Type: {}", transactionType);
        // INSERT_YOUR_CODE

        String mcc = transactionData.get(18);
        if (!isValidRuPayPreAuthMcc(mcc)) {
            errors.add("MCC_NOT_ALLOWED: Merchant Category Code (MCC) is invalid or not supported fro Preauth transactions.");
            return new ValidationResult(errors, new ArrayList<>());
        }

        }
        
        if(transactionType == TransactionType.PURCHASE_WITHCASHBACK){
            if(!purchaseWithCashback(transactionData)){
                errors.add("INVALID_TRANSACTION: PURCHASE_WITHCASHBACK validation failed ");
                return new ValidationResult(errors, new ArrayList<>());
            }
        }
        // cheking is crossboarder txn 
        int isInternationalTransaction = InternationalTxnValidation.isInternationalTransaction(transactionData);
        if(isInternationalTransaction == 0){
            errors.add("INVALID_TRANSACTION: International transaction validation failed ");
            return new ValidationResult(errors, new ArrayList<>());
        }


        // Additional purchase-specific validations can be added here
        // For example, merchant category code validation, and DE61 validation etc.
        String mcc = transactionData.get(18);
        String De61 = transactionData.get(61);
        RuPayDe61Parser.RuPayDE61Data de61data = RuPayDe61Parser.parseDE61(De61);
        if(de61data.hasErrors()){
            errors.addAll(de61data.getErrors());
            return new ValidationResult(errors, new ArrayList<>()); 
        }
        log.info("De61data: {}", de61data.getSubfieldValue(RuPayDE61Subfields.CARD_DATA_INPUT_CAPABILITY));
        Enum cardDataInputCapability = de61data.getCardDataInputCapability();

        if(transactionType == TransactionType.PURCHASE){
            String sf1=de61data.getSubfieldValue(RuPayDE61Subfields.CARD_DATA_INPUT_CAPABILITY);
            if(!List.of("0","2","4","8").contains(sf1)){
                errors.add("INVALID_DE61: SF1 Expected 0,2,4,8");
                return new ValidationResult(errors, new ArrayList<>());
            }
            String sf5=de61data.getSubfieldValue(RuPayDE61Subfields.CARDHOLDER_PRESENT);
            if(!List.of("0","1").contains(sf5)){
                errors.add("INVALID_DE61: SF5 Expected 0,1");
                return new ValidationResult(errors, new ArrayList<>());
            }
            String sf6=de61data.getSubfieldValue(RuPayDE61Subfields.CARD_PRESENT);
            if(!List.of("0","2").contains(sf6)){
                errors.add("INVALID_DE61: SF6 Expected 0,2");
                return new ValidationResult(errors, new ArrayList<>());
            }
            String sf7=de61data.getSubfieldValue(RuPayDE61Subfields.CARD_DATA_INPUT_MODE);
            if(!List.of("0","3").contains(sf7)){
                errors.add("INVALID_DE61: SF7 Expected 0,3");
                return new ValidationResult(errors, new ArrayList<>());
            }
            String sf8=de61data.getSubfieldValue(RuPayDE61Subfields.CARDHOLDER_AUTH_METHOD);
            if(!List.of("0","2").contains(sf8)){
                errors.add("INVALID_DE61: SF8 Expected 0,2");
                return new ValidationResult(errors, new ArrayList<>());
            }
            
        } else if(transactionType == TransactionType.PURCHASE_CONTACTLESS){
            // Check if the transaction amount is greater than 5000.00
            String amountStr = transactionData.get(4); // DE4: Transaction amount (usually in smallest currency unit, e.g., paise)
            if (amountStr != null && !amountStr.isEmpty()) {
                try {
                    
                    Double amount = Double.parseDouble(amountStr)/100.0;
                    log.info("Txn Amount: {}", amount);
                    // Assuming amount is in paise (INR), so 5000.00 INR = 500000 paise
                    if (amount > 5000.00 && !transactionData.containsKey(52)) {
                        errors.add("INVALID_TRANSACTION: Transaction amount exceeds 5000.00 INR for contactless purchase with out PIN");
                        return new ValidationResult(errors, new ArrayList<>());
                    }
                } catch (NumberFormatException ex) {
                    errors.add("INVALID_AMOUNT: Unable to parse transaction amount");
                    return new ValidationResult(errors, new ArrayList<>());
                }
            } else {
                errors.add("MISSING_AMOUNT: Transaction amount (DE4) is required");
                return new ValidationResult(errors, new ArrayList<>());
            }

            String sf1=de61data.getSubfieldValue(RuPayDE61Subfields.CARD_DATA_INPUT_CAPABILITY);
            if(!List.of("0","7","8").contains(sf1)){
                errors.add("INVALID_DE61: SF1 Expected 0,7,8");
                return new ValidationResult(errors, new ArrayList<>());
            }
            String sf5=de61data.getSubfieldValue(RuPayDE61Subfields.CARDHOLDER_PRESENT);
            if(!List.of("0","1").contains(sf5)){
                errors.add("INVALID_DE61: SF5 Expected 0,1");
                return new ValidationResult(errors, new ArrayList<>());
            }
            String sf6=de61data.getSubfieldValue(RuPayDE61Subfields.CARD_PRESENT);
            if(!List.of("0","1","2").contains(sf6)){
                errors.add("INVALID_DE61: SF6 Expected 0,1,2");
                return new ValidationResult(errors, new ArrayList<>());
            }
            String sf7=de61data.getSubfieldValue(RuPayDE61Subfields.CARD_DATA_INPUT_MODE);
            if(!List.of("0","4").contains(sf7)){
                errors.add("INVALID_DE61: SF7 Expected 0,4");
                return new ValidationResult(errors, new ArrayList<>());
            }
            String sf8=de61data.getSubfieldValue(RuPayDE61Subfields.CARDHOLDER_AUTH_METHOD);
            if(!List.of("0","1","2","3").contains(sf8)){
                errors.add("INVALID_DE61: SF8 Expected 0,1,2");
                return new ValidationResult(errors, new ArrayList<>());
            }
           }

           return new ValidationResult(errors, new ArrayList<>());
            
          

        
    }
    private boolean purchaseWithCashback(Map<Integer, String> transactionData){
        String sf54=transactionData.get(54);
        if(sf54 == null || sf54.isEmpty()){
            return false;
        }else {
            // Parse DE54 field to extract cashback amount and validate format
            // DE54 format: 01-02: Account type, 03-04: Amount type, 05-07: Currency code, 08: Amount sign, 09-20: Amount
           
            long amount=Long.parseLong(transactionData.get(4));
            log.info("Other Amount:{}, length:{}", sf54,sf54.length());
            De54data.RuPayDe54Data de54data = De54data.parseDe54(sf54.trim());
            log.info("De54data: {}", de54data);

            long cashback=de54data.getAmount();
            if(!de54data.getAccountType().equals("90")||!de54data.getAmountType().equals("90") || de54data.getAmountSign() != 'D' || !de54data.getCurrencyCode().equals("356") || cashback >= amount){
                log.info("PURCHASE_WITHCASHBACK: Cashback is greater than amount: {} or invalid account type or amount type or amount sign or currency code", cashback);
                return false;
            }
        return true;
        }
    }
    
    @Override
    protected double getMaxAmountForTransactionType(TransactionType transactionType) {
        if (transactionType == TransactionType.PURCHASE) {
            return MAX_PURCHASE_AMOUNT;
        }
        return Double.MAX_VALUE;
    }

    private boolean isValidRuPayPreAuthMcc(String mcc) {
        if (mcc == null || mcc.isEmpty()) {
            return false;
        }

        if (Map.of(
                "7011", "Hotel Bookings",
                "8062", "Hospitals",
                "4112", "Passenger Railways",
                "4121", "Cab Booking"
        ).containsKey(mcc)) {
            return true;
        }

        try {
            int mccValue = Integer.parseInt(mcc);
            return mccValue >= 3351 && mccValue <= 3441;
        } catch (NumberFormatException ex) {
            return false;
        }
    }
    
    @Override
    public String getValidatorType() {
        return "PURCHASE_TRANSACTION_VALIDATOR";
    }
}
