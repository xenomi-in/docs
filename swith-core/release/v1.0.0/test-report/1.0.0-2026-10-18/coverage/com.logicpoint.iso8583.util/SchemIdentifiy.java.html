package com.logicpoint.iso8583.util;

public class SchemIdentifiy {
    
    private String identifyNetwork(String pan) {
        if (pan == null || pan.length() < 6) {
            return "UNKNOWN";
        }

        String bin = pan.substring(0, 6);
        // RuPay BIN ranges
        if (bin.startsWith("60") || bin.startsWith("65") || 
            bin.startsWith("81") || bin.startsWith("82") || 
            bin.startsWith("508") || bin.startsWith("353") || 
            bin.startsWith("356")) {
            return "RUPAY";
        }
        // Visa BIN ranges
        else if (bin.startsWith("4")) {
            return "VISA";
        }
        // Mastercard BIN ranges
        else if (bin.startsWith("51") || bin.startsWith("52") || 
                 bin.startsWith("53") || bin.startsWith("54") || 
                 bin.startsWith("55") || bin.startsWith("2221") || 
                 bin.startsWith("2222") || bin.startsWith("2223") || 
                 bin.startsWith("2224") || bin.startsWith("2225") || 
                 bin.startsWith("2226") || bin.startsWith("2227") || 
                 bin.startsWith("2228") || bin.startsWith("2229") || 
                 bin.startsWith("223") || bin.startsWith("224") || 
                 bin.startsWith("225") || bin.startsWith("226") || 
                 bin.startsWith("227") || bin.startsWith("228") || 
                 bin.startsWith("229") || bin.startsWith("23") || 
                 bin.startsWith("24") || bin.startsWith("25") || 
                 bin.startsWith("26") || bin.startsWith("27")) {
            return "MASTERCARD";
        }
        return "UNKNOWN";
    }
}
