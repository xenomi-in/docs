package com.logicpoint.iso8583.util;

import java.util.HashMap;
import java.util.Map;
import java.util.regex.Matcher;
import java.util.regex.Pattern;

/**
 * Utility class for updating EMV data elements.
 * Provides methods to update specific tags in EMV data (DE55).
 * 
 * @author LogicPoint Systems
 * @version 1.0
 * @since 2025-01-27
 */
public class EmvDataUpdater {
    
    private static final Pattern TAG_PATTERN = Pattern.compile("([0-9A-F]{2})([0-9A-F]{2})([0-9A-F]*)");
    
    /**
     * Updates a specific tag in EMV data.
     * 
     * @param emvData The original EMV data (DE55)
     * @param tag The tag to update (e.g., "91" for ARPC)
     * @param newValue The new value for the tag
     * @return Updated EMV data with the new tag value
     */
    public static String updateTag(String emvData, String tag, String newValue) {
        if (emvData == null || tag == null || newValue == null) {
            return emvData;
        }
        
        // Parse EMV data into tags
        Map<String, String> tags = parseEmvData(emvData);
        
        // Update the specific tag
        tags.put(tag, newValue);
        
        // Reconstruct EMV data
        return reconstructEmvData(tags);
    }
    
    /**
     * Updates tag 91 (ARPC) in EMV data.
     * 
     * @param emvData The original EMV data (DE55)
     * @param arpcValue The new ARPC value
     * @return Updated EMV data with new ARPC value
     */
    public static String updateArpc(String emvData, String arpcValue) {
        return updateTag(emvData, "91", arpcValue);
    }
    
    /**
     * Parses EMV data into a map of tag-value pairs.
     * 
     * @param emvData The EMV data to parse
     * @return Map of tag-value pairs
     */
    public static Map<String, String> parseEmvData(String emvData) {
        Map<String, String> tags = new HashMap<>();
        
        if (emvData == null || emvData.isEmpty()) {
            return tags;
        }
        
        Matcher matcher = TAG_PATTERN.matcher(emvData);
        while (matcher.find()) {
            String tag = matcher.group(1);
            String length = matcher.group(2);
            String value = matcher.group(3);
            
            if (value != null && value.length() > 0) {
                tags.put(tag, value);
            }
        }
        
        return tags;
    }
    
    /**
     * Reconstructs EMV data from a map of tag-value pairs.
     * 
     * @param tags Map of tag-value pairs
     * @return Reconstructed EMV data
     */
    public static String reconstructEmvData(Map<String, String> tags) {
        if (tags == null || tags.isEmpty()) {
            return "";
        }
        
        StringBuilder emvData = new StringBuilder();
        
        for (Map.Entry<String, String> entry : tags.entrySet()) {
            String tag = entry.getKey();
            String value = entry.getValue();
            
            if (value != null && !value.isEmpty()) {
                String length = String.format("%02X", value.length());
                emvData.append(tag).append(length).append(value);
            }
        }
        
        return emvData.toString();
    }
    
    /**
     * Gets a specific tag value from EMV data.
     * 
     * @param emvData The EMV data to search
     * @param tag The tag to find
     * @return The tag value or null if not found
     */
    public static String getTagValue(String emvData, String tag) {
        if (emvData == null || tag == null) {
            return null;
        }
        
        Map<String, String> tags = parseEmvData(emvData);
        return tags.get(tag);
    }
    
    /**
     * Checks if a specific tag exists in EMV data.
     * 
     * @param emvData The EMV data to check
     * @param tag The tag to look for
     * @return true if the tag exists, false otherwise
     */
    public static boolean hasTag(String emvData, String tag) {
        if (emvData == null || tag == null) {
            return false;
        }
        
        Map<String, String> tags = parseEmvData(emvData);
        return tags.containsKey(tag);
    }
}
