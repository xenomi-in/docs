package com.logicpoint.iso8583.util;

import io.swagger.v3.oas.annotations.media.Schema;

@Schema(description = "ISO 8583 Card Entry Modes")
public enum EntryMode {
    @Schema(description = "Manual Entry")
    MANUAL("01", "Manual Entry"),
    
    @Schema(description = "Magnetic Stripe Read")
    MAGNETIC_STRIPE("02", "Magnetic Stripe Read"),
    
    @Schema(description = "Bar Code")
    BAR_CODE("03", "Bar Code"),
    
    @Schema(description = "OCR")
    OCR("04", "OCR"),
    
    @Schema(description = "ICC with Chip")
    ICC("05", "ICC with Chip"),
    
    @Schema(description = "Contactless ICC")
    CONTACTLESS_ICC("07", "Contactless ICC"),
    
    @Schema(description = "Contactless Magnetic Stripe")
    CONTACTLESS_MSR("91", "Contactless Magnetic Stripe"),
    
    @Schema(description = "QR Code")
    QR_CODE("92", "QR Code"),
    
    @Schema(description = "NFC")
    NFC("93", "NFC"),
    
    @Schema(description = "UPI")
    UPI("94", "UPI"),
    
    @Schema(description = "Unknown Entry Mode")
    UNKNOWN("99", "Unknown Entry Mode");

    private final String code;
    private final String description;

    EntryMode(String code, String description) {
        this.code = code;
        this.description = description;
    }

    @Schema(description = "Entry Mode Code")
    public String getCode() {
        return code;
    }

    @Schema(description = "Entry Mode Description")
    public String getDescription() {
        return description;
    }

    public static EntryMode fromCode(String code) {
        if (code == null) {
            return UNKNOWN;
        }
        
        for (EntryMode mode : values()) {
            if (mode.getCode().equals(code)) {
                return mode;
            }
        }
        return UNKNOWN;
    }

    public boolean isContactless() {
        return this == CONTACTLESS_ICC || this == CONTACTLESS_MSR || this == NFC;
    }

    public boolean isChip() {
        return this == ICC || this == CONTACTLESS_ICC;
    }

    public boolean isMagneticStripe() {
        return this == MAGNETIC_STRIPE || this == CONTACTLESS_MSR;
    }

    public boolean isManual() {
        return this == MANUAL;
    }

    public boolean isDigital() {
        return this == QR_CODE || this == UPI;
    }
}
