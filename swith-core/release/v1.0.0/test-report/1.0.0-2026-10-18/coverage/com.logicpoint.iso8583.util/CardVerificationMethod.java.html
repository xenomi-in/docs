package com.logicpoint.iso8583.util;

import io.swagger.v3.oas.annotations.media.Schema;

@Schema(description = "ISO 8583 Card Verification Methods")
public enum CardVerificationMethod {
    @Schema(description = "No CVM Required")
    NO_CVM("0", "No CVM Required"),
    
    @Schema(description = "Signature")
    SIGNATURE("1", "Signature"),
    
    @Schema(description = "Online PIN")
    ONLINE_PIN("2", "Online PIN"),
    
    @Schema(description = "Offline PIN")
    OFFLINE_PIN("3", "Offline PIN"),
    
    @Schema(description = "Online PIN and Signature")
    ONLINE_PIN_SIGNATURE("4", "Online PIN and Signature"),
    
    @Schema(description = "Offline PIN and Signature")
    OFFLINE_PIN_SIGNATURE("5", "Offline PIN and Signature"),
    
    @Schema(description = "Biometric")
    BIOMETRIC("6", "Biometric"),
    
    @Schema(description = "Biometric and PIN")
    BIOMETRIC_PIN("7", "Biometric and PIN"),
    
    @Schema(description = "Biometric and Signature")
    BIOMETRIC_SIGNATURE("8", "Biometric and Signature"),
    
    @Schema(description = "No CVM Required (Contactless)")
    NO_CVM_CONTACTLESS("9", "No CVM Required (Contactless)"),
    
    @Schema(description = "Unknown CVM")
    UNKNOWN("99", "Unknown CVM");

    private final String code;
    private final String description;

    CardVerificationMethod(String code, String description) {
        this.code = code;
        this.description = description;
    }

    @Schema(description = "CVM Code")
    public String getCode() {
        return code;
    }

    @Schema(description = "CVM Description")
    public String getDescription() {
        return description;
    }

    public static CardVerificationMethod fromCode(String code) {
        if (code == null) {
            return UNKNOWN;
        }
        
        for (CardVerificationMethod cvm : values()) {
            if (cvm.getCode().equals(code)) {
                return cvm;
            }
        }
        return UNKNOWN;
    }

    public boolean requiresPin() {
        return this == ONLINE_PIN || this == OFFLINE_PIN || 
               this == ONLINE_PIN_SIGNATURE || this == OFFLINE_PIN_SIGNATURE ||
               this == BIOMETRIC_PIN;
    }

    public boolean requiresSignature() {
        return this == SIGNATURE || this == ONLINE_PIN_SIGNATURE || 
               this == OFFLINE_PIN_SIGNATURE || this == BIOMETRIC_SIGNATURE;
    }

    public boolean requiresBiometric() {
        return this == BIOMETRIC || this == BIOMETRIC_PIN || 
               this == BIOMETRIC_SIGNATURE;
    }

    public boolean isOnline() {
        return this == ONLINE_PIN || this == ONLINE_PIN_SIGNATURE;
    }

    public boolean isOffline() {
        return this == OFFLINE_PIN || this == OFFLINE_PIN_SIGNATURE;
    }

    public boolean isNoCVM() {
        return this == NO_CVM || this == NO_CVM_CONTACTLESS;
    }
}
