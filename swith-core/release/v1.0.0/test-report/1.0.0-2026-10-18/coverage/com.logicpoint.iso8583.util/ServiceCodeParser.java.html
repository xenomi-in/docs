package com.logicpoint.iso8583.util;

import com.logicpoint.iso8583.util.Scheme;

/**
 * @author LogicPoint Systems
 * @version 1.0
* @since 2025-07-03
 * Utility for parsing and interpreting service codes.
 * Service codes are 3-digit codes that indicate card capabilities.
 */
public class ServiceCodeParser {
    
    /**
     * Parses service code and returns interpretation.
     */
    public static ServiceCodeInfo parseServiceCode(String serviceCode, Scheme scheme) {
        if (serviceCode == null || serviceCode.length() != 3) {
            throw new IllegalArgumentException("Service code must be 3 digits");
        }
        
        char firstDigit = serviceCode.charAt(0);
        char secondDigit = serviceCode.charAt(1);
        char thirdDigit = serviceCode.charAt(2);
        
        return new ServiceCodeInfo(
            serviceCode,
            scheme,
            interpretInterchangeQualifier(firstDigit),
            interpretAuthorizationProcessing(secondDigit),
            interpretRangeOfServices(thirdDigit, scheme)
        );
    }
    
    /**
     * Interprets interchange qualifier (first digit).
     */
    private static String interpretInterchangeQualifier(char digit) {
        switch (digit) {
            case '1': return "INTERNATIONAL_IC";
            case '2': return "INTERNATIONAL_MAGSTRIPE";
            case '5': return "NATIONAL_IC";
            case '6': return "NATIONAL_MAGSTRIPE";
            case '7': return "PRIVATE_NETWORK";
            case '9': return "TEST";
            default: return "UNKNOWN";
        }
    }
    
    /**
     * Interprets authorization processing (second digit).
     */
    private static String interpretAuthorizationProcessing(char digit) {
        switch (digit) {
            case '0': return "NORMAL";
            case '2': return "BY_ISSUER";
            case '4': return "BY_ISSUER_UNLESS_EXPLICIT_APPROVAL";
            case '6': return "BY_ISSUER_EXCEPT_UNDER_CONDITIONS";
            default: return "UNKNOWN";
        }
    }
    
    /**
     * Interprets range of services (third digit).
     */
    private static String interpretRangeOfServices(char digit, Scheme scheme) {
        switch (scheme) {
            case RUPAY:
                return interpretRuPayRangeOfServices(digit);
            case VISA:
                return interpretVisaRangeOfServices(digit);
            case MASTERCARD:
                return interpretMastercardRangeOfServices(digit);
            default:
                return "UNKNOWN";
        }
    }
    
    /**
     * Interprets RuPay range of services.
     */
    private static String interpretRuPayRangeOfServices(char digit) {
        switch (digit) {
            case '0': return "NO_RESTRICTIONS";
            case '1': return "NO_CASHBACK";
            case '2': return "CASHBACK_ALLOWED";
            case '3': return "GOODS_AND_SERVICES_ONLY";
            case '4': return "ATM_ONLY";
            case '5': return "CASH_ONLY";
            case '6': return "GOODS_AND_SERVICES_CASHBACK";
            case '7': return "GOODS_AND_SERVICES_CASHBACK_ATM";
            default: return "UNKNOWN";
        }
    }
    
    /**
     * Interprets Visa range of services.
     */
    private static String interpretVisaRangeOfServices(char digit) {
        switch (digit) {
            case '0': return "NO_RESTRICTIONS";
            case '1': return "NO_CASHBACK";
            case '2': return "CASHBACK_ALLOWED";
            case '3': return "GOODS_AND_SERVICES_ONLY";
            case '4': return "ATM_ONLY";
            case '5': return "CASH_ONLY";
            case '6': return "GOODS_AND_SERVICES_CASHBACK";
            case '7': return "GOODS_AND_SERVICES_CASHBACK_ATM";
            default: return "UNKNOWN";
        }
    }
    
    /**
     * Interprets Mastercard range of services.
     */
    private static String interpretMastercardRangeOfServices(char digit) {
        switch (digit) {
            case '0': return "NO_RESTRICTIONS";
            case '1': return "NO_CASHBACK";
            case '2': return "CASHBACK_ALLOWED";
            case '3': return "GOODS_AND_SERVICES_ONLY";
            case '4': return "ATM_ONLY";
            case '5': return "CASH_ONLY";
            case '6': return "GOODS_AND_SERVICES_CASHBACK";
            case '7': return "GOODS_AND_SERVICES_CASHBACK_ATM";
            default: return "UNKNOWN";
        }
    }
    
    /**
     * Service code information class.
     */
    public static class ServiceCodeInfo {
        private final String serviceCode;
        private final Scheme scheme;
        private final String interchangeQualifier;
        private final String authorizationProcessing;
        private final String rangeOfServices;
        
        public ServiceCodeInfo(String serviceCode, Scheme scheme, String interchangeQualifier, 
                             String authorizationProcessing, String rangeOfServices) {
            this.serviceCode = serviceCode;
            this.scheme = scheme;
            this.interchangeQualifier = interchangeQualifier;
            this.authorizationProcessing = authorizationProcessing;
            this.rangeOfServices = rangeOfServices;
        }
        
        public String getServiceCode() { return serviceCode; }
        public Scheme getScheme() { return scheme; }
        public String getInterchangeQualifier() { return interchangeQualifier; }
        public String getAuthorizationProcessing() { return authorizationProcessing; }
        public String getRangeOfServices() { return rangeOfServices; }
        
        @Override
        public String toString() {
            return String.format("ServiceCodeInfo{serviceCode='%s', scheme=%s, interchange='%s', auth='%s', range='%s'}", 
                serviceCode, scheme, interchangeQualifier, authorizationProcessing, rangeOfServices);
        }
    }
}
