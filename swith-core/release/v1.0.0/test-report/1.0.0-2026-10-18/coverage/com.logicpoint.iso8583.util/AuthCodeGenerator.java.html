package com.logicpoint.iso8583.util;
import java.security.SecureRandom;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

public class AuthCodeGenerator {
    private static final Logger log = LoggerFactory.getLogger(AuthCodeGenerator.class);
    private static final char[] CHARSET = "ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789".toCharArray();
    private static final int LENGTH = 6;
    private static final SecureRandom RANDOM = new SecureRandom();

    public static String generateAuthCode() {
        char[] result = new char[LENGTH];
        for (int i = 0; i < LENGTH; i++) {
            result[i] = CHARSET[RANDOM.nextInt(CHARSET.length)];
        }
        log.debug("Generated DE 38 Code: {}", new String(result));
        return new String(result);
    }

    public static void main(String[] args) {
        System.out.println("Generated DE 38 Code: " + generateAuthCode());
    }
}
