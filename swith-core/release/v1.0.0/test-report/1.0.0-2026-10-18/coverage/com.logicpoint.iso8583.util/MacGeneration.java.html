package com.logicpoint.iso8583.util;

import javax.crypto.Cipher;
import javax.crypto.SecretKey;
import javax.crypto.SecretKeyFactory;
import javax.crypto.spec.DESKeySpec;
import javax.crypto.spec.IvParameterSpec;
import java.util.Arrays;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import com.logicpoint.iso8583.config.RawDataLogging;

/*
 * This class is used to generate the MAC value for the Rupay transaction.
 * It uses the retailMac method to generate the MAC value.
 * It uses the hexStringToByteArray method to convert the hex string to byte array.
 * It uses the toHex method to convert the byte array to hex string.
 * It uses the retailMac method to generate the MAC value.
 * It uses the addPadding method to add the padding to the data.
 * Author: LogicPoint Systems
 * Version: 1.0
 * Date: 2025-07-16
 * 
 */
public class MacGeneration {

    private static final Logger log = LoggerFactory.getLogger(MacGeneration.class);

    public static void main(String[] args) {
        try {

            String mac = generateMACvalue("3EBF5B2F98FD3886BA4F34857504A7C8", "0110817319312087747400000000000000010042097016460052421642097030");
            //0110817319312087747400000000000000010004096015252352421504096030
            //01108173193120877474000000000000000100311440163718524216311440VQLZWH00
            System.out.println("MAC value: " + mac);

        } catch (Exception e) {
            e.getStackTrace();
        }
    }

    public static String generateMACvalue(String key, String inputText) {
        try {

            byte[] keyBytes = hexStringToByteArray(key);
            byte[] texToBytes = inputText.getBytes("utf-8");

            byte[] allBytes = retailMac(keyBytes, texToBytes);
            byte[] firmaBytes = Arrays.copyOf(allBytes, 8);
            String firma = toHex(firmaBytes);
            return firma.toUpperCase();
        } catch (Exception e) {
            log.error("Error generating MAC value", e);
            return null;
        }
    }

    private static byte[] hexStringToByteArray(String hex) {
        int len = hex.length();
        byte[] data = new byte[len / 2];
        for (int i = 0; i < len; i += 2) {
            data[i / 2] = (byte) ((Character.digit(hex.charAt(i), 16) << 4) + Character
                    .digit(hex.charAt(i + 1), 16));
        }
        return data;
    }

    final private static String toHex(byte[] bytes) {
        StringBuilder sb = new StringBuilder(bytes.length * 2);
        for (byte b : bytes) {
            sb.append(String.format("%02x", b & 0xff));
        }
        return sb.toString();
    }

    private static byte[] retailMac(byte[] key, byte[] data) {
        try {
            // Create Keys
            byte[] key1 = Arrays.copyOf(key, 8);
            byte[] key2 = Arrays.copyOfRange(key, 8, 16);

            // ISO/IEC 9797-1 or ISO 7816d4 Padding for data (adding 80 00 ..)
            byte[] pdata = null;
            if (data.length % 8 != 0) {
                pdata = addPadding(data);
                RawDataLogging.value("Padded MAC data in hex", toHex(pdata));
            } else {
                pdata = data;
            }

            SecretKeyFactory mySecretKeyFactory = SecretKeyFactory.getInstance("DES");

            DESKeySpec myKeySpec1 = new DESKeySpec(key1);
            SecretKey myKey1 = mySecretKeyFactory.generateSecret(myKeySpec1);
            Cipher cipher1 = Cipher.getInstance("DES/CBC/NoPadding");
            cipher1.init(Cipher.ENCRYPT_MODE, myKey1, new IvParameterSpec(new byte[8]));

            DESKeySpec myKeySpec2 = new DESKeySpec(key2);
            SecretKey myKey2 = mySecretKeyFactory.generateSecret(myKeySpec2);
            Cipher cipher2 = Cipher.getInstance("DES/CBC/NoPadding");
            cipher2.init(Cipher.DECRYPT_MODE, myKey2, new IvParameterSpec(new byte[8]));

            byte[] result = cipher1.doFinal(pdata);

            byte[] block = Arrays.copyOfRange(result, result.length - 8, result.length);

            // Decrypt the resulting block with Key-2
            block = cipher2.doFinal(block);

            // Encrypt the resulting block with Key-1
            block = cipher1.doFinal(block);

            return block;
        } catch (Exception e) {
            log.error("Error generating MAC value", e);
            return null;
        }
    }

    private static byte[] addPadding(byte[] in) {
        int extra = 8 - (in.length % 8);
        int newLength = in.length + extra;
        log.debug("Padding data length: {}", in.length + " to " + newLength);
        byte[] out = Arrays.copyOf(in, newLength);
        int offset = in.length;
//        out[offset] = (byte) 0x80;
//        offset++;
        while (offset < newLength) {
            out[offset] = (byte) 0;
            offset++;
        }
        return out;
    }

}
