package com.logicpoint.iso8583.util;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import java.util.HashMap;
import java.util.Map;

public class Iso8583MessageHandler {
    
    private static final Logger log = LoggerFactory.getLogger(Iso8583MessageHandler.class);
    
    public static Map<Integer, String> parseIso8583Message(String message) {
        Map<Integer, String> fields = new HashMap<>();
        try {
            // Basic ISO 8583 message parsing
            // MTI is first 4 characters
            String mti = message.substring(0, 4);
            fields.put(0, mti);
            
            // Parse bitmap and fields
            int currentPos = 4;
            String primaryBitmap = message.substring(currentPos, currentPos + 16);
            currentPos += 16;
            
            // Convert bitmap to binary and check which fields are present
            String binaryBitmap = hexToBinary(primaryBitmap);
            for (int i = 0; i < binaryBitmap.length(); i++) {
                if (binaryBitmap.charAt(i) == '1') {
                    int fieldNumber = i + 1;
                    // Add field parsing logic here based on field type
                    // This is a simplified version
                    fields.put(fieldNumber, "FIELD_" + fieldNumber);
                }
            }
        } catch (Exception e) {
            log.error("Error parsing ISO 8583 message", e);
            throw new RuntimeException("Failed to parse ISO 8583 message", e);
        }
        return fields;
    }
    
    public static String formatIso8583Message(Map<Integer, String> fields) {
        StringBuilder message = new StringBuilder();
        try {
            // Add MTI
            message.append(fields.getOrDefault(0, "0200"));
            
            // Create bitmap and add fields
            boolean[] bitmap = new boolean[64];
            for (Map.Entry<Integer, String> entry : fields.entrySet()) {
                if (entry.getKey() > 0 && entry.getKey() <= 64) {
                    bitmap[entry.getKey() - 1] = true;
                }
            }
            
            // Convert bitmap to hex
            String bitmapHex = binaryToHex(bitmap);
            message.append(bitmapHex);
            
            // Add fields
            for (int i = 0; i < bitmap.length; i++) {
                if (bitmap[i]) {
                    message.append(fields.get(i + 1));
                }
            }
        } catch (Exception e) {
            log.error("Error formatting ISO 8583 message", e);
            throw new RuntimeException("Failed to format ISO 8583 message", e);
        }
        return message.toString();
    }
    
    private static String hexToBinary(String hex) {
        StringBuilder binary = new StringBuilder();
        for (char c : hex.toCharArray()) {
            String binaryChar = Integer.toBinaryString(Integer.parseInt(String.valueOf(c), 16));
            binary.append(String.format("%4s", binaryChar).replace(' ', '0'));
        }
        return binary.toString();
    }
    
    private static String binaryToHex(boolean[] bitmap) {
        StringBuilder hex = new StringBuilder();
        for (int i = 0; i < bitmap.length; i += 4) {
            int value = 0;
            for (int j = 0; j < 4 && (i + j) < bitmap.length; j++) {
                if (bitmap[i + j]) {
                    value |= (1 << (3 - j));
                }
            }
            hex.append(Integer.toHexString(value).toUpperCase());
        }
        return hex.toString();
    }
}
