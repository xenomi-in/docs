package com.logicpoint.iso8583.validation.context;

import java.util.Map;
import java.util.HashMap;
import java.util.concurrent.ConcurrentHashMap;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import com.logicpoint.iso8583.validation.TlvUtils;

/**
 * Context object to hold parsed DE48 data and avoid multiple parsing
 */
public class De48Context {
    private static final Logger logger = LoggerFactory.getLogger(De48Context.class);
    
    private final String originalDe48;
    private final Map<String, String> tlvMap;
    private final boolean isValid;
    private final String parseError;
    private volatile boolean modified = false;
    
    private De48Context(String originalDe48, Map<String, String> tlvMap, boolean isValid, String parseError) {
        this.originalDe48 = originalDe48;
        this.tlvMap = tlvMap != null ? new ConcurrentHashMap<>(tlvMap) : new ConcurrentHashMap<>();
        this.isValid = isValid;
        this.parseError = parseError;
    }
    
    /**
     * Factory method to parse DE48 once and create context
     */
    public static De48Context parse(String de48) {
        if (de48 == null || de48.trim().isEmpty()) {
            return new De48Context(de48, null, false, "DE48 is null or empty");
        }
        
        try {
            Map<String, String> tlvMap = TlvUtils.parseTlv(de48);
            logger.debug("DE48 parsed successfully with {} tags", tlvMap.size());
            return new De48Context(de48, tlvMap, true, null);
        } catch (Exception e) {
            logger.error("Failed to parse DE48: {}", e.getMessage());
            return new De48Context(de48, null, false, e.getMessage());
        }
    }
    
    // Getters
    public String getOriginalDe48() { return originalDe48; }
    public Map<String, String> getTlvMap() { return tlvMap; }
    public boolean isValid() { return isValid; }
    public String getParseError() { return parseError; }
    public boolean isModified() { return modified; }
    
    // Tag access methods
    public String getTag(String tagId) {
        return isValid ? tlvMap.get(tagId) : null;
    }
    
    public boolean hasTag(String tagId) {
        return isValid && tlvMap.containsKey(tagId);
    }
    
    public void setTag(String tagId, String value) {
        if (isValid) {
            tlvMap.put(tagId, value);
            modified = true;
            logger.debug("Tag {} set to: {}", tagId, value);
        }
    }
    
    public void removeTag(String tagId) {
        if (isValid && tlvMap.remove(tagId) != null) {
            modified = true;
            logger.debug("Tag {} removed", tagId);
        }
    }
    
    /**
     * Rebuilds DE48 string if modified
     */
    public String buildDe48() {
        if (!isValid) {
            return originalDe48;
        }
        
        if (!modified) {
            return originalDe48;
        }
        
        StringBuilder sb = new StringBuilder();
        for (Map.Entry<String, String> entry : tlvMap.entrySet()) {
            String tag = entry.getKey();
            String value = entry.getValue();
            String length = String.format("%03d", value.length());
            sb.append(tag).append(length).append(value);
        }
        
        logger.debug("DE48 rebuilt with {} tags", tlvMap.size());
        return sb.toString();
    }
}
