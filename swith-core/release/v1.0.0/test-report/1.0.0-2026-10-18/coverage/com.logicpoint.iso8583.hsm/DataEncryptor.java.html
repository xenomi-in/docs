package com.logicpoint.iso8583.hsm;

import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Component;
import javax.crypto.Cipher;
import javax.crypto.SecretKey;
import java.util.Base64;

/**
 * Utility for encrypting and decrypting sensitive data.
 * Author: @logicpoint  
 * Date: 2025-07-13
 * Version: 1.0.0
 * Description: This class is used to encrypt and decrypt sensitive data.
 * 
 * 
 * 
 * 
 * 
 */
@Component
public class DataEncryptor {

  

    /**
     * Encrypts the given plaintext.
     *
     * @param plaintext Data to encrypt.
     * @return Base64-encoded ciphertext.
     * @throws Exception If encryption fails.
     */
    public String encrypt(String plaintext) throws Exception {
        SecretKey key = null;//keyManager.getEncryptionKey();
        Cipher cipher = Cipher.getInstance("AES/ECB/PKCS5Padding"); // Or GCM for better security
        cipher.init(Cipher.ENCRYPT_MODE, key);
        byte[] encrypted = cipher.doFinal(plaintext.getBytes("UTF-8"));
        return Base64.getEncoder().encodeToString(encrypted);
    }

    /**
     * Decrypts the given ciphertext.
     *
     * @param ciphertext Base64-encoded encrypted data.
     * @return Decrypted plaintext.
     * @throws Exception If decryption fails.
     */
    public String decrypt(String ciphertext) throws Exception {
        SecretKey key = null; //keyManager.getEncryptionKey();
        Cipher cipher = Cipher.getInstance("AES/ECB/PKCS5Padding");
        cipher.init(Cipher.DECRYPT_MODE, key);
        byte[] decoded = Base64.getDecoder().decode(ciphertext);
        byte[] decrypted = cipher.doFinal(decoded);
        return new String(decrypted, "UTF-8");
    }
}
