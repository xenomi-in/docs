package com.logicpoint.iso8583.hsm;

import io.netty.bootstrap.Bootstrap;
import io.netty.buffer.ByteBuf;
import io.netty.buffer.Unpooled;
import io.netty.channel.*;
import io.netty.channel.nio.NioEventLoopGroup;
import io.netty.channel.socket.SocketChannel;
import io.netty.channel.socket.nio.NioSocketChannel;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.stereotype.Component;

import java.util.concurrent.*;
import java.util.concurrent.atomic.AtomicInteger;
import java.util.concurrent.RejectedExecutionException;
import java.util.concurrent.atomic.AtomicBoolean;
import java.util.concurrent.ScheduledFuture;

/**
 * Simplified Netty-based HSM client without SSL for testing.
 * Provides async communication without SSL/TLS for development.
 * 
 * @author LogicPoint Systems
 * @version 1.0
 */

public class SimpleNettyHsmClientNoSsl {
    
    private static final Logger logger = LoggerFactory.getLogger(SimpleNettyHsmClientNoSsl.class);
    private static final int DEFAULT_MAX_CONNECTIONS = 10;
    private static final int DEFAULT_TIMEOUT = 30000;
    
    private final String host;
    private final int port;
    private final int maxConnections;
    private final int timeout;
    
    private EventLoopGroup eventLoopGroup;
    private Bootstrap bootstrap;
    private final ConcurrentLinkedQueue<Channel> connectionPool = new ConcurrentLinkedQueue<>();
    private final AtomicInteger activeConnections = new AtomicInteger(0);
    private final Semaphore connectionSemaphore;
    private volatile boolean isShutdown = false;
    
    // Add connection state tracking
    private static final io.netty.util.AttributeKey<AtomicBoolean> CONNECTION_RETURNED = 
        io.netty.util.AttributeKey.valueOf("connectionReturned");
    
    /**
     * Constructs a SimpleNettyHsmClientNoSsl with default settings.
     */
    public SimpleNettyHsmClientNoSsl(String host, int port) {
        this(host, port, DEFAULT_MAX_CONNECTIONS, DEFAULT_TIMEOUT);
    }
    
    /**
     * Constructs a SimpleNettyHsmClientNoSsl with custom settings.
     */
    public SimpleNettyHsmClientNoSsl(String host, int port, int maxConnections, int timeout) {
        this.host = host;
        this.port = port;
        this.maxConnections = maxConnections;
        this.timeout = timeout;
        this.connectionSemaphore = new Semaphore(maxConnections);
        
        initializeNetty();
    }
    
    /**
     * Initializes Netty components without SSL.
     */
    private void initializeNetty() {
        try {
            eventLoopGroup = new NioEventLoopGroup();
            
            // Create bootstrap without SSL
            bootstrap = new Bootstrap();
            bootstrap.group(eventLoopGroup)
                    .channel(NioSocketChannel.class)
                    .option(ChannelOption.CONNECT_TIMEOUT_MILLIS, timeout)
                    .option(ChannelOption.SO_KEEPALIVE, true)
                    .option(ChannelOption.TCP_NODELAY, true)
                    .handler(new ChannelInitializer<SocketChannel>() {
                        @Override
                        protected void initChannel(SocketChannel ch) {
                            ChannelPipeline pipeline = ch.pipeline();
                            
                            // Add simple handler for echo protocol (no SSL)
                            pipeline.addLast(new SimpleHsmHandler());
                        }
                    });
            
            logger.info("Simple Netty HSM client (no SSL) initialized for {}:{}", host, port);
            
        } catch (Exception e) {
            logger.error("Failed to initialize Netty client: {}", e.getMessage());
            throw new RuntimeException("Failed to initialize Netty client", e);
        }
    }
    
    /**
     * Gets a connection from the pool or creates a new one.
     */
    private CompletableFuture<Channel> getConnection() {
        CompletableFuture<Channel> future = new CompletableFuture<>();
        
        // Check if client is shutdown
        if (isShutdown) {
            future.completeExceptionally(new IllegalStateException("Client is shutdown"));
            return future;
        }
        
        // Check if event loop group is available
        if (eventLoopGroup == null || eventLoopGroup.isShutdown()) {
            future.completeExceptionally(new IllegalStateException("EventLoopGroup is not available"));
            return future;
        }
        
        // Try to get from pool first
        Channel pooledChannel = connectionPool.poll();
        if (pooledChannel != null && pooledChannel.isActive()) {
            // Reset the connection state for reuse
            AtomicBoolean returned = pooledChannel.attr(CONNECTION_RETURNED).get();
            if (returned != null) {
                returned.set(false);
            }
            logger.debug("Reusing pooled connection");
            future.complete(pooledChannel);
            return future;
        }
        
        // Create new connection if pool is empty
        if (activeConnections.get() < maxConnections) {
            createNewConnection(future);
        } else {
            // Wait for available connection
            try {
                eventLoopGroup.schedule(() -> {
                    if (!future.isDone()) {
                        createNewConnection(future);
                    }
                }, 100, TimeUnit.MILLISECONDS);
            } catch (RejectedExecutionException e) {
                future.completeExceptionally(new IllegalStateException("Executor not accepting tasks", e));
            }
        }
        
        return future;
    }
    
    /**
     * Creates a new connection.
     */
    private void createNewConnection(CompletableFuture<Channel> future) {
        try {
            connectionSemaphore.acquire();
            activeConnections.incrementAndGet();
            
            bootstrap.connect(host, port).addListener((ChannelFutureListener) channelFuture -> {
                if (channelFuture.isSuccess()) {
                    Channel channel = channelFuture.channel();
                    logger.debug("Created new connection to {}:{}", host, port);
                    future.complete(channel);
                } else {
                    activeConnections.decrementAndGet();
                    connectionSemaphore.release();
                    future.completeExceptionally(channelFuture.cause());
                }
            });
            
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
            future.completeExceptionally(e);
        }
    }
    
    /**
     * Returns a connection to the pool.
     */
    private void returnConnection(Channel channel) {
        if (channel != null && channel.isActive()) {
            // Check if the connection has already been returned
            AtomicBoolean returned = channel.attr(CONNECTION_RETURNED).get();
            if (returned == null) {
                returned = new AtomicBoolean(false);
                channel.attr(CONNECTION_RETURNED).set(returned);
            }
            
            if (returned.compareAndSet(false, true)) {
                connectionPool.offer(channel);
                logger.debug("Returned connection to pool. Pool size: {}", connectionPool.size());
            } else {
                logger.debug("Connection {} already returned, skipping return.", channel.remoteAddress());
            }
        } else {
            activeConnections.decrementAndGet();
            connectionSemaphore.release();
            if (channel != null) {
                channel.close();
            }
        }
    }
    
    /**
     * Sends a message asynchronously.
     */
    public CompletableFuture<String> sendMessageAsync(String message) {
        CompletableFuture<String> responseFuture = new CompletableFuture<>();
        
        // Check if client is shutdown
        if (isShutdown) {
            responseFuture.completeExceptionally(new IllegalStateException("Client is shutdown"));
            return responseFuture;
        }
        
        // Check if event loop group is available
        if (eventLoopGroup == null || eventLoopGroup.isShutdown()) {
            responseFuture.completeExceptionally(new IllegalStateException("EventLoopGroup is not available"));
            return responseFuture;
        }
        
        getConnection().thenAccept(channel -> {
            // Send message
            ByteBuf messageBuf = Unpooled.copiedBuffer(message, java.nio.charset.StandardCharsets.UTF_8);
            channel.writeAndFlush(messageBuf).addListener((ChannelFutureListener) writeFuture -> {
                if (!writeFuture.isSuccess()) {
                    responseFuture.completeExceptionally(writeFuture.cause());
                    returnConnection(channel);
                }
            });
            
            // Set timeout
            channel.eventLoop().schedule(() -> {
                if (!responseFuture.isDone()) {
                    responseFuture.completeExceptionally(new TimeoutException("Response timeout"));
                    returnConnection(channel);
                }
            }, timeout, TimeUnit.MILLISECONDS);
            
        }).exceptionally(throwable -> {
            responseFuture.completeExceptionally(throwable);
            return null;
        });
        
        return responseFuture;
    }
    
    /**
     * Sends a message synchronously.
     */
    public String sendMessage(String message) throws Exception {
        try {
            return sendMessageAsync(message).get(timeout, TimeUnit.MILLISECONDS);
        } catch (TimeoutException e) {
            throw new java.io.IOException("Request timeout", e);
        } catch (ExecutionException e) {
            throw new java.io.IOException("Communication failed", e.getCause());
        }
    }
    
    /**
     * Gets connection pool statistics.
     */
    public String getPoolStats() {
        return String.format("Active: %d, Pooled: %d, Max: %d", 
                           activeConnections.get(), connectionPool.size(), maxConnections);
    }
    
    /**
     * Shuts down the client.
     */
    public void shutdown() {
        logger.info("Shutting down Simple Netty HSM client (no SSL)");
        isShutdown = true;
        
        // Close pooled connections
        Channel channel;
        while ((channel = connectionPool.poll()) != null) {
            channel.close();
        }
        
        // Shutdown event loop group gracefully
        if (eventLoopGroup != null && !eventLoopGroup.isShutdown()) {
            try {
                // Wait for graceful shutdown
                eventLoopGroup.shutdownGracefully(0, 5, TimeUnit.SECONDS).sync();
                logger.info("EventLoopGroup shutdown completed");
            } catch (InterruptedException e) {
                logger.warn("EventLoopGroup shutdown interrupted");
                Thread.currentThread().interrupt();
            }
        }
        
        logger.info("Simple Netty HSM client (no SSL) shutdown completed");
    }
    
    /**
     * Simple channel handler for HSM communication.
     */
    private class SimpleHsmHandler extends SimpleChannelInboundHandler<ByteBuf> {
        
        @Override
        protected void channelRead0(ChannelHandlerContext ctx, ByteBuf msg) {
            String response = msg.toString(java.nio.charset.StandardCharsets.UTF_8);
            logger.debug("Received response: {}", response);
            
            // For simple echo protocol, just log the response
            // In a real implementation, you'd handle the response properly
            
            // Return connection to pool
            returnConnection(ctx.channel());
        }
        
        @Override
        public void exceptionCaught(ChannelHandlerContext ctx, Throwable cause) {
            logger.error("Channel exception: {}", cause.getMessage());
            ctx.close();
        }
    }
    
    /**
     * Example usage method.
     */
    public static void main(String[] args) {
        SimpleNettyHsmClientNoSsl client = new SimpleNettyHsmClientNoSsl(
            "ap-india-central-crypto-uat.virtucrypt.com", 4121);
        
        try {
            // Async send
            client.sendMessageAsync("[AOECHO;BJ0;AGEcho Hello this is a test message.]")
                .thenAccept(response -> {
                    System.out.println("Async Response: " + response);
                })
                .exceptionally(throwable -> {
                    System.err.println("Async Error: " + throwable.getCause().getMessage());
                    return null;
                });
            
            // Sync send
            String response = client.sendMessage("[AOECHO;BJ0;AGEcho Hello this is a test message.]");
            System.out.println("Sync Response: " + response);
            
            // Print pool stats
            System.out.println("Pool Stats: " + client.getPoolStats());
            
        } catch (Exception e) {
            logger.error("Communication error: {}", e.getMessage());
        } finally {
            client.shutdown();
        }
    }
}
