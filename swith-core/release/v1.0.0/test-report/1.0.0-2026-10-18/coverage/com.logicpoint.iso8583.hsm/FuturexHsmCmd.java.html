package com.logicpoint.iso8583.hsm;

import java.util.LinkedHashMap;
import java.util.Map;
import java.util.StringJoiner;

import io.netty.handler.codec.mqtt.MqttMessageBuilders;
/*
 * This class is used to build and parse Futurex HSM commands.
 * @author: LogicPoint systems  
 * @version: 1.0
 * @since: 2025-08-06
 
 */

public class FuturexHsmCmd {
    private static final char START = '[';
    private static final char END = ']';
    private static final char SEPARATOR = ';';

    private String commandId;
    private final Map<String, String> fields = new LinkedHashMap<>();

    public FuturexHsmCmd(String commandId) {
        this.commandId = commandId;
    }

    public void addField(String token, String value) {
        if (token.length() != 2) {
            throw new IllegalArgumentException("Token must be exactly 2 characters");
        }
        fields.put(token, value);
    }

    public String build() {
        StringJoiner joiner = new StringJoiner(String.valueOf(SEPARATOR));
        joiner.add("AO"+commandId);
        for (Map.Entry<String, String> entry : fields.entrySet()) {
            joiner.add(entry.getKey() + entry.getValue());
        }
        return START + joiner.toString() + SEPARATOR + END;
    }

    public static FuturexHsmCmd parse(String commandStr) {
        if (commandStr == null || commandStr.isEmpty()) {
            throw new IllegalArgumentException("Command string cannot be null or empty");
        }else if (commandStr.charAt(0) != START || commandStr.charAt(commandStr.length() - 1) != END) {
            throw new IllegalArgumentException("Invalid format: must start with '[' and end with ']'");
        }

        String content = commandStr.substring(1, commandStr.length() - 1);
        String[] parts = content.split(String.valueOf(SEPARATOR));

        if (parts.length < 1) {
            throw new IllegalArgumentException("Empty command");
        }

        FuturexHsmCmd cmd = new FuturexHsmCmd(parts[0]);

        for (int i = 0; i < parts.length; i++) {
            if (parts[i].isEmpty()) continue;
            if (parts[i].length() < 2) {
                throw new IllegalArgumentException("Invalid field, too short: " + parts[i]);
            }
            String token = parts[i].substring(0, 2);
            String value = parts[i].substring(2);
            cmd.addField(token, value);
        }

        return cmd;
    }

    public String getCommandId() {
        return commandId;
    }

    public Map<String, String> getFields() {
        return fields;
    }

    public String getField(String token) {
        return fields.get(token);
    }

    @Override
    public String toString() {
        return "CommandID: " + commandId + ", Fields: " + fields;
    }

    public boolean isSuccess() {
        return !fields.get("AO").equals("ERRO");
    }

    public String getErroMsg() {
        String errorMsg = "token :" + fields.get("AN") + " error message :" + fields.get("BS");
        return errorMsg;
    }

        //cmd.addField("BJ0", "AGEcho Msg sent from APP service class");
       // System.out.println(cmd.build());    

        FuturexHsmCmd parsedCmd = FuturexHsmCmd.parse("[AOGWKS;BGA0088P0TE00E00003952751F1CD785C4930B2F9081BAFDD21FE504F5405CE96744CED32C092C5565F538069C;BHA0088P0TE00E0000A2130D0C75C27615DFE97B509C227F46706BE8DAF63C7B5878D89A1D8D32363F01107511;AE4CCA;]");
        System.out.println(parsedCmd.toString());
        parsedCmd.addField("AB", "DEFG");
        System.out.println(parsedCmd.build());
    }   

}
