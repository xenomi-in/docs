    /**
     * Starts a periodic task to cleanup idle connections from the pool.
     */
    private void startIdleConnectionCleanup() {
        if (eventLoopGroup != null && !eventLoopGroup.isShutdown()) {
            eventLoopGroup.scheduleAtFixedRate(() -> {
                if (!isShutdown) {
                    cleanupIdleConnections();
                }
            }, 30, 30, TimeUnit.SECONDS);
            logger.info("Started idle connection cleanup task (runs every 30 seconds)");
        }
    }
    
    /**
     * Cleans up idle or stale connections from the pool.
     */
    private void cleanupIdleConnections() {
        try {
            int initialSize = connectionPool.size();
            int removedCount = 0;
            
            // Create a temporary list to hold valid connections
            ConcurrentLinkedQueue<Channel> validConnections = new ConcurrentLinkedQueue<>();
            
            Channel channel;
            while ((channel = connectionPool.poll()) != null) {
                if (isConnectionValid(channel)) {
                    // Connection is still valid, keep it
                    validConnections.offer(channel);
                } else {
                    // Connection is stale, close it
                    logger.debug("Removing stale connection from pool (age: {}ms, idle: {}ms)", 
                        getConnectionAge(channel), getConnectionIdleTime(channel));
                    closeConnection(channel);
                    removedCount++;
                }
            }
            
            // Put valid connections back into the pool
            Channel validChannel;
            while ((validChannel = validConnections.poll()) != null) {
                connectionPool.offer(validChannel);
            }
            
            if (removedCount > 0) {
                logger.info("Cleaned up {} stale connections from pool (pool size: {} -> {})", 
                    removedCount, initialSize, connectionPool.size());
            } else {
                logger.debug("Idle connection cleanup: no stale connections found (pool size: {})", 
                    connectionPool.size());
            }
            
        } catch (Exception e) {
            logger.error("Error during idle connection cleanup: {}", e.getMessage(), e);
        }
    }

    /**
     * Initializes Netty components.
     */
    private void initializeNetty() {
        try {
            eventLoopGroup = new NioEventLoopGroup();
            
            // Create SSL context
            sslContext = createSslContext();
            
            // Create bootstrap
            bootstrap = new Bootstrap();
            bootstrap.group(eventLoopGroup)
                    .channel(NioSocketChannel.class)
                    .option(ChannelOption.CONNECT_TIMEOUT_MILLIS, timeout)
                    .option(ChannelOption.SO_KEEPALIVE, true)
                    .option(ChannelOption.TCP_NODELAY, true)
                    .handler(new ChannelInitializer<SocketChannel>() {
                        @Override
                        protected void initChannel(SocketChannel ch) {
                            ChannelPipeline pipeline = ch.pipeline();
                            
                            // Add SSL handler if SSL is enabled
                            if (sslContext != null) {
                                pipeline.addLast(sslContext.newHandler(ch.alloc(), host, port));
                            }
                            
                            // Add simple handler for echo protocol
                            pipeline.addLast(new SimpleHsmHandler());
                        }
                    });
            
            logger.info("Simple Netty HSM client initialized for {}:{}", host, port);
            
        } catch (Exception e) {
            logger.error("Failed to initialize Netty client: {}", e.getMessage());
            throw new RuntimeException("Failed to initialize Netty client", e);
        }
    }
    
    /**
     * Creates SSL context from PKCS#12 certificate.
     */
    private SslContext createSslContext() throws Exception {
        // Check if certificate path is empty or file doesn't exist
        if (p12Path == null || p12Path.trim().isEmpty()) {
            logger.info("No certificate path provided. Using trust-all SSL context without client certificate.");
            return createTrustAllSslContext();
        }
        
        // Load PKCS#12 keystore from either filesystem path (Docker/EC2) or classpath resource (local/dev).
        try (InputStream in = openP12InputStream(p12Path)) {
            KeyStore keyStore = KeyStore.getInstance("PKCS12");
            keyStore.load(in, p12Password.toCharArray());
        
            for (String alias : java.util.Collections.list(keyStore.aliases())) {
                Certificate cert = keyStore.getCertificate(alias);

                if (cert instanceof X509Certificate) {
                    X509Certificate x509Cert = (X509Certificate) cert;

                    logger.info("Loaded HSM client certificate alias={}, subject={}, issuer={}, validFrom={}, validTo={}",
                            alias,
                            x509Cert.getSubjectDN(),
                            x509Cert.getIssuerDN(),
                            x509Cert.getNotBefore(),
                            x509Cert.getNotAfter());
                }
            }
            // Create KeyManager for client authentication
            javax.net.ssl.KeyManagerFactory kmf = javax.net.ssl.KeyManagerFactory.getInstance("SunX509");
            kmf.init(keyStore, p12Password.toCharArray());

            // Build Netty SSL context with client certificate
            return SslContextBuilder.forClient()
                    .keyManager(kmf)
                    .trustManager(InsecureTrustManagerFactory.INSTANCE)
                    .build();
        }
    }

    private InputStream openP12InputStream(String configuredPath) throws Exception {
        String normalizedPath = configuredPath.trim();

        File certFile = new File(normalizedPath);
        if (certFile.isFile()) {
            logger.info("Loading HSM client certificate from filesystem path: {}", certFile.getAbsolutePath());
            return new FileInputStream(certFile);
        }

        String classpathPath = normalizedPath.startsWith("classpath:")
                ? normalizedPath.substring("classpath:".length())
                : normalizedPath;
        while (classpathPath.startsWith("/")) {
            classpathPath = classpathPath.substring(1);
        }

        InputStream classpathResource = getClass().getClassLoader().getResourceAsStream(classpathPath);
        if (classpathResource != null) {
            logger.info("Loading HSM client certificate from classpath resource: {}", classpathPath);
            return classpathResource;
        }

        throw new IllegalStateException("HSM client certificate not found at filesystem path or classpath resource: " + configuredPath);
    }
    
    /**
     * Creates a trust-all SSL context without client certificate.
     * Used when certificate file is not available.
     */
    private SslContext createTrustAllSslContext() throws Exception {
        logger.info("Creating trust-all SSL context without client certificate");
        
        // Create trust-all TrustManager to ignore server certificate validation
        javax.net.ssl.TrustManager[] trustAllCerts = new javax.net.ssl.TrustManager[] {
            new javax.net.ssl.X509TrustManager() {
                public java.security.cert.X509Certificate[] getAcceptedIssuers() {
                    return new java.security.cert.X509Certificate[0];
                }
                public void checkClientTrusted(java.security.cert.X509Certificate[] certs, String authType) {
                    // Trust all client certificates
                }
                public void checkServerTrusted(java.security.cert.X509Certificate[] certs, String authType) {
                    // Trust all server certificates
                }
            }
        };
        
        // Create SSL context with trust-all configuration only
        javax.net.ssl.SSLContext sslContext = javax.net.ssl.SSLContext.getInstance("TLS");
        sslContext.init(null, trustAllCerts, new java.security.SecureRandom());
        
        // Build Netty SSL context
        return SslContextBuilder.forClient()
                .trustManager(InsecureTrustManagerFactory.INSTANCE)
                .build();
    }

    /**
     * Gets a connection from the pool or creates a new one with retry capability.
     * 
     * @param retryCount Number of retries attempted
     */
    private CompletableFuture<Channel> getConnection(int retryCount) {
        CompletableFuture<Channel> future = new CompletableFuture<>();
        
        // Check if client is shutdown
        if (isShutdown) {
            future.completeExceptionally(new IllegalStateException("Client is shutdown"));
            return future;
        }
        
        // Check if event loop group is available
        if (eventLoopGroup == null || eventLoopGroup.isShutdown()) {
            future.completeExceptionally(new IllegalStateException("EventLoopGroup is not available"));
            return future;
        }
        
        // Try to get from pool first
        logger.debug("Getting connection from pool (pool size: {})", connectionPool.size());
        Channel pooledChannel = connectionPool.poll();
        if (pooledChannel != null) {
            // Validate the connection before using it
            if (isConnectionValid(pooledChannel)) {
                // Update last used time
                pooledChannel.attr(CONNECTION_LAST_USED_TIME).set(System.currentTimeMillis());
                
                // Reset the connection state for reuse
                AtomicBoolean returned = pooledChannel.attr(CONNECTION_RETURNED).get();
                if (returned != null) {
                    returned.set(false);
                }
                
                logger.debug("Reusing pooled connection (age: {}ms, idle: {}ms) pooled channel size: {}", 
                    getConnectionAge(pooledChannel), getConnectionIdleTime(pooledChannel), connectionPool.size());
                future.complete(pooledChannel);
                return future;
            } else {
                // Connection is stale or invalid, close it and try again
                logger.warn("Pooled connection is stale or invalid, discarding and getting new connection");
                closeConnection(pooledChannel);
                
                // Retry with a new connection (limit retries to avoid infinite loop)
                if (retryCount < 3) {
                    return getConnection(retryCount + 1);
                } else {
                    // Force create new connection after retries
                    logger.warn("Max retries reached for pooled connections, creating fresh connection");
                }
            }
        }
        
        // Create new connection if pool is empty or all pooled connections are invalid
        if (activeConnections.get() < maxConnections) {
            createNewConnection(future);
        } else {
            // Wait for available connection
            try {
                eventLoopGroup.schedule(() -> {
                    if (!future.isDone()) {
                        getConnection(retryCount).thenAccept(future::complete)
                            .exceptionally(throwable -> {
                                future.completeExceptionally(throwable);
                                return null;
                            });
                    }
                }, 100, TimeUnit.MILLISECONDS);
            } catch (RejectedExecutionException e) {
                future.completeExceptionally(new IllegalStateException("Executor not accepting tasks", e));
            }
        }
        
        return future;
    }
    
    /**
     * Validates if a connection is still usable.
     * Checks: active, writable, not too old, not idle too long, and optionally tests with echo.
     * 
     * @param channel The channel to validate
     * @return true if connection is valid and usable
     */
    private boolean isConnectionValid(Channel channel) {
        if (channel == null) {
            return false;
        }
        
        // Check if channel is active and writable
        if (!channel.isActive() || !channel.isWritable() || !channel.isOpen()) {
            logger.debug("Connection validation failed: active={}, writable={}, open={}", 
                channel.isActive(), channel.isWritable(), channel.isOpen());
            return false;
        }
        
        long now = System.currentTimeMillis();
        
        // Check connection age
        Long createdTime = channel.attr(CONNECTION_CREATED_TIME).get();
        if (createdTime != null && (now - createdTime) > CONNECTION_MAX_AGE_MS) {
            logger.debug("Connection too old (age: {}ms), discarding", (now - createdTime));
            return false;
        }
        
        // Check idle time
        Long lastUsedTime = channel.attr(CONNECTION_LAST_USED_TIME).get();
        if (lastUsedTime != null) {
            long idleTime = now - lastUsedTime;
            if (idleTime > maxIdleTime) {
                logger.debug("Connection idle too long (idle: {}ms), discarding", idleTime);
                return false;
            }
        }
        
        // If test-on-borrow is enabled, perform a quick validation
        if (testOnBorrow) {
            return validateConnectionWithEcho(channel);
        }
        
        return true;
    }
    
    /**
     * Validates a connection by sending a quick echo command.
     * This is a lightweight test to ensure the connection is responsive.
     * 
     * @param channel The channel to test
     * @return true if the connection responds successfully
     */
    private boolean validateConnectionWithEcho(Channel channel) {
        try {
            // For test-on-borrow, we do a quick channel state check instead of full echo
            // to avoid performance overhead. Full echo tests are done by keep-alive service.
            // Just verify the channel can accept writes and hasn't been closed remotely
            return channel.isActive() && channel.isOpen() && channel.isWritable();
        } catch (Exception e) {
            logger.debug("Connection validation failed: {}", e.getMessage());
            return false;
        }
    }
    
    /**
     * Gets the age of a connection in milliseconds.
     */
    private long getConnectionAge(Channel channel) {
        Long createdTime = channel.attr(CONNECTION_CREATED_TIME).get();
        if (createdTime != null) {
            return System.currentTimeMillis() - createdTime;
        }
        return 0;
    }
    
    /**
     * Gets the idle time of a connection in milliseconds.
     */
    private long getConnectionIdleTime(Channel channel) {
        Long lastUsedTime = channel.attr(CONNECTION_LAST_USED_TIME).get();
        if (lastUsedTime != null) {
            return System.currentTimeMillis() - lastUsedTime;
        }
        return 0;
    }
    
    /**
     * Closes a connection and updates counters.
     */
    private void closeConnection(Channel channel) {
        if (channel != null) {
            try {
                activeConnections.decrementAndGet();
                connectionSemaphore.release();
                unregisterConnectionFromKeepAlive(channel);
                channel.close();
                logger.debug("Closed connection. Active connections: {}", activeConnections.get());
            } catch (Exception e) {
                logger.warn("Error closing connection: {}", e.getMessage());
            }
        }
    }
    
    /**
     * Creates a new connection.
     */
    private void createNewConnection(CompletableFuture<Channel> future) {
        try {
            connectionSemaphore.acquire();
            activeConnections.incrementAndGet();
            
            bootstrap.connect(host, port).addListener((ChannelFutureListener) channelFuture -> {
                if (channelFuture.isSuccess()) {
                    Channel channel = channelFuture.channel();
                    
                    // Set connection timestamps
                    long now = System.currentTimeMillis();
                    channel.attr(CONNECTION_CREATED_TIME).set(now);
                    channel.attr(CONNECTION_LAST_USED_TIME).set(now);
                    channel.attr(CONNECTION_RETURNED).set(new AtomicBoolean(false));
                    
                    logger.info("Created new connection to {}:{} (total active: {})", 
                        host, port, activeConnections.get());
                    
                    // Register connection with keep-alive service if available
                    registerConnectionWithKeepAlive(channel);
                    
                    future.complete(channel);
                } else {
                    activeConnections.decrementAndGet();
                    connectionSemaphore.release();
                    logger.error("Failed to create connection: {}", channelFuture.cause().getMessage());
                    future.completeExceptionally(channelFuture.cause());
                }
            });
            
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
            future.completeExceptionally(e);
        }
    }
    
    /**
     * Returns a connection to the pool.
     */
    private void returnConnection(Channel channel) {
        if (channel != null && channel.isActive()) {
            // Check if the connection has already been returned
            AtomicBoolean returned = channel.attr(CONNECTION_RETURNED).get();
            if (returned == null) {
                returned = new AtomicBoolean(false);
                channel.attr(CONNECTION_RETURNED).set(returned);
            }
            
            if (returned.compareAndSet(false, true)) {
                // Update last used time before returning to pool
                channel.attr(CONNECTION_LAST_USED_TIME).set(System.currentTimeMillis());
                
                // Validate before returning to pool
                if (isConnectionValid(channel)) {
                    connectionPool.offer(channel);
                    logger.debug("Returned connection to pool. Pool size: {}, Active: {}", 
                        connectionPool.size(), activeConnections.get());
                } else {
                    // Don't return invalid connections to pool
                    logger.debug("Connection validation failed on return, closing instead of pooling");
                    closeConnection(channel);
                }
            } else {
                logger.debug("Connection {} already returned, skipping return.", channel.remoteAddress());
            }
        } else {
            // Connection is not active, close it properly
            logger.debug("Connection not active on return, closing");
            closeConnection(channel);
        }
    }

    /**
     * Sends a message asynchronously with retry capability.
     * 
     * @param message The message to send
     * @param retryCount Number of retries attempted
     * @return CompletableFuture with the response
     */
    private CompletableFuture<String> sendMessageAsyncWithRetry(String message, int retryCount) {
        CompletableFuture<String> responseFuture = new CompletableFuture<>();
        
        // Check if client is shutdown
        if (isShutdown) {
            responseFuture.completeExceptionally(new IllegalStateException("Client is shutdown"));
            return responseFuture;
        }
        
        // Check if event loop group is available
        if (eventLoopGroup == null || eventLoopGroup.isShutdown()) {
            responseFuture.completeExceptionally(new IllegalStateException("EventLoopGroup is not available"));
            return responseFuture;
        }
        
        getConnection().thenAccept(channel -> {
            logger.debug("Sending HSM request (attempt {})", (retryCount + 1));
            
            // Verify channel is still valid before using
            if (!channel.isActive() || !channel.isWritable()) {
                logger.warn("Channel not ready for write (active: {}, writable: {}), getting new connection", 
                    channel.isActive(), channel.isWritable());
                closeConnection(channel);
                
                // Retry with a new connection if within retry limit
                if (retryCount < 2) {
                    sendMessageAsyncWithRetry(message, retryCount + 1)
                        .thenAccept(responseFuture::complete)
                        .exceptionally(throwable -> {
                            responseFuture.completeExceptionally(throwable);
                            return null;
                        });
                } else {
                    responseFuture.completeExceptionally(
                        new IllegalStateException("Channel not ready after retries"));
                }
                return;
            }
            
            // Update connection activity with keep-alive service
            updateConnectionActivity(channel);
            channel.attr(RESPONSE_FUTURE).set(responseFuture);
            
            // Send message
            ByteBuf messageBuf = Unpooled.copiedBuffer(message, java.nio.charset.StandardCharsets.UTF_8);
            channel.writeAndFlush(messageBuf).addListener((ChannelFutureListener) writeFuture -> {
                if (!writeFuture.isSuccess()) {
                    Throwable cause = writeFuture.cause();
                    logger.error("Failed to send message (attempt {}): {}", (retryCount + 1), cause.getMessage());
                    
                    // Close the failed connection
                    closeConnection(channel);
                    
                    // Retry with a new connection if it's a connection issue and within retry limit
                    if (retryCount < 2 && isConnectionError(cause)) {
                        logger.info("Retrying with fresh connection due to connection error");
                        sendMessageAsyncWithRetry(message, retryCount + 1)
                            .thenAccept(responseFuture::complete)
                            .exceptionally(throwable -> {
                                responseFuture.completeExceptionally(throwable);
                                return null;
                            });
                    } else {
                        responseFuture.completeExceptionally(cause);
                    }
                    channel.attr(RESPONSE_FUTURE).set(null);
                    channel.attr(RESPONSE_TIMEOUT_FUTURE).set(null);
                } else {
                    logger.info("Message sent successfully to HSM (attempt {})", (retryCount + 1));
                }
            });
            
            // Set timeout and response handler
            ScheduledFuture<?> timeoutFuture = channel.eventLoop().schedule(() -> {
                if (!responseFuture.isDone()) {
                    logger.warn("HSM response timeout (attempt {})", (retryCount + 1));
                    responseFuture.completeExceptionally(new TimeoutException("Response timeout"));
                    channel.attr(RESPONSE_FUTURE).set(null);
                    channel.attr(RESPONSE_TIMEOUT_FUTURE).set(null);
                    closeConnection(channel);
                }
            }, timeout, TimeUnit.MILLISECONDS);
            channel.attr(RESPONSE_TIMEOUT_FUTURE).set(timeoutFuture);
            
        }).exceptionally(throwable -> {
            logger.error("Connection acquisition failed (attempt {}): {}", (retryCount + 1), throwable.getMessage());
            
            // Retry if within retry limit
            if (retryCount < 2) {
                logger.info("Retrying connection acquisition");
                sendMessageAsyncWithRetry(message, retryCount + 1)
                    .thenAccept(responseFuture::complete)
                    .exceptionally(error -> {
                        responseFuture.completeExceptionally(error);
                        return null;
                    });
            } else {
                responseFuture.completeExceptionally(throwable);
            }
            return null;
        });
        
        return responseFuture;
    }
    
    /**
     * Checks if an exception is a connection-related error that warrants retry.
     * 
     * @param throwable The exception to check
     * @return true if it's a connection error
     */
    private boolean isConnectionError(Throwable throwable) {
        if (throwable == null) {
            return false;
        }
        
        String message = throwable.getMessage();
        if (message != null) {
            message = message.toLowerCase();
            return message.contains("connection") || 
                   message.contains("broken pipe") ||
                   message.contains("socket closed") ||
                   message.contains("channel closed") ||
                   message.contains("reset by peer");
        }
        
        return throwable instanceof java.io.IOException ||
               throwable instanceof java.net.SocketException ||
               throwable instanceof ChannelException;
    }

     * Sends a message synchronously.
     */
    public String sendMessage(String message) throws Exception {
        try {
            logger.info("hsm cmd: {} ",message);
            return sendMessageAsync(message).get(timeout, TimeUnit.MILLISECONDS);
        } catch (TimeoutException e) {
            logger.error("Request timeout: {}", e.getMessage());
            return null;
            //throw new IOException("Request timeout", e);
        } catch (ExecutionException e) {
            logger.error("Communication failed: {}", e.getMessage());
            return null;
            //throw new IOException("Communication failed", e.getCause());
        }
    }
    
    /**
     * Gets connection pool statistics.
     */
    public String getPoolStats() {
        return String.format("Active: %d, Pooled: %d, Max: %d", 
                           activeConnections.get(), connectionPool.size(), maxConnections);
    }

    /** Number of open HSM connections (borrowed and idle). */
    public int getTotalConnectionCount() {
        return activeConnections.get();
    }

    /** Number of connections currently available in the pool. */
    public int getIdleConnectionCount() {
        return connectionPool.size();
    }

    /** Number of connections currently borrowed by callers. */
    public int getBorrowedConnectionCount() {
        return Math.max(0, activeConnections.get() - connectionPool.size());
    }

    public int getMaxConnectionCount() {
        return maxConnections;
    }

    /** Number of callers waiting for a pool permit. */
    public int getPendingConnectionCount() {
        return connectionSemaphore == null ? 0 : connectionSemaphore.getQueueLength();
    }
    
    /**
     * Shuts down the client.
     */
    public void shutdown() {
        logger.info("Shutting down Simple Netty HSM client");
        isShutdown = true;
        
        // Close pooled connections
        Channel channel;
        while ((channel = connectionPool.poll()) != null) {
            // Unregister from keep-alive service before closing
            unregisterConnectionFromKeepAlive(channel);
            channel.close();
        }
        
        // Shutdown event loop group gracefully
        if (eventLoopGroup != null && !eventLoopGroup.isShutdown()) {
            try {
                // Wait for graceful shutdown
                eventLoopGroup.shutdownGracefully(0, 5, TimeUnit.SECONDS).sync();
                logger.info("EventLoopGroup shutdown completed");
            } catch (InterruptedException e) {
                logger.warn("EventLoopGroup shutdown interrupted");
                Thread.currentThread().interrupt();
            }
        }
        
        logger.info("Simple Netty HSM client shutdown completed");
    }
    
    /**
     * Registers a connection with the keep-alive service.
     * 
     * @param channel The connection channel to register
     */
    private void registerConnectionWithKeepAlive(Channel channel) {
        if (keepAliveService != null) {
            try {
                keepAliveService.registerConnection(channel);
                logger.debug("Registered connection {} with keep-alive service", channel.remoteAddress());
            } catch (Exception e) {
                logger.debug("Could not register connection with keep-alive service: {}", e.getMessage());
            }
        }
    }
    
    /**
     * Unregisters a connection from the keep-alive service.
     * 
     * @param channel The connection channel to unregister
     */
    private void unregisterConnectionFromKeepAlive(Channel channel) {
        if (keepAliveService != null) {
            try {
                keepAliveService.unregisterConnection(channel);
                logger.debug("Unregistered connection {} from keep-alive service", channel.remoteAddress());
            } catch (Exception e) {
                logger.debug("Could not unregister connection from keep-alive service: {}", e.getMessage());
            }
        }
    }
    
    /**
     * Updates connection activity with the keep-alive service.
     * 
     * @param channel The connection channel that was used
     */
    private void updateConnectionActivity(Channel channel) {
        if (keepAliveService != null) {
            try {
                keepAliveService.updateConnectionActivity(channel);
            } catch (Exception e) {
                logger.debug("Could not update connection activity with keep-alive service: {}", e.getMessage());
            }
        }
    }
    
    /**
     * Sets the keep-alive service reference.
     * This method will be called by Spring to inject the keep-alive service.
     * 
     * @param keepAliveService The keep-alive service instance
     */
    @Autowired(required = false)
    public void setKeepAliveService(HsmKeepAliveServiceInterface keepAliveService) {
        this.keepAliveService = keepAliveService;
        if (keepAliveService != null) {
            logger.debug("Keep-alive service injected into SimpleNettyHsmClient");
        }
    }
    
    /**
     * Simple channel handler for HSM communication.
     */
    private class SimpleHsmHandler extends SimpleChannelInboundHandler<ByteBuf> {
        
        @Override
        protected void channelRead0(ChannelHandlerContext ctx, ByteBuf msg) {
            String response = msg.toString(java.nio.charset.StandardCharsets.UTF_8);
            logger.debug("Received HSM response");

            ScheduledFuture<?> timeoutFuture = ctx.channel().attr(RESPONSE_TIMEOUT_FUTURE).getAndSet(null);
            if (timeoutFuture != null) {
                timeoutFuture.cancel(false);
            }

            CompletableFuture<String> responseFuture = ctx.channel().attr(RESPONSE_FUTURE).getAndSet(null);
            if (responseFuture != null && !responseFuture.isDone()) {
                responseFuture.complete(response);
            }

            returnConnection(ctx.channel());
        }
        
        @Override
        public void exceptionCaught(ChannelHandlerContext ctx, Throwable cause) {
            logger.error("Channel exception: {}", cause.getMessage());
            ctx.close();
        }
        
        @Override
        public void channelActive(ChannelHandlerContext ctx) {
            logger.info("Channel connected to HSM: {}", ctx.channel().remoteAddress());
        }
        
        @Override
        public void channelInactive(ChannelHandlerContext ctx) {
            logger.info("Channel disconnected from HSM: {}", ctx.channel().remoteAddress());
        }
    }
    
    /**
     * Example usage method.
     */
    public static void main(String[] args) {
        String p12Path = "src/main/resources/Falcon_Test-YY1T4ZOT.p12";
        String p12Password = "Falcon@2025";
        
        SimpleNettyHsmClient client = new SimpleNettyHsmClient(
            "ap-india-central-crypto-uat.virtucrypt.com", 4121, p12Path, p12Password);
        
        try {
            // Print initial pool stats
            System.out.println("Initial Pool Stats: " + client.getPoolStats());
            
            // Test multiple async sends
            for (int i = 0; i < 5; i++) {
                final int messageIndex = i;
                client.sendMessageAsync("[AOSKEY;]")  //Key status test
                    .thenAccept(response -> {
                        System.out.println("Async Response " + messageIndex + ": " + response);
                        System.out.println("Pool Stats after response " + messageIndex + ": " + client.getPoolStats());
                    })
                    .exceptionally(throwable -> {
                        System.err.println("Async Error " + messageIndex + ": " + throwable.getCause().getMessage());
                        System.out.println("Pool Stats after error " + messageIndex + ": " + client.getPoolStats());
                        return null;
                    });
                
                // Small delay between requests
                Thread.sleep(1000);
            }
            
            // Wait for all responses
            Thread.sleep(10000);
            
            // Print final pool stats
            System.out.println("Final Pool Stats: " + client.getPoolStats());
            
        } catch (Exception e) {
            logger.error("Communication error: {}", e.getMessage());
        } finally {
            client.shutdown();
        }
    }
