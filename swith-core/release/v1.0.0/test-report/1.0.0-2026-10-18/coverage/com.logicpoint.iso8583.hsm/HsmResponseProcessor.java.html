package com.logicpoint.iso8583.hsm;

import tools.jackson.databind.JsonNode;
import tools.jackson.databind.json.JsonMapper;
import java.io.IOException;

/**
 * Utility class for processing responses from Futurex HSM.
 */
public class HsmResponseProcessor {

    private static final JsonMapper mapper = JsonMapper.builderWithJackson2Defaults().build();

    /**
     * Parses the HSM response and extracts the result code.
     *
     * @param responseJson The JSON response string
     * @return The result code or null if parsing fails
     * @throws IOException If JSON parsing fails
     */
    public static String getResultCode(String responseJson) throws IOException {
        JsonNode root = mapper.readTree(responseJson);
        return root.path("resultCode").asText();
    }

    // Add more processing methods, e.g., extract data fields, check for errors, etc.

    /**
     * Checks if the response indicates success.
     *
     * @param responseJson The JSON response string
     * @return true if successful, false otherwise
     * @throws IOException If JSON parsing fails
     */
    public static boolean isSuccess(String responseJson) throws IOException {
        String code = getResultCode(responseJson);
        return "00".equals(code); // Assuming '00' is success code
    }

    /**
     * Extracts the generated key from HSM response.
     *
     * @param responseJson The JSON response string
     * @return The base64-encoded key or null if not found
     * @throws IOException If JSON parsing fails
     */
    public static String extractGeneratedKey(String responseJson) throws IOException {
        JsonNode root = mapper.readTree(responseJson);
        return root.path("data").path("key").asText();
    }
}
