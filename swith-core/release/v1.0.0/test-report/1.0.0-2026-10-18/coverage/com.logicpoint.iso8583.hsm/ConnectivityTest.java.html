package com.logicpoint.iso8583.hsm;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

import java.io.IOException;
import java.net.Socket;
import java.net.InetSocketAddress;

/**
 * Simple connectivity test to verify basic TCP connection to HSM.
 * 
 * @author LogicPoint Systems
 * @version 1.0
 */
public class ConnectivityTest {
    
    private static final Logger logger = LoggerFactory.getLogger(ConnectivityTest.class);
    
    public static void main(String[] args) {
        String host = "ap-india-central-crypto-uat.virtucrypt.com";
        int port = 4121;
        
        logger.info("=== Basic TCP Connectivity Test ===");
        logger.info("Testing connection to {}:{}", host, port);
        
        Socket socket = null;
        try {
            // Test basic TCP connectivity
            logger.info("Attempting TCP connection...");
            socket = new Socket();
            socket.connect(new InetSocketAddress(host, port), 10000); // 10 second timeout
            
            if (socket.isConnected()) {
                logger.info("✅ TCP connection successful!");
                logger.info("Local address: {}", socket.getLocalAddress());
                logger.info("Remote address: {}", socket.getInetAddress());
                
                // Test if we can send data
                logger.info("Testing data transmission...");
                String testMessage = "[AOECHO;BJ0;AGEcho Test.;]";
                socket.getOutputStream().write(testMessage.getBytes("UTF-8"));
                socket.getOutputStream().flush();
                logger.info("✅ Data sent successfully");
                
                // Try to read response
                logger.info("Attempting to read response...");
                byte[] buffer = new byte[1024];
                int bytesRead = socket.getInputStream().read(buffer);
                
                if (bytesRead > 0) {
                    String response = new String(buffer, 0, bytesRead, "UTF-8");
                    logger.info("✅ Received response: {}", response);
                } else {
                    logger.warn("⚠️ No response received from server");
                }
                
            } else {
                logger.error("❌ TCP connection failed");
            }
            
        } catch (IOException e) {
            logger.error("❌ Connection failed: {}", e.getMessage());
            if (e.getCause() != null) {
                logger.error("Root cause: {}", e.getCause().getMessage());
            }
        } finally {
            if (socket != null) {
                try {
                    socket.close();
                    logger.info("Socket closed");
                } catch (IOException e) {
                    logger.warn("Error closing socket: {}", e.getMessage());
                }
            }
        }
        
        logger.info("=== Connectivity Test Completed ===");
    }
}
