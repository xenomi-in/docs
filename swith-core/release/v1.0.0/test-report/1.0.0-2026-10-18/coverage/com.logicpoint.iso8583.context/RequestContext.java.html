package com.logicpoint.iso8583.context;

import com.logicpoint.iso8583.emv.ChipData;
import com.logicpoint.iso8583.emv.ChipDataParser;
import com.logicpoint.iso8583.responsecode.InternalResCode;
import com.logicpoint.iso8583.util.Scheme;
import com.logicpoint.iso8583.util.rupay.RuPayDe61Parser;
import com.logicpoint.iso8583.validation.TransactionType;
import com.logicpoint.iso8583.validation.TlvUtils;

import java.util.Collections;
import java.util.HashMap;
import java.util.Map;

public class RequestContext {
    private final String requestId;
    private final String uniqueTransactionId;
    private final String institutionId;
    private final String mti;
    private final Scheme scheme;
    private final Map<Integer, String> fields;
    private final Map<String, Object> attributes;
    private final String cardNumber;
    private final String bin;
    private final Double amount;
    private final String merchantId;
    private String panHash;
    private TransactionType transactionType;
    private TransactionChannel channel;
    private InternalResCode internalResponseCode;

    public RequestContext(
            String requestId,
            String uniqueTransactionId,
            String institutionId,
            String mti,
            Scheme scheme,
            Map<Integer, String> fields
    ) {
        this.requestId = requestId;
        this.uniqueTransactionId = uniqueTransactionId;
        this.institutionId = institutionId;
        this.mti = mti;
        this.scheme = scheme;
        this.fields = fields;
        this.attributes = new HashMap<>();
        this.cardNumber = fields.get(2);
        this.bin = cardNumber != null && cardNumber.length() >= 6 ? cardNumber.substring(0, 6) : null;
        this.amount = parseAmount(fields.get(4));
        this.merchantId = fields.get(42);
        this.transactionType = TransactionType.UNKNOWN;
        this.channel = TransactionChannel.UNKNOWN;
        this.internalResponseCode = null;
    }

    private Double parseAmount(String amountField) {
        if (amountField == null || amountField.isBlank()) {
            return null;
        }
        try {
            return Double.parseDouble(amountField) / 100.0;
        } catch (NumberFormatException ex) {
            return null;
        }
    }

    public String getRequestId() {
        return requestId;
    }

    public String getUniqueTransactionId() {
        return uniqueTransactionId;
    }

    public String getInstitutionId() {
        return institutionId;
    }

    public String getMti() {
        return mti;
    }

    public Scheme getScheme() {
        return scheme;
    }

    public Map<Integer, String> getFields() {
        return fields;
    }

    public Map<String, Object> getAttributes() {
        return attributes;
    }

    public String getCardNumber() {
        return cardNumber;
    }

    public String getBin() {
        return bin;
    }

    public Double getAmount() {
        return amount;
    }

    public String getMerchantId() {
        return merchantId;
    }

    public String getPanHash() {
        return panHash;
    }

    public void setPanHash(String panHash) {
        this.panHash = panHash;
    }

    public Object getAttribute(String key) {
        return attributes.get(key);
    }

    public void setAttribute(String key, Object value) {
        attributes.put(key, value);
    }

    public TransactionType getTransactionType() {
        return transactionType;
    }

    public void setTransactionType(TransactionType transactionType) {
        this.transactionType = transactionType == null ? TransactionType.UNKNOWN : transactionType;
    }

    public TransactionChannel getChannel() {
        return channel;
    }

    public void setChannel(TransactionChannel channel) {
        this.channel = channel == null ? TransactionChannel.UNKNOWN : channel;
    }

    public InternalResCode getInternalResponseCode() {
        return internalResponseCode;
    }

    public void setInternalResponseCode(InternalResCode internalResponseCode) {
        this.internalResponseCode = internalResponseCode;
    }

    @SuppressWarnings("unchecked")
    public Map<String, String> getParsedTlv(int fieldNumber) {
        String rawValue = fields.get(fieldNumber);
        if (rawValue == null || rawValue.isBlank()) {
            return Collections.emptyMap();
        }

        String rawKey = "parsed_tlv_raw_" + fieldNumber;
        String cacheKey = "parsed_tlv_" + fieldNumber;
        Object cachedRaw = attributes.get(rawKey);
        Object cachedValue = attributes.get(cacheKey);
        if (rawValue.equals(cachedRaw) && cachedValue instanceof Map<?, ?>) {
            return (Map<String, String>) cachedValue;
        }

        Map<String, String> parsed = TlvUtils.parseTlv(rawValue);
        attributes.put(rawKey, rawValue);
        attributes.put(cacheKey, parsed);
        return parsed;
    }

    public ChipData getParsedChipData(int fieldNumber) {
        String rawValue = fields.get(fieldNumber);
        if (rawValue == null || rawValue.isBlank()) {
            return null;
        }

        String rawKey = "parsed_chip_raw_" + fieldNumber;
        String cacheKey = "parsed_chip_" + fieldNumber;
        Object cachedRaw = attributes.get(rawKey);
        Object cachedValue = attributes.get(cacheKey);
        if (rawValue.equals(cachedRaw) && cachedValue instanceof ChipData chipData) {
            return chipData;
        }

        ChipData chipData = new ChipDataParser().parseChipData(rawValue);
        attributes.put(rawKey, rawValue);
        attributes.put(cacheKey, chipData);
        return chipData;
    }

    public RuPayDe61Parser.RuPayDE61Data getParsedDe61(int fieldNumber) {
        String rawValue = fields.get(fieldNumber);
        if (rawValue == null || rawValue.isBlank()) {
            return null;
        }

        String rawKey = "parsed_de61_raw_" + fieldNumber;
        String cacheKey = "parsed_de61_" + fieldNumber;
        Object cachedRaw = attributes.get(rawKey);
        Object cachedValue = attributes.get(cacheKey);
        if (rawValue.equals(cachedRaw) && cachedValue instanceof RuPayDe61Parser.RuPayDE61Data de61Data) {
            return de61Data;
        }

        RuPayDe61Parser.RuPayDE61Data de61Data = RuPayDe61Parser.parseDE61(rawValue);
        attributes.put(rawKey, rawValue);
        attributes.put(cacheKey, de61Data);
        return de61Data;
    }
}
