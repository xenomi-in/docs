package com.xenomi.cms.core.v2.product.controller;

import java.util.UUID;
import com.xenomi.cms.core.api.ApiResponseEnvelope;
import com.xenomi.cms.core.api.ApiResponseHelper;
import com.xenomi.cms.core.api.PageableDefaults;
import com.xenomi.cms.core.api.PagedResponse;
import com.xenomi.cms.core.v2.product.dto.*;
import com.xenomi.cms.core.v2.product.service.ProductDesignAssignmentV2Service;
import com.xenomi.cms.core.v2.product.service.ProductPaymentProfileV2Service;
import io.swagger.v3.oas.annotations.Operation;
import io.swagger.v3.oas.annotations.tags.Tag;
import jakarta.validation.Valid;
import org.springframework.data.domain.PageRequest;
import org.springframework.data.domain.Pageable;
import org.springframework.data.domain.Sort;
import org.springframework.http.ResponseEntity;
import org.springframework.validation.annotation.Validated;
import org.springframework.web.bind.annotation.*;

@Validated
@RestController
@RequestMapping("/cms/api/v2/institutions/{institutionId}/products/{productId}/versions/{versionId}")
@Tag(name = "V2 - Issuance Profiles", description = "Card designs, assignments, technology capabilities and CVM rules")
public class ProductIssuanceProfileV2Controller {
    private final ProductDesignAssignmentV2Service designService;
    private final ProductPaymentProfileV2Service profileService;
    public ProductIssuanceProfileV2Controller(ProductDesignAssignmentV2Service designService,
            ProductPaymentProfileV2Service profileService) {
        this.designService = designService;
        this.profileService = profileService;
    }

    @PostMapping("/card-design-assignments")
    @Operation(operationId = "createProductCardDesignAssignmentV2", summary = "Assign a design to a product version",
            description = "Creates a version-scoped design assignment. Only a design owned by the same institution can be assigned.")
    public ResponseEntity<ApiResponseEnvelope<ProductCardDesignAssignmentV2Response>> createDesignAssignment(
            @PathVariable UUID institutionId, @PathVariable UUID productId, @PathVariable UUID versionId,
            @Valid @RequestBody CreateProductCardDesignAssignmentV2Request request) {
        return ApiResponseHelper.created(designService.create(institutionId, productId, versionId, request));
    }
    @GetMapping("/card-design-assignments")
    @Operation(operationId = "listProductCardDesignAssignmentsV2", summary = "List product-version design assignments",
            description = "Returns the paginated physical or virtual card designs assigned to this product configuration version.")
    public ResponseEntity<ApiResponseEnvelope<PagedResponse<ProductCardDesignAssignmentV2Response>>> listDesignAssignments(
            @PathVariable UUID institutionId, @PathVariable UUID productId, @PathVariable UUID versionId,
            @RequestParam(required = false) Integer page, @RequestParam(required = false) Integer limit) {
        return ApiResponseHelper.ok(PagedResponse.of(designService.list(institutionId, productId, versionId,
                page(page, limit, "cardForm"))));
    }
    @GetMapping("/card-design-assignments/{assignmentId}")
    @Operation(operationId = "getProductCardDesignAssignmentV2", summary = "Get a product-version design assignment",
            description = "Returns one tenant-scoped design assignment by UUID.")
    public ResponseEntity<ApiResponseEnvelope<ProductCardDesignAssignmentV2Response>> getDesignAssignment(
            @PathVariable UUID institutionId, @PathVariable UUID productId, @PathVariable UUID versionId,
            @PathVariable UUID assignmentId) {
        return ApiResponseHelper.ok(designService.get(institutionId, productId, versionId, assignmentId));
    }
    @PostMapping("/card-design-assignments/{assignmentId}/actions/update")
    @Operation(operationId = "updateProductCardDesignAssignmentV2", summary = "Update or transition a design assignment",
            description = "Applies editable fields or an explicit lifecycle transition without replacing assignment history.")
    public ResponseEntity<ApiResponseEnvelope<ProductCardDesignAssignmentV2Response>> updateDesignAssignment(
            @PathVariable UUID institutionId, @PathVariable UUID productId, @PathVariable UUID versionId,
            @PathVariable UUID assignmentId,
            @Valid @RequestBody UpdateProductCardDesignAssignmentV2Request request) {
        return ApiResponseHelper.ok(designService.update(institutionId, productId, versionId, assignmentId, request));
    }

    @PostMapping("/technology-profiles")
    @Operation(operationId = "createProductTechnologyProfileV2", summary = "Create the next technology profile version",
            description = "Creates an immutable version describing magnetic-stripe, EMV contact and contactless capabilities.")
    public ResponseEntity<ApiResponseEnvelope<ProductTechnologyProfileV2Response>> createTechnology(
            @PathVariable UUID institutionId, @PathVariable UUID productId, @PathVariable UUID versionId,
            @Valid @RequestBody CreateProductTechnologyProfileV2Request request) {
        return ApiResponseHelper.created(profileService.createTechnology(institutionId, productId, versionId, request));
    }
    @GetMapping("/technology-profiles")
    @Operation(operationId = "listProductTechnologyProfilesV2", summary = "List product technology profiles",
            description = "Returns all technology-profile versions for this product configuration version in version order.")
    public ResponseEntity<ApiResponseEnvelope<PagedResponse<ProductTechnologyProfileV2Response>>> listTechnology(
            @PathVariable UUID institutionId, @PathVariable UUID productId, @PathVariable UUID versionId,
            @RequestParam(required = false) Integer page, @RequestParam(required = false) Integer limit) {
        return ApiResponseHelper.ok(PagedResponse.of(profileService.listTechnology(institutionId, productId, versionId,
                page(page, limit, "profileVersion"))));
    }
    @GetMapping("/technology-profiles/{profileId}")
    @Operation(operationId = "getProductTechnologyProfileV2", summary = "Get a product technology profile",
            description = "Returns one tenant-scoped technology-profile version by UUID.")
    public ResponseEntity<ApiResponseEnvelope<ProductTechnologyProfileV2Response>> getTechnology(
            @PathVariable UUID institutionId, @PathVariable UUID productId, @PathVariable UUID versionId,
            @PathVariable UUID profileId) {
        return ApiResponseHelper.ok(profileService.getTechnology(institutionId, productId, versionId, profileId));
    }
    @PostMapping("/technology-profiles/{profileId}/actions/update")
    @Operation(operationId = "updateProductTechnologyProfileV2", summary = "Update or transition a technology profile",
            description = "Updates an editable draft or applies an explicit lifecycle transition; activated profile history is preserved.")
    public ResponseEntity<ApiResponseEnvelope<ProductTechnologyProfileV2Response>> updateTechnology(
            @PathVariable UUID institutionId, @PathVariable UUID productId, @PathVariable UUID versionId,
            @PathVariable UUID profileId, @Valid @RequestBody UpdateProductTechnologyProfileV2Request request) {
        return ApiResponseHelper.ok(profileService.updateTechnology(institutionId, productId, versionId, profileId, request));
    }

    @PostMapping("/cvm-profiles")
    @Operation(operationId = "createProductCvmProfileV2", summary = "Create a CVM profile and ordered rules",
            description = "Creates a versioned cardholder-verification profile. Rules are evaluated by ascending ruleSequence.")
    public ResponseEntity<ApiResponseEnvelope<ProductCvmProfileV2Response>> createCvm(
            @PathVariable UUID institutionId, @PathVariable UUID productId, @PathVariable UUID versionId,
            @Valid @RequestBody CreateProductCvmProfileV2Request request) {
        return ApiResponseHelper.created(profileService.createCvm(institutionId, productId, versionId, request));
    }
    @GetMapping("/cvm-profiles")
    @Operation(operationId = "listProductCvmProfilesV2", summary = "List product CVM profiles",
            description = "Returns all CVM-profile versions and their ordered rules for this product configuration version.")
    public ResponseEntity<ApiResponseEnvelope<PagedResponse<ProductCvmProfileV2Response>>> listCvm(
            @PathVariable UUID institutionId, @PathVariable UUID productId, @PathVariable UUID versionId,
            @RequestParam(required = false) Integer page, @RequestParam(required = false) Integer limit) {
        return ApiResponseHelper.ok(PagedResponse.of(profileService.listCvm(institutionId, productId, versionId,
                page(page, limit, "profileVersion"))));
    }
    @GetMapping("/cvm-profiles/{profileId}")
    @Operation(operationId = "getProductCvmProfileV2", summary = "Get a product CVM profile",
            description = "Returns one tenant-scoped CVM profile and its ordered rules by UUID.")
    public ResponseEntity<ApiResponseEnvelope<ProductCvmProfileV2Response>> getCvm(
            @PathVariable UUID institutionId, @PathVariable UUID productId, @PathVariable UUID versionId,
            @PathVariable UUID profileId) {
        return ApiResponseHelper.ok(profileService.getCvm(institutionId, productId, versionId, profileId));
    }
    @PostMapping("/cvm-profiles/{profileId}/actions/update")
    @Operation(operationId = "updateProductCvmProfileV2", summary = "Update or transition a CVM profile",
            description = "Updates an editable profile and ordered rule set or applies an explicit lifecycle transition.")
    public ResponseEntity<ApiResponseEnvelope<ProductCvmProfileV2Response>> updateCvm(
            @PathVariable UUID institutionId, @PathVariable UUID productId, @PathVariable UUID versionId,
            @PathVariable UUID profileId, @Valid @RequestBody UpdateProductCvmProfileV2Request request) {
        return ApiResponseHelper.ok(profileService.updateCvm(institutionId, productId, versionId, profileId, request));
    }

    private static Pageable page(Integer page, Integer limit, String property) {
        Pageable base = PageableDefaults.ofOneBased(page, limit);
        return PageRequest.of(base.getPageNumber(), base.getPageSize(), Sort.by(Sort.Direction.ASC, property));
    }
}
