@Validated
@RestController
@RequestMapping("/cms/api/v2/institutions/{institutionId}/products/{productId}")
@Tag(name = "V2 - Product Policies",
        description = "BIN allocation, transaction controls, limits and versioned product policies")
public class ProductConfigurationV2Controller {
    private static final String ACTOR_HEADER = "X-Actor-Id";
    private final ProductBinRangeAssignmentV2Service assignmentService;
    private final ProductVersionConfigurationV2Service versionConfigurationService;
    private final ProductPinPolicyV2Service pinPolicyService;
    private final ProductReplacementPolicyV2Service replacementPolicyService;
    private final ProductIssuancePolicyV2Service issuancePolicyService;

    public ProductConfigurationV2Controller(
            ProductBinRangeAssignmentV2Service assignmentService,
            ProductVersionConfigurationV2Service versionConfigurationService,
            ProductPinPolicyV2Service pinPolicyService,
            ProductReplacementPolicyV2Service replacementPolicyService,
            ProductIssuancePolicyV2Service issuancePolicyService) {
        this.assignmentService = assignmentService;
        this.versionConfigurationService = versionConfigurationService;
        this.pinPolicyService = pinPolicyService;
        this.replacementPolicyService = replacementPolicyService;
        this.issuancePolicyService = issuancePolicyService;
    }

    @PostMapping("/bin-range-assignments")
    @Operation(operationId = "createProductBinRangeAssignmentV2",
            summary = "Assign an institution-owned BIN range to a product")
    public ResponseEntity<ApiResponseEnvelope<ProductBinRangeAssignmentV2Response>> createAssignment(
            @PathVariable UUID institutionId, @PathVariable UUID productId,
            @Valid @RequestBody CreateProductBinRangeAssignmentV2Request request,
            @RequestHeader(ACTOR_HEADER) @NotBlank @Size(max = 100) String actorId) {
        return ApiResponseHelper.created(assignmentService.create(institutionId, productId, request, actorId));
    }

    @GetMapping("/bin-range-assignments")
    @Operation(operationId = "listProductBinRangeAssignmentsV2",
            summary = "List BIN-range assignments for a product")
    public ResponseEntity<ApiResponseEnvelope<PagedResponse<ProductBinRangeAssignmentV2Response>>> listAssignments(
            @PathVariable UUID institutionId, @PathVariable UUID productId,
            @RequestParam(required = false) Integer page,
            @RequestParam(required = false) Integer limit) {
        Pageable base = PageableDefaults.ofOneBased(page, limit);
        Pageable pageable = PageRequest.of(base.getPageNumber(), base.getPageSize(),
                Sort.by(Sort.Direction.ASC, "priority"));
        return ApiResponseHelper.ok(PagedResponse.of(
                assignmentService.list(institutionId, productId, pageable)));
    }

    @GetMapping("/bin-range-assignments/{assignmentId}")
    @Operation(operationId = "getProductBinRangeAssignmentV2",
            summary = "Get a product BIN-range assignment")
    public ResponseEntity<ApiResponseEnvelope<ProductBinRangeAssignmentV2Response>> getAssignment(
            @PathVariable UUID institutionId, @PathVariable UUID productId,
            @PathVariable UUID assignmentId) {
        return ApiResponseHelper.ok(assignmentService.get(institutionId, productId, assignmentId));
    }

    @PostMapping("/bin-range-assignments/{assignmentId}/actions/update")
    @Operation(operationId = "updateProductBinRangeAssignmentV2",
            summary = "Update or transition a product BIN-range assignment")
    public ResponseEntity<ApiResponseEnvelope<ProductBinRangeAssignmentV2Response>> updateAssignment(

            @PathVariable UUID assignmentId,

    }
