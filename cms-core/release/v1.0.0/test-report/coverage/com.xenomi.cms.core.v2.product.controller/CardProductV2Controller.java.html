package com.xenomi.cms.core.v2.product.controller;

import java.util.UUID;

import com.xenomi.cms.core.api.ApiResponseEnvelope;
import com.xenomi.cms.core.api.ApiResponseHelper;
import com.xenomi.cms.core.api.PageableDefaults;
import com.xenomi.cms.core.api.PagedResponse;
import com.xenomi.cms.core.v2.bin.model.CardNetwork;
import com.xenomi.cms.core.v2.product.dto.CardProductV2Response;
import com.xenomi.cms.core.v2.product.dto.CreateCardProductV2Request;
import com.xenomi.cms.core.v2.product.dto.CreateProductVersionV2Request;
import com.xenomi.cms.core.v2.product.dto.ProductVersionV2Response;
import com.xenomi.cms.core.v2.product.dto.UpdateCardProductV2Request;
import com.xenomi.cms.core.v2.product.dto.UpdateProductVersionV2Request;
import com.xenomi.cms.core.v2.product.model.FundingType;
import com.xenomi.cms.core.v2.product.model.ProductStatus;
import com.xenomi.cms.core.v2.product.service.CardProductV2Service;
import com.xenomi.cms.core.v2.product.service.ProductVersionV2Service;

import io.swagger.v3.oas.annotations.Operation;
import io.swagger.v3.oas.annotations.tags.Tag;
import jakarta.validation.Valid;
import jakarta.validation.constraints.NotBlank;
import jakarta.validation.constraints.Size;
import org.springframework.data.domain.PageRequest;
import org.springframework.data.domain.Pageable;
import org.springframework.data.domain.Sort;
import org.springframework.http.ResponseEntity;
import org.springframework.validation.annotation.Validated;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RequestHeader;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.bind.annotation.RestController;

@Validated
@RestController
@RequestMapping("/cms/api/v2/institutions/{institutionId}/products")
@Tag(name = "V2 - Products", description = "Stable products, configuration versions and lifecycle")
public class CardProductV2Controller {

    private static final String ACTOR_HEADER = "X-Actor-Id";
    private final CardProductV2Service productService;
    private final ProductVersionV2Service versionService;

    public CardProductV2Controller(
            CardProductV2Service productService, ProductVersionV2Service versionService) {
        this.productService = productService;
        this.versionService = versionService;
    }

    @PostMapping
    @Operation(operationId = "createCardProductV2", summary = "Create a card product")
    public ResponseEntity<ApiResponseEnvelope<CardProductV2Response>> create(
            @PathVariable UUID institutionId,
            @Valid @RequestBody CreateCardProductV2Request request,
            @RequestHeader(ACTOR_HEADER) @NotBlank @Size(max = 100) String actorId) {
        return ApiResponseHelper.created(productService.create(institutionId, request, actorId));
    }

    @GetMapping
    @Operation(operationId = "searchCardProductsV2", summary = "Search and list card products")
    public ResponseEntity<ApiResponseEnvelope<PagedResponse<CardProductV2Response>>> search(
            @PathVariable UUID institutionId,
            @RequestParam(required = false) String query,
            @RequestParam(required = false) CardNetwork cardNetwork,
            @RequestParam(required = false) FundingType fundingType,
            @RequestParam(required = false) ProductStatus productStatus,
            @RequestParam(required = false) Integer page,
            @RequestParam(required = false) Integer limit) {
        Pageable base = PageableDefaults.ofOneBased(page, limit);
        Pageable pageable = PageRequest.of(base.getPageNumber(), base.getPageSize(),
                Sort.by(Sort.Direction.ASC, "productCode"));
        return ApiResponseHelper.ok(PagedResponse.of(productService.search(
                institutionId, query, cardNetwork, fundingType, productStatus, pageable)));
    }

    @GetMapping("/{productId}")
    @Operation(operationId = "getCardProductV2", summary = "Get a card product")
    public ResponseEntity<ApiResponseEnvelope<CardProductV2Response>> get(
            @PathVariable UUID institutionId, @PathVariable UUID productId) {
        return ApiResponseHelper.ok(productService.get(institutionId, productId));
    }

    @GetMapping("/code/{productCode}")
    @Operation(operationId = "getCardProductByCodeV2", summary = "Get a card product by code")
    public ResponseEntity<ApiResponseEnvelope<CardProductV2Response>> getByCode(
            @PathVariable UUID institutionId, @PathVariable String productCode) {
        return ApiResponseHelper.ok(productService.getByCode(institutionId, productCode));
    }

    @PostMapping("/{productId}/actions/update")
    @Operation(operationId = "updateCardProductV2", summary = "Apply an explicit update or lifecycle action to a card product",
            description = "This is the single product lifecycle path; V2 has no overlapping deactivate endpoint. ACTIVE permits new issuance and virtual-to-physical requests. SUSPENDED and RETIRED reject those new operations, but do not alter live cards or cancel production requests already created. Existing cards remain serviceable, including policy-controlled replacement. Invalid transitions return a business validation error.")
    public ResponseEntity<ApiResponseEnvelope<CardProductV2Response>> update(
            @PathVariable UUID institutionId, @PathVariable UUID productId,
            @Valid @RequestBody UpdateCardProductV2Request request,
            @RequestHeader(ACTOR_HEADER) @NotBlank @Size(max = 100) String actorId) {
        return ApiResponseHelper.ok(productService.update(institutionId, productId, request, actorId));
    }

    @PostMapping("/{productId}/versions")
    @Operation(operationId = "createProductVersionV2", summary = "Create the next configuration version for a product")
    public ResponseEntity<ApiResponseEnvelope<ProductVersionV2Response>> createVersion(
            @PathVariable UUID institutionId, @PathVariable UUID productId,
            @Valid @RequestBody CreateProductVersionV2Request request,
            @RequestHeader(ACTOR_HEADER) @NotBlank @Size(max = 100) String actorId) {
        return ApiResponseHelper.created(versionService.create(institutionId, productId, request, actorId));
    }

    @GetMapping("/{productId}/versions")
    @Operation(operationId = "listProductVersionsV2", summary = "List configuration versions for a product")
    public ResponseEntity<ApiResponseEnvelope<PagedResponse<ProductVersionV2Response>>> listVersions(
            @PathVariable UUID institutionId, @PathVariable UUID productId,
            @RequestParam(required = false) Integer page,
            @RequestParam(required = false) Integer limit) {
        Pageable base = PageableDefaults.ofOneBased(page, limit);
        Pageable pageable = PageRequest.of(base.getPageNumber(), base.getPageSize(),
                Sort.by(Sort.Direction.DESC, "versionNumber"));
        return ApiResponseHelper.ok(PagedResponse.of(
                versionService.list(institutionId, productId, pageable)));
    }

    @GetMapping("/{productId}/versions/{versionId}")
    @Operation(operationId = "getProductVersionV2", summary = "Get a product configuration version")
    public ResponseEntity<ApiResponseEnvelope<ProductVersionV2Response>> getVersion(
            @PathVariable UUID institutionId, @PathVariable UUID productId,
            @PathVariable UUID versionId) {
        return ApiResponseHelper.ok(versionService.get(institutionId, productId, versionId));
    }

    @PostMapping("/{productId}/versions/{versionId}/actions/update")
    @Operation(operationId = "updateProductVersionV2", summary = "Apply an explicit update or lifecycle action to a draft version")
    public ResponseEntity<ApiResponseEnvelope<ProductVersionV2Response>> updateVersion(
            @PathVariable UUID institutionId, @PathVariable UUID productId,
            @PathVariable UUID versionId,
            @Valid @RequestBody UpdateProductVersionV2Request request,
            @RequestHeader(ACTOR_HEADER) @NotBlank @Size(max = 100) String actorId) {
        return ApiResponseHelper.ok(versionService.update(
                institutionId, productId, versionId, request, actorId));
    }
}
