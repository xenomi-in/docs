package com.xenomi.cms.core.v2.product.controller;

import java.util.UUID;
import com.xenomi.cms.core.api.ApiResponseEnvelope;
import com.xenomi.cms.core.api.ApiResponseHelper;
import com.xenomi.cms.core.api.PageableDefaults;
import com.xenomi.cms.core.api.PagedResponse;
import com.xenomi.cms.core.v2.bin.model.CardNetwork;
import com.xenomi.cms.core.v2.product.dto.CardDesignV2Response;
import com.xenomi.cms.core.v2.product.dto.CreateCardDesignV2Request;
import com.xenomi.cms.core.v2.product.dto.UpdateCardDesignV2Request;
import com.xenomi.cms.core.v2.product.model.CardDesignStatus;
import com.xenomi.cms.core.v2.product.service.CardDesignV2Service;
import io.swagger.v3.oas.annotations.Operation;
import io.swagger.v3.oas.annotations.tags.Tag;
import jakarta.validation.Valid;
import jakarta.validation.constraints.NotBlank;
import jakarta.validation.constraints.Size;
import org.springframework.data.domain.PageRequest;
import org.springframework.data.domain.Pageable;
import org.springframework.data.domain.Sort;
import org.springframework.http.ResponseEntity;
import org.springframework.validation.annotation.Validated;
import org.springframework.web.bind.annotation.*;

@Validated
@RestController
@RequestMapping("/cms/api/v2/institutions/{institutionId}/card-designs")
@Tag(name = "V2 - Issuance Profiles", description = "Card designs, assignments, technology capabilities and CVM rules")
public class CardDesignV2Controller {
    private static final String ACTOR_HEADER = "X-Actor-Id";
    private final CardDesignV2Service service;
    public CardDesignV2Controller(CardDesignV2Service service) { this.service = service; }

    @PostMapping
    @Operation(operationId = "createCardDesignV2", summary = "Create a draft card design version")
    public ResponseEntity<ApiResponseEnvelope<CardDesignV2Response>> create(
            @PathVariable UUID institutionId, @Valid @RequestBody CreateCardDesignV2Request request,
            @RequestHeader(ACTOR_HEADER) @NotBlank @Size(max = 100) String actorId) {
        return ApiResponseHelper.created(service.create(institutionId, request, actorId));
    }
    @GetMapping
    @Operation(operationId = "listCardDesignsV2", summary = "List institution card designs")
    public ResponseEntity<ApiResponseEnvelope<PagedResponse<CardDesignV2Response>>> list(
            @PathVariable UUID institutionId,
            @RequestParam(required = false) CardNetwork cardNetwork,
            @RequestParam(required = false) CardDesignStatus designStatus,
            @RequestParam(required = false) Integer page, @RequestParam(required = false) Integer limit) {
        Pageable base = PageableDefaults.ofOneBased(page, limit);
        Pageable pageable = PageRequest.of(base.getPageNumber(), base.getPageSize(),
                Sort.by(Sort.Direction.ASC, "designCode", "designVersion"));
        return ApiResponseHelper.ok(PagedResponse.of(service.search(institutionId, cardNetwork, designStatus, pageable)));
    }
    @GetMapping("/{designId}")
    @Operation(operationId = "getCardDesignV2", summary = "Get a card design version")
    public ResponseEntity<ApiResponseEnvelope<CardDesignV2Response>> get(
            @PathVariable UUID institutionId, @PathVariable UUID designId) {
        return ApiResponseHelper.ok(service.get(institutionId, designId));
    }
    @PostMapping("/{designId}/actions/update")
    @Operation(operationId = "updateCardDesignV2", summary = "Update or transition a card design")
    public ResponseEntity<ApiResponseEnvelope<CardDesignV2Response>> update(
            @PathVariable UUID institutionId, @PathVariable UUID designId,
            @Valid @RequestBody UpdateCardDesignV2Request request,
            @RequestHeader(ACTOR_HEADER) @NotBlank @Size(max = 100) String actorId) {
        return ApiResponseHelper.ok(service.update(institutionId, designId, request, actorId));
    }
}
