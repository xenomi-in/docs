package com.xenomi.cms.core.v2.card.repository;

import com.xenomi.cms.core.v2.card.entity.CardV2;
import jakarta.persistence.EntityManager;
import jakarta.persistence.PersistenceContext;
import org.springframework.transaction.annotation.Transactional;

public class CardFactsV2RepositoryImpl implements CardFactsV2Repository {
    @PersistenceContext
    private EntityManager entityManager;

    public CardFactsV2RepositoryImpl() {}

    public CardFactsV2RepositoryImpl(EntityManager entityManager) {
        this.entityManager = entityManager;
    }

    @Override
    @Transactional
    public CardV2 saveFactsAndRefresh(CardV2 card) {
        CardV2 managed = card;
        if (!entityManager.contains(card)) {
            if (card.getLifecycleVersion() == null) {
                entityManager.persist(card);
            } else {
                managed = entityManager.merge(card);
            }
        }
        // Hibernate 6.6 UPDATE RETURNING for @Generated can hide the usual stale-row
        // exception. A normal versioned update followed by refresh preserves that check.
        entityManager.flush();
        entityManager.refresh(managed);
        return managed;
    }
}
