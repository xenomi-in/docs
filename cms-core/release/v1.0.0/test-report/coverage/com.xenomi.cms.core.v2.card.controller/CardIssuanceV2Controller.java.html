package com.xenomi.cms.core.v2.card.controller;

import java.util.UUID;

import com.xenomi.cms.core.api.ApiResponseEnvelope;
import com.xenomi.cms.core.api.ApiResponseHelper;
import com.xenomi.cms.core.api.PageableDefaults;
import com.xenomi.cms.core.api.PagedResponse;
import com.xenomi.cms.core.v2.card.dto.CardV2Response;
import com.xenomi.cms.core.v2.card.dto.CreateCardV2Request;
import com.xenomi.cms.core.v2.card.dto.RequestPhysicalCardV2Request;
import com.xenomi.cms.core.v2.card.service.CardIssuanceV2Service;
import io.swagger.v3.oas.annotations.Operation;
import io.swagger.v3.oas.annotations.tags.Tag;
import jakarta.validation.Valid;
import jakarta.validation.constraints.NotBlank;
import jakarta.validation.constraints.Size;
import org.springframework.data.domain.PageRequest;
import org.springframework.data.domain.Pageable;
import org.springframework.data.domain.Sort;
import org.springframework.http.ResponseEntity;
import org.springframework.validation.annotation.Validated;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RequestHeader;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.bind.annotation.RestController;

@Validated
@RestController
@RequestMapping("/cms/api/v2/institutions/{institutionId}/cards")
@Tag(name = "V2 - Cards", description = "Tenant-scoped card issuance and safe card metadata retrieval")
public class CardIssuanceV2Controller {
    private static final String ACTOR_HEADER = "X-Actor-Id";
    private final CardIssuanceV2Service service;
    private final com.xenomi.cms.core.v2.card.lifecycle.command.OriginalCardIssuanceApiV2 issuance;

    public CardIssuanceV2Controller(CardIssuanceV2Service service,
            com.xenomi.cms.core.v2.card.lifecycle.command.OriginalCardIssuanceApiV2 issuance) {
        this.service = service;
        this.issuance = issuance;
    }

    @PostMapping
    @Operation(
            operationId = "issueCardV2",
            summary = "Issue a card",
            description = "Accepts a durable issuance operation. Returns HTTP 202 and a receipt; poll issuance-operations/{operationId} until CONFIRMED. UNKNOWN requires provider reconciliation and is never automatically redispatched. Actor and tenant permissions come from authenticated JWT claims.")
    public ResponseEntity<ApiResponseEnvelope<com.xenomi.cms.core.v2.card.lifecycle.command.LifecycleCommandReceiptV2>> issue(
            @PathVariable UUID institutionId,
            @Valid @RequestBody CreateCardV2Request request,
            @RequestHeader("Idempotency-Key") @NotBlank @Size(max = 128) String idempotencyKey,
            @RequestHeader(ACTOR_HEADER) @NotBlank @Size(max = 100) String actorId) {
        return ResponseEntity.accepted().body(ApiResponseHelper.success(issuance.submit(institutionId, request, idempotencyKey)));
    }

    @GetMapping("/issuance-operations/{operationId}")
    public ResponseEntity<ApiResponseEnvelope<com.xenomi.cms.core.v2.card.lifecycle.command.LifecycleCommandReceiptV2>> operation(
            @PathVariable UUID institutionId, @PathVariable UUID operationId) {
        return ApiResponseHelper.ok(issuance.get(institutionId, operationId));
    }

    @PostMapping("/{cardId}/physical-production-requests")
    @Operation(
            operationId = "requestPhysicalCardV2",
            summary = "Request the physical card after virtual-first issuance",
            description = "Creates one production request for a VIRTUAL_THEN_PHYSICAL_LATER card. The existing card ID and PAN allocation are preserved; no new PAN is generated.")
    public ResponseEntity<ApiResponseEnvelope<CardV2Response>> requestPhysicalCard(
            @PathVariable UUID institutionId,
            @PathVariable UUID cardId,
            @Valid @RequestBody RequestPhysicalCardV2Request request,
            @RequestHeader("Idempotency-Key") @NotBlank @Size(max = 128) String idempotencyKey,
            @RequestHeader(ACTOR_HEADER) @NotBlank @Size(max = 100) String actorId) {
        return ApiResponseHelper.created(service.requestPhysicalCard(
                institutionId, cardId, request, idempotencyKey, actorId));
    }

    @GetMapping("/{cardId}")
    @Operation(operationId = "getCardV2", summary = "Get safe card metadata by UUID")
    public ResponseEntity<ApiResponseEnvelope<CardV2Response>> get(
            @PathVariable UUID institutionId, @PathVariable UUID cardId) {
        return ApiResponseHelper.ok(service.get(institutionId, cardId));
    }

    @GetMapping
    @Operation(
            operationId = "listCardsV2",
            summary = "List cards",
            description = "Lists institution cards, optionally filtered by customer or external account reference. Only masked PAN metadata is returned.")
    public ResponseEntity<ApiResponseEnvelope<PagedResponse<CardV2Response>>> list(
            @PathVariable UUID institutionId,
            @RequestParam(required = false) UUID customerReferenceId,
            @RequestParam(required = false) UUID accountReferenceId,
            @RequestParam(required = false) Integer page,
            @RequestParam(required = false) Integer limit) {
        Pageable base = PageableDefaults.ofOneBased(page, limit);
        Pageable pageable = PageRequest.of(base.getPageNumber(), base.getPageSize(),
                Sort.by(Sort.Direction.DESC, "createdAt"));
        return ApiResponseHelper.ok(PagedResponse.of(
                service.search(institutionId, customerReferenceId, accountReferenceId, pageable)));
    }
}
