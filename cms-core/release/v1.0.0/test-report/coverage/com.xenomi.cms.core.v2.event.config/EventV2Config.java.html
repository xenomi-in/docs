package com.xenomi.cms.core.v2.event.config;

import java.util.concurrent.TimeUnit;

import com.xenomi.cms.core.v2.event.publisher.RetryPolicy;
import com.xenomi.cms.core.v2.event.publisher.auth.ClientCredentialsProvider;
import com.xenomi.cms.core.v2.event.publisher.auth.HubCredentialsProvider;
import com.xenomi.cms.core.v2.event.publisher.auth.StaticBearerCredentialsProvider;
import io.github.resilience4j.circuitbreaker.CircuitBreaker;
import io.github.resilience4j.circuitbreaker.CircuitBreakerConfig;
import org.apache.hc.client5.http.config.ConnectionConfig;
import org.apache.hc.client5.http.config.RequestConfig;
import org.apache.hc.client5.http.impl.classic.CloseableHttpClient;
import org.apache.hc.client5.http.impl.classic.HttpClients;
import org.apache.hc.client5.http.impl.io.PoolingHttpClientConnectionManager;
import org.apache.hc.client5.http.impl.io.PoolingHttpClientConnectionManagerBuilder;
import org.apache.hc.core5.util.TimeValue;
import org.apache.hc.core5.util.Timeout;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.beans.factory.annotation.Qualifier;
import org.springframework.http.client.HttpComponentsClientHttpRequestFactory;
import org.springframework.scheduling.concurrent.ThreadPoolTaskScheduler;
import org.springframework.web.client.RestClient;

/**
 * Wiring for the v2 outbox publisher. Everything here is private to the event module; the
 * application's scheduling and HTTP configuration for other integrations is untouched.
 */
@Configuration
public class EventV2Config {

    public static final String OUTBOX_SCHEDULER = "outboxTaskScheduler";
    public static final String HUB_REST_CLIENT = "cmsNotificationRestClient";
    public static final String HUB_CIRCUIT_BREAKER = "cmsNotificationCircuitBreaker";

    @Bean(name = OUTBOX_SCHEDULER)
    ThreadPoolTaskScheduler outboxTaskScheduler() {
        ThreadPoolTaskScheduler scheduler = new ThreadPoolTaskScheduler();
        scheduler.setPoolSize(2);
        scheduler.setThreadNamePrefix("cms-outbox-");
        scheduler.setDaemon(true);
        scheduler.setWaitForTasksToCompleteOnShutdown(true);
        scheduler.setAwaitTerminationSeconds(35);
        scheduler.initialize();
        return scheduler;
    }

    @Bean(destroyMethod = "close")
    CloseableHttpClient cmsNotificationHttpClient(
            CmsNotificationProperties properties,
            OutboxV2Properties outboxProperties) {
        properties.validateSecureTransport();
        outboxProperties.validateAgainst(properties);
        CmsNotificationProperties.Pool pool = properties.getPool();
        PoolingHttpClientConnectionManager connectionManager = PoolingHttpClientConnectionManagerBuilder.create()
                .setMaxConnTotal(pool.getMaxTotal())
                .setMaxConnPerRoute(pool.getMaxPerRoute())
                .setDefaultConnectionConfig(ConnectionConfig.custom()
                        .setConnectTimeout(Timeout.ofMilliseconds(properties.getConnectTimeout().toMillis()))
                        .setSocketTimeout(Timeout.ofMilliseconds(properties.getReadTimeout().toMillis()))
                        .setValidateAfterInactivity(TimeValue.ofSeconds(5))
                        .build())
                .build();
        return HttpClients.custom()
                .setConnectionManager(connectionManager)
                .setDefaultRequestConfig(RequestConfig.custom()
                        .setResponseTimeout(Timeout.ofMilliseconds(properties.getReadTimeout().toMillis()))
                        .setConnectionRequestTimeout(Timeout.ofMilliseconds(properties.getConnectTimeout().toMillis()))
                        .build())
                .evictIdleConnections(TimeValue.of(pool.getIdleEviction().toMillis(), TimeUnit.MILLISECONDS))
                .evictExpiredConnections()
                .disableAutomaticRetries()
                .build();
    }

    @Bean(name = HUB_REST_CLIENT)
    RestClient cmsNotificationRestClient(
            RestClient.Builder builder,
            CmsNotificationProperties properties,
            @Qualifier("cmsNotificationHttpClient") CloseableHttpClient httpClient) {
        return builder.clone()
                .baseUrl(properties.getBaseUrl().trim())
                .requestFactory(requestFactory(httpClient))
                .build();
    }

    @Bean(name = "cmsNotificationTokenRestClient")
    RestClient cmsNotificationTokenRestClient(
            RestClient.Builder builder,
            @Qualifier("cmsNotificationHttpClient") CloseableHttpClient httpClient) {
        return builder.clone()
                .requestFactory(requestFactory(httpClient))
                .build();
    }

    @Bean
    HubCredentialsProvider hubCredentialsProvider(
            CmsNotificationProperties properties,
            @Qualifier("cmsNotificationTokenRestClient") RestClient tokenClient) {
        if (!properties.isEnabled()) {
            return java.util.Optional::empty;
        }
        CmsNotificationProperties.Auth auth = properties.getAuth();
        return switch (auth.getMode()) {
            case NONE -> java.util.Optional::empty;
            case STATIC_BEARER -> new StaticBearerCredentialsProvider(auth.getStaticToken());
            case OAUTH2_CLIENT_CREDENTIALS -> new ClientCredentialsProvider(tokenClient, auth);
        };
    }

    @Bean(name = HUB_CIRCUIT_BREAKER)
    CircuitBreaker cmsNotificationCircuitBreaker(OutboxV2Properties properties) {
        OutboxV2Properties.CircuitBreaker cb = properties.getCircuitBreaker();
        return CircuitBreaker.of(HUB_CIRCUIT_BREAKER, CircuitBreakerConfig.custom()
                .slidingWindowType(CircuitBreakerConfig.SlidingWindowType.COUNT_BASED)
                .slidingWindowSize(cb.getSlidingWindowSize())
                .minimumNumberOfCalls(cb.getMinimumNumberOfCalls())
                .failureRateThreshold(cb.getFailureRateThreshold())
                .waitDurationInOpenState(cb.getWaitDurationInOpenState())
                .permittedNumberOfCallsInHalfOpenState(cb.getPermittedNumberOfCallsInHalfOpenState())
                .automaticTransitionFromOpenToHalfOpenEnabled(true)
                .build());
    }

    @Bean
    RetryPolicy outboxRetryPolicy(OutboxV2Properties properties) {
        return new RetryPolicy(properties.getRetry());
    }

    private static HttpComponentsClientHttpRequestFactory requestFactory(
            CloseableHttpClient httpClient) {
        return new HttpComponentsClientHttpRequestFactory(httpClient);
    }
}
