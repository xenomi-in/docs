package com.xenomi.cms.core.v2.event.config;

import java.time.Duration;

import org.springframework.boot.context.properties.ConfigurationProperties;
import org.springframework.stereotype.Component;

import lombok.Data;

@Data
@Component
@ConfigurationProperties(prefix = "cms.outbox")
public class OutboxV2Properties {

    private boolean enabled = true;

    /** Delay between poll cycles when the previous cycle found nothing to do. */
    private Duration pollInterval = Duration.ofSeconds(2);

    /** Rows claimed per database round-trip. */
    private int batchSize = 50;

    /** Maximum batches drained in one poll cycle before yielding. */
    private int maxBatchesPerCycle = 20;

    /** Parallel publishers per dispatcher instance. Events never share an aggregate within a batch. */
    private int workerThreads = 4;

    /**
     * Lease held on claimed rows. It covers task queueing and outbound calls
     * across the complete batch, not only the request currently in flight.
     */
    private Duration lease = Duration.ofMinutes(2);

    /** Wait for in-flight publishes on shutdown. */
    private Duration shutdownGrace = Duration.ofSeconds(30);

    /**
     * Allows capture without dispatch only during a deliberate maintenance
     * window. It prevents silently accumulating rows when notification is off.
     */
    private boolean captureOnly;

    private Retry retry = new Retry();
    private CircuitBreaker circuitBreaker = new CircuitBreaker();
    private Retention retention = new Retention();

    public void validateAgainst(CmsNotificationProperties notification) {
        if (batchSize < 1 || batchSize > 1_000) {
            throw new IllegalStateException("cms.outbox.batch-size must be between 1 and 1000");
        }
        if (maxBatchesPerCycle < 1 || maxBatchesPerCycle > 1_000) {
            throw new IllegalStateException("cms.outbox.max-batches-per-cycle must be between 1 and 1000");
        }
        if (workerThreads < 1 || workerThreads > 64) {
            throw new IllegalStateException("cms.outbox.worker-threads must be between 1 and 64");
        }
        requirePositive(pollInterval, "cms.outbox.poll-interval");
        requirePositive(lease, "cms.outbox.lease");
        requirePositive(shutdownGrace, "cms.outbox.shutdown-grace");
        requirePositive(retry.window, "cms.outbox.retry.window");
        requirePositive(retry.initialBackoff, "cms.outbox.retry.initial-backoff");
        requirePositive(retry.maxBackoff, "cms.outbox.retry.max-backoff");
        requirePositive(circuitBreaker.waitDurationInOpenState,
                "cms.outbox.circuit-breaker.wait-duration-in-open-state");
        requirePositive(retention.publishedRetention, "cms.outbox.retention.published-retention");
        if (retry.initialBackoff.compareTo(retry.maxBackoff) > 0) {
            throw new IllegalStateException(
                    "cms.outbox.retry.initial-backoff must be <= cms.outbox.retry.max-backoff");
        }
        if (retry.jitter < 0 || retry.jitter > 1) {
            throw new IllegalStateException("cms.outbox.retry.jitter must be between 0 and 1");
        }
        if (circuitBreaker.slidingWindowSize < 1 || circuitBreaker.minimumNumberOfCalls < 1
                || circuitBreaker.minimumNumberOfCalls > circuitBreaker.slidingWindowSize
                || circuitBreaker.permittedNumberOfCallsInHalfOpenState < 1
                || circuitBreaker.failureRateThreshold <= 0 || circuitBreaker.failureRateThreshold > 100) {
            throw new IllegalStateException("cms.outbox circuit-breaker values are invalid");
        }
        if (retention.purgeBatchSize < 1 || retention.purgeBatchSize > 100_000) {
            throw new IllegalStateException("cms.outbox.retention.purge-batch-size must be between 1 and 100000");
        }
        if (enabled && !notification.isEnabled() && !captureOnly) {
            throw new IllegalStateException(
                    "cms.outbox.enabled requires cms.notification.enabled unless cms.outbox.capture-only=true");
        }
        if (!enabled || !notification.isEnabled()) {
            return;
        }
        // Every task may wait for earlier tasks in the batch. Include one
        // additional bounded token request and a scheduling margin.
        long waves = (batchSize + (long) workerThreads - 1) / workerThreads;
        Duration minimumLease = notification.maximumOutboundCallDuration()
                .multipliedBy(waves + 1)
                .plusSeconds(5);
        if (lease.compareTo(minimumLease) < 0) {
            throw new IllegalStateException(
                    "cms.outbox.lease must be at least " + minimumLease
                            + " for the configured batch, workers, and notification timeouts");
        }
        if (shutdownGrace.compareTo(notification.maximumOutboundCallDuration()) < 0) {
            throw new IllegalStateException(
                    "cms.outbox.shutdown-grace must cover one notification request");
        }
    }

    private static void requirePositive(Duration value, String property) {
        if (value == null || value.isZero() || value.isNegative()) {
            throw new IllegalStateException(property + " must be positive");
        }
    }

    @Data
    public static class Retry {
        /** Transient failures are retried until this window elapses, then dead-lettered. */
        private Duration window = Duration.ofHours(72);
        private Duration initialBackoff = Duration.ofSeconds(2);
        private Duration maxBackoff = Duration.ofMinutes(5);
        /** Jitter fraction (0..1) applied to each backoff to avoid thundering herds across instances. */
        private double jitter = 0.2;
    }

    @Data
    public static class CircuitBreaker {
        private int slidingWindowSize = 20;
        private int minimumNumberOfCalls = 10;
        private float failureRateThreshold = 50f;
        private Duration waitDurationInOpenState = Duration.ofSeconds(30);
        private int permittedNumberOfCallsInHalfOpenState = 3;
    }

    @Data
    public static class Retention {
        private boolean enabled = true;
        /** Published rows older than this are purged. card_event remains the audit record. */
        private Duration publishedRetention = Duration.ofDays(14);
        private int purgeBatchSize = 5000;
        /** Spring cron expression, evaluated in UTC. */
        private String cron = "0 30 2 * * *";
    }
}
