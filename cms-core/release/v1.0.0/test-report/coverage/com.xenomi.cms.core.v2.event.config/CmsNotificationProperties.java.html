package com.xenomi.cms.core.v2.event.config;

import java.time.Duration;
import java.net.URI;

import org.springframework.boot.context.properties.ConfigurationProperties;
import org.springframework.stereotype.Component;

import lombok.Data;

@Data
@Component
@ConfigurationProperties(prefix = "cms.notification")
public class CmsNotificationProperties {

    private boolean enabled = true;
    private String baseUrl = "http://localhost:8085";
    private String publishPath = "/api/v2/institutions/{institutionId}/notifications/publish";
    private Duration connectTimeout = Duration.ofSeconds(2);
    private Duration readTimeout = Duration.ofSeconds(5);

    private Pool pool = new Pool();
    private Auth auth = new Auth();

    /**
     * Validates configuration that cannot be expressed with field annotations.
     * Loopback HTTP is permitted only for local development; credentials are always required.
     */
    public void validateSecureTransport() {
        URI uri;
        try {
            uri = URI.create(baseUrl);
        } catch (IllegalArgumentException exception) {
            throw new IllegalStateException("cms.notification.base-url must be an absolute URI", exception);
        }
        String scheme = uri.getScheme();
        String host = uri.getHost();
        if (scheme == null || host == null) {
            throw new IllegalStateException("cms.notification.base-url must be an absolute URI");
        }
        if (isNonPositive(connectTimeout) || isNonPositive(readTimeout)
                || isNonPositive(pool.idleEviction)) {
            throw new IllegalStateException("cms.notification timeouts must be positive");
        }
        if (pool.maxTotal < 1 || pool.maxPerRoute < 1 || pool.maxPerRoute > pool.maxTotal) {
            throw new IllegalStateException("cms.notification pool sizes must be positive and max-per-route <= max-total");
        }
        boolean loopback = "localhost".equalsIgnoreCase(host)
                || "127.0.0.1".equals(host)
                || "::1".equals(host);
        if (enabled && !loopback && !"https".equalsIgnoreCase(scheme)) {
            throw new IllegalStateException(
                    "cms.notification.base-url must use HTTPS outside local development");
        }
        if (enabled && auth.mode == Auth.Mode.NONE) {
            throw new IllegalStateException(
                    "cms.notification.auth.mode must not be NONE: notification requires a Keycloak bearer token");
        }
        if (!enabled) {
            return;
        }
        if (publishPath == null || !publishPath.startsWith("/") || publishPath.startsWith("//")
                || !publishPath.contains("/institutions/{institutionId}/")) {
            throw new IllegalStateException("cms.notification.publish-path must contain /institutions/{institutionId}/");
        }
        switch (auth.mode) {
            case STATIC_BEARER -> require(auth.staticToken, "cms.notification.auth.static-token");
            case OAUTH2_CLIENT_CREDENTIALS -> {
                require(auth.tokenUri, "cms.notification.auth.token-uri");
                requireHttpsOutsideLoopback(auth.tokenUri, "cms.notification.auth.token-uri");
                require(auth.clientId, "cms.notification.auth.client-id");
                require(auth.clientSecret, "cms.notification.auth.client-secret");
                if (isNonPositive(auth.refreshSkew)) {
                    throw new IllegalStateException("cms.notification.auth.refresh-skew must be positive");
                }
            }
            case NONE -> {
                // Only reachable when publishing is disabled (handled above).
            }
        }
    }

    Duration maximumOutboundCallDuration() {
        return connectTimeout.plus(readTimeout);
    }

    private static boolean isNonPositive(Duration value) {
        return value == null || value.isZero() || value.isNegative();
    }

    private static void require(String value, String property) {
        if (value == null || value.isBlank()) {
            throw new IllegalStateException(property + " is required");
        }
    }

    private static void requireHttpsOutsideLoopback(String value, String property) {
        URI uri;
        try {
            uri = URI.create(value);
        } catch (IllegalArgumentException exception) {
            throw new IllegalStateException(property + " must be an absolute URI", exception);
        }
        String host = uri.getHost();
        boolean loopback = "localhost".equalsIgnoreCase(host)
                || "127.0.0.1".equals(host)
                || "::1".equals(host);
        if (uri.getScheme() == null || host == null
                || (!loopback && !"https".equalsIgnoreCase(uri.getScheme()))) {
            throw new IllegalStateException(property + " must use HTTPS outside local development");
        }
    }

    @Data
    public static class Pool {
        private int maxTotal = 32;
        private int maxPerRoute = 32;
        private Duration idleEviction = Duration.ofSeconds(30);
    }

    @Data
    public static class Auth {
        public enum Mode { NONE, STATIC_BEARER, OAUTH2_CLIENT_CREDENTIALS }

        private Mode mode = Mode.OAUTH2_CLIENT_CREDENTIALS;
        /** Pre-issued service token, sourced from a secret store. Used when mode = STATIC_BEARER. */
        private String staticToken;
        /** OAuth2 client-credentials settings. Used when mode = OAUTH2_CLIENT_CREDENTIALS. */
        private String tokenUri;
        private String clientId;
        private String clientSecret;
        private String scope;
        /** Refresh the cached token this long before it expires. */
        private Duration refreshSkew = Duration.ofSeconds(30);
    }
}
