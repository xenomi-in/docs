package com.xenomi.cms.core.v2.event.writer;

import java.util.UUID;

import com.fasterxml.jackson.core.JsonProcessingException;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.xenomi.cms.core.v2.common.DatabaseUuidV7Generator;
import com.xenomi.cms.core.v2.event.config.OutboxV2Properties;
import com.xenomi.cms.core.v2.event.correlation.CorrelationIdResolver;
import com.xenomi.cms.core.v2.event.mapper.NotificationEnvelopeFactory;
import com.xenomi.cms.core.v2.event.model.NotificationEnvelope;
import com.xenomi.cms.core.v2.event.model.OutboxEventCommand;
import com.xenomi.cms.core.v2.event.repository.OutboxEventV2Repository;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.stereotype.Component;
import org.springframework.transaction.support.TransactionSynchronizationManager;

/**
 * Writes the Falcon envelope into {@code cms_v2.outbox_event} as part of the caller's
 * transaction. The write is atomic with the business mutation: if it fails, the business
 * change fails too. That is the guarantee the outbox pattern exists to provide, so no
 * failure is swallowed here.
 */
@Component
public class OutboxEventWriter {

    private static final Logger log = LoggerFactory.getLogger(OutboxEventWriter.class);

    private final OutboxV2Properties properties;
    private final OutboxEventV2Repository repository;
    private final DatabaseUuidV7Generator uuidGenerator;
    private final CorrelationIdResolver correlationIdResolver;
    private final NotificationEnvelopeFactory envelopeFactory;
    private final ObjectMapper objectMapper;

    public OutboxEventWriter(
            OutboxV2Properties properties,
            OutboxEventV2Repository repository,
            DatabaseUuidV7Generator uuidGenerator,
            CorrelationIdResolver correlationIdResolver,
            NotificationEnvelopeFactory envelopeFactory,
            ObjectMapper objectMapper) {
        this.properties = properties;
        this.repository = repository;
        this.uuidGenerator = uuidGenerator;
        this.correlationIdResolver = correlationIdResolver;
        this.envelopeFactory = envelopeFactory;
        this.objectMapper = objectMapper;
    }

    /**
     * @return the outbox event id (also the envelope {@code eventId}) when a row was
     *         inserted, or {@code null} when the outbox is disabled or an identical event
     *         already exists
     */
    public UUID enqueue(OutboxEventCommand command) {
        if (!properties.isEnabled()) {
            return null;
        }
        if (!TransactionSynchronizationManager.isActualTransactionActive()) {
            throw new IllegalStateException(
                    "Outbox events must be written inside the business transaction: "
                            + command.eventType() + " " + command.aggregateId());
        }
        String correlation = correlationIdResolver.resolve(command.correlationId());
        UUID eventId = uuidGenerator.generate();
        long aggregateSequence = repository.nextAggregateSequence(
                command.institutionId(), command.aggregateType(), command.aggregateId());
        NotificationEnvelope envelope = envelopeFactory.create(eventId, command, correlation);

        int inserted = repository.insertIfAbsent(
                eventId,
                command.institutionId(),
                command.aggregateType(),
                command.aggregateId(),
                command.eventType(),
                aggregateSequence,
                toJson(envelope),
                correlationIdResolver.toUuid(correlation),
                command.idempotencyKey());
        if (inserted == 0) {
            log.info("Outbox event already recorded, skipping eventType={} aggregateId={} idempotencyKey={}",
                    command.eventType(), command.aggregateId(), command.idempotencyKey());
            return null;
        }
        return eventId;
    }

    private String toJson(NotificationEnvelope envelope) {
        try {
            return objectMapper.writeValueAsString(envelope);
        } catch (JsonProcessingException exception) {
            throw new IllegalStateException("Notification envelope is not serialisable", exception);
        }
    }
}
