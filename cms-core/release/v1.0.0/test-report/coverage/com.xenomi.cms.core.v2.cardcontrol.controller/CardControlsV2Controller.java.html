package com.xenomi.cms.core.v2.cardcontrol.controller;

import java.util.UUID;

import com.xenomi.cms.core.api.ApiResponseEnvelope;
import com.xenomi.cms.core.api.ApiResponseHelper;
import com.xenomi.cms.core.v2.cardcontrol.dto.CardControlsV2Response;
import com.xenomi.cms.core.v2.cardcontrol.dto.UpdateCardControlsV2Request;
import com.xenomi.cms.core.v2.cardcontrol.service.CardControlsV2Service;
import io.swagger.v3.oas.annotations.Operation;
import io.swagger.v3.oas.annotations.tags.Tag;
import jakarta.validation.Valid;
import jakarta.validation.constraints.NotBlank;
import jakarta.validation.constraints.Size;
import org.springframework.http.ResponseEntity;
import org.springframework.validation.annotation.Validated;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PutMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RequestHeader;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RestController;

@Validated
@RestController
@RequestMapping("/cms/api/v2/institutions/{institutionId}/cards/{cardId}/controls")
@Tag(name = "V2 - Card Controls",
        description = "Domestic/international ATM, POS, ECOM, MOTO and contactless controls")
public class CardControlsV2Controller {
    private static final String ACTOR_HEADER = "X-Actor-Id";
    private final CardControlsV2Service service;

    public CardControlsV2Controller(CardControlsV2Service service) {
        this.service = service;
    }

    @GetMapping
    @Operation(
            operationId = "getCardControlsV2",
            summary = "Get current card controls",
            description = "Returns ten dimensions: ATM, POS_CONTACT, ECOM, MOTO and CONTACTLESS_POS for domestic and international use. Includes international master switch and synchronization status. Limits are per transaction.")
    public ResponseEntity<ApiResponseEnvelope<CardControlsV2Response>> get(
            @PathVariable UUID institutionId, @PathVariable UUID cardId) {
        return ApiResponseHelper.ok(service.get(institutionId, cardId));
    }

    @PutMapping
    @Operation(
            operationId = "updateCardControlsV2",
            summary = "Update card controls and synchronize with Switch",
            description = "Validates product capabilities and Accounting limits. With switch-sync enabled, commits the complete snapshot to a durable queue and returns PENDING_SWITCH until Switch acknowledges it. GET the card switch-sync endpoint to inspect progress. Without switch-sync, uses the legacy synchronous client.")
    public ResponseEntity<ApiResponseEnvelope<CardControlsV2Response>> update(
            @PathVariable UUID institutionId,
            @PathVariable UUID cardId,
            @Valid @RequestBody UpdateCardControlsV2Request request,
            @RequestHeader("Idempotency-Key") @NotBlank @Size(max = 128) String idempotencyKey,
            @RequestHeader(value = "X-Request-Id", required = false) @Size(max = 128) String requestId,
            @RequestHeader(ACTOR_HEADER) @NotBlank @Size(max = 100) String actorId) {
        return ApiResponseHelper.ok(service.update(
                institutionId, cardId, request, idempotencyKey, requestId, actorId));
    }
}
