package com.xenomi.cms.core.v2.institution.controller;

import java.util.UUID;

import com.xenomi.cms.core.api.ApiResponseEnvelope;
import com.xenomi.cms.core.api.ApiResponseHelper;
import com.xenomi.cms.core.api.PageableDefaults;
import com.xenomi.cms.core.api.PagedResponse;
import com.xenomi.cms.core.v2.institution.dto.CreateInstitutionV2Request;
import com.xenomi.cms.core.v2.institution.dto.InstitutionV2Response;
import com.xenomi.cms.core.v2.institution.dto.UpdateInstitutionV2Request;
import com.xenomi.cms.core.v2.institution.model.InstitutionStatus;
import com.xenomi.cms.core.v2.institution.model.InstitutionType;
import com.xenomi.cms.core.v2.institution.service.InstitutionV2Service;

import io.swagger.v3.oas.annotations.Operation;
import io.swagger.v3.oas.annotations.tags.Tag;
import jakarta.validation.Valid;
import jakarta.validation.constraints.NotBlank;
import jakarta.validation.constraints.Size;

import org.springframework.data.domain.PageRequest;
import org.springframework.data.domain.Pageable;
import org.springframework.data.domain.Sort;
import org.springframework.http.ResponseEntity;
import org.springframework.validation.annotation.Validated;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RequestHeader;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.bind.annotation.RestController;

@Validated
@RestController
@RequestMapping("/cms/api/v2/institutions")
@Tag(name = "V2 - Institutions", description = "Institution onboarding, discovery and lifecycle")
public class InstitutionV2Controller {

    private static final String ACTOR_HEADER = "X-Actor-Id";

    private final InstitutionV2Service service;

    public InstitutionV2Controller(InstitutionV2Service service) {
        this.service = service;
    }

    @PostMapping
    @Operation(summary = "Onboard an institution")
    public ResponseEntity<ApiResponseEnvelope<InstitutionV2Response>> create(
            @Valid @RequestBody CreateInstitutionV2Request request,
            @RequestHeader(ACTOR_HEADER) @NotBlank @Size(max = 50) String actorId) {
        return ApiResponseHelper.created(service.create(request, actorId));
    }

    @GetMapping("/{institutionId}")
    @Operation(summary = "Get an institution by UUID")
    public ResponseEntity<ApiResponseEnvelope<InstitutionV2Response>> get(
            @PathVariable UUID institutionId) {
        return ApiResponseHelper.ok(service.get(institutionId));
    }

    @GetMapping("/code/{institutionCode}")
    @Operation(summary = "Get an institution by code")
    public ResponseEntity<ApiResponseEnvelope<InstitutionV2Response>> getByCode(
            @PathVariable String institutionCode) {
        return ApiResponseHelper.ok(service.getByCode(institutionCode));
    }

    @GetMapping
    @Operation(summary = "Search and list institutions")
    public ResponseEntity<ApiResponseEnvelope<PagedResponse<InstitutionV2Response>>> search(
            @RequestParam(required = false) String query,
            @RequestParam(required = false) InstitutionType institutionType,
            @RequestParam(required = false) InstitutionStatus status,
            @RequestParam(required = false) Boolean visaMemberFlag,
            @RequestParam(required = false) Boolean rupayMemberFlag,
            @RequestParam(required = false) Integer page,
            @RequestParam(required = false) Integer limit) {
        Pageable base = PageableDefaults.ofOneBased(page, limit);
        Pageable pageable = PageRequest.of(base.getPageNumber(), base.getPageSize(),
                Sort.by(Sort.Direction.ASC, "institutionCode"));
        return ApiResponseHelper.ok(PagedResponse.of(service.search(
                query, institutionType, status, visaMemberFlag, rupayMemberFlag, pageable)));
    }

    @PostMapping("/{institutionId}/actions/update")
    @Operation(summary = "Apply an explicit partial-update action to an institution")
    public ResponseEntity<ApiResponseEnvelope<InstitutionV2Response>> update(
            @PathVariable UUID institutionId,
            @Valid @RequestBody UpdateInstitutionV2Request request,
            @RequestHeader(ACTOR_HEADER) @NotBlank @Size(max = 50) String actorId) {
        return ApiResponseHelper.ok(service.update(institutionId, request, actorId));
    }
}
