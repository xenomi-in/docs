package com.xenomi.cms.core.v2.institution.controller;

import java.util.UUID;

import com.xenomi.cms.core.api.ApiResponseEnvelope;
import com.xenomi.cms.core.api.ApiResponseHelper;
import com.xenomi.cms.core.api.PageableDefaults;
import com.xenomi.cms.core.api.PagedResponse;
import com.xenomi.cms.core.v2.institution.dto.CreateDataKeyAssignmentV2Request;
import com.xenomi.cms.core.v2.institution.dto.DataKeyAssignmentV2Response;
import com.xenomi.cms.core.v2.institution.dto.OperationalPolicyV2Response;
import com.xenomi.cms.core.v2.institution.dto.SecurityPolicyV2Response;
import com.xenomi.cms.core.v2.institution.dto.UpdateDataKeyAssignmentStatusV2Request;
import com.xenomi.cms.core.v2.institution.dto.UpdateOperationalPolicyV2Request;
import com.xenomi.cms.core.v2.institution.dto.UpdateSecurityPolicyV2Request;
import com.xenomi.cms.core.v2.institution.model.DataKeyAssignmentStatus;
import com.xenomi.cms.core.v2.hsm.model.KeyPurpose;
import com.xenomi.cms.core.v2.institution.service.InstitutionDataKeyAssignmentV2Service;
import com.xenomi.cms.core.v2.institution.service.InstitutionPolicyV2Service;

import io.swagger.v3.oas.annotations.Operation;
import io.swagger.v3.oas.annotations.tags.Tag;
import jakarta.validation.Valid;
import jakarta.validation.constraints.NotBlank;
import jakarta.validation.constraints.Size;

import org.springframework.data.domain.PageRequest;
import org.springframework.data.domain.Pageable;
import org.springframework.data.domain.Sort;
import org.springframework.http.ResponseEntity;
import org.springframework.validation.annotation.Validated;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.PutMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RequestHeader;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.bind.annotation.RestController;

@Validated
@RestController
@RequestMapping("/cms/api/v2/institutions/{institutionId}")
@Tag(name = "V2 - Institution Policies", description = "Operational, security and institution data-key policies")
public class InstitutionPolicyV2Controller {

    private static final String ACTOR_HEADER = "X-Actor-Id";

    private final InstitutionPolicyV2Service policyService;
    private final InstitutionDataKeyAssignmentV2Service dataKeyService;

    public InstitutionPolicyV2Controller(
            InstitutionPolicyV2Service policyService,
            InstitutionDataKeyAssignmentV2Service dataKeyService) {
        this.policyService = policyService;
        this.dataKeyService = dataKeyService;
    }

    @GetMapping("/operational-policy")
    @Operation(summary = "Get the institution operational policy")
    public ResponseEntity<ApiResponseEnvelope<OperationalPolicyV2Response>> getOperational(
            @PathVariable UUID institutionId) {
        return ApiResponseHelper.ok(policyService.getOperational(institutionId));
    }

    @PutMapping("/operational-policy")
    @Operation(summary = "Replace the institution operational policy")
    public ResponseEntity<ApiResponseEnvelope<OperationalPolicyV2Response>> updateOperational(
            @PathVariable UUID institutionId,
            @Valid @RequestBody UpdateOperationalPolicyV2Request request,
            @RequestHeader(ACTOR_HEADER) @NotBlank @Size(max = 100) String actorId) {
        return ApiResponseHelper.ok(policyService.updateOperational(institutionId, request, actorId));
    }

    @GetMapping("/security-policy")
    @Operation(summary = "Get the institution security policy",
            description = "Session and token lifetimes are application-level identity-provider settings; "
                    + "they are not part of the institution security policy.")
    public ResponseEntity<ApiResponseEnvelope<SecurityPolicyV2Response>> getSecurity(
            @PathVariable UUID institutionId) {
        return ApiResponseHelper.ok(policyService.getSecurity(institutionId));
    }

    @PutMapping("/security-policy")
    @Operation(summary = "Replace the institution security policy",
            description = "Session timeout is not configurable through this endpoint. Configure session "
                    + "and token lifetimes in the application identity provider.")
    public ResponseEntity<ApiResponseEnvelope<SecurityPolicyV2Response>> updateSecurity(
            @PathVariable UUID institutionId,
            @Valid @RequestBody UpdateSecurityPolicyV2Request request,
            @RequestHeader(ACTOR_HEADER) @NotBlank @Size(max = 100) String actorId) {
        return ApiResponseHelper.ok(policyService.updateSecurity(institutionId, request, actorId));
    }

    @PostMapping("/data-key-assignments")
    @Operation(summary = "Create an institution data-key assignment")
    public ResponseEntity<ApiResponseEnvelope<DataKeyAssignmentV2Response>> createDataKeyAssignment(
            @PathVariable UUID institutionId,
            @Valid @RequestBody CreateDataKeyAssignmentV2Request request,
            @RequestHeader(ACTOR_HEADER) @NotBlank @Size(max = 100) String actorId) {
        return ApiResponseHelper.created(dataKeyService.create(institutionId, request, actorId));
    }

    @GetMapping("/data-key-assignments")
    @Operation(summary = "List institution data-key assignments")
    public ResponseEntity<ApiResponseEnvelope<PagedResponse<DataKeyAssignmentV2Response>>> searchDataKeys(
            @PathVariable UUID institutionId,
            @RequestParam(required = false) KeyPurpose keyPurpose,
            @RequestParam(required = false) DataKeyAssignmentStatus status,
            @RequestParam(required = false) Integer page,
            @RequestParam(required = false) Integer limit) {
        Pageable base = PageableDefaults.ofOneBased(page, limit);
        Pageable pageable = PageRequest.of(base.getPageNumber(), base.getPageSize(),
                Sort.by(Sort.Direction.DESC, "assignmentVersion"));
        return ApiResponseHelper.ok(PagedResponse.of(
                dataKeyService.search(institutionId, keyPurpose, status, pageable)));
    }

    @GetMapping("/data-key-assignments/{assignmentId}")
    @Operation(summary = "Get an institution data-key assignment")
    public ResponseEntity<ApiResponseEnvelope<DataKeyAssignmentV2Response>> getDataKeyAssignment(
            @PathVariable UUID institutionId,
            @PathVariable UUID assignmentId) {
        return ApiResponseHelper.ok(dataKeyService.get(institutionId, assignmentId));
    }

    @PutMapping("/data-key-assignments/{assignmentId}/status")
    @Operation(summary = "Change an institution data-key assignment status")
    public ResponseEntity<ApiResponseEnvelope<DataKeyAssignmentV2Response>> changeDataKeyStatus(
            @PathVariable UUID institutionId,
            @PathVariable UUID assignmentId,
            @Valid @RequestBody UpdateDataKeyAssignmentStatusV2Request request,
            @RequestHeader(ACTOR_HEADER) @NotBlank @Size(max = 100) String actorId) {
        return ApiResponseHelper.ok(
                dataKeyService.changeStatus(institutionId, assignmentId, request, actorId));
    }
}
