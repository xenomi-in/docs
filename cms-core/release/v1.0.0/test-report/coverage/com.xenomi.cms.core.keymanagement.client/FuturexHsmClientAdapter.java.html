package com.xenomi.cms.core.keymanagement.client;

import com.xenomi.cms.core.hms.service.NettyHsmService;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.stereotype.Component;

/**
 * HSM client adapter for Futurex HSM via Netty-based communication.
 * Delegates key generation to NettyHsmService and maps results to the abstraction.
 *
 * @author LogicPoint Systems
 * @version 1.0.0
 */
@Component
public class FuturexHsmClientAdapter implements HsmClient {

    private static final Logger log = LoggerFactory.getLogger(FuturexHsmClientAdapter.class);
    private static final String PROVIDER_ID = "FUTUREX";

    private final NettyHsmService nettyHsmService;

    public FuturexHsmClientAdapter(NettyHsmService nettyHsmService) {
        this.nettyHsmService = nettyHsmService;
    }

    @Override
    public KeyGenerationResult generateKey(KeyGenerationRequest request) {
        validateSupportedGenerationFormat(request);
        try {
            String rawResult = nettyHsmService.generateWorkingKey(resolveKeyType(request.getKeyType()));
            if (rawResult == null || rawResult.isEmpty()) {
                throw new HsmKeyGenerationException("HSM returned no result for key generation");
            }
            String[] parts = rawResult.split(";");
            if (parts.length < 2 || parts[0].isBlank()
                    || !parts[1].trim().matches("[0-9A-Fa-f]{4,16}")) {
                log.warn("HSM returned malformed key-generation response: segments={}, "
                                + "keyBlockPresent={}, kcvLength={}",
                        parts.length,
                        parts.length > 0 && !parts[0].isBlank(),
                        parts.length > 1 ? parts[1].trim().length() : 0);
                throw new HsmKeyGenerationException("HSM returned an invalid key-generation response");
            }
            String encryptedKeyBlock = parts[0].trim();
            String keyCheckValue = parts[1].trim();
            String keyIndex = deriveKeyIndex(request);

            return KeyGenerationResult.builder()
                    .keyCheckValue(keyCheckValue)
                    .keyIndex(keyIndex)
                    .keyType(request.getKeyType())
                    .hsmIdentifier(null)
                    .hsmPartition(null)
                    // The currently configured GWKS command returns a legacy TDES key block.
                    .tr31Format(false)
                    .encryptedKeyBlock(encryptedKeyBlock)
                    .build();
        } catch (HsmKeyGenerationException e) {
            throw e;
        } catch (Exception e) {
            log.error("HSM key generation failed for keyType={}, scheme={}",
                    request.getKeyType(), request.getScheme(), e);
            throw new HsmKeyGenerationException("HSM provider request failed", e);
        }
    }

    @Override
    public boolean testConnection() {
        try {
            return nettyHsmService.testConnection();
        } catch (Exception e) {
            log.warn("HSM connection test failed: {}", e.getMessage());
            return false;
        }
    }

    @Override
    public String getProviderId() {
        return PROVIDER_ID;
    }

    private static String resolveKeyType(String keyType) {
        if (keyType == null) return "PVK";
        return switch (keyType.toUpperCase()) {
            case "PVK" -> "PVK";
            case "DKE", "DEK", "HMAC" -> "DKE";
            case "PGK" -> "PGK";
            case "CVK", "CVV" -> "CVK";
            case "ZPK" -> "ZPK";
            default -> keyType;
        };
    }

    private static void validateSupportedGenerationFormat(KeyGenerationRequest request) {
        String scheme = request.getScheme();
        if (request.isTr31Format() || "TR31".equalsIgnoreCase(scheme)) {
            throw new HsmKeyGenerationException(
                    "TR-31 generation is not configured for the Futurex GWKS adapter");
        }
        if ("AES".equalsIgnoreCase(scheme)) {
            throw new HsmKeyGenerationException(
                    "AES generation is not configured for the Futurex GWKS adapter");
        }
        if ("TDES".equalsIgnoreCase(scheme)
                && !Integer.valueOf(112).equals(request.getKeyLengthBits())) {
            throw new HsmKeyGenerationException(
                    "The Futurex GWKS adapter supports only double-length TDES generation");
        }
        // V1 uses scheme for the card network (for example VISA), so unknown scheme
        // values retain the existing legacy double-length TDES behaviour.
    }

    private static String deriveKeyIndex(KeyGenerationRequest request) {
        String prefix = request.getKeyScope() != null && "BIN".equalsIgnoreCase(request.getKeyScope())
                ? "BIN"
                : "GLB";
        String bin = request.getBinNumber() != null ? request.getBinNumber().replaceAll("\\D", "") : "000";
        return String.format("%s_%s_%s", prefix, request.getKeyType(), bin.substring(Math.max(0, bin.length() - 4)));
    }
}
