package com.xenomi.cms.core.v2.acs;

import java.util.Arrays;
import java.util.Set;
import java.util.stream.Collectors;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Component;
import org.springframework.security.authorization.AuthorizationDecision;
import org.springframework.security.authorization.AuthorizationManager;
import org.springframework.security.authorization.AuthorizationResult;
import org.springframework.security.core.Authentication;
import org.springframework.security.oauth2.server.resource.authentication.JwtAuthenticationToken;
import org.springframework.security.web.access.intercept.RequestAuthorizationContext;
import java.util.function.Supplier;

@Component
public class AcsContactAuthorization implements AuthorizationManager<RequestAuthorizationContext> {
    private final Set<String> subjects;
    public AcsContactAuthorization(@Value("${cms.security.acs.allowed-subjects:}") String subjects) {
        this.subjects = Arrays.stream(subjects.split(",")).map(String::trim)
                .filter(s -> !s.isEmpty()).collect(Collectors.toUnmodifiableSet());
    }
    @Override
    public AuthorizationResult authorize(Supplier<? extends Authentication> supplier, RequestAuthorizationContext context) {
        Authentication auth = supplier.get();
        boolean allowed = auth instanceof JwtAuthenticationToken jwt && auth.isAuthenticated()
                && subjects.contains(jwt.getToken().getSubject())
                && Boolean.TRUE.equals(jwt.getToken().getClaims().get("platform_access"))
                && auth.getAuthorities().stream().anyMatch(a -> a.getAuthority().equals("PERM_cms:acs-contact:read"));
        return new AuthorizationDecision(allowed);
    }
}
