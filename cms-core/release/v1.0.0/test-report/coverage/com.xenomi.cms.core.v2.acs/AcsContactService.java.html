package com.xenomi.cms.core.v2.acs;

import java.time.OffsetDateTime;
import java.time.ZoneOffset;
import java.util.*;
import com.xenomi.cms.core.bin.util.LuhnAlgorithm;
import com.xenomi.cms.core.exception.ResourceNotFoundException;
import com.xenomi.cms.core.v2.bin.repository.BinRangeV2Repository;
import com.xenomi.cms.core.v2.card.repository.CardV2Repository;
import com.xenomi.cms.core.v2.card.repository.CustomerReferenceV2Repository;
import com.xenomi.cms.core.v2.card.lifecycle.repository.PanLookupIdentityV2Repository;
import com.xenomi.cms.core.v2.card.security.CmsCardSecurityService;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

@Service
public class AcsContactService {
    private final BinRangeV2Repository ranges;
    private final PanLookupIdentityV2Repository identities;
    private final CmsCardSecurityService security;
    private final CardV2Repository cards;
    private final CustomerReferenceV2Repository customers;
    public AcsContactService(BinRangeV2Repository ranges, PanLookupIdentityV2Repository identities,
            CmsCardSecurityService security, CardV2Repository cards, CustomerReferenceV2Repository customers) {
        this.ranges=ranges; this.identities=identities; this.security=security; this.cards=cards; this.customers=customers;
    }
    @Transactional(readOnly=true)
    public Contact lookup(String pan) {
        if (pan == null || !pan.matches("[0-9]{12,19}") || !LuhnAlgorithm.validate(pan))
            throw new IllegalArgumentException("A valid card number is required");
        List<UUID> institutions = ranges.findInstitutionsForPan(pan, pan.length());
        if (institutions.size() > 1) throw new IllegalArgumentException("Card institution resolution is ambiguous");
        if (institutions.isEmpty()) throw unavailable();
        UUID institution = institutions.getFirst();
        OffsetDateTime now = OffsetDateTime.now(ZoneOffset.UTC);
        Set<UUID> families = new HashSet<>();
        for (UUID key : identities.findEffectiveKeyIds(institution, now)) {
            for (boolean legacy : new boolean[]{false, true}) {
                String hash = security.hashForLookup(institution, key, pan, legacy);
                identities.findEffective(institution, key, hash, now).ifPresent(i -> families.add(i.getPanAllocationId()));
            }
        }
        if (families.size() > 1) throw new IllegalArgumentException("Card identity resolution is ambiguous");
        if (families.isEmpty()) throw unavailable();
        // Renewal/replacement can leave several card rows for one PAN. All must identify the same customer.
        Set<UUID> customerIds = new HashSet<>();
        cards.findByInstitutionIdAndPanAllocationIdOrderByIssuedAtAscCardIdAsc(institution, families.iterator().next())
                .forEach(c -> customerIds.add(c.getCustomerReferenceId()));
        if (customerIds.size() != 1) throw unavailable();
        var customer = customers.findByCustomerReferenceIdAndInstitutionId(customerIds.iterator().next(), institution)
                .orElseThrow(AcsContactService::unavailable);
        if (blank(customer.getRegisteredMobile()) && blank(customer.getRegisteredEmail())) throw unavailable();
        return new Contact(blank(customer.getRegisteredMobile()) ? null : customer.getRegisteredMobile(),
                blank(customer.getRegisteredEmail()) ? null : customer.getRegisteredEmail());
    }
    private static boolean blank(String value) { return value == null || value.isBlank(); }
    private static ResourceNotFoundException unavailable() { return new ResourceNotFoundException("Cardholder contact is unavailable"); }
    public record Contact(String mobile, String email) {
        @Override public String toString() { return "Contact[redacted]"; }
    }
}
