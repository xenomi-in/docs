package com.xenomi.cms.core.v2.event.admin;

import java.time.OffsetDateTime;
import java.util.Map;
import java.util.UUID;

import com.xenomi.cms.core.v2.event.entity.OutboxEventV2;
import com.xenomi.cms.core.v2.event.model.OutboxPublicationStatus;
import io.swagger.v3.oas.annotations.media.Schema;
import lombok.AllArgsConstructor;
import lombok.Builder;
import lombok.Data;
import lombok.NoArgsConstructor;

@Data @Builder @NoArgsConstructor @AllArgsConstructor
@Schema(description = "Operator view of one outbox publication record")
public class OutboxEventV2Response {
    private UUID outboxEventId;
    private UUID institutionId;
    private String aggregateType;
    private String aggregateId;
    private String eventType;
    private OutboxPublicationStatus publicationStatus;
    private OffsetDateTime occurredAt;
    private OffsetDateTime publishedAt;
    private int retryCount;
    private OffsetDateTime firstFailedAt;
    private OffsetDateTime nextRetryAt;
    private String lastError;
    private String lockedBy;
    private OffsetDateTime lockedUntil;
    @Schema(description = "Falcon envelope exactly as it is sent to cms_notification")
    private Map<String, Object> envelope;

    static OutboxEventV2Response from(OutboxEventV2 event) {
        return OutboxEventV2Response.builder()
                .outboxEventId(event.getOutboxEventId())
                .institutionId(event.getInstitutionId())
                .aggregateType(event.getAggregateType())
                .aggregateId(event.getAggregateId())
                .eventType(event.getEventType())
                .publicationStatus(event.getPublicationStatus())
                .occurredAt(event.getOccurredAt())
                .publishedAt(event.getPublishedAt())
                .retryCount(event.getRetryCount())
                .firstFailedAt(event.getFirstFailedAt())
                .nextRetryAt(event.getNextRetryAt())
                .lastError(event.getLastError())
                .lockedBy(event.getLockedBy())
                .lockedUntil(event.getLockedUntil())
                .envelope(event.getEventPayload())
                .build();
    }
}
