package com.xenomi.cms.core.v2.event.admin;

import java.util.List;
import java.util.Objects;
import java.util.UUID;

import com.xenomi.cms.core.exception.BusinessRuleViolationException;
import com.xenomi.cms.core.exception.ResourceNotFoundException;
import com.xenomi.cms.core.v2.event.entity.OutboxEventV2;
import com.xenomi.cms.core.v2.event.model.OutboxPublicationStatus;
import com.xenomi.cms.core.v2.event.repository.OutboxEventV2Repository;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.data.domain.Page;
import org.springframework.data.domain.PageRequest;
import org.springframework.data.domain.Pageable;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

/** Operator tooling for the outbox: inspect and replay dead-lettered events. */
@Service
public class OutboxAdminV2Service {

    private static final Logger log = LoggerFactory.getLogger(OutboxAdminV2Service.class);
    private static final String REPLAY_ONE = "REPLAY_ONE";
    private static final String REPLAY_BATCH = "REPLAY_BATCH";

    private final OutboxEventV2Repository repository;
    private final OutboxAdminCommandV2Repository commandRepository;

    public OutboxAdminV2Service(
            OutboxEventV2Repository repository,
            OutboxAdminCommandV2Repository commandRepository) {
        this.repository = repository;
        this.commandRepository = commandRepository;
    }

    @Transactional(readOnly = true)
    public Page<OutboxEventV2Response> list(UUID institutionId, OutboxPublicationStatus status, Pageable pageable) {
        return repository.findByInstitutionIdAndPublicationStatus(institutionId, status, pageable)
                .map(OutboxEventV2Response::from);
    }

    @Transactional(readOnly = true)
    public OutboxEventV2Response get(UUID institutionId, UUID outboxEventId) {
        return OutboxEventV2Response.from(find(institutionId, outboxEventId));
    }

    /** Returns a DEAD_LETTER event to PENDING so the dispatcher picks it up on the next cycle. */
    @Transactional
    public OutboxEventV2Response replay(
            UUID institutionId, UUID outboxEventId, String idempotencyKey, String actorId) {
        idempotencyKey = required(idempotencyKey, "Idempotency-Key");
        actorId = required(actorId, "X-Actor-Id");
        if (!registerCommand(institutionId, idempotencyKey, REPLAY_ONE, outboxEventId, actorId)) {
            return OutboxEventV2Response.from(find(institutionId, outboxEventId));
        }
        OutboxEventV2 event = repository.findForUpdate(institutionId, outboxEventId)
                .orElseThrow(() -> new ResourceNotFoundException(
                        "Outbox event", outboxEventId.toString()));
        if (event.getPublicationStatus() != OutboxPublicationStatus.DEAD_LETTER) {
            throw new BusinessRuleViolationException(
                    "Only DEAD_LETTER outbox events can be replayed; current status is " + event.getPublicationStatus());
        }
        reset(event);
        log.warn("Outbox event {} ({}) replayed by {}", outboxEventId, event.getEventType(), actorId);
        return OutboxEventV2Response.from(event);
    }

    /** Replays one bounded batch, e.g. after a hub contract fix. */
    @Transactional
    public int replayAllDeadLetters(
            UUID institutionId, String idempotencyKey, String actorId, int limit) {
        if (limit < 1 || limit > 1_000) {
            throw new BusinessRuleViolationException("Replay limit must be between 1 and 1000");
        }
        idempotencyKey = required(idempotencyKey, "Idempotency-Key");
        actorId = required(actorId, "X-Actor-Id");
        if (!registerCommand(institutionId, idempotencyKey, REPLAY_BATCH, null, actorId)) {
            return commandRepository.findByInstitutionIdAndIdempotencyKey(institutionId, idempotencyKey)
                    .orElseThrow().getReplayedCount();
        }
        List<OutboxEventV2> deadLetters = repository.findDeadLettersForUpdate(
                institutionId, OutboxPublicationStatus.DEAD_LETTER, PageRequest.of(0, limit));
        deadLetters.forEach(OutboxAdminV2Service::reset);
        OutboxAdminCommandV2 command = commandRepository
                .findByInstitutionIdAndIdempotencyKey(institutionId, idempotencyKey)
                .orElseThrow();
        command.setReplayedCount(deadLetters.size());
        log.warn("{} dead-lettered outbox events for institution {} replayed by {}",
                deadLetters.size(), institutionId, actorId);
        return deadLetters.size();
    }

    private boolean registerCommand(UUID institutionId, String idempotencyKey, String operation,
            UUID outboxEventId, String actorId) {
        int inserted = commandRepository.insertIfAbsent(
                UUID.randomUUID(), institutionId, idempotencyKey, operation, outboxEventId, actorId);
        if (inserted == 1) {
            return true;
        }
        OutboxAdminCommandV2 existing = commandRepository
                .findByInstitutionIdAndIdempotencyKey(institutionId, idempotencyKey)
                .orElseThrow();
        if (!operation.equals(existing.getOperation())
                || !Objects.equals(outboxEventId, existing.getOutboxEventId())) {
            throw new BusinessRuleViolationException(
                    "Idempotency-Key was already used for a different outbox replay operation");
        }
        return false;
    }

    private static String required(String value, String name) {
        if (value == null || value.isBlank()) {
            throw new BusinessRuleViolationException(name + " is required");
        }
        return value.trim();
    }

    private static void reset(OutboxEventV2 event) {
        event.setPublicationStatus(OutboxPublicationStatus.PENDING);
        event.setRetryCount(0);
        event.setFirstFailedAt(null);
        event.setNextRetryAt(null);
        event.setLockedUntil(null);
        event.setLockedBy(null);
        event.setLastError(null);
    }

    private OutboxEventV2 find(UUID institutionId, UUID outboxEventId) {
        return repository.findById(outboxEventId)
                .filter(event -> institutionId.equals(event.getInstitutionId()))
                .orElseThrow(() -> new ResourceNotFoundException("Outbox event", outboxEventId.toString()));
    }
}
