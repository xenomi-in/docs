package com.xenomi.cms.core.v2.event.admin;

import java.time.OffsetDateTime;
import java.util.UUID;

import jakarta.persistence.Column;
import jakarta.persistence.Entity;
import jakarta.persistence.Id;
import jakarta.persistence.Table;
import lombok.Builder;
import lombok.Getter;
import lombok.NoArgsConstructor;
import lombok.Setter;

/**
 * Idempotency ledger for destructive operator commands. A replay can be safely
 * retried even after it changes the outbox event from DEAD_LETTER to PENDING.
 */
@Getter
@Setter
@Entity
@NoArgsConstructor
@Table(name = "outbox_admin_command", schema = "cms_v2")
public class OutboxAdminCommandV2 {

    @Id
    @Column(name = "outbox_admin_command_id", nullable = false, updatable = false)
    private UUID outboxAdminCommandId;

    @Column(name = "institution_id", nullable = false, updatable = false)
    private UUID institutionId;

    @Column(name = "idempotency_key", nullable = false, length = 128, updatable = false)
    private String idempotencyKey;

    @Column(name = "operation", nullable = false, length = 32, updatable = false)
    private String operation;

    @Column(name = "outbox_event_id", updatable = false)
    private UUID outboxEventId;

    @Column(name = "replayed_count", nullable = false)
    private int replayedCount;

    @Column(name = "actor_id", nullable = false, length = 100, updatable = false)
    private String actorId;

    @Column(name = "created_at", nullable = false, updatable = false)
    private OffsetDateTime createdAt;

    @Builder
    public OutboxAdminCommandV2(
            UUID outboxAdminCommandId,
            UUID institutionId,
            String idempotencyKey,
            String operation,
            UUID outboxEventId,
            int replayedCount,
            String actorId,
            OffsetDateTime createdAt) {
        this.outboxAdminCommandId = outboxAdminCommandId;
        this.institutionId = institutionId;
        this.idempotencyKey = idempotencyKey;
        this.operation = operation;
        this.outboxEventId = outboxEventId;
        this.replayedCount = replayedCount;
        this.actorId = actorId;
        this.createdAt = createdAt;
    }
}
