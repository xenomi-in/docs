package com.xenomi.cms.core.v2.event.admin;

import java.util.Map;
import java.util.UUID;

import com.xenomi.cms.core.api.ApiResponseEnvelope;
import com.xenomi.cms.core.api.ApiResponseHelper;
import com.xenomi.cms.core.api.PageableDefaults;
import com.xenomi.cms.core.api.PagedResponse;
import com.xenomi.cms.core.v2.event.model.OutboxPublicationStatus;
import io.swagger.v3.oas.annotations.Operation;
import io.swagger.v3.oas.annotations.tags.Tag;
import jakarta.validation.constraints.Max;
import jakarta.validation.constraints.Min;
import jakarta.validation.constraints.NotBlank;
import jakarta.validation.constraints.Size;
import org.springframework.data.domain.PageRequest;
import org.springframework.data.domain.Pageable;
import org.springframework.data.domain.Sort;
import org.springframework.http.ResponseEntity;
import org.springframework.validation.annotation.Validated;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestHeader;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.bind.annotation.RestController;

/**
 * Operator endpoints for outbox publication records. Authorization is inherited from the
 * existing {@code /cms/api/v2/institutions/**} rules: GET requires INSTITUTION_READ or
 * INSTITUTION_MANAGE, replay requires INSTITUTION_MANAGE.
 */
@Validated
@RestController
@RequestMapping("/cms/api/v2/institutions/{institutionId}/outbox-events")
@Tag(name = "V2 - Outbox Operations", description = "Inspect and replay Falcon notification publication records")
public class OutboxAdminV2Controller {

    private static final String ACTOR_HEADER = "X-Actor-Id";
    private final OutboxAdminV2Service service;

    public OutboxAdminV2Controller(OutboxAdminV2Service service) {
        this.service = service;
    }

    @GetMapping
    @Operation(summary = "List outbox events by publication status (defaults to DEAD_LETTER)")
    public ResponseEntity<ApiResponseEnvelope<PagedResponse<OutboxEventV2Response>>> list(
            @PathVariable UUID institutionId,
            @RequestParam(required = false) OutboxPublicationStatus status,
            @RequestParam(required = false) Integer page,
            @RequestParam(required = false) Integer limit) {
        Pageable base = PageableDefaults.ofOneBased(page, limit);
        Pageable pageable = PageRequest.of(base.getPageNumber(), base.getPageSize(),
                Sort.by(Sort.Direction.DESC, "occurredAt"));
        OutboxPublicationStatus effective = status == null ? OutboxPublicationStatus.DEAD_LETTER : status;
        return ApiResponseHelper.ok(PagedResponse.of(service.list(institutionId, effective, pageable)));
    }

    @GetMapping("/{outboxEventId}")
    @Operation(summary = "Get one outbox event including the envelope sent to cms_notification")
    public ResponseEntity<ApiResponseEnvelope<OutboxEventV2Response>> get(
            @PathVariable UUID institutionId, @PathVariable UUID outboxEventId) {
        return ApiResponseHelper.ok(service.get(institutionId, outboxEventId));
    }

    @PostMapping("/{outboxEventId}/actions/replay")
    @Operation(summary = "Return a dead-lettered event to PENDING for re-publication")
    public ResponseEntity<ApiResponseEnvelope<OutboxEventV2Response>> replay(
            @PathVariable UUID institutionId,
            @PathVariable UUID outboxEventId,
            @RequestHeader("Idempotency-Key") @NotBlank @Size(max = 128) String idempotencyKey,
            @RequestHeader(ACTOR_HEADER) @NotBlank @Size(max = 100) String actorId) {
        return ApiResponseHelper.ok(service.replay(institutionId, outboxEventId, idempotencyKey, actorId));
    }

    @PostMapping("/actions/replay-dead-letters")
    @Operation(summary = "Return one bounded batch of dead-lettered events to PENDING")
    public ResponseEntity<ApiResponseEnvelope<Map<String, Integer>>> replayAll(
            @PathVariable UUID institutionId,
            @RequestHeader("Idempotency-Key") @NotBlank @Size(max = 128) String idempotencyKey,
            @RequestHeader(ACTOR_HEADER) @NotBlank @Size(max = 100) String actorId,
            @RequestParam(defaultValue = "100") @Min(1) @Max(1_000) int limit) {
        return ApiResponseHelper.ok(Map.of(
                "replayed", service.replayAllDeadLetters(institutionId, idempotencyKey, actorId, limit)));
    }
}
