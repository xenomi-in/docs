package com.xenomi.cms.core.v2.card.lifecycle.dto;

import com.fasterxml.jackson.annotation.JsonIgnore;
import com.xenomi.cms.core.v2.card.lifecycle.model.CardHotlistReason;
import io.swagger.v3.oas.annotations.media.Schema;
import jakarta.validation.constraints.AssertTrue;
import jakarta.validation.constraints.NotNull;
import jakarta.validation.constraints.Size;
import lombok.AllArgsConstructor;
import lombok.Builder;
import lombok.Data;
import lombok.NoArgsConstructor;

@Data @Builder @NoArgsConstructor @AllArgsConstructor
@Schema(description = "Immediately hot-lists a card. Never include PAN, PIN, CVV or track data.")
public class HotlistCardV2Request {
    @NotNull
    @Schema(description = "Why the card must be blocked", example = "STOLEN")
    private CardHotlistReason reason;

    @Size(max = 250)
    @Schema(description = "Non-sensitive explanation. Required when reason is OTHER.",
            example = "Customer reported an unrecognized physical-card event")
    private String reasonDetail;

    @JsonIgnore
    @AssertTrue(message = "reasonDetail is required when reason is OTHER")
    public boolean isReasonDetailValid() {
        return reason != CardHotlistReason.OTHER
                || (reasonDetail != null && !reasonDetail.isBlank());
    }
}
