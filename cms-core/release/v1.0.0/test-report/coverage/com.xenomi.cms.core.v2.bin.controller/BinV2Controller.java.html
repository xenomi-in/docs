package com.xenomi.cms.core.v2.bin.controller;

import java.util.UUID;

import com.xenomi.cms.core.api.ApiResponseEnvelope;
import com.xenomi.cms.core.api.ApiErrorEnvelope;
import com.xenomi.cms.core.api.ApiResponseHelper;
import com.xenomi.cms.core.api.PageableDefaults;
import com.xenomi.cms.core.api.PagedResponse;
import com.xenomi.cms.core.v2.bin.dto.BinConfigurationV2Response;
import com.xenomi.cms.core.v2.bin.dto.BinRangeV2Response;
import com.xenomi.cms.core.v2.bin.dto.CreateBinConfigurationV2Request;
import com.xenomi.cms.core.v2.bin.dto.CreateBinRangeV2Request;
import com.xenomi.cms.core.v2.bin.dto.UpdateBinConfigurationV2Request;
import com.xenomi.cms.core.v2.bin.dto.UpdateBinRangeV2Request;
import com.xenomi.cms.core.v2.bin.model.BinRangeStatus;
import com.xenomi.cms.core.v2.bin.model.BinStatus;
import com.xenomi.cms.core.v2.bin.model.CardNetwork;
import com.xenomi.cms.core.v2.bin.service.BinConfigurationV2Service;
import com.xenomi.cms.core.v2.bin.service.BinRangeV2Service;

import io.swagger.v3.oas.annotations.Operation;
import io.swagger.v3.oas.annotations.media.Content;
import io.swagger.v3.oas.annotations.media.Schema;
import io.swagger.v3.oas.annotations.responses.ApiResponse;
import io.swagger.v3.oas.annotations.responses.ApiResponses;
import io.swagger.v3.oas.annotations.tags.Tag;
import jakarta.validation.Valid;
import jakarta.validation.constraints.NotBlank;
import jakarta.validation.constraints.Size;
import org.springframework.data.domain.PageRequest;
import org.springframework.data.domain.Pageable;
import org.springframework.data.domain.Sort;
import org.springframework.http.ResponseEntity;
import org.springframework.validation.annotation.Validated;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.PutMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RequestHeader;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.bind.annotation.RestController;

@Validated
@RestController
@RequestMapping("/cms/api/v2/institutions/{institutionId}/bins")
@Tag(name = "V2 - BIN Configuration", description = "BIN configuration, PAN ranges and lifecycle")
public class BinV2Controller {

    private static final String ACTOR_HEADER = "X-Actor-Id";
    private final BinConfigurationV2Service binService;
    private final BinRangeV2Service rangeService;

    public BinV2Controller(BinConfigurationV2Service binService, BinRangeV2Service rangeService) {
        this.binService = binService;
        this.rangeService = rangeService;
    }

    @PostMapping
    @Operation(summary = "Create a BIN configuration")
    public ResponseEntity<ApiResponseEnvelope<BinConfigurationV2Response>> createBin(
            @PathVariable UUID institutionId,
            @Valid @RequestBody CreateBinConfigurationV2Request request,
            @RequestHeader(ACTOR_HEADER) @NotBlank @Size(max = 50) String actorId) {
        return ApiResponseHelper.created(binService.create(institutionId, request, actorId));
    }

    @GetMapping
    @Operation(summary = "List BIN configurations for an institution")
    public ResponseEntity<ApiResponseEnvelope<PagedResponse<BinConfigurationV2Response>>> listBins(
            @PathVariable UUID institutionId,
            @RequestParam(required = false) CardNetwork cardNetwork,
            @RequestParam(required = false) BinStatus binStatus,
            @RequestParam(required = false) Integer page,
            @RequestParam(required = false) Integer limit) {
        Pageable base = PageableDefaults.ofOneBased(page, limit);
        Pageable pageable = PageRequest.of(base.getPageNumber(), base.getPageSize(),
                Sort.by(Sort.Direction.ASC, "binNumber"));
        return ApiResponseHelper.ok(PagedResponse.of(
                binService.search(institutionId, cardNetwork, binStatus, pageable)));
    }

    @GetMapping("/{binId}")
    @Operation(summary = "Get a BIN configuration")
    public ResponseEntity<ApiResponseEnvelope<BinConfigurationV2Response>> getBin(
            @PathVariable UUID institutionId, @PathVariable UUID binId) {
        return ApiResponseHelper.ok(binService.get(institutionId, binId));
    }

    @PutMapping("/{binId}")
    @Operation(
            summary = "Replace the mutable BIN configuration",
            description = "Applies full PUT semantics. The request must contain every required mutable field and the latest version returned by GET. "
                    + "BIN identity, institution ownership, network classification and audit fields are immutable. "
                    + "A stale version returns 409 CONFLICT.")
    @ApiResponses({
            @ApiResponse(responseCode = "200", description = "BIN configuration replaced"),
            @ApiResponse(responseCode = "400", description = "Incomplete representation, invalid value, unknown field, or immutable field",
                    content = @Content(mediaType = "application/json", schema = @Schema(implementation = ApiErrorEnvelope.class))),
            @ApiResponse(responseCode = "404", description = "BIN not found within the institution",
                    content = @Content(mediaType = "application/json", schema = @Schema(implementation = ApiErrorEnvelope.class))),
            @ApiResponse(responseCode = "409", description = "Stale optimistic-lock version",
                    content = @Content(mediaType = "application/json", schema = @Schema(implementation = ApiErrorEnvelope.class)))
    })
    public ResponseEntity<ApiResponseEnvelope<BinConfigurationV2Response>> updateBin(
            @PathVariable UUID institutionId, @PathVariable UUID binId,
            @Valid @RequestBody UpdateBinConfigurationV2Request request,
            @RequestHeader(ACTOR_HEADER) @NotBlank @Size(max = 50) String actorId) {
        return ApiResponseHelper.ok(binService.update(institutionId, binId, request, actorId));
    }

    @PostMapping("/{binId}/ranges")
    @Operation(summary = "Create a PAN range under a BIN configuration")
    public ResponseEntity<ApiResponseEnvelope<BinRangeV2Response>> createRange(
            @PathVariable UUID institutionId, @PathVariable UUID binId,
            @Valid @RequestBody CreateBinRangeV2Request request,
            @RequestHeader(ACTOR_HEADER) @NotBlank @Size(max = 100) String actorId) {
        return ApiResponseHelper.created(rangeService.create(institutionId, binId, request, actorId));
    }

    @GetMapping("/{binId}/ranges")
    @Operation(summary = "List PAN ranges under a BIN configuration")
    public ResponseEntity<ApiResponseEnvelope<PagedResponse<BinRangeV2Response>>> listRanges(
            @PathVariable UUID institutionId, @PathVariable UUID binId,
            @RequestParam(required = false) BinRangeStatus rangeStatus,
            @RequestParam(required = false) Integer page,
            @RequestParam(required = false) Integer limit) {
        Pageable base = PageableDefaults.ofOneBased(page, limit);
        Pageable pageable = PageRequest.of(base.getPageNumber(), base.getPageSize(),
                Sort.by(Sort.Direction.ASC, "rangeStart"));
        return ApiResponseHelper.ok(PagedResponse.of(
                rangeService.search(institutionId, binId, rangeStatus, pageable)));
    }

    @GetMapping("/{binId}/ranges/{rangeId}")
    @Operation(summary = "Get a PAN range under a BIN configuration")
    public ResponseEntity<ApiResponseEnvelope<BinRangeV2Response>> getRange(
            @PathVariable UUID institutionId, @PathVariable UUID binId, @PathVariable UUID rangeId) {
        return ApiResponseHelper.ok(rangeService.get(institutionId, binId, rangeId));
    }

    @PostMapping("/{binId}/ranges/{rangeId}/actions/update")
    @Operation(summary = "Apply an explicit update or lifecycle action to a PAN range")
    public ResponseEntity<ApiResponseEnvelope<BinRangeV2Response>> updateRange(
            @PathVariable UUID institutionId, @PathVariable UUID binId, @PathVariable UUID rangeId,
            @Valid @RequestBody UpdateBinRangeV2Request request,
            @RequestHeader(ACTOR_HEADER) @NotBlank @Size(max = 100) String actorId) {
        return ApiResponseHelper.ok(rangeService.update(
                institutionId, binId, rangeId, request, actorId));
    }
}
