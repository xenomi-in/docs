package com.xenomi.cms.core.v2.event.repository;

import java.util.List;
import java.util.UUID;

import com.xenomi.cms.core.v2.event.entity.OutboxEventV2;
import jakarta.persistence.EntityManager;
import jakarta.persistence.PersistenceContext;
import org.springframework.stereotype.Repository;

/**
 * A single PostgreSQL statement claims an event and advances its fencing token.
 * The CTE locks candidates before the update; {@code RETURNING} gives the
 * dispatcher the exact token it must present while recording its outcome.
 */
@Repository
class OutboxEventV2RepositoryImpl implements OutboxEventV2RepositoryCustom {

    private static final String CLAIM_DUE_EVENTS = """
            WITH candidates AS (
                SELECT o.outbox_event_id
                FROM cms_v2.outbox_event o
                WHERE o.publication_status IN ('PENDING', 'FAILED')
                  AND (o.next_retry_at IS NULL OR o.next_retry_at <= now())
                  AND (o.locked_until IS NULL OR o.locked_until <= now())
                  AND NOT EXISTS (
                        SELECT 1
                        FROM cms_v2.outbox_event earlier
                        WHERE earlier.institution_id = o.institution_id
                          AND earlier.aggregate_type = o.aggregate_type
                          AND earlier.aggregate_id = o.aggregate_id
                          AND earlier.publication_status IN ('PENDING', 'FAILED', 'DEAD_LETTER')
                          AND earlier.aggregate_sequence < o.aggregate_sequence)
                ORDER BY o.occurred_at, o.aggregate_sequence
                LIMIT :batchSize
                FOR UPDATE OF o SKIP LOCKED
            )
            UPDATE cms_v2.outbox_event o
            SET locked_until = now() + make_interval(secs => :leaseSeconds),
                locked_by = :owner,
                dispatch_attempt = o.dispatch_attempt + 1
            FROM candidates
            WHERE o.outbox_event_id = candidates.outbox_event_id
            RETURNING o.*
            """;

    private static final String NEXT_AGGREGATE_SEQUENCE = """
            INSERT INTO cms_v2.outbox_aggregate_sequence (
                institution_id, aggregate_type, aggregate_id, next_sequence
            )
            VALUES (:institutionId, :aggregateType, :aggregateId, 2)
            ON CONFLICT (institution_id, aggregate_type, aggregate_id)
            DO UPDATE SET next_sequence = cms_v2.outbox_aggregate_sequence.next_sequence + 1
            RETURNING next_sequence - 1
            """;

    @PersistenceContext
    private EntityManager entityManager;

    @Override
    @SuppressWarnings("unchecked")
    public List<OutboxEventV2> claimDueEvents(String owner, long leaseSeconds, int batchSize) {
        return entityManager.createNativeQuery(CLAIM_DUE_EVENTS, OutboxEventV2.class)
                .setParameter("owner", owner)
                .setParameter("leaseSeconds", leaseSeconds)
                .setParameter("batchSize", batchSize)
                .getResultList();
    }

    @Override
    public long nextAggregateSequence(UUID institutionId, String aggregateType, String aggregateId) {
        Number sequence = (Number) entityManager.createNativeQuery(NEXT_AGGREGATE_SEQUENCE)
                .setParameter("institutionId", institutionId)
                .setParameter("aggregateType", aggregateType)
                .setParameter("aggregateId", aggregateId)
                .getSingleResult();
        return sequence.longValue();
    }
}
