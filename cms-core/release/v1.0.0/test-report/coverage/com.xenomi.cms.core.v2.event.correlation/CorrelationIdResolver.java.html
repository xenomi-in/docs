package com.xenomi.cms.core.v2.event.correlation;

import java.nio.charset.StandardCharsets;
import java.util.UUID;

import com.xenomi.cms.core.api.RequestIdFilter;
import org.slf4j.MDC;
import org.springframework.stereotype.Component;

/**
 * Resolves the correlation id carried in the Falcon envelope. The value is the inbound
 * HTTP request id (explicit {@code X-Request-Id} argument first, then the MDC populated by
 * {@link RequestIdFilter}). A fresh id is minted only when no request context exists,
 * e.g. internal jobs.
 */
@Component
public class CorrelationIdResolver {

    private static final UUID BLANK_CORRELATION =
            UUID.nameUUIDFromBytes("cms-v2-outbox|blank-correlation".getBytes(StandardCharsets.UTF_8));

    public String current() {
        return resolve(null);
    }

    public String resolve(String requestId) {
        String value = firstNonBlank(requestId, MDC.get(RequestIdFilter.REQUEST_ID_MDC));
        return value != null ? value : UUID.randomUUID().toString();
    }

    /** Maps the string correlation id onto the UUID {@code correlation_id} column deterministically. */
    public UUID toUuid(String correlationId) {
        if (correlationId == null || correlationId.isBlank()) {
            return BLANK_CORRELATION;
        }
        String trimmed = correlationId.trim();
        try {
            return UUID.fromString(trimmed);
        } catch (IllegalArgumentException ignored) {
            return UUID.nameUUIDFromBytes(trimmed.getBytes(StandardCharsets.UTF_8));
        }
    }

    static String firstNonBlank(String... values) {
        for (String value : values) {
            if (value != null && !value.isBlank()) {
                return value.trim();
            }
        }
        return null;
    }
}
