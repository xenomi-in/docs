package com.xenomi.cms.core.v2.hsm.model;

/** Approved operational purposes for institution- and BIN-scoped keys. */
public enum KeyPurpose {
    CARDHOLDER_DATA_ENCRYPTION(HsmKeyScope.INSTITUTION),
    PAN_DATA_ENCRYPTION(HsmKeyScope.INSTITUTION),
    PII_ENCRYPTION(HsmKeyScope.INSTITUTION),
    PAN_HMAC(HsmKeyScope.INSTITUTION),
    FILE_ENCRYPTION(HsmKeyScope.INSTITUTION),
    AUDIT_HMAC(HsmKeyScope.INSTITUTION),
    TOKEN_REFERENCE_ENCRYPTION(HsmKeyScope.INSTITUTION),

    PIN_VERIFICATION(HsmKeyScope.BIN),
    PIN_GENERATION(HsmKeyScope.BIN),
    CVV_GENERATION(HsmKeyScope.BIN),
    CVV_VERIFICATION(HsmKeyScope.BIN),
    EMV_KEY_DERIVATION(HsmKeyScope.BIN),
    PIN_BLOCK_ENCRYPTION(HsmKeyScope.BIN),
    PERSONALISATION_DATA_GENERATION(HsmKeyScope.BIN);

    private final HsmKeyScope scope;

    KeyPurpose(HsmKeyScope scope) {
        this.scope = scope;
    }

    public HsmKeyScope scope() {
        return scope;
    }
}
