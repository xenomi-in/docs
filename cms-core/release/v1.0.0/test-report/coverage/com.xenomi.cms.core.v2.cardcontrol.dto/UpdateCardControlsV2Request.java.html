package com.xenomi.cms.core.v2.cardcontrol.dto;

import java.util.List;

import io.swagger.v3.oas.annotations.media.Schema;
import jakarta.validation.Valid;
import jakarta.validation.constraints.AssertTrue;
import jakarta.validation.constraints.NotNull;
import jakarta.validation.constraints.Size;
import lombok.AllArgsConstructor;
import lombok.Builder;
import lombok.Data;
import lombok.NoArgsConstructor;

@Data @Builder @NoArgsConstructor @AllArgsConstructor
@Schema(description = "Partial card-control update. Omitted dimensions retain their current state.")
public class UpdateCardControlsV2Request {
    @Schema(description = "Card-level international master switch. It cannot be enabled when the product-level international switch is disabled.")
    private Boolean internationalEnabled;
    @NotNull @Size(max = 8) @Valid
    private List<CardControlDimensionV2Request> controls;
    @Size(max = 50)
    @Schema(description = "Auditable reason code", example = "CUSTOMER_REQUEST")
    private String reasonCode;

    @AssertTrue(message = "internationalEnabled or at least one control change is required")
    @Schema(hidden = true)
    public boolean isChangePresent() {
        return internationalEnabled != null || (controls != null && !controls.isEmpty());
    }
}
