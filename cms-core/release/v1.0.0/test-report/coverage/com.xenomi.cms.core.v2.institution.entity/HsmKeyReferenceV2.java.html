package com.xenomi.cms.core.v2.institution.entity;

import java.time.LocalDate;
import java.time.OffsetDateTime;
import java.time.ZoneOffset;
import java.util.UUID;

import com.xenomi.cms.core.v2.hsm.model.HsmKeyScheme;
import com.xenomi.cms.core.v2.hsm.model.HsmKeyScope;
import com.xenomi.cms.core.v2.hsm.model.HsmKeyStatus;
import com.xenomi.cms.core.v2.hsm.model.HsmKeyType;

import jakarta.persistence.Column;
import jakarta.persistence.Entity;
import jakarta.persistence.EnumType;
import jakarta.persistence.Enumerated;
import jakarta.persistence.Id;
import jakarta.persistence.PrePersist;
import jakarta.persistence.PreUpdate;
import jakarta.persistence.Table;
import lombok.AllArgsConstructor;
import lombok.Builder;
import lombok.Data;
import lombok.NoArgsConstructor;
import lombok.ToString;

@Entity(name = "HsmKeyReferenceV2")
@Table(name = "hsm_key_reference", schema = "cms_v2")
@Data
@Builder
@NoArgsConstructor
@AllArgsConstructor
public class HsmKeyReferenceV2 {

    @Id @Column(name = "key_id", nullable = false, updatable = false)
    private UUID keyId;
    @Enumerated(EnumType.STRING)
    @Column(name = "key_type", nullable = false, length = 50, updatable = false)
    private HsmKeyType keyType;
    @Column(name = "key_index", nullable = false, length = 20, updatable = false)
    private String keyIndex;
    @Column(name = "key_label", length = 100)
    private String keyLabel;
    @Column(name = "key_check_value", nullable = false, length = 16)
    private String keyCheckValue;
    @Column(name = "key_version", nullable = false, updatable = false)
    private int keyVersion;
    @Column(name = "bin_id", updatable = false)
    private UUID binId;
    @Enumerated(EnumType.STRING)
    @Column(name = "key_scope", nullable = false, length = 20, updatable = false)
    private HsmKeyScope keyScope;
    @Enumerated(EnumType.STRING)
    @Column(name = "scheme", nullable = false, length = 20, updatable = false)
    private HsmKeyScheme scheme;
    @Column(name = "key_length_bits", nullable = false, updatable = false)
    private int keyLengthBits;
    @Column(name = "is_tr31", nullable = false, updatable = false)
    private boolean tr31;
    @Column(name = "tr31_key_usage", length = 10, updatable = false)
    private String tr31KeyUsage;
    @Column(name = "tr31_algorithm", length = 10, updatable = false)
    private String tr31Algorithm;
    @Column(name = "tr31_mode_of_use", length = 10, updatable = false)
    private String tr31ModeOfUse;
    @Column(name = "hsm_identifier", length = 50)
    private String hsmIdentifier;
    @Column(name = "hsm_partition", length = 50)
    private String hsmPartition;
    @Column(name = "hsm_serial_number", length = 100)
    private String hsmSerialNumber;
    @Column(name = "effective_from", nullable = false)
    private LocalDate effectiveFrom;
    @Column(name = "expiry_date", nullable = false)
    private LocalDate expiryDate;
    @Enumerated(EnumType.STRING)
    @Column(name = "status", nullable = false, length = 20)
    private HsmKeyStatus status;
    @Column(name = "previous_key_id", updatable = false)
    private UUID previousKeyId;
    @Column(name = "replacement_key_id")
    private UUID replacementKeyId;
    @Column(name = "rotated_at")
    private OffsetDateTime rotatedAt;
    @Column(name = "rotation_reason", length = 50)
    private String rotationReason;
    @Column(name = "encrypted_key_block", length = 500)
    @ToString.Exclude
    private String encryptedKeyBlock;
    @Column(name = "institution_id", nullable = false, updatable = false)
    private UUID institutionId;
    @Column(name = "acs_id", updatable = false)
    private UUID acsId;
    @Column(name = "card_vendor_id", updatable = false)
    private UUID cardVendorId;
    @Column(name = "stakeholder_id", updatable = false)
    private UUID stakeholderId;
    @Column(name = "imported_via_kek_id", updatable = false)
    private UUID importedViaKekId;
    @Column(name = "external_request_id", length = 100, updatable = false)
    private String externalRequestId;
    @Column(name = "created_at", nullable = false, updatable = false)
    private OffsetDateTime createdAt;
    @Column(name = "updated_at", nullable = false)
    private OffsetDateTime updatedAt;
    @Column(name = "created_by", nullable = false, length = 50, updatable = false)
    private String createdBy;
    @Column(name = "updated_by", nullable = false, length = 50)
    private String updatedBy;

    @PrePersist
    void beforeInsert() {
        OffsetDateTime now = OffsetDateTime.now(ZoneOffset.UTC);
        if (createdAt == null) createdAt = now;
        updatedAt = now;
    }

    @PreUpdate
    void beforeUpdate() {
        updatedAt = OffsetDateTime.now(ZoneOffset.UTC);
    }
}
