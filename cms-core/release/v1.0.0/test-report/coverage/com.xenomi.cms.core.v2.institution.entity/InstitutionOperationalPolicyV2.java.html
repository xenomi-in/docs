package com.xenomi.cms.core.v2.institution.entity;

import java.time.OffsetDateTime;
import java.util.UUID;

import jakarta.persistence.Column;
import jakarta.persistence.Entity;
import jakarta.persistence.Id;
import jakarta.persistence.PrePersist;
import jakarta.persistence.PreUpdate;
import jakarta.persistence.Table;
import jakarta.persistence.Version;
import lombok.AllArgsConstructor;
import lombok.Builder;
import lombok.Data;
import lombok.NoArgsConstructor;

@Entity(name = "InstitutionOperationalPolicyV2")
@Table(name = "institution_operational_policy", schema = "cms_v2")
@Data
@Builder
@NoArgsConstructor
@AllArgsConstructor
public class InstitutionOperationalPolicyV2 {

    @Id
    @Column(name = "institution_id", nullable = false, updatable = false)
    private UUID institutionId;

    @Column(name = "card_issuance_enabled", nullable = false)
    private boolean cardIssuanceEnabled;

    @Column(name = "replacement_enabled", nullable = false)
    private boolean replacementEnabled;

    @Column(name = "renewal_enabled", nullable = false)
    private boolean renewalEnabled;

    @Column(name = "pin_operations_enabled", nullable = false)
    private boolean pinOperationsEnabled;

    @Column(name = "token_operations_enabled", nullable = false)
    private boolean tokenOperationsEnabled;

    @Column(name = "batch_processing_enabled", nullable = false)
    private boolean batchProcessingEnabled;

    @Column(name = "maintenance_mode", nullable = false)
    private boolean maintenanceMode;

    @Version
    @Column(name = "policy_version", nullable = false)
    private int policyVersion;

    @Column(name = "updated_at", nullable = false)
    private OffsetDateTime updatedAt;

    @Column(name = "updated_by", nullable = false, length = 100)
    private String updatedBy;

    @PrePersist
    @PreUpdate
    void touch() {
        updatedAt = OffsetDateTime.now(java.time.ZoneOffset.UTC);
    }
}
