package com.xenomi.cms.core.v2.institution.entity;

import java.time.LocalDate;
import java.time.OffsetDateTime;
import java.time.ZoneOffset;
import java.util.UUID;

import com.xenomi.cms.core.v2.institution.model.InstitutionStatus;
import com.xenomi.cms.core.v2.institution.model.InstitutionType;

import jakarta.persistence.Column;
import jakarta.persistence.Entity;
import jakarta.persistence.EnumType;
import jakarta.persistence.Enumerated;
import jakarta.persistence.Id;
import jakarta.persistence.PrePersist;
import jakarta.persistence.PreUpdate;
import jakarta.persistence.Table;
import lombok.AllArgsConstructor;
import lombok.Builder;
import lombok.Data;
import lombok.NoArgsConstructor;

@Entity(name = "InstitutionV2")
@Table(name = "institution", schema = "cms_v2")
@Data
@Builder
@NoArgsConstructor
@AllArgsConstructor
public class InstitutionV2 {

    @Id
    @Column(name = "institution_id", nullable = false, updatable = false)
    private UUID institutionId;

    @Column(name = "institution_name", nullable = false, length = 200)
    private String institutionName;

    @Column(name = "institution_code", nullable = false, unique = true, length = 20, updatable = false)
    private String institutionCode;

    @Column(name = "institution_short_name", length = 50)
    private String institutionShortName;

    @Enumerated(EnumType.STRING)
    @Column(name = "institution_type", nullable = false, length = 20, updatable = false)
    private InstitutionType institutionType;

    @Column(name = "visa_member_flag")
    private boolean visaMemberFlag;

    @Column(name = "visa_member_id", length = 20)
    private String visaMemberId;

    @Column(name = "rupay_member_flag")
    private boolean rupayMemberFlag;

    @Column(name = "npci_member_id", length = 20)
    private String npciMemberId;

    @Column(name = "registered_address_line1", nullable = false, length = 200)
    private String registeredAddressLine1;

    @Column(name = "registered_address_line2", length = 200)
    private String registeredAddressLine2;

    @Column(name = "city", nullable = false, length = 100)
    private String city;

    @Column(name = "state_code", nullable = false, length = 50)
    private String stateCode;

    @org.hibernate.annotations.JdbcTypeCode(org.hibernate.type.SqlTypes.CHAR)
    @Column(name = "country_code", nullable = false, length = 3, columnDefinition = "char(3)")
    private String countryCode;

    @Column(name = "postal_code", nullable = false, length = 10)
    private String postalCode;

    @Column(name = "contact_phone_primary", nullable = false, length = 20)
    private String contactPhonePrimary;

    @Column(name = "contact_phone_secondary", length = 20)
    private String contactPhoneSecondary;

    @Column(name = "contact_email_primary", nullable = false, length = 100)
    private String contactEmailPrimary;

    @Column(name = "contact_email_secondary", length = 100)
    private String contactEmailSecondary;

    @Column(name = "website_url", length = 200)
    private String websiteUrl;

    @Column(name = "timezone", nullable = false, length = 60)
    private String timezone;

    @Column(name = "report_delivery_sftp_path", length = 500)
    private String reportDeliverySftpPath;

    @Column(name = "report_delivery_email", length = 500)
    private String reportDeliveryEmail;

    @Column(name = "prepaid_card_issuing_enabled")
    private boolean prepaidCardIssuingEnabled;

    @Column(name = "credit_card_issuing_enabled")
    private boolean creditCardIssuingEnabled;

    @Column(name = "default_card_validity_months")
    private int defaultCardValidityMonths;

    @org.hibernate.annotations.JdbcTypeCode(org.hibernate.type.SqlTypes.CHAR)
    @Column(name = "base_currency", nullable = false, length = 3, columnDefinition = "char(3)")
    private String baseCurrency;

    @Column(name = "currency_exponent", nullable = false)
    private short currencyExponent;

    @Column(name = "business_date")
    private LocalDate businessDate;

    @Enumerated(EnumType.STRING)
    @Column(name = "status", nullable = false, length = 20)
    private InstitutionStatus status;

    @Column(name = "created_at", nullable = false, updatable = false)
    private OffsetDateTime createdAt;

    @Column(name = "updated_at", nullable = false)
    private OffsetDateTime updatedAt;

    @Column(name = "created_by", nullable = false, length = 50, updatable = false)
    private String createdBy;

    @Column(name = "updated_by", nullable = false, length = 50)
    private String updatedBy;

    @PrePersist
    void beforeInsert() {
        OffsetDateTime now = OffsetDateTime.now(ZoneOffset.UTC);
        if (createdAt == null) {
            createdAt = now;
        }
        updatedAt = now;
    }

    @PreUpdate
    void beforeUpdate() {
        updatedAt = OffsetDateTime.now(ZoneOffset.UTC);
    }
}
