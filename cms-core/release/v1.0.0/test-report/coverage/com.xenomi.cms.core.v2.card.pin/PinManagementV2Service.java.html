package com.xenomi.cms.core.v2.card.pin;

import java.util.UUID;

import com.xenomi.cms.core.exception.ResourceNotFoundException;
import com.xenomi.cms.core.v2.card.pin.dto.CardPinHistoryV2Response;
import com.xenomi.cms.core.v2.card.pin.entity.PinHistoryV2;
import com.xenomi.cms.core.v2.card.pin.repository.PinHistoryV2Repository;
import com.xenomi.cms.core.v2.card.repository.CardV2Repository;
import org.springframework.data.domain.Page;
import org.springframework.data.domain.Pageable;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

/**
 * Read-side access to the CMS PIN audit trail.
 *
 * <p>PIN initialization, change, and reset are owned by {@code SwitchPinService}; CMS does not
 * retain or mutate PIN verification material locally.</p>
 */
@Service
public class PinManagementV2Service {
    private final CardV2Repository cards;
    private final PinHistoryV2Repository history;

    public PinManagementV2Service(CardV2Repository cards, PinHistoryV2Repository history) {
        this.cards = cards;
        this.history = history;
    }

    @Transactional(readOnly = true)
    public Page<CardPinHistoryV2Response> history(
            UUID institutionId, UUID cardId, Pageable pageable) {
        if (cards.findByCardIdAndInstitutionId(cardId, institutionId).isEmpty()) {
            throw new ResourceNotFoundException("Card", cardId.toString());
        }
        return history.findByInstitutionIdAndCardIdOrderByOccurredAtDesc(
                institutionId, cardId, pageable).map(this::historyResponse);
    }

    private CardPinHistoryV2Response historyResponse(PinHistoryV2 event) {
        return new CardPinHistoryV2Response(event.getPinHistoryId(), event.getCardId(),
                event.getOperationType(), event.getPreviousStatus(), event.getResultingStatus(),
                event.getOccurredAt(), event.getActorReference(), event.getOperationReason(),
                event.getCorrelationId());
    }
}
