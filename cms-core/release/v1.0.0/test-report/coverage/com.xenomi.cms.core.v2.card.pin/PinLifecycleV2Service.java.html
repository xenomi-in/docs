package com.xenomi.cms.core.v2.card.pin;

/** Initializes card PIN state and delegates random PIN generation entirely to the HSM. */
@Service
public class PinLifecycleV2Service {
    @org.springframework.beans.factory.annotation.Value("${cms.switch.pin.enabled:${cms.switch.sync.enabled:false}}")
    private boolean switchPinEnabled;
    @org.springframework.beans.factory.annotation.Value("${cms.switch.sync.enabled:false}")
    private boolean switchSyncEnabled;
    @org.springframework.beans.factory.annotation.Autowired
    private com.xenomi.cms.core.v2.card.pin.switchpin.SwitchPinPolicyResolver switchPolicy;
    private static final Set<HsmKeyStatus> USABLE_KEY_STATUSES =
            Set.of(HsmKeyStatus.ACTIVE);

    private final ProductPinPolicyV2Repository policyRepository;
    private final BinConfigurationV2Repository binRepository;
    private final BinHsmKeyAssignmentV2Repository assignmentRepository;
    private final HsmKeyReferenceV2Repository keyRepository;
    private final PinDataV2Repository pinRepository;
    private final PinHistoryV2Repository historyRepository;
    private final CmsCardSecurityService cardSecurityService;
    private final PinHsmOperations hsm;
    private final DatabaseUuidV7Generator uuidGenerator;

    public PinLifecycleV2Service(ProductPinPolicyV2Repository policyRepository,
            BinConfigurationV2Repository binRepository,
            BinHsmKeyAssignmentV2Repository assignmentRepository,
            HsmKeyReferenceV2Repository keyRepository,
            PinDataV2Repository pinRepository,
            PinHistoryV2Repository historyRepository,
            CmsCardSecurityService cardSecurityService,
            PinHsmOperations hsm,
            DatabaseUuidV7Generator uuidGenerator) {
        this.policyRepository = policyRepository;
        this.binRepository = binRepository;
        this.assignmentRepository = assignmentRepository;
        this.keyRepository = keyRepository;
        this.pinRepository = pinRepository;
        this.historyRepository = historyRepository;
        this.cardSecurityService = cardSecurityService;
        this.hsm = hsm;
        this.uuidGenerator = uuidGenerator;
    }

    public PinDataV2 initializeForIssuance(CardV2 card, String actor) {
        return initialize(card, null, actor);
    }

    /** Reject missing policy/key configuration before staging any provider work. */
    public ProductPinPolicyV2 validateOriginalConfiguration(CardV2 card) {
        var policy=policyRepository.findByProductVersionIdAndInstitutionId(card.getProductVersionId(),card.getInstitutionId())
                .filter(p->p.getPolicyStatus()==com.xenomi.cms.core.v2.product.model.ProductPolicyStatus.ACTIVE)
                .orElseThrow(()->new BusinessRuleViolationException("An active PIN policy is required"));
        var bin=binRepository.findByBinIdAndInstitutionId(card.getBinId(),card.getInstitutionId()).orElseThrow();
        requireSwitchAuthority(policy);
        if (switchPinEnabled) {
            switchPolicy.resolve(card.getInstitutionId(), policy, bin, null);
            return policy;
        }
        if(policy.isPinRequired() && policy.getGenerationMethod()!=PinGenerationMethod.CUSTOMER_SELECTED) {
            validateGenerationPolicy(policy,bin); requirePvki(bin);
            activeBinKey(card,KeyPurpose.PIN_GENERATION,HsmKeyType.PVK);
            activeBinKey(card,KeyPurpose.PIN_BLOCK_ENCRYPTION,HsmKeyType.ZPK);
        }
        return policy;
    }

    public PinDataV2 initializeForReplacement(
            CardV2 card, ReplacementPinStrategy strategy, String actor) {
        return initialize(card, strategy, actor);
    }

    private PinDataV2 initialize(CardV2 card, ReplacementPinStrategy replacementStrategy,
            String actor) {
        com.xenomi.cms.core.v2.card.lifecycle.service.LifecycleCommandPendingV2.reject("initialize PIN aggregate");
        return null;
    }

    /** Prepares verification metadata only. Persist it before CARD in the finalization transaction. */
    public PinDataV2 prepareOriginal(CardV2 card, String actor, ProductPinPolicyV2 expectedPolicy) {
        return prepare(card, actor, expectedPolicy, null);
    }

    /** Prepares a new PAN's PIN aggregate while applying the replacement policy's PIN decision. */
    public PinDataV2 prepareReplacement(CardV2 card, String actor,
            ProductPinPolicyV2 expectedPolicy, ReplacementPinStrategy replacementStrategy) {
        if (replacementStrategy == null) {
            throw new BusinessRuleViolationException("Replacement PIN strategy is required");
        }
        return prepare(card, actor, expectedPolicy, replacementStrategy);
    }

    private PinDataV2 prepare(CardV2 card, String actor, ProductPinPolicyV2 expectedPolicy,
            ReplacementPinStrategy replacementStrategy) {
        if (org.springframework.transaction.support.TransactionSynchronizationManager.isActualTransactionActive()) {
            throw new IllegalStateException("PIN preparation must run outside a database transaction");
        }
        if (pinRepository.findByInstitutionIdAndPanAllocationId(card.getInstitutionId(), card.getPanAllocationId()).isPresent()) {
            throw new IllegalStateException("PIN state already exists for card " + card.getCardId());
        }
        ProductPinPolicyV2 policy = policyRepository
                .findByProductVersionIdAndInstitutionId(
                        card.getProductVersionId(), card.getInstitutionId())
                .filter(p -> p.getPolicyStatus() == com.xenomi.cms.core.v2.product.model.ProductPolicyStatus.ACTIVE)
                .orElseThrow(() -> new BusinessRuleViolationException("An active PIN policy is required"));
        if (!policy.equals(expectedPolicy)) throw new BusinessRuleViolationException("PIN policy changed after issuance acceptance");
        requireSwitchAuthority(policy);
        BinConfigurationV2 bin = binRepository
                .findByBinIdAndInstitutionId(card.getBinId(), card.getInstitutionId())
                .orElseThrow(() -> new BusinessRuleViolationException(
                        "BIN configuration is required before PIN initialization"));

        if (switchPinEnabled) {
            var profile = switchPolicy.resolve(card.getInstitutionId(), policy, bin, replacementStrategy);
            boolean selected = profile != null && "CUSTOMER_SELECTED".equals(profile.get("establishment"));
            return PinDataV2.builder().pinDataId(card.getPinDataRef()).panAllocationId(card.getPanAllocationId())
                    .institutionId(card.getInstitutionId()).pinAuthority("SWITCH").switchPinProfile(profile)
                    .pinStatus(selected ? PinStatusV2.RESET_REQUIRED : PinStatusV2.NOT_SET)
                    .failedAttemptCount(0).pinMethod(bin.getPinMethod()).pinLength((int) policy.getPinLength())
                    .pvki(bin.getPvki()).maximumAttempts((int) policy.getMaximumAttempts())
                    .generationMethod(selected ? PinGenerationMethod.CUSTOMER_SELECTED : policy.getGenerationMethod())
                    .deliveryMethod(policy.getDeliveryMethod()).updatedAt(OffsetDateTime.now(ZoneOffset.UTC))
                    .updatedBy(actor).build();
        }

        PinStatusV2 status = PinStatusV2.NOT_SET;
        PinOperationTypeV2 operation = PinOperationTypeV2.INITIALIZED;
        String offset = null;
        UUID verificationKeyId = null;
        UUID pinBlockKeyId = null;

        boolean resetRequired = policy != null && policy.isPinRequired()
                && (replacementStrategy == ReplacementPinStrategy.CUSTOMER_RESET_REQUIRED
                    || policy.getGenerationMethod() == PinGenerationMethod.CUSTOMER_SELECTED);
        boolean generate = policy != null && policy.isPinRequired() && !resetRequired
                && (replacementStrategy == null
                    || replacementStrategy == ReplacementPinStrategy.GENERATE_NEW_PIN);
        if (resetRequired) {
            status = PinStatusV2.RESET_REQUIRED;
            operation = PinOperationTypeV2.RESET_REQUIRED;
        } else if (generate) {
            validateGenerationPolicy(policy, bin);
            HsmKeyReferenceV2 verificationKey = activeBinKey(
                    card, KeyPurpose.PIN_GENERATION, HsmKeyType.PVK);
            HsmKeyReferenceV2 pinBlockKey = activeBinKey(
                    card, KeyPurpose.PIN_BLOCK_ENCRYPTION, HsmKeyType.ZPK);
            offset = cardSecurityService.withDecryptedPan(card.getInstitutionId(), card,
                    pan -> hsm.generateRandomPinOffset(pan, bin.getPinMethod(),
                            policy.getPinLength(), requirePvki(bin),
                            verificationKey, pinBlockKey));
            verificationKeyId = verificationKey.getKeyId();
            pinBlockKeyId = pinBlockKey.getKeyId();
            status = PinStatusV2.PENDING_DELIVERY;
            operation = PinOperationTypeV2.GENERATED;
        }

        OffsetDateTime now = OffsetDateTime.now(ZoneOffset.UTC);
        return PinDataV2.builder()
                .pinDataId(card.getPinDataRef()).panAllocationId(card.getPanAllocationId())
                .institutionId(card.getInstitutionId())
                .pinStatus(status)
                .failedAttemptCount(0)
                .pinOffset(offset)
                .pinMethod(bin.getPinMethod())
                .pinLength(policy == null ? bin.getPinLength() : (int) policy.getPinLength())
                .pvki(bin.getPvki())
                .maximumAttempts(policy == null ? null : (int) policy.getMaximumAttempts())
                .generationMethod(policy == null ? null : policy.getGenerationMethod())
                .deliveryMethod(policy == null ? null : policy.getDeliveryMethod())
                .lastChangedAt(generate ? now : null)
                .hsmKeyId(verificationKeyId)
                .pinBlockKeyId(pinBlockKeyId)
                .updatedAt(now)
                .updatedBy(actor)
                .build();
    }

    public void recordOriginalHistory(CardV2 card, PinDataV2 pin, UUID operationId, String actor) {
        PinOperationTypeV2 operation = pin.getPinOffset() != null ? PinOperationTypeV2.GENERATED :
                pin.getPinStatus() == PinStatusV2.RESET_REQUIRED ? PinOperationTypeV2.RESET_REQUIRED : PinOperationTypeV2.INITIALIZED;
        historyRepository.save(PinHistoryV2.builder()
                .pinHistoryId(uuidGenerator.generate())
                .institutionId(card.getInstitutionId())
                .cardId(card.getCardId()).pinDataId(card.getPinDataRef())
                .operationType(operation)
                .resultingStatus(pin.getPinStatus())
                .hsmKeyId(pin.getHsmKeyId())
                .occurredAt(OffsetDateTime.now(ZoneOffset.UTC))
                .actorReference(actor)
                .correlationId(operationId)
                .build());
    }

    private static void validateGenerationPolicy(
            ProductPinPolicyV2 policy, BinConfigurationV2 bin) {
        if (!policy.isRandomPinGenerationEnabled()) {
            throw new BusinessRuleViolationException(
                    "randomPinGenerationEnabled must be true for HSM PIN generation");
        }
        if (policy.getPinLength() != bin.getPinLength()) {
            throw new BusinessRuleViolationException(
                    "Product PIN length must match the BIN PIN length");
        }
    }

    private void requireSwitchAuthority(ProductPinPolicyV2 policy) {
        if (switchSyncEnabled && !switchPinEnabled && policy.isPinRequired())
            throw new BusinessRuleViolationException("Switch card synchronization requires Switch PIN integration for PIN-required issuance");
    }

    private HsmKeyReferenceV2 activeBinKey(
            CardV2 card, KeyPurpose purpose, HsmKeyType requiredType) {
        OffsetDateTime now = OffsetDateTime.now(ZoneOffset.UTC);
        List<BinHsmKeyAssignmentV2> assignments = assignmentRepository.findEffective(
                card.getInstitutionId(), card.getBinId(), purpose,
                BinHsmKeyAssignmentStatus.ACTIVE, now);
        if (assignments.size() != 1) {
            throw new CardSecurityException(
                    "Exactly one active BIN " + purpose + " assignment is required");
        }
        HsmKeyReferenceV2 key = keyRepository
                .findByKeyIdAndInstitutionId(
                        assignments.get(0).getHsmKeyId(), card.getInstitutionId())
                .filter(value -> USABLE_KEY_STATUSES.contains(value.getStatus()))
                .filter(value -> value.getKeyType() == requiredType)
                .filter(value -> scopeMatches(value, card.getBinId()))
                .filter(value -> effective(value, LocalDate.now(ZoneOffset.UTC)))
                .filter(value -> value.getEncryptedKeyBlock() != null
                        && !value.getEncryptedKeyBlock().isBlank())
                .orElseThrow(() -> new CardSecurityException(
                        "Assigned " + requiredType + " key is not active, effective, or HSM-protected"));
        return key;
    }

    private static boolean scopeMatches(HsmKeyReferenceV2 key, UUID binId) {
        return key.getKeyScope() == HsmKeyScope.INSTITUTION
                || (key.getKeyScope() == HsmKeyScope.BIN && binId.equals(key.getBinId()));
    }

    private static boolean effective(HsmKeyReferenceV2 key, LocalDate today) {
        return !today.isBefore(key.getEffectiveFrom()) && today.isBefore(key.getExpiryDate());
    }

    private static int requirePvki(BinConfigurationV2 bin) {
        if (bin.getPvki() == null || bin.getPvki() < 0 || bin.getPvki() > 9) {
            throw new BusinessRuleViolationException(
                    "BIN PVKI must be configured between 0 and 9 for PIN generation");
        }
        return bin.getPvki();
    }
}
