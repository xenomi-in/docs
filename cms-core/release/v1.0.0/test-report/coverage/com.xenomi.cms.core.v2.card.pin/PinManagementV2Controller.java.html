package com.xenomi.cms.core.v2.card.pin;

import java.util.UUID;

import com.xenomi.cms.core.api.ApiResponseEnvelope;
import com.xenomi.cms.core.api.ApiResponseHelper;
import com.xenomi.cms.core.api.PageableDefaults;
import com.xenomi.cms.core.api.PagedResponse;
import com.xenomi.cms.core.v2.card.pin.dto.CardPinHistoryV2Response;
import com.xenomi.cms.core.v2.card.pin.switchpin.SelectedPinRequest;
import com.xenomi.cms.core.v2.card.pin.switchpin.SwitchPinService;
import io.swagger.v3.oas.annotations.Operation;
import io.swagger.v3.oas.annotations.tags.Tag;
import jakarta.validation.Valid;
import org.springframework.data.domain.Pageable;
import org.springframework.http.ResponseEntity;
import org.springframework.security.core.Authentication;
import org.springframework.validation.annotation.Validated;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.bind.annotation.RestController;

@Validated
@RestController
@RequestMapping("/cms/api/v2/institutions/{institutionId}/cards/{cardId}/pin")
@Tag(name = "V2 - PIN Management",
        description = "HSM-backed card PIN change/reset and metadata-only audit history")
public class PinManagementV2Controller {
    private final PinManagementV2Service service;
    private final SwitchPinService switchPins;

    public PinManagementV2Controller(PinManagementV2Service service, SwitchPinService switchPins) {
        this.service = service;
        this.switchPins = switchPins;
    }

    @PostMapping("/actions/initialize")
    @Operation(operationId="initializeCardPinV2", summary="Establish the first customer-selected online PIN",
            description="Requires an issuer authorization grant bound to this operation and versions. CMS encrypts ISO_0 blocks to the Switch PIN recipient. UNKNOWN requires status lookup, not a new operation.")
    public ResponseEntity<ApiResponseEnvelope<SwitchPinService.OperationResult>> initialize(
            @PathVariable UUID institutionId, @PathVariable UUID cardId,
            @Valid @RequestBody SelectedPinRequest request, Authentication authentication) {
        return operationResponse(switchPins.submit(institutionId,cardId,"INITIALIZE_SELECTED_PIN",request,authentication.getName()));
    }

    @PostMapping("/actions/configure")
    @Operation(operationId="configureCardPinProfileV2", summary="Retry initial Switch PIN profile configuration",
            description="Uses the issuance-pinned profile after confirmed card registration. Does not migrate an existing CMS PIN or change an established profile.")
    public ResponseEntity<ApiResponseEnvelope<SwitchPinService.OperationResult>> configure(
            @PathVariable UUID institutionId,@PathVariable UUID cardId,
            @Valid @RequestBody SwitchPinService.ConfigureRequest request,Authentication authentication) {
        return operationResponse(switchPins.configureProfile(institutionId,cardId,request,authentication.getName()));
    }

    @PostMapping("/actions/change")
    @Operation(operationId = "changeCardPinV2", summary = "Change the card PIN",
            description = "Switch verifies and changes the PIN using ISO_0 blocks and explicit input-key aliases. Requires a bound issuer grant and versions; returns a safe operation receipt.")
    public ResponseEntity<ApiResponseEnvelope<SwitchPinService.OperationResult>> change(
            @PathVariable UUID institutionId, @PathVariable UUID cardId,
            @Valid @RequestBody SelectedPinRequest request,
            Authentication authentication) {
        return operationResponse(switchPins.submit(institutionId, cardId,"CHANGE_PIN", request, authentication.getName()));
    }

    @PostMapping("/actions/reset")
    @Operation(operationId = "resetCardPinV2", summary = "Reset the card PIN",
            description = "Switch resets an established PIN using a new encrypted ISO_0 block and a bound recovery grant. First-time selection uses initialize. CMS does not issue grants or store PIN material.")
    public ResponseEntity<ApiResponseEnvelope<SwitchPinService.OperationResult>> reset(
            @PathVariable UUID institutionId, @PathVariable UUID cardId,
            @Valid @RequestBody SelectedPinRequest request,
            Authentication authentication) {
        return operationResponse(switchPins.submit(institutionId,cardId,"RESET_PIN",request,authentication.getName()));
    }

    @GetMapping
    @Operation(operationId="getCardPinStateV2", summary="Get last-confirmed Switch PIN metadata",
            description="Not live Switch state; ATM changes/counters are not included in the current Switch API contract.")
    public ResponseEntity<ApiResponseEnvelope<java.util.Map<String,Object>>> state(
            @PathVariable UUID institutionId,@PathVariable UUID cardId) {
        return ApiResponseHelper.ok(switchPins.state(institutionId,cardId));
    }

    @GetMapping("/operations/{operationId}")
    @Operation(operationId="getCardPinOperationV2", summary="Look up/reconcile a PIN operation",
            description="Queries the Switch receipt for uncertain operations. A missing receipt does not prove no HSM work ran; CMS never resends PIN material automatically.")
    public ResponseEntity<ApiResponseEnvelope<SwitchPinService.OperationResult>> operation(
            @PathVariable UUID institutionId,@PathVariable UUID cardId,@PathVariable UUID operationId) {
        return operationResponse(switchPins.operation(institutionId,cardId,operationId));
    }

    private ResponseEntity<ApiResponseEnvelope<SwitchPinService.OperationResult>> operationResponse(SwitchPinService.OperationResult result) {
        var response=ApiResponseHelper.ok(result);
        return java.util.Set.of("PREPARED","UNKNOWN").contains(result.status())
                ? ResponseEntity.accepted().body(response.getBody()) : response;
    }

    @GetMapping("/history")
    @Operation(operationId = "getCardPinHistoryV2", summary = "Get card PIN history",
            description = "Returns newest-first PIN lifecycle audit events. PIN values, encrypted PIN blocks and PIN verification offsets are never returned.")
    public ResponseEntity<ApiResponseEnvelope<PagedResponse<CardPinHistoryV2Response>>> history(
            @PathVariable UUID institutionId, @PathVariable UUID cardId,
            @RequestParam(required = false) Integer page,
            @RequestParam(required = false) Integer limit) {
        Pageable pageable = PageableDefaults.ofOneBased(page, limit);
        return ApiResponseHelper.ok(PagedResponse.of(
                service.history(institutionId, cardId, pageable)));
    }
}
