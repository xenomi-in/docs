package com.xenomi.cms.core.v2.card.pin;

import com.xenomi.cms.core.hms.service.NettyHsmService;
import com.xenomi.cms.core.v2.bin.model.PinMethod;
import com.xenomi.cms.core.v2.card.security.CardSecurityException;
import com.xenomi.cms.core.v2.institution.entity.HsmKeyReferenceV2;
import org.springframework.stereotype.Component;

@Component
public class FuturexPinHsmOperations implements PinHsmOperations {
    private final NettyHsmService hsm;

    public FuturexPinHsmOperations(NettyHsmService hsm) {
        this.hsm = hsm;
    }

    @Override
    public String generateRandomPinOffset(String pan, PinMethod pinMethod, int pinLength, int pvki,
            HsmKeyReferenceV2 verificationKey, HsmKeyReferenceV2 pinBlockKey) {
        String offset = hsm.generateRandomPinOffset(pan, normalize(pinMethod), pinLength,
                verificationKey.getEncryptedKeyBlock(), String.valueOf(pvki),
                pinBlockKey.getEncryptedKeyBlock());
        if (offset == null || !offset.matches("[0-9]{4,12}")) {
            throw new CardSecurityException("HSM returned invalid PIN-generation metadata");
        }
        return offset;
    }

    @Override
    public String generateOffsetFromEncryptedPinBlock(String pan, String encryptedPinBlock,
            PinMethod pinMethod, int pinLength, int pvki, HsmKeyReferenceV2 verificationKey,
            HsmKeyReferenceV2 pinBlockKey) {
        String offset;
        try {
            offset = hsm.generatePinOffsetFromProtectedBlock(pan, encryptedPinBlock,
                    normalize(pinMethod), pinLength, String.valueOf(pvki),
                    verificationKey.getEncryptedKeyBlock(), pinBlockKey.getEncryptedKeyBlock());
        } catch (RuntimeException exception) {
            throw new CardSecurityException("HSM PIN-reset processing failed", exception);
        }
        return requireOffset(offset);
    }

    @Override
    public PinChangeResult changeEncryptedPin(String pan, String currentEncryptedPinBlock,
            String newEncryptedPinBlock, String currentOffset, PinMethod pinMethod,
            int pinLength, int pvki, HsmKeyReferenceV2 verificationKey,
            HsmKeyReferenceV2 pinBlockKey) {
        String result;
        try {
            result = hsm.changePinUsingProtectedBlocks(pan, currentEncryptedPinBlock,
                    newEncryptedPinBlock, currentOffset, normalize(pinMethod), pinLength,
                    String.valueOf(pvki), verificationKey.getEncryptedKeyBlock(),
                    pinBlockKey.getEncryptedKeyBlock());
        } catch (RuntimeException exception) {
            throw new CardSecurityException("HSM PIN-change processing failed", exception);
        }
        if (result == null || !result.contains(";")) {
            throw new CardSecurityException("HSM returned an invalid PIN-change response");
        }
        String[] values = result.split(";", -1);
        boolean verified = "Y".equalsIgnoreCase(values[0]);
        return new PinChangeResult(verified, verified ? requireOffset(values[1]) : null);
    }

    private static String requireOffset(String offset) {
        if (offset == null || !offset.matches("[0-9]{4,12}")) {
            throw new CardSecurityException("HSM returned invalid PIN verification metadata");
        }
        return offset;
    }

    private static String normalize(PinMethod method) {
        return switch (method) {
            case VISA_PVV -> "VISA";
            case IBM3624 -> "IBM3624";
            case ISO9564 -> "ISO9564";
        };
    }
}
