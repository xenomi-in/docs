package com.xenomi.cms.core.v2.cardcontrol.client;

import java.util.List;
import java.util.UUID;

import org.springframework.beans.factory.annotation.Value;
import org.springframework.http.MediaType;
import org.springframework.stereotype.Component;
import org.springframework.web.client.RestClient;
import org.springframework.web.client.RestClientException;
import org.springframework.http.client.SimpleClientHttpRequestFactory;

@Component
public class HttpSwitchCardControlClient implements SwitchCardControlClient {
    @Value("${cms.switch.sync.enabled:false}")
    private boolean durableSyncEnabled;
    private final RestClient restClient;
    private final String baseUrl;

    public HttpSwitchCardControlClient(RestClient.Builder builder,
            @Value("${cms.switch.base-url:}") String baseUrl,
            @Value("${cms.switch.timeout-ms:3000}") int timeoutMs) {
        SimpleClientHttpRequestFactory requestFactory = new SimpleClientHttpRequestFactory();
        requestFactory.setConnectTimeout(timeoutMs);
        requestFactory.setReadTimeout(timeoutMs);
        this.restClient = builder.requestFactory(requestFactory).build();
        this.baseUrl = baseUrl == null ? "" : baseUrl.trim();
    }

    @Override
    public void apply(UUID institutionId, UUID cardId, int controlVersion,
            boolean internationalEnabled,
            List<SwitchCardControl> controls, String idempotencyKey, String requestId) {
        if (durableSyncEnabled) return;
        if (baseUrl.isBlank()) {
            throw new IllegalStateException("Authorization switch is not configured");
        }
        try {
            SwitchApplyResponse response = restClient.put()
                    .uri(baseUrl + "/internal/v1/institutions/{institutionId}/cards/{cardId}/controls",
                            institutionId, cardId)
                    .contentType(MediaType.APPLICATION_JSON)
                    .header("Idempotency-Key", idempotencyKey)
                    .header("X-Request-Id", requestId)
                    .body(new SwitchApplyRequest(controlVersion, internationalEnabled, controls))
                    .retrieve()
                    .body(SwitchApplyResponse.class);
            if (response == null || !response.applied()
                    || response.appliedVersion() != controlVersion) {
                throw new IllegalStateException(
                        "Authorization switch did not confirm the requested control version");
            }
        } catch (RestClientException ex) {
            throw new IllegalStateException("Authorization switch control update failed", ex);
        }
    }

    private record SwitchApplyRequest(int controlVersion, boolean internationalEnabled,
            List<SwitchCardControl> controls) {
    }

    private record SwitchApplyResponse(boolean applied, int appliedVersion) {
    }
}
