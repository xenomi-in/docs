package com.xenomi.cms.core.v2.cardcontrol.client;

import java.time.OffsetDateTime;
import java.time.ZoneOffset;
import java.util.UUID;

import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Component;
import org.springframework.web.client.RestClient;
import org.springframework.web.client.RestClientException;
import org.springframework.http.client.SimpleClientHttpRequestFactory;

@Component
public class HttpAccountingCardLimitClient implements AccountingCardLimitClient {
    private final RestClient restClient;
    private final String baseUrl;
    private final long freshnessSeconds;

    public HttpAccountingCardLimitClient(RestClient.Builder builder,
            @Value("${cms.accounting.base-url:}") String baseUrl,
            @Value("${cms.accounting.limit-freshness-seconds:60}") long freshnessSeconds,
            @Value("${cms.accounting.timeout-ms:3000}") int timeoutMs) {
        SimpleClientHttpRequestFactory requestFactory = new SimpleClientHttpRequestFactory();
        requestFactory.setConnectTimeout(timeoutMs);
        requestFactory.setReadTimeout(timeoutMs);
        this.restClient = builder.requestFactory(requestFactory).build();
        this.baseUrl = baseUrl == null ? "" : baseUrl.trim();
        this.freshnessSeconds = freshnessSeconds;
    }

    @Override
    public AccountingCardLimit getAssignedLimit(
            UUID institutionId, UUID accountReferenceId, UUID cardId, String requestId) {
        if (baseUrl.isBlank()) {
            throw new IllegalStateException("Accounting card-limit wrapper is not configured");
        }
        try {
            AccountingCardLimit result = restClient.get()
                    .uri(baseUrl + "/internal/v1/institutions/{institutionId}/accounts/{accountId}/card-limit?cardId={cardId}",
                            institutionId, accountReferenceId, cardId)
                    .header("X-Request-Id", requestId)
                    .retrieve()
                    .body(AccountingCardLimit.class);
            validate(result);
            return result;
        } catch (RestClientException ex) {
            throw new IllegalStateException("Accounting card-limit lookup failed", ex);
        }
    }

    private void validate(AccountingCardLimit result) {
        OffsetDateTime now = OffsetDateTime.now(ZoneOffset.UTC);
        if (result == null || result.assignedLimitMinor() < 0
                || result.currencyCode() == null
                || !result.currencyCode().matches("[A-Z]{3}")
                || result.availableAmountMinor() != null && result.availableAmountMinor() < 0
                || result.sourceSystem() != null && result.sourceSystem().length() > 50
                || result.sourceVersion() != null && result.sourceVersion().length() > 120
                || result.retrievedAt() == null || result.expiresAt() == null
                || result.retrievedAt().isAfter(now.plusSeconds(5))
                || !result.expiresAt().isAfter(result.retrievedAt())
                || !result.expiresAt().isAfter(now)
                || result.retrievedAt().isBefore(now.minusSeconds(freshnessSeconds))) {
            throw new IllegalStateException("Accounting returned an invalid or stale card limit");
        }
    }
}
