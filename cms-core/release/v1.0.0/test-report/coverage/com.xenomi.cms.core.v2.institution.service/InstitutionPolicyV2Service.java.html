package com.xenomi.cms.core.v2.institution.service;

import java.util.LinkedHashMap;
import java.util.UUID;

import com.xenomi.cms.core.exception.ResourceNotFoundException;
import com.xenomi.cms.core.v2.institution.dto.OperationalPolicyV2Response;
import com.xenomi.cms.core.v2.institution.dto.SecurityPolicyV2Response;
import com.xenomi.cms.core.v2.institution.dto.UpdateOperationalPolicyV2Request;
import com.xenomi.cms.core.v2.institution.dto.UpdateSecurityPolicyV2Request;
import com.xenomi.cms.core.v2.institution.entity.InstitutionOperationalPolicyV2;
import com.xenomi.cms.core.v2.institution.entity.InstitutionSecurityPolicyV2;
import com.xenomi.cms.core.v2.institution.repository.InstitutionOperationalPolicyV2Repository;
import com.xenomi.cms.core.v2.institution.repository.InstitutionSecurityPolicyV2Repository;
import com.xenomi.cms.core.v2.institution.repository.InstitutionV2Repository;

import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

@Service
public class InstitutionPolicyV2Service {

    private final InstitutionV2Repository institutionRepository;
    private final InstitutionOperationalPolicyV2Repository operationalRepository;
    private final InstitutionSecurityPolicyV2Repository securityRepository;

    public InstitutionPolicyV2Service(
            InstitutionV2Repository institutionRepository,
            InstitutionOperationalPolicyV2Repository operationalRepository,
            InstitutionSecurityPolicyV2Repository securityRepository) {
        this.institutionRepository = institutionRepository;
        this.operationalRepository = operationalRepository;
        this.securityRepository = securityRepository;
    }

    @Transactional(readOnly = true)
    public OperationalPolicyV2Response getOperational(UUID institutionId) {
        return toResponse(operationalRepository.findById(institutionId)
                .orElseThrow(() -> new ResourceNotFoundException(
                        "Institution operational policy", institutionId.toString())));
    }

    @Transactional
    public OperationalPolicyV2Response updateOperational(
            UUID institutionId, UpdateOperationalPolicyV2Request request, String actorId) {
        requireInstitution(institutionId);
        InstitutionOperationalPolicyV2 policy = operationalRepository.findById(institutionId)
                .orElseGet(() -> newOperationalDefaults(institutionId, actorId));
        policy.setCardIssuanceEnabled(request.getCardIssuanceEnabled());
        policy.setReplacementEnabled(request.getReplacementEnabled());
        policy.setRenewalEnabled(request.getRenewalEnabled());
        policy.setPinOperationsEnabled(request.getPinOperationsEnabled());
        policy.setTokenOperationsEnabled(request.getTokenOperationsEnabled());
        policy.setBatchProcessingEnabled(request.getBatchProcessingEnabled());
        policy.setMaintenanceMode(request.getMaintenanceMode());
        policy.setUpdatedBy(actorId.trim());
        return toResponse(operationalRepository.saveAndFlush(policy));
    }

    @Transactional(readOnly = true)
    public SecurityPolicyV2Response getSecurity(UUID institutionId) {
        return toResponse(securityRepository.findById(institutionId)
                .orElseThrow(() -> new ResourceNotFoundException(
                        "Institution security policy", institutionId.toString())));
    }

    @Transactional
    public SecurityPolicyV2Response updateSecurity(
            UUID institutionId, UpdateSecurityPolicyV2Request request, String actorId) {
        requireInstitution(institutionId);
        InstitutionSecurityPolicyV2 policy = securityRepository.findById(institutionId)
                .orElseGet(() -> newSecurityDefaults(institutionId, actorId));
        policy.setMakerCheckerEnabled(request.getMakerCheckerEnabled());
        policy.setMfaRequired(request.getMfaRequired());
        policy.setMaximumLoginAttempts(request.getMaximumLoginAttempts());
        policy.setPasswordPolicyReference(trimToNull(request.getPasswordPolicyReference()));
        policy.setDataMaskingPolicy(new LinkedHashMap<>(request.getDataMaskingPolicy()));
        policy.setAuditRetentionDays(request.getAuditRetentionDays());
        policy.setUpdatedBy(actorId.trim());
        return toResponse(securityRepository.saveAndFlush(policy));
    }

    private void requireInstitution(UUID institutionId) {
        if (!institutionRepository.existsById(institutionId)) {
            throw new ResourceNotFoundException("Institution", institutionId.toString());
        }
    }

    private static InstitutionOperationalPolicyV2 newOperationalDefaults(UUID institutionId, String actorId) {
        return InstitutionOperationalPolicyV2.builder()
                .institutionId(institutionId)
                .cardIssuanceEnabled(true)
                .replacementEnabled(true)
                .renewalEnabled(true)
                .pinOperationsEnabled(true)
                .tokenOperationsEnabled(true)
                .batchProcessingEnabled(true)
                .maintenanceMode(false)
                .policyVersion(1)
                .updatedBy(actorId.trim())
                .build();
    }

    private static InstitutionSecurityPolicyV2 newSecurityDefaults(UUID institutionId, String actorId) {
        return InstitutionSecurityPolicyV2.builder()
                .institutionId(institutionId)
                .makerCheckerEnabled(true)
                .mfaRequired(true)
                .maximumLoginAttempts(5)
                .dataMaskingPolicy(new LinkedHashMap<>())
                .auditRetentionDays(2555)
                .policyVersion(1)
                .updatedBy(actorId.trim())
                .build();
    }

    private static OperationalPolicyV2Response toResponse(InstitutionOperationalPolicyV2 policy) {
        return OperationalPolicyV2Response.builder()
                .institutionId(policy.getInstitutionId())
                .cardIssuanceEnabled(policy.isCardIssuanceEnabled())
                .replacementEnabled(policy.isReplacementEnabled())
                .renewalEnabled(policy.isRenewalEnabled())
                .pinOperationsEnabled(policy.isPinOperationsEnabled())
                .tokenOperationsEnabled(policy.isTokenOperationsEnabled())
                .batchProcessingEnabled(policy.isBatchProcessingEnabled())
                .maintenanceMode(policy.isMaintenanceMode())
                .policyVersion(policy.getPolicyVersion())
                .updatedAt(policy.getUpdatedAt())
                .updatedBy(policy.getUpdatedBy())
                .build();
    }

    private static SecurityPolicyV2Response toResponse(InstitutionSecurityPolicyV2 policy) {
        return SecurityPolicyV2Response.builder()
                .institutionId(policy.getInstitutionId())
                .makerCheckerEnabled(policy.isMakerCheckerEnabled())
                .mfaRequired(policy.isMfaRequired())
                .maximumLoginAttempts(policy.getMaximumLoginAttempts())
                .passwordPolicyReference(policy.getPasswordPolicyReference())
                .dataMaskingPolicy(policy.getDataMaskingPolicy())
                .auditRetentionDays(policy.getAuditRetentionDays())
                .policyVersion(policy.getPolicyVersion())
                .updatedAt(policy.getUpdatedAt())
                .updatedBy(policy.getUpdatedBy())
                .build();
    }

    private static String trimToNull(String value) {
        return value == null || value.isBlank() ? null : value.trim();
    }
}
