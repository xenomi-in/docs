package com.xenomi.cms.core.v2.institution.service;

import java.util.UUID;
import java.util.Locale;

import com.xenomi.cms.core.exception.ResourceAlreadyExistsException;
import com.xenomi.cms.core.exception.ResourceNotFoundException;
import com.xenomi.cms.core.v2.common.DatabaseUuidV7Generator;
import com.xenomi.cms.core.v2.institution.dto.CreateInstitutionV2Request;
import com.xenomi.cms.core.v2.institution.dto.InstitutionAddressV2;
import com.xenomi.cms.core.v2.institution.dto.InstitutionV2Response;
import com.xenomi.cms.core.v2.institution.dto.UpdateInstitutionV2Request;
import com.xenomi.cms.core.v2.institution.entity.InstitutionV2;
import com.xenomi.cms.core.v2.institution.mapper.InstitutionV2Mapper;
import com.xenomi.cms.core.v2.institution.model.InstitutionStatus;
import com.xenomi.cms.core.v2.institution.model.InstitutionType;
import com.xenomi.cms.core.v2.institution.repository.InstitutionV2Repository;

import org.springframework.data.domain.Page;
import org.springframework.data.domain.Pageable;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

@Service
public class InstitutionV2Service {

    private final InstitutionV2Repository repository;
    private final InstitutionV2Mapper mapper;
    private final DatabaseUuidV7Generator uuidGenerator;
    private final InstitutionPolicyProvisioner policyProvisioner;

    public InstitutionV2Service(
            InstitutionV2Repository repository,
            InstitutionV2Mapper mapper,
            DatabaseUuidV7Generator uuidGenerator,
            InstitutionPolicyProvisioner policyProvisioner) {
        this.repository = repository;
        this.mapper = mapper;
        this.uuidGenerator = uuidGenerator;
        this.policyProvisioner = policyProvisioner;
    }

    @Transactional
    public InstitutionV2Response create(CreateInstitutionV2Request request, String actorId) {
        String institutionCode = InstitutionV2Mapper.upper(request.getInstitutionCode());
        if (repository.existsByInstitutionCode(institutionCode)) {
            throw new ResourceAlreadyExistsException("Institution", institutionCode);
        }

        InstitutionV2 institution = mapper.toEntity(request, uuidGenerator.generate(), actorId.trim());
        validateMemberships(institution);
        InstitutionV2 saved = repository.save(institution);
        policyProvisioner.createDefaults(saved.getInstitutionId(), actorId.trim());
        return mapper.toResponse(saved);
    }

    @Transactional(readOnly = true)
    public InstitutionV2Response get(UUID institutionId) {
        return mapper.toResponse(find(institutionId));
    }

    @Transactional(readOnly = true)
    public InstitutionV2Response getByCode(String institutionCode) {
        String normalizedCode = InstitutionV2Mapper.upper(institutionCode);
        return repository.findByInstitutionCode(normalizedCode)
                .map(mapper::toResponse)
                .orElseThrow(() -> new ResourceNotFoundException("Institution", normalizedCode));
    }

    @Transactional(readOnly = true)
    public Page<InstitutionV2Response> search(
            String query,
            InstitutionType institutionType,
            InstitutionStatus status,
            Boolean visaMemberFlag,
            Boolean rupayMemberFlag,
            Pageable pageable) {
        String normalizedQuery = InstitutionV2Mapper.trimToNull(query);
        return repository.search(normalizedQuery, institutionType, status,
                        visaMemberFlag, rupayMemberFlag, pageable)
                .map(mapper::toResponse);
    }

    @Transactional
    public InstitutionV2Response update(UUID institutionId, UpdateInstitutionV2Request request, String actorId) {
        InstitutionV2 institution = find(institutionId);

        if (request.getInstitutionName() != null) {
            institution.setInstitutionName(request.getInstitutionName().trim());
        }
        if (request.getInstitutionShortName() != null) {
            institution.setInstitutionShortName(InstitutionV2Mapper.trimToNull(request.getInstitutionShortName()));
        }
        if (request.getVisaMemberFlag() != null) {
            institution.setVisaMemberFlag(request.getVisaMemberFlag());
            if (!request.getVisaMemberFlag()) {
                institution.setVisaMemberId(null);
            }
        }
        if (request.getVisaMemberId() != null) {
            institution.setVisaMemberId(InstitutionV2Mapper.trimToNull(request.getVisaMemberId()));
        }
        if (request.getRupayMemberFlag() != null) {
            institution.setRupayMemberFlag(request.getRupayMemberFlag());
            if (!request.getRupayMemberFlag()) {
                institution.setNpciMemberId(null);
            }
        }
        if (request.getNpciMemberId() != null) {
            institution.setNpciMemberId(InstitutionV2Mapper.trimToNull(request.getNpciMemberId()));
        }
        if (request.getRegisteredAddress() != null) {
            applyAddress(institution, request.getRegisteredAddress());
        }
        if (request.getContactPhonePrimary() != null) {
            institution.setContactPhonePrimary(request.getContactPhonePrimary().trim());
        }
        if (request.getContactPhoneSecondary() != null) {
            institution.setContactPhoneSecondary(InstitutionV2Mapper.trimToNull(request.getContactPhoneSecondary()));
        }
        if (request.getContactEmailPrimary() != null) {
            institution.setContactEmailPrimary(
                    request.getContactEmailPrimary().trim().toLowerCase(Locale.ROOT));
        }
        if (request.getContactEmailSecondary() != null) {
            institution.setContactEmailSecondary(lowerOrNull(request.getContactEmailSecondary()));
        }
        if (request.getWebsiteUrl() != null) {
            institution.setWebsiteUrl(InstitutionV2Mapper.trimToNull(request.getWebsiteUrl()));
        }
        if (request.getTimezone() != null) {
            institution.setTimezone(request.getTimezone().trim());
        }
        if (request.getReportDeliverySftpPath() != null) {
            institution.setReportDeliverySftpPath(
                    InstitutionV2Mapper.trimToNull(request.getReportDeliverySftpPath()));
        }
        if (request.getReportDeliveryEmail() != null) {
            institution.setReportDeliveryEmail(lowerOrNull(request.getReportDeliveryEmail()));
        }
      //  if (request.getPrepaidCardIssuingEnabled() != null) {
     //       institution.setPrepaidCardIssuingEnabled(request.getPrepaidCardIssuingEnabled());
     //   }
        if (request.getCreditCardIssuingEnabled() != null) {
            institution.setCreditCardIssuingEnabled(request.getCreditCardIssuingEnabled());
        }
        if (request.getDefaultCardValidityMonths() != null) {
            institution.setDefaultCardValidityMonths(request.getDefaultCardValidityMonths());
        }
        if (request.getBaseCurrency() != null) {
            institution.setBaseCurrency(InstitutionV2Mapper.upper(request.getBaseCurrency()));
        }
        if (request.getCurrencyExponent() != null) {
            institution.setCurrencyExponent(request.getCurrencyExponent());
        }
        if (request.getBusinessDate() != null) {
            institution.setBusinessDate(request.getBusinessDate());
        }
        if (request.getStatus() != null) {
            institution.setStatus(request.getStatus());
        }

        validateMemberships(institution);
        institution.setUpdatedBy(actorId.trim());
        return mapper.toResponse(repository.save(institution));
    }

    private InstitutionV2 find(UUID institutionId) {
        return repository.findById(institutionId)
                .orElseThrow(() -> new ResourceNotFoundException("Institution", institutionId.toString()));
    }

    private static void applyAddress(InstitutionV2 institution, InstitutionAddressV2 address) {
        institution.setRegisteredAddressLine1(address.getAddressLine1().trim());
        institution.setRegisteredAddressLine2(InstitutionV2Mapper.trimToNull(address.getAddressLine2()));
        institution.setCity(address.getCity().trim());
        institution.setStateCode(InstitutionV2Mapper.upper(address.getStateCode()));
        institution.setCountryCode(address.getCountryCode() == null
                ? "IND" : InstitutionV2Mapper.upper(address.getCountryCode()));
        institution.setPostalCode(address.getPostalCode().trim());
    }

    private static void validateMemberships(InstitutionV2 institution) {
        if (institution.isVisaMemberFlag() && institution.getVisaMemberId() == null) {
            throw new IllegalArgumentException("visaMemberId is required when visaMemberFlag is true");
        }
        if (!institution.isVisaMemberFlag() && institution.getVisaMemberId() != null) {
            throw new IllegalArgumentException("visaMemberId is allowed only when visaMemberFlag is true");
        }
        if (institution.isRupayMemberFlag() && institution.getNpciMemberId() == null) {
            throw new IllegalArgumentException("npciMemberId is required when rupayMemberFlag is true");
        }
        if (!institution.isRupayMemberFlag() && institution.getNpciMemberId() != null) {
            throw new IllegalArgumentException("npciMemberId is allowed only when rupayMemberFlag is true");
        }
    }

    private static String lowerOrNull(String value) {
        String normalized = InstitutionV2Mapper.trimToNull(value);
        return normalized == null ? null : normalized.toLowerCase(Locale.ROOT);
    }
}
