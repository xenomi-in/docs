package com.xenomi.cms.core.v2.institution.service;

import java.time.OffsetDateTime;
import java.util.EnumMap;
import java.util.Map;
import java.util.Set;
import java.util.UUID;

import com.xenomi.cms.core.exception.ResourceNotFoundException;
import com.xenomi.cms.core.v2.common.DatabaseUuidV7Generator;
import com.xenomi.cms.core.v2.institution.dto.CreateDataKeyAssignmentV2Request;
import com.xenomi.cms.core.v2.institution.dto.DataKeyAssignmentV2Response;
import com.xenomi.cms.core.v2.institution.dto.UpdateDataKeyAssignmentStatusV2Request;
import com.xenomi.cms.core.v2.institution.entity.InstitutionDataKeyAssignmentV2;
import com.xenomi.cms.core.v2.institution.model.DataKeyAssignmentStatus;
import com.xenomi.cms.core.v2.institution.repository.HsmKeyReferenceV2Repository;
import com.xenomi.cms.core.v2.institution.repository.InstitutionDataKeyAssignmentV2Repository;
import com.xenomi.cms.core.v2.institution.repository.InstitutionSecurityPolicyV2Repository;
import com.xenomi.cms.core.v2.institution.repository.InstitutionV2Repository;
import com.xenomi.cms.core.v2.hsm.model.HsmKeyStatus;
import com.xenomi.cms.core.v2.hsm.model.HsmKeyType;
import com.xenomi.cms.core.v2.hsm.model.HsmKeyScope;
import com.xenomi.cms.core.v2.hsm.model.KeyPurpose;

import org.springframework.data.domain.Page;
import org.springframework.data.domain.Pageable;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

@Service
public class InstitutionDataKeyAssignmentV2Service {

    private static final Set<HsmKeyStatus> ASSIGNABLE_KEY_STATUSES = Set.of(
            HsmKeyStatus.PENDING_APPROVAL, HsmKeyStatus.SCHEDULED,
            HsmKeyStatus.ACTIVE, HsmKeyStatus.DECRYPT_ONLY);

    private static final Map<DataKeyAssignmentStatus, Set<DataKeyAssignmentStatus>> ALLOWED_TRANSITIONS =
            transitions();

    private final InstitutionV2Repository institutionRepository;
    private final HsmKeyReferenceV2Repository hsmKeyRepository;
    private final InstitutionDataKeyAssignmentV2Repository assignmentRepository;
    private final InstitutionSecurityPolicyV2Repository securityPolicyRepository;
    private final DatabaseUuidV7Generator uuidGenerator;

    public InstitutionDataKeyAssignmentV2Service(
            InstitutionV2Repository institutionRepository,
            HsmKeyReferenceV2Repository hsmKeyRepository,
            InstitutionDataKeyAssignmentV2Repository assignmentRepository,
            InstitutionSecurityPolicyV2Repository securityPolicyRepository,
            DatabaseUuidV7Generator uuidGenerator) {
        this.institutionRepository = institutionRepository;
        this.hsmKeyRepository = hsmKeyRepository;
        this.assignmentRepository = assignmentRepository;
        this.securityPolicyRepository = securityPolicyRepository;
        this.uuidGenerator = uuidGenerator;
    }

    @Transactional
    public DataKeyAssignmentV2Response create(
            UUID institutionId, CreateDataKeyAssignmentV2Request request, String actorId) {
        institutionRepository.findByIdForUpdate(institutionId)
                .orElseThrow(() -> new ResourceNotFoundException("Institution", institutionId.toString()));

        if (!hsmKeyRepository.existsByKeyIdAndInstitutionIdAndStatusIn(
                request.getHsmKeyId(), institutionId, ASSIGNABLE_KEY_STATUSES)) {
            throw new IllegalArgumentException(
                    "hsmKeyId must reference an assignable HSM key owned by the institution");
        }
        var key = hsmKeyRepository.findByKeyIdAndInstitutionId(
                request.getHsmKeyId(), institutionId).orElseThrow();
        if (key.getKeyType() == HsmKeyType.KEK) {
            throw new IllegalArgumentException(
                    "KEK cannot be used as an operational institution data key");
        }
        validateDates(request.getEffectiveFrom(), request.getEffectiveTo());

        KeyPurpose keyPurpose = request.getKeyPurpose();
        if (keyPurpose.scope() != HsmKeyScope.INSTITUTION) {
            throw new IllegalArgumentException("keyPurpose must be approved for institution scope");
        }
        if (key.getKeyScope() != HsmKeyScope.INSTITUTION) {
            throw new IllegalArgumentException("Institution data keys must use INSTITUTION scope");
        }
        HsmKeyType requiredType = switch (keyPurpose) {
            case PAN_HMAC, AUDIT_HMAC -> HsmKeyType.HMAC;
            case PAN_DATA_ENCRYPTION, CARDHOLDER_DATA_ENCRYPTION, PII_ENCRYPTION,
                    FILE_ENCRYPTION, TOKEN_REFERENCE_ENCRYPTION -> HsmKeyType.DEK;
            default -> throw new IllegalArgumentException("Unsupported institution key purpose");
        };
        if (key.getKeyType() != requiredType) {
            throw new IllegalArgumentException(keyPurpose + " requires an " + requiredType + " key");
        }
        int nextVersion = assignmentRepository.findMaximumVersion(institutionId, keyPurpose) + 1;
        String actor = actorId.trim();
        InstitutionDataKeyAssignmentV2 assignment = InstitutionDataKeyAssignmentV2.builder()
                .assignmentId(uuidGenerator.generate())
                .institutionId(institutionId)
                .hsmKeyId(request.getHsmKeyId())
                .keyPurpose(keyPurpose)
                .dataClassification(request.getDataClassification())
                .assignmentVersion(nextVersion)
                .assignmentStatus(DataKeyAssignmentStatus.PENDING_APPROVAL)
                .effectiveFrom(request.getEffectiveFrom())
                .effectiveTo(request.getEffectiveTo())
                .rotationPolicyReference(trimToNull(request.getRotationPolicyReference()))
                .createdBy(actor)
                .updatedBy(actor)
                .build();
        return toResponse(assignmentRepository.save(assignment));
    }

    @Transactional(readOnly = true)
    public DataKeyAssignmentV2Response get(UUID institutionId, UUID assignmentId) {
        return toResponse(find(institutionId, assignmentId));
    }

    @Transactional(readOnly = true)
    public Page<DataKeyAssignmentV2Response> search(
            UUID institutionId,
            KeyPurpose keyPurpose,
            DataKeyAssignmentStatus status,
            Pageable pageable) {
        if (!institutionRepository.existsById(institutionId)) {
            throw new ResourceNotFoundException("Institution", institutionId.toString());
        }
        return assignmentRepository.search(institutionId, keyPurpose, status, pageable)
                .map(InstitutionDataKeyAssignmentV2Service::toResponse);
    }

    @Transactional
    public DataKeyAssignmentV2Response changeStatus(
            UUID institutionId,
            UUID assignmentId,
            UpdateDataKeyAssignmentStatusV2Request request,
            String actorId) {
        InstitutionDataKeyAssignmentV2 assignment = find(institutionId, assignmentId);
        DataKeyAssignmentStatus current = assignment.getAssignmentStatus();
        DataKeyAssignmentStatus requested = request.getStatus();
        if (!ALLOWED_TRANSITIONS.getOrDefault(current, Set.of()).contains(requested)) {
            throw new IllegalArgumentException(
                    "Invalid data-key assignment transition: " + current + " -> " + requested);
        }
        enforceApprovalGate(institutionId, assignment, current, requested);

        if (requested == DataKeyAssignmentStatus.ACTIVE) {
            var key = hsmKeyRepository.findByKeyIdAndInstitutionId(
                    assignment.getHsmKeyId(), institutionId)
                    .orElseThrow(() -> new ResourceNotFoundException(
                            "HSM key", assignment.getHsmKeyId().toString()));
            if (key.getStatus() != HsmKeyStatus.ACTIVE) {
                throw new IllegalArgumentException(
                        "The referenced HSM key must be ACTIVE before assignment activation");
            }
            OffsetDateTime now = OffsetDateTime.now(java.time.ZoneOffset.UTC);
            assignmentRepository
                    .findFirstByInstitutionIdAndKeyPurposeAndAssignmentStatusAndAssignmentIdNot(
                            institutionId, assignment.getKeyPurpose(),
                            DataKeyAssignmentStatus.ACTIVE, assignmentId)
                    .ifPresent(previous -> {
                        previous.setAssignmentStatus(DataKeyAssignmentStatus.DECRYPT_ONLY);
                        previous.setEffectiveTo(now);
                        previous.setUpdatedBy(actorId.trim());
                        assignmentRepository.save(previous);
                    });
        }

        OffsetDateTime effectiveTo = request.getEffectiveTo();
        if ((requested == DataKeyAssignmentStatus.RETIRED
                || requested == DataKeyAssignmentStatus.REVOKED) && effectiveTo == null) {
            effectiveTo = OffsetDateTime.now(java.time.ZoneOffset.UTC);
        }
        validateDates(assignment.getEffectiveFrom(), effectiveTo);
        assignment.setAssignmentStatus(requested);
        assignment.setEffectiveTo(effectiveTo);
        assignment.setUpdatedBy(actorId.trim());
        return toResponse(assignmentRepository.save(assignment));
    }

    private InstitutionDataKeyAssignmentV2 find(UUID institutionId, UUID assignmentId) {
        return assignmentRepository.findByAssignmentIdAndInstitutionId(assignmentId, institutionId)
                .orElseThrow(() -> new ResourceNotFoundException(
                        "Institution data-key assignment", assignmentId.toString()));
    }

    private void enforceApprovalGate(
            UUID institutionId,
            InstitutionDataKeyAssignmentV2 assignment,
            DataKeyAssignmentStatus current,
            DataKeyAssignmentStatus requested) {
        boolean activationFromPending = current == DataKeyAssignmentStatus.PENDING_APPROVAL
                && (requested == DataKeyAssignmentStatus.SCHEDULED
                    || requested == DataKeyAssignmentStatus.ACTIVE);
        if (!activationFromPending) {
            return;
        }
        boolean makerCheckerEnabled = securityPolicyRepository.findById(institutionId)
                .map(policy -> policy.isMakerCheckerEnabled())
                .orElse(true);
        if (makerCheckerEnabled && assignment.getApprovalRequestId() == null) {
            throw new IllegalArgumentException(
                    "An approved approvalRequestId is required before activating this assignment");
        }
    }

    private static void validateDates(OffsetDateTime effectiveFrom, OffsetDateTime effectiveTo) {
        if (effectiveTo != null && !effectiveTo.isAfter(effectiveFrom)) {
            throw new IllegalArgumentException("effectiveTo must be later than effectiveFrom");
        }
    }

    private static String trimToNull(String value) {
        return value == null || value.isBlank() ? null : value.trim();
    }

    private static DataKeyAssignmentV2Response toResponse(InstitutionDataKeyAssignmentV2 assignment) {
        return DataKeyAssignmentV2Response.builder()
                .assignmentId(assignment.getAssignmentId())
                .institutionId(assignment.getInstitutionId())
                .hsmKeyId(assignment.getHsmKeyId())
                .keyPurpose(assignment.getKeyPurpose())
                .dataClassification(assignment.getDataClassification())
                .assignmentVersion(assignment.getAssignmentVersion())
                .assignmentStatus(assignment.getAssignmentStatus())
                .effectiveFrom(assignment.getEffectiveFrom())
                .effectiveTo(assignment.getEffectiveTo())
                .rotationPolicyReference(assignment.getRotationPolicyReference())
                .approvalRequestId(assignment.getApprovalRequestId())
                .createdAt(assignment.getCreatedAt())
                .updatedAt(assignment.getUpdatedAt())
                .createdBy(assignment.getCreatedBy())
                .updatedBy(assignment.getUpdatedBy())
                .build();
    }

    private static Map<DataKeyAssignmentStatus, Set<DataKeyAssignmentStatus>> transitions() {
        Map<DataKeyAssignmentStatus, Set<DataKeyAssignmentStatus>> transitions =
                new EnumMap<>(DataKeyAssignmentStatus.class);
        transitions.put(DataKeyAssignmentStatus.PENDING_APPROVAL,
                Set.of(DataKeyAssignmentStatus.SCHEDULED, DataKeyAssignmentStatus.ACTIVE,
                        DataKeyAssignmentStatus.REVOKED));
        transitions.put(DataKeyAssignmentStatus.SCHEDULED,
                Set.of(DataKeyAssignmentStatus.ACTIVE, DataKeyAssignmentStatus.SUSPENDED,
                        DataKeyAssignmentStatus.REVOKED));
        transitions.put(DataKeyAssignmentStatus.ACTIVE,
                Set.of(DataKeyAssignmentStatus.DECRYPT_ONLY, DataKeyAssignmentStatus.SUSPENDED,
                        DataKeyAssignmentStatus.RETIRED, DataKeyAssignmentStatus.REVOKED));
        transitions.put(DataKeyAssignmentStatus.DECRYPT_ONLY,
                Set.of(DataKeyAssignmentStatus.RETIRED, DataKeyAssignmentStatus.REVOKED));
        transitions.put(DataKeyAssignmentStatus.SUSPENDED,
                Set.of(DataKeyAssignmentStatus.ACTIVE, DataKeyAssignmentStatus.RETIRED,
                        DataKeyAssignmentStatus.REVOKED));
        transitions.put(DataKeyAssignmentStatus.RETIRED, Set.of());
        transitions.put(DataKeyAssignmentStatus.REVOKED, Set.of());
        return Map.copyOf(transitions);
    }
}
