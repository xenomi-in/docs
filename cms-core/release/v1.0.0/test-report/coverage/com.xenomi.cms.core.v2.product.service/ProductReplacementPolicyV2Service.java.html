package com.xenomi.cms.core.v2.product.service;

import java.util.EnumSet;
import java.util.HashSet;
import java.util.Set;
import java.util.UUID;

import com.xenomi.cms.core.exception.ResourceNotFoundException;
import com.xenomi.cms.core.v2.common.DatabaseUuidV7Generator;
import com.xenomi.cms.core.v2.product.dto.CreateProductReplacementPolicyV2Request;
import com.xenomi.cms.core.v2.product.dto.ProductReplacementPolicyV2Response;
import com.xenomi.cms.core.v2.product.dto.UpdateProductReplacementPolicyV2Request;
import com.xenomi.cms.core.v2.product.entity.ProductReplacementPolicyV2;
import com.xenomi.cms.core.v2.product.entity.ProductVersionV2;
import com.xenomi.cms.core.v2.product.model.OldCardUsagePolicy;
import com.xenomi.cms.core.v2.product.model.ProductPolicyStatus;
import com.xenomi.cms.core.v2.product.model.ProductVersionStatus;
import com.xenomi.cms.core.v2.product.model.ReplacementAccountLinkStrategy;
import com.xenomi.cms.core.v2.product.model.ReplacementControlStrategy;
import com.xenomi.cms.core.v2.product.model.ReplacementPanStrategy;
import com.xenomi.cms.core.v2.product.model.ReplacementPinStrategy;
import com.xenomi.cms.core.v2.product.model.ReplacementReason;
import com.xenomi.cms.core.v2.product.repository.ProductReplacementPolicyV2Repository;
import com.xenomi.cms.core.v2.product.repository.ProductVersionV2Repository;
import org.springframework.data.domain.Page;
import org.springframework.data.domain.Pageable;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

@Service
public class ProductReplacementPolicyV2Service {
    private static final EnumSet<ReplacementReason> SECURITY_REPLACEMENT_REASONS =
            EnumSet.of(ReplacementReason.LOST, ReplacementReason.STOLEN,
                    ReplacementReason.COMPROMISED);
    private final ProductReplacementPolicyV2Repository repository;
    private final ProductVersionV2Repository versionRepository;
    private final DatabaseUuidV7Generator uuidGenerator;

    public ProductReplacementPolicyV2Service(
            ProductReplacementPolicyV2Repository repository,
            ProductVersionV2Repository versionRepository,
            DatabaseUuidV7Generator uuidGenerator) {
        this.repository = repository;
        this.versionRepository = versionRepository;
        this.uuidGenerator = uuidGenerator;
    }

    @Transactional
    public ProductReplacementPolicyV2Response create(UUID institutionId, UUID productId,
            UUID versionId, CreateProductReplacementPolicyV2Request request) {
        requireDraftVersion(institutionId, productId, versionId);
        if (request.permittedChannels() == null || request.permittedChannels().isEmpty()) {
            throw new IllegalArgumentException("permittedChannels must contain at least one channel");
        }
        if (repository.existsByInstitutionIdAndProductVersionIdAndReplacementReason(
                institutionId, versionId, request.replacementReason())) {
            throw new IllegalArgumentException(
                    "A replacement policy already exists for reason " + request.replacementReason());
        }
        ProductReplacementPolicyV2 entity = ProductReplacementPolicyV2.builder()
                .policyId(uuidGenerator.generate())
                .institutionId(institutionId)
                .productVersionId(versionId)
                .replacementReason(request.replacementReason())
                .panStrategy(request.panStrategy())
                .expiryStrategy(request.expiryStrategy())
                .oldCardUsagePolicy(request.oldCardUsagePolicy())
                .pinStrategy(request.pinStrategy() == null
                        ? ReplacementPinStrategy.GENERATE_NEW_PIN : request.pinStrategy())
                .controlStrategy(request.controlStrategy() == null
                        ? ReplacementControlStrategy.PRESERVE_AND_REVALIDATE : request.controlStrategy())
                .accountLinkStrategy(request.accountLinkStrategy() == null
                        ? ReplacementAccountLinkStrategy.PRESERVE_ACCOUNT_LINK : request.accountLinkStrategy())
                .permittedChannels(new HashSet<>(request.permittedChannels()))
                .replacementAllowed(request.replacementAllowed() == null || request.replacementAllowed())
                .approvalRequired(request.approvalRequired() != null && request.approvalRequired())
                .policyStatus(ProductPolicyStatus.DRAFT)
                .build();
        validateSecurityCombinations(entity);
        return toResponse(repository.save(entity));
    }

    @Transactional(readOnly = true)
    public Page<ProductReplacementPolicyV2Response> list(
            UUID institutionId, UUID productId, UUID versionId, Pageable pageable) {
        requireVersion(institutionId, productId, versionId);
        return repository.findByInstitutionIdAndProductVersionId(institutionId, versionId, pageable)
                .map(this::toResponse);
    }

    @Transactional(readOnly = true)
    public ProductReplacementPolicyV2Response get(UUID institutionId, UUID productId,
            UUID versionId, UUID policyId) {
        requireVersion(institutionId, productId, versionId);
        return toResponse(find(institutionId, versionId, policyId));
    }

    @Transactional
    public ProductReplacementPolicyV2Response update(UUID institutionId, UUID productId,
            UUID versionId, UUID policyId, UpdateProductReplacementPolicyV2Request request) {
        requireDraftVersion(institutionId, productId, versionId);
        ProductReplacementPolicyV2 entity = find(institutionId, versionId, policyId);
        if (request.panStrategy() != null) entity.setPanStrategy(request.panStrategy());
        if (request.expiryStrategy() != null) entity.setExpiryStrategy(request.expiryStrategy());
        if (request.oldCardUsagePolicy() != null) entity.setOldCardUsagePolicy(request.oldCardUsagePolicy());
        if (request.pinStrategy() != null) entity.setPinStrategy(request.pinStrategy());
        if (request.controlStrategy() != null) entity.setControlStrategy(request.controlStrategy());
        if (request.accountLinkStrategy() != null) entity.setAccountLinkStrategy(request.accountLinkStrategy());
        if (request.permittedChannels() != null) {
            if (request.permittedChannels().isEmpty()) {
                throw new IllegalArgumentException("permittedChannels must contain at least one channel");
            }
            entity.setPermittedChannels(new HashSet<>(request.permittedChannels()));
        }
        if (request.replacementAllowed() != null) entity.setReplacementAllowed(request.replacementAllowed());
        if (request.approvalRequired() != null) entity.setApprovalRequired(request.approvalRequired());
        entity.setPolicyStatus(ProductPolicyStatus.DRAFT);
        validateSecurityCombinations(entity);
        return toResponse(repository.save(entity));
    }

    private ProductVersionV2 requireVersion(UUID institutionId, UUID productId, UUID versionId) {
        return versionRepository.findByProductVersionIdAndInstitutionIdAndProductId(
                        versionId, institutionId, productId)
                .orElseThrow(() -> new ResourceNotFoundException("Product version", versionId.toString()));
    }

    private void requireDraftVersion(UUID institutionId, UUID productId, UUID versionId) {
        if (requireVersion(institutionId, productId, versionId).getVersionStatus() != ProductVersionStatus.DRAFT) {
            throw new IllegalArgumentException(
                    "Product replacement policies can be changed only while the version is DRAFT");
        }
    }

    private ProductReplacementPolicyV2 find(UUID institutionId, UUID versionId, UUID policyId) {
        return repository.findByPolicyIdAndInstitutionIdAndProductVersionId(
                        policyId, institutionId, versionId)
                .orElseThrow(() -> new ResourceNotFoundException(
                        "Product replacement policy", policyId.toString()));
    }

    private static void validateSecurityCombinations(ProductReplacementPolicyV2 policy) {
        if (!policy.isReplacementAllowed()) return;
        ReplacementReason reason = policy.getReplacementReason();
        if (SECURITY_REPLACEMENT_REASONS.contains(reason)
                && policy.getOldCardUsagePolicy() != OldCardUsagePolicy.BLOCK_IMMEDIATELY) {
            throw new IllegalArgumentException(reason + " replacements must block the old card immediately");
        }
        if (SECURITY_REPLACEMENT_REASONS.contains(reason)
                && policy.getPanStrategy() != ReplacementPanStrategy.GENERATE_NEW_PAN) {
            throw new IllegalArgumentException(
                    reason + " replacements must generate a new PAN");
        }
        boolean manualOrRiskDecision = policy.getPanStrategy() == ReplacementPanStrategy.RISK_BASED
                || policy.getPanStrategy() == ReplacementPanStrategy.MANUAL_DECISION;
        if (manualOrRiskDecision && !policy.isApprovalRequired()) {
            throw new IllegalArgumentException(
                    "approvalRequired must be true for risk-based or manual replacement decisions");
        }
    }

    private ProductReplacementPolicyV2Response toResponse(ProductReplacementPolicyV2 entity) {
        return new ProductReplacementPolicyV2Response(
                entity.getPolicyId(), entity.getInstitutionId(), entity.getProductVersionId(),
                entity.getReplacementReason(), entity.getPanStrategy(), entity.getExpiryStrategy(),
                entity.getOldCardUsagePolicy(), entity.getPinStrategy(), entity.getControlStrategy(),
                entity.getAccountLinkStrategy(),
                Set.copyOf(entity.getPermittedChannels()),
                entity.isReplacementAllowed(), entity.isApprovalRequired(), entity.getPolicyStatus());
    }
}
