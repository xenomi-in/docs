package com.xenomi.cms.core.v2.product.service;

import java.time.OffsetDateTime;
import java.util.UUID;

import com.xenomi.cms.core.exception.ResourceNotFoundException;
import com.xenomi.cms.core.v2.bin.model.CardNetwork;
import com.xenomi.cms.core.v2.common.DatabaseUuidV7Generator;
import com.xenomi.cms.core.v2.product.dto.CardDesignV2Response;
import com.xenomi.cms.core.v2.product.dto.CreateCardDesignV2Request;
import com.xenomi.cms.core.v2.product.dto.UpdateCardDesignV2Request;
import com.xenomi.cms.core.v2.product.entity.CardDesignV2;
import com.xenomi.cms.core.v2.product.model.CardDesignStatus;
import com.xenomi.cms.core.v2.product.repository.CardDesignV2Repository;
import org.springframework.data.domain.Page;
import org.springframework.data.domain.Pageable;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

@Service
public class CardDesignV2Service {
    private final CardDesignV2Repository repository;
    private final DatabaseUuidV7Generator uuidGenerator;

    public CardDesignV2Service(CardDesignV2Repository repository, DatabaseUuidV7Generator uuidGenerator) {
        this.repository = repository;
        this.uuidGenerator = uuidGenerator;
    }

    @Transactional
    public CardDesignV2Response create(UUID institutionId, CreateCardDesignV2Request request, String actorId) {
        String code = request.designCode().trim().toUpperCase();
        if (repository.existsByInstitutionIdAndDesignCodeAndDesignVersion(
                institutionId, code, request.designVersion())) {
            throw new IllegalArgumentException("The card design code and version already exist");
        }
        validateDates(request.effectiveFrom(), request.effectiveTo());
        CardDesignV2 entity = CardDesignV2.builder()
                .cardDesignId(uuidGenerator.generate()).institutionId(institutionId)
                .designCode(code).designName(request.designName().trim())
                .designVersion(request.designVersion()).cardNetwork(request.cardNetwork())
                .orientation(request.orientation()).designStatus(CardDesignStatus.DRAFT)
                .effectiveFrom(request.effectiveFrom()).effectiveTo(request.effectiveTo())
                .createdBy(actorId.trim()).updatedBy(actorId.trim()).build();
        return toResponse(repository.save(entity));
    }

    @Transactional(readOnly = true)
    public Page<CardDesignV2Response> search(UUID institutionId, CardNetwork network,
            CardDesignStatus status, Pageable pageable) {
        return repository.search(institutionId, network, status, pageable).map(this::toResponse);
    }

    @Transactional(readOnly = true)
    public CardDesignV2Response get(UUID institutionId, UUID designId) {
        return toResponse(find(institutionId, designId));
    }

    @Transactional
    public CardDesignV2Response update(UUID institutionId, UUID designId,
            UpdateCardDesignV2Request request, String actorId) {
        CardDesignV2 entity = find(institutionId, designId);
        boolean configurationChange = request.designName() != null || request.orientation() != null
                || request.effectiveFrom() != null || request.effectiveTo() != null;
        if (configurationChange && entity.getDesignStatus() != CardDesignStatus.DRAFT) {
            throw new IllegalArgumentException("Card design configuration can be changed only while DRAFT");
        }
        if (request.designName() != null) entity.setDesignName(request.designName().trim());
        if (request.orientation() != null) entity.setOrientation(request.orientation());
        if (request.effectiveFrom() != null) entity.setEffectiveFrom(request.effectiveFrom());
        if (request.effectiveTo() != null) entity.setEffectiveTo(request.effectiveTo());
        if (request.designStatus() != null && request.designStatus() != entity.getDesignStatus()) {
            validateTransition(entity.getDesignStatus(), request.designStatus());
            entity.setDesignStatus(request.designStatus());
        }
        validateDates(entity.getEffectiveFrom(), entity.getEffectiveTo());
        entity.setUpdatedBy(actorId.trim());
        return toResponse(repository.save(entity));
    }

    private CardDesignV2 find(UUID institutionId, UUID designId) {
        return repository.findByCardDesignIdAndInstitutionId(designId, institutionId)
                .orElseThrow(() -> new ResourceNotFoundException("Card design", designId.toString()));
    }

    private static void validateDates(OffsetDateTime from, OffsetDateTime to) {
        if (to != null && !to.isAfter(from)) throw new IllegalArgumentException("effectiveTo must be after effectiveFrom");
    }

    private static void validateTransition(CardDesignStatus current, CardDesignStatus target) {
        boolean allowed = switch (current) {
            case DRAFT -> target == CardDesignStatus.ACTIVE || target == CardDesignStatus.RETIRED;
            case ACTIVE -> target == CardDesignStatus.SUSPENDED || target == CardDesignStatus.RETIRED;
            case SUSPENDED -> target == CardDesignStatus.ACTIVE || target == CardDesignStatus.RETIRED;
            case RETIRED -> false;
        };
        if (!allowed) throw new IllegalArgumentException("Invalid card design status transition: " + current + " -> " + target);
    }

    private CardDesignV2Response toResponse(CardDesignV2 d) {
        return new CardDesignV2Response(d.getCardDesignId(), d.getInstitutionId(), d.getDesignCode(),
                d.getDesignName(), d.getDesignVersion(), d.getCardNetwork(), d.getOrientation(),
                d.getDesignStatus(), d.getEffectiveFrom(), d.getEffectiveTo(), d.getCreatedAt(),
                d.getUpdatedAt(), d.getCreatedBy(), d.getUpdatedBy());
    }
}
