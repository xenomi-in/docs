package com.xenomi.cms.core.v2.product.service;

import java.util.UUID;

import com.xenomi.cms.core.exception.ResourceNotFoundException;
import com.xenomi.cms.core.v2.product.dto.ProductCardFormPolicyV2Response;
import com.xenomi.cms.core.v2.product.dto.ProductPersonalisationPolicyV2Response;
import com.xenomi.cms.core.v2.product.dto.UpsertProductCardFormPolicyV2Request;
import com.xenomi.cms.core.v2.product.dto.UpsertProductPersonalisationPolicyV2Request;
import com.xenomi.cms.core.v2.product.entity.ProductCardFormPolicyV2;
import com.xenomi.cms.core.v2.product.entity.ProductPersonalisationPolicyV2;
import com.xenomi.cms.core.v2.product.entity.ProductPinPolicyV2;
import com.xenomi.cms.core.v2.product.entity.ProductVersionV2;
import com.xenomi.cms.core.v2.product.model.CardIssuanceMode;
import com.xenomi.cms.core.v2.product.model.PersonalisationMode;
import com.xenomi.cms.core.v2.product.model.PhysicalIssueTiming;
import com.xenomi.cms.core.v2.product.model.ProductPolicyStatus;
import com.xenomi.cms.core.v2.product.model.ProductVersionStatus;
import com.xenomi.cms.core.v2.product.model.VirtualPanStrategy;
import com.xenomi.cms.core.v2.product.repository.ProductCardFormPolicyV2Repository;
import com.xenomi.cms.core.v2.product.repository.ProductPersonalisationPolicyV2Repository;
import com.xenomi.cms.core.v2.product.repository.ProductPinPolicyV2Repository;
import com.xenomi.cms.core.v2.product.repository.ProductVersionV2Repository;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

@Service
public class ProductIssuancePolicyV2Service {
    private final ProductVersionV2Repository versionRepository;
    private final ProductPersonalisationPolicyV2Repository personalisationRepository;
    private final ProductCardFormPolicyV2Repository cardFormRepository;
    private final ProductPinPolicyV2Repository pinPolicyRepository;

    public ProductIssuancePolicyV2Service(
            ProductVersionV2Repository versionRepository,
            ProductPersonalisationPolicyV2Repository personalisationRepository,
            ProductCardFormPolicyV2Repository cardFormRepository,
            ProductPinPolicyV2Repository pinPolicyRepository) {
        this.versionRepository = versionRepository;
        this.personalisationRepository = personalisationRepository;
        this.cardFormRepository = cardFormRepository;
        this.pinPolicyRepository = pinPolicyRepository;
    }

    @Transactional(readOnly = true)
    public ProductPersonalisationPolicyV2Response getPersonalisation(
            UUID institutionId, UUID productId, UUID versionId) {
        requireVersion(institutionId, productId, versionId);
        return personalisationRepository.findByProductVersionIdAndInstitutionId(versionId, institutionId)
                .map(this::toPersonalisationResponse)
                .orElseThrow(() -> new ResourceNotFoundException(
                        "Product personalisation policy", versionId.toString()));
    }

    @Transactional
    public ProductPersonalisationPolicyV2Response putPersonalisation(
            UUID institutionId, UUID productId, UUID versionId,
            UpsertProductPersonalisationPolicyV2Request request) {
        requireDraftVersion(institutionId, productId, versionId);
        validatePersonalisationRequest(request);
        cardFormRepository.findByProductVersionIdAndInstitutionId(versionId, institutionId)
                .ifPresent(form -> validateFormAndPersonalisation(form, request));
        if (request.pinMailerRequired()) {
            ProductPinPolicyV2 pinPolicy = pinPolicyRepository
                    .findByProductVersionIdAndInstitutionId(versionId, institutionId)
                    .orElseThrow(() -> new IllegalArgumentException(
                            "A PIN policy with pinMailerEnabled=true is required when a PIN mailer is requested"));
            if (!pinPolicy.isPinMailerEnabled()) {
                throw new IllegalArgumentException(
                        "The product PIN policy must enable PIN mailer before personalisation can require it");
            }
        }
        ProductPersonalisationPolicyV2 entity = personalisationRepository
                .findByProductVersionIdAndInstitutionId(versionId, institutionId)
                .orElseGet(() -> ProductPersonalisationPolicyV2.builder()
                        .productVersionId(versionId).institutionId(institutionId).build());
        entity.setPersonalisedAllowed(request.personalisedAllowed());
        entity.setNonPersonalisedAllowed(false);
        entity.setDefaultMode(request.defaultMode());
        entity.setVendorFormatProfileReference(normalize(request.vendorFormatProfileReference()));
        entity.setEmbossingNameRule(normalize(request.embossingNameRule()));
        entity.setMaximumEmbossingNameLength(request.maximumEmbossingNameLength());
        entity.setEmbossingNameLineCount(request.embossingNameLineCount());
        entity.setPrintCardNumber(request.printCardNumber());
        entity.setPrintExpiry(request.printExpiry());
        entity.setPrintCvv2(request.printCvv2());
        entity.setEncodeMagneticStripe(request.encodeMagneticStripe());
        entity.setPersonaliseEmvChip(request.personaliseEmvChip());
        entity.setPinMailerRequired(request.pinMailerRequired());
        entity.setDefaultVendorId(request.defaultVendorId());
        entity.setProductionPriority(request.productionPriority());
        entity.setPolicyStatus(ProductPolicyStatus.DRAFT);
        return toPersonalisationResponse(personalisationRepository.save(entity));
    }

    @Transactional(readOnly = true)
    public ProductCardFormPolicyV2Response getCardForm(
            UUID institutionId, UUID productId, UUID versionId) {
        requireVersion(institutionId, productId, versionId);
        return cardFormRepository.findByProductVersionIdAndInstitutionId(versionId, institutionId)
                .map(this::toCardFormResponse)
                .orElseThrow(() -> new ResourceNotFoundException(
                        "Product card form policy", versionId.toString()));
    }

    @Transactional
    public ProductCardFormPolicyV2Response putCardForm(
            UUID institutionId, UUID productId, UUID versionId,
            UpsertProductCardFormPolicyV2Request request) {
        requireDraftVersion(institutionId, productId, versionId);
        DerivedCardForm derived = derive(request);
        personalisationRepository.findByProductVersionIdAndInstitutionId(versionId, institutionId)
                .ifPresent(policy -> validateFormAndPersonalisation(derived, policy));
        ProductCardFormPolicyV2 entity = cardFormRepository
                .findByProductVersionIdAndInstitutionId(versionId, institutionId)
                .orElseGet(() -> ProductCardFormPolicyV2.builder()
                        .productVersionId(versionId).institutionId(institutionId).build());
        entity.setPhysicalAllowed(derived.physicalAllowed());
        entity.setVirtualAllowed(derived.virtualAllowed());
        entity.setIssuanceMode(derived.issuanceMode());
        entity.setPhysicalIssueTiming(derived.physicalIssueTiming());
        entity.setVirtualPanStrategy(derived.virtualPanStrategy());
        entity.setMaximumActiveVirtualPans(derived.maximumActiveVirtualPans());
        entity.setPhysicalConversionAllowed(derived.physicalConversionAllowed());
        entity.setSamePanForPhysicalConversion(derived.samePanForPhysicalConversion());
        return toCardFormResponse(cardFormRepository.save(entity));
    }

    private ProductVersionV2 requireVersion(UUID institutionId, UUID productId, UUID versionId) {
        return versionRepository.findByProductVersionIdAndInstitutionIdAndProductId(
                        versionId, institutionId, productId)
                .orElseThrow(() -> new ResourceNotFoundException("Product version", versionId.toString()));
    }

    private void requireDraftVersion(UUID institutionId, UUID productId, UUID versionId) {
        if (requireVersion(institutionId, productId, versionId).getVersionStatus() != ProductVersionStatus.DRAFT) {
            throw new IllegalArgumentException(
                    "Product issuance policies can be changed only while the version is DRAFT");
        }
    }

    private static void validatePersonalisationRequest(
            UpsertProductPersonalisationPolicyV2Request request) {
        if (!request.personalisedAllowed()) {
            throw new IllegalArgumentException(
                    "personalisedAllowed must be true because non-personalised issuance is currently disabled");
        }
        if (request.defaultMode() != PersonalisationMode.PERSONALISED) {
            throw new IllegalArgumentException(
                    "Only PERSONALISED is currently supported as defaultMode");
        }
    }

    private static DerivedCardForm derive(UpsertProductCardFormPolicyV2Request request) {
        int requestedMaximum = request.maximumActiveVirtualPans() == null
                ? 1 : request.maximumActiveVirtualPans();
        return switch (request.issuanceMode()) {
            case VIRTUAL_AND_PHYSICAL_SAME_DAY -> new DerivedCardForm(
                    true, true, request.issuanceMode(), PhysicalIssueTiming.SAME_DAY,
                    VirtualPanStrategy.SHARED_WITH_PHYSICAL, 1, false, true);
            case VIRTUAL_THEN_PHYSICAL_LATER -> new DerivedCardForm(
                    true, true, request.issuanceMode(), PhysicalIssueTiming.LATER,
                    VirtualPanStrategy.SHARED_WITH_PHYSICAL, 1, true, true);
            case PHYSICAL_WITH_MULTIPLE_VIRTUAL_PANS -> {
                if (request.maximumActiveVirtualPans() == null) {
                    throw new IllegalArgumentException(
                            "maximumActiveVirtualPans is required for separate virtual PAN issuance");
                }
                yield new DerivedCardForm(
                        true, true, request.issuanceMode(), PhysicalIssueTiming.SAME_DAY,
                        VirtualPanStrategy.SEPARATE_PAN_PER_VIRTUAL,
                        requestedMaximum, false, false);
            }
            case PHYSICAL_ONLY -> new DerivedCardForm(
                    true, false, request.issuanceMode(), PhysicalIssueTiming.SAME_DAY,
                    VirtualPanStrategy.NOT_APPLICABLE, 0, false, true);
            case VIRTUAL_ONLY -> new DerivedCardForm(
                    false, true, request.issuanceMode(), PhysicalIssueTiming.NOT_APPLICABLE,
                    VirtualPanStrategy.SEPARATE_PAN_PER_VIRTUAL,
                    requestedMaximum, false, false);
        };
    }

    private static void validateFormAndPersonalisation(
            ProductCardFormPolicyV2 form, UpsertProductPersonalisationPolicyV2Request policy) {
        validateVirtualOnlyPhysicalFeatures(form.getIssuanceMode(), policy.printCardNumber(),
                policy.printExpiry(), policy.printCvv2(), policy.encodeMagneticStripe(),
                policy.personaliseEmvChip(), policy.pinMailerRequired());
    }

    private static void validateFormAndPersonalisation(
            DerivedCardForm form, ProductPersonalisationPolicyV2 policy) {
        validateVirtualOnlyPhysicalFeatures(form.issuanceMode(), policy.isPrintCardNumber(),
                policy.isPrintExpiry(), policy.isPrintCvv2(), policy.isEncodeMagneticStripe(),
                policy.isPersonaliseEmvChip(), policy.isPinMailerRequired());
    }

    private static void validateVirtualOnlyPhysicalFeatures(
            CardIssuanceMode mode, boolean printCardNumber, boolean printExpiry,
            boolean printCvv2, boolean encodeMagneticStripe,
            boolean personaliseEmvChip, boolean pinMailerRequired) {
        if (mode == CardIssuanceMode.VIRTUAL_ONLY
                && (printCardNumber || printExpiry || printCvv2 || encodeMagneticStripe
                    || personaliseEmvChip || pinMailerRequired)) {
            throw new IllegalArgumentException(
                    "Physical printing, encoding, chip personalisation and PIN mailer must be disabled for VIRTUAL_ONLY");
        }
    }

    private static String normalize(String value) {
        if (value == null || value.isBlank()) return null;
        return value.trim();
    }

    private ProductPersonalisationPolicyV2Response toPersonalisationResponse(
            ProductPersonalisationPolicyV2 entity) {
        return new ProductPersonalisationPolicyV2Response(
                entity.getProductVersionId(), entity.getInstitutionId(),
                entity.isPersonalisedAllowed(), entity.isNonPersonalisedAllowed(),
                entity.getDefaultMode(), entity.getVendorFormatProfileReference(),
                entity.getEmbossingNameRule(), entity.getMaximumEmbossingNameLength(),
                entity.getEmbossingNameLineCount(), entity.isPrintCardNumber(),
                entity.isPrintExpiry(), entity.isPrintCvv2(), entity.isEncodeMagneticStripe(),
                entity.isPersonaliseEmvChip(), entity.isPinMailerRequired(),
                entity.getDefaultVendorId(), entity.getProductionPriority(), entity.getPolicyStatus());
    }

    private ProductCardFormPolicyV2Response toCardFormResponse(ProductCardFormPolicyV2 entity) {
        return new ProductCardFormPolicyV2Response(
                entity.getProductVersionId(), entity.getInstitutionId(),
                entity.isPhysicalAllowed(), entity.isVirtualAllowed(), entity.getIssuanceMode(),
                entity.getPhysicalIssueTiming(), entity.getVirtualPanStrategy(),
                entity.getMaximumActiveVirtualPans(), entity.isPhysicalConversionAllowed(),
                entity.isSamePanForPhysicalConversion());
    }

    private record DerivedCardForm(
            boolean physicalAllowed,
            boolean virtualAllowed,
            CardIssuanceMode issuanceMode,
            PhysicalIssueTiming physicalIssueTiming,
            VirtualPanStrategy virtualPanStrategy,
            int maximumActiveVirtualPans,
            boolean physicalConversionAllowed,
            boolean samePanForPhysicalConversion) {
    }
}
