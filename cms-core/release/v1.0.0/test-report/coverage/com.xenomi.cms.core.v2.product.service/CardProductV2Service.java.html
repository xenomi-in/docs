package com.xenomi.cms.core.v2.product.service;

import java.time.OffsetDateTime;
import java.util.Locale;
import java.util.UUID;

import com.xenomi.cms.core.exception.ResourceAlreadyExistsException;
import com.xenomi.cms.core.exception.ResourceNotFoundException;
import com.xenomi.cms.core.v2.bin.model.CardNetwork;
import com.xenomi.cms.core.v2.common.DatabaseUuidV7Generator;
import com.xenomi.cms.core.v2.event.catalog.ProductEventSink;
import com.xenomi.cms.core.v2.event.catalog.ProductEventSink.ProductSnapshot;
import com.xenomi.cms.core.v2.event.correlation.CorrelationIdResolver;
import com.xenomi.cms.core.v2.institution.repository.InstitutionV2Repository;
import com.xenomi.cms.core.v2.product.dto.CardProductV2Response;
import com.xenomi.cms.core.v2.product.dto.CreateCardProductV2Request;
import com.xenomi.cms.core.v2.product.dto.UpdateCardProductV2Request;
import com.xenomi.cms.core.v2.product.entity.CardProductV2;
import com.xenomi.cms.core.v2.product.model.FundingType;
import com.xenomi.cms.core.v2.product.model.ProductStatus;
import com.xenomi.cms.core.v2.product.repository.CardProductV2Repository;

import org.springframework.data.domain.Page;
import org.springframework.data.domain.Pageable;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

@Service
public class CardProductV2Service {

    private final CardProductV2Repository repository;
    private final InstitutionV2Repository institutionRepository;
    private final DatabaseUuidV7Generator uuidGenerator;
    private final ProductEventSink eventSink;
    private final CorrelationIdResolver correlationIdResolver;

    public CardProductV2Service(CardProductV2Repository repository,
            InstitutionV2Repository institutionRepository, DatabaseUuidV7Generator uuidGenerator,
            ProductEventSink eventSink, CorrelationIdResolver correlationIdResolver) {
        this.repository = repository;
        this.institutionRepository = institutionRepository;
        this.uuidGenerator = uuidGenerator;
        this.eventSink = eventSink;
        this.correlationIdResolver = correlationIdResolver;
    }

    @Transactional
    public CardProductV2Response create(UUID institutionId,
            CreateCardProductV2Request request, String actorId) {
        requireInstitution(institutionId);
        String productCode = upper(request.getProductCode());
        if (repository.existsByInstitutionIdAndProductCode(institutionId, productCode)) {
            throw new ResourceAlreadyExistsException("Card product", productCode);
        }
        CardProductV2 product = CardProductV2.builder()
                .productId(uuidGenerator.generate())
                .institutionId(institutionId)
                .productCode(productCode)
                .productName(request.getProductName().trim())
                .description(trimToNull(request.getDescription()))
                .cardNetwork(request.getCardNetwork())
                .fundingType(request.getFundingType())
                .cardVariant(request.getCardVariant())
                .productStatus(ProductStatus.DRAFT)
                .createdBy(actorId.trim())
                .updatedBy(actorId.trim())
                .build();
        CardProductV2Response response = toResponse(repository.save(product));
        eventSink.onCreated(response, correlationIdResolver.current());
        return response;
    }

    @Transactional(readOnly = true)
    public CardProductV2Response get(UUID institutionId, UUID productId) {
        return toResponse(find(institutionId, productId));
    }

    @Transactional(readOnly = true)
    public CardProductV2Response getByCode(UUID institutionId, String productCode) {
        String normalized = upper(productCode);
        return repository.findByInstitutionIdAndProductCode(institutionId, normalized)
                .map(this::toResponse)
                .orElseThrow(() -> new ResourceNotFoundException("Card product", normalized));
    }

    @Transactional(readOnly = true)
    public Page<CardProductV2Response> search(UUID institutionId, String query,
            CardNetwork network, FundingType fundingType, ProductStatus status, Pageable pageable) {
        requireInstitution(institutionId);
        return repository.search(institutionId, trimToNull(query), network, fundingType, status, pageable)
                .map(this::toResponse);
    }

    @Transactional
    public CardProductV2Response update(UUID institutionId, UUID productId,
            UpdateCardProductV2Request request, String actorId) {
        CardProductV2 product = find(institutionId, productId);
        ProductSnapshot before = new ProductSnapshot(
                product.getProductName(), product.getDescription(), product.getProductStatus());
        if (request.getProductName() != null) product.setProductName(request.getProductName().trim());
        if (request.getDescription() != null) product.setDescription(trimToNull(request.getDescription()));
        if (request.getProductStatus() != null && request.getProductStatus() != product.getProductStatus()) {
            validateTransition(product.getProductStatus(), request.getProductStatus());
            product.setProductStatus(request.getProductStatus());
        }
        product.setUpdatedBy(actorId.trim());
        product.setUpdatedAt(OffsetDateTime.now(java.time.ZoneOffset.UTC));
        CardProductV2Response response = toResponse(repository.save(product));
        eventSink.onUpdated(response, before, correlationIdResolver.current());
        return response;
    }

    CardProductV2 find(UUID institutionId, UUID productId) {
        return repository.findByProductIdAndInstitutionId(productId, institutionId)
                .orElseThrow(() -> new ResourceNotFoundException("Card product", productId.toString()));
    }

    private void requireInstitution(UUID institutionId) {
        if (!institutionRepository.existsById(institutionId)) {
            throw new ResourceNotFoundException("Institution", institutionId.toString());
        }
    }

    private static void validateTransition(ProductStatus current, ProductStatus target) {
        boolean allowed = switch (current) {
            case DRAFT -> target == ProductStatus.PENDING_APPROVAL;
            case PENDING_APPROVAL -> target == ProductStatus.DRAFT || target == ProductStatus.ACTIVE;
            case ACTIVE -> target == ProductStatus.SUSPENDED || target == ProductStatus.RETIRED;
            case SUSPENDED -> target == ProductStatus.ACTIVE || target == ProductStatus.RETIRED;
            case RETIRED -> false;
        };
        if (!allowed) {
            throw new IllegalArgumentException("Invalid product status transition: " + current + " -> " + target);
        }
    }

    private CardProductV2Response toResponse(CardProductV2 p) {
        return CardProductV2Response.builder()
                .productId(p.getProductId()).institutionId(p.getInstitutionId())
                .productCode(p.getProductCode()).productName(p.getProductName())
                .description(p.getDescription()).cardNetwork(p.getCardNetwork())
                .fundingType(p.getFundingType()).productStatus(p.getProductStatus())
                .cardVariant(p.getCardVariant())
                .createdAt(p.getCreatedAt()).updatedAt(p.getUpdatedAt())
                .createdBy(p.getCreatedBy()).updatedBy(p.getUpdatedBy()).build();
    }

    private static String upper(String value) { return value.trim().toUpperCase(Locale.ROOT); }
    private static String trimToNull(String value) {
        if (value == null) return null;
        String normalized = value.trim();
        return normalized.isEmpty() ? null : normalized;
    }
}
