package com.xenomi.cms.core.v2.product.service;

import java.util.UUID;

import com.xenomi.cms.core.exception.ResourceNotFoundException;
import com.xenomi.cms.core.v2.common.DatabaseUuidV7Generator;
import com.xenomi.cms.core.v2.product.dto.CreateProductCardLimitV2Request;
import com.xenomi.cms.core.v2.product.dto.ProductCardLimitV2Response;
import com.xenomi.cms.core.v2.product.dto.ProductDefaultControlsV2Response;
import com.xenomi.cms.core.v2.product.dto.UpdateProductCardLimitV2Request;
import com.xenomi.cms.core.v2.product.dto.UpsertProductDefaultControlsV2Request;
import com.xenomi.cms.core.v2.product.entity.ProductCardLimitV2;
import com.xenomi.cms.core.v2.product.entity.ProductDefaultControlsV2;
import com.xenomi.cms.core.v2.product.entity.ProductVersionV2;
import com.xenomi.cms.core.v2.product.model.ProductLimitStatus;
import com.xenomi.cms.core.v2.product.model.ProductVersionStatus;
import com.xenomi.cms.core.v2.product.model.LimitPeriodType;
import com.xenomi.cms.core.v2.product.repository.ProductCardLimitV2Repository;
import com.xenomi.cms.core.v2.product.repository.ProductDefaultControlsV2Repository;
import com.xenomi.cms.core.v2.product.repository.ProductVersionV2Repository;
import org.springframework.data.domain.Page;
import org.springframework.data.domain.Pageable;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

@Service
public class ProductVersionConfigurationV2Service {
    private final ProductVersionV2Repository versionRepository;
    private final ProductDefaultControlsV2Repository controlsRepository;
    private final ProductCardLimitV2Repository limitRepository;
    private final DatabaseUuidV7Generator uuidGenerator;

    public ProductVersionConfigurationV2Service(
            ProductVersionV2Repository versionRepository,
            ProductDefaultControlsV2Repository controlsRepository,
            ProductCardLimitV2Repository limitRepository,
            DatabaseUuidV7Generator uuidGenerator) {
        this.versionRepository = versionRepository;
        this.controlsRepository = controlsRepository;
        this.limitRepository = limitRepository;
        this.uuidGenerator = uuidGenerator;
    }

    @Transactional(readOnly = true)
    public ProductDefaultControlsV2Response getControls(
            UUID institutionId, UUID productId, UUID versionId) {
        requireVersion(institutionId, productId, versionId);
        return controlsRepository.findByProductVersionIdAndInstitutionId(versionId, institutionId)
                .map(this::toControlsResponse)
                .orElseThrow(() -> new ResourceNotFoundException("Product default controls", versionId.toString()));
    }

    @Transactional
    public ProductDefaultControlsV2Response putControls(UUID institutionId, UUID productId,
            UUID versionId, UpsertProductDefaultControlsV2Request request, String actorId) {
        requireDraftVersion(institutionId, productId, versionId);
        validateControlCapabilities(request);
        ProductDefaultControlsV2 entity = controlsRepository
                .findByProductVersionIdAndInstitutionId(versionId, institutionId)
                .orElseGet(() -> ProductDefaultControlsV2.builder()
                        .productVersionId(versionId).institutionId(institutionId).build());
        entity.setDomesticAtmEnabled(request.domesticAtmEnabled());
        entity.setDomesticAtmAllowed(request.domesticAtmAllowed());
        entity.setDomesticPosEnabled(request.domesticPosEnabled());
        entity.setDomesticPosAllowed(request.domesticPosAllowed());
        entity.setDomesticEcommerceEnabled(request.domesticEcommerceEnabled());
        entity.setDomesticEcommerceAllowed(request.domesticEcommerceAllowed());
        entity.setDomesticMotoEnabled(request.domesticMotoEnabled());
        entity.setDomesticMotoAllowed(request.domesticMotoAllowed());
        entity.setDomesticContactlessEnabled(request.domesticContactlessEnabled());
        entity.setDomesticContactlessAllowed(request.domesticContactlessAllowed());
        entity.setDomesticCashWithdrawalEnabled(request.domesticCashWithdrawalEnabled());
        entity.setInternationalEnabled(request.internationalEnabled());
        entity.setInternationalAtmEnabled(request.internationalAtmEnabled());
        entity.setInternationalAtmAllowed(request.internationalAtmAllowed());
        entity.setInternationalPosEnabled(request.internationalPosEnabled());
        entity.setInternationalPosAllowed(request.internationalPosAllowed());
        entity.setInternationalEcommerceEnabled(request.internationalEcommerceEnabled());
        entity.setInternationalEcommerceAllowed(request.internationalEcommerceAllowed());
        entity.setInternationalMotoEnabled(request.internationalMotoEnabled());
        entity.setInternationalMotoAllowed(request.internationalMotoAllowed());
        entity.setInternationalContactlessEnabled(request.internationalContactlessEnabled());
        entity.setInternationalContactlessAllowed(request.internationalContactlessAllowed());
        entity.setInternationalCashWithdrawalEnabled(request.internationalCashWithdrawalEnabled());
        entity.setUpdatedBy(actorId.trim());
        return toControlsResponse(controlsRepository.save(entity));
    }

    private static void validateControlCapabilities(UpsertProductDefaultControlsV2Request request) {
        requireAllowed(request.domesticAtmEnabled(), request.domesticAtmAllowed(), "DOMESTIC ATM");
        requireAllowed(request.domesticPosEnabled(), request.domesticPosAllowed(), "DOMESTIC POS");
        requireAllowed(request.domesticEcommerceEnabled(), request.domesticEcommerceAllowed(), "DOMESTIC ECOM");
        requireAllowed(request.domesticMotoEnabled(), request.domesticMotoAllowed(), "DOMESTIC MOTO");
        requireAllowed(request.domesticContactlessEnabled(), request.domesticContactlessAllowed(),
                "DOMESTIC CONTACTLESS POS");
        requireAllowed(request.internationalAtmEnabled(), request.internationalAtmAllowed(),
                "INTERNATIONAL ATM");
        requireAllowed(request.internationalPosEnabled(), request.internationalPosAllowed(),
                "INTERNATIONAL POS");
        requireAllowed(request.internationalEcommerceEnabled(), request.internationalEcommerceAllowed(),
                "INTERNATIONAL ECOM");
        requireAllowed(request.internationalMotoEnabled(), request.internationalMotoAllowed(),
                "INTERNATIONAL MOTO");
        requireAllowed(request.internationalContactlessEnabled(),
                request.internationalContactlessAllowed(), "INTERNATIONAL CONTACTLESS POS");
        if (!request.internationalEnabled()
                && (request.internationalAtmEnabled() || request.internationalPosEnabled()
                    || request.internationalEcommerceEnabled() || request.internationalMotoEnabled()
                    || request.internationalContactlessEnabled())) {
            throw new IllegalArgumentException(
                    "International transaction defaults must be disabled when internationalEnabled is false");
        }
    }

    private static void requireAllowed(boolean enabled, boolean allowed, String dimension) {
        if (enabled && !allowed) {
            throw new IllegalArgumentException(
                    dimension + " cannot be enabled when isAllowed is false");
        }
    }

    @Transactional
    public ProductCardLimitV2Response createLimit(UUID institutionId, UUID productId,
            UUID versionId, CreateProductCardLimitV2Request request) {
        requireDraftVersion(institutionId, productId, versionId);
        validateSupportedPeriod(request.periodType());
        validateAmounts(request.defaultAmountMinor(), request.maximumAmountMinor());
        if (limitRepository.existsByInstitutionIdAndProductVersionIdAndLimitTypeAndPeriodType(
                institutionId, versionId, request.limitType(), request.periodType())) {
            throw new IllegalArgumentException("A limit already exists for this limitType and periodType");
        }
        ProductCardLimitV2 entity = ProductCardLimitV2.builder()
                .productCardLimitId(uuidGenerator.generate())
                .institutionId(institutionId)
                .productVersionId(versionId)
                .limitType(request.limitType())
                .periodType(request.periodType())
                .defaultAmountMinor(request.defaultAmountMinor())
                .maximumAmountMinor(request.maximumAmountMinor())
                .limitStatus(request.limitStatus() == null ? ProductLimitStatus.ACTIVE : request.limitStatus())
                .build();
        return toLimitResponse(limitRepository.save(entity));
    }

    @Transactional(readOnly = true)
    public Page<ProductCardLimitV2Response> listLimits(
            UUID institutionId, UUID productId, UUID versionId, Pageable pageable) {
        requireVersion(institutionId, productId, versionId);
        return limitRepository.findByInstitutionIdAndProductVersionId(institutionId, versionId, pageable)
                .map(this::toLimitResponse);
    }

    @Transactional(readOnly = true)
    public ProductCardLimitV2Response getLimit(UUID institutionId, UUID productId,
            UUID versionId, UUID limitId) {
        requireVersion(institutionId, productId, versionId);
        return toLimitResponse(findLimit(institutionId, versionId, limitId));
    }

    @Transactional
    public ProductCardLimitV2Response updateLimit(UUID institutionId, UUID productId,
            UUID versionId, UUID limitId, UpdateProductCardLimitV2Request request) {
        requireDraftVersion(institutionId, productId, versionId);
        ProductCardLimitV2 entity = findLimit(institutionId, versionId, limitId);
        long defaultAmount = request.defaultAmountMinor() == null
                ? entity.getDefaultAmountMinor() : request.defaultAmountMinor();
        Long maximumAmount = request.maximumAmountMinor() == null
                ? entity.getMaximumAmountMinor() : request.maximumAmountMinor();
        validateAmounts(defaultAmount, maximumAmount);
        entity.setDefaultAmountMinor(defaultAmount);
        entity.setMaximumAmountMinor(maximumAmount);
        if (request.limitStatus() != null) entity.setLimitStatus(request.limitStatus());
        return toLimitResponse(limitRepository.save(entity));
    }

    private ProductVersionV2 requireVersion(UUID institutionId, UUID productId, UUID versionId) {
        return versionRepository.findByProductVersionIdAndInstitutionIdAndProductId(
                        versionId, institutionId, productId)
                .orElseThrow(() -> new ResourceNotFoundException("Product version", versionId.toString()));
    }

    private void requireDraftVersion(UUID institutionId, UUID productId, UUID versionId) {
        if (requireVersion(institutionId, productId, versionId).getVersionStatus() != ProductVersionStatus.DRAFT) {
            throw new IllegalArgumentException("Product version policies can be changed only while DRAFT");
        }
    }

    private ProductCardLimitV2 findLimit(UUID institutionId, UUID versionId, UUID limitId) {
        return limitRepository.findByProductCardLimitIdAndInstitutionIdAndProductVersionId(
                        limitId, institutionId, versionId)
                .orElseThrow(() -> new ResourceNotFoundException("Product card limit", limitId.toString()));
    }

    private static void validateAmounts(long defaultAmount, Long maximumAmount) {
        if (defaultAmount < 0) throw new IllegalArgumentException("defaultAmountMinor must not be negative");
        if (maximumAmount != null && maximumAmount < defaultAmount) {
            throw new IllegalArgumentException("maximumAmountMinor must be greater than or equal to defaultAmountMinor");
        }
    }

    private static void validateSupportedPeriod(LimitPeriodType periodType) {
        if (periodType != LimitPeriodType.PER_TRANSACTION) {
            throw new IllegalArgumentException(
                    "Only PER_TRANSACTION is currently supported for product card limits");
        }
    }

    private ProductDefaultControlsV2Response toControlsResponse(ProductDefaultControlsV2 entity) {
        return new ProductDefaultControlsV2Response(
                entity.getProductVersionId(), entity.getInstitutionId(),
                entity.isDomesticAtmEnabled(), entity.isDomesticPosEnabled(),
                entity.isDomesticEcommerceEnabled(), entity.isDomesticContactlessEnabled(),
                entity.isDomesticCashWithdrawalEnabled(), entity.isInternationalAtmEnabled(),
                entity.isInternationalPosEnabled(), entity.isInternationalEcommerceEnabled(),
                entity.isInternationalContactlessEnabled(), entity.isInternationalCashWithdrawalEnabled(),
                entity.getUpdatedAt(), entity.getUpdatedBy(), entity.isInternationalEnabled(),
                entity.isDomesticAtmAllowed(), entity.isDomesticPosAllowed(),
                entity.isDomesticEcommerceAllowed(), entity.isDomesticMotoEnabled(),
                entity.isDomesticMotoAllowed(), entity.isDomesticContactlessAllowed(),
                entity.isInternationalAtmAllowed(), entity.isInternationalPosAllowed(),
                entity.isInternationalEcommerceAllowed(), entity.isInternationalMotoEnabled(),
                entity.isInternationalMotoAllowed(), entity.isInternationalContactlessAllowed());
    }

    private ProductCardLimitV2Response toLimitResponse(ProductCardLimitV2 entity) {
        return new ProductCardLimitV2Response(
                entity.getProductCardLimitId(), entity.getInstitutionId(), entity.getProductVersionId(),
                entity.getLimitType(), entity.getPeriodType(), entity.getDefaultAmountMinor(),
                entity.getMaximumAmountMinor(), entity.getLimitStatus());
    }
}
