package com.xenomi.cms.core.v2.product.service;

import java.time.OffsetDateTime;
import java.util.UUID;

import com.xenomi.cms.core.exception.ResourceNotFoundException;
import com.xenomi.cms.core.v2.bin.entity.BinConfigurationV2;
import com.xenomi.cms.core.v2.bin.entity.BinRangeV2;
import com.xenomi.cms.core.v2.bin.model.BinRangeStatus;
import com.xenomi.cms.core.v2.bin.repository.BinConfigurationV2Repository;
import com.xenomi.cms.core.v2.bin.repository.BinRangeV2Repository;
import com.xenomi.cms.core.v2.common.DatabaseUuidV7Generator;
import com.xenomi.cms.core.v2.product.dto.CreateProductBinRangeAssignmentV2Request;
import com.xenomi.cms.core.v2.product.dto.ProductBinRangeAssignmentV2Response;
import com.xenomi.cms.core.v2.product.dto.UpdateProductBinRangeAssignmentV2Request;
import com.xenomi.cms.core.v2.product.entity.CardProductV2;
import com.xenomi.cms.core.v2.product.entity.ProductBinRangeAssignmentV2;
import com.xenomi.cms.core.v2.product.model.ProductBinAssignmentStatus;
import com.xenomi.cms.core.v2.product.model.ProductStatus;
import com.xenomi.cms.core.v2.product.repository.CardProductV2Repository;
import com.xenomi.cms.core.v2.product.repository.ProductBinRangeAssignmentV2Repository;
import org.springframework.data.domain.Page;
import org.springframework.data.domain.Pageable;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

@Service
public class ProductBinRangeAssignmentV2Service {
    private final ProductBinRangeAssignmentV2Repository repository;
    private final CardProductV2Repository productRepository;
    private final BinRangeV2Repository rangeRepository;
    private final BinConfigurationV2Repository binRepository;
    private final DatabaseUuidV7Generator uuidGenerator;

    public ProductBinRangeAssignmentV2Service(
            ProductBinRangeAssignmentV2Repository repository,
            CardProductV2Repository productRepository,
            BinRangeV2Repository rangeRepository,
            BinConfigurationV2Repository binRepository,
            DatabaseUuidV7Generator uuidGenerator) {
        this.repository = repository;
        this.productRepository = productRepository;
        this.rangeRepository = rangeRepository;
        this.binRepository = binRepository;
        this.uuidGenerator = uuidGenerator;
    }

    @Transactional
    public ProductBinRangeAssignmentV2Response create(UUID institutionId, UUID productId,
            CreateProductBinRangeAssignmentV2Request request, String actorId) {
        CardProductV2 product = requireProduct(institutionId, productId);
        if (product.getProductStatus() == ProductStatus.RETIRED) {
            throw new IllegalArgumentException("BIN ranges cannot be assigned to a RETIRED product");
        }
        BinRangeV2 range = requireRange(institutionId, request.binRangeId());
        validateProductNetwork(product, range);
        validateDates(request.effectiveFrom(), request.effectiveTo());
        if (repository.existsByInstitutionIdAndProductIdAndBinRangeIdAndAssignmentStatusNot(
                institutionId, productId, request.binRangeId(), ProductBinAssignmentStatus.CLOSED)) {
            throw new IllegalArgumentException("This BIN range already has an open assignment to the product");
        }
        ProductBinRangeAssignmentV2 entity = ProductBinRangeAssignmentV2.builder()
                .assignmentId(uuidGenerator.generate())
                .institutionId(institutionId)
                .productId(productId)
                .binRangeId(request.binRangeId())
                .priority(request.priority() == null ? 100 : request.priority())
                .allocationWeight(request.allocationWeight())
                .serviceCodeOverride(request.serviceCodeOverride())
                .assignmentStatus(ProductBinAssignmentStatus.DRAFT)
                .effectiveFrom(request.effectiveFrom())
                .effectiveTo(request.effectiveTo())
                .createdBy(actorId.trim())
                .updatedBy(actorId.trim())
                .build();
        return toResponse(repository.save(entity));
    }

    @Transactional(readOnly = true)
    public Page<ProductBinRangeAssignmentV2Response> list(
            UUID institutionId, UUID productId, Pageable pageable) {
        requireProduct(institutionId, productId);
        return repository.findByInstitutionIdAndProductId(institutionId, productId, pageable)
                .map(this::toResponse);
    }

    @Transactional(readOnly = true)
    public ProductBinRangeAssignmentV2Response get(
            UUID institutionId, UUID productId, UUID assignmentId) {
        return toResponse(find(institutionId, productId, assignmentId));
    }

    @Transactional
    public ProductBinRangeAssignmentV2Response update(UUID institutionId, UUID productId,
            UUID assignmentId, UpdateProductBinRangeAssignmentV2Request request, String actorId) {
        CardProductV2 product = requireProduct(institutionId, productId);
        ProductBinRangeAssignmentV2 entity = find(institutionId, productId, assignmentId);
        boolean configurationChange = request.priority() != null
                || request.allocationWeight() != null
                || request.serviceCodeOverride() != null
                || request.effectiveFrom() != null
                || request.effectiveTo() != null;
        if (configurationChange && entity.getAssignmentStatus() != ProductBinAssignmentStatus.DRAFT) {
            throw new IllegalArgumentException("Assignment configuration can be changed only while DRAFT");
        }
        if (request.priority() != null) entity.setPriority(request.priority());
        if (request.allocationWeight() != null) entity.setAllocationWeight(request.allocationWeight());
        if (request.serviceCodeOverride() != null) entity.setServiceCodeOverride(request.serviceCodeOverride());
        if (request.effectiveFrom() != null) entity.setEffectiveFrom(request.effectiveFrom());
        if (request.effectiveTo() != null) entity.setEffectiveTo(request.effectiveTo());
        if (request.approvalRequestId() != null) entity.setApprovalRequestId(request.approvalRequestId());

        if (request.assignmentStatus() != null
                && request.assignmentStatus() != entity.getAssignmentStatus()) {
            validateTransition(entity.getAssignmentStatus(), request.assignmentStatus());
            if (request.assignmentStatus() == ProductBinAssignmentStatus.ACTIVE) {
                if (product.getProductStatus() != ProductStatus.ACTIVE) {
                    throw new IllegalArgumentException("The product must be ACTIVE before activating a BIN assignment");
                }
                BinRangeV2 range = requireRange(institutionId, entity.getBinRangeId());
                if (range.getRangeStatus() != BinRangeStatus.ACTIVE) {
                    throw new IllegalArgumentException("The BIN range must be ACTIVE before activating its assignment");
                }
            }
            entity.setAssignmentStatus(request.assignmentStatus());
        }
        validateDates(entity.getEffectiveFrom(), entity.getEffectiveTo());
        entity.setUpdatedBy(actorId.trim());
        entity.setUpdatedAt(OffsetDateTime.now(java.time.ZoneOffset.UTC));
        return toResponse(repository.save(entity));
    }

    private CardProductV2 requireProduct(UUID institutionId, UUID productId) {
        return productRepository.findByProductIdAndInstitutionId(productId, institutionId)
                .orElseThrow(() -> new ResourceNotFoundException("Card product", productId.toString()));
    }

    private BinRangeV2 requireRange(UUID institutionId, UUID rangeId) {
        return rangeRepository.findById(rangeId)
                .filter(range -> institutionId.equals(range.getInstitutionId()))
                .orElseThrow(() -> new ResourceNotFoundException("BIN range", rangeId.toString()));
    }

    private void validateProductNetwork(CardProductV2 product, BinRangeV2 range) {
        BinConfigurationV2 bin = binRepository.findByBinIdAndInstitutionId(
                        range.getBinId(), range.getInstitutionId())
                .orElseThrow(() -> new ResourceNotFoundException("BIN", range.getBinId().toString()));
        if (bin.getCardNetwork() != product.getCardNetwork()) {
            throw new IllegalArgumentException("Product and BIN range must use the same card network");
        }
    }

    private ProductBinRangeAssignmentV2 find(UUID institutionId, UUID productId, UUID assignmentId) {
        return repository.findByAssignmentIdAndInstitutionIdAndProductId(
                        assignmentId, institutionId, productId)
                .orElseThrow(() -> new ResourceNotFoundException(
                        "Product BIN range assignment", assignmentId.toString()));
    }

    private static void validateDates(OffsetDateTime from, OffsetDateTime to) {
        if (to != null && !to.isAfter(from)) {
            throw new IllegalArgumentException("effectiveTo must be after effectiveFrom");
        }
    }

    private static void validateTransition(ProductBinAssignmentStatus current,
            ProductBinAssignmentStatus target) {
        boolean allowed = switch (current) {
            case DRAFT -> target == ProductBinAssignmentStatus.PENDING_APPROVAL
                    || target == ProductBinAssignmentStatus.CLOSED;
            case PENDING_APPROVAL -> target == ProductBinAssignmentStatus.DRAFT
                    || target == ProductBinAssignmentStatus.ACTIVE
                    || target == ProductBinAssignmentStatus.CLOSED;
            case ACTIVE -> target == ProductBinAssignmentStatus.SUSPENDED
                    || target == ProductBinAssignmentStatus.CLOSED;
            case SUSPENDED -> target == ProductBinAssignmentStatus.ACTIVE
                    || target == ProductBinAssignmentStatus.CLOSED;
            case CLOSED -> false;
        };
        if (!allowed) {
            throw new IllegalArgumentException("Invalid BIN assignment status transition: " + current + " -> " + target);
        }
    }

    private ProductBinRangeAssignmentV2Response toResponse(ProductBinRangeAssignmentV2 entity) {
        return new ProductBinRangeAssignmentV2Response(
                entity.getAssignmentId(), entity.getInstitutionId(), entity.getProductId(),
                entity.getBinRangeId(), entity.getPriority(), entity.getAllocationWeight(),
                entity.getServiceCodeOverride(), entity.getAssignmentStatus(),
                entity.getEffectiveFrom(), entity.getEffectiveTo(), entity.getApprovalRequestId(),
                entity.getCreatedAt(), entity.getUpdatedAt(), entity.getCreatedBy(), entity.getUpdatedBy());
    }
}
