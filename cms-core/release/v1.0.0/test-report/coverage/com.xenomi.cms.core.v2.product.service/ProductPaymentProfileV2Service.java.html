@Service
public class ProductPaymentProfileV2Service {
    private final ProductVersionV2Repository versionRepository;
    private final ProductCardTechnologyProfileV2Repository technologyRepository;
    private final ProductCvmProfileV2Repository cvmRepository;
    private final CvmRuleV2Repository ruleRepository;
    private final ProductCardFormPolicyV2Repository cardFormRepository;
    private final ProductPersonalisationPolicyV2Repository personalisationRepository;
    private final ProductPinPolicyV2Repository pinPolicyRepository;
    private final DatabaseUuidV7Generator uuidGenerator;

    public ProductPaymentProfileV2Service(ProductVersionV2Repository versionRepository,
            ProductCardTechnologyProfileV2Repository technologyRepository,
            ProductCvmProfileV2Repository cvmRepository, CvmRuleV2Repository ruleRepository,
            ProductCardFormPolicyV2Repository cardFormRepository,
            ProductPersonalisationPolicyV2Repository personalisationRepository,
            ProductPinPolicyV2Repository pinPolicyRepository, DatabaseUuidV7Generator uuidGenerator) {
        this.versionRepository = versionRepository;
        this.technologyRepository = technologyRepository;
        this.cvmRepository = cvmRepository;
        this.ruleRepository = ruleRepository;
        this.cardFormRepository = cardFormRepository;
        this.personalisationRepository = personalisationRepository;
        this.pinPolicyRepository = pinPolicyRepository;
        this.uuidGenerator = uuidGenerator;
    }

    @Transactional
    public ProductTechnologyProfileV2Response createTechnology(UUID institutionId, UUID productId,
            UUID versionId, CreateProductTechnologyProfileV2Request request) {
        requireDraftVersion(institutionId, productId, versionId);
        validateTechnology(institutionId, versionId, request.magneticStripeEnabled(),
                request.emvContactEnabled(), request.emvContactlessEnabled());
        validateDates(request.effectiveFrom(), request.effectiveTo());
        ProductCardTechnologyProfileV2 entity = ProductCardTechnologyProfileV2.builder()
                .technologyProfileId(uuidGenerator.generate()).institutionId(institutionId)
                .productVersionId(versionId).profileVersion(technologyRepository.findMaximumVersion(versionId) + 1)
                .magneticStripeEnabled(request.magneticStripeEnabled())
                .emvContactEnabled(request.emvContactEnabled())
                .emvContactlessEnabled(request.emvContactlessEnabled())
                .chipProfileReference(normalize(request.chipProfileReference()))
                .contactlessProfileReference(normalize(request.contactlessProfileReference()))
                .applicationProfileReference(normalize(request.applicationProfileReference()))
                .profileStatus(ProductProfileStatus.DRAFT)
                .effectiveFrom(request.effectiveFrom()).effectiveTo(request.effectiveTo()).build();
        return toTechnologyResponse(technologyRepository.save(entity));
    }

    @Transactional(readOnly = true)
    public Page<ProductTechnologyProfileV2Response> listTechnology(UUID institutionId, UUID productId,
            UUID versionId, Pageable pageable) {
        requireVersion(institutionId, productId, versionId);
        return technologyRepository.findByInstitutionIdAndProductVersionId(institutionId, versionId, pageable)
                .map(this::toTechnologyResponse);
    }

    @Transactional(readOnly = true)
    public ProductTechnologyProfileV2Response getTechnology(UUID institutionId, UUID productId,
            UUID versionId, UUID profileId) {
        requireVersion(institutionId, productId, versionId);
        return toTechnologyResponse(findTechnology(institutionId, versionId, profileId));
    }

    @Transactional
    public ProductTechnologyProfileV2Response updateTechnology(UUID institutionId, UUID productId,
            UUID versionId, UUID profileId, UpdateProductTechnologyProfileV2Request request) {
        requireDraftVersion(institutionId, productId, versionId);
        ProductCardTechnologyProfileV2 entity = findTechnology(institutionId, versionId, profileId);
        boolean config = request.magneticStripeEnabled() != null || request.emvContactEnabled() != null
                || request.emvContactlessEnabled() != null || request.chipProfileReference() != null
                || request.contactlessProfileReference() != null || request.applicationProfileReference() != null
                || request.effectiveFrom() != null || request.effectiveTo() != null;
        if (config && entity.getProfileStatus() != ProductProfileStatus.DRAFT) {
            throw new IllegalArgumentException("Technology profile configuration can be changed only while DRAFT");
        }
        if (request.magneticStripeEnabled() != null) entity.setMagneticStripeEnabled(request.magneticStripeEnabled());
        if (request.emvContactEnabled() != null) entity.setEmvContactEnabled(request.emvContactEnabled());
        if (request.emvContactlessEnabled() != null) entity.setEmvContactlessEnabled(request.emvContactlessEnabled());
        if (request.chipProfileReference() != null) entity.setChipProfileReference(normalize(request.chipProfileReference()));
        if (request.contactlessProfileReference() != null) entity.setContactlessProfileReference(normalize(request.contactlessProfileReference()));
        if (request.applicationProfileReference() != null) entity.setApplicationProfileReference(normalize(request.applicationProfileReference()));
        if (request.effectiveFrom() != null) entity.setEffectiveFrom(request.effectiveFrom());
        if (request.effectiveTo() != null) entity.setEffectiveTo(request.effectiveTo());
        if (request.profileStatus() != null && request.profileStatus() != entity.getProfileStatus()) {
            validateProfileTransition(entity.getProfileStatus(), request.profileStatus());
            if (request.profileStatus() == ProductProfileStatus.ACTIVE
                    && technologyRepository.existsByInstitutionIdAndProductVersionIdAndProfileStatus(
                            institutionId, versionId, ProductProfileStatus.ACTIVE)) {
                throw new IllegalArgumentException("The product version already has an ACTIVE technology profile");
            }
            entity.setProfileStatus(request.profileStatus());
        }
        validateTechnology(institutionId, versionId, entity.isMagneticStripeEnabled(),
                entity.isEmvContactEnabled(), entity.isEmvContactlessEnabled());
        validateDates(entity.getEffectiveFrom(), entity.getEffectiveTo());
        return toTechnologyResponse(technologyRepository.save(entity));
    }

    @Transactional
    public ProductCvmProfileV2Response createCvm(UUID institutionId, UUID productId, UUID versionId,
            CreateProductCvmProfileV2Request request) {
        requireDraftVersion(institutionId, productId, versionId);
        validateDates(request.effectiveFrom(), request.effectiveTo());
        validateCvmHeader(request.contactlessCvmLimitMinor(), request.currencyCode());
        validateRules(institutionId, versionId, request.rules());
        UUID profileId = uuidGenerator.generate();
        ProductCvmProfileV2 profile = ProductCvmProfileV2.builder()
                .cvmProfileId(profileId).institutionId(institutionId).productVersionId(versionId)
                .profileVersion(cvmRepository.findMaximumVersion(versionId) + 1)
                .contactlessCvmLimitMinor(request.contactlessCvmLimitMinor())
                .currencyCode(normalizeCurrency(request.currencyCode()))
                .profileStatus(ProductProfileStatus.DRAFT)
                .effectiveFrom(request.effectiveFrom()).effectiveTo(request.effectiveTo()).build();
        cvmRepository.save(profile);
        saveRules(institutionId, profileId, request.rules());
        return toCvmResponse(profile);
    }

    @Transactional(readOnly = true)

    @Transactional
    public ProductCvmProfileV2Response updateCvm(UUID institutionId, UUID productId, UUID versionId,
            UUID profileId, UpdateProductCvmProfileV2Request request) {
        requireDraftVersion(institutionId, productId, versionId);
        ProductCvmProfileV2 entity = findCvm(institutionId, versionId, profileId);
        boolean config = request.contactlessCvmLimitMinor() != null || request.currencyCode() != null
                || request.effectiveFrom() != null || request.effectiveTo() != null || request.rules() != null;
        if (config && entity.getProfileStatus() != ProductProfileStatus.DRAFT) {
            throw new IllegalArgumentException("CVM profile configuration can be changed only while DRAFT");
        }
        if (request.contactlessCvmLimitMinor() != null) entity.setContactlessCvmLimitMinor(request.contactlessCvmLimitMinor());
        if (request.currencyCode() != null) entity.setCurrencyCode(normalizeCurrency(request.currencyCode()));
        if (request.effectiveFrom() != null) entity.setEffectiveFrom(request.effectiveFrom());
        if (request.effectiveTo() != null) entity.setEffectiveTo(request.effectiveTo());
        if (request.profileStatus() != null && request.profileStatus() != entity.getProfileStatus()) {
            validateProfileTransition(entity.getProfileStatus(), request.profileStatus());
            if (request.profileStatus() == ProductProfileStatus.ACTIVE
                    && cvmRepository.existsByInstitutionIdAndProductVersionIdAndProfileStatus(
                            institutionId, versionId, ProductProfileStatus.ACTIVE)) {
                throw new IllegalArgumentException("The product version already has an ACTIVE CVM profile");
            }
            entity.setProfileStatus(request.profileStatus());
        }
        validateCvmHeader(entity.getContactlessCvmLimitMinor(), entity.getCurrencyCode());
        validateDates(entity.getEffectiveFrom(), entity.getEffectiveTo());
        if (request.rules() != null) {
            if (request.rules().isEmpty()) throw new IllegalArgumentException("CVM profile must contain at least one rule");
            validateRules(institutionId, versionId, request.rules());
            ruleRepository.deleteByInstitutionIdAndCvmProfileId(institutionId, profileId);
            saveRules(institutionId, profileId, request.rules());
        }
        return toCvmResponse(cvmRepository.save(entity));
    }

    private void validateTechnology(UUID institutionId, UUID versionId, boolean magnetic,
            boolean contact, boolean contactless) {
        if (!magnetic && !contact && !contactless) {
            throw new IllegalArgumentException("At least one physical card technology must be enabled");
        }
        ProductCardFormPolicyV2 form = cardFormRepository
                .findByProductVersionIdAndInstitutionId(versionId, institutionId)
                .orElseThrow(() -> new IllegalArgumentException("Configure card-form policy before technology profile"));
        if (form.getIssuanceMode() == CardIssuanceMode.VIRTUAL_ONLY) {
            throw new IllegalArgumentException("VIRTUAL_ONLY products cannot have a physical technology profile");
        }
        ProductPersonalisationPolicyV2 p = personalisationRepository
                .findByProductVersionIdAndInstitutionId(versionId, institutionId)
                .orElseThrow(() -> new IllegalArgumentException("Configure personalisation policy before technology profile"));
        if (magnetic != p.isEncodeMagneticStripe()) {
            throw new IllegalArgumentException("Magnetic-stripe technology must match personalisation encoding");
        }
        boolean chipEnabled = contact || contactless;
        if (chipEnabled != p.isPersonaliseEmvChip()) {
            throw new IllegalArgumentException("EMV technology must match chip personalisation");
        }
    }

    private void validateRules(UUID institutionId, UUID versionId, List<CvmRuleV2Request> rules) {
        Set<Integer> sequences = new HashSet<>();
        ProductCardTechnologyProfileV2 tech = technologyRepository
                .findTopByInstitutionIdAndProductVersionIdOrderByProfileVersionDesc(institutionId, versionId)
                .orElse(null);
        ProductPinPolicyV2 pin = pinPolicyRepository
                .findByProductVersionIdAndInstitutionId(versionId, institutionId).orElse(null);
        for (CvmRuleV2Request rule : rules) {
            if (!sequences.add(rule.ruleSequence())) {
                throw new IllegalArgumentException("CVM ruleSequence values must be unique");
            }
            if (rule.minimumAmountMinor() != null && rule.maximumAmountMinor() != null
                    && rule.maximumAmountMinor() < rule.minimumAmountMinor()) {
                throw new IllegalArgumentException("CVM maximumAmountMinor must be greater than or equal to minimumAmountMinor");
            }
            if (rule.cardInterface() != CardInterface.ECOMMERCE && tech == null) {
                throw new IllegalArgumentException("A technology profile is required for physical CVM interfaces");
            }
            if (tech != null) validateInterface(rule.cardInterface(), tech);
            if (rule.cvmMethod() == CvmMethod.ONLINE_PIN && (pin == null || !pin.isOnlinePinSupported())) {
                throw new IllegalArgumentException("ONLINE_PIN CVM requires onlinePinSupported in PIN policy");
            }
            if (rule.cvmMethod() == CvmMethod.OFFLINE_PIN && (pin == null || !pin.isOfflinePinSupported())) {
                throw new IllegalArgumentException("OFFLINE_PIN CVM requires offlinePinSupported in PIN policy");
            }
            if (rule.cvmMethod() == CvmMethod.CDCVM && rule.cardInterface() != CardInterface.CONTACTLESS) {
                throw new IllegalArgumentException("CDCVM is supported only for CONTACTLESS rules");
            }
        }
    }

    private static void validateInterface(CardInterface cardInterface, ProductCardTechnologyProfileV2 tech) {
        if (cardInterface == CardInterface.CONTACT && !tech.isEmvContactEnabled())
            throw new IllegalArgumentException("CONTACT CVM requires EMV contact technology");
        if (cardInterface == CardInterface.CONTACTLESS && !tech.isEmvContactlessEnabled())
            throw new IllegalArgumentException("CONTACTLESS CVM requires EMV contactless technology");
        if (cardInterface == CardInterface.MAGSTRIPE && !tech.isMagneticStripeEnabled())
            throw new IllegalArgumentException("MAGSTRIPE CVM requires magnetic-stripe technology");
    }

    private void saveRules(UUID institutionId, UUID profileId, List<CvmRuleV2Request> requests) {
        List<CvmRuleV2> rules = requests.stream().map(r -> CvmRuleV2.builder()
                .cvmRuleId(uuidGenerator.generate()).institutionId(institutionId).cvmProfileId(profileId)
                .ruleSequence(r.ruleSequence()).cardInterface(r.cardInterface()).cvmMethod(r.cvmMethod())
                .cvmCondition(r.cvmCondition()).continueOnFailure(r.continueOnFailure())
                .transactionType(normalize(r.transactionType())).minimumAmountMinor(r.minimumAmountMinor())
                .maximumAmountMinor(r.maximumAmountMinor()).currencyCode(normalizeCurrency(r.currencyCode()))
                .build()).toList();
        ruleRepository.saveAll(rules);
    }

    private ProductVersionV2 requireVersion(UUID institutionId, UUID productId, UUID versionId) {
        return versionRepository.findByProductVersionIdAndInstitutionIdAndProductId(versionId, institutionId, productId)
                .orElseThrow(() -> new ResourceNotFoundException("Product version", versionId.toString()));
    }
    private void requireDraftVersion(UUID institutionId, UUID productId, UUID versionId) {
        if (requireVersion(institutionId, productId, versionId).getVersionStatus() != ProductVersionStatus.DRAFT)
            throw new IllegalArgumentException("Payment profiles can be changed only while the version is DRAFT");
    }
    private ProductCardTechnologyProfileV2 findTechnology(UUID institutionId, UUID versionId, UUID profileId) {
        return technologyRepository.findByTechnologyProfileIdAndInstitutionIdAndProductVersionId(profileId, institutionId, versionId)
                .orElseThrow(() -> new ResourceNotFoundException("Technology profile", profileId.toString()));
    }
    private ProductCvmProfileV2 findCvm(UUID institutionId, UUID versionId, UUID profileId) {
        return cvmRepository.findByCvmProfileIdAndInstitutionIdAndProductVersionId(profileId, institutionId, versionId)
                .orElseThrow(() -> new ResourceNotFoundException("CVM profile", profileId.toString()));
    }
    private static void validateDates(OffsetDateTime from, OffsetDateTime to) {
        if (to != null && !to.isAfter(from)) throw new IllegalArgumentException("effectiveTo must be after effectiveFrom");
    }
    private static void validateCvmHeader(Long limit, String currency) {
        if ((limit == null) != (currency == null || currency.isBlank())) {
            throw new IllegalArgumentException("contactlessCvmLimitMinor and currencyCode must be provided together");
        }
    }
    private static void validateProfileTransition(ProductProfileStatus current, ProductProfileStatus target) {
        boolean allowed = current == ProductProfileStatus.DRAFT
                ? target == ProductProfileStatus.ACTIVE || target == ProductProfileStatus.RETIRED
                : current == ProductProfileStatus.ACTIVE && target == ProductProfileStatus.RETIRED;
        if (!allowed) throw new IllegalArgumentException("Invalid profile status transition: " + current + " -> " + target);
    }
    private static String normalize(String value) { return value == null || value.isBlank() ? null : value.trim(); }
    private static String normalizeCurrency(String value) { return value == null || value.isBlank() ? null : value.trim().toUpperCase(); }
    private ProductTechnologyProfileV2Response toTechnologyResponse(ProductCardTechnologyProfileV2 p) {
        return new ProductTechnologyProfileV2Response(p.getTechnologyProfileId(), p.getInstitutionId(),
                p.getProductVersionId(), p.getProfileVersion(), p.isMagneticStripeEnabled(),
                p.isEmvContactEnabled(), p.isEmvContactlessEnabled(), p.getChipProfileReference(),
                p.getContactlessProfileReference(), p.getApplicationProfileReference(), p.getProfileStatus(),
                p.getEffectiveFrom(), p.getEffectiveTo());
    }
    private ProductCvmProfileV2Response toCvmResponse(ProductCvmProfileV2 p) {
        List<CvmRuleV2Response> rules = ruleRepository
                .findByInstitutionIdAndCvmProfileIdOrderByRuleSequence(p.getInstitutionId(), p.getCvmProfileId())
                .stream().map(r -> new CvmRuleV2Response(r.getCvmRuleId(), r.getRuleSequence(),
                        r.getCardInterface(), r.getCvmMethod(), r.getCvmCondition(), r.isContinueOnFailure(),
                        r.getTransactionType(), r.getMinimumAmountMinor(), r.getMaximumAmountMinor(),
                        r.getCurrencyCode())).toList();
        return new ProductCvmProfileV2Response(p.getCvmProfileId(), p.getInstitutionId(),
                p.getProductVersionId(), p.getProfileVersion(), p.getContactlessCvmLimitMinor(),
                p.getCurrencyCode(), p.getProfileStatus(), p.getEffectiveFrom(), p.getEffectiveTo(), rules);
    }
