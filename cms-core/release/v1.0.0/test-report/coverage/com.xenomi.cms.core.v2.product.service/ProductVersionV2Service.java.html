package com.xenomi.cms.core.v2.product.service;

import java.time.OffsetDateTime;
import java.util.UUID;

import com.xenomi.cms.core.exception.ResourceNotFoundException;
import com.xenomi.cms.core.v2.common.DatabaseUuidV7Generator;
import com.xenomi.cms.core.v2.event.catalog.ProductVersionEventSink;
import com.xenomi.cms.core.v2.event.catalog.ProductVersionEventSink.VersionSnapshot;
import com.xenomi.cms.core.v2.event.correlation.CorrelationIdResolver;
import com.xenomi.cms.core.v2.product.dto.CreateProductVersionV2Request;
import com.xenomi.cms.core.v2.product.dto.ProductVersionV2Response;
import com.xenomi.cms.core.v2.product.dto.UpdateProductVersionV2Request;
import com.xenomi.cms.core.v2.product.entity.CardProductV2;
import com.xenomi.cms.core.v2.product.entity.ProductVersionV2;
import com.xenomi.cms.core.v2.product.model.ExpiryCalculationRule;
import com.xenomi.cms.core.v2.product.model.ProductStatus;
import com.xenomi.cms.core.v2.product.model.ProductVersionStatus;
import com.xenomi.cms.core.v2.product.model.ValidityStartRule;
import com.xenomi.cms.core.v2.product.repository.CardProductV2Repository;
import com.xenomi.cms.core.v2.product.repository.ProductVersionV2Repository;
import com.xenomi.cms.core.v2.institution.repository.InstitutionV2Repository;

import org.springframework.data.domain.Page;
import org.springframework.data.domain.Pageable;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

@Service
public class ProductVersionV2Service {

    private final ProductVersionV2Repository repository;
    private final CardProductV2Repository productRepository;
    private final DatabaseUuidV7Generator uuidGenerator;
    private final InstitutionV2Repository institutionRepository;
    private final ProductVersionEventSink eventSink;
    private final CorrelationIdResolver correlationIdResolver;

    public ProductVersionV2Service(ProductVersionV2Repository repository,
            CardProductV2Repository productRepository, DatabaseUuidV7Generator uuidGenerator,
            InstitutionV2Repository institutionRepository, ProductVersionEventSink eventSink,
            CorrelationIdResolver correlationIdResolver) {
        this.repository = repository;
        this.productRepository = productRepository;
        this.uuidGenerator = uuidGenerator;
        this.institutionRepository = institutionRepository;
        this.eventSink = eventSink;
        this.correlationIdResolver = correlationIdResolver;
    }

    @Transactional
    public ProductVersionV2Response create(UUID institutionId, UUID productId,
            CreateProductVersionV2Request request, String actorId) {
        CardProductV2 product = lockProduct(institutionId, productId);
        if (product.getProductStatus() == ProductStatus.RETIRED) {
            throw new IllegalArgumentException("A version cannot be created for a RETIRED product");
        }
        validateDates(request.getEffectiveFrom(), request.getEffectiveTo());
        ProductVersionV2 version = ProductVersionV2.builder()
                .productVersionId(uuidGenerator.generate())
                .institutionId(institutionId)
                .productId(productId)
                .versionNumber(repository.findMaximumVersion(productId) + 1)
                .versionStatus(ProductVersionStatus.DRAFT)
                .validityMonths(resolveValidityMonths(institutionId, request.getValidityMonths()))
                .validityStartRule(request.getValidityStartRule() == null
                        ? ValidityStartRule.ISSUANCE_DATE : request.getValidityStartRule())
                .expiryCalculationRule(request.getExpiryCalculationRule() == null
                        ? ExpiryCalculationRule.END_OF_EXPIRY_MONTH : request.getExpiryCalculationRule())
                .renewalNoticeDays(request.getRenewalNoticeDays())
                .renewalProductionDays(request.getRenewalProductionDays())
                .automaticRenewalEnabled(request.getAutomaticRenewalEnabled() == null
                        || request.getAutomaticRenewalEnabled())
                .defaultServiceCode(request.getDefaultServiceCode())
                .icvvServiceCode(request.getIcvvServiceCode())
                .effectiveFrom(request.getEffectiveFrom())
                .effectiveTo(request.getEffectiveTo())
                .createdBy(actorId.trim())
                .updatedBy(actorId.trim())
                .build();
        ProductVersionV2Response response = toResponse(repository.save(version));
        eventSink.onCreated(response, correlationIdResolver.current());
        return response;
    }

    @Transactional(readOnly = true)
    public ProductVersionV2Response get(UUID institutionId, UUID productId, UUID versionId) {
        return toResponse(find(institutionId, productId, versionId));
    }

    @Transactional(readOnly = true)
    public Page<ProductVersionV2Response> list(
            UUID institutionId, UUID productId, Pageable pageable) {
        requireProduct(institutionId, productId);
        return repository.findByInstitutionIdAndProductId(institutionId, productId, pageable)
                .map(this::toResponse);
    }

    @Transactional
    public ProductVersionV2Response update(UUID institutionId, UUID productId, UUID versionId,
            UpdateProductVersionV2Request request, String actorId) {
        CardProductV2 product = lockProduct(institutionId, productId);
        ProductVersionV2 version = find(institutionId, productId, versionId);
        VersionSnapshot before = snapshot(version);
        boolean configurationUpdate = request.getValidityMonths() != null
                || request.getValidityStartRule() != null
                || request.getExpiryCalculationRule() != null
                || request.getRenewalNoticeDays() != null
                || request.getRenewalProductionDays() != null
                || request.getAutomaticRenewalEnabled() != null
                || request.getDefaultServiceCode() != null
                || request.getIcvvServiceCode() != null
                || request.getEffectiveFrom() != null
                || request.getEffectiveTo() != null;
        if (configurationUpdate && version.getVersionStatus() != ProductVersionStatus.DRAFT) {
            throw new IllegalArgumentException("Product version configuration can be changed only while DRAFT");
        }
        if (request.getValidityMonths() != null) version.setValidityMonths(request.getValidityMonths());
        if (request.getValidityStartRule() != null) version.setValidityStartRule(request.getValidityStartRule());
        if (request.getExpiryCalculationRule() != null) version.setExpiryCalculationRule(request.getExpiryCalculationRule());
        if (request.getRenewalNoticeDays() != null) version.setRenewalNoticeDays(request.getRenewalNoticeDays());
        if (request.getRenewalProductionDays() != null) version.setRenewalProductionDays(request.getRenewalProductionDays());
        if (request.getAutomaticRenewalEnabled() != null) version.setAutomaticRenewalEnabled(request.getAutomaticRenewalEnabled());
        if (request.getDefaultServiceCode() != null) version.setDefaultServiceCode(request.getDefaultServiceCode());
        if (request.getIcvvServiceCode() != null) version.setIcvvServiceCode(request.getIcvvServiceCode());
        if (request.getEffectiveFrom() != null) version.setEffectiveFrom(request.getEffectiveFrom());
        if (request.getEffectiveTo() != null) version.setEffectiveTo(request.getEffectiveTo());

        if (request.getVersionStatus() != null && request.getVersionStatus() != version.getVersionStatus()) {
            validateTransition(version.getVersionStatus(), request.getVersionStatus());
            if (request.getVersionStatus() == ProductVersionStatus.ACTIVE) {
                if (product.getProductStatus() != ProductStatus.ACTIVE) {
                    throw new IllegalArgumentException("The parent product must be ACTIVE before activating a version");
                }
                if (repository.existsByInstitutionIdAndProductIdAndVersionStatus(
                        institutionId, productId, ProductVersionStatus.ACTIVE)) {
                    throw new IllegalArgumentException("The product already has an ACTIVE version");
                }
                version.setActivatedAt(OffsetDateTime.now(java.time.ZoneOffset.UTC));
            }
            version.setVersionStatus(request.getVersionStatus());
        }
        validateDates(version.getEffectiveFrom(), version.getEffectiveTo());
        version.setUpdatedBy(actorId.trim());
        version.setUpdatedAt(OffsetDateTime.now(java.time.ZoneOffset.UTC));
        ProductVersionV2Response response = toResponse(repository.save(version));
        eventSink.onUpdated(response, before, correlationIdResolver.current());
        return response;
    }

    private static VersionSnapshot snapshot(ProductVersionV2 version) {
        return new VersionSnapshot(
                version.getVersionStatus(),
                version.getValidityMonths(),
                version.getValidityStartRule(),
                version.getExpiryCalculationRule(),
                version.getRenewalNoticeDays(),
                version.getRenewalProductionDays(),
                version.isAutomaticRenewalEnabled(),
                version.getDefaultServiceCode(),
                version.getIcvvServiceCode(),
                version.getEffectiveFrom(),
                version.getEffectiveTo());
    }

    private CardProductV2 lockProduct(UUID institutionId, UUID productId) {
        return productRepository.findByTenantAndIdForUpdate(institutionId, productId)
                .orElseThrow(() -> new ResourceNotFoundException("Card product", productId.toString()));
    }

    private int resolveValidityMonths(UUID institutionId, Integer productValidityMonths) {
        if (productValidityMonths != null) {
            return productValidityMonths;
        }
        return institutionRepository.findById(institutionId)
                .orElseThrow(() -> new ResourceNotFoundException(
                        "Institution", institutionId.toString()))
                .getDefaultCardValidityMonths();
    }

    private void requireProduct(UUID institutionId, UUID productId) {
        if (productRepository.findByProductIdAndInstitutionId(productId, institutionId).isEmpty()) {
            throw new ResourceNotFoundException("Card product", productId.toString());
        }
    }

    private ProductVersionV2 find(UUID institutionId, UUID productId, UUID versionId) {
        return repository.findByProductVersionIdAndInstitutionIdAndProductId(
                        versionId, institutionId, productId)
                .orElseThrow(() -> new ResourceNotFoundException("Product version", versionId.toString()));
    }

    private static void validateDates(OffsetDateTime from, OffsetDateTime to) {
        if (to != null && !to.isAfter(from)) {
            throw new IllegalArgumentException("effectiveTo must be after effectiveFrom");
        }
    }

    private static void validateTransition(ProductVersionStatus current, ProductVersionStatus target) {
        boolean allowed = switch (current) {
            case DRAFT -> target == ProductVersionStatus.PENDING_APPROVAL
                    || target == ProductVersionStatus.CANCELLED;
            case PENDING_APPROVAL -> target == ProductVersionStatus.DRAFT
                    || target == ProductVersionStatus.ACTIVE
                    || target == ProductVersionStatus.CANCELLED;
            case ACTIVE -> target == ProductVersionStatus.RETIRED;
            case RETIRED, CANCELLED -> false;
        };
        if (!allowed) {
            throw new IllegalArgumentException(
                    "Invalid product version status transition: " + current + " -> " + target);
        }
    }

    private ProductVersionV2Response toResponse(ProductVersionV2 v) {
        return ProductVersionV2Response.builder()
                .productVersionId(v.getProductVersionId()).institutionId(v.getInstitutionId())
                .productId(v.getProductId()).versionNumber(v.getVersionNumber())
                .versionStatus(v.getVersionStatus()).validityMonths(v.getValidityMonths())
                .validityStartRule(v.getValidityStartRule())
                .expiryCalculationRule(v.getExpiryCalculationRule())
                .renewalNoticeDays(v.getRenewalNoticeDays())
                .renewalProductionDays(v.getRenewalProductionDays())
                .automaticRenewalEnabled(v.isAutomaticRenewalEnabled())
                .defaultServiceCode(v.getDefaultServiceCode()).icvvServiceCode(v.getIcvvServiceCode())
                .effectiveFrom(v.getEffectiveFrom()).effectiveTo(v.getEffectiveTo())
                .activatedAt(v.getActivatedAt()).createdAt(v.getCreatedAt()).updatedAt(v.getUpdatedAt())
                .createdBy(v.getCreatedBy()).updatedBy(v.getUpdatedBy()).build();
    }
}
