package com.xenomi.cms.core.v2.product.service;

import java.time.OffsetDateTime;
import java.util.UUID;

import com.xenomi.cms.core.exception.ResourceNotFoundException;
import com.xenomi.cms.core.v2.common.DatabaseUuidV7Generator;
import com.xenomi.cms.core.v2.product.dto.CreateProductCardDesignAssignmentV2Request;
import com.xenomi.cms.core.v2.product.dto.ProductCardDesignAssignmentV2Response;
import com.xenomi.cms.core.v2.product.dto.UpdateProductCardDesignAssignmentV2Request;
import com.xenomi.cms.core.v2.product.entity.CardDesignV2;
import com.xenomi.cms.core.v2.product.entity.CardProductV2;
import com.xenomi.cms.core.v2.product.entity.ProductCardDesignAssignmentV2;
import com.xenomi.cms.core.v2.product.entity.ProductCardFormPolicyV2;
import com.xenomi.cms.core.v2.product.entity.ProductVersionV2;
import com.xenomi.cms.core.v2.product.model.CardDesignStatus;
import com.xenomi.cms.core.v2.product.model.CardForm;
import com.xenomi.cms.core.v2.product.model.DesignAssignmentStatus;
import com.xenomi.cms.core.v2.product.model.ProductVersionStatus;
import com.xenomi.cms.core.v2.product.repository.CardDesignV2Repository;
import com.xenomi.cms.core.v2.product.repository.CardProductV2Repository;
import com.xenomi.cms.core.v2.product.repository.ProductCardDesignAssignmentV2Repository;
import com.xenomi.cms.core.v2.product.repository.ProductCardFormPolicyV2Repository;
import com.xenomi.cms.core.v2.product.repository.ProductVersionV2Repository;
import org.springframework.data.domain.Page;
import org.springframework.data.domain.Pageable;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

@Service
public class ProductDesignAssignmentV2Service {
    private final ProductCardDesignAssignmentV2Repository repository;
    private final ProductVersionV2Repository versionRepository;
    private final CardProductV2Repository productRepository;
    private final CardDesignV2Repository designRepository;
    private final ProductCardFormPolicyV2Repository cardFormRepository;
    private final DatabaseUuidV7Generator uuidGenerator;

    public ProductDesignAssignmentV2Service(ProductCardDesignAssignmentV2Repository repository,
            ProductVersionV2Repository versionRepository, CardProductV2Repository productRepository,
            CardDesignV2Repository designRepository, ProductCardFormPolicyV2Repository cardFormRepository,
            DatabaseUuidV7Generator uuidGenerator) {
        this.repository = repository;
        this.versionRepository = versionRepository;
        this.productRepository = productRepository;
        this.designRepository = designRepository;
        this.cardFormRepository = cardFormRepository;
        this.uuidGenerator = uuidGenerator;
    }

    @Transactional
    public ProductCardDesignAssignmentV2Response create(UUID institutionId, UUID productId,
            UUID versionId, CreateProductCardDesignAssignmentV2Request request) {
        ProductVersionV2 version = requireDraftVersion(institutionId, productId, versionId);
        CardProductV2 product = requireProduct(institutionId, version.getProductId());
        CardDesignV2 design = requireDesign(institutionId, request.cardDesignId());
        if (design.getCardNetwork() != product.getCardNetwork()) {
            throw new IllegalArgumentException("Product and card design must use the same card network");
        }
        validateCardForm(institutionId, versionId, request.cardForm());
        validateDates(request.effectiveFrom(), request.effectiveTo());
        ProductCardDesignAssignmentV2 entity = ProductCardDesignAssignmentV2.builder()
                .assignmentId(uuidGenerator.generate()).institutionId(institutionId)
                .productVersionId(versionId).cardDesignId(request.cardDesignId())
                .cardForm(request.cardForm()).defaultDesign(Boolean.TRUE.equals(request.defaultDesign()))
                .assignmentStatus(DesignAssignmentStatus.DRAFT)
                .effectiveFrom(request.effectiveFrom()).effectiveTo(request.effectiveTo()).build();
        return toResponse(repository.save(entity));
    }

    @Transactional(readOnly = true)
    public Page<ProductCardDesignAssignmentV2Response> list(UUID institutionId, UUID productId,
            UUID versionId, Pageable pageable) {
        requireVersion(institutionId, productId, versionId);
        return repository.findByInstitutionIdAndProductVersionId(institutionId, versionId, pageable)
                .map(this::toResponse);
    }

    @Transactional(readOnly = true)
    public ProductCardDesignAssignmentV2Response get(UUID institutionId, UUID productId,
            UUID versionId, UUID assignmentId) {
        requireVersion(institutionId, productId, versionId);
        return toResponse(find(institutionId, versionId, assignmentId));
    }

    @Transactional
    public ProductCardDesignAssignmentV2Response update(UUID institutionId, UUID productId,
            UUID versionId, UUID assignmentId, UpdateProductCardDesignAssignmentV2Request request) {
        requireDraftVersion(institutionId, productId, versionId);
        ProductCardDesignAssignmentV2 entity = find(institutionId, versionId, assignmentId);
        boolean configurationChange = request.defaultDesign() != null
                || request.effectiveFrom() != null || request.effectiveTo() != null;
        if (configurationChange && entity.getAssignmentStatus() != DesignAssignmentStatus.DRAFT) {
            throw new IllegalArgumentException("Design assignment configuration can be changed only while DRAFT");
        }
        if (request.defaultDesign() != null) entity.setDefaultDesign(request.defaultDesign());
        if (request.effectiveFrom() != null) entity.setEffectiveFrom(request.effectiveFrom());
        if (request.effectiveTo() != null) entity.setEffectiveTo(request.effectiveTo());
        if (request.assignmentStatus() != null && request.assignmentStatus() != entity.getAssignmentStatus()) {
            validateTransition(entity.getAssignmentStatus(), request.assignmentStatus());
            if (request.assignmentStatus() == DesignAssignmentStatus.ACTIVE) {
                CardDesignV2 design = requireDesign(institutionId, entity.getCardDesignId());
                if (design.getDesignStatus() != CardDesignStatus.ACTIVE) {
                    throw new IllegalArgumentException("Card design must be ACTIVE before activating its assignment");
                }
                if (entity.isDefaultDesign() && repository
                        .existsByInstitutionIdAndProductVersionIdAndCardFormAndDefaultDesignTrueAndAssignmentStatus(
                                institutionId, versionId, entity.getCardForm(), DesignAssignmentStatus.ACTIVE)) {
                    throw new IllegalArgumentException("An ACTIVE default design already exists for this card form");
                }
            }
            entity.setAssignmentStatus(request.assignmentStatus());
        }
        validateDates(entity.getEffectiveFrom(), entity.getEffectiveTo());
        return toResponse(repository.save(entity));
    }

    private ProductVersionV2 requireVersion(UUID institutionId, UUID productId, UUID versionId) {
        return versionRepository.findByProductVersionIdAndInstitutionIdAndProductId(versionId, institutionId, productId)
                .orElseThrow(() -> new ResourceNotFoundException("Product version", versionId.toString()));
    }
    private ProductVersionV2 requireDraftVersion(UUID institutionId, UUID productId, UUID versionId) {
        ProductVersionV2 version = requireVersion(institutionId, productId, versionId);
        if (version.getVersionStatus() != ProductVersionStatus.DRAFT) {
            throw new IllegalArgumentException("Design assignments can be changed only while the version is DRAFT");
        }
        return version;
    }
    private CardProductV2 requireProduct(UUID institutionId, UUID productId) {
        return productRepository.findByProductIdAndInstitutionId(productId, institutionId)
                .orElseThrow(() -> new ResourceNotFoundException("Card product", productId.toString()));
    }
    private CardDesignV2 requireDesign(UUID institutionId, UUID designId) {
        return designRepository.findByCardDesignIdAndInstitutionId(designId, institutionId)
                .orElseThrow(() -> new ResourceNotFoundException("Card design", designId.toString()));
    }
    private ProductCardDesignAssignmentV2 find(UUID institutionId, UUID versionId, UUID assignmentId) {
        return repository.findByAssignmentIdAndInstitutionIdAndProductVersionId(assignmentId, institutionId, versionId)
                .orElseThrow(() -> new ResourceNotFoundException("Card design assignment", assignmentId.toString()));
    }
    private void validateCardForm(UUID institutionId, UUID versionId, CardForm requested) {
        ProductCardFormPolicyV2 form = cardFormRepository
                .findByProductVersionIdAndInstitutionId(versionId, institutionId)
                .orElseThrow(() -> new IllegalArgumentException("Configure card-form policy before assigning a design"));
        if (requested == CardForm.PHYSICAL && !form.isPhysicalAllowed()) {
            throw new IllegalArgumentException("PHYSICAL design cannot be assigned when physical cards are disabled");
        }
        if (requested == CardForm.VIRTUAL && !form.isVirtualAllowed()) {
            throw new IllegalArgumentException("VIRTUAL design cannot be assigned when virtual cards are disabled");
        }
    }
    private static void validateDates(OffsetDateTime from, OffsetDateTime to) {
        if (to != null && !to.isAfter(from)) throw new IllegalArgumentException("effectiveTo must be after effectiveFrom");
    }
    private static void validateTransition(DesignAssignmentStatus current, DesignAssignmentStatus target) {
        boolean allowed = switch (current) {
            case DRAFT -> target == DesignAssignmentStatus.ACTIVE || target == DesignAssignmentStatus.CLOSED;
            case ACTIVE -> target == DesignAssignmentStatus.SUSPENDED || target == DesignAssignmentStatus.CLOSED;
            case SUSPENDED -> target == DesignAssignmentStatus.ACTIVE || target == DesignAssignmentStatus.CLOSED;
            case CLOSED -> false;
        };
        if (!allowed) throw new IllegalArgumentException("Invalid design assignment transition: " + current + " -> " + target);
    }
    private ProductCardDesignAssignmentV2Response toResponse(ProductCardDesignAssignmentV2 a) {
        return new ProductCardDesignAssignmentV2Response(a.getAssignmentId(), a.getInstitutionId(),
                a.getProductVersionId(), a.getCardDesignId(), a.getCardForm(), a.isDefaultDesign(),
                a.getAssignmentStatus(), a.getEffectiveFrom(), a.getEffectiveTo());
    }
}
