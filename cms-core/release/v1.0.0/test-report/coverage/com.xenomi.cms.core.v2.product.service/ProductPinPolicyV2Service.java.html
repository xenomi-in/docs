package com.xenomi.cms.core.v2.product.service;

import java.util.UUID;

import com.xenomi.cms.core.exception.ResourceNotFoundException;
import com.xenomi.cms.core.v2.product.dto.ProductPinPolicyV2Response;
import com.xenomi.cms.core.v2.product.dto.UpsertProductPinPolicyV2Request;
import com.xenomi.cms.core.v2.product.entity.ProductPinPolicyV2;
import com.xenomi.cms.core.v2.product.entity.ProductVersionV2;
import com.xenomi.cms.core.v2.product.model.PinDeliveryMethod;
import com.xenomi.cms.core.v2.product.model.PinGenerationMethod;
import com.xenomi.cms.core.v2.product.model.ProductPolicyStatus;
import com.xenomi.cms.core.v2.product.model.ProductVersionStatus;
import com.xenomi.cms.core.v2.product.repository.ProductPinPolicyV2Repository;
import com.xenomi.cms.core.v2.product.repository.ProductVersionV2Repository;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

@Service
public class ProductPinPolicyV2Service {
    private final ProductPinPolicyV2Repository repository;
    private final ProductVersionV2Repository versionRepository;

    public ProductPinPolicyV2Service(
            ProductPinPolicyV2Repository repository,
            ProductVersionV2Repository versionRepository) {
        this.repository = repository;
        this.versionRepository = versionRepository;
    }

    @Transactional(readOnly = true)
    public ProductPinPolicyV2Response get(UUID institutionId, UUID productId, UUID versionId) {
        requireVersion(institutionId, productId, versionId);
        return repository.findByProductVersionIdAndInstitutionId(versionId, institutionId)
                .map(this::toResponse)
                .orElseThrow(() -> new ResourceNotFoundException("Product PIN policy", versionId.toString()));
    }

    @Transactional
    public ProductPinPolicyV2Response put(UUID institutionId, UUID productId, UUID versionId,
            UpsertProductPinPolicyV2Request request) {
        requireDraftVersion(institutionId, productId, versionId);
        validateConsistency(request);
        ProductPinPolicyV2 entity = repository
                .findByProductVersionIdAndInstitutionId(versionId, institutionId)
                .orElseGet(() -> ProductPinPolicyV2.builder()
                        .productVersionId(versionId)
                        .institutionId(institutionId)
                        .build());
        entity.setPinRequired(request.pinRequired());
        entity.setPinLength(request.pinLength());
        entity.setGenerationMethod(request.generationMethod());
        entity.setDeliveryMethod(request.deliveryMethod());
        entity.setCustomerPinSelectionAllowed(request.customerPinSelectionAllowed());
        entity.setPinChangeAllowed(request.pinChangeAllowed());
        entity.setPinResetAllowed(request.pinResetAllowed());
        entity.setMaximumAttempts(request.maximumAttempts());
        entity.setGreenPinEnabled(request.greenPinEnabled());
        entity.setPinMailerEnabled(request.pinMailerEnabled());
        entity.setAtmPinChangeEnabled(request.atmPinChangeEnabled());
        entity.setRandomPinGenerationEnabled(request.randomPinGenerationEnabled());
        entity.setOfflinePinSupported(request.offlinePinSupported());
        entity.setOnlinePinSupported(request.onlinePinSupported());
        entity.setUnblockMethod(request.unblockMethod());
        entity.setHsmPinProfileReference(normalize(request.hsmPinProfileReference()));
        entity.setPolicyStatus(ProductPolicyStatus.DRAFT);
        return toResponse(repository.save(entity));
    }

    private ProductVersionV2 requireVersion(UUID institutionId, UUID productId, UUID versionId) {
        return versionRepository.findByProductVersionIdAndInstitutionIdAndProductId(
                        versionId, institutionId, productId)
                .orElseThrow(() -> new ResourceNotFoundException("Product version", versionId.toString()));
    }

    private void requireDraftVersion(UUID institutionId, UUID productId, UUID versionId) {
        if (requireVersion(institutionId, productId, versionId).getVersionStatus() != ProductVersionStatus.DRAFT) {
            throw new IllegalArgumentException("Product PIN policy can be changed only while the version is DRAFT");
        }
    }

    private static void validateConsistency(UpsertProductPinPolicyV2Request request) {
        if (!request.pinRequired()) return;
        if (!request.onlinePinSupported() && !request.offlinePinSupported()) {
            throw new IllegalArgumentException(
                    "At least one of onlinePinSupported or offlinePinSupported must be enabled when PIN is required");
        }
        if (request.generationMethod() == PinGenerationMethod.CUSTOMER_SELECTED
                && !request.customerPinSelectionAllowed()) {
            throw new IllegalArgumentException(
                    "customerPinSelectionAllowed must be enabled for CUSTOMER_SELECTED generation");
        }
        if (request.generationMethod() != PinGenerationMethod.CUSTOMER_SELECTED
                && !request.randomPinGenerationEnabled()) {
            throw new IllegalArgumentException(
                    "randomPinGenerationEnabled must be true for HSM-generated or temporary PINs");
        }
        if (request.deliveryMethod() == PinDeliveryMethod.GREEN_PIN_OTP && !request.greenPinEnabled()) {
            throw new IllegalArgumentException("greenPinEnabled must be true for GREEN_PIN_OTP delivery");
        }
        if (request.deliveryMethod() == PinDeliveryMethod.PIN_MAILER && !request.pinMailerEnabled()) {
            throw new IllegalArgumentException("pinMailerEnabled must be true for PIN_MAILER delivery");
        }
        if (request.atmPinChangeEnabled() && !request.pinChangeAllowed()) {
            throw new IllegalArgumentException(
                    "pinChangeAllowed must be true when ATM PIN change is enabled");
        }
    }

    private static String normalize(String value) {
        if (value == null || value.isBlank()) return null;
        return value.trim();
    }

    private ProductPinPolicyV2Response toResponse(ProductPinPolicyV2 entity) {
        return new ProductPinPolicyV2Response(
                entity.getProductVersionId(), entity.getInstitutionId(), entity.isPinRequired(),
                entity.getPinLength(), entity.getGenerationMethod(), entity.getDeliveryMethod(),
                entity.isCustomerPinSelectionAllowed(), entity.isPinChangeAllowed(),
                entity.isPinResetAllowed(), entity.getMaximumAttempts(), entity.isGreenPinEnabled(),
                entity.isPinMailerEnabled(), entity.isAtmPinChangeEnabled(),
                entity.isRandomPinGenerationEnabled(), entity.isOfflinePinSupported(),
                entity.isOnlinePinSupported(), entity.getUnblockMethod(),
                entity.getHsmPinProfileReference(), entity.getPolicyStatus());
    }
}
