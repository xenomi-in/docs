    /*
     * This command translates a Visa working key from encryption under a key encryption key (KEK) / ZMK to encryption under the MFK,
     */
    public String translateWorkingKeyUnderKek(String workingKey, String kekProtectedKeyBlock,
            String keyType, boolean tr31Format) {
        String modifier = "0";
        FuturexHsmCmd futurexHsmCmd = new FuturexHsmCmd("TVWK");
        try {
            switch (keyType) {
                case "ZPK":
                    modifier = "1";
                    break;
                case "AES":
                    break;
                default:
            }
            //futurexHsmCmd.addField("AO", "TVWK"); // command indicator
            futurexHsmCmd.addField("AS", modifier); // modifier
            futurexHsmCmd.addField("FD", workingKey); // working key encrypted under the KEK/ZMK
            futurexHsmCmd.addField("AP", kekProtectedKeyBlock); // KEK protected by the CMS HSM
            if(tr31Format) futurexHsmCmd.addField("OF", "T");
            else futurexHsmCmd.addField("OF", "C"); // output key fromat C: Cryptogram , T: TR-31 , A : AKB format key block
            String command = futurexHsmCmd.build();
            String response = hsmClient.sendMessage(command);
            FuturexHsmCmd responseCmd = FuturexHsmCmd.parse(response);
            if (responseCmd.isSuccess()) {
                String keyUnderMFK = responseCmd.getField("BG");
                String KCV = responseCmd.getField("AE");
                return keyUnderMFK + ";" + KCV;

            } else {
                throw new IllegalStateException("HSM rejected working-key translation: "
                        + responseCmd.getErroMsg());
            }
        } catch (Exception e) {
            logger.error("HSM working-key translation failed for keyType={}", keyType, e);
            throw new IllegalStateException("HSM working-key translation failed", e);
        }

    }

    /*
    * This command generates a working key encrypted under the MFK modifier of the key type. 
     */
    public String generateWorkingKey(String keyType) {
        try {
            FuturexHsmCmd futurexHsmCmd = new FuturexHsmCmd("GWKS");
            // futurexHsmCmd.addField("AO", "GWKS"); // command indicator

            futurexHsmCmd.addField("FS", "2"); // 3DES keyType  1 - single DES key , 2 - Double length DES key , 3 - Triple Length DES key

            switch (keyType) {
                case "PVK":
                    futurexHsmCmd.addField("AS", "4"); // modifier 4 - PIN verification Key
                    break;
                case "DKE":
                    futurexHsmCmd.addField("AS", "2"); // modifier 2 - Data Encryption Key
                    break;
                case "PGK":
                    futurexHsmCmd.addField("AS", "9"); // modifier 9 - PIN Generation Key
                    break;
                case "CVK":
                    futurexHsmCmd.addField("AS", "4"); // modifier 4 - CVK
                    break;
                case "ZPK":
                    futurexHsmCmd.addField("AS", "1"); // modifier 1 - Zone PIN Key
                    break;
                default:
                    futurexHsmCmd.addField("AS", "0"); // modifier 0 - generic
                    break;
            }
            String command = futurexHsmCmd.build();
            String response = hsmClient.sendMessage(command);
            FuturexHsmCmd responseCmd = FuturexHsmCmd.parse(response);
            if (responseCmd.isSuccess()) {
                String workingKey = responseCmd.getField("BG");
                String KCV = responseCmd.getField("AE");
                return workingKey + ";" + KCV;
            } else {
                return responseCmd.getErroMsg();
            }
        } catch (Exception e) {
            logger.error("Error building command: {}", e.getMessage());
            return null;
        }
    }

    /**
     * Generates a random PIN inside the HSM and returns the PIN offset for storage.
     * PCI-DSS: PIN is never present in application memory; generation is HSM-only.
     * Uses Futurex GPIN/offset-generation flow; exact command may vary by HSM model.
     *
     * @param pan       Primary Account Number (full PAN)
     * @param pinMethod PIN method (VISA, IBM3624, or ISO9564)
     * @param pinLength PIN length (e.g. 4)
     * @return New offset to store in PinData, or null if HSM does not support or fails
     */
    public String generateRandomPinOffset(String pan, String pinMethod, int pinLength, String pvk, String pvki, String pek) {
        try {
            String withoutCheckDigit = pan.length() >= 16
                    ? pan.substring(0, pan.length() - 1)
                    : pan;
            String partialPan = withoutCheckDigit.length() >= 11
                    ? withoutCheckDigit.substring(withoutCheckDigit.length() - 11)
                    : withoutCheckDigit;
            FuturexHsmCmd futurexHsmCmd = new FuturexHsmCmd("GPIN");
            if ("VISA".equalsIgnoreCase(pinMethod) || "VISA_PVV".equalsIgnoreCase(pinMethod)) {
                futurexHsmCmd.addField("BF", "3");
                futurexHsmCmd.addField("BY", pvki);
            } else if ("IBM3624".equalsIgnoreCase(pinMethod)) {
                futurexHsmCmd.addField("BF", "2");
                futurexHsmCmd.addField("BS", "0123456789012345");
                futurexHsmCmd.addField("BE", pan);
                futurexHsmCmd.addField("AD", String.valueOf(Math.min(12, Math.max(4, pinLength))));
            } else if ("ISO9564".equalsIgnoreCase(pinMethod)) {
                futurexHsmCmd.addField("BF", "1");
            } else {
                logger.warn("Unknown pin method for random PIN generation: {}; using VISA", pinMethod);
                futurexHsmCmd.addField("BF", "3");
            }
            futurexHsmCmd.addField("BZ", partialPan);
           // String pvk = getWorkingKeyForCard(pan, "PVK");
            if (pvk != null) {
                futurexHsmCmd.addField("CA", pvk);
            }
            String additionalPinBlockData = withoutCheckDigit.substring(withoutCheckDigit.length() - 12);

            futurexHsmCmd.addField("BT", pek);
            futurexHsmCmd.addField("AK", additionalPinBlockData);
            futurexHsmCmd.addField("CM", String.valueOf(pinLength));
            String command = futurexHsmCmd.build();
            String response = hsmClient.sendMessage(command);
            FuturexHsmCmd responseCmd = FuturexHsmCmd.parse(response);
            if (responseCmd.isSuccess()) {
                String offset = responseCmd.getField("BX");
                if (offset == null) offset = responseCmd.getField("AY");
                if (offset != null) {
                    logger.debug("HSM generated random PIN offset for PAN: {}", maskPan(pan));
                    return offset;
                }
            } else {
                logger.warn("HSM random PIN generation failed or not supported: {}", responseCmd.getErroMsg());
            }
            return null;
        } catch (Exception e) {
            logger.error("Error generating random PIN offset in HSM: {}", e.getMessage());
            return null;
        }
    }

    /** Generates PIN verification metadata from an encrypted ISO-9564 format-0 PIN block. */
    public String generatePinOffsetFromProtectedBlock(String pan, String pinBlock,
            String pinMethod, int pinLength, String pvki, String pvk, String zpk) {
        try {
            String withoutCheckDigit = pan.substring(0, pan.length() - 1);
            String partialPan = withoutCheckDigit.substring(withoutCheckDigit.length() - 11);
            String additionalPinBlockData = withoutCheckDigit.substring(withoutCheckDigit.length() - 12);
            FuturexHsmCmd command = new FuturexHsmCmd("GOFF");
            addPinMethodFields(command, pinMethod, pan, partialPan, pinLength, pvki, pvk);
            command.addField("AW", "1");
            command.addField("AL", pinBlock);
            command.addField("AX", zpk);
            command.addField("AK", additionalPinBlockData);
            FuturexHsmCmd response = FuturexHsmCmd.parse(hsmClient.sendMessage(command.build()));
            if (!response.isSuccess()) {
                throw new IllegalStateException("HSM rejected PIN offset generation: " + response.getErroMsg());
            }
            String offset = "IBM3624".equalsIgnoreCase(pinMethod)
                    ? response.getField("AY") : response.getField("BX");
            if (offset == null) offset = response.getField("AY");
            return offset;
        } catch (Exception exception) {
            logger.error("HSM protected PIN-block offset generation failed", exception);
            throw new IllegalStateException("HSM protected PIN-block offset generation failed", exception);
        }
    }

    /** Verifies the current encrypted PIN block and derives metadata for the new block. */
    public String changePinUsingProtectedBlocks(String pan, String currentPinBlock,
            String newPinBlock, String currentOffset, String pinMethod, int pinLength,
            String pvki, String pvk, String zpk) {
        try {
            String withoutCheckDigit = pan.substring(0, pan.length() - 1);
            String partialPan = withoutCheckDigit.substring(withoutCheckDigit.length() - 11);
            String additionalPinBlockData = withoutCheckDigit.substring(withoutCheckDigit.length() - 12);
            FuturexHsmCmd command = new FuturexHsmCmd("RPIN");
            addPinMethodFields(command, pinMethod, pan, partialPan, pinLength, pvki, pvk);
            command.addField("AW", "1");
            command.addField("AL", currentPinBlock);
            command.addField("FW", newPinBlock);
            command.addField("AX", zpk);
            command.addField("BX", currentOffset);
            command.addField("AK", additionalPinBlockData);
            FuturexHsmCmd response = FuturexHsmCmd.parse(hsmClient.sendMessage(command.build()));
            if (!response.isSuccess()) {
                throw new IllegalStateException("HSM rejected PIN change: " + response.getErroMsg());
            }
            String verified = response.getField("BB");
            String offset = "IBM3624".equalsIgnoreCase(pinMethod)
                    ? response.getField("AY") : response.getField("BX");
            if (offset == null) offset = response.getField("AY");
            return verified + ";" + (offset == null ? "" : offset);
        } catch (Exception exception) {
            logger.error("HSM protected PIN-change processing failed", exception);
            throw new IllegalStateException("HSM protected PIN-change processing failed", exception);
        }
    }

    private static void addPinMethodFields(FuturexHsmCmd command, String pinMethod,
            String pan, String partialPan, int pinLength, String pvki, String pvk) {
        if ("VISA".equalsIgnoreCase(pinMethod) || "VISA_PVV".equalsIgnoreCase(pinMethod)) {
            command.addField("BF", "3");
            command.addField("BY", pvki);
            command.addField("CA", pvk);
            command.addField("BZ", partialPan);
        } else if ("IBM3624".equalsIgnoreCase(pinMethod)) {
            command.addField("BJ", "1");
            command.addField("BF", "2");
            command.addField("BS", "0123456789012345");
            command.addField("AT", "F");
            command.addField("BE", pan);
            command.addField("AZ", pvk);
            command.addField("AD", String.valueOf(pinLength));
        } else if ("ISO9564".equalsIgnoreCase(pinMethod)) {
            command.addField("BF", "1");
            command.addField("CA", pvk);
        } else {
            throw new IllegalArgumentException("Unsupported PIN method");
        }
    }

    /**
     * Sends an echo command to test HSM connectivity.
     *
     * @return true if successful, false otherwise
     */
    public boolean testConnection() {
        try {
            FuturexHsmCmd futurexHsmCmd = new FuturexHsmCmd("ECHO");
            futurexHsmCmd.addField("BJ", "0");
            futurexHsmCmd.addField("AG", "Echo Test connection.");
            String response = hsmClient.sendMessage(futurexHsmCmd.build());
            if (response == null || response.trim().isEmpty()) {
                logger.error("HSM connection test failed: No response received (timeout or connection error)");
                return false;
            }
            FuturexHsmCmd responseCmd = FuturexHsmCmd.parse(response);
            logger.debug("HSM echo test completed; success={}", responseCmd.isSuccess());
            if (responseCmd.isSuccess()) {
                logger.info("HSM echo test success: {}", responseCmd.getField("AG"));
                return true;
            } else {
                logger.error("HSM echo test failed : {}", responseCmd.getErroMsg());
                return false;
            }
        } catch (Exception e) {
            logger.error("Error testing HSM connection: {}", e.getMessage());
            e.printStackTrace();
            return false;
        }
    }

    /**
     * Sends multiple messages concurrently to demonstrate connection pooling.
     *
     * @param messages Array of messages to send
     * @return Array of CompletableFuture responses
     */
    public CompletableFuture<String>[] sendMessagesConcurrently(String... messages) {
        @SuppressWarnings("unchecked")
        CompletableFuture<String>[] futures = new CompletableFuture[messages.length];

        for (int i = 0; i < messages.length; i++) {
            final int messageIndex = i;
            futures[i] = sendMessageAsync(messages[i])
                    .thenApply(response -> {
                        logger.debug("Received HSM batch response for message {} (payloadLength={})",
                                messageIndex, response == null ? 0 : response.length());
                        return response;
                    })
                    .exceptionally(throwable -> {
                        logger.error("Error sending message {}: {}", messageIndex, throwable.getMessage());
                        return null;
                    });
        }

        return futures;
    }

    /**
     * Generates CVV (Card Verification Value) for a card.
     *
     * @param pan Primary Account Number
     * @param expiryDate Card expiry date (MMYY format)
     * @param serviceCode Service code (e.g., "201" for ATM, "101" for POS)
     * @param cvvKey CVV generation key
     * @return Generated CVV or null if failed
     */
    public String generateCvv(String pan, String expiryDate, String serviceCode, String cvvKey) {
        try {
            if (cvvKey == null || cvvKey.isBlank()) return null;
            FuturexHsmCmd futurexHsmCmd = new FuturexHsmCmd("GCVV");
            // Add required fields for CVV generation
            futurexHsmCmd.addField("AV", pan); // PAN
            futurexHsmCmd.addField("FA", expiryDate); // Expiry date
            futurexHsmCmd.addField("FB", serviceCode); // Service code
            futurexHsmCmd.addField("CA", cvvKey); // CVV key
            futurexHsmCmd.addField("FT", "1"); // CVV type

            String command = futurexHsmCmd.build();
            String response = hsmClient.sendMessage(command);
            if (response != null) {
                FuturexHsmCmd responseCmd = FuturexHsmCmd.parse(response);

                if (responseCmd.isSuccess()) {
                    String cvv1 = responseCmd.getField("FC"); // Generated CVV

                } else {
                    logger.error("CVV generation failed: {}", responseCmd.getErroMsg());

                }
            } else {
                return null;
            }

        } catch (Exception e) {
            logger.error("Error generating CVV: {}", e.getMessage());
            return null;
        }
    }

    /** Generates a deterministic MAC using only an HSM-protected key reference/block. */
    public String generateMacWithKeyReference(String data, String protectedKeyBlock) {
        if (data == null || protectedKeyBlock == null || protectedKeyBlock.isBlank()) return null;
        try {
            FuturexHsmCmd command = new FuturexHsmCmd("GMAC");
            command.addField("BN", "0");
            command.addField("AR", protectedKeyBlock);
            command.addField("BO", HexFormat.of().formatHex(data.getBytes(StandardCharsets.UTF_8)));
            command.addField("AD", "8");
            command.addField("BF", "1");
            FuturexHsmCmd response = FuturexHsmCmd.parse(hsmClient.sendMessage(command.build()));
            return response.isSuccess() ? response.getField("BP") : null;
        } catch (Exception exception) {
            logger.error("HSM MAC generation failed");
            return null;
        }
    }

    /** Encrypts UTF-8 data using an HSM-protected assignment key and a fresh IV. */
    public String encryptDataWithKeyReference(String clearText, String protectedKeyBlock, String algorithm) {
        if (clearText == null || protectedKeyBlock == null || protectedKeyBlock.isBlank()) return null;
        try {
            int ivBytes = "AES".equalsIgnoreCase(algorithm) ? 16 : 8;
            byte[] iv = new byte[ivBytes];
            new SecureRandom().nextBytes(iv);
            String ivHex = HexFormat.of().formatHex(iv).toUpperCase();
            FuturexHsmCmd command = new FuturexHsmCmd("CBCE");
            command.addField("GG", protectedKeyBlock);
            command.addField("GH", ivHex);
            command.addField("BO", HexFormat.of().formatHex(
                    clearText.getBytes(StandardCharsets.UTF_8)).toUpperCase());
            command.addField("BF", "0");
            command.addField("ZA", "1");
            FuturexHsmCmd response = FuturexHsmCmd.parse(hsmClient.sendMessage(command.build()));
            String cipherText = response.isSuccess() ? response.getField("BP") : null;
            return cipherText == null ? null : "HSM1:" + algorithm + ":" + ivHex + ":" + cipherText;
        } catch (Exception exception) {
            logger.error("HSM data encryption failed");
            return null;
        }
    }

    /** Decrypts an envelope produced by {@link #encryptDataWithKeyReference}. */
    public String decryptDataWithKeyReference(String envelope, String protectedKeyBlock, String algorithm) {
        if (envelope == null || protectedKeyBlock == null || protectedKeyBlock.isBlank()) return null;
        try {
            String[] parts = envelope.split(":", 4);
            if (parts.length != 4 || !"HSM1".equals(parts[0])
                    || !algorithm.equalsIgnoreCase(parts[1])) return null;
            FuturexHsmCmd command = new FuturexHsmCmd("CBCD");
            command.addField("GG", protectedKeyBlock);
            command.addField("GH", parts[2]);
            command.addField("BO", parts[3]);
            command.addField("BF", "0");
            command.addField("ZA", "1");
            FuturexHsmCmd response = FuturexHsmCmd.parse(hsmClient.sendMessage(command.build()));
            String clearHex = response.isSuccess() ? response.getField("BP") : null;
            return clearHex == null ? null : new String(HexFormat.of().parseHex(clearHex), StandardCharsets.UTF_8);
        } catch (Exception exception) {
            logger.error("HSM data decryption failed");
            return null;
        }
    }
    /**
     * Masks PAN for logging purposes.
     *
     * @param pan PAN to mask
     * @return Masked PAN
     */
    private String maskPan(String pan) {
        if (pan == null || pan.length() < 8) {
            return "****";
        }
        return pan.substring(0, 4) + "****" + pan.substring(pan.length() - 4);
    }

    /**
     * Shuts down the HSM client gracefully.
     */
    @PreDestroy
    public void shutdown() {
        logger.info("Shutting down Netty HSM service");
        hsmClient.shutdown();
    }
