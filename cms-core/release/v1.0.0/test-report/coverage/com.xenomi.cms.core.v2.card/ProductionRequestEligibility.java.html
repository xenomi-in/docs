package com.xenomi.cms.core.v2.card;

import java.time.OffsetDateTime;
import java.util.UUID;

import org.springframework.jdbc.core.JdbcTemplate;

import com.xenomi.cms.core.v2.card.entity.CardProductionRequestV2;
import com.xenomi.cms.core.v2.card.model.ProductionRequestStatus;

/**
 * ELIGIBLE means the embossing scheduler may claim the request.
 * A request stays PENDING until Switch registration has been applied and card, address, and approval are ready.
 */
public final class ProductionRequestEligibility {
    private ProductionRequestEligibility() {}

    public static void stamp(CardProductionRequestV2 request, long appliedSwitchRevision, OffsetDateTime now) {
        boolean ready = request.isCardDataReady() && request.isAddressReady() && request.isApprovalReady();
        if (appliedSwitchRevision > 0 && ready) {
            request.setSwitchOnboardingReady(true);
            request.setRequestStatus(ProductionRequestStatus.ELIGIBLE);
            request.setEligibleAt(now);
        } else {
            request.setSwitchOnboardingReady(false);
            request.setRequestStatus(ProductionRequestStatus.PENDING);
            request.setEligibleAt(null);
        }
    }

    public static int promoteAfterRegistration(JdbcTemplate db, UUID institutionId, UUID cardId) {
        return db.update("""
                update cms_v2.card_production_request
                set switch_onboarding_ready = true,
                    request_status = 'ELIGIBLE',
                    eligible_at = now(),
                    updated_at = now()
                where institution_id = ?
                  and card_id = ?
                  and request_status = 'PENDING'
                  and card_data_ready
                  and address_ready
                  and approval_ready
                """, institutionId, cardId);
    }
}
