    public Draft prepareDraft(UUID institutionId, CreateCardV2Request request, String actor, UUID operationId, boolean newPan) {
        OffsetDateTime now = OffsetDateTime.now(ZoneOffset.UTC);
        InstitutionV2 institution = requireInstitution(institutionId);
        if (institution.getStatus() != com.xenomi.cms.core.v2.institution.model.InstitutionStatus.ACTIVE) {
            throw new BusinessRuleViolationException("Institution must be ACTIVE for issuance");
        }
        if (request.getAccountLinkType() != null && request.getAccountLinkType() != CardAccountLinkType.PRIMARY_FUNDING) {
            throw new BusinessRuleViolationException("Original issuance requires PRIMARY_FUNDING account link");
        }
        if (request.getActivationPolicyVersionId() == null) {
            throw new BusinessRuleViolationException("An activation policy version is required for card issuance");
        }
        ActivationPolicyVersionV2 activationPolicy = activationPolicyVersionRepository
                .findByInstitutionIdAndActivationPolicyVersionId(institutionId, request.getActivationPolicyVersionId())
                .orElseThrow(() -> new ResourceNotFoundException("Activation policy version",
                        request.getActivationPolicyVersionId().toString()));
        if (activationPolicy.getEffectiveFrom().isAfter(now)
                || !"Asia/Kolkata".equals(activationPolicy.getExpiryTimezone())) {
            throw new BusinessRuleViolationException("Activation policy version is not currently effective");
        }
        CustomerReferenceV2 customer = requireCustomer(institutionId, request.getCustomerReferenceId());
        ExternalAccountReferenceV2 account = requireAccount(institutionId, request.getAccountReferenceId());
        AddressReferenceV2 address = requireAddressIfPresent(
                institutionId, customer.getCustomerReferenceId(), request.getShippingAddressReferenceId());
        CardProductV2 product = requireProduct(institutionId, request.getProductId());
        ProductVersionV2 version = requireActiveVersion(
                institutionId, product.getProductId(), request.getProductVersionId(), now);
        ProductCardFormPolicyV2 formPolicy = formPolicyRepository
                .findByProductVersionIdAndInstitutionId(version.getProductVersionId(), institutionId)
                .orElseThrow(() -> new BusinessRuleViolationException(
                        "Active product version does not have a card-form policy"));
        CardFormV2 cardForm = resolveCardForm(formPolicy.getIssuanceMode());
        if ((requiresImmediatePhysical(cardForm) && !formPolicy.isPhysicalAllowed())
                || (hasVirtualCredential(cardForm) && !formPolicy.isVirtualAllowed())) {
            throw new BusinessRuleViolationException("Product form capability does not permit issuance mode");
        }

        BinRangeV2 range = binRangeRepository
                .findForPanAllocation(institutionId, request.getBinRangeId())
                .orElseThrow(() -> new ResourceNotFoundException(
                        "BIN range", request.getBinRangeId().toString()));
        if (range.getRangeStatus() != BinRangeStatus.ACTIVE && (newPan || range.getRangeStatus() != BinRangeStatus.EXHAUSTED)) {
            throw new BusinessRuleViolationException("BIN range must be ACTIVE for card issuance");
        }
        validateBinAndRangeEffective(institutionId, range, now);
        List<ProductBinRangeAssignmentV2> assignments = rangeAssignmentRepository.findEffectiveAssignments(
                institutionId, product.getProductId(), range.getBinRangeId(),
                ProductBinAssignmentStatus.ACTIVE, now);
        if (assignments.isEmpty()) {
            throw new BusinessRuleViolationException(
                    "BIN range is not actively assigned to the selected product");
        }

        ResolvedProfiles profiles = resolveProfiles(institutionId, version.getProductVersionId(),
                cardForm, request);
        ProductionContext production = resolveProduction(
                institutionId, version.getProductVersionId(), cardForm, request);

        LocalDate issueDate = now.toLocalDate();
        LocalDate expiryDate = calculateExpiry(issueDate, version);
        String serviceCode = assignments.get(0).getServiceCodeOverride() == null
                ? version.getDefaultServiceCode() : assignments.get(0).getServiceCodeOverride();
        cardSecurityService.validateIssuanceKeys(institutionId, range.getBinId());
        var pinPolicy=pinLifecycleService.validateOriginalConfiguration(CardV2.builder().institutionId(institutionId)
                .binId(range.getBinId()).productVersionId(version.getProductVersionId()).build());
        controlProfileInitializer.validateOriginalConfiguration(institutionId, version.getProductVersionId());
        return new Draft(institutionId, request, actor, operationId, uuidGenerator.generate(),
                uuidGenerator.generate(), uuidGenerator.generate(), now, issueDate, expiryDate,
                serviceCode, cardForm, formPolicy, profiles, production, range, pinPolicy);
    }

            com.xenomi.cms.core.v2.card.pin.entity.PinDataV2 pin,
            com.xenomi.cms.core.v2.card.lifecycle.entity.AccountRestrictionSnapshotV2 holds,
            jakarta.persistence.EntityManager em, CardV2 predecessor) {

    private CardV2 finalizeDraftInternal(Draft draft, CardSecurityMaterial security,
            com.xenomi.cms.core.v2.card.pin.entity.PinDataV2 pin,
            com.xenomi.cms.core.v2.card.lifecycle.entity.AccountRestrictionSnapshotV2 holds,
            jakarta.persistence.EntityManager em, CardV2 predecessor,
            com.xenomi.cms.core.v2.product.model.ReplacementReason replacementReason,
            boolean newPanReplacement) {
        UUID institutionId = draft.institutionId();
        var request = draft.request();
        String actor = draft.actor();
        var now = draft.issuedAt();
        var range = draft.range();
        var cardForm = draft.cardForm();
        var formPolicy = draft.formPolicy();
        var profiles = draft.profiles();
        var production = draft.production();
        var issueDate = draft.issueDate();
        var expiryDate = draft.expiryDate();
        var institution = requireInstitution(institutionId);
        if(institution.getStatus()!=com.xenomi.cms.core.v2.institution.model.InstitutionStatus.ACTIVE)
            throw new BusinessRuleViolationException("Institution must remain ACTIVE at finalization");
        var customer = requireCustomer(institutionId, request.getCustomerReferenceId());
        var account = requireAccount(institutionId, request.getAccountReferenceId());
        var product = requireProduct(institutionId, request.getProductId());
        var version = requireActiveVersion(institutionId, product.getProductId(), request.getProductVersionId(), OffsetDateTime.now(ZoneOffset.UTC));
        var address = requireAddressIfPresent(institutionId, customer.getCustomerReferenceId(), request.getShippingAddressReferenceId());
        if(!formPolicy.equals(formPolicyRepository.findByProductVersionIdAndInstitutionId(version.getProductVersionId(),institutionId).orElseThrow())
                || !profiles.equals(resolveProfiles(institutionId,version.getProductVersionId(),cardForm,request))
                || !java.util.Objects.equals(production,resolveProduction(institutionId,version.getProductVersionId(),cardForm,request)))
            throw new BusinessRuleViolationException("Prepared issuance configuration changed; reconciliation required");
        var currentRange=binRangeRepository.findForPanAllocation(institutionId,range.getBinRangeId()).orElseThrow();
        if ((currentRange.getRangeStatus()!=BinRangeStatus.ACTIVE && currentRange.getRangeStatus()!=BinRangeStatus.EXHAUSTED)
                || !currentRange.getRangeStart().equals(range.getRangeStart()) || !currentRange.getRangeEnd().equals(range.getRangeEnd()))
            throw new BusinessRuleViolationException("Reserved BIN range is no longer eligible");
        validateBinAndRangeEffective(institutionId,currentRange,OffsetDateTime.now(ZoneOffset.UTC));
        if(rangeAssignmentRepository.findEffectiveAssignments(institutionId,product.getProductId(),range.getBinRangeId(),
                ProductBinAssignmentStatus.ACTIVE,OffsetDateTime.now(ZoneOffset.UTC)).isEmpty())
            throw new BusinessRuleViolationException("Product BIN assignment is no longer effective");
        UUID domainIdempotency = draft.operationId();
        String idempotencyKey = draft.operationId().toString();

        UUID panAllocationId = draft.panAllocationId();
        PanAllocationV2 panAllocation = predecessor != null && !newPanReplacement
                ? em.find(PanAllocationV2.class, predecessor.getPanAllocationId())
                : panAllocationRepository.save(PanAllocationV2.builder()
                .panAllocationId(panAllocationId)
                .institutionId(institutionId)
                .binRangeId(range.getBinRangeId())
                .allocationStrategy(range.getAllocationStrategy())
                .panFingerprint(security.panFingerprint())
                .securePanReference(security.secureDataReference())
                .panHmac(security.panHmac())
                // Older prepared RANDOM results contain the reserved offset; it is not a sequence.
                .sequenceValue(range.getAllocationStrategy() == com.xenomi.cms.core.v2.bin.model.AllocationStrategy.SEQUENTIAL
                        ? security.sequenceValue() : null)
                .allocationStatus(PanAllocationStatus.ISSUED)
                .idempotencyKey(domainIdempotency)
                .reservedAt(now)
                .issuedAt(now)
                .accountReferenceId(account.getExternalAccountReferenceId())
                .createdBy(actor)
                .build());

        em.flush();
        if (predecessor == null || newPanReplacement) em.persist(pin);
        em.flush();
        UUID cardId = draft.cardId();
        boolean immediatePhysical = requiresImmediatePhysical(cardForm);
        CardV2 card = cardRepository.save(CardV2.builder()
                .cardId(cardId)
                .institutionId(institutionId)
                .customerReferenceId(customer.getCustomerReferenceId())
                .productId(product.getProductId())
                .productVersionId(version.getProductVersionId())
                .binId(range.getBinId())
                .binRangeId(range.getBinRangeId())
                .panAllocationId(panAllocation.getPanAllocationId())
                .maskedPan(security.maskedPan())
                .panLastFour(security.panLastFour())
                .cardDesignId(profiles.cardDesignId())
                .technologyProfileId(profiles.technologyProfileId())
                .cvmProfileId(profiles.cvmProfileId())
                .shippingAddressReferenceId(address == null ? null : address.getAddressReferenceId())
                .cardForm(cardForm)
                .issuanceMode(formPolicy.getIssuanceMode())
                .physicalIssueTiming(formPolicy.getPhysicalIssueTiming())
                .cardSequenceNumber(null)
                .embossedNameLine1(trim(request.getEmbossedNameLine1()))
                .embossedNameLine2(trim(request.getEmbossedNameLine2()))
                .embossedNameLine3(trim(request.getEmbossedNameLine3()))
                .embossedNameLine4(trim(request.getEmbossedNameLine4()))
                .panEncrypted(security.encryptedPan())
                .cvv1Encrypted(security.encryptedCvv1())
                .cvv2Encrypted(security.encryptedCvv2())
                .track2Encrypted(security.encryptedTrack2())
                .dataEncryptionKeyId(security.encryptionKeyId())
                .panHmacKeyId(security.panHmacKeyId())
                .cvvKeyId(security.cvvKeyId())
                .issueDate(issueDate)
                .expiryDate(expiryDate)
                .issuedAt(now)
                .issuanceReason(issuanceReason(predecessor, replacementReason, newPanReplacement))
                .previousCardId(predecessor == null ? null : predecessor.getCardId())
                .rootCardId(predecessor == null ? null : predecessor.getRootCardId() == null ? predecessor.getCardId() : predecessor.getRootCardId())
                .replacementReason(predecessor == null ? null : replacementReason)
                .panReplacementStrategy(predecessor == null ? null : newPanReplacement
                        ? com.xenomi.cms.core.v2.product.model.ReplacementPanStrategy.GENERATE_NEW_PAN
                        : com.xenomi.cms.core.v2.product.model.ReplacementPanStrategy.PRESERVE_EXISTING_PAN)
                .pinDataRef(draft.pinDataId())
                .credentialIssuanceRef(draft.operationId())
                .activationPolicyVersion(request.getActivationPolicyVersionId())
                .collectionsHoldReason(predecessor != null && !newPanReplacement && holds.getCollectionsHoldReason() == null
                        ? predecessor.getCollectionsHoldReason() : holds.getCollectionsHoldReason())
                .complianceHoldReason(predecessor != null && !newPanReplacement && (holds.getComplianceHoldReason() == null
                        || predecessor.getComplianceHoldReason() == com.xenomi.cms.core.v2.card.model.ComplianceHoldReasonV2.REGULATORY)
                        ? predecessor.getComplianceHoldReason() : holds.getComplianceHoldReason())
                .customerHoldReason(predecessor == null || newPanReplacement ? null : predecessor.getCustomerHoldReason())
                .opsHoldReason(predecessor == null || newPanReplacement ? null : predecessor.getOpsHoldReason())
                .securityHoldReason(predecessor == null || newPanReplacement ? null : predecessor.getSecurityHoldReason())
                .physicalFulfilmentStatus(immediatePhysical ?
                        com.xenomi.cms.core.v2.card.model.PhysicalFulfilmentStatusV2.REQUESTED :
                        com.xenomi.cms.core.v2.card.model.PhysicalFulfilmentStatusV2.NOT_APPLICABLE)
                .issueSequence(predecessor == null ? 1 : Math.addExact(predecessor.getIssueSequence(), 1))
                .digitalAvailable(hasVirtualCredential(cardForm))
                .createdBy(actor)
                .updatedBy(actor)
                .build());

        CardAccountLinkV2 link = accountLinkRepository.save(CardAccountLinkV2.builder()
                .cardAccountLinkId(uuidGenerator.generate())
                .institutionId(institutionId)
                .cardId(cardId)
                .externalAccountReferenceId(account.getExternalAccountReferenceId())
                .linkType(request.getAccountLinkType() == null
                        ? CardAccountLinkType.PRIMARY_FUNDING : request.getAccountLinkType())
                .primary(true)
                .linkStatus(ReferenceStatus.ACTIVE)
                .effectiveFrom(now)
                .sourceReference(trim(request.getAccountLinkSourceReference()))
                .build());

        controlProfileInitializer.initialize(card, institution, actor);
        if (predecessor == null || newPanReplacement) pinLifecycleService.recordOriginalHistory(card, pin, draft.operationId(), actor);
        var lookup = em.createQuery("select p from PanLookupIdentityV2 p where p.institutionId=:t and p.panAllocationId=:f and p.hmacKeyId=:k",
                        com.xenomi.cms.core.v2.card.lifecycle.entity.PanLookupIdentityV2.class)
                .setParameter("t", institutionId).setParameter("f", panAllocationId).setParameter("k", security.panHmacKeyId()).getResultList();
        if (!lookup.isEmpty() && !lookup.getFirst().getPanHash().equals(security.panHmac()))
            throw new BusinessRuleViolationException("PAN lookup identity does not match the prepared credentials");
        if (lookup.isEmpty()) em.persist(com.xenomi.cms.core.v2.card.lifecycle.entity.PanLookupIdentityV2.builder()
                .panLookupIdentityId(uuidGenerator.generate()).institutionId(institutionId)
                .panAllocationId(panAllocationId).hmacKeyId(security.panHmacKeyId())
                .panHash(security.panHmac()).validFrom(now).build());
        if (immediatePhysical) em.persist(com.xenomi.cms.core.v2.card.lifecycle.entity.PhysicalSequenceReservationV2.builder()
                .reservationId(uuidGenerator.generate()).institutionId(institutionId).panAllocationId(panAllocationId)
                .cardId(cardId).operationId(draft.operationId()).reservedAt(now).build());

        CardProductionRequestV2 productionRequest = immediatePhysical
                ? createProductionRequest(institutionId, card, production, address,
                        idempotencyKey, actor, now)
                : null;
        em.flush();
        em.refresh(card);
        return card;
    }

    /** Finalizes a new-PAN replacement and then records lineage without sharing PAN or PIN state. */
    public CardV2 finalizeNewPanReplacementDraft(Draft draft, CardSecurityMaterial security,

            com.xenomi.cms.core.v2.card.lifecycle.entity.AccountRestrictionSnapshotV2 holds,
            jakarta.persistence.EntityManager em, CardV2 predecessor,
            com.xenomi.cms.core.v2.product.model.ReplacementReason reason) {
        return finalizeDraftInternal(draft, security, pin, holds, em, predecessor,
                reason, true);
    }

    private static com.xenomi.cms.core.v2.card.model.IssuanceReasonV2 issuanceReason(
            CardV2 predecessor,
            com.xenomi.cms.core.v2.product.model.ReplacementReason reason,
            boolean newPanReplacement) {
        if (predecessor == null) {
            return com.xenomi.cms.core.v2.card.model.IssuanceReasonV2.ORIGINAL;
        }
        if (!newPanReplacement) {
            return com.xenomi.cms.core.v2.card.model.IssuanceReasonV2.RENEWAL_SAME_PAN;
        }
        return switch (reason) {
            case LOST -> com.xenomi.cms.core.v2.card.model.IssuanceReasonV2.REPLACEMENT_LOST;
            case STOLEN -> com.xenomi.cms.core.v2.card.model.IssuanceReasonV2.REPLACEMENT_STOLEN;
            case COMPROMISED -> com.xenomi.cms.core.v2.card.model.IssuanceReasonV2.REPLACEMENT_COMPROMISED;
            case FRAUD -> com.xenomi.cms.core.v2.card.model.IssuanceReasonV2.REPLACEMENT_FRAUD;
            default -> throw new BusinessRuleViolationException(
                    "This finalizer supports permanent-security replacement only");
        };
    }

    @Transactional(readOnly = true)
    public CardV2Response get(UUID institutionId, UUID cardId) {
        CardV2 card = cardRepository.findByCardIdAndInstitutionId(cardId, institutionId)
                .orElseThrow(() -> new ResourceNotFoundException("Card", cardId.toString()));
        requireProduct(institutionId, card.getProductId());
        return toResponse(card,
                accountLinkRepository.findFirstByInstitutionIdAndCardIdAndPrimaryTrue(
                        institutionId, cardId).orElse(null),
                productionRequestRepository.findByInstitutionIdAndCardId(
                        institutionId, cardId).orElse(null));
    }

    @Transactional
    public CardV2Response requestPhysicalCard(UUID institutionId, UUID cardId,
            RequestPhysicalCardV2Request request, String idempotencyKey, String actorId) {
        String actor = actorId.trim();
        OffsetDateTime now = OffsetDateTime.now(ZoneOffset.UTC);
        if (request.getShippingAddressReferenceId() == null
                || request.getProductionBatchConfigurationId() == null) {
            throw new BusinessRuleViolationException(
                    "Shipping address and production batch configuration are required");
        }
        CardV2 card = cardRepository.findByCardIdAndInstitutionId(cardId, institutionId)
                .orElseThrow(() -> new ResourceNotFoundException("Card", cardId.toString()));
        if (card.getIssuanceMode() != CardIssuanceMode.VIRTUAL_THEN_PHYSICAL_LATER) {
            throw new BusinessRuleViolationException(
                    "Physical production can be requested here only for VIRTUAL_THEN_PHYSICAL_LATER cards");
        }
        if (card.getCardStatus() == CardStatusV2.CLOSED
                || card.getCardStatus() == CardStatusV2.BLOCKED) {
            throw new BusinessRuleViolationException(
                    "A blocked or closed card cannot be submitted for physical production");
        }
        if (productionRequestRepository.findByInstitutionIdAndCardId(institutionId, cardId).isPresent()) {
            throw new BusinessRuleViolationException(
                    "A physical production request already exists for this card");
        }

        AddressReferenceV2 address = requireAddressIfPresent(institutionId,
                card.getCustomerReferenceId(), request.getShippingAddressReferenceId());
        ProductionContext production = requireProductionContext(institutionId,
                card.getProductVersionId(), request.getProductionBatchConfigurationId());
        card.setShippingAddressReferenceId(address.getAddressReferenceId());
        // A virtual-first card becomes a dual-form card once physical production is requested.
        // physical_activation_status is generated from card_form/activated_at/retired_at.
        card.setCardForm(CardFormV2.VIRTUAL_PHYSICAL);
        card.setPhysicalFulfilmentStatus(com.xenomi.cms.core.v2.card.model.PhysicalFulfilmentStatusV2.REQUESTED);
        card.setUpdatedBy(actor);
        cardRepository.save(card);

        CardProductionRequestV2 productionRequest = createProductionRequest(
                institutionId, card, production, address, idempotencyKey, actor, now,
                ProductionRequestType.VIRTUAL_TO_PHYSICAL,
                ProductionRequestTrigger.CUSTOMER_REQUEST);
        CardAccountLinkV2 link = accountLinkRepository
                .findFirstByInstitutionIdAndCardIdAndPrimaryTrue(institutionId, cardId)
                .orElse(null);
        return toResponse(card, link, productionRequest);
    }

    private ProductionContext resolveProduction(UUID institutionId, UUID versionId,
            CardFormV2 cardForm, CreateCardV2Request request) {
        if (!requiresImmediatePhysical(cardForm)) return null;
        if (request.getProductionBatchConfigurationId() == null) {
            throw new BusinessRuleViolationException(
                    "productionBatchConfigurationId is required for immediate physical issuance");
        }
        return requireProductionContext(
                institutionId, versionId, request.getProductionBatchConfigurationId());
    }

    private ProductionContext requireProductionContext(UUID institutionId, UUID versionId,
            UUID productionBatchConfigurationId) {
        ProductPersonalisationPolicyV2 policy = personalisationPolicyRepository
                .findByProductVersionIdAndInstitutionId(versionId, institutionId)
                .orElseThrow(() -> new BusinessRuleViolationException(
                        "Physical production requires a personalisation policy"));
        if (policy.getPolicyStatus() != ProductPolicyStatus.ACTIVE || !policy.isPersonalisedAllowed()) {
            throw new BusinessRuleViolationException(
                    "Personalised physical issuance is not active for the product version");
        }
        if (policy.getDefaultVendorId() == null) {
            throw new BusinessRuleViolationException(
                    "Physical issuance requires a default card vendor");
        }
        CardVendorV2 vendor = vendorRepository
                .findByVendorIdAndInstitutionId(policy.getDefaultVendorId(), institutionId)
                .orElseThrow(() -> new ResourceNotFoundException(
                        "Default card vendor", policy.getDefaultVendorId().toString()));
        if (vendor.getVendorStatus() != ReferenceStatus.ACTIVE) {
            throw new BusinessRuleViolationException("Default card vendor must be ACTIVE");
        }
        PersonalisationBatchConfigurationV2 configuration = batchConfigurationRepository
                .findByBatchConfigurationIdAndInstitutionIdAndVendorId(
                        productionBatchConfigurationId, institutionId, vendor.getVendorId())
                .orElseThrow(() -> new ResourceNotFoundException(
                        "Personalisation batch configuration",
                        productionBatchConfigurationId.toString()));
        if (configuration.getConfigurationStatus() != BatchConfigurationStatus.ACTIVE) {
            throw new BusinessRuleViolationException(
                    "Personalisation batch configuration must be ACTIVE");
        }
        return new ProductionContext(vendor.getVendorId(), configuration.getBatchConfigurationId(),
                policy.getVendorFormatProfileReference());
    }

    private ResolvedProfiles resolveProfiles(UUID institutionId, UUID versionId,
            CardFormV2 cardForm, CreateCardV2Request request) {
        CardForm designForm = cardForm == CardFormV2.VIRTUAL ? CardForm.VIRTUAL : CardForm.PHYSICAL;
        ProductCardDesignAssignmentV2 design = request.getCardDesignId() == null
                ? designAssignmentRepository
                        .findFirstByInstitutionIdAndProductVersionIdAndCardFormAndDefaultDesignTrueAndAssignmentStatusOrderByEffectiveFromDesc(
                                institutionId, versionId, designForm, DesignAssignmentStatus.ACTIVE)
                        .orElse(null)
                : designAssignmentRepository
                        .findFirstByInstitutionIdAndProductVersionIdAndCardDesignIdAndCardFormAndAssignmentStatusOrderByEffectiveFromDesc(
                                institutionId, versionId, request.getCardDesignId(), designForm,
                                DesignAssignmentStatus.ACTIVE)
                        .orElseThrow(() -> new BusinessRuleViolationException(
                                "Selected card design is not actively assigned to the product version"));
        ProductCardTechnologyProfileV2 technology = request.getTechnologyProfileId() == null
                ? technologyRepository
                        .findTopByInstitutionIdAndProductVersionIdAndProfileStatusOrderByProfileVersionDesc(
                                institutionId, versionId, ProductProfileStatus.ACTIVE)
                        .orElse(null)
                : technologyRepository
                        .findByTechnologyProfileIdAndInstitutionIdAndProductVersionId(
                                request.getTechnologyProfileId(), institutionId, versionId)
                        .filter(value -> value.getProfileStatus() == ProductProfileStatus.ACTIVE)
                        .orElseThrow(() -> new BusinessRuleViolationException(
                                "Selected technology profile is not ACTIVE for the product version"));
        ProductCvmProfileV2 cvm = request.getCvmProfileId() == null
                ? cvmRepository
                        .findTopByInstitutionIdAndProductVersionIdAndProfileStatusOrderByProfileVersionDesc(
                                institutionId, versionId, ProductProfileStatus.ACTIVE)
                        .orElse(null)
                : cvmRepository
                        .findByCvmProfileIdAndInstitutionIdAndProductVersionId(
                                request.getCvmProfileId(), institutionId, versionId)
                        .filter(value -> value.getProfileStatus() == ProductProfileStatus.ACTIVE)
                        .orElseThrow(() -> new BusinessRuleViolationException(
                                "Selected CVM profile is not ACTIVE for the product version"));
        OffsetDateTime now=OffsetDateTime.now(ZoneOffset.UTC);
        if(design!=null && (design.getEffectiveFrom().isAfter(now)
                || (design.getEffectiveTo()!=null && !design.getEffectiveTo().isAfter(now))))
            throw new BusinessRuleViolationException("Card design assignment is not currently effective");
        return new ResolvedProfiles(design == null ? null : design.getCardDesignId(),
                technology == null ? null : technology.getTechnologyProfileId(),
                cvm == null ? null : cvm.getCvmProfileId());
    }

    private CardProductionRequestV2 createProductionRequest(UUID institutionId, CardV2 card,
            ProductionContext production, AddressReferenceV2 address, String idempotencyKey,
            String actor, OffsetDateTime now) {
        return createProductionRequest(institutionId, card, production, address,
                idempotencyKey, actor, now, ProductionRequestType.INITIAL_PHYSICAL,
                ProductionRequestTrigger.CARD_CREATION);
    }

    private CardProductionRequestV2 createProductionRequest(UUID institutionId, CardV2 card,
            ProductionContext production, AddressReferenceV2 address, String idempotencyKey,
            String actor, OffsetDateTime now, ProductionRequestType requestType,
            ProductionRequestTrigger requestTrigger) {
        CardProductionRequestV2 request = CardProductionRequestV2.builder()
                .productionRequestId(uuidGenerator.generate())
                .institutionId(institutionId)
                .cardId(card.getCardId())
                .vendorId(production.vendorId())
                .batchConfigurationId(production.batchConfigurationId())
                .productionProfileReference(production.productionProfileReference())
                .requestType(requestType)
                .requestTrigger(requestTrigger)
                .cardDataReady(true)
                .addressReady(address != null)
                .approvalReady(true)
                .requestedAt(now)
                .idempotencyKey(deterministicId(institutionId, "PRODUCTION:" + idempotencyKey))
                .createdBy(actor)
                .updatedAt(now)
                .build();
        ProductionRequestEligibility.stamp(request,
                productionRequestRepository.appliedSwitchRevision(institutionId, card.getCardId()), now);
        return productionRequestRepository.save(request);
    }

    private CustomerReferenceV2 requireCustomer(UUID institutionId, UUID customerId) {
        CustomerReferenceV2 customer = customerRepository
                .findByCustomerReferenceIdAndInstitutionId(customerId, institutionId)
                .orElseThrow(() -> new ResourceNotFoundException("Customer reference", customerId.toString()));
        if (customer.getCustomerStatus() != ReferenceStatus.ACTIVE) {
            throw new BusinessRuleViolationException("Customer reference must be ACTIVE");
        }
        return customer;
    }

    private ExternalAccountReferenceV2 requireAccount(UUID institutionId, UUID accountId) {
        ExternalAccountReferenceV2 account = accountRepository

                .orElseThrow(() -> new ResourceNotFoundException("Account reference", accountId.toString()));
        if (account.getAccountStatus() != ReferenceStatus.ACTIVE) {

    private AddressReferenceV2 requireAddressIfPresent(UUID institutionId, UUID customerId,
            UUID addressId) {
        if (addressId == null) return null;
        AddressReferenceV2 address = addressRepository
                .findByAddressReferenceIdAndInstitutionIdAndCustomerReferenceId(
                        addressId, institutionId, customerId)
                .orElseThrow(() -> new ResourceNotFoundException("Address reference", addressId.toString()));
        if (address.getAddressStatus() != ReferenceStatus.ACTIVE) {
            throw new BusinessRuleViolationException("Shipping address reference must be ACTIVE");
        }
        return address;
    }

    private CardProductV2 requireProduct(UUID institutionId, UUID productId) {
        CardProductV2 product = productRepository.findByProductIdAndInstitutionId(productId, institutionId)
                .orElseThrow(() -> new ResourceNotFoundException("Card product", productId.toString()));
        if (product.getProductStatus() != ProductStatus.ACTIVE) {
            throw new BusinessRuleViolationException("Card product must be ACTIVE for issuance");
        }
        return product;
    }

    private ProductVersionV2 requireActiveVersion(UUID institutionId, UUID productId,
            UUID versionId, OffsetDateTime now) {
        ProductVersionV2 version = versionRepository
                .findByProductVersionIdAndInstitutionIdAndProductId(versionId, institutionId, productId)
                .orElseThrow(() -> new ResourceNotFoundException("Product version", versionId.toString()));
        if (version.getVersionStatus() != ProductVersionStatus.ACTIVE
                || version.getEffectiveFrom().isAfter(now)
                || (version.getEffectiveTo() != null && !version.getEffectiveTo().isAfter(now))) {
            throw new BusinessRuleViolationException("Product version is not currently ACTIVE");
        }
        return version;
    }

    private InstitutionV2 requireInstitution(UUID institutionId) {
        return institutionRepository.findById(institutionId)
                .orElseThrow(() -> new ResourceNotFoundException(
                        "Institution", institutionId.toString()));
    }

    private void validateBinAndRangeEffective(UUID institutionId, BinRangeV2 range,
            OffsetDateTime now) {
        BinConfigurationV2 bin = binRepository.findByBinIdAndInstitutionId(
                        range.getBinId(), institutionId)
                .orElseThrow(() -> new ResourceNotFoundException(
                        "BIN configuration", range.getBinId().toString()));
        LocalDate today = now.toLocalDate();
        if (bin.getBinStatus() != BinStatus.ACTIVE
                || today.isBefore(bin.getEffectiveFrom())
                || (bin.getEffectiveTo() != null && today.isAfter(bin.getEffectiveTo()))) {
            throw new BusinessRuleViolationException(
                    "BIN configuration is not active and effective for new card issuance");
        }
        if (now.isBefore(range.getEffectiveFrom())
                || (range.getEffectiveTo() != null && !now.isBefore(range.getEffectiveTo()))) {
            throw new BusinessRuleViolationException(
                    "BIN range is not currently effective for new card issuance");
        }
    }

    private static CardFormV2 resolveCardForm(CardIssuanceMode mode) {
        return switch (mode) {
            case VIRTUAL_AND_PHYSICAL_SAME_DAY -> CardFormV2.VIRTUAL_PHYSICAL;
            case VIRTUAL_THEN_PHYSICAL_LATER, VIRTUAL_ONLY -> CardFormV2.VIRTUAL;
            case PHYSICAL_WITH_MULTIPLE_VIRTUAL_PANS, PHYSICAL_ONLY -> CardFormV2.PHYSICAL;
        };
    }

    private static boolean requiresImmediatePhysical(CardFormV2 form) {
        return form == CardFormV2.PHYSICAL || form == CardFormV2.VIRTUAL_PHYSICAL;
    }

    private static boolean hasVirtualCredential(CardFormV2 form) {
        return form == CardFormV2.VIRTUAL || form == CardFormV2.VIRTUAL_PHYSICAL;
    }

    private static LocalDate calculateExpiry(LocalDate issueDate, ProductVersionV2 version) {
        LocalDate expiry = issueDate.plusMonths(version.getValidityMonths());
        return version.getExpiryCalculationRule() == ExpiryCalculationRule.END_OF_EXPIRY_MONTH
                ? expiry.with(TemporalAdjusters.lastDayOfMonth()) : expiry;
    }

    private static UUID deterministicId(UUID institutionId, String value) {
        return UUID.nameUUIDFromBytes((institutionId + ":" + value)
                .getBytes(StandardCharsets.UTF_8));
    }

    private CardV2Response toResponse(CardV2 card, CardAccountLinkV2 link,
            CardProductionRequestV2 production) {
        return CardV2Response.builder()
                .cardId(card.getCardId())
                .institutionId(card.getInstitutionId())
                .customerReferenceId(card.getCustomerReferenceId())
                .accountReferenceId(link == null ? null : link.getExternalAccountReferenceId())
                .productId(card.getProductId())
                .productVersionId(card.getProductVersionId())
                .binId(card.getBinId())
                .binRangeId(card.getBinRangeId())
                .maskedPan(card.getMaskedPan())
                .panLastFour(card.getPanLastFour())
                .cardForm(card.getCardForm())
                .issuanceMode(card.getIssuanceMode())
                .physicalIssueTiming(card.getPhysicalIssueTiming())
                .cardStatus(card.getCardStatus())
                .lifecycleVersion(card.getLifecycleVersion())
                .digitalAvailable(card.isDigitalAvailable())
                .issueDate(card.getIssueDate())
                .expiryDate(card.getExpiryDate())
                .productionRequestId(production == null ? null : production.getProductionRequestId())
                .productionRequestStatus(production == null ? null : production.getRequestStatus())
                .createdAt(card.getCreatedAt())
                .build();
    }
