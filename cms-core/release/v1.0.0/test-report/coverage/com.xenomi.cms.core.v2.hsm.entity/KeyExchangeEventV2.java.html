package com.xenomi.cms.core.v2.hsm.entity;

import java.time.OffsetDateTime;
import java.util.UUID;

import com.xenomi.cms.core.v2.hsm.model.HsmKeyScope;
import com.xenomi.cms.core.v2.hsm.model.HsmKeyType;
import com.xenomi.cms.core.v2.hsm.model.KeyExchangeDirection;
import com.xenomi.cms.core.v2.hsm.model.KeyExchangeStatus;
import com.xenomi.cms.core.v2.hsm.model.KeyPurpose;

import jakarta.persistence.Column;
import jakarta.persistence.Entity;
import jakarta.persistence.EnumType;
import jakarta.persistence.Enumerated;
import jakarta.persistence.Id;
import jakarta.persistence.PrePersist;
import jakarta.persistence.Table;
import lombok.AllArgsConstructor;
import lombok.Builder;
import lombok.Data;
import lombok.NoArgsConstructor;

@Entity(name = "KeyExchangeEventV2")
@Table(name = "key_exchange_event", schema = "cms_v2")
@Data
@Builder
@NoArgsConstructor
@AllArgsConstructor
public class KeyExchangeEventV2 {

    @Id
    @Column(name = "exchange_event_id", nullable = false, updatable = false)
    private UUID exchangeEventId;
    @Column(name = "institution_id", nullable = false, updatable = false)
    private UUID institutionId;
    @Column(name = "kek_key_id", nullable = false, updatable = false)
    private UUID kekKeyId;
    @Column(name = "external_request_id", nullable = false, length = 100, updatable = false)
    private String externalRequestId;
    @org.hibernate.annotations.JdbcTypeCode(org.hibernate.type.SqlTypes.CHAR)
    @Column(name = "idempotency_key_hash", nullable = false, length = 64, updatable = false, columnDefinition = "char(64)")
    private String idempotencyKeyHash;
    @org.hibernate.annotations.JdbcTypeCode(org.hibernate.type.SqlTypes.CHAR)
    @Column(name = "request_fingerprint", nullable = false, length = 64, updatable = false, columnDefinition = "char(64)")
    private String requestFingerprint;
    @Enumerated(EnumType.STRING)
    @Column(name = "direction", nullable = false, length = 12, updatable = false)
    private KeyExchangeDirection direction;
    @Enumerated(EnumType.STRING)
    @Column(name = "working_key_type", nullable = false, length = 50, updatable = false)
    private HsmKeyType workingKeyType;
    @Enumerated(EnumType.STRING)
    @Column(name = "target_scope", nullable = false, length = 20, updatable = false)
    private HsmKeyScope targetScope;
    @Enumerated(EnumType.STRING)
    @Column(name = "key_purpose", nullable = false, length = 60, updatable = false)
    private KeyPurpose keyPurpose;
    @Column(name = "expected_kcv", nullable = false, length = 16, updatable = false)
    private String expectedKcv;
    @Column(name = "calculated_kcv", length = 16)
    private String calculatedKcv;
    @Column(name = "resulting_key_id")
    private UUID resultingKeyId;
    @Column(name = "assignment_id")
    private UUID assignmentId;
    @Enumerated(EnumType.STRING)
    @Column(name = "exchange_status", nullable = false, length = 20)
    private KeyExchangeStatus exchangeStatus;
    @Column(name = "failure_code", length = 50)
    private String failureCode;
    @Column(name = "created_at", nullable = false, updatable = false)
    private OffsetDateTime createdAt;
    @Column(name = "completed_at")
    private OffsetDateTime completedAt;
    @Column(name = "created_by", nullable = false, length = 100, updatable = false)
    private String createdBy;

    @PrePersist
    void beforeInsert() {
        if (createdAt == null) {
            createdAt = OffsetDateTime.now(java.time.ZoneOffset.UTC);
        }
    }
}
