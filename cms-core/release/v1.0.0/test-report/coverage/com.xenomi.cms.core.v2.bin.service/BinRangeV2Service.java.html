package com.xenomi.cms.core.v2.bin.service;

import java.math.BigInteger;
import java.time.OffsetDateTime;
import java.util.UUID;

import com.xenomi.cms.core.exception.ResourceNotFoundException;
import com.xenomi.cms.core.v2.bin.dto.BinRangeV2Response;
import com.xenomi.cms.core.v2.bin.dto.CreateBinRangeV2Request;
import com.xenomi.cms.core.v2.bin.dto.UpdateBinRangeV2Request;
import com.xenomi.cms.core.v2.bin.entity.BinConfigurationV2;
import com.xenomi.cms.core.v2.bin.entity.BinRangeV2;
import com.xenomi.cms.core.v2.bin.model.AllocationStrategy;
import com.xenomi.cms.core.v2.bin.model.BinRangeStatus;
import com.xenomi.cms.core.v2.bin.repository.BinConfigurationV2Repository;
import com.xenomi.cms.core.v2.bin.repository.BinRangeV2Repository;
import com.xenomi.cms.core.v2.card.repository.CardV2Repository;
import com.xenomi.cms.core.v2.common.DatabaseUuidV7Generator;
import com.xenomi.cms.core.v2.event.catalog.BinEventSink;
import com.xenomi.cms.core.v2.event.correlation.CorrelationIdResolver;
import com.xenomi.cms.core.bin.util.LuhnAlgorithm;

import org.springframework.data.domain.Page;
import org.springframework.data.domain.Pageable;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

@Service
public class BinRangeV2Service {

    private final BinRangeV2Repository repository;
    private final BinConfigurationV2Repository binRepository;
    private final DatabaseUuidV7Generator uuidGenerator;
    private final CardV2Repository cardRepository;
    private final BinEventSink eventSink;
    private final CorrelationIdResolver correlationIdResolver;

    public BinRangeV2Service(BinRangeV2Repository repository,
            BinConfigurationV2Repository binRepository, DatabaseUuidV7Generator uuidGenerator,
            BinEventSink eventSink, CorrelationIdResolver correlationIdResolver) {
        this(repository, binRepository, uuidGenerator, null, eventSink, correlationIdResolver);
    }

    @Autowired
    public BinRangeV2Service(BinRangeV2Repository repository,
            BinConfigurationV2Repository binRepository, DatabaseUuidV7Generator uuidGenerator,
            CardV2Repository cardRepository, BinEventSink eventSink,
            CorrelationIdResolver correlationIdResolver) {
        this.repository = repository;
        this.binRepository = binRepository;
        this.uuidGenerator = uuidGenerator;
        this.cardRepository = cardRepository;
        this.eventSink = eventSink;
        this.correlationIdResolver = correlationIdResolver;
    }

    @Transactional
    public BinRangeV2Response create(UUID institutionId, UUID binId,
            CreateBinRangeV2Request request, String actorId) {
        BinConfigurationV2 bin = lockBin(institutionId, binId);
        String start = request.getRangeStart().trim();
        String end = request.getRangeEnd().trim();
        BigInteger totalCapacity = validateRange(bin, start, end);
        assertNoOverlap(institutionId, binId, start, end);
        if (repository.countPlatformOverlaps(start, end, start.length()) > 0) {
            throw new IllegalArgumentException("PAN range overlaps an existing configured range");
        }

        // usable_capacity is derived from the supplied range; callers cannot override it.
        long usableCapacity = totalCapacity.longValueExact();
        validateThreshold(request.getExhaustionThreshold(), usableCapacity);
        validateDates(request.getEffectiveFrom(), request.getEffectiveTo());

        BinRangeV2 range = BinRangeV2.builder()
                .binRangeId(uuidGenerator.generate())
                .institutionId(institutionId)
                .binId(binId)
                .rangeStart(start)
                .rangeEnd(end)
                .allocationStrategy(request.getAllocationStrategy() == null
                        ? AllocationStrategy.RANDOM : request.getAllocationStrategy())
                .usableCapacity(usableCapacity)
                .exhaustionThreshold(request.getExhaustionThreshold())
                .issuedPanCount(0L)
                .nextSequenceValue(0L)
                .rangeStatus(BinRangeStatus.DRAFT)
                .effectiveFrom(request.getEffectiveFrom())
                .effectiveTo(request.getEffectiveTo())
                .createdBy(actorId.trim())
                .updatedBy(actorId.trim())
                .build();
        BinRangeV2 saved;
        try {
            saved = repository.save(range);
            // Surface concurrent platform overlaps before emitting a successful creation event.
            repository.flush();
        } catch (org.springframework.dao.DataIntegrityViolationException failure) {
            for (Throwable cause = failure; cause != null; cause = cause.getCause()) {
                if (cause instanceof org.hibernate.exception.ConstraintViolationException constraint
                        && "ex_pan_range_platform".equals(constraint.getConstraintName())) {
                    throw new IllegalArgumentException("PAN range overlaps an existing configured range");
                }
            }
            throw failure;
        }
        BinRangeV2Response response = toResponse(saved);
        eventSink.onRangeCreated(response, correlationIdResolver.current());
        return response;
    }

    @Transactional(readOnly = true)
    public BinRangeV2Response get(UUID institutionId, UUID binId, UUID rangeId) {
        return toResponse(find(institutionId, binId, rangeId));
    }

    @Transactional(readOnly = true)
    public Page<BinRangeV2Response> search(UUID institutionId, UUID binId,
            BinRangeStatus status, Pageable pageable) {
        requireBin(institutionId, binId);
        return repository.search(institutionId, binId, status, pageable).map(this::toResponse);
    }

    @Transactional
    public BinRangeV2Response update(UUID institutionId, UUID binId, UUID rangeId,
            UpdateBinRangeV2Request request, String actorId) {
        lockBin(institutionId, binId);
        BinRangeV2 range = find(institutionId, binId, rangeId);
        if (request.getAllocationStrategy() != null) {
            requireDraft(range, "allocationStrategy");
            range.setAllocationStrategy(request.getAllocationStrategy());
        }
        if (request.getExhaustionThreshold() != null) {
            range.setExhaustionThreshold(request.getExhaustionThreshold());
        }
        if (request.getEffectiveFrom() != null) range.setEffectiveFrom(request.getEffectiveFrom());
        if (request.getEffectiveTo() != null) range.setEffectiveTo(request.getEffectiveTo());
        if (request.getRangeStatus() != null && request.getRangeStatus() != range.getRangeStatus()) {
            validateTransition(range.getRangeStatus(), request.getRangeStatus());
            range.setRangeStatus(request.getRangeStatus());
        }
        validateThreshold(range.getExhaustionThreshold(), range.getUsableCapacity());
        validateDates(range.getEffectiveFrom(), range.getEffectiveTo());
        range.setUpdatedBy(actorId.trim());
        range.setUpdatedAt(OffsetDateTime.now(java.time.ZoneOffset.UTC));
        BinRangeV2Response response = toResponse(repository.save(range));
        eventSink.onRangeUpdated(response, correlationIdResolver.current());
        return response;
    }

    private BinConfigurationV2 lockBin(UUID institutionId, UUID binId) {
        return binRepository.findByTenantAndIdForUpdate(institutionId, binId)
                .orElseThrow(() -> new ResourceNotFoundException("BIN configuration", binId.toString()));
    }

    private void requireBin(UUID institutionId, UUID binId) {
        if (binRepository.findByBinIdAndInstitutionId(binId, institutionId).isEmpty()) {
            throw new ResourceNotFoundException("BIN configuration", binId.toString());
        }
    }

    private BinRangeV2 find(UUID institutionId, UUID binId, UUID rangeId) {
        return repository.findByBinRangeIdAndInstitutionIdAndBinId(rangeId, institutionId, binId)
                .orElseThrow(() -> new ResourceNotFoundException("BIN range", rangeId.toString()));
    }

    private static BigInteger validateRange(BinConfigurationV2 bin, String start, String end) {
        if (start.length() != end.length()) {
            throw new IllegalArgumentException("rangeStart and rangeEnd must have the same length");
        }
        if (start.length() < 12 || start.length() > 19) {
            throw new IllegalArgumentException("PAN range values must contain between 12 and 19 digits");
        }
        if (start.length() != bin.getPanLength()) {
            throw new IllegalArgumentException("PAN range values must match configured PAN length " + bin.getPanLength());
        }
        if (!start.startsWith(bin.getBinNumber()) || !end.startsWith(bin.getBinNumber())) {
            throw new IllegalArgumentException("rangeStart and rangeEnd must start with BIN " + bin.getBinNumber());
        }
        if (start.compareTo(end) > 0) {
            throw new IllegalArgumentException("rangeStart must be less than or equal to rangeEnd");
        }
        return luhnCapacity(start, end);
    }

    private void assertNoOverlap(UUID institutionId, UUID binId, String start, String end) {
        if (repository.countOpenOverlaps(institutionId, binId, start, end,
                start.length(), BinRangeStatus.CLOSED) > 0) {
            throw new IllegalArgumentException("PAN range overlaps an existing non-closed range for this BIN");
        }
    }

    private static BigInteger capacity(String start, String end) {
        return luhnCapacity(start, end);
    }

    /** Every complete group of ten values contains exactly one valid Luhn check digit. */
    private static BigInteger luhnCapacity(String startValue, String endValue) {
        BigInteger start = new BigInteger(startValue);
        BigInteger end = new BigInteger(endValue);
        BigInteger ten = BigInteger.TEN;
        BigInteger firstGroup = start.divide(ten);
        BigInteger lastGroup = end.divide(ten);
        if (firstGroup.equals(lastGroup)) {
            return countBoundary(start, end, startValue.length());
        }
        BigInteger firstEnd = firstGroup.multiply(ten).add(BigInteger.valueOf(9));
        BigInteger lastStart = lastGroup.multiply(ten);
        BigInteger middleGroups = lastGroup.subtract(firstGroup).subtract(BigInteger.ONE)
                .max(BigInteger.ZERO);
        return middleGroups
                .add(countBoundary(start, firstEnd, startValue.length()))
                .add(countBoundary(lastStart, end, startValue.length()));
    }

    private static BigInteger countBoundary(BigInteger start, BigInteger end, int length) {
        long count = 0;
        for (BigInteger value = start; value.compareTo(end) <= 0;
                value = value.add(BigInteger.ONE)) {
            if (LuhnAlgorithm.validate(leftPad(value.toString(), length))) count++;
        }
        return BigInteger.valueOf(count);
    }

    private static String leftPad(String value, int length) {
        return "0".repeat(Math.max(0, length - value.length())) + value;
    }

    private static long resolveUsableCapacity(Long requested, BigInteger total) {
        if (requested == null) {
            try {
                return total.longValueExact();
            } catch (ArithmeticException ex) {
                throw new IllegalArgumentException(
                        "PAN range capacity exceeds the supported numeric capacity");
            }
        }
        if (BigInteger.valueOf(requested).compareTo(total) > 0) {
            throw new IllegalArgumentException("usableCapacity cannot exceed the PAN range capacity");
        }
        return requested;
    }

    private static void validateThreshold(Long threshold, Long usableCapacity) {
        if (threshold != null && threshold > usableCapacity) {
            throw new IllegalArgumentException("exhaustionThreshold cannot exceed usableCapacity");
        }
    }

    private static void validateDates(OffsetDateTime from, OffsetDateTime to) {
        if (to != null && !to.isAfter(from)) {
            throw new IllegalArgumentException("effectiveTo must be after effectiveFrom");
        }
    }

    private static void requireDraft(BinRangeV2 range, String field) {
        if (range.getRangeStatus() != BinRangeStatus.DRAFT) {
            throw new IllegalArgumentException(field + " can be changed only while the BIN range is DRAFT");
        }
    }

    private static void validateTransition(BinRangeStatus current, BinRangeStatus target) {
        boolean allowed = switch (current) {
            case DRAFT -> target == BinRangeStatus.PENDING_APPROVAL || target == BinRangeStatus.CLOSED;
            case PENDING_APPROVAL -> target == BinRangeStatus.ACTIVE
                    || target == BinRangeStatus.SUSPENDED || target == BinRangeStatus.CLOSED;
            case ACTIVE -> target == BinRangeStatus.EXHAUSTED
                    || target == BinRangeStatus.SUSPENDED || target == BinRangeStatus.CLOSED;
            case SUSPENDED -> target == BinRangeStatus.ACTIVE || target == BinRangeStatus.CLOSED;
            case EXHAUSTED -> target == BinRangeStatus.CLOSED;
            case CLOSED -> false;
        };
        if (!allowed) {
            throw new IllegalArgumentException("Invalid BIN range status transition: " + current + " -> " + target);
        }
    }

    private BinRangeV2Response toResponse(BinRangeV2 e) {
        long totalPanCapacity = luhnCapacity(e.getRangeStart(), e.getRangeEnd()).longValueExact();
        long availablePanCapacity = Math.max(0L, totalPanCapacity - e.getIssuedPanCount());
        long active = cardRepository == null ? 0L : cardRepository.countActiveCards(e.getInstitutionId(), e.getBinRangeId());
        long blocked = cardRepository == null ? 0L : cardRepository.countBlockedOrSuspendedCards(e.getInstitutionId(), e.getBinRangeId());
        long closed = cardRepository == null ? 0L : cardRepository.countClosedCards(e.getInstitutionId(), e.getBinRangeId());
        return BinRangeV2Response.builder()
                .binRangeId(e.getBinRangeId()).institutionId(e.getInstitutionId()).binId(e.getBinId())
                .rangeStart(e.getRangeStart()).rangeEnd(e.getRangeEnd())
                .allocationStrategy(e.getAllocationStrategy()).usableCapacity(e.getUsableCapacity())
                .totalPanCapacity(totalPanCapacity)
                .exhaustionThreshold(e.getExhaustionThreshold())
                .issuedPanCount(e.getIssuedPanCount())
                .remainingPanCapacity(availablePanCapacity)
                .availablePanCapacity(availablePanCapacity)
                .activeCardCount(active).blockedSuspendedCardCount(blocked).closedCardCount(closed)
                .nextSequenceValue(e.getNextSequenceValue()).rangeStatus(e.getRangeStatus())
                .effectiveFrom(e.getEffectiveFrom()).effectiveTo(e.getEffectiveTo())
                .createdAt(e.getCreatedAt()).updatedAt(e.getUpdatedAt())
                .createdBy(e.getCreatedBy()).updatedBy(e.getUpdatedBy()).build();
    }
}
