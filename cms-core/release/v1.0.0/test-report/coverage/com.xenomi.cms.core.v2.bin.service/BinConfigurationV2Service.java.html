package com.xenomi.cms.core.v2.bin.service;

import java.time.LocalDate;
import java.time.OffsetDateTime;
import java.time.ZoneOffset;
import java.util.Locale;
import java.util.Objects;
import java.util.Set;
import java.util.UUID;

import com.xenomi.cms.core.exception.ImmutableFieldException;
import com.xenomi.cms.core.exception.ResourceAlreadyExistsException;
import com.xenomi.cms.core.exception.ResourceNotFoundException;
import com.xenomi.cms.core.exception.ResourceVersionConflictException;
import com.xenomi.cms.core.exception.UnknownRequestFieldException;
import com.xenomi.cms.core.v2.bin.dto.BinConfigurationV2Response;
import com.xenomi.cms.core.v2.bin.dto.CreateBinConfigurationV2Request;
import com.xenomi.cms.core.v2.bin.dto.UpdateBinConfigurationV2Request;
import com.xenomi.cms.core.v2.bin.entity.BinConfigurationV2;
import com.xenomi.cms.core.v2.bin.model.BinStatus;
import com.xenomi.cms.core.v2.bin.model.CardTechnology;
import com.xenomi.cms.core.v2.bin.model.CardNetwork;
import com.xenomi.cms.core.v2.bin.model.PinMethod;
import com.xenomi.cms.core.v2.bin.repository.BinConfigurationV2Repository;
import com.xenomi.cms.core.v2.common.DatabaseUuidV7Generator;
import com.xenomi.cms.core.v2.event.catalog.BinEventSink;
import com.xenomi.cms.core.v2.event.correlation.CorrelationIdResolver;
import com.xenomi.cms.core.v2.institution.repository.InstitutionV2Repository;

import org.springframework.data.domain.Page;
import org.springframework.data.domain.Pageable;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

@Service
public class BinConfigurationV2Service {

    private static final Set<String> IMMUTABLE_BIN_FIELDS = Set.of(
            "binId", "institutionId", "binNumber", "binLength", "cardNetwork", "binType",
            "cardProductClass", "createdAt", "createdBy", "updatedAt", "updatedBy");

    private final BinConfigurationV2Repository repository;
    private final InstitutionV2Repository institutionRepository;
    private final DatabaseUuidV7Generator uuidGenerator;
    private final BinEventSink eventSink;
    private final CorrelationIdResolver correlationIdResolver;

    public BinConfigurationV2Service(BinConfigurationV2Repository repository,
            InstitutionV2Repository institutionRepository, DatabaseUuidV7Generator uuidGenerator,
            BinEventSink eventSink, CorrelationIdResolver correlationIdResolver) {
        this.repository = repository;
        this.institutionRepository = institutionRepository;
        this.uuidGenerator = uuidGenerator;
        this.eventSink = eventSink;
        this.correlationIdResolver = correlationIdResolver;
    }

    @Transactional
    public BinConfigurationV2Response create(UUID institutionId,
            CreateBinConfigurationV2Request request, String actorId) {
        requireInstitution(institutionId);
        String binNumber = request.getBinNumber().trim();
        int binLength = request.getBinLength() == null ? binNumber.length() : request.getBinLength();
        if (binLength != binNumber.length()) {
            throw new IllegalArgumentException("binLength must equal the number of digits in binNumber");
        }
        if (binLength != 6 && binLength != 8) {
            throw new IllegalArgumentException("binLength must be exactly 6 or 8");
        }
        if (repository.existsByBinNumber(binNumber)) {
            throw new ResourceAlreadyExistsException("BIN configuration", binNumber);
        }

        BinConfigurationV2 entity = BinConfigurationV2.builder()
                .binId(uuidGenerator.generate())
                .institutionId(institutionId)
                .binNumber(binNumber)
                .binLength(binLength)
                .panLength(request.getPanLength())
                .cardNetwork(request.getCardNetwork())
                .binType(request.getBinType())
                .cardProductClass(upper(request.getCardProductClass()))
                .cardTechnology(request.getCardTechnology() == null
                        ? CardTechnology.DUAL_INTERFACE : request.getCardTechnology())
                .pinMethod(request.getPinMethod() == null ? PinMethod.VISA_PVV : request.getPinMethod())
                .pinLength(request.getPinLength() == null ? 4 : request.getPinLength())
                .issuingCountryCode(upperOrDefault(request.getIssuingCountryCode(), "IND"))
                .defaultCurrencyCode(upperOrDefault(request.getDefaultCurrencyCode(), "INR"))
                .defaultChipEnabled(defaultTrue(request.getDefaultChipEnabled()))
                .defaultDomesticContactEnabled(defaultTrue(request.getDefaultDomesticContactEnabled()))
                .defaultDomesticContactlessEnabled(Boolean.TRUE.equals(request.getDefaultDomesticContactlessEnabled()))
                .defaultDomesticEcommerceEnabled(Boolean.TRUE.equals(request.getDefaultDomesticEcommerceEnabled()))
                .defaultInternationalEnabled(Boolean.TRUE.equals(request.getDefaultInternationalEnabled()))
                .binStatus(request.getBinStatus() == null ? BinStatus.ACTIVE : request.getBinStatus())
                .effectiveFrom(request.getEffectiveFrom())
                .effectiveTo(request.getEffectiveTo())
                .networkProgramIdentifier(trimToNull(request.getNetworkProgramIdentifier()))
                .networkCertificationReference(trimToNull(request.getNetworkCertificationReference()))
                .track2Separator(trimToNull(request.getTrack2Separator()))
                .decimalizationTable(trimToNull(request.getDecimalizationTable()))
                .pvki(request.getPvki())
                .createdBy(actorId.trim())
                .updatedBy(actorId.trim())
                .build();
        validateDates(entity.getEffectiveFrom(), entity.getEffectiveTo());
        try {
            // Flush here so a concurrent duplicate receives the same API conflict response.
            BinConfigurationV2Response response = toResponse(repository.saveAndFlush(entity));
            eventSink.onBinCreated(response, correlationIdResolver.current());
            return response;
        } catch (org.springframework.dao.DataIntegrityViolationException failure) {
            for (Throwable cause = failure; cause != null; cause = cause.getCause()) {
                if (cause instanceof org.hibernate.exception.ConstraintViolationException constraint
                        && Set.of("uk_bin_number_platform", "uk_bin_institution_number")
                                .contains(Objects.toString(constraint.getConstraintName(), ""))) {
                    throw new ResourceAlreadyExistsException("BIN configuration", binNumber);
                }
            }
            throw failure;
        }
    }

    @Transactional(readOnly = true)
    public BinConfigurationV2Response get(UUID institutionId, UUID binId) {
        return toResponse(find(institutionId, binId));
    }

    @Transactional(readOnly = true)
    public Page<BinConfigurationV2Response> search(UUID institutionId, CardNetwork cardNetwork,
            BinStatus binStatus, Pageable pageable) {
        requireInstitution(institutionId);
        return repository.search(institutionId, cardNetwork, binStatus, pageable).map(this::toResponse);
    }

    @Transactional
    public BinConfigurationV2Response update(UUID institutionId, UUID binId,
            UpdateBinConfigurationV2Request request, String actorId) {
        rejectUnsupportedFields(request);
        BinConfigurationV2 entity = find(institutionId, binId);
        if (!Objects.equals(entity.getVersion(), request.getVersion())) {
            throw new ResourceVersionConflictException(
                    "BIN configuration", entity.getVersion(), request.getVersion());
        }

        entity.setCardTechnology(request.getCardTechnology());
        entity.setPinMethod(request.getPinMethod());
        entity.setPinLength(request.getPinLength());
        entity.setIssuingCountryCode(upper(request.getIssuingCountryCode()));
        entity.setDefaultCurrencyCode(upper(request.getDefaultCurrencyCode()));
        entity.setDefaultChipEnabled(request.getDefaultChipEnabled());
        entity.setDefaultDomesticContactEnabled(request.getDefaultDomesticContactEnabled());
        entity.setDefaultDomesticContactlessEnabled(request.getDefaultDomesticContactlessEnabled());
        entity.setDefaultDomesticEcommerceEnabled(request.getDefaultDomesticEcommerceEnabled());
        entity.setDefaultInternationalEnabled(request.getDefaultInternationalEnabled());
        entity.setBinStatus(request.getBinStatus());
        entity.setEffectiveFrom(request.getEffectiveFrom());
        entity.setEffectiveTo(request.getEffectiveTo());
        entity.setNetworkProgramIdentifier(trimToNull(request.getNetworkProgramIdentifier()));
        entity.setNetworkCertificationReference(trimToNull(request.getNetworkCertificationReference()));
        entity.setTrack2Separator(trimToNull(request.getTrack2Separator()));
        entity.setDecimalizationTable(trimToNull(request.getDecimalizationTable()));
        entity.setPvki(request.getPvki());
        validateDates(request.getEffectiveFrom(), request.getEffectiveTo());
        entity.setUpdatedBy(actorId.trim());
        entity.setUpdatedAt(OffsetDateTime.now(ZoneOffset.UTC));
        BinConfigurationV2Response response = toResponse(repository.saveAndFlush(entity));
        eventSink.onBinUpdated(response, correlationIdResolver.current());
        return response;
    }

    BinConfigurationV2 find(UUID institutionId, UUID binId) {
        return repository.findByBinIdAndInstitutionId(binId, institutionId)
                .orElseThrow(() -> new ResourceNotFoundException("BIN configuration", binId.toString()));
    }

    private void requireInstitution(UUID institutionId) {
        if (!institutionRepository.existsById(institutionId)) {
            throw new ResourceNotFoundException("Institution", institutionId.toString());
        }
    }

    private static void validateDates(LocalDate from, LocalDate to) {
        if (to != null && !to.isAfter(from)) {
            throw new IllegalArgumentException("effectiveTo must be after effectiveFrom");
        }
    }

    private BinConfigurationV2Response toResponse(BinConfigurationV2 e) {
        return BinConfigurationV2Response.builder()
                .binId(e.getBinId()).institutionId(e.getInstitutionId()).binNumber(e.getBinNumber())
                .binLength(e.getBinLength()).panLength(e.getPanLength()).cardNetwork(e.getCardNetwork()).binType(e.getBinType())
                .cardProductClass(e.getCardProductClass()).cardTechnology(e.getCardTechnology())
                .pinMethod(e.getPinMethod()).pinLength(e.getPinLength())
                .issuingCountryCode(e.getIssuingCountryCode()).defaultCurrencyCode(e.getDefaultCurrencyCode())
                .defaultChipEnabled(e.isDefaultChipEnabled())
                .defaultDomesticContactEnabled(e.isDefaultDomesticContactEnabled())
                .defaultDomesticContactlessEnabled(e.isDefaultDomesticContactlessEnabled())
                .defaultDomesticEcommerceEnabled(e.isDefaultDomesticEcommerceEnabled())
                .defaultInternationalEnabled(e.isDefaultInternationalEnabled()).binStatus(e.getBinStatus())
                .effectiveFrom(e.getEffectiveFrom()).effectiveTo(e.getEffectiveTo())
                .networkProgramIdentifier(e.getNetworkProgramIdentifier())
                .networkCertificationReference(e.getNetworkCertificationReference())
                .track2Separator(e.getTrack2Separator()).decimalizationTable(e.getDecimalizationTable())
                .pvki(e.getPvki()).createdAt(e.getCreatedAt()).updatedAt(e.getUpdatedAt())
                .createdBy(e.getCreatedBy()).updatedBy(e.getUpdatedBy()).version(e.getVersion()).build();
    }

    private static void rejectUnsupportedFields(UpdateBinConfigurationV2Request request) {
        if (request.getUnsupportedFields() == null || request.getUnsupportedFields().isEmpty()) {
            return;
        }
        var immutable = request.getUnsupportedFields().keySet().stream()
                .filter(IMMUTABLE_BIN_FIELDS::contains)
                .sorted()
                .toList();
        if (!immutable.isEmpty()) {
            throw new ImmutableFieldException(immutable);
        }
        throw new UnknownRequestFieldException(request.getUnsupportedFields().keySet().stream()
                .sorted()
                .toList());
    }

    private static boolean defaultTrue(Boolean value) { return value == null || value; }
    private static String upper(String value) { return value.trim().toUpperCase(Locale.ROOT); }
    private static String upperOrDefault(String value, String defaultValue) {
        return value == null || value.isBlank() ? defaultValue : upper(value);
    }
    private static String trimToNull(String value) {
        if (value == null) return null;
        String result = value.trim();
        return result.isEmpty() ? null : result;
    }
}
