package com.xenomi.cms.core.exception;

import org.springframework.http.HttpStatus;
import org.springframework.web.bind.annotation.ResponseStatus;

/**
 * Thrown when the requesting institution (from the {@code X-Institution-Id}
 * header) does not own the target resource.
 */
@ResponseStatus(HttpStatus.FORBIDDEN)
public class InstitutionAccessDeniedException extends RuntimeException {

    private static final String GENERIC_CLIENT_MESSAGE =
            "The requested resource was not found, or you do not have permission to access it.";

    /**
     * Server-side detail (may include resource identifiers). Never echo this to API clients
     * unchanged, to avoid leaking whether an ID exists in another tenant.
     */
    private final String logDetail;

    public InstitutionAccessDeniedException(String resourceType, Object resourceId) {
        super(GENERIC_CLIENT_MESSAGE);
        this.logDetail = "Access denied: " + resourceType + " [" + resourceId
                + "] does not belong to the requesting institution";
    }

    public InstitutionAccessDeniedException(String message) {
        super(message);
        this.logDetail = message;
    }

    public String getLogDetail() {
        return logDetail;
    }
}
