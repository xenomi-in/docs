package com.xenomi.cms.core.exception;

/** Raised when a client submits a stale optimistic-lock version. */
public class ResourceVersionConflictException extends RuntimeException {

    public ResourceVersionConflictException(String message) {
        super(message);
    }

    public ResourceVersionConflictException(String resourceType, long expectedVersion, long suppliedVersion) {
        super(resourceType + " was modified by another request; expected version "
                + expectedVersion + " but received " + suppliedVersion);
    }
}
