package com.xenomi.cms.core.exception;

import java.util.List;

/** Raised when a mutation contains fields outside its published request contract. */
public class UnknownRequestFieldException extends RuntimeException {

    private final List<String> fields;

    public UnknownRequestFieldException(List<String> fields) {
        super("Unknown request fields: " + String.join(", ", fields));
        this.fields = List.copyOf(fields);
    }

    public List<String> getFields() {
        return fields;
    }
}
