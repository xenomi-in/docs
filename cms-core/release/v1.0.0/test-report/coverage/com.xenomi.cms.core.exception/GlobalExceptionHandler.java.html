package com.xenomi.cms.core.exception;

import com.xenomi.cms.core.api.ApiErrorEnvelope;
import com.xenomi.cms.core.api.ApiErrorCode;
import com.xenomi.cms.core.api.ApiResponseHelper;
import jakarta.servlet.http.HttpServletRequest;
import lombok.extern.slf4j.Slf4j;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.http.converter.HttpMessageNotReadableException;
import org.springframework.orm.ObjectOptimisticLockingFailureException;
import org.springframework.validation.FieldError;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.MissingRequestHeaderException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;
import org.springframework.web.method.annotation.HandlerMethodValidationException;
import org.springframework.web.method.annotation.MethodArgumentTypeMismatchException;

import com.xenomi.cms.core.keymanagement.client.HsmKeyGenerationException;
import com.xenomi.cms.core.v2.card.lifecycle.client.CardLifecycleSwitchException;
import com.xenomi.cms.core.v2.card.security.CardSecurityException;

import jakarta.validation.ConstraintViolationException;

import java.util.List;
import java.util.Map;
import java.util.stream.Collectors;

/**
 * Global exception handler for the CMS Core Service.
 * Centralizes exception handling across all controllers and converts exceptions
 * to standardized API error responses using ApiErrorEnvelope.
 * Handles validation errors, resource not found, conflicts, and generic exceptions.
 *
 * @author LogicPoint Systems
 * @version 1.0.0
 * @since 2026-01-21
 */
@Slf4j
@RestControllerAdvice
public class GlobalExceptionHandler {

    @ExceptionHandler(ResourceNotFoundException.class)
    public ResponseEntity<ApiErrorEnvelope> handleResourceNotFoundException(
            ResourceNotFoundException ex, HttpServletRequest request) {
        log.error("Resource not found: {}", ex.getMessage());
        ApiErrorEnvelope body = ApiResponseHelper.error(request, ApiErrorCode.NOT_FOUND, ex.getMessage(), null);
        return ResponseEntity.status(HttpStatus.NOT_FOUND).body(body);
    }

    @ExceptionHandler(ResourceAlreadyExistsException.class)
    public ResponseEntity<ApiErrorEnvelope> handleResourceAlreadyExistsException(
            ResourceAlreadyExistsException ex, HttpServletRequest request) {
        log.error("Resource already exists: {}", ex.getMessage());
        ApiErrorEnvelope body = ApiResponseHelper.error(request, ApiErrorCode.CONFLICT, ex.getMessage(), null);
        return ResponseEntity.status(HttpStatus.CONFLICT).body(body);
    }

    @ExceptionHandler(ObjectOptimisticLockingFailureException.class)
    public ResponseEntity<ApiErrorEnvelope> handleOptimisticLockingFailure(
            ObjectOptimisticLockingFailureException ex, HttpServletRequest request) {
        ApiErrorEnvelope body = ApiResponseHelper.error(request, ApiErrorCode.CONFLICT,
                "The resource was updated by another request; reload and retry", null);
        return ResponseEntity.status(HttpStatus.CONFLICT).body(body);
    }

    @ExceptionHandler(com.xenomi.cms.core.v2.card.lifecycle.command.LifecycleCommandConflictV2.class)
    public ResponseEntity<ApiErrorEnvelope> handleLifecycleCommandConflict(
            com.xenomi.cms.core.v2.card.lifecycle.command.LifecycleCommandConflictV2 ex,
            HttpServletRequest request) {
        ApiErrorCode code = ex.code() == com.xenomi.cms.core.v2.card.lifecycle.command.LifecycleCommandConflictV2.Code.IDEMPOTENCY_CONFLICT
                ? ApiErrorCode.IDEMPOTENCY_CONFLICT : ApiErrorCode.VERSION_CONFLICT;
        return ResponseEntity.status(code.status()).body(ApiResponseHelper.error(request, code, ex.getMessage(), null));
    }

    @ExceptionHandler(org.springframework.security.access.AccessDeniedException.class)
    public ResponseEntity<ApiErrorEnvelope> handleCommandAccessDenied(
            org.springframework.security.access.AccessDeniedException ex, HttpServletRequest request) {
        return ResponseEntity.status(HttpStatus.FORBIDDEN).body(
                ApiResponseHelper.error(request, ApiErrorCode.ACCESS_DENIED, ex.getMessage(), null));
    }

    @ExceptionHandler(ResourceVersionConflictException.class)
    public ResponseEntity<ApiErrorEnvelope> handleResourceVersionConflict(
            ResourceVersionConflictException ex, HttpServletRequest request) {
        ApiErrorEnvelope body = ApiResponseHelper.error(request, ApiErrorCode.CONFLICT,
                ex.getMessage(), null);
        return ResponseEntity.status(HttpStatus.CONFLICT).body(body);
    }

    @ExceptionHandler(ImmutableFieldException.class)
    public ResponseEntity<ApiErrorEnvelope> handleImmutableField(
            ImmutableFieldException ex, HttpServletRequest request) {
        List<Object> details = ex.getFields().stream()
                .map(field -> (Object) Map.of("field", field, "message", "Field is immutable"))
                .toList();
        ApiErrorEnvelope body = ApiResponseHelper.error(request, ApiErrorCode.IMMUTABLE_FIELD,
                ex.getMessage(), details);
        return ResponseEntity.status(HttpStatus.BAD_REQUEST).body(body);
    }

    @ExceptionHandler(UnknownRequestFieldException.class)
    public ResponseEntity<ApiErrorEnvelope> handleUnknownRequestField(
            UnknownRequestFieldException ex, HttpServletRequest request) {
        List<Object> details = ex.getFields().stream()
                .map(field -> (Object) Map.of("field", field, "message", "Field is not supported"))
                .toList();
        ApiErrorEnvelope body = ApiResponseHelper.error(request, ApiErrorCode.VALIDATION_ERROR,
                ex.getMessage(), details);
        return ResponseEntity.status(HttpStatus.BAD_REQUEST).body(body);
    }

    @ExceptionHandler(InvalidOperationException.class)
    public ResponseEntity<ApiErrorEnvelope> handleInvalidOperationException(
            InvalidOperationException ex, HttpServletRequest request) {
        log.error("Invalid operation: {}", ex.getMessage());
        ApiErrorEnvelope body = ApiResponseHelper.error(request, ApiErrorCode.BAD_REQUEST, ex.getMessage(), null);
        return ResponseEntity.status(HttpStatus.BAD_REQUEST).body(body);
    }

    @ExceptionHandler(BusinessRuleViolationException.class)
    public ResponseEntity<ApiErrorEnvelope> handleBusinessRuleViolation(
            BusinessRuleViolationException ex, HttpServletRequest request) {
        log.error("Business rule violation: {}", ex.getMessage());
        ApiErrorEnvelope body = ApiResponseHelper.error(
                request, ApiErrorCode.BUSINESS_RULE_VIOLATION, ex.getMessage(), null);
        return ResponseEntity.status(HttpStatus.UNPROCESSABLE_ENTITY).body(body);
    }

    @ExceptionHandler(HsmKeyGenerationException.class)
    public ResponseEntity<ApiErrorEnvelope> handleHsmKeyGenerationException(
            HsmKeyGenerationException ex, HttpServletRequest request) {
        log.error("HSM key processing failed", ex);
        ApiErrorEnvelope body = ApiResponseHelper.error(request, ApiErrorCode.INTERNAL_ERROR,
                "HSM key processing failed. Please try again later.", null);
        return ResponseEntity.status(HttpStatus.INTERNAL_SERVER_ERROR).body(body);
    }

    @ExceptionHandler(CardLifecycleSwitchException.class)
    public ResponseEntity<ApiErrorEnvelope> handleCardLifecycleSwitchException(
            CardLifecycleSwitchException ex, HttpServletRequest request) {
        log.error("Card lifecycle switch processing failed", ex);
        ApiErrorEnvelope body = ApiResponseHelper.error(request, ApiErrorCode.INTERNAL_ERROR,
                "The card could not be hot-listed at the authorization switch. Please retry.", null);
        return ResponseEntity.status(HttpStatus.INTERNAL_SERVER_ERROR).body(body);
    }

    @ExceptionHandler(CardSecurityException.class)
    public ResponseEntity<ApiErrorEnvelope> handleCardSecurityException(
            CardSecurityException ex, HttpServletRequest request) {
        log.error("HSM-backed card security processing failed", ex);
        ApiErrorEnvelope body = ApiResponseHelper.error(request, ApiErrorCode.INTERNAL_ERROR,
                "Secure card-data processing failed. Please retry.", null);
        return ResponseEntity.status(HttpStatus.INTERNAL_SERVER_ERROR).body(body);
    }

    @ExceptionHandler(MethodArgumentNotValidException.class)
    public ResponseEntity<ApiErrorEnvelope> handleValidationException(
            MethodArgumentNotValidException ex, HttpServletRequest request) {
        // Binding exception messages include rejected values (including encrypted PIN blocks).
        log.warn("Request validation failed fields={}", ex.getBindingResult().getErrorCount());
        List<Object> details = ex.getBindingResult().getAllErrors().stream()
                .map(err -> {
                    if (err instanceof FieldError fe) {
                        String msg = fe.getDefaultMessage();
                        return Map.<String, Object>of("field", fe.getField(), "message", msg != null ? msg : "");
                    }
                    String msg = err.getDefaultMessage();
                    return Map.<String, Object>of("message", msg != null ? msg : "");
                })
                .collect(Collectors.toList());
        ApiErrorEnvelope body = ApiResponseHelper.error(request, ApiErrorCode.VALIDATION_ERROR,
                "Request validation failed", details);
        return ResponseEntity.status(HttpStatus.BAD_REQUEST).body(body);
    }

    @ExceptionHandler(HttpMessageNotReadableException.class)
    public ResponseEntity<ApiErrorEnvelope> handleUnreadableMessage(
            HttpMessageNotReadableException ex, HttpServletRequest request) {
        ApiErrorEnvelope body = ApiResponseHelper.error(request, ApiErrorCode.BAD_REQUEST,
                "Malformed or unreadable JSON request", null);
        return ResponseEntity.status(HttpStatus.BAD_REQUEST).body(body);
    }

    @ExceptionHandler(MethodArgumentTypeMismatchException.class)
    public ResponseEntity<ApiErrorEnvelope> handleTypeMismatchException(
            MethodArgumentTypeMismatchException ex, HttpServletRequest request) {
        log.error("Type mismatch error: {}", ex.getMessage());
        String msg = String.format("Invalid value '%s' for parameter '%s'", ex.getValue(), ex.getName());
        ApiErrorEnvelope body = ApiResponseHelper.error(request, ApiErrorCode.BAD_REQUEST, msg, null);
        return ResponseEntity.status(HttpStatus.BAD_REQUEST).body(body);
    }

    @ExceptionHandler(MissingRequestHeaderException.class)
    public ResponseEntity<ApiErrorEnvelope> handleMissingRequestHeader(
            MissingRequestHeaderException ex, HttpServletRequest request) {
        ApiErrorEnvelope body = ApiResponseHelper.error(request, ApiErrorCode.BAD_REQUEST,
                "Missing required header: " + ex.getHeaderName(), null);
        return ResponseEntity.status(HttpStatus.BAD_REQUEST).body(body);
    }

    @ExceptionHandler({HandlerMethodValidationException.class, ConstraintViolationException.class})
    public ResponseEntity<ApiErrorEnvelope> handleMethodValidation(
            Exception ex, HttpServletRequest request) {
        ApiErrorEnvelope body = ApiResponseHelper.error(request, ApiErrorCode.VALIDATION_ERROR,
                "Request parameter validation failed", null);
        return ResponseEntity.status(HttpStatus.BAD_REQUEST).body(body);
    }

    @ExceptionHandler(IllegalArgumentException.class)
    public ResponseEntity<ApiErrorEnvelope> handleIllegalArgumentException(
            IllegalArgumentException ex, HttpServletRequest request) {
        log.error("Invalid argument: {}", ex.getMessage());
        ApiErrorEnvelope body = ApiResponseHelper.error(request, ApiErrorCode.BAD_REQUEST, ex.getMessage(), null);
        return ResponseEntity.status(HttpStatus.BAD_REQUEST).body(body);
    }

    @ExceptionHandler(InvalidControlTokenException.class)
    public ResponseEntity<ApiErrorEnvelope> handleInvalidControlToken(
            InvalidControlTokenException ex, HttpServletRequest request) {
        log.debug("Invalid control token: {}", ex.getMessage());
        ApiErrorEnvelope body = ApiResponseHelper.error(request, ApiErrorCode.INVALID_TOKEN,
                "Invalid or expired control token.", null);
        return ResponseEntity.status(HttpStatus.BAD_REQUEST).body(body);
    }

    @ExceptionHandler(PinChangeNotAllowedException.class)
    public ResponseEntity<ApiErrorEnvelope> handlePinChangeNotAllowed(
            PinChangeNotAllowedException ex, HttpServletRequest request) {
        log.debug("PIN change not allowed: {}", ex.getMessage());
        ApiErrorEnvelope body = ApiResponseHelper.error(request, ApiErrorCode.PIN_CHANGE_NOT_ALLOWED,
                "PIN change not allowed.", null);
        return ResponseEntity.status(HttpStatus.UNPROCESSABLE_ENTITY).body(body);
    }

    @ExceptionHandler(InvalidCurrentPinException.class)
    public ResponseEntity<ApiErrorEnvelope> handleInvalidCurrentPin(
            InvalidCurrentPinException ex, HttpServletRequest request) {
        log.warn("Current PIN verification failed");
        ApiErrorEnvelope body = ApiResponseHelper.error(request, ApiErrorCode.INVALID_CURRENT_PIN,
                "Current PIN could not be verified.", null);
        return ResponseEntity.status(HttpStatus.UNPROCESSABLE_ENTITY).body(body);
    }

    @ExceptionHandler(InstitutionAccessDeniedException.class)
    public ResponseEntity<ApiErrorEnvelope> handleInstitutionAccessDenied(
            InstitutionAccessDeniedException ex, HttpServletRequest request) {
        log.warn("Institution access denied: {}", ex.getLogDetail());
        ApiErrorEnvelope body = ApiResponseHelper.error(request, ApiErrorCode.ACCESS_DENIED, ex.getMessage(), null);
        return ResponseEntity.status(HttpStatus.FORBIDDEN).body(body);
    }

    @ExceptionHandler(IllegalStateException.class)
    public ResponseEntity<ApiErrorEnvelope> handleIllegalStateException(
            IllegalStateException ex, HttpServletRequest request) {
        log.error("Illegal state: {}", ex.getMessage());
        ApiErrorEnvelope body = ApiResponseHelper.error(request, ApiErrorCode.BAD_REQUEST, ex.getMessage(), null);
        return ResponseEntity.status(HttpStatus.BAD_REQUEST).body(body);
    }

    @ExceptionHandler(org.springframework.web.HttpMediaTypeNotSupportedException.class)
    public ResponseEntity<ApiErrorEnvelope> handleUnsupportedMediaType(
            org.springframework.web.HttpMediaTypeNotSupportedException ex, HttpServletRequest request) {
        ApiErrorEnvelope body = ApiResponseHelper.error(request, ApiErrorCode.UNSUPPORTED_MEDIA_TYPE,
                "Unsupported content type", null);
        return ResponseEntity.status(HttpStatus.UNSUPPORTED_MEDIA_TYPE).body(body);
    }

    @ExceptionHandler(Exception.class)
    public ResponseEntity<ApiErrorEnvelope> handleGenericException(Exception ex, HttpServletRequest request) {
        log.error("Unexpected error occurred: ", ex);
        ApiErrorEnvelope body = ApiResponseHelper.error(request, ApiErrorCode.INTERNAL_ERROR,
                "An unexpected error occurred. Please try again later.", null);
        return ResponseEntity.status(HttpStatus.INTERNAL_SERVER_ERROR).body(body);
    }
}
