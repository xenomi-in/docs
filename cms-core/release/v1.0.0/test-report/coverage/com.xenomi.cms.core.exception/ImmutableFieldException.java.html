package com.xenomi.cms.core.exception;

import java.util.List;

/** Raised when a request attempts to change system-controlled resource identity. */
public class ImmutableFieldException extends RuntimeException {

    private final List<String> fields;

    public ImmutableFieldException(List<String> fields) {
        super("Immutable fields cannot be updated: " + String.join(", ", fields));
        this.fields = List.copyOf(fields);
    }

    public List<String> getFields() {
        return fields;
    }
}
