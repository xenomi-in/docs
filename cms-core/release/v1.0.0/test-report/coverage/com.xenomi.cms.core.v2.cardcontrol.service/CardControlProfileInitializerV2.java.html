package com.xenomi.cms.core.v2.cardcontrol.service;

import java.time.OffsetDateTime;
import java.time.ZoneOffset;
import java.util.ArrayList;
import java.util.EnumMap;
import java.util.List;
import java.util.Map;
import java.util.UUID;

import com.xenomi.cms.core.exception.BusinessRuleViolationException;
import com.xenomi.cms.core.v2.card.entity.CardV2;
import com.xenomi.cms.core.v2.cardcontrol.entity.CardControlProfileV2;
import com.xenomi.cms.core.v2.cardcontrol.entity.CardTransactionControlV2;
import com.xenomi.cms.core.v2.cardcontrol.model.ControlChannel;
import com.xenomi.cms.core.v2.cardcontrol.model.ControlGeography;
import com.xenomi.cms.core.v2.cardcontrol.model.ControlProfileStatus;
import com.xenomi.cms.core.v2.cardcontrol.repository.CardControlProfileV2Repository;
import com.xenomi.cms.core.v2.cardcontrol.repository.CardTransactionControlV2Repository;
import com.xenomi.cms.core.v2.common.DatabaseUuidV7Generator;
import com.xenomi.cms.core.v2.institution.entity.InstitutionV2;
import com.xenomi.cms.core.v2.product.entity.ProductCardLimitV2;
import com.xenomi.cms.core.v2.product.entity.ProductDefaultControlsV2;
import com.xenomi.cms.core.v2.product.model.LimitPeriodType;
import com.xenomi.cms.core.v2.product.model.ProductLimitStatus;
import com.xenomi.cms.core.v2.product.model.ProductLimitType;
import com.xenomi.cms.core.v2.product.repository.ProductCardLimitV2Repository;
import com.xenomi.cms.core.v2.product.repository.ProductDefaultControlsV2Repository;
import org.springframework.stereotype.Component;

@Component
public class CardControlProfileInitializerV2 {
    private final CardControlProfileV2Repository profileRepository;
    private final CardTransactionControlV2Repository controlRepository;
    private final ProductDefaultControlsV2Repository defaultsRepository;
    private final ProductCardLimitV2Repository productLimitRepository;
    private final DatabaseUuidV7Generator uuidGenerator;

    public CardControlProfileInitializerV2(
            CardControlProfileV2Repository profileRepository,
            CardTransactionControlV2Repository controlRepository,
            ProductDefaultControlsV2Repository defaultsRepository,
            ProductCardLimitV2Repository productLimitRepository,
            DatabaseUuidV7Generator uuidGenerator) {
        this.profileRepository = profileRepository;
        this.controlRepository = controlRepository;
        this.defaultsRepository = defaultsRepository;
        this.productLimitRepository = productLimitRepository;
        this.uuidGenerator = uuidGenerator;
    }

    public InitializedControls initialize(CardV2 card, InstitutionV2 institution, String actor) {
        ProductDefaultControlsV2 defaults = defaultsRepository
                .findByProductVersionIdAndInstitutionId(
                        card.getProductVersionId(), card.getInstitutionId())
                .orElseThrow(() -> new BusinessRuleViolationException(
                        "Product version must define default controls before card issuance"));
        Map<ProductLimitType, ProductCardLimitV2> limits = new EnumMap<>(ProductLimitType.class);
        productLimitRepository
                .findByInstitutionIdAndProductVersionIdAndPeriodTypeAndLimitStatus(
                        card.getInstitutionId(), card.getProductVersionId(),
                        LimitPeriodType.PER_TRANSACTION, ProductLimitStatus.ACTIVE)
                .forEach(value -> limits.put(value.getLimitType(), value));

        UUID profileId = uuidGenerator.generate();
        CardControlProfileV2 profile = profileRepository.save(CardControlProfileV2.builder()
                .cardControlProfileId(profileId)
                .institutionId(card.getInstitutionId())
                .cardId(card.getCardId())
                .controlVersion(1)
                .internationalEnabled(false)
                .profileStatus(ControlProfileStatus.PENDING_SWITCH)
                .updatedBy(actor)
                .build());
        OffsetDateTime now = OffsetDateTime.now(ZoneOffset.UTC);
        List<CardTransactionControlV2> controls = new ArrayList<>();
        for (ControlGeography geography : ControlGeography.values()) {
            for (ControlChannel channel : ControlChannel.values()) {
                ProductLimitType limitType = limitType(geography, channel, false);
                ProductCardLimitV2 limit = limits.get(limitType);
                boolean enabled = geography == ControlGeography.DOMESTIC
                        && (channel == ControlChannel.ATM || channel == ControlChannel.POS_CONTACT)
                        && enabled(defaults, geography, channel);
                boolean allowed = allowed(defaults, geography, channel);
                controls.add(CardTransactionControlV2.builder()
                        .cardTransactionControlId(uuidGenerator.generate())
                        .institutionId(card.getInstitutionId())
                        .cardControlProfileId(profileId)
                        .geography(geography)
                        .channel(channel)
                        .allowed(allowed)
                        .enabled(enabled)
                        .perTransactionLimitMinor(!allowed || limit == null
                                ? null : limit.getDefaultAmountMinor())
                        .currencyCode(institution.getBaseCurrency())
                        .effectiveAt(now)
                        .build());
            }
        }
        return new InitializedControls(profile, controlRepository.saveAll(controls));
    }

    public void validateOriginalConfiguration(UUID tenant, UUID version) {
        if(defaultsRepository.findByProductVersionIdAndInstitutionId(version,tenant).isEmpty())
            throw new BusinessRuleViolationException("Product version must define default controls before issuance");
    }

    public InitializedControls initializeFromSource(
            CardV2 source, CardV2 replacement, InstitutionV2 institution, String actor,
            long assignedLimitMinor) {
        CardControlProfileV2 sourceProfile = profileRepository
                .findByInstitutionIdAndCardId(source.getInstitutionId(), source.getCardId())
                .orElse(null);
        if (sourceProfile == null) return initialize(replacement, institution, actor);
        List<CardTransactionControlV2> sourceControls = controlRepository
                .findByInstitutionIdAndCardControlProfileIdOrderByGeographyAscChannelAsc(
                        source.getInstitutionId(), sourceProfile.getCardControlProfileId());
        if (sourceControls.size() != ControlGeography.values().length * ControlChannel.values().length) {
            throw new BusinessRuleViolationException(
                    "Source card does not contain all ten control dimensions");
        }
        if (assignedLimitMinor < 0 || sourceControls.stream().anyMatch(value ->
                exceeds(value.getPerTransactionLimitMinor(), assignedLimitMinor))) {
            throw new BusinessRuleViolationException(
                    "Source card controls exceed the current Accounting assigned card limit");
        }
        UUID profileId = uuidGenerator.generate();
        CardControlProfileV2 profile = profileRepository.save(CardControlProfileV2.builder()
                .cardControlProfileId(profileId)
                .institutionId(replacement.getInstitutionId())
                .cardId(replacement.getCardId())
                .controlVersion(1)
                .internationalEnabled(sourceProfile.isInternationalEnabled())
                .profileStatus(ControlProfileStatus.PENDING_SWITCH)
                .updatedBy(actor)
                .build());
        OffsetDateTime now = OffsetDateTime.now(ZoneOffset.UTC);
        List<CardTransactionControlV2> controls = sourceControls.stream()
                .map(value -> CardTransactionControlV2.builder()
                        .cardTransactionControlId(uuidGenerator.generate())
                        .institutionId(replacement.getInstitutionId())
                        .cardControlProfileId(profileId)
                        .geography(value.getGeography())
                        .channel(value.getChannel())
                        .allowed(value.isAllowed())
                        .enabled(value.isEnabled())
                        .perTransactionLimitMinor(value.getPerTransactionLimitMinor())
                        .currencyCode(institution.getBaseCurrency())
                        .effectiveAt(now)
                        .build())
                .toList();
        return new InitializedControls(profile, controlRepository.saveAll(controls));
    }

    private static boolean exceeds(Long value, long maximum) {
        return value != null && value > maximum;
    }

    static ProductLimitType limitType(
            ControlGeography geography, ControlChannel channel, boolean contactless) {
        if (contactless || channel == ControlChannel.CONTACTLESS_POS) {
            return geography == ControlGeography.DOMESTIC
                    ? ProductLimitType.DOMESTIC_CONTACTLESS
                    : ProductLimitType.INTERNATIONAL_CONTACTLESS;
        }
        return switch (geography) {
            case DOMESTIC -> switch (channel) {
                case ATM -> ProductLimitType.DOMESTIC_ATM;
                case POS_CONTACT -> ProductLimitType.DOMESTIC_POS;
                case CONTACTLESS_POS -> ProductLimitType.DOMESTIC_CONTACTLESS;
                case ECOM -> ProductLimitType.DOMESTIC_ECOMMERCE;
                case MOTO -> ProductLimitType.DOMESTIC_MOTO;
            };
            case INTERNATIONAL -> switch (channel) {
                case ATM -> ProductLimitType.INTERNATIONAL_ATM;
                case POS_CONTACT -> ProductLimitType.INTERNATIONAL_POS;
                case CONTACTLESS_POS -> ProductLimitType.INTERNATIONAL_CONTACTLESS;
                case ECOM -> ProductLimitType.INTERNATIONAL_ECOMMERCE;
                case MOTO -> ProductLimitType.INTERNATIONAL_MOTO;
            };
        };
    }

    static boolean enabled(ProductDefaultControlsV2 value,
            ControlGeography geography, ControlChannel channel) {
        return geography == ControlGeography.DOMESTIC
                ? switch (channel) {
                    case ATM -> value.isDomesticAtmEnabled();
                    case POS_CONTACT -> value.isDomesticPosEnabled();
                    case CONTACTLESS_POS -> value.isDomesticContactlessEnabled();
                    case ECOM -> value.isDomesticEcommerceEnabled();
                    case MOTO -> value.isDomesticMotoEnabled();
                }
                : switch (channel) {
                    case ATM -> value.isInternationalAtmEnabled();
                    case POS_CONTACT -> value.isInternationalPosEnabled();
                    case CONTACTLESS_POS -> value.isInternationalContactlessEnabled();
                    case ECOM -> value.isInternationalEcommerceEnabled();
                    case MOTO -> value.isInternationalMotoEnabled();
                };
    }

    static boolean allowed(ProductDefaultControlsV2 value,
            ControlGeography geography, ControlChannel channel) {
        return geography == ControlGeography.DOMESTIC
                ? switch (channel) {
                    case ATM -> value.isDomesticAtmAllowed();
                    case POS_CONTACT -> value.isDomesticPosAllowed();
                    case CONTACTLESS_POS -> value.isDomesticContactlessAllowed();
                    case ECOM -> value.isDomesticEcommerceAllowed();
                    case MOTO -> value.isDomesticMotoAllowed();
                }
                : switch (channel) {
                    case ATM -> value.isInternationalAtmAllowed();
                    case POS_CONTACT -> value.isInternationalPosAllowed();
                    case CONTACTLESS_POS -> value.isInternationalContactlessAllowed();
                    case ECOM -> value.isInternationalEcommerceAllowed();
                    case MOTO -> value.isInternationalMotoAllowed();
                };
    }

    static boolean contactlessEnabled(
            ProductDefaultControlsV2 value, ControlGeography geography) {
        return geography == ControlGeography.DOMESTIC
                ? value.isDomesticContactlessEnabled()
                : value.isInternationalContactlessEnabled();
    }

    static boolean contactlessAllowed(
            ProductDefaultControlsV2 value, ControlGeography geography) {
        return geography == ControlGeography.DOMESTIC
                ? value.isDomesticContactlessAllowed()
                : value.isInternationalContactlessAllowed();
    }

    public record InitializedControls(
            CardControlProfileV2 profile, List<CardTransactionControlV2> controls) {
    }
}
