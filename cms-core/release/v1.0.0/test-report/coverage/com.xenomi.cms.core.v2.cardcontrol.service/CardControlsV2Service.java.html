    @Transactional(readOnly = true)
    public CardControlsV2Response get(UUID institutionId, UUID cardId) {
        requireCard(institutionId, cardId);
        CardControlProfileV2 profile = profileRepository
                .findByInstitutionIdAndCardId(institutionId, cardId)
                .orElseThrow(() -> new ResourceNotFoundException(
                        "Card control profile", cardId.toString()));
        return toResponse(profile, loadControls(profile), latestSnapshot(institutionId, cardId));
    }

    @Transactional
    public CardControlsV2Response update(UUID institutionId, UUID cardId,
            UpdateCardControlsV2Request request, String idempotencyKey,
            String requestId, String actorId) {
        String actor = actorId.trim();
        String correlation = requestId == null || requestId.isBlank()
                ? uuidGenerator.generate().toString() : requestId.trim();
        InstitutionV2 institution = institutionRepository.findById(institutionId)
                .orElseThrow(() -> new ResourceNotFoundException(
                        "Institution", institutionId.toString()));
        CardV2 card = cardRepository.findForUpdate(institutionId, cardId)
                .orElseThrow(() -> new ResourceNotFoundException("Card", cardId.toString()));
        if (card.getCardStatus() == CardStatusV2.CLOSED) {
            throw new BusinessRuleViolationException("Closed-card controls cannot be modified");
        }
        if (card.getCardStatus() == CardStatusV2.BLOCKED) {
            throw new BusinessRuleViolationException(
                    "Blocked or hot-listed card controls cannot be modified");
        }

        CardControlProfileV2 profile = profileRepository.findForUpdate(institutionId, cardId)
                .orElseGet(() -> initializer.initialize(card, institution, actor).profile());
        List<CardTransactionControlV2> controls = loadControls(profile);
        if (controls.size() != ControlGeography.values().length * ControlChannel.values().length) {
            throw new IllegalStateException("Card control profile does not contain all ten dimensions");
        }
        Map<String, Object> previousState = state(profile, controls);
        validateUniqueDimensions(request.getControls());

        ProductDefaultControlsV2 defaults = defaultsRepository
                .findByProductVersionIdAndInstitutionId(card.getProductVersionId(), institutionId)
                .orElseThrow(() -> new BusinessRuleViolationException(
                        "Product version does not define default controls"));
        if (request.getInternationalEnabled() != null) {
            if (request.getInternationalEnabled() && !defaults.isInternationalEnabled()) {
                throw new BusinessRuleViolationException(
                        "Card international controls cannot be enabled when the product international scope is disabled");
            }
            profile.setInternationalEnabled(request.getInternationalEnabled());
        }
        Map<ProductLimitType, ProductCardLimitV2> productLimits = productLimitRepository
                .findByInstitutionIdAndProductVersionIdAndPeriodTypeAndLimitStatus(
                        institutionId, card.getProductVersionId(),
                        LimitPeriodType.PER_TRANSACTION, ProductLimitStatus.ACTIVE)
                .stream().collect(Collectors.toMap(ProductCardLimitV2::getLimitType, Function.identity()));
        Map<Dimension, CardTransactionControlV2> byDimension = controls.stream()
                .collect(Collectors.toMap(
                        value -> new Dimension(value.getGeography(), value.getChannel()),
                        Function.identity()));

        boolean limitChanged = request.getControls().stream().anyMatch(value ->
                value.getPerTransactionLimitMinor() != null);
        AccountingCardLimit accountingLimit = limitChanged
                ? loadAccountingLimit(institutionId, cardId, institution, correlation) : null;
        OffsetDateTime now = OffsetDateTime.now(ZoneOffset.UTC);
        for (CardControlDimensionV2Request change : request.getControls()) {
            CardTransactionControlV2 current = byDimension.get(
                    new Dimension(change.getGeography(), change.getChannel()));
            validateAndApply(change, current, profile.isInternationalEnabled(),
                    defaults, productLimits, accountingLimit, now);
        }

        int nextVersion = profile.getControlVersion() + 1;
        profile.setControlVersion(nextVersion);
        profile.setProfileStatus(ControlProfileStatus.PENDING_SWITCH);
        profile.setUpdatedAt(now);
        profile.setUpdatedBy(actor);
        controlRepository.saveAll(controls);
        profileRepository.save(profile);

        List<SwitchCardControl> switchControls = controls.stream()
                .map(this::toSwitchControl).toList();
        switchClient.apply(institutionId, cardId, nextVersion,
                profile.isInternationalEnabled(), switchControls,
                idempotencyKey, correlation);

        profile.setProfileStatus(durableSyncEnabled ? ControlProfileStatus.PENDING_SWITCH : ControlProfileStatus.ACTIVE);
        profileRepository.save(profile);
        CardLimitSnapshotV2 appliedSnapshot = accountingLimit == null
                ? latestSnapshot(institutionId, cardId)
                : snapshotRepository.save(toSnapshot(
                        institutionId, cardId, accountingLimit, correlation));
        CardControlEventV2 audit = eventRepository.save(CardControlEventV2.builder()
                .cardControlEventId(uuidGenerator.generate())
                .institutionId(institutionId)
                .cardId(cardId)
                .controlVersion(nextVersion)
                .previousState(previousState)
                .resultingState(state(profile, controls))
                .reasonCode(trim(request.getReasonCode()))
                .occurredAt(now)
                .actorReference(actor)
                .correlationId(correlationId(correlation))
                .build());
        CardControlsV2Response response = toResponse(profile, controls, appliedSnapshot, audit);
        eventSink.onUpdated(response, correlation);
        return response;
    }

    private AccountingCardLimit loadAccountingLimit(UUID institutionId, UUID cardId,
            InstitutionV2 institution, String requestId) {
        CardAccountLinkV2 link = accountLinkRepository
                .findFirstByInstitutionIdAndCardIdAndPrimaryTrue(institutionId, cardId)
                .orElseThrow(() -> new BusinessRuleViolationException(
                        "Card must have a primary account link before transaction limits can be changed"));
        AccountingCardLimit result = accountingClient.getAssignedLimit(
                institutionId, link.getExternalAccountReferenceId(), cardId, requestId);
        if (!institution.getBaseCurrency().equals(result.currencyCode())) {
            throw new BusinessRuleViolationException(
                    "Accounting card-limit currency does not match the institution base currency");
        }
        return result;
    }

    private void validateAndApply(CardControlDimensionV2Request change,
            CardTransactionControlV2 current, boolean internationalEnabled,
            ProductDefaultControlsV2 defaults,
            Map<ProductLimitType, ProductCardLimitV2> productLimits,
            AccountingCardLimit accountingLimit, OffsetDateTime now) {
        if (current == null) {
            throw new IllegalStateException("Card control dimension is missing");
        }
        if (!current.isAllowed()) {
            throw new BusinessRuleViolationException(
                    change.getGeography() + " " + change.getChannel()
                            + " is disabled and immutable because product isAllowed is false");
        }
        if (change.getGeography() == ControlGeography.INTERNATIONAL
                && Boolean.TRUE.equals(change.getEnabled()) && !internationalEnabled) {
            throw new BusinessRuleViolationException(
                    "An international control cannot be enabled while the card international scope is disabled");
        }
        if (change.getContactlessEnabled() != null || change.getContactlessLimitMinor() != null) {
            throw new BusinessRuleViolationException("Use an independent CONTACTLESS_POS dimension; nested contactless fields are no longer supported");
        }
        if (Boolean.TRUE.equals(change.getEnabled())
                && !CardControlProfileInitializerV2.allowed(defaults, change.getGeography(), change.getChannel())) {
            throw new BusinessRuleViolationException("The product does not allow this channel");
        }
        current.setEnabled(change.getEnabled());
        if (change.getPerTransactionLimitMinor() != null) {
            validateLimit(change.getPerTransactionLimitMinor(),
                    CardControlProfileInitializerV2.limitType(change.getGeography(), change.getChannel(), false),
                    productLimits, accountingLimit);
            current.setPerTransactionLimitMinor(change.getPerTransactionLimitMinor());
        }
        current.setEffectiveAt(now);

    }

    private static void validateLimit(long value, ProductLimitType type,
            Map<ProductLimitType, ProductCardLimitV2> productLimits,
            AccountingCardLimit accountingLimit) {
        if (accountingLimit == null) {
            throw new IllegalStateException("Accounting limit is required for transaction-limit changes");
        }
        ProductCardLimitV2 specific = productLimits.get(type);
        ProductCardLimitV2 overall = productLimits.get(ProductLimitType.OVERALL_CARD);
        if (specific != null && specific.getMaximumAmountMinor() != null
                && value > specific.getMaximumAmountMinor()) {
            throw new BusinessRuleViolationException(
                    "Requested limit exceeds the product maximum for " + type);
        }
        if (overall != null && overall.getMaximumAmountMinor() != null
                && value > overall.getMaximumAmountMinor()) {
            throw new BusinessRuleViolationException(
                    "Requested limit exceeds the product overall-card maximum");
        }
        if (value > accountingLimit.assignedLimitMinor()) {
            throw new BusinessRuleViolationException(
                    "Requested limit exceeds the issuer-assigned Accounting card limit");
        }
    }

    private CardLimitSnapshotV2 toSnapshot(UUID institutionId, UUID cardId,
            AccountingCardLimit value, String requestId) {
        UUID accountId = accountLinkRepository
                .findFirstByInstitutionIdAndCardIdAndPrimaryTrue(institutionId, cardId)
                .orElseThrow(() -> new IllegalStateException("Primary account link disappeared"))
                .getExternalAccountReferenceId();
        return CardLimitSnapshotV2.builder()
                .cardLimitSnapshotId(uuidGenerator.generate())
                .institutionId(institutionId)
                .cardId(cardId)
                .externalAccountReferenceId(accountId)
                .assignedLimitMinor(value.assignedLimitMinor())
                .availableAmountMinor(value.availableAmountMinor())
                .currencyCode(value.currencyCode())
                .sourceSystem(value.sourceSystem() == null ? "ACCOUNTING" : value.sourceSystem())
                .sourceVersion(value.sourceVersion() == null ? requestId : value.sourceVersion())
                .retrievedAt(value.retrievedAt())
                .expiresAt(value.expiresAt())
                .build();
    }

    private CardV2 requireCard(UUID institutionId, UUID cardId) {
        return cardRepository.findByCardIdAndInstitutionId(cardId, institutionId)
                .orElseThrow(() -> new ResourceNotFoundException("Card", cardId.toString()));
    }

    private List<CardTransactionControlV2> loadControls(CardControlProfileV2 profile) {
        return controlRepository
                .findByInstitutionIdAndCardControlProfileIdOrderByGeographyAscChannelAsc(
                        profile.getInstitutionId(), profile.getCardControlProfileId());
    }

    private CardLimitSnapshotV2 latestSnapshot(UUID institutionId, UUID cardId) {
        return snapshotRepository.findFirstByInstitutionIdAndCardIdOrderByRetrievedAtDesc(
                institutionId, cardId).orElse(null);
    }

    private CardControlsV2Response toResponse(CardControlProfileV2 profile,
            List<CardTransactionControlV2> controls, CardLimitSnapshotV2 snapshot) {
        return toResponse(profile, controls, snapshot, null);
    }

    private CardControlsV2Response toResponse(CardControlProfileV2 profile,
            List<CardTransactionControlV2> controls, CardLimitSnapshotV2 snapshot,
            CardControlEventV2 audit) {
        return CardControlsV2Response.builder()
                .cardControlProfileId(profile.getCardControlProfileId())
                .institutionId(profile.getInstitutionId())
                .cardId(profile.getCardId())
                .cardControlEventId(audit == null ? null : audit.getCardControlEventId())
                .controlVersion(profile.getControlVersion())
                .internationalEnabled(profile.isInternationalEnabled())
                .profileStatus(profile.getProfileStatus())
                .reasonCode(audit == null ? null : audit.getReasonCode())
                .updatedAt(profile.getUpdatedAt())
                .updatedBy(profile.getUpdatedBy())
                .accountingAssignedLimitMinor(snapshot == null ? null : snapshot.getAssignedLimitMinor())
                .accountingAvailableAmountMinor(snapshot == null ? null : snapshot.getAvailableAmountMinor())
                .accountingCurrencyCode(snapshot == null ? null : snapshot.getCurrencyCode())
                .accountingLimitRetrievedAt(snapshot == null ? null : snapshot.getRetrievedAt())
                .controls(controls.stream().map(this::toResponse).toList())
                .build();
    }

    private CardControlDimensionV2Response toResponse(CardTransactionControlV2 value) {
        return CardControlDimensionV2Response.builder()
                .cardTransactionControlId(value.getCardTransactionControlId())
                .geography(value.getGeography())
                .channel(value.getChannel())
                .isAllowed(value.isAllowed())
                .enabled(value.isEnabled())
                .perTransactionLimitMinor(value.getPerTransactionLimitMinor())
                .currencyCode(value.getCurrencyCode())
                .effectiveAt(value.getEffectiveAt())
                .build();
    }

    private SwitchCardControl toSwitchControl(CardTransactionControlV2 value) {
        return new SwitchCardControl(value.getGeography(), value.getChannel(), value.isAllowed(),
                value.isEnabled(), value.getPerTransactionLimitMinor(), null,
                null,
                null, value.getCurrencyCode());
    }

    private static void validateUniqueDimensions(List<CardControlDimensionV2Request> controls) {
        Set<Dimension> dimensions = new HashSet<>();
        for (CardControlDimensionV2Request value : controls) {
            if (!dimensions.add(new Dimension(value.getGeography(), value.getChannel()))) {
                throw new BusinessRuleViolationException(
                        "Each geography and channel dimension may appear only once per request");
            }
        }
    }

    private static Map<String, Object> state(CardControlProfileV2 profile,
            List<CardTransactionControlV2> controls) {
        Map<String, Object> result = new LinkedHashMap<>();
        result.put("controlVersion", profile.getControlVersion());
        result.put("profileStatus", profile.getProfileStatus().name());
        result.put("internationalEnabled", profile.isInternationalEnabled());
        List<Map<String, Object>> dimensions = new ArrayList<>();
        for (CardTransactionControlV2 value : controls) {
            Map<String, Object> dimension = new LinkedHashMap<>();
            dimension.put("geography", value.getGeography().name());
            dimension.put("channel", value.getChannel().name());
            dimension.put("isAllowed", value.isAllowed());
            dimension.put("enabled", value.isEnabled());
            dimension.put("perTransactionLimitMinor", value.getPerTransactionLimitMinor());
            dimension.put("currencyCode", value.getCurrencyCode());
            dimensions.add(dimension);
        }
        result.put("controls", dimensions);
        return result;
    }

    private static UUID correlationId(String requestId) {
        try {
            return UUID.fromString(requestId);
        } catch (IllegalArgumentException ignored) {
            return UUID.nameUUIDFromBytes(requestId.getBytes(StandardCharsets.UTF_8));
        }
    }

    private static String trim(String value) {
        return value == null || value.isBlank() ? null : value.trim();
    }

    }
