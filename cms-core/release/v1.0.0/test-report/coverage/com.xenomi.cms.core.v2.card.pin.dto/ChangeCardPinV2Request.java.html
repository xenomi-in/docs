package com.xenomi.cms.core.v2.card.pin.dto;

import io.swagger.v3.oas.annotations.media.Schema;
import jakarta.validation.constraints.NotBlank;
import jakarta.validation.constraints.Pattern;

@Schema(description = "Changes a card PIN using encrypted ISO-9564 format-0 PIN blocks. Clear PIN values are prohibited.")
public record ChangeCardPinV2Request(
        @NotBlank
        @Pattern(regexp = "^[0-9A-Fa-f]{16}$")
        @Schema(description = "Current encrypted ISO-9564 format-0 PIN block under the active BIN ZPK",
                example = "0123456789ABCDEF", accessMode = Schema.AccessMode.WRITE_ONLY)
        String currentEncryptedPinBlock,
        @NotBlank
        @Pattern(regexp = "^[0-9A-Fa-f]{16}$")
        @Schema(description = "New encrypted ISO-9564 format-0 PIN block under the active BIN ZPK",
                example = "FEDCBA9876543210", accessMode = Schema.AccessMode.WRITE_ONLY)
        String newEncryptedPinBlock) {}
