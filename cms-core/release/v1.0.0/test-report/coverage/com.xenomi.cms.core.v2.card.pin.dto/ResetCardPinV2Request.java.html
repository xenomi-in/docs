package com.xenomi.cms.core.v2.card.pin.dto;

import io.swagger.v3.oas.annotations.media.Schema;
import jakarta.validation.constraints.NotBlank;
import jakarta.validation.constraints.Pattern;
import jakarta.validation.constraints.Size;

@Schema(description = "Resets a card PIN after identity verification outside CMS. Clear PIN values are prohibited.")
public record ResetCardPinV2Request(
        @NotBlank
        @Pattern(regexp = "^[0-9A-Fa-f]{16}$")
        @Schema(description = "New encrypted ISO-9564 format-0 PIN block under the active BIN ZPK",
                example = "FEDCBA9876543210", accessMode = Schema.AccessMode.WRITE_ONLY)
        String newEncryptedPinBlock,
        @NotBlank @Size(max = 200)
        @Schema(description = "Auditable reason for resetting the PIN; do not include PIN or PAN data",
                example = "Customer authenticated through assisted-service OTP")
        String reason) {}
