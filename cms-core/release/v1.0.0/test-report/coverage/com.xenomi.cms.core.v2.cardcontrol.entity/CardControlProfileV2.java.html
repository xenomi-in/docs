package com.xenomi.cms.core.v2.cardcontrol.entity;

import java.time.OffsetDateTime;
import java.util.UUID;

import com.xenomi.cms.core.v2.cardcontrol.model.ControlProfileStatus;
import jakarta.persistence.Column;
import jakarta.persistence.Entity;
import jakarta.persistence.EnumType;
import jakarta.persistence.Enumerated;
import jakarta.persistence.Id;
import jakarta.persistence.PrePersist;
import jakarta.persistence.PreUpdate;
import jakarta.persistence.Table;
import lombok.AllArgsConstructor;
import lombok.Builder;
import lombok.Data;
import lombok.NoArgsConstructor;

@Entity(name = "CardControlProfileV2")
@Table(name = "card_control_profile", schema = "cms_v2")
@Data @Builder @NoArgsConstructor @AllArgsConstructor
public class CardControlProfileV2 {
    @Id @Column(name = "card_control_profile_id", nullable = false, updatable = false)
    private UUID cardControlProfileId;
    @Column(name = "institution_id", nullable = false, updatable = false)
    private UUID institutionId;
    @Column(name = "card_id", nullable = false, updatable = false)
    private UUID cardId;
    @Column(name = "control_version", nullable = false)
    private int controlVersion;
    @Column(name = "international_enabled", nullable = false)
    private boolean internationalEnabled;
    @Enumerated(EnumType.STRING)
    @Column(name = "profile_status", nullable = false)
    private ControlProfileStatus profileStatus;
    @Column(name = "updated_at", nullable = false)
    private OffsetDateTime updatedAt;
    @Column(name = "updated_by", nullable = false)
    private String updatedBy;

    @lombok.Builder.Default
    @Column(name = "family_revision", nullable = false)
    private long familyRevision = 1L;

    @PrePersist @PreUpdate
    void touch() {
        updatedAt = OffsetDateTime.now(java.time.ZoneOffset.UTC);
    }
}
