package com.xenomi.cms.core.api;

import org.springframework.http.HttpStatus;

/**
 * Versioned, machine-readable CMS API error catalogue.
 *
 * <p>Names are part of the public API contract and must not be renamed or
 * reused with a different meaning within an API major version.</p>
 */
public enum ApiErrorCode {
    BAD_REQUEST(HttpStatus.BAD_REQUEST),
    VALIDATION_ERROR(HttpStatus.BAD_REQUEST),
    INVALID_TOKEN(HttpStatus.BAD_REQUEST),
    IDEMPOTENCY_KEY_REQUIRED(HttpStatus.BAD_REQUEST),
    PROHIBITED_HEADER(HttpStatus.BAD_REQUEST),
    IMMUTABLE_FIELD(HttpStatus.BAD_REQUEST),

    UNAUTHENTICATED(HttpStatus.UNAUTHORIZED),
    INVALID_TOKEN_CLAIMS(HttpStatus.UNAUTHORIZED),

    ACCESS_DENIED(HttpStatus.FORBIDDEN),
    TENANT_ACCESS_DENIED(HttpStatus.FORBIDDEN),
    PLATFORM_ACCESS_REQUIRED(HttpStatus.FORBIDDEN),

    NOT_FOUND(HttpStatus.NOT_FOUND),
    CONFLICT(HttpStatus.CONFLICT),
    VERSION_CONFLICT(HttpStatus.CONFLICT),
    IDEMPOTENCY_CONFLICT(HttpStatus.CONFLICT),
    IDEMPOTENCY_IN_PROGRESS(HttpStatus.CONFLICT),

    REQUEST_BODY_TOO_LARGE(HttpStatus.PAYLOAD_TOO_LARGE),
    UNSUPPORTED_MEDIA_TYPE(HttpStatus.UNSUPPORTED_MEDIA_TYPE),

    BUSINESS_RULE_VIOLATION(HttpStatus.UNPROCESSABLE_ENTITY),
    PIN_CHANGE_NOT_ALLOWED(HttpStatus.UNPROCESSABLE_ENTITY),
    INVALID_CURRENT_PIN(HttpStatus.UNPROCESSABLE_ENTITY),

    INTERNAL_ERROR(HttpStatus.INTERNAL_SERVER_ERROR);

    private final HttpStatus status;

    ApiErrorCode(HttpStatus status) {
        this.status = status;
    }

    public HttpStatus status() {
        return status;
    }
}
