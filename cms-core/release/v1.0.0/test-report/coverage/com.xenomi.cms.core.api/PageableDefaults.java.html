package com.xenomi.cms.core.api;

import org.springframework.data.domain.PageRequest;
import org.springframework.data.domain.Pageable;
import org.springframework.data.domain.Sort;

/**
 * Utility class for creating Pageable instances with default and maximum size limits.
 * Provides methods for 0-based and 1-based pagination, with automatic size capping.
 *
 * @author LogicPoint Systems
 * @version 1.0.0
 * @since 2026-01-21
 */
public final class PageableDefaults {

    public static final int DEFAULT_PAGE = 0;
    public static final int DEFAULT_SIZE = 20;
    public static final int MAX_SIZE = 100;

    private PageableDefaults() {}

    /**
     * Build Pageable with defaults and cap. Use for list APIs.
     * Page is 0-based (Spring Data standard).
     */
    public static Pageable of(Integer page, Integer size) {
        int p = page != null && page >= 0 ? page : DEFAULT_PAGE;
        int s = size != null && size > 0 ? Math.min(size, MAX_SIZE) : DEFAULT_SIZE;
        return PageRequest.of(p, s);
    }
    
    /**
     * Build Pageable with 1-based page number (converts to 0-based internally).
     * Use when API accepts page starting from 1.
     */
    public static Pageable ofOneBased(Integer page, Integer size) {
        int p = page != null && page > 0 ? page - 1 : DEFAULT_PAGE;
        int s = size != null && size > 0 ? Math.min(size, MAX_SIZE) : DEFAULT_SIZE;
        return PageRequest.of(p, s);
    }

    /**
     * Build Pageable with sort.
     */
    public static Pageable of(Integer page, Integer size, Sort sort) {
        int p = page != null && page >= 0 ? page : DEFAULT_PAGE;
        int s = size != null && size > 0 ? Math.min(size, MAX_SIZE) : DEFAULT_SIZE;
        return PageRequest.of(p, s, sort != null ? sort : Sort.unsorted());
    }
}
