package com.xenomi.cms.core.api;

import jakarta.servlet.http.HttpServletRequest;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.web.context.request.RequestContextHolder;
import org.springframework.web.context.request.ServletRequestAttributes;

import java.time.Instant;
import java.util.List;
import java.util.Optional;
import java.util.UUID;

/**
 * Helper utility for creating standardized API response envelopes.
 * Provides methods for building success and error responses with consistent structure,
 * including request ID tracking and timestamp metadata.
 *
 * @author LogicPoint Systems
 * @version 1.0.0
 * @since 2026-01-21
 */
public final class ApiResponseHelper {

    private ApiResponseHelper() {}

    public static String getRequestId() {
        return Optional.ofNullable(RequestContextHolder.getRequestAttributes())
                .filter(ServletRequestAttributes.class::isInstance)
                .map(ServletRequestAttributes.class::cast)
                .map(ServletRequestAttributes::getRequest)
                .map(req -> (String) req.getAttribute(RequestIdFilter.REQUEST_ID_ATTRIBUTE))
                .orElse(null);
    }

    public static String getRequestId(HttpServletRequest request) {
        if (request == null) return null;
        return (String) request.getAttribute(RequestIdFilter.REQUEST_ID_ATTRIBUTE);
    }

    public static ApiResponseMeta meta() {
        return meta(getRequestId());
    }

    public static ApiResponseMeta meta(String requestId) {
        return ApiResponseMeta.builder()
                .requestId(requestId != null ? requestId : UUID.randomUUID().toString())
                .timestamp(Instant.now().toString())
                .build();
    }

    public static <T> ApiResponseEnvelope<T> success(T data) {
        return ApiResponseEnvelope.<T>builder()
                .success(true)
                .data(data)
                .meta(meta())
                .build();
    }

    public static <T> ResponseEntity<ApiResponseEnvelope<T>> ok(T data) {
        return ResponseEntity.ok(success(data));
    }

    public static <T> ResponseEntity<ApiResponseEnvelope<T>> created(T data) {
        return ResponseEntity.status(HttpStatus.CREATED).body(success(data));
    }

    public static ApiErrorEnvelope error(String code, String message) {
        return error(code, message, null);
    }

    public static ApiErrorEnvelope error(ApiErrorCode code, String message) {
        return error(code.name(), message, null);
    }

    public static ApiErrorEnvelope error(String code, String message, List<Object> details) {
        return ApiErrorEnvelope.builder()
                .success(false)
                .error(ApiErrorInfo.builder()
                        .code(code)
                        .message(message)
                        .details(details)
                        .build())
                .meta(meta())
                .build();
    }

    public static ApiErrorEnvelope error(ApiErrorCode code, String message, List<Object> details) {
        return error(code.name(), message, details);
    }

    public static ApiErrorEnvelope error(HttpServletRequest request, String code, String message, List<Object> details) {
        return ApiErrorEnvelope.builder()
                .success(false)
                .error(ApiErrorInfo.builder()
                        .code(code)
                        .message(message)
                        .details(details)
                        .build())
                .meta(meta(getRequestId(request)))
                .build();
    }

    public static ApiErrorEnvelope error(
            HttpServletRequest request, ApiErrorCode code, String message, List<Object> details) {
        return error(request, code.name(), message, details);
    }
}
