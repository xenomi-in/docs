package com.xenomi.cms.core.api;

import com.xenomi.cms.core.exception.InstitutionAccessDeniedException;

/**
 * Shared tenant ownership checks for institution-scoped resources.
 * <p>
 * {@code X-Institution-Id: 0} is reserved for platform / cross-tenant admin (e.g. dashboard):
 * listing and resolving institutions is allowed without matching a concrete tenant row.
 */
public final class TenantAccessValidator {

    private TenantAccessValidator() {
    }

    /**
     * {@code true} when the gateway sent platform scope ({@code 0}) — not a real institution PK.
     */
    public static boolean isPlatformScope(Long headerInstitutionId) {
        return headerInstitutionId != null && headerInstitutionId == 0L;
    }

    public static void requireInstitution(Long headerInstitutionId) {
        if (headerInstitutionId == null) {
            throw new IllegalStateException("Missing required header: " + TenantContext.HEADER_INSTITUTION_ID);
        }
    }

    /**
     * Requires a real tenant institution id (non-null and not platform sentinel {@code 0}).
     */
    public static void requireTenantInstitution(Long headerInstitutionId) {
        requireInstitution(headerInstitutionId);
        if (headerInstitutionId == 0L) {
            throw new IllegalStateException(
                    "Invalid tenant header: " + TenantContext.HEADER_INSTITUTION_ID
                            + " must be a positive institution id for this operation");
        }
    }

    public static void validateInstitutionAccess(Long resourceInstitutionId,
                                                 Long headerInstitutionId,
                                                 String resourceType,
                                                 Object resourceId) {
        if (isPlatformScope(headerInstitutionId)) {
            return;
        }
        requireTenantInstitution(headerInstitutionId);
        if (resourceInstitutionId == null || !headerInstitutionId.equals(resourceInstitutionId)) {
            throw new InstitutionAccessDeniedException(resourceType, resourceId);
        }
    }
}
