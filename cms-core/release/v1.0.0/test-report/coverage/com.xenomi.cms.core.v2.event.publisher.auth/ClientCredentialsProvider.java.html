package com.xenomi.cms.core.v2.event.publisher.auth;

import java.time.Duration;
import java.time.Instant;
import java.util.Optional;

import com.fasterxml.jackson.annotation.JsonIgnoreProperties;
import com.fasterxml.jackson.annotation.JsonProperty;
import com.xenomi.cms.core.v2.event.config.CmsNotificationProperties;
import org.springframework.http.MediaType;
import org.springframework.util.LinkedMultiValueMap;
import org.springframework.util.MultiValueMap;
import org.springframework.web.client.RestClient;

/**
 * OAuth2 client-credentials token source with in-memory caching. Tokens are refreshed
 * {@code refreshSkew} before expiry. Failures propagate so the publish is retried later
 * instead of being sent unauthenticated.
 */
public final class ClientCredentialsProvider implements HubCredentialsProvider {

    private final RestClient tokenClient;
    private final CmsNotificationProperties.Auth auth;
    private final Object lock = new Object();
    private volatile CachedToken cached;

    public ClientCredentialsProvider(RestClient tokenClient, CmsNotificationProperties.Auth auth) {
        if (isBlank(auth.getTokenUri()) || isBlank(auth.getClientId()) || isBlank(auth.getClientSecret())) {
            throw new IllegalStateException(
                    "cms.notification.auth.token-uri, client-id and client-secret are required for OAUTH2_CLIENT_CREDENTIALS");
        }
        this.tokenClient = tokenClient;
        this.auth = auth;
    }

    @Override
    public Optional<String> bearerToken() {
        CachedToken current = cached;
        if (current != null && current.isFresh(auth.getRefreshSkew())) {
            return Optional.of(current.value);
        }
        synchronized (lock) {
            current = cached;
            if (current != null && current.isFresh(auth.getRefreshSkew())) {
                return Optional.of(current.value);
            }
            cached = fetch();
            return Optional.of(cached.value);
        }
    }

    @Override
    public void invalidate() {
        cached = null;
    }

    @Override
    public boolean canRefreshAfterUnauthorized() {
        return true;
    }

    private CachedToken fetch() {
        MultiValueMap<String, String> form = new LinkedMultiValueMap<>();
        form.add("grant_type", "client_credentials");
        if (!isBlank(auth.getScope())) {
            form.add("scope", auth.getScope());
        }
        TokenResponse response = tokenClient.post()
                .uri(auth.getTokenUri())
                .headers(headers -> headers.setBasicAuth(auth.getClientId(), auth.getClientSecret()))
                .contentType(MediaType.APPLICATION_FORM_URLENCODED)
                .body(form)
                .retrieve()
                .body(TokenResponse.class);
        if (response == null || isBlank(response.accessToken)) {
            throw new IllegalStateException("Token endpoint returned no access_token");
        }
        long ttl = response.expiresIn == null ? 300 : response.expiresIn;
        return new CachedToken(response.accessToken, Instant.now().plusSeconds(ttl));
    }

    private static boolean isBlank(String value) {
        return value == null || value.isBlank();
    }

    @JsonIgnoreProperties(ignoreUnknown = true)
    private static final class TokenResponse {
        @JsonProperty("access_token") String accessToken;
        @JsonProperty("expires_in") Long expiresIn;
    }

    private record CachedToken(String value, Instant expiresAt) {
        boolean isFresh(Duration skew) {
            return Instant.now().plus(skew).isBefore(expiresAt);
        }
    }
}
