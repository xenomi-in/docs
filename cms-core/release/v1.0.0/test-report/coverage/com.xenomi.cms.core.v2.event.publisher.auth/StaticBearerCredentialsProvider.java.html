package com.xenomi.cms.core.v2.event.publisher.auth;

import java.util.Optional;

public final class StaticBearerCredentialsProvider implements HubCredentialsProvider {

    private final String token;

    public StaticBearerCredentialsProvider(String token) {
        if (token == null || token.isBlank()) {
            throw new IllegalStateException(
                    "cms.notification.auth.static-token is required when auth mode is STATIC_BEARER");
        }
        this.token = token.trim();
    }

    @Override
    public Optional<String> bearerToken() {
        return Optional.of(token);
    }
}
