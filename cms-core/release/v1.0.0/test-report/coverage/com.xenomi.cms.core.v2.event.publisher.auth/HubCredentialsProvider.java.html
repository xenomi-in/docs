package com.xenomi.cms.core.v2.event.publisher.auth;

import java.util.Optional;

/** Supplies the bearer credential presented to cms_notification. */
public interface HubCredentialsProvider {

    /** @return the bearer token, or empty when the hub is called unauthenticated */
    Optional<String> bearerToken();

    /** Invalidate any cached credential, e.g. after a 401 from the hub. */
    default void invalidate() {
    }

    /**
     * Whether a 401 can be resolved by obtaining a different token. Static and
     * absent credentials require an operator/configuration change instead.
     */
    default boolean canRefreshAfterUnauthorized() {
        return false;
    }
}
