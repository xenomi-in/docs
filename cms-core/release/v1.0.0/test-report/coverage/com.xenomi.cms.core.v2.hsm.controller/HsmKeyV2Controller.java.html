package com.xenomi.cms.core.v2.hsm.controller;

import java.util.UUID;

import com.xenomi.cms.core.api.ApiResponseEnvelope;
import com.xenomi.cms.core.api.ApiResponseHelper;
import com.xenomi.cms.core.api.PageableDefaults;
import com.xenomi.cms.core.api.PagedResponse;
import com.xenomi.cms.core.v2.hsm.dto.HsmKeyV2Response;
import com.xenomi.cms.core.v2.hsm.dto.GenerateHsmKeyV2Request;
import com.xenomi.cms.core.v2.hsm.dto.RegisterHsmKeyV2Request;
import com.xenomi.cms.core.v2.hsm.dto.RotateHsmKeyV2Request;
import com.xenomi.cms.core.v2.hsm.dto.UpdateHsmKeyStatusV2Request;
import com.xenomi.cms.core.v2.hsm.model.HsmKeyScope;
import com.xenomi.cms.core.v2.hsm.model.HsmKeyStatus;
import com.xenomi.cms.core.v2.hsm.model.HsmKeyType;
import com.xenomi.cms.core.v2.hsm.service.HsmKeyV2Service;

import io.swagger.v3.oas.annotations.Operation;
import io.swagger.v3.oas.annotations.tags.Tag;
import jakarta.validation.Valid;
import jakarta.validation.constraints.NotBlank;
import jakarta.validation.constraints.Size;
import org.springframework.data.domain.PageRequest;
import org.springframework.data.domain.Pageable;
import org.springframework.data.domain.Sort;
import org.springframework.http.ResponseEntity;
import org.springframework.validation.annotation.Validated;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.PutMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RequestHeader;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.bind.annotation.RestController;

@Validated
@RestController
@RequestMapping("/cms/api/v2/institutions/{institutionId}/hsm-keys")
@Tag(name = "V2 - Key Management", description = "HSM key generation, registration, exchange and assignment; key material is never returned")
public class HsmKeyV2Controller {
    private static final String ACTOR_HEADER = "X-Actor-Id";
    private final HsmKeyV2Service service;

    public HsmKeyV2Controller(HsmKeyV2Service service) { this.service = service; }

    @PostMapping
    @Operation(summary = "Generate a key using the configured HSM",
            description = "Accepts TR-31, double-length TDES, or AES according to the configured HSM provider capabilities. "
                    + "Clear key material is never returned; provider processing failures return HTTP 500.")
    public ResponseEntity<ApiResponseEnvelope<HsmKeyV2Response>> generate(
            @PathVariable UUID institutionId,
            @Valid @RequestBody GenerateHsmKeyV2Request request,
            @RequestHeader(ACTOR_HEADER) @NotBlank @Size(max = 50) String actorId) {
        return ApiResponseHelper.created(service.generate(institutionId, request, actorId));
    }

    @PostMapping("/registrations")
    @Operation(summary = "Register an existing HSM-protected key reference",
            description = "Import metadata for a key already generated or loaded in an HSM")
    public ResponseEntity<ApiResponseEnvelope<HsmKeyV2Response>> register(
            @PathVariable UUID institutionId,
            @Valid @RequestBody RegisterHsmKeyV2Request request,
            @RequestHeader(ACTOR_HEADER) @NotBlank @Size(max = 50) String actorId) {
        return ApiResponseHelper.created(service.register(institutionId, request, actorId));
    }

    @GetMapping
    @Operation(summary = "List HSM key references for an institution")
    public ResponseEntity<ApiResponseEnvelope<PagedResponse<HsmKeyV2Response>>> search(
            @PathVariable UUID institutionId,
            @RequestParam(required = false) HsmKeyType keyType,
            @RequestParam(required = false) HsmKeyScope keyScope,
            @RequestParam(required = false) HsmKeyStatus status,
            @RequestParam(required = false) UUID binId,
            @RequestParam(required = false) Integer page,
            @RequestParam(required = false) Integer limit) {
        Pageable base = PageableDefaults.ofOneBased(page, limit);
        Pageable pageable = PageRequest.of(base.getPageNumber(), base.getPageSize(),
                Sort.by(Sort.Direction.DESC, "createdAt"));
        return ApiResponseHelper.ok(PagedResponse.of(
                service.search(institutionId, keyType, keyScope, status, binId, pageable)));
    }

    @GetMapping("/{keyId}")
    @Operation(summary = "Get HSM key metadata by UUID")
    public ResponseEntity<ApiResponseEnvelope<HsmKeyV2Response>> get(
            @PathVariable UUID institutionId, @PathVariable UUID keyId) {
        return ApiResponseHelper.ok(service.get(institutionId, keyId));
    }

    @PutMapping("/{keyId}/status")
    @Operation(summary = "Move an HSM key through its lifecycle")
    public ResponseEntity<ApiResponseEnvelope<HsmKeyV2Response>> changeStatus(
            @PathVariable UUID institutionId, @PathVariable UUID keyId,
            @Valid @RequestBody UpdateHsmKeyStatusV2Request request,
            @RequestHeader(ACTOR_HEADER) @NotBlank @Size(max = 50) String actorId) {
        return ApiResponseHelper.ok(service.changeStatus(institutionId, keyId, request, actorId));
    }

    @PostMapping("/{keyId}/rotations")
    @Operation(summary = "Register a replacement version without overwriting key history")
    public ResponseEntity<ApiResponseEnvelope<HsmKeyV2Response>> rotate(
            @PathVariable UUID institutionId, @PathVariable UUID keyId,
            @Valid @RequestBody RotateHsmKeyV2Request request,
            @RequestHeader(ACTOR_HEADER) @NotBlank @Size(max = 50) String actorId) {
        return ApiResponseHelper.created(service.rotate(institutionId, keyId, request, actorId));
    }
}
