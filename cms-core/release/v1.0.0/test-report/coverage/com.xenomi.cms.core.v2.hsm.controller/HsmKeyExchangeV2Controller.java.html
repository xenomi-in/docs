package com.xenomi.cms.core.v2.hsm.controller;

import java.util.UUID;

import com.xenomi.cms.core.api.ApiResponseEnvelope;
import com.xenomi.cms.core.api.ApiResponseHelper;
import com.xenomi.cms.core.v2.hsm.dto.ImportHsmKeyV2Request;
import com.xenomi.cms.core.v2.hsm.dto.ImportHsmKeyV2Response;
import com.xenomi.cms.core.v2.hsm.service.HsmKeyExchangeV2Service;

import io.swagger.v3.oas.annotations.Operation;
import io.swagger.v3.oas.annotations.tags.Tag;
import jakarta.validation.Valid;
import jakarta.validation.constraints.NotBlank;
import jakarta.validation.constraints.Pattern;
import jakarta.validation.constraints.Size;
import org.springframework.http.ResponseEntity;
import org.springframework.validation.annotation.Validated;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RequestHeader;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RestController;

@Validated
@RestController
@RequestMapping("/cms/api/v2/institutions/{institutionId}/hsm-keys")
@Tag(name = "V2 - Key Management",
        description = "Import stakeholder working keys under an active KEK/ZMK without exposing clear key material")
public class HsmKeyExchangeV2Controller {

    private static final String ACTOR_HEADER = "X-Actor-Id";
    private static final String IDEMPOTENCY_HEADER = "Idempotency-Key";
    private final HsmKeyExchangeV2Service service;

    public HsmKeyExchangeV2Controller(HsmKeyExchangeV2Service service) {
        this.service = service;
    }

    @PostMapping("/imports")
    @Operation(summary = "Import a stakeholder working key wrapped under a KEK/ZMK",
            description = "The HSM translates the transported working key under the CMS LMK/MFK. "
                    + "The clear working key and KEK are never returned or logged. Reusing the same "
                    + "Idempotency-Key with the identical request replays the original result.")
    public ResponseEntity<ApiResponseEnvelope<ImportHsmKeyV2Response>> importKey(
            @PathVariable UUID institutionId,
            @Valid @RequestBody ImportHsmKeyV2Request request,
            @RequestHeader(IDEMPOTENCY_HEADER) @NotBlank @Size(max = 128)
            @Pattern(regexp = "[A-Za-z0-9._:-]+") String idempotencyKey,
            @RequestHeader(ACTOR_HEADER) @NotBlank @Size(max = 100) String actorId) {
        ImportHsmKeyV2Response response = service.importKey(
                institutionId, request, idempotencyKey, actorId);
        return response.isReplayed()
                ? ApiResponseHelper.ok(response)
                : ApiResponseHelper.created(response);
    }
}
