package com.xenomi.cms.core.v2.hsm.controller;

import java.util.Map;

import com.xenomi.cms.core.api.ApiResponseEnvelope;
import com.xenomi.cms.core.api.ApiResponseHelper;
import com.xenomi.cms.core.v2.hsm.service.HsmKeyV2Service;

import io.swagger.v3.oas.annotations.Operation;
import io.swagger.v3.oas.annotations.tags.Tag;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RestController;

/** Application-level health check for the configured HSM provider. */
@RestController
@RequestMapping("/cms/api/v2/hsm")
@Tag(name = "V2 - Key Management")
public class HsmProviderHealthV2Controller {

    private final HsmKeyV2Service service;

    public HsmProviderHealthV2Controller(HsmKeyV2Service service) {
        this.service = service;
    }

    @GetMapping("/provider-health")
    @Operation(
            operationId = "getHsmProviderHealth",
            summary = "Get HSM provider health",
            description = "Tests connectivity to the application-configured HSM provider. "
                    + "This is an application-level check and is not scoped to an institution.")
    public ResponseEntity<ApiResponseEnvelope<Map<String, Object>>> providerHealth() {
        boolean connected = service.testHsmConnection();
        return ApiResponseHelper.ok(Map.of(
                "connected", connected,
                "message", connected ? "HSM reachable" : "HSM unreachable"));
    }
}
