package com.xenomi.cms.core.v2.hsm.controller;

import java.util.UUID;

import com.xenomi.cms.core.api.ApiResponseEnvelope;
import com.xenomi.cms.core.api.ApiResponseHelper;
import com.xenomi.cms.core.api.PageableDefaults;
import com.xenomi.cms.core.api.PagedResponse;
import com.xenomi.cms.core.v2.hsm.dto.BinHsmKeyAssignmentV2Response;
import com.xenomi.cms.core.v2.hsm.dto.CreateBinHsmKeyAssignmentV2Request;
import com.xenomi.cms.core.v2.hsm.dto.UpdateBinHsmKeyAssignmentV2Request;
import com.xenomi.cms.core.v2.hsm.model.BinHsmKeyAssignmentStatus;
import com.xenomi.cms.core.v2.hsm.model.KeyPurpose;
import com.xenomi.cms.core.v2.hsm.service.BinHsmKeyAssignmentV2Service;

import io.swagger.v3.oas.annotations.Operation;
import io.swagger.v3.oas.annotations.tags.Tag;
import jakarta.validation.Valid;
import jakarta.validation.constraints.NotBlank;
import jakarta.validation.constraints.Size;
import org.springframework.data.domain.PageRequest;
import org.springframework.data.domain.Pageable;
import org.springframework.data.domain.Sort;
import org.springframework.http.ResponseEntity;
import org.springframework.validation.annotation.Validated;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.PutMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RequestHeader;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.bind.annotation.RestController;

@Validated
@RestController
@RequestMapping("/cms/api/v2/institutions/{institutionId}/bins/{binId}/hsm-key-assignments")
@Tag(name = "V2 - Key Management", description = "HSM key generation, registration, exchange and assignment; key material is never returned")
public class BinHsmKeyAssignmentV2Controller {
    private static final String ACTOR_HEADER = "X-Actor-Id";
    private final BinHsmKeyAssignmentV2Service service;

    public BinHsmKeyAssignmentV2Controller(BinHsmKeyAssignmentV2Service service) { this.service = service; }

    @PostMapping
    @Operation(summary = "Create a pending BIN HSM key assignment version")
    public ResponseEntity<ApiResponseEnvelope<BinHsmKeyAssignmentV2Response>> create(
            @PathVariable UUID institutionId, @PathVariable UUID binId,
            @Valid @RequestBody CreateBinHsmKeyAssignmentV2Request request,
            @RequestHeader(ACTOR_HEADER) @NotBlank @Size(max = 100) String actorId) {
        return ApiResponseHelper.created(service.create(institutionId, binId, request, actorId));
    }

    @GetMapping
    @Operation(summary = "List versioned HSM key assignments for a BIN")
    public ResponseEntity<ApiResponseEnvelope<PagedResponse<BinHsmKeyAssignmentV2Response>>> search(
            @PathVariable UUID institutionId, @PathVariable UUID binId,
            @RequestParam(required = false) KeyPurpose keyPurpose,
            @RequestParam(required = false) BinHsmKeyAssignmentStatus status,
            @RequestParam(required = false) Integer page,
            @RequestParam(required = false) Integer limit) {
        Pageable base = PageableDefaults.ofOneBased(page, limit);
        Pageable pageable = PageRequest.of(base.getPageNumber(), base.getPageSize(),
                Sort.by(Sort.Direction.DESC, "assignmentVersion"));
        return ApiResponseHelper.ok(PagedResponse.of(
                service.search(institutionId, binId, keyPurpose, status, pageable)));
    }

    @GetMapping("/{assignmentId}")
    @Operation(summary = "Get a BIN HSM key assignment")
    public ResponseEntity<ApiResponseEnvelope<BinHsmKeyAssignmentV2Response>> get(
            @PathVariable UUID institutionId, @PathVariable UUID binId,
            @PathVariable UUID assignmentId) {
        return ApiResponseHelper.ok(service.get(institutionId, binId, assignmentId));
    }

    @PutMapping("/{assignmentId}/status")
    @Operation(summary = "Move a BIN HSM key assignment through its lifecycle")
    public ResponseEntity<ApiResponseEnvelope<BinHsmKeyAssignmentV2Response>> changeStatus(
            @PathVariable UUID institutionId, @PathVariable UUID binId,
            @PathVariable UUID assignmentId,
            @Valid @RequestBody UpdateBinHsmKeyAssignmentV2Request request,
            @RequestHeader(ACTOR_HEADER) @NotBlank @Size(max = 100) String actorId) {
        return ApiResponseHelper.ok(service.changeStatus(
                institutionId, binId, assignmentId, request, actorId));
    }
}
