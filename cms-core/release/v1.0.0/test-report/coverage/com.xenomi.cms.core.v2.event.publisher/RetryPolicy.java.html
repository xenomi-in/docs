package com.xenomi.cms.core.v2.event.publisher;

import java.time.Duration;
import java.time.OffsetDateTime;
import java.util.concurrent.ThreadLocalRandom;

import com.xenomi.cms.core.v2.event.config.OutboxV2Properties;
import com.xenomi.cms.core.v2.event.entity.OutboxEventV2;
import com.xenomi.cms.core.v2.event.model.OutboxPublicationStatus;

/**
 * Applies a publish outcome to an outbox row.
 *
 * <p>Transient failures are retried with capped exponential backoff plus jitter for as long
 * as the configured retry window allows, measured from the first failure. A hub outage of
 * any length shorter than the window therefore never dead-letters an event. Permanent
 * failures (contract rejections) dead-letter immediately for operator review.
 */
public final class RetryPolicy {

    private final OutboxV2Properties.Retry retry;

    public RetryPolicy(OutboxV2Properties.Retry retry) {
        this.retry = retry;
    }

    public void apply(OutboxEventV2 event, NotificationPublishResult result, OffsetDateTime now) {
        event.setLockedUntil(null);
        event.setLockedBy(null);
        switch (result.outcome()) {
            case PUBLISHED -> {
                event.setPublicationStatus(OutboxPublicationStatus.PUBLISHED);
                event.setPublishedAt(now);
                event.setNextRetryAt(null);
                event.setLastError(null);
            }
            case PERMANENT_FAILURE -> deadLetter(event, result.error());
            case TRANSIENT_FAILURE -> transientFailure(event, result.error(), now);
        }
    }

    private void transientFailure(OutboxEventV2 event, String error, OffsetDateTime now) {
        if (event.getFirstFailedAt() == null) {
            event.setFirstFailedAt(now);
        }
        int attempt = event.getRetryCount() + 1;
        event.setRetryCount(attempt);
        event.setLastError(truncate(error));

        OffsetDateTime windowEnd = event.getFirstFailedAt().plus(retry.getWindow());
        if (!now.isBefore(windowEnd)) {
            deadLetter(event, "retry window " + retry.getWindow() + " exhausted; last error: " + error);
            return;
        }
        Duration delay = backoff(attempt);
        OffsetDateTime next = now.plus(delay);
        event.setPublicationStatus(OutboxPublicationStatus.FAILED);
        event.setNextRetryAt(next.isAfter(windowEnd) ? windowEnd : next);
    }

    Duration backoff(int attempt) {
        long initial = Math.max(1, retry.getInitialBackoff().toMillis());
        long max = Math.max(initial, retry.getMaxBackoff().toMillis());
        int shift = Math.min(Math.max(attempt - 1, 0), 30);
        long base = Math.min(max, initial << shift);
        double jitter = Math.max(0d, Math.min(1d, retry.getJitter()));
        if (jitter == 0d) {
            return Duration.ofMillis(base);
        }
        double factor = 1d + (ThreadLocalRandom.current().nextDouble(-jitter, jitter));
        return Duration.ofMillis(Math.max(initial, Math.min(max, Math.round(base * factor))));
    }

    private static void deadLetter(OutboxEventV2 event, String error) {
        event.setPublicationStatus(OutboxPublicationStatus.DEAD_LETTER);
        event.setNextRetryAt(null);
        event.setLastError(truncate(error));
    }

    private static String truncate(String value) {
        if (value == null) {
            return null;
        }
        return value.length() <= 1000 ? value : value.substring(0, 1000);
    }
}
