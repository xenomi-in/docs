package com.xenomi.cms.core.v2.event.publisher;

import java.util.LinkedHashMap;
import java.util.List;
import java.util.Locale;
import java.util.Map;
import java.util.Set;

import com.fasterxml.jackson.annotation.JsonIgnoreProperties;
import com.xenomi.cms.core.v2.event.config.CmsNotificationProperties;
import com.xenomi.cms.core.v2.event.entity.OutboxEventV2;
import com.xenomi.cms.core.v2.event.publisher.auth.HubCredentialsProvider;
import org.springframework.beans.factory.annotation.Qualifier;
import org.springframework.http.HttpHeaders;
import org.springframework.http.HttpStatusCode;
import org.springframework.http.MediaType;
import org.springframework.stereotype.Component;
import org.springframework.web.client.ResourceAccessException;
import org.springframework.web.client.RestClient;
import org.springframework.web.client.RestClientResponseException;

/**
 * Posts the stored envelope to {@code POST {base-url}{publish-path}} exactly as persisted,
 * so hub-side de-duplication on {@code eventId} sees identical bodies across retries.
 */
@Component
public class CmsNotificationClient {

    static final String IDEMPOTENCY_KEY_HEADER = "Idempotency-Key";
    static final String REQUEST_ID_HEADER = "X-Request-Id";
    static final String EVENT_TYPE_HEADER = "X-Event-Type";

    private static final Set<String> REJECTED_STATUSES = Set.of("REJECTED", "FAILED", "INVALID");
    private static final Set<String> ACCEPTED_STATUSES = Set.of("ACCEPTED", "PUBLISHED", "QUEUED", "DUPLICATE", "OK");

    private final CmsNotificationProperties properties;
    private final RestClient restClient;
    private final HubCredentialsProvider credentials;

    public CmsNotificationClient(
            CmsNotificationProperties properties,
            @Qualifier("cmsNotificationRestClient") RestClient restClient,
            HubCredentialsProvider credentials) {
        this.properties = properties;
        this.restClient = restClient;
        this.credentials = credentials;
    }

    public NotificationPublishResult publish(OutboxEventV2 event) {
        if (!properties.isEnabled()) {
            return NotificationPublishResult.localFailure("cms.notification.enabled=false");
        }
        if (properties.getBaseUrl() == null || properties.getBaseUrl().isBlank()) {
            return NotificationPublishResult.localFailure("cms.notification.base-url is blank");
        }
        String token;
        try {
            token = credentials.bearerToken().orElse(null);
        } catch (Exception exception) {
            // The token endpoint is an upstream dependency too. Configuration
            // errors fail during startup; runtime acquisition failures are retryable.
            return NotificationPublishResult.transientFailure(
                    "hub credential acquisition failed: " + message(exception));
        }
        try {
            Map<String, Object> body = publicationBody(event);
            if (event.getInstitutionId() == null
                    || !event.getInstitutionId().toString().equals(body.get("tenantId"))) {
                return NotificationPublishResult.permanentFailure("Envelope tenantId must match the outbox institution");
            }
            String correlationId = String.valueOf(body.getOrDefault("correlationId", ""));
            PublishAcknowledgement ack = restClient.post()
                    .uri(properties.getPublishPath(), Map.of("institutionId", event.getInstitutionId()))
                    .contentType(MediaType.APPLICATION_JSON)
                    .accept(MediaType.APPLICATION_JSON)
                    .headers(headers -> {
                        headers.set(IDEMPOTENCY_KEY_HEADER, event.getOutboxEventId().toString());
                        headers.set(EVENT_TYPE_HEADER, event.getEventType());
                        if (!correlationId.isBlank()) {
                            headers.set(REQUEST_ID_HEADER, correlationId);
                        }
                        if (token != null) {
                            headers.set(HttpHeaders.AUTHORIZATION, "Bearer " + token);
                        }
                    })
                    .body(body)
                    .retrieve()
                    .body(PublishAcknowledgement.class);
            return fromAcknowledgement(ack);
        } catch (RestClientResponseException exception) {
            int status = exception.getStatusCode().value();
            if (status == 401) {
                credentials.invalidate();
                if (!credentials.canRefreshAfterUnauthorized()) {
                    return NotificationPublishResult.permanentFailure(
                            status + " " + exception.getStatusText());
                }
            }
            if (status == 403) {
                return NotificationPublishResult.permanentFailure(
                        status + " " + exception.getStatusText());
            }
            return fromHttpStatus(exception.getStatusCode(),
                    status + " " + exception.getStatusText());
        } catch (ResourceAccessException exception) {
            return NotificationPublishResult.transientFailure("hub unreachable: " + message(exception));
        } catch (Exception exception) {
            return NotificationPublishResult.transientFailure(message(exception));
        }
    }

    /**
     * Lifecycle commands already wrote domain payloads to the shared outbox before
     * notification envelopes were introduced. Preserve those events and adapt them
     * at the transport boundary; newly written envelopes continue to be sent exactly
     * as stored.
     */
    static Map<String, Object> publicationBody(OutboxEventV2 event) {
        Map<String, Object> stored = event.getEventPayload();
        if (stored != null && stored.containsKey("eventType")
                && stored.containsKey("tenantId") && stored.containsKey("payload")) {
            return stored;
        }
        Map<String, Object> envelope = new LinkedHashMap<>();
        envelope.put("eventId", event.getOutboxEventId().toString());
        envelope.put("eventType", event.getEventType());
        envelope.put("tenantId", event.getInstitutionId().toString());
        envelope.put("aggregateType", event.getAggregateType());
        envelope.put("aggregateId", event.getAggregateId());
        envelope.put("customerId", null);
        envelope.put("eventSource", "CMS");
        Object storedCorrelation = stored == null ? null : stored.get("correlationId");
        envelope.put("correlationId", storedCorrelation == null
                ? event.getCorrelationId().toString() : storedCorrelation.toString());
        envelope.put("channels", List.of());
        envelope.put("priority", "MEDIUM");
        envelope.put("payload", stored == null ? Map.of() : stored);
        return envelope;
    }

    /** Maps a 2xx body to an outcome. An empty body is an acceptance; an explicit rejection is permanent. */
    static NotificationPublishResult fromAcknowledgement(PublishAcknowledgement ack) {
        if (ack == null) {
            return NotificationPublishResult.published();
        }
        if (Boolean.FALSE.equals(ack.accepted())) {
            return NotificationPublishResult.permanentFailure("hub rejected event: " + ack.message());
        }
        if (ack.status() != null) {
            String status = ack.status().trim().toUpperCase(Locale.ROOT);
            if (REJECTED_STATUSES.contains(status)) {
                return NotificationPublishResult.permanentFailure("hub status " + status + ": " + ack.message());
            }
            if (!ACCEPTED_STATUSES.contains(status)) {
                return NotificationPublishResult.transientFailure("unrecognised hub status " + status);
            }
        }
        return NotificationPublishResult.published();
    }

    static NotificationPublishResult fromHttpStatus(HttpStatusCode status, String error) {
        int code = status.value();
        if (status.is2xxSuccessful()) {
            return NotificationPublishResult.published();
        }
        if (code == 409) {
            // The hub has an in-flight claim, not a confirmed Kafka publication.
            // Retry the same event without counting contention as a hub outage.
            return NotificationPublishResult.localFailure(error);
        }
        if (code == 401 || code == 403) {
            // Our credential is wrong or expired; the event itself is fine.
            return NotificationPublishResult.localFailure(error);
        }
        if (code == 408 || code == 429 || status.is5xxServerError()) {
            return NotificationPublishResult.transientFailure(error);
        }
        if (status.is4xxClientError()) {
            return NotificationPublishResult.permanentFailure(error);
        }
        return NotificationPublishResult.transientFailure(error);
    }

    private static String message(Exception exception) {
        String value = exception.getMessage();
        if (value == null) {
            return exception.getClass().getSimpleName();
        }
        return value.length() <= 500 ? value : value.substring(0, 500);
    }

    @JsonIgnoreProperties(ignoreUnknown = true)
    record PublishAcknowledgement(Boolean accepted, String status, String message, String eventId) {
    }
}
