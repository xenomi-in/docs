package com.xenomi.cms.core.v2.event.publisher;

import java.time.OffsetDateTime;
import java.time.ZoneOffset;
import java.util.TimeZone;
import java.util.concurrent.ScheduledFuture;
import java.util.concurrent.atomic.AtomicBoolean;

import com.xenomi.cms.core.v2.event.config.EventV2Config;
import com.xenomi.cms.core.v2.event.config.OutboxV2Properties;
import com.xenomi.cms.core.v2.event.repository.OutboxEventV2Repository;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.beans.factory.annotation.Qualifier;
import org.springframework.context.SmartLifecycle;
import org.springframework.scheduling.concurrent.ThreadPoolTaskScheduler;
import org.springframework.scheduling.support.CronTrigger;
import org.springframework.stereotype.Component;
import org.springframework.transaction.PlatformTransactionManager;
import org.springframework.transaction.support.TransactionTemplate;

/**
 * Purges PUBLISHED outbox rows past their retention in bounded batches, each in its own
 * short transaction so the table is never locked for long. DEAD_LETTER rows are kept until
 * an operator resolves them; {@code card_event} remains the permanent audit record.
 */
@Component
public class OutboxRetentionJob implements SmartLifecycle {

    private static final Logger log = LoggerFactory.getLogger(OutboxRetentionJob.class);

    private final OutboxV2Properties properties;
    private final OutboxEventV2Repository repository;
    private final ThreadPoolTaskScheduler scheduler;
    private final TransactionTemplate transactionTemplate;
    private final AtomicBoolean running = new AtomicBoolean(false);
    private volatile ScheduledFuture<?> handle;

    public OutboxRetentionJob(
            OutboxV2Properties properties,
            OutboxEventV2Repository repository,
            @Qualifier(EventV2Config.OUTBOX_SCHEDULER) ThreadPoolTaskScheduler scheduler,
            PlatformTransactionManager transactionManager) {
        this.properties = properties;
        this.repository = repository;
        this.scheduler = scheduler;
        this.transactionTemplate = new TransactionTemplate(transactionManager);
    }

    @Override
    public void start() {
        OutboxV2Properties.Retention retention = properties.getRetention();
        if (!properties.isEnabled() || !retention.isEnabled()) {
            return;
        }
        if (!running.compareAndSet(false, true)) {
            return;
        }
        handle = scheduler.schedule(this::purgeSafely,
                new CronTrigger(retention.getCron(), TimeZone.getTimeZone("UTC")));
        log.info("Outbox retention scheduled cron='{}' retention={} batch={}",
                retention.getCron(), retention.getPublishedRetention(), retention.getPurgeBatchSize());
    }

    @Override
    public void stop() {
        if (running.compareAndSet(true, false) && handle != null) {
            handle.cancel(false);
        }
    }

    @Override
    public boolean isRunning() {
        return running.get();
    }

    @Override
    public int getPhase() {
        return Integer.MAX_VALUE - 512;
    }

    void purgeSafely() {
        try {
            long deleted = purge();
            if (deleted > 0) {
                log.info("Outbox retention purged {} published events", deleted);
            }
        } catch (Exception exception) {
            log.error("Outbox retention purge failed", exception);
        }
    }

    long purge() {
        OutboxV2Properties.Retention retention = properties.getRetention();
        OffsetDateTime cutoff = OffsetDateTime.now(ZoneOffset.UTC).minus(retention.getPublishedRetention());
        long total = 0;
        while (running.get()) {
            Integer deleted = transactionTemplate.execute(status ->
                    repository.purgePublishedBefore(cutoff, Math.max(1, retention.getPurgeBatchSize())));
            if (deleted == null || deleted == 0) {
                break;
            }
            total += deleted;
        }
        return total;
    }
}
