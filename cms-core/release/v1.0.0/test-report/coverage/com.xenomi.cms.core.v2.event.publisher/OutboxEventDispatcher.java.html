package com.xenomi.cms.core.v2.event.publisher;

import java.net.InetAddress;
import java.time.Duration;
import java.time.OffsetDateTime;
import java.time.ZoneOffset;
import java.util.ArrayList;
import java.util.List;
import java.util.Set;
import java.util.UUID;
import java.util.concurrent.Callable;
import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;
import java.util.concurrent.Future;
import java.util.concurrent.ScheduledFuture;
import java.util.concurrent.TimeUnit;
import java.util.concurrent.atomic.AtomicBoolean;
import java.util.concurrent.locks.ReentrantLock;

import com.xenomi.cms.core.v2.event.config.CmsNotificationProperties;
import com.xenomi.cms.core.v2.event.config.EventV2Config;
import com.xenomi.cms.core.v2.event.config.OutboxV2Properties;
import com.xenomi.cms.core.v2.event.entity.OutboxEventV2;
import com.xenomi.cms.core.v2.event.repository.OutboxEventV2Repository;
import io.github.resilience4j.circuitbreaker.CircuitBreaker;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.beans.factory.annotation.Qualifier;
import org.springframework.context.SmartLifecycle;
import org.springframework.scheduling.concurrent.ThreadPoolTaskScheduler;
import org.springframework.stereotype.Component;
import org.springframework.transaction.PlatformTransactionManager;
import org.springframework.transaction.support.TransactionTemplate;

/**
 * Polls {@code cms_v2.outbox_event} and publishes to cms_notification.
 *
 * <p>Design:
 * <ul>
 *   <li>Claims are {@code FOR UPDATE SKIP LOCKED} with a lease, so several instances share
 *       the table without double-publishing while one is mid-flight.</li>
 *   <li>Only the head event of each aggregate is claimable, so per-card / per-product order
 *       is preserved across instances. Events in a batch therefore never share an aggregate
 *       and are published in parallel.</li>
 *   <li>A cycle drains until the table is empty (bounded by {@code maxBatchesPerCycle}) and
 *       only then sleeps, so throughput is limited by the hub, not by the poll interval.</li>
 *   <li>A circuit breaker pauses claiming while the hub is down instead of burning retries.</li>
 *   <li>{@link SmartLifecycle} starts after the context is ready and waits for in-flight
 *       publishes on shutdown.</li>
 * </ul>
 */
@Component
public class OutboxEventDispatcher implements SmartLifecycle {

    private static final Logger log = LoggerFactory.getLogger(OutboxEventDispatcher.class);

    private final OutboxV2Properties outboxProperties;
    private final CmsNotificationProperties notificationProperties;
    private final OutboxEventV2Repository repository;
    private final CmsNotificationClient notificationClient;
    private final RetryPolicy retryPolicy;
    private final CircuitBreaker circuitBreaker;
    private final ThreadPoolTaskScheduler scheduler;
    private final TransactionTemplate transactionTemplate;
    private final String instanceId;

    private final AtomicBoolean running = new AtomicBoolean(false);
    private final AtomicBoolean stopRequested = new AtomicBoolean(false);
    private final ReentrantLock cycleLock = new ReentrantLock();
    private volatile ScheduledFuture<?> pollHandle;
    private volatile ExecutorService workers;

    public OutboxEventDispatcher(
            OutboxV2Properties outboxProperties,
            CmsNotificationProperties notificationProperties,
            OutboxEventV2Repository repository,
            CmsNotificationClient notificationClient,
            RetryPolicy retryPolicy,
            @Qualifier(EventV2Config.HUB_CIRCUIT_BREAKER) CircuitBreaker circuitBreaker,
            @Qualifier(EventV2Config.OUTBOX_SCHEDULER) ThreadPoolTaskScheduler scheduler,
            PlatformTransactionManager transactionManager) {
        this.outboxProperties = outboxProperties;
        this.notificationProperties = notificationProperties;
        this.repository = repository;
        this.notificationClient = notificationClient;
        this.retryPolicy = retryPolicy;
        this.circuitBreaker = circuitBreaker;
        this.scheduler = scheduler;
        this.transactionTemplate = new TransactionTemplate(transactionManager);
        this.instanceId = instanceId();
    }

    // ---------------------------------------------------------------- lifecycle

    @Override
    public void start() {
        if (!outboxProperties.isEnabled() || !notificationProperties.isEnabled()) {
            log.info("Outbox dispatcher disabled (cms.outbox.enabled={}, cms.notification.enabled={})",
                    outboxProperties.isEnabled(), notificationProperties.isEnabled());
            return;
        }
        if (!running.compareAndSet(false, true)) {
            return;
        }
        stopRequested.set(false);
        workers();
        Duration interval = outboxProperties.getPollInterval().compareTo(Duration.ofMillis(200)) < 0
                ? Duration.ofMillis(200) : outboxProperties.getPollInterval();
        pollHandle = scheduler.scheduleWithFixedDelay(this::runCycleSafely, interval);
        log.info("Outbox dispatcher {} started: interval={} batch={} workers={} lease={}",
                instanceId, interval, outboxProperties.getBatchSize(),
                outboxProperties.getWorkerThreads(), outboxProperties.getLease());
    }

    @Override
    public void stop() {
        stopRequested.set(true);
        if (!running.compareAndSet(true, false)) {
            shutdownWorkers(Duration.ZERO);
            return;
        }
        if (pollHandle != null) {
            pollHandle.cancel(false);
        }
        Duration grace = outboxProperties.getShutdownGrace();
        boolean quiesced = false;
        try {
            quiesced = cycleLock.tryLock(grace.toMillis(), TimeUnit.MILLISECONDS);
        } catch (InterruptedException interrupted) {
            Thread.currentThread().interrupt();
        } finally {
            if (quiesced) {
                cycleLock.unlock();
            }
        }
        shutdownWorkers(grace);
        log.info("Outbox dispatcher {} stopped (quiesced={})", instanceId, quiesced);
    }

    private synchronized ExecutorService workers() {
        ExecutorService pool = workers;
        if (pool == null || pool.isShutdown()) {
            pool = Executors.newFixedThreadPool(Math.max(1, outboxProperties.getWorkerThreads()), runnable -> {
                Thread thread = new Thread(runnable,
                        "cms-outbox-publish-" + UUID.randomUUID().toString().substring(0, 8));
                thread.setDaemon(true);
                return thread;
            });
            workers = pool;
        }
        return pool;
    }

    private void shutdownWorkers(Duration grace) {
        ExecutorService pool = workers;
        if (pool == null) {
            return;
        }
        pool.shutdown();
        try {
            if (!pool.awaitTermination(grace.toMillis(), TimeUnit.MILLISECONDS)) {
                pool.shutdownNow();
            }
        } catch (InterruptedException interrupted) {
            pool.shutdownNow();
            Thread.currentThread().interrupt();
        }
    }

    @Override
    public boolean isRunning() {
        return running.get();
    }

    @Override
    public int getPhase() {
        // Start after web server and data infrastructure; stop before them.
        return Integer.MAX_VALUE - 512;
    }

    // ---------------------------------------------------------------- polling

    void runCycleSafely() {
        if (!running.get() || stopRequested.get() || !cycleLock.tryLock()) {
            return;
        }
        try {
            runCycle();
        } catch (Exception exception) {
            log.error("Outbox dispatch cycle failed", exception);
        } finally {
            cycleLock.unlock();
        }
    }

    /** Drains due events until a batch comes back short or the cycle budget is used. */
    void runCycle() {
        int budget = Math.max(1, outboxProperties.getMaxBatchesPerCycle());
        for (int batch = 0; batch < budget && !stopRequested.get(); batch++) {
            if (!canBeginCycle()) {
                return;
            }
            List<OutboxEventV2> claimed = claim();
            if (claimed.isEmpty()) {
                return;
            }
            publishAll(claimed);
            if (claimed.size() < outboxProperties.getBatchSize()) {
                return;
            }
        }
    }

    /**
     * Only probes an open circuit to perform its timed transition to HALF_OPEN.
     * Actual permits are taken by {@link #publishOne(OutboxEventV2)}, one per
     * outbound HTTP call.
     */
    private boolean canBeginCycle() {
        if (circuitBreaker.getState() != CircuitBreaker.State.OPEN) {
            return true;
        }
        if (!circuitBreaker.tryAcquirePermission()) {
            log.warn("cms_notification circuit is {}; outbox dispatch paused", circuitBreaker.getState());
            return false;
        }
        circuitBreaker.releasePermission();
        return true;
    }

    List<OutboxEventV2> claim() {
        List<OutboxEventV2> claimed = transactionTemplate.execute(status -> {
            return repository.claimDueEvents(
                    instanceId,
                    Math.max(1, outboxProperties.getLease().toSeconds()),
                    outboxProperties.getBatchSize());
        });
        return claimed == null ? List.of() : claimed;
    }

    private void publishAll(List<OutboxEventV2> events) {
        List<Callable<Void>> tasks = new ArrayList<>(events.size());
        for (OutboxEventV2 event : events) {
            tasks.add(() -> {
                publishOne(event);
                return null;
            });
        }
        try {
            List<Future<Void>> futures = workers().invokeAll(tasks);
            for (Future<Void> future : futures) {
                try {
                    future.get();
                } catch (Exception exception) {
                    log.error("Outbox publish task failed", exception.getCause() == null ? exception : exception.getCause());
                }
            }
        } catch (InterruptedException interrupted) {
            Thread.currentThread().interrupt();
        }
    }

    void publishOne(OutboxEventV2 event) {
        if (!circuitBreaker.tryAcquirePermission()) {
            releaseClaim(event);
            return;
        }
        long started = System.nanoTime();
        NotificationPublishResult result;
        try {
            result = notificationClient.publish(event);
        } catch (RuntimeException exception) {
            result = NotificationPublishResult.transientFailure(
                    exception.getClass().getSimpleName() + ": " + exception.getMessage());
        }
        long elapsed = System.nanoTime() - started;
        if (result.isPublished()) {
            circuitBreaker.onSuccess(elapsed, TimeUnit.NANOSECONDS);
        } else if (result.transportFailure()) {
            circuitBreaker.onError(elapsed, TimeUnit.NANOSECONDS,
                    new HubUnavailableException(result.error()));
        } else {
            // The hub is reachable (or the failure is local), so release a
            // HALF_OPEN permit without recording a transport failure.
            circuitBreaker.onSuccess(elapsed, TimeUnit.NANOSECONDS);
        }
        record(event, result);
    }

    private void releaseClaim(OutboxEventV2 event) {
        transactionTemplate.executeWithoutResult(status -> repository.releaseOwnedClaim(
                event.getOutboxEventId(),
                instanceId,
                event.getDispatchAttempt(),
                dispatchableStatuses()));
    }

    void record(OutboxEventV2 claimedEvent, NotificationPublishResult result) {
        transactionTemplate.executeWithoutResult(status -> {
            OutboxEventV2 event = repository.findOwnedClaimForUpdate(
                    claimedEvent.getOutboxEventId(),
                    instanceId,
                    claimedEvent.getDispatchAttempt(),
                    dispatchableStatuses()).orElse(null);
            if (event == null) {
                return;
            }
            OffsetDateTime now = OffsetDateTime.now(ZoneOffset.UTC);
            retryPolicy.apply(event, result, now);
            switch (event.getPublicationStatus()) {
                case DEAD_LETTER -> log.error("Outbox event dead-lettered eventId={} eventType={} aggregateId={} error={}",
                        claimedEvent.getOutboxEventId(), event.getEventType(), event.getAggregateId(), result.error());
                case FAILED -> log.warn("Outbox event retry scheduled eventId={} eventType={} attempt={} nextRetryAt={} error={}",
                        claimedEvent.getOutboxEventId(), event.getEventType(), event.getRetryCount(), event.getNextRetryAt(), result.error());
                default -> log.debug("Outbox event published eventId={} eventType={}",
                        claimedEvent.getOutboxEventId(), event.getEventType());
            }
        });
    }

    private static Set<com.xenomi.cms.core.v2.event.model.OutboxPublicationStatus> dispatchableStatuses() {
        return Set.of(
                com.xenomi.cms.core.v2.event.model.OutboxPublicationStatus.PENDING,
                com.xenomi.cms.core.v2.event.model.OutboxPublicationStatus.FAILED);
    }

    private static String instanceId() {
        String host;
        try {
            host = InetAddress.getLocalHost().getHostName();
        } catch (Exception ignored) {
            host = "unknown-host";
        }
        String suffix = "/" + ProcessHandle.current().pid() + "/"
                + UUID.randomUUID().toString().substring(0, 8);
        int maximumHostLength = 120 - suffix.length();
        return host.length() <= maximumHostLength
                ? host + suffix
                : host.substring(0, maximumHostLength) + suffix;
    }

    static final class HubUnavailableException extends RuntimeException {
        HubUnavailableException(String message) {
            super(message, null, false, false);
        }
    }
}
