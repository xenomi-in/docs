package com.xenomi.cms.core.v2.event.publisher;

/**
 * Outcome of one publish attempt.
 *
 * <ul>
 *   <li>{@code PUBLISHED}: hub accepted the event (or already had it).</li>
 *   <li>{@code TRANSIENT_FAILURE}: hub unavailable, throttled, or timed out; retry later.</li>
 *   <li>{@code PERMANENT_FAILURE}: hub rejected the event contract; dead-letter for operator review.</li>
 * </ul>
 */
public record NotificationPublishResult(Outcome outcome, String error, boolean transportFailure) {

    public enum Outcome {
        PUBLISHED,
        TRANSIENT_FAILURE,
        PERMANENT_FAILURE
    }

    public static NotificationPublishResult published() {
        return new NotificationPublishResult(Outcome.PUBLISHED, null, false);
    }

    /** Hub reachable but asked us to back off (429/503) or failed server-side (5xx). */
    public static NotificationPublishResult transientFailure(String error) {
        return new NotificationPublishResult(Outcome.TRANSIENT_FAILURE, error, true);
    }

    /** Hub reachable and answered, but this instance could not authenticate or was misconfigured. */
    public static NotificationPublishResult localFailure(String error) {
        return new NotificationPublishResult(Outcome.TRANSIENT_FAILURE, error, false);
    }

    public static NotificationPublishResult permanentFailure(String error) {
        return new NotificationPublishResult(Outcome.PERMANENT_FAILURE, error, false);
    }

    public boolean isPublished() {
        return outcome == Outcome.PUBLISHED;
    }
}
