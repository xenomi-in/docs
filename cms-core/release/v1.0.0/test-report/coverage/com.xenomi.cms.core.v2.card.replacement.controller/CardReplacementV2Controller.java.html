package com.xenomi.cms.core.v2.card.replacement.controller;

import java.util.UUID;

import com.xenomi.cms.core.api.ApiResponseEnvelope;
import com.xenomi.cms.core.api.ApiResponseHelper;
import com.xenomi.cms.core.config.CmsPermissions;
import com.xenomi.cms.core.v2.card.lifecycle.model.CardHotlistCallerType;
import com.xenomi.cms.core.v2.card.replacement.dto.CardReplacementV2Response;
import com.xenomi.cms.core.v2.card.replacement.dto.CreateCardReplacementV2Request;
import com.xenomi.cms.core.v2.card.replacement.dto.ReplacementDecisionV2Request;
import com.xenomi.cms.core.v2.card.replacement.service.CardReplacementV2Service;
import io.swagger.v3.oas.annotations.Operation;
import io.swagger.v3.oas.annotations.tags.Tag;
import jakarta.validation.Valid;
import jakarta.validation.constraints.NotBlank;
import jakarta.validation.constraints.Size;
import org.springframework.http.ResponseEntity;
import org.springframework.security.access.AccessDeniedException;
import org.springframework.security.core.Authentication;
import org.springframework.validation.annotation.Validated;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RequestHeader;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RestController;

@Validated
@RestController
@RequestMapping("/cms/api/v2/institutions/{institutionId}")
@Tag(name = "V2 - Cards", description = "Tenant-scoped card issuance and lifecycle")
public class CardReplacementV2Controller {
    private static final String ACTOR_HEADER = "X-Actor-Id";
    private final CardReplacementV2Service service;
    private final com.xenomi.cms.core.v2.card.lifecycle.command.SamePanRenewalApiV2 renewal;
    private final com.xenomi.cms.core.v2.card.lifecycle.command.SecurityReplacementApiV2 securityReplacement;

    public CardReplacementV2Controller(CardReplacementV2Service service,
            com.xenomi.cms.core.v2.card.lifecycle.command.SamePanRenewalApiV2 renewal,
            com.xenomi.cms.core.v2.card.lifecycle.command.SecurityReplacementApiV2 securityReplacement) {
        this.service = service;
        this.renewal = renewal;
        this.securityReplacement = securityReplacement;
    }

    @PostMapping("/cards/{cardId}/replacement")
    @Operation(
            operationId = "createCardReplacementV2",
            summary = "Request card replacement or reissue",
            description = "Same-PAN RENEWAL and new-PAN LOST, STOLEN, COMPROMISED or FRAUD replacement accept asynchronous operations (202). Poll the replacement request until COMPLETED. Other reasons are checked for eligibility; an accepted non-security replacement is still pending its lifecycle command.")
    public ResponseEntity<ApiResponseEnvelope<CardReplacementV2Response>> create(
            @PathVariable UUID institutionId,
            @PathVariable UUID cardId,
            @Valid @RequestBody CreateCardReplacementV2Request request,
            @RequestHeader("Idempotency-Key") @NotBlank @Size(max = 128) String idempotencyKey,
            @RequestHeader(value = "X-Request-Id", required = false) @Size(max = 128) String requestId,
            @RequestHeader(ACTOR_HEADER) @NotBlank @Size(max = 100) String actorId,
            Authentication authentication) {
        if (request.getReplacementReason() == com.xenomi.cms.core.v2.product.model.ReplacementReason.RENEWAL
                && (request.getRequestedPanStrategy() == null || request.getRequestedPanStrategy() == com.xenomi.cms.core.v2.card.replacement.model.CardReplacementPanStrategy.PRESERVE_EXISTING_PAN)) {
            callerType(authentication);
            return ResponseEntity.accepted().body(com.xenomi.cms.core.api.ApiResponseHelper.success(renewal.submit(institutionId,cardId,request,idempotencyKey)));
        }
        if (java.util.Set.of(
                com.xenomi.cms.core.v2.product.model.ReplacementReason.LOST,
                com.xenomi.cms.core.v2.product.model.ReplacementReason.STOLEN,
                com.xenomi.cms.core.v2.product.model.ReplacementReason.COMPROMISED,
                com.xenomi.cms.core.v2.product.model.ReplacementReason.FRAUD)
                .contains(request.getReplacementReason())) {
            CardHotlistCallerType callerType = callerType(authentication);
            return ResponseEntity.accepted().body(com.xenomi.cms.core.api.ApiResponseHelper.success(
                    securityReplacement.submit(institutionId, cardId, request, idempotencyKey,
                            callerType)));
        }
        return ApiResponseHelper.created(service.create(institutionId, cardId, request,
                callerType(authentication), idempotencyKey, requestId, actorId));
    }

    @GetMapping("/cards/{cardId}/replacement/{replacementRequestId}")
    @Operation(operationId = "getCardReplacementV2",
            summary = "Get a card replacement request")
    public ResponseEntity<ApiResponseEnvelope<CardReplacementV2Response>> get(
            @PathVariable UUID institutionId,
            @PathVariable UUID cardId,
            @PathVariable UUID replacementRequestId) {
        return ApiResponseHelper.ok(renewal.get(institutionId, cardId, replacementRequestId));
    }

    @PostMapping("/cards/{cardId}/replacement/{replacementRequestId}/actions/approve")
    @Operation(operationId = "approveCardReplacement",
            summary = "Approve a pending card replacement request",
            description = "Transitions PENDING_APPROVAL to PROCESSING. The decision is recorded as an immutable card event; card creation occurs only when the completion action is called.")
    public ResponseEntity<ApiResponseEnvelope<CardReplacementV2Response>> approve(
            @PathVariable UUID institutionId, @PathVariable UUID cardId,
            @PathVariable UUID replacementRequestId,
            @Valid @RequestBody ReplacementDecisionV2Request request,
            @RequestHeader(value = "X-Request-Id", required = false) @Size(max = 128) String requestId,
            @RequestHeader(ACTOR_HEADER) @NotBlank @Size(max = 100) String actorId) {
        return ApiResponseHelper.ok(service.approve(
                institutionId, cardId, replacementRequestId, request, requestId, actorId));
    }

    @PostMapping("/cards/{cardId}/replacement/{replacementRequestId}/actions/reject")
    @Operation(operationId = "rejectCardReplacement",
            summary = "Reject a pending card replacement request",
            description = "Transitions PENDING_APPROVAL to terminal REJECTED and records the decision as an immutable card event. No replacement card is created.")
    public ResponseEntity<ApiResponseEnvelope<CardReplacementV2Response>> reject(
            @PathVariable UUID institutionId, @PathVariable UUID cardId,
            @PathVariable UUID replacementRequestId,
            @Valid @RequestBody ReplacementDecisionV2Request request,
            @RequestHeader(value = "X-Request-Id", required = false) @Size(max = 128) String requestId,
            @RequestHeader(ACTOR_HEADER) @NotBlank @Size(max = 100) String actorId) {
        return ApiResponseHelper.ok(service.reject(
                institutionId, cardId, replacementRequestId, request, requestId, actorId));
    }

    @PostMapping("/cards/{cardId}/replacement/{replacementRequestId}/actions/complete")
    @Operation(operationId = "completeCardReplacement",
            summary = "Complete an approved card replacement request",
            description = "Processes an approved PROCESSING request using its immutable policy snapshot, creates the replacement card and related records, and transitions the request to COMPLETED.")
    public ResponseEntity<ApiResponseEnvelope<CardReplacementV2Response>> complete(
            @PathVariable UUID institutionId, @PathVariable UUID cardId,
            @PathVariable UUID replacementRequestId,
            @RequestHeader("Idempotency-Key") @NotBlank @Size(max = 128) String idempotencyKey,
            @RequestHeader(value = "X-Request-Id", required = false) @Size(max = 128) String requestId,
            @RequestHeader(ACTOR_HEADER) @NotBlank @Size(max = 100) String actorId) {
        return ApiResponseHelper.ok(service.complete(institutionId, cardId,
                replacementRequestId, idempotencyKey, requestId, actorId));
    }

    private static CardHotlistCallerType callerType(Authentication authentication) {
        if (has(authentication, CmsPermissions.CARD_REPLACE_OPERATIONS)) {
            return CardHotlistCallerType.OPERATIONS;
        }
        if (has(authentication, CmsPermissions.CARD_REPLACE_CUSTOMER)) {
            return CardHotlistCallerType.CUSTOMER;
        }
        throw new AccessDeniedException("Card replacement permission is required");
    }

    private static boolean has(Authentication authentication, String authority) {
        return authentication != null && authentication.getAuthorities().stream()
                .anyMatch(value -> authority.equals(value.getAuthority()));
    }
}
