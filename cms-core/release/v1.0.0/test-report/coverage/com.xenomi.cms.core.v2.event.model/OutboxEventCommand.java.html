package com.xenomi.cms.core.v2.event.model;

import java.util.Map;
import java.util.Objects;
import java.util.UUID;

/**
 * Everything the outbox writer needs to persist one Falcon notification.
 *
 * <p>{@code idempotencyKey} must be derived from the <em>domain</em> identity of the
 * event (card_event id, replacement request id, product id + updatedAt, ...), never from
 * transport metadata such as the HTTP request id. Two business operations that produce
 * the same key are, by definition, the same event.
 */
public record OutboxEventCommand(
        UUID institutionId,
        String eventType,
        String aggregateType,
        String aggregateId,
        String customerId,
        String correlationId,
        NotificationPriority priority,
        UUID idempotencyKey,
        Map<String, Object> payload) {

    public OutboxEventCommand {
        Objects.requireNonNull(institutionId, "institutionId");
        Objects.requireNonNull(eventType, "eventType");
        Objects.requireNonNull(aggregateType, "aggregateType");
        Objects.requireNonNull(aggregateId, "aggregateId");
        Objects.requireNonNull(idempotencyKey, "idempotencyKey");
        priority = priority == null ? NotificationPriority.MEDIUM : priority;
        payload = payload == null ? Map.of() : payload;
    }

    public static Builder builder() {
        return new Builder();
    }

    public static final class Builder {
        private UUID institutionId;
        private String eventType;
        private String aggregateType;
        private String aggregateId;
        private String customerId;
        private String correlationId;
        private NotificationPriority priority = NotificationPriority.MEDIUM;
        private UUID idempotencyKey;
        private Map<String, Object> payload = Map.of();

        private Builder() {
        }

        public Builder institutionId(UUID value) { this.institutionId = value; return this; }
        public Builder eventType(String value) { this.eventType = value; return this; }
        public Builder aggregateType(String value) { this.aggregateType = value; return this; }
        public Builder aggregateId(UUID value) { this.aggregateId = value == null ? null : value.toString(); return this; }
        public Builder customerId(String value) { this.customerId = value; return this; }
        public Builder correlationId(String value) { this.correlationId = value; return this; }
        public Builder priority(NotificationPriority value) { this.priority = value; return this; }
        public Builder idempotencyKey(UUID value) { this.idempotencyKey = value; return this; }
        public Builder payload(Map<String, Object> value) { this.payload = value; return this; }

        public OutboxEventCommand build() {
            return new OutboxEventCommand(institutionId, eventType, aggregateType, aggregateId,
                    customerId, correlationId, priority, idempotencyKey, payload);
        }
    }
}
