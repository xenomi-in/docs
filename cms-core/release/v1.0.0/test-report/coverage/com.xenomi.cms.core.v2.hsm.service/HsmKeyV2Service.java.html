    @Transactional
    public HsmKeyV2Response generate(UUID institutionId, GenerateHsmKeyV2Request request,
            String actorId) {
        requireInstitution(institutionId);
        BinConfigurationV2 bin = validateGenerationRequest(institutionId, request);
        String keyIndex = deriveGeneratedKeyIndex(
                institutionId, bin, request.getKeyType(), request.getKeyScope());
        int version = keyRepository.findMaximumVersion(request.getKeyType(), keyIndex) + 1;
        if (keyRepository.existsByKeyTypeAndKeyIndexAndKeyVersion(
                request.getKeyType(), keyIndex, version)) {
            throw new ResourceAlreadyExistsException("HSM key version",
                    request.getKeyType() + "/" + keyIndex + "/" + version);
        }

        boolean tr31 = request.getKeyScheme() == HsmKeyScheme.TR31;
        KeyGenerationRequest hsmRequest = KeyGenerationRequest.builder()
                .keyType(request.getKeyType().name())
                .scheme(request.getKeyScheme().name())
                .binNumber(bin == null ? null : bin.getBinNumber())
                .keyScope(request.getKeyScope().name())
                .keyLengthBits(request.getKeyLengthBits())
                .tr31Format(tr31)
                .tr31KeyUsage(normalizeNullable(request.getTr31KeyUsage()))
                .tr31Algorithm(normalizeNullable(request.getTr31Algorithm()))
                .tr31ModeOfUse(normalizeNullable(request.getTr31ModeOfUse()))
                .build();
        KeyGenerationResult result;
        result = hsmClient.generateKey(hsmRequest);
        validateHsmResult(result, tr31);

        LocalDate effectiveFrom = LocalDate.now();
        String actor = actorId.trim();
        HsmKeyReferenceV2 key = HsmKeyReferenceV2.builder()
                .keyId(uuidGenerator.generate()).institutionId(institutionId)
                .keyType(request.getKeyType()).keyIndex(keyIndex)
                .keyLabel(resolveGeneratedLabel(request, bin))
                .keyCheckValue(result.getKeyCheckValue().trim().toUpperCase(Locale.ROOT))
                .keyVersion(version).keyScope(request.getKeyScope()).scheme(request.getKeyScheme())
                .keyLengthBits(request.getKeyLengthBits()).tr31(result.isTr31Format())
                .tr31KeyUsage(normalizeNullable(request.getTr31KeyUsage()))
                .tr31Algorithm(normalizeNullable(request.getTr31Algorithm()))
                .tr31ModeOfUse(normalizeNullable(request.getTr31ModeOfUse()))
                .hsmIdentifier(firstNonBlank(result.getHsmIdentifier(), hsmClient.getProviderId()))
                .hsmPartition(trimToNull(result.getHsmPartition()))
                .effectiveFrom(effectiveFrom)
                .expiryDate(effectiveFrom.plusYears(request.getValidityYears()))
                .status(HsmKeyStatus.PENDING_APPROVAL)
                .binId(bin == null ? null : bin.getBinId())
                .encryptedKeyBlock(result.getEncryptedKeyBlock().trim())
                .createdBy(actor).updatedBy(actor).build();
        return toResponse(keyRepository.save(key));
    }

    public boolean testHsmConnection() {
        return hsmClient.testConnection();
    }

    @Transactional
    public HsmKeyV2Response register(UUID institutionId, RegisterHsmKeyV2Request request,
            String actorId) {
        requireInstitution(institutionId);
        validateRequest(institutionId, request);
        HsmKeyType keyType = request.getKeyType();
        String keyIndex = normalize(request.getKeyIndex());
        int version = keyRepository.findMaximumVersion(keyType, keyIndex) + 1;
        if (keyRepository.existsByKeyTypeAndKeyIndexAndKeyVersion(keyType, keyIndex, version)) {
            throw new ResourceAlreadyExistsException("HSM key version", keyType + "/" + keyIndex + "/" + version);
        }
        String actor = actorId.trim();
        HsmKeyReferenceV2 key = HsmKeyReferenceV2.builder()
                .keyId(uuidGenerator.generate()).institutionId(institutionId)
                .keyType(keyType).keyIndex(keyIndex).keyLabel(trimToNull(request.getKeyLabel()))
                .keyCheckValue(request.getKeyCheckValue().toUpperCase(Locale.ROOT))
                .keyVersion(version).keyScope(request.getKeyScope()).scheme(request.getKeyScheme())
                .keyLengthBits(request.getKeyLengthBits()).tr31(request.isTr31())
                .tr31KeyUsage(normalizeNullable(request.getTr31KeyUsage()))
                .tr31Algorithm(normalizeNullable(request.getTr31Algorithm()))
                .tr31ModeOfUse(normalizeNullable(request.getTr31ModeOfUse()))
                .hsmIdentifier(trimToNull(request.getHsmIdentifier()))
                .hsmPartition(trimToNull(request.getHsmPartition()))
                .hsmSerialNumber(trimToNull(request.getHsmSerialNumber()))
                .effectiveFrom(request.getEffectiveFrom()).expiryDate(request.getExpiryDate())
                .status(HsmKeyStatus.PENDING_APPROVAL)
                .binId(request.getBinId()).acsId(request.getAcsId())
                .cardVendorId(request.getCardVendorId()).stakeholderId(request.getStakeholderId())
                .encryptedKeyBlock(trimToNull(request.getEncryptedKeyBlock()))
                .createdBy(actor).updatedBy(actor).build();
        return toResponse(keyRepository.save(key));
    }

    @Transactional(readOnly = true)
    public HsmKeyV2Response get(UUID institutionId, UUID keyId) {
        return toResponse(find(institutionId, keyId));
    }

    @Transactional(readOnly = true)
    public Page<HsmKeyV2Response> search(UUID institutionId, HsmKeyType keyType, HsmKeyScope scope,
            HsmKeyStatus status, UUID binId, Pageable pageable) {
        requireInstitution(institutionId);
        if (binId != null && binRepository.findByBinIdAndInstitutionId(binId, institutionId).isEmpty()) {
            throw new ResourceNotFoundException("BIN configuration", binId.toString());
        }
        return keyRepository.search(institutionId, keyType, scope, status, binId, pageable)
                .map(HsmKeyV2Service::toResponse);
    }

    @Transactional
    public HsmKeyV2Response changeStatus(UUID institutionId, UUID keyId,
            UpdateHsmKeyStatusV2Request request, String actorId) {
        HsmKeyReferenceV2 key = keyRepository.findByTenantAndIdForUpdate(institutionId, keyId)
                .orElseThrow(() -> new ResourceNotFoundException("HSM key", keyId.toString()));
        HsmKeyStatus current = key.getStatus();
        HsmKeyStatus requested = request.getStatus();
        if (!TRANSITIONS.getOrDefault(current, Set.of()).contains(requested)) {
            throw new IllegalArgumentException("Invalid HSM key transition: " + current + " -> " + requested);
        }
        LocalDate today = LocalDate.now();
        if (requested == HsmKeyStatus.SCHEDULED && !key.getEffectiveFrom().isAfter(today)) {
            throw new IllegalArgumentException("SCHEDULED keys require a future effectiveFrom date");
        }
        if (requested == HsmKeyStatus.ACTIVE
                && (key.getEffectiveFrom().isAfter(today) || !key.getExpiryDate().isAfter(today))) {
            throw new IllegalArgumentException("ACTIVE keys must be within their cryptoperiod");
        }
        if (request.getExpiryDate() != null) {
            if (!request.getExpiryDate().isAfter(key.getEffectiveFrom())) {
                throw new IllegalArgumentException("expiryDate must be later than effectiveFrom");
            }
            key.setExpiryDate(request.getExpiryDate());
        }
        if ((requested == HsmKeyStatus.RETIRED || requested == HsmKeyStatus.REVOKED)
                && request.getReason() == null) {
            throw new IllegalArgumentException("reason is required when retiring or revoking a key");
        }
        key.setStatus(requested);
        key.setRotationReason(trimToNull(request.getReason()));
        if (requested == HsmKeyStatus.DECRYPT_ONLY || requested == HsmKeyStatus.RETIRED
                || requested == HsmKeyStatus.REVOKED) {
            key.setRotatedAt(OffsetDateTime.now(ZoneOffset.UTC));
        }
        key.setUpdatedBy(actorId.trim());
        if (requested == HsmKeyStatus.ACTIVE && key.getPreviousKeyId() != null) {
            HsmKeyReferenceV2 previous = keyRepository
                    .findByTenantAndIdForUpdate(institutionId, key.getPreviousKeyId())
                    .orElseThrow(() -> new ResourceNotFoundException(
                            "Previous HSM key", key.getPreviousKeyId().toString()));
            if (previous.getStatus() == HsmKeyStatus.ACTIVE) {
                previous.setStatus(HsmKeyStatus.DECRYPT_ONLY);
                previous.setRotatedAt(OffsetDateTime.now(ZoneOffset.UTC));
                previous.setUpdatedBy(actorId.trim());
                keyRepository.save(previous);
            }
        }
        return toResponse(keyRepository.save(key));
    }

    @Transactional
    public HsmKeyV2Response rotate(UUID institutionId, UUID keyId,
            RotateHsmKeyV2Request request, String actorId) {
        HsmKeyReferenceV2 previous = keyRepository.findByTenantAndIdForUpdate(institutionId, keyId)
                .orElseThrow(() -> new ResourceNotFoundException("HSM key", keyId.toString()));
        if (!Set.of(HsmKeyStatus.ACTIVE, HsmKeyStatus.DECRYPT_ONLY,
                HsmKeyStatus.SUSPENDED).contains(previous.getStatus())) {
            throw new IllegalArgumentException("Only ACTIVE, DECRYPT_ONLY or SUSPENDED keys can be rotated");
        }
        if (previous.getReplacementKeyId() != null) {
            throw new IllegalArgumentException("This key already has a registered replacement");
        }
        if (!request.getExpiryDate().isAfter(request.getEffectiveFrom())) {
            throw new IllegalArgumentException("expiryDate must be later than effectiveFrom");
        }
        if (previous.isTr31() && isBlank(request.getEncryptedKeyBlock())) {
            throw new IllegalArgumentException("A protected replacement key block is required for a TR-31 key");
        }
        int version = keyRepository.findMaximumVersion(previous.getKeyType(), previous.getKeyIndex()) + 1;
        UUID replacementId = uuidGenerator.generate();
        String actor = actorId.trim();
        HsmKeyReferenceV2 replacement = HsmKeyReferenceV2.builder()
                .keyId(replacementId).institutionId(institutionId)
                .keyType(previous.getKeyType()).keyIndex(previous.getKeyIndex())
                .keyLabel(trimToNull(request.getKeyLabel()))
                .keyCheckValue(request.getKeyCheckValue().toUpperCase(Locale.ROOT)).keyVersion(version)
                .keyScope(previous.getKeyScope()).scheme(previous.getScheme())
                .keyLengthBits(previous.getKeyLengthBits()).tr31(previous.isTr31())
                .tr31KeyUsage(previous.getTr31KeyUsage()).tr31Algorithm(previous.getTr31Algorithm())
                .tr31ModeOfUse(previous.getTr31ModeOfUse())
                .hsmIdentifier(trimToNull(request.getHsmIdentifier()))
                .hsmPartition(trimToNull(request.getHsmPartition()))
                .hsmSerialNumber(trimToNull(request.getHsmSerialNumber()))
                .effectiveFrom(request.getEffectiveFrom()).expiryDate(request.getExpiryDate())
                .status(HsmKeyStatus.PENDING_APPROVAL).previousKeyId(previous.getKeyId())
                .rotationReason(request.getRotationReason().trim())
                .encryptedKeyBlock(trimToNull(request.getEncryptedKeyBlock()))
                .binId(previous.getBinId()).acsId(previous.getAcsId())
                .cardVendorId(previous.getCardVendorId()).stakeholderId(previous.getStakeholderId())
                .createdBy(actor).updatedBy(actor).build();
        keyRepository.save(replacement);
        previous.setReplacementKeyId(replacementId);
        previous.setRotationReason(request.getRotationReason().trim());
        previous.setUpdatedBy(actor);
        keyRepository.save(previous);
        return toResponse(replacement);
    }

    private void validateRequest(UUID institutionId, RegisterHsmKeyV2Request request) {
        if (!request.getExpiryDate().isAfter(request.getEffectiveFrom())) {
            throw new IllegalArgumentException("expiryDate must be later than effectiveFrom");
        }
        boolean validLength = switch (request.getKeyScheme()) {
            case TDES -> Set.of(112, 168).contains(request.getKeyLengthBits());
            case AES -> Set.of(128, 192, 256).contains(request.getKeyLengthBits());
            case TR31 -> Set.of(112, 128, 168, 192, 256).contains(request.getKeyLengthBits());
        };
        if (!validLength) throw new IllegalArgumentException("keyLengthBits is invalid for " + request.getKeyScheme());
        if (request.getKeyScheme() == HsmKeyScheme.TR31 && !request.isTr31()) {
            throw new IllegalArgumentException("tr31 must be true when keyScheme=TR31");
        }
        if (request.getKeyScheme() != HsmKeyScheme.TR31 && request.isTr31()) {
            throw new IllegalArgumentException("tr31 is allowed only when keyScheme=TR31");
        }
        if (request.isTr31()) {
            if (isBlank(request.getTr31KeyUsage()) || isBlank(request.getTr31Algorithm())
                    || isBlank(request.getTr31ModeOfUse()) || isBlank(request.getEncryptedKeyBlock())) {
                throw new IllegalArgumentException("TR-31 keys require usage, algorithm, modeOfUse and a protected key block");
            }
        } else if (!isBlank(request.getTr31KeyUsage()) || !isBlank(request.getTr31Algorithm())
                || !isBlank(request.getTr31ModeOfUse())) {
            throw new IllegalArgumentException("TR-31 attributes are only allowed when tr31=true");
        }
        validateScope(institutionId, request);
    }

    private BinConfigurationV2 validateGenerationRequest(
            UUID institutionId, GenerateHsmKeyV2Request request) {
        if (request.getKeyType() == HsmKeyType.KEK) {
            throw new IllegalArgumentException(
                    "KEK must be provisioned through the secure registration process");
        }
        if (request.getKeyType() != null
                && request.getKeyScope() != request.getKeyType().generationScope()) {
            throw new IllegalArgumentException(request.getKeyType() + " keys must use "
                    + request.getKeyType().generationScope() + " scope");
        }
        validateGenerationFormat(request);
        if (request.getKeyScope() == HsmKeyScope.BIN) {
            if (request.getBinId() == null) {
                throw new IllegalArgumentException("binId is required when keyScope=BIN");
            }
            return binRepository.findByTenantAndIdForUpdate(institutionId, request.getBinId())
                    .orElseThrow(() -> new ResourceNotFoundException(
                            "BIN configuration", request.getBinId().toString()));
        }
        if (request.getKeyScope() != HsmKeyScope.INSTITUTION || request.getBinId() != null) {
            throw new IllegalArgumentException(
                    "HSM generation currently supports only INSTITUTION or BIN scope");
        }
        institutionRepository.findByIdForUpdate(institutionId)
                .orElseThrow(() -> new ResourceNotFoundException(
                        "Institution", institutionId.toString()));
        return null;
    }

    private static void validateGenerationFormat(GenerateHsmKeyV2Request request) {
        Integer length = request.getKeyLengthBits();
        switch (request.getKeyScheme()) {
            case TDES -> {
                if (!Integer.valueOf(112).equals(length)) {
                    throw new IllegalArgumentException(
                            "TDES generation requires a double-length key with keyLengthBits=112");
                }
                rejectTr31Metadata(request);
            }
            case AES -> {
                if (!Set.of(128, 192, 256).contains(length)) {
                    throw new IllegalArgumentException(
                            "AES generation requires keyLengthBits of 128, 192, or 256");
                }
                rejectTr31Metadata(request);
            }
            case TR31 -> {
                if (!Set.of(112, 128, 168, 192, 256).contains(length)) {
                    throw new IllegalArgumentException("Invalid keyLengthBits for TR-31 generation");
                }
                if (isBlank(request.getTr31KeyUsage()) || isBlank(request.getTr31Algorithm())
                        || isBlank(request.getTr31ModeOfUse())) {
                    throw new IllegalArgumentException(
                            "TR-31 generation requires keyUsage, algorithm, and modeOfUse");
                }
            }
        }
    }

    private static void rejectTr31Metadata(GenerateHsmKeyV2Request request) {
        if (!isBlank(request.getTr31KeyUsage()) || !isBlank(request.getTr31Algorithm())
                || !isBlank(request.getTr31ModeOfUse())) {
            throw new IllegalArgumentException("TR-31 metadata is allowed only when keyScheme=TR31");
        }
    }

    private static void validateHsmResult(KeyGenerationResult result, boolean tr31Expected) {
        if (result == null || isBlank(result.getEncryptedKeyBlock())) {
            throw new HsmKeyGenerationException("HSM returned no protected key block");
        }
        if (isBlank(result.getKeyCheckValue())
                || !result.getKeyCheckValue().trim().matches("[0-9A-Fa-f]{4,16}")) {
            throw new HsmKeyGenerationException("HSM returned an invalid key check value");
        }
        if (result.isTr31Format() != tr31Expected) {
            throw new HsmKeyGenerationException(
                    "HSM response format does not match the requested generation format");
        }
    }

    private static String deriveGeneratedKeyIndex(UUID institutionId, BinConfigurationV2 bin,
            HsmKeyType keyType, HsmKeyScope scope) {
        if (scope == HsmKeyScope.BIN) {
            String digits = bin.getBinNumber().replaceAll("\\D", "");
            String suffix = digits.substring(Math.max(0, digits.length() - 6));
            return "BIN_" + keyType + "_" + suffix;
        }
        String tenantSuffix = institutionId.toString().replace("-", "");
        return "INST_" + keyType + "_" + tenantSuffix.substring(tenantSuffix.length() - 8);
    }

    private static String resolveGeneratedLabel(
            GenerateHsmKeyV2Request request, BinConfigurationV2 bin) {
        String supplied = trimToNull(request.getKeyLabel());
        return supplied != null ? supplied
                : request.getKeyType() + "_" + (bin == null ? "INSTITUTION" : bin.getBinNumber());
    }

    private static String firstNonBlank(String preferred, String fallback) {
        String value = trimToNull(preferred);
        return value == null ? trimToNull(fallback) : value;
    }

    private void validateScope(UUID institutionId, RegisterHsmKeyV2Request request) {
        int scopedIds = count(request.getBinId()) + count(request.getAcsId())
                + count(request.getCardVendorId()) + count(request.getStakeholderId());
        switch (request.getKeyScope()) {
            case INSTITUTION -> { if (scopedIds != 0) throw scopeError(); }
            case BIN -> {
                if (request.getBinId() == null || scopedIds != 1) throw scopeError();
                if (binRepository.findByBinIdAndInstitutionId(request.getBinId(), institutionId).isEmpty())
                    throw new ResourceNotFoundException("BIN configuration", request.getBinId().toString());
            }
            case ACS -> { if (request.getAcsId() == null || scopedIds != 1) throw scopeError(); }
            case CARD_VENDOR -> { if (request.getCardVendorId() == null || scopedIds != 1) throw scopeError(); }
            case STAKEHOLDER -> { if (request.getStakeholderId() == null || scopedIds != 1) throw scopeError(); }
            case CARD_NETWORK -> throw new IllegalArgumentException(
                    "CARD_NETWORK scope requires a dedicated scope identifier before it can be registered");
        }
        if (request.getKeyType() == HsmKeyType.KEK) {
            if (!Set.of(HsmKeyScope.ACS, HsmKeyScope.CARD_VENDOR,
                    HsmKeyScope.STAKEHOLDER).contains(request.getKeyScope())) {
                throw new IllegalArgumentException(
                        "KEK must be scoped to ACS, CARD_VENDOR, or STAKEHOLDER");
            }
            if (isBlank(request.getEncryptedKeyBlock())) {
                throw new IllegalArgumentException(
                        "KEK registration requires a CMS-HSM-protected key block");
            }
        }
    }

    private static IllegalArgumentException scopeError() {
        return new IllegalArgumentException("Exactly the identifier matching keyScope must be supplied");
    }
    private static int count(UUID value) { return value == null ? 0 : 1; }
    private static boolean isBlank(String value) { return value == null || value.isBlank(); }
    private void requireInstitution(UUID id) {
        if (!institutionRepository.existsById(id)) throw new ResourceNotFoundException("Institution", id.toString());
    }
    private HsmKeyReferenceV2 find(UUID institutionId, UUID keyId) {
        return keyRepository.findByKeyIdAndInstitutionId(keyId, institutionId)
                .orElseThrow(() -> new ResourceNotFoundException("HSM key", keyId.toString()));
    }
    private static String normalize(String value) { return value.trim().toUpperCase(Locale.ROOT); }
    private static String normalizeNullable(String value) { return isBlank(value) ? null : normalize(value); }
    private static String trimToNull(String value) { return isBlank(value) ? null : value.trim(); }

    static HsmKeyV2Response toResponse(HsmKeyReferenceV2 k) {
        return HsmKeyV2Response.builder().keyId(k.getKeyId()).institutionId(k.getInstitutionId())
                .keyType(k.getKeyType()).keyIndex(k.getKeyIndex()).keyLabel(k.getKeyLabel())
                .keyCheckValue(k.getKeyCheckValue()).keyVersion(k.getKeyVersion())
                .keyScope(k.getKeyScope()).keyScheme(k.getScheme()).keyLengthBits(k.getKeyLengthBits())
                .tr31(k.isTr31()).tr31KeyUsage(k.getTr31KeyUsage()).tr31Algorithm(k.getTr31Algorithm())
                .tr31ModeOfUse(k.getTr31ModeOfUse()).hsmIdentifier(k.getHsmIdentifier())
                .hsmPartition(k.getHsmPartition()).hsmSerialNumber(k.getHsmSerialNumber())
                .effectiveFrom(k.getEffectiveFrom()).expiryDate(k.getExpiryDate()).status(k.getStatus())
                .binId(k.getBinId()).acsId(k.getAcsId()).cardVendorId(k.getCardVendorId())
                .stakeholderId(k.getStakeholderId()).importedViaKekId(k.getImportedViaKekId())
                .externalRequestId(k.getExternalRequestId()).previousKeyId(k.getPreviousKeyId())
                .replacementKeyId(k.getReplacementKeyId()).rotatedAt(k.getRotatedAt())
                .rotationReason(k.getRotationReason())
                .protectedKeyBlockStored(k.getEncryptedKeyBlock() != null)
                .createdAt(k.getCreatedAt()).updatedAt(k.getUpdatedAt())
                .createdBy(k.getCreatedBy()).updatedBy(k.getUpdatedBy()).build();
    }

    private static Map<HsmKeyStatus, Set<HsmKeyStatus>> transitions() {
        Map<HsmKeyStatus, Set<HsmKeyStatus>> m = new EnumMap<>(HsmKeyStatus.class);
        m.put(HsmKeyStatus.PENDING_APPROVAL, Set.of(HsmKeyStatus.SCHEDULED, HsmKeyStatus.ACTIVE, HsmKeyStatus.REVOKED));
        m.put(HsmKeyStatus.SCHEDULED, Set.of(HsmKeyStatus.ACTIVE, HsmKeyStatus.SUSPENDED, HsmKeyStatus.REVOKED));
        m.put(HsmKeyStatus.ACTIVE, Set.of(HsmKeyStatus.DECRYPT_ONLY, HsmKeyStatus.SUSPENDED, HsmKeyStatus.RETIRED, HsmKeyStatus.REVOKED));
        m.put(HsmKeyStatus.DECRYPT_ONLY, Set.of(HsmKeyStatus.RETIRED, HsmKeyStatus.REVOKED));
        m.put(HsmKeyStatus.SUSPENDED, Set.of(HsmKeyStatus.ACTIVE, HsmKeyStatus.RETIRED, HsmKeyStatus.REVOKED));
        m.put(HsmKeyStatus.RETIRED, Set.of()); m.put(HsmKeyStatus.REVOKED, Set.of());
        return Map.copyOf(m);
    }
