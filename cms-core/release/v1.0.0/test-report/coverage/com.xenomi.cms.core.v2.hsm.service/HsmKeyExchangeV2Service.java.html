@Service
public class HsmKeyExchangeV2Service {

    private static final Logger log = LoggerFactory.getLogger(HsmKeyExchangeV2Service.class);
    private static final Set<HsmKeyScope> EXTERNAL_KEK_SCOPES = Set.of(
            HsmKeyScope.ACS, HsmKeyScope.CARD_VENDOR, HsmKeyScope.STAKEHOLDER);

    private final HsmKeyReferenceV2Repository keyRepository;
    private final KeyExchangeEventV2Repository eventRepository;
    private final HsmKeyExchangeClient exchangeClient;
    private final InstitutionDataKeyAssignmentV2Service institutionAssignmentService;
    private final BinHsmKeyAssignmentV2Service binAssignmentService;
    private final KeyExchangeEventAuditV2Service auditService;
    private final DatabaseUuidV7Generator uuidGenerator;

    public HsmKeyExchangeV2Service(HsmKeyReferenceV2Repository keyRepository,
            KeyExchangeEventV2Repository eventRepository,
            HsmKeyExchangeClient exchangeClient,
            InstitutionDataKeyAssignmentV2Service institutionAssignmentService,
            BinHsmKeyAssignmentV2Service binAssignmentService,
            KeyExchangeEventAuditV2Service auditService,
            DatabaseUuidV7Generator uuidGenerator) {
        this.keyRepository = keyRepository;
        this.eventRepository = eventRepository;
        this.exchangeClient = exchangeClient;
        this.institutionAssignmentService = institutionAssignmentService;
        this.binAssignmentService = binAssignmentService;
        this.auditService = auditService;
        this.uuidGenerator = uuidGenerator;
    }

    @Transactional
    public ImportHsmKeyV2Response importKey(UUID institutionId,
            ImportHsmKeyV2Request request, String idempotencyKey, String actorId) {
        validateRequest(request);
        String actor = actorId.trim();
        KeyPurpose purpose = request.getKeyPurpose();
        String expectedKcv = normalize(request.getExpectedKeyCheckValue());
        String idempotencyHash = sha256(idempotencyKey.trim());
        String fingerprint = fingerprint(request, purpose, expectedKcv);

        Optional<KeyExchangeEventV2> existing = eventRepository
                .findByInstitutionIdAndIdempotencyKeyHash(institutionId, idempotencyHash);
        if (existing.isEmpty()) {
            existing = eventRepository.findByInstitutionIdAndExternalRequestId(
                    institutionId, request.getExternalRequestId());
        }
        if (existing.isPresent()) {
            return replay(existing.get(), fingerprint, institutionId);
        }

        HsmKeyReferenceV2 kek = keyRepository.findByKeyIdAndInstitutionId(
                request.getKekKeyId(), institutionId)
                .orElseThrow(() -> new ResourceNotFoundException(
                        "KEK", request.getKekKeyId().toString()));
        validateKek(kek);

        ImportedKeyResult imported;
        try {
            imported = exchangeClient.importKeyUnderKek(ImportKeyUnderKekRequest.builder()
                    .kekProtectedKeyBlock(kek.getEncryptedKeyBlock())
                    .transportedWorkingKeyBlock(request.getTransportedKeyBlock())
                    .workingKeyType(request.getKeyType().name())
                    .keyScheme(request.getKeyScheme().name())
                    .keyLengthBits(request.getKeyLengthBits())
                    .tr31(request.getKeyScheme() == HsmKeyScheme.TR31)
                    .tr31KeyUsage(normalizeNullable(request.getTr31KeyUsage()))
                    .tr31Algorithm(normalizeNullable(request.getTr31Algorithm()))
                    .tr31ModeOfUse(normalizeNullable(request.getTr31ModeOfUse()))
                    .build());
        } catch (HsmKeyGenerationException e) {
            recordFailureSafely(institutionId, request, idempotencyHash, fingerprint,
                    purpose, expectedKcv, null, KeyExchangeStatus.FAILED,
                    "HSM_IMPORT_FAILED", actor);
            throw e;
        }
        try {
            validateImportedResult(imported, request.getKeyScheme() == HsmKeyScheme.TR31);
        } catch (HsmKeyGenerationException e) {
            recordFailureSafely(institutionId, request, idempotencyHash, fingerprint,
                    purpose, expectedKcv, null, KeyExchangeStatus.FAILED,
                    "INVALID_HSM_RESPONSE", actor);
            throw e;
        }
        String calculatedKcv = normalize(imported.getKeyCheckValue());
        if (!expectedKcv.equals(calculatedKcv)) {
            recordFailureSafely(institutionId, request, idempotencyHash, fingerprint,
                    purpose, expectedKcv, calculatedKcv, KeyExchangeStatus.KCV_MISMATCH,
                    "KCV_MISMATCH", actor);
            throw new BusinessRuleViolationException(
                    "The imported key check value does not match the expected value");
        }

        UUID keyId = uuidGenerator.generate();
        String keyIndex = "IMP_" + request.getKeyType() + "_"
                + keyId.toString().replace("-", "").substring(0, 8).toUpperCase(Locale.ROOT);
        HsmKeyReferenceV2 importedKey = HsmKeyReferenceV2.builder()
                .keyId(keyId).institutionId(institutionId)
                .keyType(request.getKeyType()).keyIndex(keyIndex)
                .keyLabel(trimToNull(request.getKeyLabel()))
                .keyCheckValue(calculatedKcv).keyVersion(1)
                .keyScope(request.getTargetScope())
                .scheme(request.getKeyScheme()).keyLengthBits(request.getKeyLengthBits())
                .tr31(imported.isTr31())
                .tr31KeyUsage(normalizeNullable(request.getTr31KeyUsage()))
                .tr31Algorithm(normalizeNullable(request.getTr31Algorithm()))
                .tr31ModeOfUse(normalizeNullable(request.getTr31ModeOfUse()))
                .hsmIdentifier(trimToNull(imported.getHsmIdentifier()))
                .hsmPartition(trimToNull(imported.getHsmPartition()))
                .effectiveFrom(request.getEffectiveFrom().toLocalDate())
                .expiryDate(request.getExpiryDate())
                .status(HsmKeyStatus.PENDING_APPROVAL)
                .binId(request.getTargetScope() == HsmKeyScope.BIN ? request.getBinId() : null)
                .encryptedKeyBlock(imported.getCmsProtectedKeyBlock())
                .importedViaKekId(kek.getKeyId())
                .externalRequestId(request.getExternalRequestId())
                .createdBy(actor).updatedBy(actor).build();
        keyRepository.save(importedKey);

        UUID assignmentId = createAssignment(
                institutionId, importedKey, request, purpose, actor);
        KeyExchangeEventV2 event = eventRepository.save(KeyExchangeEventV2.builder()
                .exchangeEventId(uuidGenerator.generate())
                .institutionId(institutionId).kekKeyId(kek.getKeyId())
                .externalRequestId(request.getExternalRequestId())
                .idempotencyKeyHash(idempotencyHash).requestFingerprint(fingerprint)
                .direction(KeyExchangeDirection.INBOUND)
                .workingKeyType(request.getKeyType()).targetScope(request.getTargetScope())
                .keyPurpose(purpose).expectedKcv(expectedKcv).calculatedKcv(calculatedKcv)
                .resultingKeyId(keyId).assignmentId(assignmentId)
                .exchangeStatus(KeyExchangeStatus.IMPORTED)
                .completedAt(OffsetDateTime.now(java.time.ZoneOffset.UTC)).createdBy(actor).build());
        return response(event, importedKey, false);
    }

    private UUID createAssignment(UUID institutionId, HsmKeyReferenceV2 key,
            ImportHsmKeyV2Request request, KeyPurpose purpose, String actor) {
        if (request.getTargetScope() == HsmKeyScope.BIN) {
            BinHsmKeyAssignmentV2Response assignment = binAssignmentService.create(
                    institutionId, request.getBinId(),
                    CreateBinHsmKeyAssignmentV2Request.builder()
                            .hsmKeyId(key.getKeyId()).keyPurpose(purpose)
                            .effectiveFrom(request.getEffectiveFrom())
                            .rotationGroup(trimToNull(request.getRotationGroup()))
                            .approvalRequestId(request.getApprovalRequestId()).build(),
                    actor);
            return assignment.getAssignmentId();
        }
        DataKeyAssignmentV2Response assignment = institutionAssignmentService.create(
                institutionId, CreateDataKeyAssignmentV2Request.builder()
                        .hsmKeyId(key.getKeyId()).keyPurpose(purpose)
                        .dataClassification(defaultClassification(request.getDataClassification()))
                        .effectiveFrom(request.getEffectiveFrom())
                        .rotationPolicyReference(
                                trimToNull(request.getRotationPolicyReference()))
                        .build(), actor);
        return assignment.getAssignmentId();
    }

    private ImportHsmKeyV2Response replay(KeyExchangeEventV2 event,
            String fingerprint, UUID institutionId) {
        if (!event.getRequestFingerprint().equals(fingerprint)) {
            throw new ResourceAlreadyExistsException(
                    "Idempotency key or external request", event.getExternalRequestId());
        }
        if (event.getExchangeStatus() != KeyExchangeStatus.IMPORTED
                || event.getResultingKeyId() == null) {
            throw new ResourceAlreadyExistsException(
                    "Failed key-exchange request", event.getExternalRequestId());
        }
        HsmKeyReferenceV2 key = keyRepository.findByKeyIdAndInstitutionId(
                event.getResultingKeyId(), institutionId)
                .orElseThrow(() -> new ResourceNotFoundException(
                        "Imported HSM key", event.getResultingKeyId().toString()));
        return response(event, key, true);
    }

    private static ImportHsmKeyV2Response response(KeyExchangeEventV2 event,
            HsmKeyReferenceV2 key, boolean replayed) {
        return ImportHsmKeyV2Response.builder()
                .exchangeEventId(event.getExchangeEventId())
                .exchangeStatus(event.getExchangeStatus())
                .hsmKey(HsmKeyV2Service.toResponse(key))
                .assignmentId(event.getAssignmentId())
                .assignmentScope(event.getTargetScope())
                .replayed(replayed).build();
    }

    private static void validateRequest(ImportHsmKeyV2Request request) {
        if (request.getKeyType() == HsmKeyType.KEK) {
            throw new IllegalArgumentException(
                    "KEK cannot be imported as an operational working key");
        }
        if (!Set.of(HsmKeyScope.INSTITUTION, HsmKeyScope.BIN)
                .contains(request.getTargetScope())) {
            throw new IllegalArgumentException(
                    "targetScope must be INSTITUTION or BIN");
        }
        if ((request.getTargetScope() == HsmKeyScope.BIN) != (request.getBinId() != null)) {
            throw new IllegalArgumentException(
                    "binId is required only when targetScope=BIN");
        }
        if (request.getKeyPurpose().scope() != request.getTargetScope()) {
            throw new IllegalArgumentException(
                    "keyPurpose is not approved for the requested targetScope");
        }
        if (!request.getExpiryDate().isAfter(request.getEffectiveFrom().toLocalDate())) {
            throw new IllegalArgumentException(
                    "expiryDate must be later than effectiveFrom");
        }
        validateFormat(request);
    }

    private static void validateFormat(ImportHsmKeyV2Request request) {
        boolean hasTr31Metadata = !isBlank(request.getTr31KeyUsage())
                || !isBlank(request.getTr31Algorithm())
                || !isBlank(request.getTr31ModeOfUse());
        switch (request.getKeyScheme()) {
            case TDES -> {
                if (!Set.of(112, 168).contains(request.getKeyLengthBits())) {
                    throw new IllegalArgumentException(
                            "TDES import requires keyLengthBits of 112 or 168");
                }
                if (hasTr31Metadata) throw tr31MetadataError();
            }
            case AES -> {
                if (!Set.of(128, 192, 256).contains(request.getKeyLengthBits())) {
                    throw new IllegalArgumentException(
                            "AES import requires keyLengthBits of 128, 192, or 256");
                }
                if (hasTr31Metadata) throw tr31MetadataError();
            }
            case TR31 -> {
                if (!Set.of(112, 128, 168, 192, 256).contains(request.getKeyLengthBits())
                        || isBlank(request.getTr31KeyUsage())
                        || isBlank(request.getTr31Algorithm())
                        || isBlank(request.getTr31ModeOfUse())) {
                    throw new IllegalArgumentException(
                            "TR-31 import requires a valid key length, usage, algorithm, and modeOfUse");
                }
            }
        }
    }

    private static IllegalArgumentException tr31MetadataError() {
        return new IllegalArgumentException(
                "TR-31 metadata is allowed only when keyScheme=TR31");
    }

    private static void validateKek(HsmKeyReferenceV2 kek) {
        LocalDate today = LocalDate.now();
        if (kek.getKeyType() != HsmKeyType.KEK
                || !EXTERNAL_KEK_SCOPES.contains(kek.getKeyScope())) {
            throw new IllegalArgumentException(
                    "kekKeyId must reference an external-stakeholder KEK");
        }
        if (kek.getStatus() != HsmKeyStatus.ACTIVE
                || kek.getEffectiveFrom().isAfter(today)
                || !kek.getExpiryDate().isAfter(today)) {
            throw new IllegalArgumentException(
                    "kekKeyId must reference an active KEK within its cryptoperiod");
        }
        if (isBlank(kek.getEncryptedKeyBlock())) {
            throw new IllegalArgumentException(
                    "The KEK has no CMS-HSM-protected key block");
        }
    }

    private static void validateImportedResult(ImportedKeyResult result,
            boolean tr31Expected) {
        if (result == null || isBlank(result.getCmsProtectedKeyBlock())
                || isBlank(result.getKeyCheckValue())
                || !result.getKeyCheckValue().matches("[0-9A-Fa-f]{4,16}")) {
            throw new HsmKeyGenerationException(
                    "HSM returned an invalid key-import response");
        }
        if (result.isTr31() != tr31Expected) {
            throw new HsmKeyGenerationException(
                    "HSM import response format does not match the request");
        }
    }

    private void recordFailureSafely(UUID institutionId, ImportHsmKeyV2Request request,
            String idempotencyHash, String fingerprint, KeyPurpose purpose,
            String expectedKcv, String calculatedKcv, KeyExchangeStatus status,
            String failureCode, String actor) {
        try {
            auditService.recordFailure(institutionId, request.getKekKeyId(),
                    request.getExternalRequestId(), idempotencyHash, fingerprint,
                    request.getKeyType(), request.getTargetScope(), purpose,
                    expectedKcv, calculatedKcv, status, failureCode, actor);
        } catch (RuntimeException auditFailure) {
            log.error("Could not persist sanitized key-exchange failure audit", auditFailure);
        }
    }

    private static String fingerprint(ImportHsmKeyV2Request request,
            KeyPurpose purpose, String expectedKcv) {
        String value = String.join("|", request.getExternalRequestId(),
                request.getKekKeyId().toString(), request.getKeyType().name(),
                request.getKeyScheme().name(), request.getKeyLengthBits().toString(),
                request.getTransportedKeyBlock(), expectedKcv,
                request.getTargetScope().name(), String.valueOf(request.getBinId()),
                purpose.name(), request.getEffectiveFrom().toString(),
                request.getExpiryDate().toString(),
                String.valueOf(request.getTr31KeyUsage()),
                String.valueOf(request.getTr31Algorithm()),
                String.valueOf(request.getTr31ModeOfUse()));
        return sha256(value);
    }

    private static String sha256(String value) {
        try {
            byte[] digest = MessageDigest.getInstance("SHA-256")
                    .digest(value.getBytes(StandardCharsets.UTF_8));
            return java.util.HexFormat.of().formatHex(digest);
        } catch (NoSuchAlgorithmException e) {
            throw new IllegalStateException("SHA-256 is unavailable", e);
        }
    }

    private static DataClassification defaultClassification(DataClassification value) {
        return value == null ? DataClassification.RESTRICTED : value;
    }
    private static String normalize(String value) {
        return value.trim().toUpperCase(Locale.ROOT);
    }
    private static String normalizeNullable(String value) {
        return isBlank(value) ? null : normalize(value);
    }
    private static String trimToNull(String value) {
        return isBlank(value) ? null : value.trim();
    }
    private static boolean isBlank(String value) {
        return value == null || value.isBlank();
    }
}
