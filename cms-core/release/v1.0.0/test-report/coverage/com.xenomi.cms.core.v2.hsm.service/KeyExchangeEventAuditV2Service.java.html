package com.xenomi.cms.core.v2.hsm.service;

import java.time.OffsetDateTime;
import java.util.UUID;

import com.xenomi.cms.core.v2.common.DatabaseUuidV7Generator;
import com.xenomi.cms.core.v2.hsm.entity.KeyExchangeEventV2;
import com.xenomi.cms.core.v2.hsm.model.HsmKeyScope;
import com.xenomi.cms.core.v2.hsm.model.HsmKeyType;
import com.xenomi.cms.core.v2.hsm.model.KeyExchangeDirection;
import com.xenomi.cms.core.v2.hsm.model.KeyExchangeStatus;
import com.xenomi.cms.core.v2.hsm.model.KeyPurpose;
import com.xenomi.cms.core.v2.hsm.repository.KeyExchangeEventV2Repository;

import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Propagation;
import org.springframework.transaction.annotation.Transactional;

@Service
public class KeyExchangeEventAuditV2Service {

    private final KeyExchangeEventV2Repository repository;
    private final DatabaseUuidV7Generator uuidGenerator;

    public KeyExchangeEventAuditV2Service(KeyExchangeEventV2Repository repository,
            DatabaseUuidV7Generator uuidGenerator) {
        this.repository = repository;
        this.uuidGenerator = uuidGenerator;
    }

    @Transactional(propagation = Propagation.REQUIRES_NEW)
    public void recordFailure(UUID institutionId, UUID kekKeyId, String externalRequestId,
            String idempotencyKeyHash, String requestFingerprint, HsmKeyType keyType,
            HsmKeyScope targetScope, KeyPurpose keyPurpose, String expectedKcv,
            String calculatedKcv, KeyExchangeStatus status, String failureCode,
            String actorId) {
        if (repository.findByInstitutionIdAndIdempotencyKeyHash(
                institutionId, idempotencyKeyHash).isPresent()
                || repository.findByInstitutionIdAndExternalRequestId(
                        institutionId, externalRequestId).isPresent()) {
            return;
        }
        repository.save(KeyExchangeEventV2.builder()
                .exchangeEventId(uuidGenerator.generate())
                .institutionId(institutionId).kekKeyId(kekKeyId)
                .externalRequestId(externalRequestId)
                .idempotencyKeyHash(idempotencyKeyHash)
                .requestFingerprint(requestFingerprint)
                .direction(KeyExchangeDirection.INBOUND)
                .workingKeyType(keyType).targetScope(targetScope)
                .keyPurpose(keyPurpose).expectedKcv(expectedKcv)
                .calculatedKcv(calculatedKcv).exchangeStatus(status)
                .failureCode(failureCode).completedAt(OffsetDateTime.now(java.time.ZoneOffset.UTC))
                .createdBy(actorId).build());
    }
}
