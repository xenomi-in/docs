package com.xenomi.cms.core.v2.hsm.service;

import java.time.OffsetDateTime;
import java.util.EnumMap;
import java.util.Map;
import java.util.Set;
import java.util.UUID;

import com.xenomi.cms.core.exception.ResourceNotFoundException;
import com.xenomi.cms.core.v2.bin.repository.BinConfigurationV2Repository;
import com.xenomi.cms.core.v2.common.DatabaseUuidV7Generator;
import com.xenomi.cms.core.v2.hsm.dto.BinHsmKeyAssignmentV2Response;
import com.xenomi.cms.core.v2.hsm.dto.CreateBinHsmKeyAssignmentV2Request;
import com.xenomi.cms.core.v2.hsm.dto.UpdateBinHsmKeyAssignmentV2Request;
import com.xenomi.cms.core.v2.hsm.entity.BinHsmKeyAssignmentV2;
import com.xenomi.cms.core.v2.hsm.model.BinHsmKeyAssignmentStatus;
import com.xenomi.cms.core.v2.hsm.model.HsmKeyStatus;
import com.xenomi.cms.core.v2.hsm.model.HsmKeyScope;
import com.xenomi.cms.core.v2.hsm.model.HsmKeyType;
import com.xenomi.cms.core.v2.hsm.model.KeyPurpose;
import com.xenomi.cms.core.v2.hsm.repository.BinHsmKeyAssignmentV2Repository;
import com.xenomi.cms.core.v2.institution.repository.HsmKeyReferenceV2Repository;
import com.xenomi.cms.core.v2.institution.entity.HsmKeyReferenceV2;
import com.xenomi.cms.core.v2.institution.repository.InstitutionSecurityPolicyV2Repository;

import org.springframework.data.domain.Page;
import org.springframework.data.domain.Pageable;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

@Service
public class BinHsmKeyAssignmentV2Service {
    private static final Set<HsmKeyStatus> ASSIGNABLE_KEYS = Set.of(
            HsmKeyStatus.PENDING_APPROVAL, HsmKeyStatus.SCHEDULED,
            HsmKeyStatus.ACTIVE, HsmKeyStatus.DECRYPT_ONLY);
    private static final Map<BinHsmKeyAssignmentStatus, Set<BinHsmKeyAssignmentStatus>> TRANSITIONS = transitions();

    private final BinConfigurationV2Repository binRepository;
    private final HsmKeyReferenceV2Repository keyRepository;
    private final BinHsmKeyAssignmentV2Repository assignmentRepository;
    private final InstitutionSecurityPolicyV2Repository securityPolicyRepository;
    private final DatabaseUuidV7Generator uuidGenerator;

    public BinHsmKeyAssignmentV2Service(BinConfigurationV2Repository binRepository,
            HsmKeyReferenceV2Repository keyRepository,
            BinHsmKeyAssignmentV2Repository assignmentRepository,
            InstitutionSecurityPolicyV2Repository securityPolicyRepository,
            DatabaseUuidV7Generator uuidGenerator) {
        this.binRepository = binRepository;
        this.keyRepository = keyRepository;
        this.assignmentRepository = assignmentRepository;
        this.securityPolicyRepository = securityPolicyRepository;
        this.uuidGenerator = uuidGenerator;
    }

    @Transactional
    public BinHsmKeyAssignmentV2Response create(UUID institutionId, UUID binId,
            CreateBinHsmKeyAssignmentV2Request request, String actorId) {
        requireBin(institutionId, binId, true);
        HsmKeyReferenceV2 key = keyRepository
                .findByKeyIdAndInstitutionId(request.getHsmKeyId(), institutionId)
                .orElseThrow(() -> new IllegalArgumentException(
                        "hsmKeyId must reference a key owned by the institution"));
        if (!ASSIGNABLE_KEYS.contains(key.getStatus())) {
            throw new IllegalArgumentException("hsmKeyId must reference an assignable key owned by the institution");
        }
        if (key.getKeyType() == HsmKeyType.KEK) {
            throw new IllegalArgumentException(
                    "KEK cannot be used as an operational BIN key");
        }
        boolean compatibleScope = key.getKeyScope() == HsmKeyScope.INSTITUTION
                || (key.getKeyScope() == HsmKeyScope.BIN && binId.equals(key.getBinId()));
        if (!compatibleScope) {
            throw new IllegalArgumentException("Only institution-scoped keys or keys scoped to this BIN can be assigned");
        }
        validateDates(request.getEffectiveFrom(), request.getEffectiveTo());
        KeyPurpose purpose = request.getKeyPurpose();
        if (purpose.scope() != HsmKeyScope.BIN) {
            throw new IllegalArgumentException("keyPurpose must be approved for BIN scope");
        }
        HsmKeyType requiredType = switch (purpose) {
            case CVV_GENERATION, CVV_VERIFICATION -> HsmKeyType.CVK;
            case PIN_VERIFICATION, PIN_GENERATION -> HsmKeyType.PVK;
            case PIN_BLOCK_ENCRYPTION -> HsmKeyType.ZPK;
            case EMV_KEY_DERIVATION, PERSONALISATION_DATA_GENERATION -> HsmKeyType.PGK;
            default -> throw new IllegalArgumentException("Unsupported BIN key purpose");
        };
        if (key.getKeyType() != requiredType) {
            throw new IllegalArgumentException(purpose + " requires a " + requiredType + " key");
        }
        int version = assignmentRepository.findMaximumVersion(binId, purpose) + 1;
        String actor = actorId.trim();
        BinHsmKeyAssignmentV2 assignment = BinHsmKeyAssignmentV2.builder()
                .assignmentId(uuidGenerator.generate()).institutionId(institutionId).binId(binId)
                .hsmKeyId(request.getHsmKeyId()).keyPurpose(purpose).assignmentVersion(version)
                .assignmentStatus(BinHsmKeyAssignmentStatus.PENDING_APPROVAL)
                .effectiveFrom(request.getEffectiveFrom()).effectiveTo(request.getEffectiveTo())
                .rotationGroup(trimToNull(request.getRotationGroup()))
                .approvalRequestId(request.getApprovalRequestId())
                .createdBy(actor).updatedBy(actor).build();
        return toResponse(assignmentRepository.save(assignment));
    }

    @Transactional(readOnly = true)
    public BinHsmKeyAssignmentV2Response get(UUID institutionId, UUID binId, UUID assignmentId) {
        requireBin(institutionId, binId, false);
        return toResponse(find(institutionId, binId, assignmentId));
    }

    @Transactional(readOnly = true)
    public Page<BinHsmKeyAssignmentV2Response> search(UUID institutionId, UUID binId,
            KeyPurpose purpose, BinHsmKeyAssignmentStatus status, Pageable pageable) {
        requireBin(institutionId, binId, false);
        return assignmentRepository.search(institutionId, binId, purpose, status, pageable)
                .map(BinHsmKeyAssignmentV2Service::toResponse);
    }

    @Transactional
    public BinHsmKeyAssignmentV2Response changeStatus(UUID institutionId, UUID binId,
            UUID assignmentId, UpdateBinHsmKeyAssignmentV2Request request, String actorId) {
        requireBin(institutionId, binId, true);
        BinHsmKeyAssignmentV2 assignment = find(institutionId, binId, assignmentId);
        BinHsmKeyAssignmentStatus current = assignment.getAssignmentStatus();
        BinHsmKeyAssignmentStatus requested = request.getStatus();
        if (!TRANSITIONS.getOrDefault(current, Set.of()).contains(requested)) {
            throw new IllegalArgumentException("Invalid BIN key assignment transition: " + current + " -> " + requested);
        }
        OffsetDateTime now = OffsetDateTime.now(java.time.ZoneOffset.UTC);
        if (requested == BinHsmKeyAssignmentStatus.SCHEDULED
                && !assignment.getEffectiveFrom().isAfter(now)) {
            throw new IllegalArgumentException("SCHEDULED assignments require a future effectiveFrom");
        }
        if (requested == BinHsmKeyAssignmentStatus.ACTIVE
                && (assignment.getEffectiveFrom().isAfter(now)
                    || (assignment.getEffectiveTo() != null && !assignment.getEffectiveTo().isAfter(now)))) {
            throw new IllegalArgumentException("ACTIVE assignments must be within their effective period");
        }
        if (requested == BinHsmKeyAssignmentStatus.ACTIVE) {
            HsmKeyReferenceV2 key = keyRepository
                    .findByKeyIdAndInstitutionId(assignment.getHsmKeyId(), institutionId)
                    .orElseThrow(() -> new ResourceNotFoundException(
                            "HSM key", assignment.getHsmKeyId().toString()));
            if (key.getStatus() != HsmKeyStatus.ACTIVE) {
                throw new IllegalArgumentException("The referenced HSM key must be ACTIVE before assignment activation");
            }
        }
        UUID approvalId = request.getApprovalRequestId() != null
                ? request.getApprovalRequestId() : assignment.getApprovalRequestId();
        boolean activationFromPending = current == BinHsmKeyAssignmentStatus.PENDING_APPROVAL
                && (requested == BinHsmKeyAssignmentStatus.SCHEDULED
                    || requested == BinHsmKeyAssignmentStatus.ACTIVE);
        boolean makerChecker = securityPolicyRepository.findById(institutionId)
                .map(policy -> policy.isMakerCheckerEnabled()).orElse(true);
        if (activationFromPending && makerChecker && approvalId == null) {
            throw new IllegalArgumentException("approvalRequestId is required before activation");
        }
        if (requested == BinHsmKeyAssignmentStatus.SCHEDULED
                && assignmentRepository.countOtherAssignmentsInStatuses(binId,
                        assignment.getKeyPurpose(), assignmentId,
                        Set.of(BinHsmKeyAssignmentStatus.SCHEDULED)) > 0) {
            throw new IllegalArgumentException("Another scheduled assignment already exists for this BIN key purpose");
        }
        if (requested == BinHsmKeyAssignmentStatus.ACTIVE) {
            assignmentRepository
                    .findFirstByBinIdAndKeyPurposeAndAssignmentStatusAndAssignmentIdNot(
                            binId, assignment.getKeyPurpose(),
                            BinHsmKeyAssignmentStatus.ACTIVE, assignmentId)
                    .ifPresent(previous -> {
                        previous.setAssignmentStatus(BinHsmKeyAssignmentStatus.VERIFY_ONLY);
                        previous.setEffectiveTo(now);
                        previous.setUpdatedBy(actorId.trim());
                        assignmentRepository.save(previous);
                    });
        }
        OffsetDateTime effectiveTo = request.getEffectiveTo();
        if ((requested == BinHsmKeyAssignmentStatus.RETIRED
                || requested == BinHsmKeyAssignmentStatus.REVOKED) && effectiveTo == null) {
            effectiveTo = OffsetDateTime.now(java.time.ZoneOffset.UTC);
        }
        validateDates(assignment.getEffectiveFrom(), effectiveTo);
        assignment.setAssignmentStatus(requested);
        assignment.setEffectiveTo(effectiveTo);
        assignment.setApprovalRequestId(approvalId);
        assignment.setUpdatedBy(actorId.trim());
        return toResponse(assignmentRepository.save(assignment));
    }

    private void requireBin(UUID institutionId, UUID binId, boolean lock) {
        boolean found = lock
                ? binRepository.findByTenantAndIdForUpdate(institutionId, binId).isPresent()
                : binRepository.findByBinIdAndInstitutionId(binId, institutionId).isPresent();
        if (!found) throw new ResourceNotFoundException("BIN configuration", binId.toString());
    }
    private BinHsmKeyAssignmentV2 find(UUID institutionId, UUID binId, UUID assignmentId) {
        return assignmentRepository.findByAssignmentIdAndInstitutionIdAndBinId(assignmentId, institutionId, binId)
                .orElseThrow(() -> new ResourceNotFoundException("BIN HSM key assignment", assignmentId.toString()));
    }
    private static void validateDates(OffsetDateTime from, OffsetDateTime to) {
        if (to != null && !to.isAfter(from)) throw new IllegalArgumentException("effectiveTo must be later than effectiveFrom");
    }
    private static String trimToNull(String value) { return value == null || value.isBlank() ? null : value.trim(); }

    private static BinHsmKeyAssignmentV2Response toResponse(BinHsmKeyAssignmentV2 a) {
        return BinHsmKeyAssignmentV2Response.builder().assignmentId(a.getAssignmentId())
                .institutionId(a.getInstitutionId()).binId(a.getBinId()).hsmKeyId(a.getHsmKeyId())
                .keyPurpose(a.getKeyPurpose()).assignmentVersion(a.getAssignmentVersion())
                .assignmentStatus(a.getAssignmentStatus()).effectiveFrom(a.getEffectiveFrom())
                .effectiveTo(a.getEffectiveTo()).rotationGroup(a.getRotationGroup())
                .approvalRequestId(a.getApprovalRequestId()).createdAt(a.getCreatedAt())
                .updatedAt(a.getUpdatedAt()).createdBy(a.getCreatedBy()).updatedBy(a.getUpdatedBy()).build();
    }

    private static Map<BinHsmKeyAssignmentStatus, Set<BinHsmKeyAssignmentStatus>> transitions() {
        Map<BinHsmKeyAssignmentStatus, Set<BinHsmKeyAssignmentStatus>> m = new EnumMap<>(BinHsmKeyAssignmentStatus.class);
        m.put(BinHsmKeyAssignmentStatus.PENDING_APPROVAL, Set.of(BinHsmKeyAssignmentStatus.SCHEDULED, BinHsmKeyAssignmentStatus.ACTIVE, BinHsmKeyAssignmentStatus.REVOKED));
        m.put(BinHsmKeyAssignmentStatus.SCHEDULED, Set.of(BinHsmKeyAssignmentStatus.ACTIVE, BinHsmKeyAssignmentStatus.SUSPENDED, BinHsmKeyAssignmentStatus.REVOKED));
        m.put(BinHsmKeyAssignmentStatus.ACTIVE, Set.of(BinHsmKeyAssignmentStatus.VERIFY_ONLY, BinHsmKeyAssignmentStatus.SUSPENDED, BinHsmKeyAssignmentStatus.RETIRED, BinHsmKeyAssignmentStatus.REVOKED));
        m.put(BinHsmKeyAssignmentStatus.VERIFY_ONLY, Set.of(BinHsmKeyAssignmentStatus.RETIRED, BinHsmKeyAssignmentStatus.REVOKED));
        m.put(BinHsmKeyAssignmentStatus.SUSPENDED, Set.of(BinHsmKeyAssignmentStatus.ACTIVE, BinHsmKeyAssignmentStatus.RETIRED, BinHsmKeyAssignmentStatus.REVOKED));
        m.put(BinHsmKeyAssignmentStatus.RETIRED, Set.of()); m.put(BinHsmKeyAssignmentStatus.REVOKED, Set.of());
        return Map.copyOf(m);
    }
}
