package com.xenomi.cms.core.v2.switchsync;

import com.xenomi.cms.core.api.ApiResponseHelper;
import com.xenomi.cms.core.exception.ResourceNotFoundException;
import io.swagger.v3.oas.annotations.Operation;
import io.swagger.v3.oas.annotations.tags.Tag;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.web.bind.annotation.*;
import java.util.*;

@RestController
@RequestMapping("/cms/api/v2/institutions/{institutionId}/cards/{cardId}/switch-sync")
@Tag(name = "V2 - Cards", description = "Tenant-scoped card issuance and lifecycle")
public class SwitchSyncController {
    private final JdbcTemplate db; private final boolean enabled;
    public SwitchSyncController(JdbcTemplate db,@Value("${cms.switch.sync.enabled:false}") boolean enabled) { this.db=db; this.enabled=enabled; }
    @GetMapping
    @Operation(summary="Read Switch registration and synchronization status",description="Lifecycle success is a CMS commit. Check this endpoint for Switch installation. No PAN, encrypted envelope, or key references are returned.")
    public Object get(@PathVariable UUID institutionId,@PathVariable UUID cardId) {
        if(db.queryForList("select 1 from cms_v2.card where institution_id=? and card_id=?",institutionId,cardId).isEmpty())
            throw new ResourceNotFoundException("Card not found");
        var result=new LinkedHashMap<String,Object>();
        result.put("enabled",enabled); result.put("cardId",cardId);
        var revisions=db.queryForList("select applied_revision from cms_v2.switch_card_sync where institution_id=? and card_id=?",Long.class,institutionId,cardId);
        result.put("appliedRevision",revisions.isEmpty()?0:revisions.getFirst());
        result.put("pendingCommands",db.queryForList("select command_id,status,attempts,last_error,next_attempt_at from cms_v2.switch_sync_command where institution_id=? and card_id=? and status<>'APPLIED' order by sequence_id",institutionId,cardId));
        result.put("institutionBacklog",db.queryForObject("select count(*) from cms_v2.switch_sync_command where institution_id=? and status<>'APPLIED'",Long.class,institutionId));
        return ApiResponseHelper.ok(result);
    }
}
