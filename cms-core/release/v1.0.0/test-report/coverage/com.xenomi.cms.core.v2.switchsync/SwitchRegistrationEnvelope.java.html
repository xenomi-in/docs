package com.xenomi.cms.core.v2.switchsync;

import com.fasterxml.jackson.databind.ObjectMapper;
import com.nimbusds.jose.*;
import com.nimbusds.jose.crypto.RSAEncrypter;
import com.xenomi.cms.core.v2.card.repository.CardV2Repository;
import com.xenomi.cms.core.v2.card.security.CmsCardSecurityService;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.boot.autoconfigure.condition.ConditionalOnProperty;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.stereotype.Component;
import java.nio.file.*;
import java.security.cert.*;
import java.security.interfaces.RSAPublicKey;
import java.time.Instant;
import java.time.format.DateTimeFormatter;
import java.util.*;

@Component
@ConditionalOnProperty(name="cms.switch.sync.enabled",havingValue="true")
public class SwitchRegistrationEnvelope {
    private final CardV2Repository cards; private final CmsCardSecurityService security;
    private final JdbcTemplate db; private final ObjectMapper json; private final X509Certificate certificate;
    public SwitchRegistrationEnvelope(CardV2Repository cards,CmsCardSecurityService security,JdbcTemplate db,ObjectMapper json,
            @Value("${cms.switch.sync.recipient-certificate}") String path) throws Exception {
        this.cards=cards; this.security=security; this.db=db; this.json=json;
        var resource = path.startsWith("classpath:") || path.startsWith("file:")
                ? new org.springframework.core.io.DefaultResourceLoader().getResource(path)
                : new org.springframework.core.io.FileSystemResource(path);
        try(var in=resource.getInputStream()) {
            certificate=(X509Certificate)CertificateFactory.getInstance("X.509").generateCertificate(in);
        }
        if(!(certificate.getPublicKey() instanceof RSAPublicKey key) || key.getModulus().bitLength()<2048)
            throw new IllegalArgumentException("Switch recipient must use RSA 2048 bits or stronger");
    }
    public String create(UUID tenant,UUID cardId,UUID command) {
        var card=cards.findByCardIdAndInstitutionId(cardId,tenant).orElseThrow();
        var product=db.queryForMap("select funding_type,card_network from cms_v2.card_product where institution_id=? and product_id=?",tenant,card.getProductId());
        var material=new LinkedHashMap<String,Object>();
        material.put("institutionId",tenant); material.put("cardId",cardId); material.put("commandId",command);
        material.put("expiresAt",Instant.now().plusSeconds(240).toString());
        material.put("expiryYYMM",card.getExpiryDate().format(DateTimeFormatter.ofPattern("yyMM")));
        material.put("cardType",switch(product.get("funding_type").toString()) { case "CREDIT"->"C"; case "DEBIT"->"D"; case "PREPAID"->"P"; default->throw new IllegalStateException("Unsupported funding type"); });
        material.put("cardScheme",switch(product.get("card_network").toString()) { case "VISA"->"V"; case "MASTERCARD"->"M"; case "RUPAY"->"R"; default->throw new IllegalStateException("Unsupported card network"); });
        return security.withDecryptedPan(tenant,card,pan->{
            try {
                certificate.checkValidity(); material.put("pan",pan);
                var jwe=new JWEObject(new JWEHeader.Builder(JWEAlgorithm.RSA_OAEP_256,EncryptionMethod.A256GCM).build(),new Payload(json.writeValueAsString(material)));
                jwe.encrypt(new RSAEncrypter((RSAPublicKey)certificate.getPublicKey())); return jwe.serialize();
            } catch(Exception ex) { throw new IllegalStateException("Switch envelope preparation failed"); }
            finally { material.clear(); }
        });
    }
}
