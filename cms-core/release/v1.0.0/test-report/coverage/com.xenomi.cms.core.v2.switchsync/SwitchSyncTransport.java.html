package com.xenomi.cms.core.v2.switchsync;

import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.boot.autoconfigure.condition.ConditionalOnProperty;
import org.springframework.stereotype.Component;
import org.springframework.http.client.JdkClientHttpRequestFactory;
import org.springframework.web.client.RestClient;
import org.springframework.web.client.RestClientResponseException;
import org.springframework.http.MediaType;
import org.springframework.util.LinkedMultiValueMap;
import javax.net.ssl.*;
import java.net.URI;
import java.net.http.HttpClient;
import java.nio.file.Files;
import java.nio.file.Path;
import java.security.KeyStore;
import java.time.Duration;
import java.util.UUID;

/** Dedicated authenticated service transport. Never logs bodies or propagates remote error bodies. */
@Component
@ConditionalOnProperty(name="cms.switch.sync.enabled",havingValue="true")
public class SwitchSyncTransport {
    private static final org.slf4j.Logger log = org.slf4j.LoggerFactory.getLogger(SwitchSyncTransport.class);
    private static final ObjectMapper JSON = new ObjectMapper();
    private final RestClient client;
    private final String baseUrl, tokenUrl, clientId, secret;
    public SwitchSyncTransport(@Value("${cms.switch.base-url}") String baseUrl,
            @Value("${cms.switch.sync.token-url}") String tokenUrl,
            @Value("${cms.switch.sync.client-id:cms-switch-client}") String clientId,
            @Value("${cms.switch.sync.client-secret}") String secret,
            @Value("${cms.switch.sync.keystore:}") String keystore,
            @Value("${cms.switch.sync.keystore-password:}") String password,
            @Value("${cms.switch.sync.truststore:}") String truststore,
            @Value("${cms.switch.sync.truststore-password:}") String trustPassword,
            @Value("${cms.switch.sync.tls-enabled:true}") boolean tlsEnabled,
            @Value("${cms.switch.timeout-ms:3000}") int timeout) throws Exception {
        validateEndpoint(baseUrl,tlsEnabled); validateEndpoint(tokenUrl,tlsEnabled);
        this.baseUrl=baseUrl.replaceAll("/+$",""); this.tokenUrl=tokenUrl; this.clientId=clientId; this.secret=secret;
        var httpBuilder=HttpClient.newBuilder().connectTimeout(Duration.ofMillis(timeout))
                .followRedirects(HttpClient.Redirect.NEVER);
        if(tlsEnabled) {
        KeyStore keys=KeyStore.getInstance("PKCS12"), trust=KeyStore.getInstance("PKCS12");
        try(var in=Files.newInputStream(Path.of(keystore))) { keys.load(in,password.toCharArray()); }
        try(var in=Files.newInputStream(Path.of(truststore))) { trust.load(in,trustPassword.toCharArray()); }
        var km=KeyManagerFactory.getInstance(KeyManagerFactory.getDefaultAlgorithm()); km.init(keys,password.toCharArray());
        var tm=TrustManagerFactory.getInstance(TrustManagerFactory.getDefaultAlgorithm()); tm.init(trust);
        var ssl=SSLContext.getInstance("TLS"); ssl.init(km.getKeyManagers(),tm.getTrustManagers(),null);
        httpBuilder.sslContext(ssl);
        }
        var http=httpBuilder.build();
        var factory=new JdkClientHttpRequestFactory(http); factory.setReadTimeout(Duration.ofMillis(timeout));
        client=RestClient.builder().requestFactory(factory).requestInterceptor((request, body, execution) -> {
            // Log metadata only: the token body contains credentials and registration contains JWE.
            String endpoint = request.getURI().getScheme() + "://" + request.getURI().getAuthority()
                    + request.getURI().getPath();
            String command = request.getHeaders().getFirst("Idempotency-Key");
            long start = System.nanoTime();
            log.info("Switch sync HTTP request method={} endpoint={} commandId={} bytes={}",
                    request.getMethod(), endpoint, command, body.length);
            try {
                var response = execution.execute(request, body);
                log.info("Switch sync HTTP response method={} endpoint={} commandId={} status={} elapsedMs={}",
                        request.getMethod(), endpoint, command, response.getStatusCode().value(),
                        (System.nanoTime() - start) / 1_000_000);
                return response;
            } catch (java.io.IOException | RuntimeException ex) {
                log.warn("Switch sync HTTP failure method={} endpoint={} commandId={} exceptionType={}",
                        request.getMethod(), endpoint, command, ex.getClass().getSimpleName());
                throw ex;
            }
        }).build();
    }
    static void requireHttps(String url) {
        validateEndpoint(url,true);
    }
    static void validateEndpoint(String url,boolean tlsEnabled) {
        URI uri=URI.create(url);
        boolean schemeAllowed="https".equals(uri.getScheme()) || (!tlsEnabled && "http".equals(uri.getScheme()));
        if(!schemeAllowed || uri.getHost()==null || uri.getUserInfo()!=null || uri.getFragment()!=null)
            throw new IllegalArgumentException("Invalid Switch/token endpoint; TLS mode requires HTTPS");
    }
    private String token() {
        var form=new LinkedMultiValueMap<String,String>();
        form.add("grant_type","client_credentials"); form.add("client_id",clientId); form.add("client_secret",secret);
        form.add("scope","switch:card-authorization:write");
        JsonNode response;
        try {
            response=parseJson(client.post().uri(tokenUrl)
                    .contentType(MediaType.APPLICATION_FORM_URLENCODED).body(form)
                    .retrieve().body(String.class));
        } catch (RestClientResponseException ex) {
            log.warn("Switch sync token acquisition rejected httpStatus={} oauthError={}",
                    ex.getStatusCode().value(), safeOAuthError(ex.getResponseBodyAsString()));
            throw ex;
        }
        if(response==null || response.path("access_token").asText().isBlank()) throw new IllegalStateException("Missing service token");
        return response.path("access_token").asText();
    }
    static String safeOAuthError(String body) {
        try {
            String code = new com.fasterxml.jackson.databind.ObjectMapper().readTree(body).path("error").asText();
            return switch (code) {
                case "invalid_client", "invalid_scope", "unauthorized_client", "invalid_grant",
                     "invalid_request", "unsupported_grant_type", "server_error", "temporarily_unavailable" -> code;
                default -> "UNRECOGNIZED_ERROR";
            };
        } catch (Exception ignored) {
            return "UNRECOGNIZED_ERROR";
        }
    }
    public JsonNode receipt(UUID tenant,UUID command) {
        try { return parseJson(client.get().uri(baseUrl+"/internal/v1/institutions/"+tenant+"/card-commands/"+command)
                .headers(h->h.setBearerAuth(token())).retrieve().body(String.class)); }
        catch(RestClientResponseException ex) { if(ex.getStatusCode().value()==404) return null; throw ex; }
    }
    public JsonNode pinReceipt(UUID tenant, UUID operation) {
        try {
            return parseJson(client.get().uri(baseUrl+"/internal/v1/institutions/"+tenant+"/pin-operations/"+operation)
                    .headers(h->h.setBearerAuth(token())).retrieve().body(String.class));
        } catch (RestClientResponseException ex) {
            if (ex.getStatusCode().value()==404) return null;
            throw ex;
        }
    }
    public JsonNode sendPin(UUID tenant, UUID family, UUID operation, JsonNode body) {
        return parseJson(client.post().uri(baseUrl+"/internal/v1/institutions/"+tenant+"/pan-families/"+family+"/pin-operations")
                .headers(h->h.setBearerAuth(token())).header("Idempotency-Key",operation.toString())
                .contentType(MediaType.APPLICATION_JSON).body(body.toString()).retrieve().body(String.class));
    }
    public JsonNode send(UUID tenant,UUID card,UUID command,String kind,JsonNode body) {
        return parseJson(client.put().uri(baseUrl+"/internal/v1/institutions/"+tenant+"/cards/"+card+
                        ("REGISTER".equals(kind)?"/registration":"/authorization-profile"))
                .headers(h->h.setBearerAuth(token())).header("Idempotency-Key",command.toString())
                .contentType(MediaType.APPLICATION_JSON).body(body.toString()).retrieve().body(String.class));
    }
    private static JsonNode parseJson(String body) {
        if (body == null) return null;
        try {
            return JSON.readTree(body);
        } catch (Exception exception) {
            throw new IllegalStateException("Invalid JSON response from Switch", exception);
        }
    }
}
