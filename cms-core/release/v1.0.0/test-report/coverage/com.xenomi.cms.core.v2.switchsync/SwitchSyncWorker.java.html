package com.xenomi.cms.core.v2.switchsync;

import com.xenomi.cms.core.v2.card.ProductionRequestEligibility;
import com.fasterxml.jackson.databind.*;
import com.fasterxml.jackson.databind.node.ObjectNode;
import org.springframework.boot.autoconfigure.condition.ConditionalOnProperty;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.scheduling.annotation.*;
import org.springframework.stereotype.Component;
import org.springframework.transaction.PlatformTransactionManager;
import org.springframework.transaction.support.TransactionTemplate;
import org.springframework.web.client.RestClientResponseException;
import java.util.*;

/** Persist request before sending it. A crash after remote commit replays the identical command. */
@Component
@EnableScheduling
@ConditionalOnProperty(name="cms.switch.sync.enabled",havingValue="true")
public class SwitchSyncWorker {
    private static final org.slf4j.Logger log = org.slf4j.LoggerFactory.getLogger(SwitchSyncWorker.class);
    private final JdbcTemplate db; private final ObjectMapper json;
    private final SwitchRegistrationEnvelope envelope; private final SwitchSyncTransport transport;
    private final TransactionTemplate tx;
    private long nextDiagnosticAt;
    @jakarta.annotation.PostConstruct
    void started() {
        log.info("Switch sync worker started diagnosticsVersion=3; ordering is strict per card and independent cards do not block each other");
    }
    public SwitchSyncWorker(JdbcTemplate db,ObjectMapper json,SwitchRegistrationEnvelope envelope,
            SwitchSyncTransport transport,PlatformTransactionManager transactions) {
        this.db=db; this.json=json; this.envelope=envelope; this.transport=transport; tx=new TransactionTemplate(transactions);
    }
    @Scheduled(fixedDelayString="${cms.switch.sync.poll-ms:1000}")
    public void poll() {
        reportQueue();
        // Separate transactions ensure that the exact encrypted request survives a remote timeout.
        tx.executeWithoutResult(s->process(false));
        tx.executeWithoutResult(s->process(true));
    }
    private void reportQueue() {
        long now = System.nanoTime();
        if (nextDiagnosticAt != 0 && now - nextDiagnosticAt < 0) return;
        nextDiagnosticAt = now + java.util.concurrent.TimeUnit.SECONDS.toNanos(60);
        try {
            var counts = db.queryForMap("""
                select count(*) filter (where status='FAILED') as failed,
                       count(*) filter (where status='PENDING') as pending,
                       count(*) filter (where status='PENDING' and next_attempt_at<=now()) as due
                from cms_v2.switch_sync_command where status<>'APPLIED'
                """);
            log.info("Switch sync poll alive diagnosticsVersion=3 pending={} due={} failed={} (due commands may be blocked only by earlier commands for the same card)",
                    counts.get("pending"), counts.get("due"), counts.get("failed"));
        } catch (Exception ex) {
            log.warn("Switch sync queue diagnostics failed exceptionType={}", ex.getClass().getSimpleName());
        }
    }
    private void process(boolean sending) {
        if(!Boolean.TRUE.equals(db.queryForObject("select pg_try_advisory_xact_lock(724816239)",Boolean.class))) {
            log.debug("Switch sync poll skipped: another transaction holds the worker lock");
            return;
        }
        int deferredForPin = 0;
        // A PIN wait must not occupy the only slot. Step past it and take the next card.
        for (int examined = 0; examined < 32; examined++) {
        var rows=db.queryForList("""
            select q.* from cms_v2.switch_sync_command q
            where q.status='PENDING' and q.next_attempt_at<=now()
            and not exists(select 1 from cms_v2.switch_sync_command prior
                where prior.institution_id=q.institution_id
                and prior.status<>'APPLIED'
                and ((prior.card_id=q.card_id and prior.transaction_id<>q.transaction_id and prior.sequence_id<q.sequence_id)
                  or (prior.transaction_id=q.transaction_id and
                    (prior.card_id=q.card_id or prior.snapshot->>'panFamilyId'=q.snapshot->>'panFamilyId') and
                    (case when prior.snapshot#>>'{profile,effectiveStatus}'='ACTIVE' then 1 else 0 end,prior.sequence_id)
                    < (case when q.snapshot#>>'{profile,effectiveStatus}'='ACTIVE' then 1 else 0 end,q.sequence_id))))
            order by q.sequence_id limit 1 for update
            """);
        if(rows.isEmpty()) { logPinDeferral(deferredForPin); return; }
        var q=rows.getFirst(); long sequence=((Number)q.get("sequence_id")).longValue();
        UUID tenant=(UUID)q.get("institution_id"),card=(UUID)q.get("card_id"),command=(UUID)q.get("command_id");
        String stage = "PREPARE";
        try {
            if(q.get("request_body")==null) {
                if(sending) { logPinDeferral(deferredForPin); return; }
                long revision=db.queryForObject("select applied_revision from cms_v2.switch_card_sync where institution_id=? and card_id=?",Long.class,tenant,card);
                ObjectNode snapshot=(ObjectNode)json.readTree(q.get("snapshot").toString());
                stage = "NORMALIZE_PROFILE";
                ObjectNode profile=normalize((ObjectNode)snapshot.get("profile"));
                if(revision>0 && enablesUse(profile) && waitingForPinEnrollment(tenant,card)) {
                    postponeEnrollment(sequence); deferredForPin++; continue;
                }
                log.info("Switch sync command selected institutionId={} cardId={} commandId={} phase=PREPARE previousAttempts={}",
                        tenant, card, command, q.get("attempts"));
                ObjectNode body=json.createObjectNode(); body.put("commandId",command.toString());
                body.set("physicalSequence",snapshot.get("physicalSequence"));
                String kind=revision==0?"REGISTER":"UPDATE";
                if(revision==0) {
                    body.set("accountReferenceId",snapshot.get("accountReferenceId"));
                    body.set("panFamilyId",snapshot.get("panFamilyId"));
                    body.set("predecessorCardId",snapshot.get("predecessorCardId"));
                    profile.put("effectiveStatus","PENDING"); profile.putNull("statusReason");
                    profile.put("physicalEnabled",false); profile.put("digitalEnabled",false); profile.put("lifecycleVersion",1);
                    stage = "ENCRYPT_REGISTRATION";
                    body.put("protectedMaterial",envelope.create(tenant,card,command));
                } else { body.put("expectedRevision",revision); body.put("authorizationRevision",Math.addExact(revision,1)); }
                body.set("profile",profile);
                db.update("update cms_v2.switch_sync_command set request_body=?::jsonb,request_kind=?,prepared_at=now() where sequence_id=?",body.toString(),kind,sequence);
                logPinDeferral(deferredForPin); return;
            }
            if(!sending) { logPinDeferral(deferredForPin); return; }
            JsonNode body=json.readTree(q.get("request_body").toString());
            String kind=q.get("request_kind").toString();
            if("UPDATE".equals(kind) && enablesUse(body.path("profile")) && waitingForPinEnrollment(tenant,card)) {
                postponeEnrollment(sequence); deferredForPin++; continue;
            }
            log.info("Switch sync command selected institutionId={} cardId={} commandId={} phase=SEND previousAttempts={}",
                    tenant, card, command, q.get("attempts"));
            stage = "RECEIPT_LOOKUP";
            JsonNode receipt=transport.receipt(tenant,command);
            if(receipt==null && "REGISTER".equals(kind)) {
                int cleared=db.update("""
                    update cms_v2.switch_sync_command
                    set request_body=null,request_kind=null,prepared_at=null,last_error=null,next_attempt_at=now()
                    where sequence_id=? and request_kind='REGISTER'
                      and (prepared_at is null or prepared_at<=now()-interval '180 seconds')
                    """,sequence);
                if(cleared==1) {
                    log.info("Switch registration envelope expired and will be regenerated institutionId={} cardId={} commandId={}",
                            tenant,card,command);
                    logPinDeferral(deferredForPin);
                    return;
                }
            }
            if(receipt==null) {
                stage = kind;
                receipt=transport.send(tenant,card,command,kind,body);
            }
            stage = "VALIDATE_RECEIPT";
            long expected="REGISTER".equals(kind)?1:body.path("authorizationRevision").asLong();
            if(receipt==null || !receipt.path("applied").asBoolean() || receipt.path("appliedRevision").asLong()!=expected
                    || !command.toString().equals(receipt.path("commandId").asText())
                    || !card.toString().equals(receipt.path("cardId").asText())
                    || !body.path("profile").path("effectiveStatus").asText().equals(receipt.path("effectiveStatus").asText()))
                throw new IllegalStateException("Invalid Switch receipt");
            db.update("update cms_v2.switch_card_sync set applied_revision=? where institution_id=? and card_id=?",expected,tenant,card);
            if("REGISTER".equals(kind)) {
                int promoted=ProductionRequestEligibility.promoteAfterRegistration(db,tenant,card);
                if(promoted>0) log.info("Production request eligible after Switch registration institutionId={} cardId={} count={}",tenant,card,promoted);
                db.update("update cms_v2.switch_sync_command set command_id=?,request_body=null,request_kind=null,prepared_at=null,attempts=0,last_error=null where sequence_id=?",UUID.randomUUID(),sequence);
            } else {
                db.update("update cms_v2.switch_sync_command set status='APPLIED',receipt=?::jsonb,request_body=null,prepared_at=null,applied_at=now(),last_error=null where sequence_id=?",receipt.toString(),sequence);
                db.update("update cms_v2.card_control_profile set profile_status='ACTIVE' where institution_id=? and card_id=? and control_version=? and profile_status in ('PENDING_SWITCH','PENDING_RETRY')",tenant,card,body.path("profile").path("controlVersion").asInt());
            }
            logPinDeferral(deferredForPin);
            return;
        } catch(Exception ex) {
            int attempts=((Number)q.get("attempts")).intValue()+1;
            int code=ex instanceof RestClientResponseException http?http.getStatusCode().value():0;
            boolean permanent=code>=400 && code<500 && code!=408 && code!=429;
            log.warn("Switch sync command failed institutionId={} cardId={} commandId={} stage={} attempt={} httpStatus={} exceptionType={} state={}",
                    tenant, card, command, stage, attempts, code, ex.getClass().getSimpleName(), permanent ? "FAILED" : "PENDING");
            // Do not retain remote bodies, plaintext material, tokens, or exception messages.
            db.update("update cms_v2.switch_sync_command set attempts=?,status=?,last_error=?,next_attempt_at=now()+make_interval(secs=>?) where sequence_id=?",
                    attempts,permanent?"FAILED":"PENDING",code==0?"PREPARATION_OR_TRANSPORT_FAILURE":"HTTP_"+code,
                    Math.min(300,5*(1<<Math.min(attempts,6))),sequence);
            logPinDeferral(deferredForPin);
            return;
        }
        }
        logPinDeferral(deferredForPin);
    }
    private void logPinDeferral(int deferredForPin) {
        if (deferredForPin > 0) log.info("Switch sync deferred {} command(s) for PIN enrollment and continued with the next card", deferredForPin);
    }
    static ObjectNode normalize(ObjectNode profile) {
        if(profile==null || !profile.path("controls").isArray() || profile.path("controls").size()!=10
                || profile.path("controlVersion").asInt()<1) throw new IllegalArgumentException("Ten controls are required");
        String currency=profile.path("currencyCode").asText();
        profile.put("currencyCode",numericCurrency(currency));
        Set<String> dimensions=new HashSet<>();
        for(JsonNode value:profile.withArray("controls")) {
            ObjectNode control=(ObjectNode)value;
            if(!currency.equals(control.path("currencyCode").asText()) || !dimensions.add(control.path("geography").asText()+":"+control.path("channel").asText()))
                throw new IllegalArgumentException("Control currency or dimension mismatch");
            control.remove("currencyCode");
        }
        return profile;
    }
    private boolean waitingForPinEnrollment(UUID tenant,UUID card) {
        return Boolean.TRUE.equals(db.queryForObject("""
            select exists(select 1 from cms_v2.pin_data p join cms_v2.card c
              on c.institution_id=p.institution_id and c.pan_allocation_id=p.pan_allocation_id
              where c.institution_id=? and c.card_id=? and p.pin_authority='SWITCH'
                and p.switch_pin_profile is not null and p.switch_policy_version=0)
            """,Boolean.class,tenant,card));
    }
    private void postponeEnrollment(long sequence) {
        db.update("update cms_v2.switch_sync_command set next_attempt_at=now()+interval '5 seconds' where sequence_id=?",sequence);
    }
    private static boolean enablesUse(JsonNode profile) {
        return profile.path("physicalEnabled").asBoolean() || profile.path("digitalEnabled").asBoolean();
    }
    static String numericCurrency(String alphabetic) {
        return String.format(Locale.ROOT,"%03d",Currency.getInstance(alphabetic).getNumericCode());
    }
}
