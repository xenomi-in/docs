/** Switch is the single writer. Durable state contains metadata only; network IO owns no DB transaction. */
@Service
public class SwitchPinService {
    private static final Set<String> CUSTOMER_OPERATIONS=Set.of("INITIALIZE_SELECTED_PIN","CHANGE_PIN","RESET_PIN");
    private static final Set<String> TERMINAL=Set.of("COMPLETED","FAILED","REJECTED","UNSUPPORTED");

    private final TransactionTemplate tx;
    private final ObjectProvider<SwitchSyncTransport> transports;

    public SwitchPinService(JdbcTemplate db, ObjectMapper json, PlatformTransactionManager tm,
            ObjectProvider<SwitchSyncTransport> transports, SwitchPinEnvelope envelope,
            @Value("${cms.switch.pin.enabled:${cms.switch.sync.enabled:false}}") boolean enabled) {
        this.db=db; this.json=json; tx=new TransactionTemplate(tm);
        this.transports=transports; this.envelope=envelope; this.enabled=enabled;

    public record OperationResult(UUID operationId, UUID institutionId, UUID panFamilyReference,

    /** Explicit retry after a definite configuration rejection; never migrates an existing CMS-owned PIN. */
    public OperationResult configureProfile(UUID tenant,UUID card,ConfigureRequest request,String actor) {
        requireTransport(); outsideTransaction();
        Map<String,Object> op=tx.execute(s -> {
            db.queryForObject("select pg_advisory_xact_lock(hashtextextended(?, 3781))",Object.class,request.operationId().toString());
            var prior=find(request.operationId());
            if(prior!=null) {
                if(!tenant.equals(prior.get("institution_id")) || !card.equals(prior.get("card_id"))
                        || !"CONFIGURE_PIN_PROFILE".equals(prior.get("operation")))
                    throw new ResourceVersionConflictException("PIN operation ID is already in use");
                return prior;
            }
            var pin=cardPin(tenant,card,true);
            if(!"SWITCH".equals(pin.get("pin_authority")) || pin.get("switch_pin_profile")==null
                    || number(pin,"switch_policy_version")!=0 || number(pin,"switch_pin_version")!=0
                    || number(pin,"applied_revision")<1)
                throw new BusinessRuleViolationException("Only registered, unenrolled Switch PIN profiles may be configured");
            if(!Set.of("PENDING","ACTIVE").contains(pin.get("card_status")) || pin.get("security_hold_reason")!=null)
                throw new BusinessRuleViolationException("Card is not eligible for PIN configuration");
            requireNoPending(tenant,(UUID)pin.get("pan_allocation_id"));
            insert(request.operationId(),tenant,card,(UUID)pin.get("pan_allocation_id"),"CONFIGURE_PIN_PROFILE",1,0,
                    fingerprint(pin.get("switch_pin_profile").toString()),actor,"PREPARED");
            return find(request.operationId());
        });
        if("PREPARED".equals(op.get("status"))) {
            var pin=cardPin(tenant,card,false);
            var body=command(op); body.set("profile",parse(pin.get("switch_pin_profile").toString()));
            sendOnce(op,body);
        } else if("UNKNOWN".equals(op.get("status"))) reconcile(request.operationId());
        return result(find(request.operationId()));
    }

    public OperationResult submit(UUID tenant, UUID card, String operation, SelectedPinRequest request, String actor) {
        requireTransport(); outsideTransaction();
        if (!CUSTOMER_OPERATIONS.contains(operation)) throw new IllegalArgumentException("Unsupported PIN action");
        if (("CHANGE_PIN".equals(operation)) != (request.currentPin()!=null))
            throw new IllegalArgumentException("Current PIN is required only for CHANGE_PIN");
        String fingerprint=fingerprint(tenant+":"+card+":"+operation+":"+encode(request));
        Map<String,Object> op=tx.execute(s -> {
            // Serialize identical operation IDs before checking replay, including cross-family attempts.
            db.queryForObject("select pg_advisory_xact_lock(hashtextextended(?, 3781))",Object.class,request.operationId().toString());
            var prior=find(request.operationId());
            if (prior!=null) {
                if (!tenant.equals(prior.get("institution_id")) || !card.equals(prior.get("card_id"))
                        || !fingerprint.equals(prior.get("request_fingerprint")))
                    throw new ResourceVersionConflictException("PIN operation ID was already used for a different request");
                return prior;
            }
            var pin=cardPin(tenant,card,true);
            validateRequest(pin,operation,request);
            requireNoPending(tenant,(UUID)pin.get("pan_allocation_id"));
            insert(request.operationId(),tenant,card,(UUID)pin.get("pan_allocation_id"),operation,
                    request.policyProfileVersion(),request.expectedPinVersion(),fingerprint,actor,"PREPARED");
            return find(request.operationId());
        });
        if ("PREPARED".equals(op.get("status"))) {
            ObjectNode command=command(op);
            try { command.put("protectedPinMaterial",envelope.create(command,request)); }
            catch (RuntimeException ex) {
                rejectPrepared(request.operationId(),"ENVELOPE_UNAVAILABLE");
                return result(find(request.operationId()));
            }
            sendOnce(op,command);
        } else if ("UNKNOWN".equals(op.get("status"))) reconcile(request.operationId());
        return result(find(request.operationId()));
    }

    public OperationResult operation(UUID tenant, UUID card, UUID operation) {
        var op=find(operation);
        if (op==null || !tenant.equals(op.get("institution_id")) || !card.equals(op.get("card_id")))
            throw new ResourceNotFoundException("PIN operation",operation.toString());
        // GET is safe reconciliation; a 404 at Switch never causes a new submission.
        if (enabled && "UNKNOWN".equals(op.get("status"))) reconcile(operation);
        return result(find(operation));
    }

    public Map<String,Object> state(UUID tenant, UUID card) {
        var pin=cardPin(tenant,card,false);
        var state=new LinkedHashMap<String,Object>();
        state.put("cardId",card); state.put("panFamilyReference",pin.get("pan_allocation_id"));
        state.put("authority",pin.get("pin_authority")); state.put("pinStatus",pin.get("pin_status"));
        state.put("pinRequired",pin.get("pin_required"));
        state.put("syncStatus","SWITCH".equals(pin.get("pin_authority")) && pin.get("switch_pin_profile")==null
                ? "NOT_REQUIRED" : pin.get("switch_pin_sync_status"));
        state.put("policyProfileVersion",pin.get("switch_policy_version"));
        state.put("lastConfirmedPinVersion",pin.get("switch_pin_version"));
        state.put("lastConfirmedAt",pin.get("switch_pin_confirmed_at"));
        state.put("mailerReference",pin.get("pin_mailer_reference"));
        state.put("liveSwitchState",false);
        state.put("operations",db.queryForList("""
                select * from cms_v2.switch_pin_operation where institution_id=? and pan_family_id=?
                order by created_at desc limit 20
                """,tenant,pin.get("pan_allocation_id")).stream().map(this::result).toList());
        return state;
    }

    /** Bounded background provisioning and metadata-only reconciliation. */
    public void poll() {
        if (!enabled || transports.getIfAvailable()==null) return;
        outsideTransaction();
        var candidates=db.queryForList("""
                select p.institution_id, min(c.card_id::text)::uuid as card_id
                from cms_v2.pin_data p join cms_v2.card c
                  on c.institution_id=p.institution_id and c.pin_data_ref=p.pin_data_id
                join cms_v2.switch_card_sync s on s.institution_id=c.institution_id and s.card_id=c.card_id
                where p.pin_authority='SWITCH' and p.switch_pin_profile is not null
                  and p.switch_policy_version=0 and s.applied_revision>0
                  and c.card_status not in ('CLOSED','BLOCKED','REPLACED','EXPIRED')
                  and not exists(select 1 from cms_v2.switch_pin_operation o
                    where o.institution_id=p.institution_id and o.pan_family_id=p.pan_allocation_id)
                group by p.institution_id,p.pin_data_id limit 10
                """);
        for(var candidate:candidates) configure((UUID)candidate.get("institution_id"),(UUID)candidate.get("card_id"));
        var pending=db.queryForList("""
                select operation_id from cms_v2.switch_pin_operation
                where (status='UNKNOWN' or (status='PREPARED' and operation='CONFIGURE_PIN_PROFILE'))
                  and next_lookup_at<=now() order by created_at limit 20
                """,UUID.class);
        for(UUID id:pending) {
            var op=find(id);
            if ("PREPARED".equals(op.get("status"))) {
                var pin=cardPin((UUID)op.get("institution_id"),(UUID)op.get("card_id"),false);
                var body=command(op); body.set("profile",parse(pin.get("switch_pin_profile").toString()));
                sendOnce(op,body);
            } else reconcile(id);
        }
    }

    private void configure(UUID tenant, UUID card) {
        tx.executeWithoutResult(s -> {
            var pin=cardPin(tenant,card,true);
            if (number(pin,"switch_policy_version")!=0 || Boolean.TRUE.equals(db.queryForObject("""
                    select exists(select 1 from cms_v2.switch_pin_operation where institution_id=? and pan_family_id=?)
                    """,Boolean.class,tenant,pin.get("pan_allocation_id")))) return;
            UUID id=UUID.randomUUID();
            insert(id,tenant,card,(UUID)pin.get("pan_allocation_id"),"CONFIGURE_PIN_PROFILE",1,0,
                    fingerprint(pin.get("switch_pin_profile").toString()),"switch-pin-worker","PREPARED");
            db.update("update cms_v2.pin_data set switch_pin_sync_status='CONFIGURING',version=version+1 where pin_data_id=?",pin.get("pin_data_id"));
        });
    }

    private void sendOnce(Map<String,Object> op, ObjectNode body) {
        outsideTransaction();
        UUID id=(UUID)op.get("operation_id");
        // Commit UNKNOWN before IO. Only the winner may send. A crash never triggers a second HSM call.
        int claimed=db.update("""
                update cms_v2.switch_pin_operation set status='UNKNOWN',updated_at=now(),next_lookup_at=now()+interval '30 seconds'
                where operation_id=? and status='PREPARED'
                """,id);
        if (claimed!=1) return;
        try {
            JsonNode receipt=requireTransport().sendPin((UUID)op.get("institution_id"),(UUID)op.get("pan_family_id"),id,body);
            accept(op,receipt);
        } catch (RestClientResponseException ex) {
            int code=ex.getStatusCode().value();
            if (Set.of(400,401,403,404,409,422).contains(code)) reject(id,"HTTP_"+code);
            else uncertain(id,"SWITCH_UNAVAILABLE");
        } catch (RuntimeException ex) { uncertain(id,"OUTCOME_UNCONFIRMED"); }
        finally { body.removeAll(); }
    }

    private void reconcile(UUID id) {
        outsideTransaction();
        var op=find(id);
        if (op==null || !"UNKNOWN".equals(op.get("status"))) return;
        try {
            JsonNode receipt=requireTransport().pinReceipt((UUID)op.get("institution_id"),id);
            if (receipt!=null) accept(op,receipt);
            else if ("CONFIGURE_PIN_PROFILE".equals(op.get("operation"))) {
                // This command has no encrypted input, expiry, or HSM side effect. Its pinned profile is replayable.
                db.update("""
                        update cms_v2.switch_pin_operation set status='PREPARED',next_lookup_at=now()+interval '30 seconds'
                        where operation_id=? and status='UNKNOWN'
                        """,id);
            }
            else uncertain(id,"RECEIPT_NOT_FOUND");
        } catch (RuntimeException ex) { uncertain(id,"LOOKUP_UNAVAILABLE"); }
    }

    private void accept(Map<String,Object> op, JsonNode receipt) {
        validateReceipt(op,receipt);
        if ("PENDING".equals(receipt.path("status").asText())) {
            uncertain((UUID)op.get("operation_id"),"SWITCH_PENDING"); return;
        }
        tx.executeWithoutResult(s -> {
            // Consistent family-before-ledger lock order.
            var pin=cardPin((UUID)op.get("institution_id"),(UUID)op.get("card_id"),true);
            var current=db.queryForMap("select * from cms_v2.switch_pin_operation where operation_id=? for update",op.get("operation_id"));
            if (TERMINAL.contains(current.get("status"))) return;
            String status=receipt.path("status").asText();
            long pinVersion=receipt.path("pinVersion").longValue();
            boolean configured="CONFIGURE_PIN_PROFILE".equals(op.get("operation"));
            boolean completed="COMPLETED".equals(status);
            String failure=completed ? null : safeFailure(receipt.path("failureCode").asText());
            String mailer=receipt.path("mailerReference").isTextual()?receipt.path("mailerReference").asText():null;
            db.update("""
                    update cms_v2.switch_pin_operation set status=?,result_pin_version=?,mailer_reference=?,failure_code=?,updated_at=now()
                    where operation_id=?
                    """,status,pinVersion,mailer,failure,op.get("operation_id"));
            if (completed && configured) {
                boolean generated="HSM_GENERATED".equals(parse(pin.get("switch_pin_profile").toString()).path("establishment").asText());
                db.update("""
                        update cms_v2.pin_data set switch_policy_version=?,switch_pin_sync_status=?,
                          switch_pin_confirmed_at=now(),updated_at=now(),version=version+1 where pin_data_id=?
                        """,op.get("policy_version"),generated?"UNSUPPORTED":"READY",pin.get("pin_data_id"));
                if (generated) {
                    UUID generation=UUID.randomUUID();
                    insert(generation,(UUID)op.get("institution_id"),(UUID)op.get("card_id"),(UUID)op.get("pan_family_id"),
                            "GENERATE_INITIAL_PIN",number(op,"policy_version"),0,fingerprint("unsupported-generation:"+generation),
                            "switch-pin-worker","UNSUPPORTED");
                    db.update("update cms_v2.switch_pin_operation set failure_code='GENERATED_PIN_MAILER_UNSUPPORTED' where operation_id=?",generation);
                }
            } else if (completed && pinVersion>=number(pin,"switch_pin_version")) {
                String resulting="CLOSED".equals(pin.get("pin_status")) ? "CLOSED"
                        : "GENERATE_INITIAL_PIN".equals(op.get("operation")) ? "PENDING_DELIVERY" : "ACTIVE";
                db.update("""
                        update cms_v2.pin_data set switch_pin_version=?,switch_pin_sync_status='READY',pin_status=?,
                          pin_mailer_reference=?,switch_pin_confirmed_at=now(),last_changed_at=?,updated_at=now(),
                          updated_by=?,version=version+1 where pin_data_id=?
                        """,pinVersion,resulting,mailer,java.sql.Timestamp.from(receiptTime(receipt.get("updatedAt"))),op.get("actor"),pin.get("pin_data_id"));
                history(op,pin,resulting,true);
            } else if (!completed && "CHANGE_PIN".equals(op.get("operation"))) {
                history(op,pin,pin.get("pin_status").toString(),false);
            }
            // Failed PIN verification does not infer retry count/block state from a receipt lacking those fields.
        });
    }

    static void validateReceipt(Map<String,Object> op, JsonNode receipt) {
        if (receipt==null || !receipt.isObject()) throw new IllegalStateException("Invalid Switch PIN receipt");
        for(var entry:Map.of("operationId","operation_id","institutionId","institution_id",
                "panFamilyReference","pan_family_id","operation","operation").entrySet())
            if (!op.get(entry.getValue()).toString().equals(receipt.path(entry.getKey()).asText()))
                throw new IllegalStateException("Mismatched Switch PIN receipt");
        String status=receipt.path("status").asText();
        if (!Set.of("COMPLETED","FAILED","PENDING").contains(status)
                || !receipt.path("policyProfileVersion").isIntegralNumber()
                || !receipt.path("policyProfileVersion").canConvertToLong()
                || receipt.path("policyProfileVersion").longValue()!=number(op,"policy_version")
                || !receipt.path("pinVersion").isIntegralNumber() || !receipt.path("pinVersion").canConvertToLong())
            throw new IllegalStateException("Invalid Switch PIN receipt version/status");
        long expected=number(op,"expected_pin_version");
        if ("COMPLETED".equals(status) && !"CONFIGURE_PIN_PROFILE".equals(op.get("operation"))) expected=Math.addExact(expected,1);
        if (receipt.path("pinVersion").longValue()!=expected) throw new IllegalStateException("Unexpected Switch PIN version");
        if ("COMPLETED".equals(status) && "GENERATE_INITIAL_PIN".equals(op.get("operation"))
                && (!receipt.path("mailerReference").isTextual() || receipt.path("mailerReference").asText().isBlank()
                    || receipt.path("mailerReference").asText().length()>300))
            throw new IllegalStateException("Missing Switch mailer reference");
        receiptTime(receipt.get("updatedAt"));
    }

    static Instant receiptTime(JsonNode value) {
        if (value==null) throw new IllegalStateException("Missing receipt time");
        if (value.isTextual()) return Instant.parse(value.asText());
        if (value.isNumber()) {
            var seconds=value.decimalValue();
            long whole=seconds.setScale(0,java.math.RoundingMode.FLOOR).longValueExact();
            long nanos=seconds.subtract(java.math.BigDecimal.valueOf(whole)).movePointRight(9).longValueExact();
            return Instant.ofEpochSecond(whole,nanos);
        }
        throw new IllegalStateException("Invalid receipt time");
    }

    private void history(Map<String,Object> op,Map<String,Object> pin,String resulting,boolean success) {
        String type=!success ? "PIN_CHANGE_FAILED" : switch(op.get("operation").toString()) {
            case "INITIALIZE_SELECTED_PIN" -> "INITIALIZED";
            case "CHANGE_PIN" -> "PIN_CHANGED";
            case "RESET_PIN" -> "PIN_RESET";
            default -> "GENERATED";
        };
        db.update("""
                insert into cms_v2.pin_history(pin_history_id,institution_id,card_id,pin_data_id,operation_type,
                  previous_status,resulting_status,occurred_at,actor_reference,correlation_id,operation_reason)
                values(?,?,?,?,?,?,?,now(),?,?, 'SWITCH_CONFIRMED')
                """,UUID.randomUUID(),op.get("institution_id"),op.get("card_id"),pin.get("pin_data_id"),type,
                pin.get("pin_status"),resulting,op.get("actor"),op.get("operation_id"));
    }

    private void validateRequest(Map<String,Object> pin,String operation,SelectedPinRequest request) {
        if (!"SWITCH".equals(pin.get("pin_authority")) || pin.get("switch_pin_profile")==null)
            throw new BusinessRuleViolationException("PIN family requires explicit Switch enrollment");
        if (!"ACTIVE".equals(pin.get("policy_status")) || !Boolean.TRUE.equals(pin.get("pin_required")))
            throw new BusinessRuleViolationException("An active PIN policy is required");
        if (!Set.of("PENDING","ACTIVE").contains(pin.get("card_status")) || pin.get("security_hold_reason")!=null
                || pin.get("retired_at")!=null || "CLOSED".equals(pin.get("pin_status")))
            throw new BusinessRuleViolationException("Card is not eligible for PIN operations");
        String flag=switch(operation) {
            case "INITIALIZE_SELECTED_PIN" -> "customer_pin_selection_allowed";
            case "CHANGE_PIN" -> "pin_change_allowed";
            default -> "pin_reset_allowed";
        };
        if (!Boolean.TRUE.equals(pin.get(flag))) throw new BusinessRuleViolationException("PIN operation is disabled by product policy");
        if (number(pin,"applied_revision")<1 || number(pin,"switch_policy_version")!=request.policyProfileVersion())
            throw new ResourceVersionConflictException("Switch PIN profile is not confirmed at the requested policy version");
        if (request.expectedPinVersion()<number(pin,"switch_pin_version"))
            throw new ResourceVersionConflictException("PIN version is older than the last confirmed Switch version");
        if ("INITIALIZE_SELECTED_PIN".equals(operation)) {
            if (request.expectedPinVersion()!=0 || !"CUSTOMER_SELECTED".equals(parse(pin.get("switch_pin_profile").toString()).path("establishment").asText()))
                throw new BusinessRuleViolationException("Initial selection requires an unestablished customer-selected profile");
        } else if (request.expectedPinVersion()==0) throw new BusinessRuleViolationException("Change/reset requires an established PIN");
    }

    private Map<String,Object> cardPin(UUID tenant,UUID card,boolean lock) {
        var rows=db.queryForList("""
                select p.*, c.card_status,c.security_hold_reason,c.retired_at,
                  policy.policy_status,policy.pin_required,policy.customer_pin_selection_allowed,
                  policy.pin_change_allowed,policy.pin_reset_allowed,coalesce(s.applied_revision,0) as applied_revision
                from cms_v2.card c join cms_v2.pin_data p on p.institution_id=c.institution_id and p.pin_data_id=c.pin_data_ref
                join cms_v2.product_pin_policy policy on policy.institution_id=c.institution_id and policy.product_version_id=c.product_version_id
                left join cms_v2.switch_card_sync s on s.institution_id=c.institution_id and s.card_id=c.card_id
                where c.institution_id=? and c.card_id=?
                """+(lock?" for update of p":""),tenant,card);
        if(rows.isEmpty()) throw new ResourceNotFoundException("Card PIN profile",card.toString());
        return rows.getFirst();
    }
    private void requireNoPending(UUID tenant,UUID family) {
        if (Boolean.TRUE.equals(db.queryForObject("""
                select exists(select 1 from cms_v2.switch_pin_operation where institution_id=? and pan_family_id=?
                  and status in ('PREPARED','UNKNOWN'))
                """,Boolean.class,tenant,family)))
            throw new ResourceVersionConflictException("A PIN operation is pending reconciliation for this PAN family");
    }
    private void insert(UUID id,UUID tenant,UUID card,UUID family,String operation,long policy,long expected,
            String fingerprint,String actor,String status) {
        db.update("""
                insert into cms_v2.switch_pin_operation(operation_id,institution_id,card_id,pan_family_id,operation,
                  policy_version,expected_pin_version,request_fingerprint,status,actor) values(?,?,?,?,?,?,?,?,?,?)
                """,id,tenant,card,family,operation,policy,expected,fingerprint,status,actor);
    }
    private Map<String,Object> find(UUID id) {
        var rows=db.queryForList("select * from cms_v2.switch_pin_operation where operation_id=?",id);
        return rows.isEmpty()?null:rows.getFirst();
    }
    private ObjectNode command(Map<String,Object> op) {
        return json.createObjectNode().put("operationId",op.get("operation_id").toString())
                .put("institutionId",op.get("institution_id").toString()).put("panFamilyReference",op.get("pan_family_id").toString())
                .put("policyProfileVersion",number(op,"policy_version")).put("expectedPinVersion",number(op,"expected_pin_version"))
                .put("operation",op.get("operation").toString());
    }
    private OperationResult result(Map<String,Object> op) {
        return new OperationResult((UUID)op.get("operation_id"),(UUID)op.get("institution_id"),(UUID)op.get("pan_family_id"),
                (String)op.get("operation"),(String)op.get("status"),number(op,"policy_version"),number(op,"expected_pin_version"),
                op.get("result_pin_version")==null?null:number(op,"result_pin_version"),(String)op.get("mailer_reference"),(String)op.get("failure_code"));
    }
    private void rejectPrepared(UUID id,String code) {
        db.update("update cms_v2.switch_pin_operation set status='REJECTED',failure_code=?,updated_at=now() where operation_id=? and status='PREPARED'",code,id);
    }
    private void reject(UUID id,String code) {
        tx.executeWithoutResult(s -> {
            var op=find(id);
            var pin=cardPin((UUID)op.get("institution_id"),(UUID)op.get("card_id"),true);
            int changed=db.update("update cms_v2.switch_pin_operation set status='REJECTED',failure_code=?,updated_at=now() where operation_id=? and status='UNKNOWN'",code,id);
            if(changed==1) db.update("""
                    update cms_v2.pin_data set switch_pin_sync_status='RECONCILIATION_REQUIRED',version=version+1
                    where pin_data_id=?
                    """,pin.get("pin_data_id"));
        });
    }
    private void uncertain(UUID id,String code) {
        db.update("""
                update cms_v2.switch_pin_operation set failure_code=?,updated_at=now(),next_lookup_at=now()+interval '60 seconds'
                where operation_id=? and status='UNKNOWN'
                """,code,id);
    }
    private SwitchSyncTransport requireTransport() {
        var transport=enabled?transports.getIfAvailable():null;
        if(transport==null) throw new BusinessRuleViolationException("Switch PIN integration is not enabled");
        return transport;
    }
    private static void outsideTransaction() {
        if(TransactionSynchronizationManager.isActualTransactionActive()) throw new IllegalStateException("PIN network operation must run outside a database transaction");
    }
    private JsonNode parse(String value) { try { return json.readTree(value); } catch(Exception ex) { throw new IllegalStateException("Invalid PIN profile metadata"); } }
    private String encode(Object value) { try { return json.writeValueAsString(value); } catch(Exception ex) { throw new IllegalArgumentException("Invalid PIN request"); } }
    private static long number(Map<String,Object> row,String key) { return ((Number)row.get(key)).longValue(); }
    private static String fingerprint(String value) {
        try { return HexFormat.of().formatHex(MessageDigest.getInstance("SHA-256").digest(value.getBytes(StandardCharsets.UTF_8))); }
        catch(Exception ex) { throw new IllegalStateException("Fingerprint unavailable"); }
    }
    private static String safeFailure(String code) {
        return Set.of("INVALID_PIN","PIN_BLOCKED").contains(code)?code:"SWITCH_EXECUTION_FAILED";
    }
