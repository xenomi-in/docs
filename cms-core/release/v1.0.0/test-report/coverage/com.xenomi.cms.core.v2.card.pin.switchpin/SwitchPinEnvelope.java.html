package com.xenomi.cms.core.v2.card.pin.switchpin;

import com.fasterxml.jackson.databind.ObjectMapper;
import com.fasterxml.jackson.databind.node.ObjectNode;
import com.nimbusds.jose.*;
import com.nimbusds.jose.crypto.RSAEncrypter;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Component;
import org.springframework.core.io.DefaultResourceLoader;
import org.springframework.core.io.FileSystemResource;
import java.security.cert.CertificateFactory;
import java.security.cert.X509Certificate;
import java.security.interfaces.RSAPublicKey;
import java.time.Instant;

@Component
public class SwitchPinEnvelope {
    private final ObjectMapper json;
    private final String certificatePath, kid;
    public SwitchPinEnvelope(ObjectMapper json,
            @Value("${cms.switch.pin.recipient-certificate:}") String certificatePath,
            @Value("${cms.switch.pin.recipient-key-id:cms-pin-v1}") String kid) {
        this.json=json; this.certificatePath=certificatePath; this.kid=kid;
    }

    public String create(ObjectNode command, SelectedPinRequest request) {
        if (certificatePath.isBlank() || kid.isBlank())
            throw new IllegalStateException("Switch PIN recipient certificate/key ID is not configured");
        ObjectNode material=command.deepCopy();
        material.put("expiresAt",Instant.now().plusSeconds(240).toString());
        material.put("authorizationReference",request.authorizationReference());
        if (request.currentPin()!=null) material.set("currentPin",json.valueToTree(request.currentPin()));
        material.set("newPin",json.valueToTree(request.newPin()));
        try {
            var resource=certificatePath.startsWith("classpath:") || certificatePath.startsWith("file:")
                    ? new DefaultResourceLoader().getResource(certificatePath) : new FileSystemResource(certificatePath);
            X509Certificate certificate;
            try(var in=resource.getInputStream()) {
                certificate=(X509Certificate)CertificateFactory.getInstance("X.509").generateCertificate(in);
            }
            certificate.checkValidity();
            if (!(certificate.getPublicKey() instanceof RSAPublicKey key) || key.getModulus().bitLength()<2048)
                throw new IllegalStateException("Invalid Switch PIN recipient key");
            var jwe=new JWEObject(new JWEHeader.Builder(JWEAlgorithm.RSA_OAEP_256,EncryptionMethod.A256GCM)
                    .keyID(kid).build(),new Payload(json.writeValueAsString(material)));
            jwe.encrypt(new RSAEncrypter(key));
            return jwe.serialize();
        } catch(Exception ex) {
            // Do not attach provider/serialization exceptions that could contain PIN inputs.
            throw new IllegalStateException("Switch PIN envelope preparation failed");
        } finally { material.removeAll(); }
    }
}
