package com.xenomi.cms.core.v2.card.pin.switchpin;

import com.xenomi.cms.core.exception.BusinessRuleViolationException;
import com.xenomi.cms.core.v2.bin.entity.BinConfigurationV2;
import com.xenomi.cms.core.v2.bin.model.PinMethod;
import com.xenomi.cms.core.v2.product.entity.ProductPinPolicyV2;
import com.xenomi.cms.core.v2.product.model.*;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.stereotype.Component;
import java.util.Map;
import java.util.UUID;

@Component
public class SwitchPinPolicyResolver {
    private final JdbcTemplate db;
    public SwitchPinPolicyResolver(JdbcTemplate db) { this.db = db; }

    public Map<String, Object> resolve(UUID tenant, ProductPinPolicyV2 policy,
            BinConfigurationV2 bin, ReplacementPinStrategy replacement) {
        if (!policy.isPinRequired()) return null;
        if (!policy.isOnlinePinSupported() || policy.isOfflinePinSupported())
            throw new BusinessRuleViolationException("Switch PIN integration currently supports online PIN only");
        if (bin.getPinMethod() != PinMethod.VISA_PVV || policy.getPinLength() != 4 || bin.getPinLength() != 4
                || bin.getPvki() == null || bin.getPvki() < 0 || bin.getPvki() > 6)
            throw new BusinessRuleViolationException("Switch PIN integration requires VISA_PVV, a four-digit PIN and PVKI 0..6");
        boolean selected = replacement == ReplacementPinStrategy.CUSTOMER_RESET_REQUIRED
                || policy.getGenerationMethod() == PinGenerationMethod.CUSTOMER_SELECTED;
        if (selected && !policy.isCustomerPinSelectionAllowed())
            throw new BusinessRuleViolationException("Customer PIN selection is disabled by the product policy");
        if (!selected && (policy.getGenerationMethod() != PinGenerationMethod.HSM_GENERATED
                || !policy.isRandomPinGenerationEnabled() || !policy.isPinMailerEnabled()
                || policy.getDeliveryMethod() != PinDeliveryMethod.PIN_MAILER))
            throw new BusinessRuleViolationException("Switch-generated PIN policy requires HSM_GENERATED and PIN_MAILER");
        var aliases = db.queryForList("""
                select key_binding_reference from cms_v2.switch_pin_profile_mapping
                where institution_id=? and profile_reference=?
                """, String.class, tenant, policy.getHsmPinProfileReference());
        if (aliases.size() != 1)
            throw new BusinessRuleViolationException("A Switch PIN profile/key binding mapping is required");
        return Map.of("establishment", selected ? "CUSTOMER_SELECTED" : "HSM_GENERATED",
                "delivery", selected ? "NONE" : "PIN_MAILER", "verificationMethod", "VISA_PVV",
                "keyBindingReference", aliases.getFirst(), "pvki", bin.getPvki());
    }
}
