package com.xenomi.cms.core.v2.bin.entity;

import java.time.OffsetDateTime;
import java.util.UUID;

import com.xenomi.cms.core.v2.bin.model.AllocationStrategy;
import com.xenomi.cms.core.v2.bin.model.BinRangeStatus;

import jakarta.persistence.Column;
import jakarta.persistence.Entity;
import jakarta.persistence.EnumType;
import jakarta.persistence.Enumerated;
import jakarta.persistence.Id;
import jakarta.persistence.PrePersist;
import jakarta.persistence.PreUpdate;
import jakarta.persistence.Table;
import lombok.AllArgsConstructor;
import lombok.Builder;
import lombok.Data;
import lombok.NoArgsConstructor;

@Entity(name = "BinRangeV2")
@Table(name = "bin_range", schema = "cms_v2")
@Data
@Builder
@NoArgsConstructor
@AllArgsConstructor
public class BinRangeV2 {

    @Id
    @Column(name = "bin_range_id", nullable = false, updatable = false)
    private UUID binRangeId;

    @Column(name = "institution_id", nullable = false, updatable = false)
    private UUID institutionId;

    @Column(name = "bin_id", nullable = false, updatable = false)
    private UUID binId;

    @Column(name = "range_start", nullable = false, length = 19, updatable = false)
    private String rangeStart;

    @Column(name = "range_end", nullable = false, length = 19, updatable = false)
    private String rangeEnd;

    @Enumerated(EnumType.STRING)
    @Column(name = "allocation_strategy", nullable = false, length = 24)
    private AllocationStrategy allocationStrategy;

    @Column(name = "usable_capacity", nullable = false)
    private Long usableCapacity;

    @Column(name = "exhaustion_threshold")
    private Long exhaustionThreshold;

    @Column(name = "issued_pan_count", nullable = false)
    @Builder.Default
    private Long issuedPanCount = 0L;

    @Column(name = "next_sequence_value", nullable = false)
    @Builder.Default
    private Long nextSequenceValue = 0L;

    @Enumerated(EnumType.STRING)
    @Column(name = "range_status", nullable = false, length = 24)
    private BinRangeStatus rangeStatus;

    @Column(name = "effective_from", nullable = false)
    private OffsetDateTime effectiveFrom;

    @Column(name = "effective_to")
    private OffsetDateTime effectiveTo;

    @Column(name = "created_at", nullable = false, updatable = false)
    private OffsetDateTime createdAt;

    @Column(name = "updated_at", nullable = false)
    private OffsetDateTime updatedAt;

    @Column(name = "created_by", nullable = false, length = 100, updatable = false)
    private String createdBy;

    @Column(name = "updated_by", nullable = false, length = 100)
    private String updatedBy;

    @PrePersist
    void beforeInsert() {
        OffsetDateTime now = OffsetDateTime.now(java.time.ZoneOffset.UTC);
        if (createdAt == null) createdAt = now;
        updatedAt = now;
    }

    @PreUpdate
    void beforeUpdate() {
        updatedAt = OffsetDateTime.now(java.time.ZoneOffset.UTC);
    }
}
