package com.xenomi.cms.core.v2.bin.entity;

import java.time.LocalDate;
import java.time.OffsetDateTime;
import java.time.ZoneOffset;
import java.util.UUID;

import com.xenomi.cms.core.v2.bin.model.BinStatus;
import com.xenomi.cms.core.v2.bin.model.BinType;
import com.xenomi.cms.core.v2.bin.model.CardNetwork;
import com.xenomi.cms.core.v2.bin.model.CardTechnology;
import com.xenomi.cms.core.v2.bin.model.PinMethod;

import jakarta.persistence.Column;
import jakarta.persistence.Entity;
import jakarta.persistence.EnumType;
import jakarta.persistence.Enumerated;
import jakarta.persistence.Id;
import jakarta.persistence.PrePersist;
import jakarta.persistence.PreUpdate;
import jakarta.persistence.Table;
import jakarta.persistence.Version;
import lombok.AllArgsConstructor;
import lombok.Builder;
import lombok.Data;
import lombok.NoArgsConstructor;

@Entity(name = "BinConfigurationV2")
@Table(name = "bin_configuration", schema = "cms_v2",
        uniqueConstraints = @jakarta.persistence.UniqueConstraint(name = "uk_bin_number_platform", columnNames = "bin_number"))
@Data
@Builder
@NoArgsConstructor
@AllArgsConstructor
public class BinConfigurationV2 {

    @Id
    @Column(name = "bin_id", nullable = false, updatable = false)
    private UUID binId;

    @Column(name = "institution_id", nullable = false, updatable = false)
    private UUID institutionId;

    @Column(name = "bin_number", nullable = false, length = 8, updatable = false)
    private String binNumber;

    @Column(name = "bin_length", nullable = false, updatable = false)
    private int binLength;

    /** Total PAN length, including the BIN prefix and Luhn check digit. */
    @Column(name = "pan_length", nullable = false, updatable = false)
    private int panLength;

    @Enumerated(EnumType.STRING)
    @Column(name = "card_network", nullable = false, length = 20, updatable = false)
    private CardNetwork cardNetwork;

    @Enumerated(EnumType.STRING)
    @Column(name = "bin_type", nullable = false, length = 20, updatable = false)
    private BinType binType;

    @Column(name = "card_product_class", nullable = false, length = 20, updatable = false)
    private String cardProductClass;

    @Enumerated(EnumType.STRING)
    @Column(name = "card_technology", nullable = false, length = 20)
    private CardTechnology cardTechnology;

    @Enumerated(EnumType.STRING)
    @Column(name = "pin_method", nullable = false, length = 20)
    private PinMethod pinMethod;

    @Column(name = "pin_length", nullable = false)
    private int pinLength;

    @org.hibernate.annotations.JdbcTypeCode(org.hibernate.type.SqlTypes.CHAR)
    @Column(name = "issuing_country_code", nullable = false, length = 3, columnDefinition = "char(3)")
    private String issuingCountryCode;

    @org.hibernate.annotations.JdbcTypeCode(org.hibernate.type.SqlTypes.CHAR)
    @Column(name = "default_currency_code", nullable = false, length = 3, columnDefinition = "char(3)")
    private String defaultCurrencyCode;

    @Column(name = "default_chip_enabled", nullable = false)
    private boolean defaultChipEnabled;

    @Column(name = "default_domestic_contact_enabled", nullable = false)
    private boolean defaultDomesticContactEnabled;

    @Column(name = "default_domestic_contactless_enabled", nullable = false)
    private boolean defaultDomesticContactlessEnabled;

    @Column(name = "default_domestic_ecommerce_enabled", nullable = false)
    private boolean defaultDomesticEcommerceEnabled;

    @Column(name = "default_international_enabled", nullable = false)
    private boolean defaultInternationalEnabled;

    @Enumerated(EnumType.STRING)
    @Column(name = "bin_status", nullable = false, length = 20)
    private BinStatus binStatus;

    @Column(name = "effective_from", nullable = false)
    private LocalDate effectiveFrom;

    @Column(name = "effective_to")
    private LocalDate effectiveTo;

    @Column(name = "network_program_identifier", length = 50)
    private String networkProgramIdentifier;

    @Column(name = "network_certification_reference", length = 100)
    private String networkCertificationReference;

    @Column(name = "track2_separator", length = 1)
    private String track2Separator;

    @Column(name = "decimalization_table", length = 20)
    private String decimalizationTable;

    @Column(name = "pvki")
    private Integer pvki;

    @Column(name = "created_at", nullable = false, updatable = false)
    private OffsetDateTime createdAt;

    @Column(name = "updated_at", nullable = false)
    private OffsetDateTime updatedAt;

    @Column(name = "created_by", nullable = false, length = 50, updatable = false)
    private String createdBy;

    @Column(name = "updated_by", nullable = false, length = 50)
    private String updatedBy;

    @Version
    @Column(name = "version", nullable = false)
    @Builder.Default
    private Long version = 0L;

    @PrePersist
    void beforeInsert() {
        OffsetDateTime now = OffsetDateTime.now(ZoneOffset.UTC);
        if (createdAt == null) createdAt = now;
        updatedAt = now;
    }

    @PreUpdate
    void beforeUpdate() {
        updatedAt = OffsetDateTime.now(ZoneOffset.UTC);
    }
}
