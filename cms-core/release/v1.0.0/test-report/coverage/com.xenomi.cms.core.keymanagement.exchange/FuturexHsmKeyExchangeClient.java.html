package com.xenomi.cms.core.keymanagement.exchange;

import com.xenomi.cms.core.hms.service.NettyHsmService;
import com.xenomi.cms.core.keymanagement.client.HsmKeyGenerationException;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.stereotype.Component;

@Component
public class FuturexHsmKeyExchangeClient implements HsmKeyExchangeClient {

    private static final Logger log = LoggerFactory.getLogger(FuturexHsmKeyExchangeClient.class);
    private static final String PROVIDER_ID = "FUTUREX";

    private final NettyHsmService nettyHsmService;

    public FuturexHsmKeyExchangeClient(NettyHsmService nettyHsmService) {
        this.nettyHsmService = nettyHsmService;
    }

    @Override
    public ImportedKeyResult importKeyUnderKek(ImportKeyUnderKekRequest request) {
        validate(request);
        try {
            String rawResult = nettyHsmService.translateWorkingKeyUnderKek(
                    request.getTransportedWorkingKeyBlock(),
                    request.getKekProtectedKeyBlock(),
                    request.getWorkingKeyType(),
                    request.isTr31());
            if (rawResult == null || rawResult.isBlank()) {
                throw new HsmKeyGenerationException("HSM returned no key-import result");
            }
            String[] parts = rawResult.split(";", -1);
            if (parts.length < 2 || parts[0].isBlank()
                    || !parts[1].trim().matches("[0-9A-Fa-f]{4,16}")) {
                log.warn("HSM returned malformed key-import response: segments={}, "
                        + "keyBlockPresent={}, kcvLength={}",
                        parts.length, parts.length > 0 && !parts[0].isBlank(),
                        parts.length > 1 ? parts[1].trim().length() : 0);
                throw new HsmKeyGenerationException("HSM returned an invalid key-import response");
            }
            return ImportedKeyResult.builder()
                    .cmsProtectedKeyBlock(parts[0].trim())
                    .keyCheckValue(parts[1].trim().toUpperCase(java.util.Locale.ROOT))
                    .hsmIdentifier(PROVIDER_ID)
                    .tr31(request.isTr31())
                    .build();
        } catch (HsmKeyGenerationException e) {
            throw e;
        } catch (Exception e) {
            log.error("HSM key import failed for keyType={}, keyScheme={}",
                    request.getWorkingKeyType(), request.getKeyScheme(), e);
            throw new HsmKeyGenerationException("HSM key import request failed", e);
        }
    }

    private static void validate(ImportKeyUnderKekRequest request) {
        if (request == null || isBlank(request.getKekProtectedKeyBlock())
                || isBlank(request.getTransportedWorkingKeyBlock())) {
            throw new HsmKeyGenerationException(
                    "Protected KEK and transported working-key blocks are required");
        }
        if (isBlank(request.getWorkingKeyType()) || isBlank(request.getKeyScheme())) {
            throw new HsmKeyGenerationException("Working-key type and keyScheme are required");
        }
    }

    private static boolean isBlank(String value) {
        return value == null || value.isBlank();
    }
}
