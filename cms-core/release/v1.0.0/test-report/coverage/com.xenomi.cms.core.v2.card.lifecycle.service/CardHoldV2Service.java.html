package com.xenomi.cms.core.v2.card.lifecycle.service;

import java.nio.charset.StandardCharsets;
import java.security.MessageDigest;
import java.time.OffsetDateTime;
import java.time.ZoneOffset;
import java.util.LinkedHashMap;
import java.util.Map;
import java.util.UUID;

import com.xenomi.cms.core.exception.BusinessRuleViolationException;
import com.xenomi.cms.core.exception.ResourceNotFoundException;
import com.xenomi.cms.core.v2.card.entity.CardV2;
import com.xenomi.cms.core.v2.card.lifecycle.dto.CardHoldV2Request;
import com.xenomi.cms.core.v2.card.lifecycle.dto.CardLifecycleActionV2Response;
import com.xenomi.cms.core.v2.card.lifecycle.entity.CardEventV2;
import com.xenomi.cms.core.v2.card.lifecycle.entity.CardLifecycleOperationV2;
import com.xenomi.cms.core.v2.card.lifecycle.model.CardHoldReasonV2;
import com.xenomi.cms.core.v2.card.model.LifecycleActorTypeV2;
import com.xenomi.cms.core.v2.card.model.LifecycleOperationStatusV2;
import com.xenomi.cms.core.v2.card.model.PhysicalFulfilmentStatusV2;
import com.xenomi.cms.core.v2.card.model.PhysicalRetiredReasonV2;
import com.xenomi.cms.core.v2.card.lifecycle.repository.CardEventV2Repository;
import com.xenomi.cms.core.v2.card.repository.CardV2Repository;
import com.xenomi.cms.core.v2.card.model.CardStatusV2;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import jakarta.persistence.EntityManager;
import jakarta.persistence.PersistenceContext;

@Service
public class CardHoldV2Service {
    private final CardV2Repository cards;
    private final CardEventV2Repository events;
    @PersistenceContext
    private EntityManager entityManager;

    public CardHoldV2Service(CardV2Repository cards, CardEventV2Repository events) {
        this.cards = cards;
        this.events = events;
    }

    @Transactional
    public CardLifecycleActionV2Response place(UUID institutionId, UUID cardId,
            CardHoldV2Request request, String idempotencyKey, String requestId, String actorId) {
        return mutate(institutionId, cardId, request, "HOLD_PLACED", false, idempotencyKey, actorId, requestId);
    }

    @Transactional
    public CardLifecycleActionV2Response escalate(UUID institutionId, UUID cardId,
            CardHoldV2Request request, String idempotencyKey, String requestId, String actorId) {
        return mutate(institutionId, cardId, request, "HOLD_ESCALATED", true, idempotencyKey, actorId, requestId);
    }

    @Transactional
    public CardLifecycleActionV2Response release(UUID institutionId, UUID cardId,
            CardHoldV2Request request, String idempotencyKey, String requestId, String actorId) {
        CardV2 card = load(institutionId, cardId);
        CardHoldReasonV2 reason = request.getReason();
        if (permanent(reason)) throw new BusinessRuleViolationException("Permanent security holds cannot be released");
        if (!has(card, reason)) throw new BusinessRuleViolationException("The requested hold is not set");
        clear(card, reason);
        return save(card, "HOLD_RELEASED", reason, request, idempotencyKey, actorId, requestId);
    }

    private CardLifecycleActionV2Response mutate(UUID institutionId, UUID cardId,
            CardHoldV2Request request, String eventType, boolean escalation,
            String idempotencyKey, String actorId, String requestId) {
        CardV2 card = load(institutionId, cardId);
        if (card.getClosedAt() != null || card.getReplacedAt() != null)
            throw new BusinessRuleViolationException("Terminal cards cannot receive holds");
        CardHoldReasonV2 reason = request.getReason();
        CardHoldReasonV2 current = current(card, reason);
        if (escalation && (current == null || priority(current) >= priority(reason)))
            throw new BusinessRuleViolationException("Escalation requires a strictly higher same-column reason");
        if (!escalation && current != null)
            throw new BusinessRuleViolationException("A hold column is already set; use escalation");
        set(card, reason);
        // Permanent security reasons also retire any physical representation in the same transaction.
        // The card check constraint requires this for non-virtual forms.
        if (permanent(reason) && card.getCardForm() != com.xenomi.cms.core.v2.card.model.CardFormV2.VIRTUAL
                && card.getRetiredAt() == null) {
            card.setRetiredAt(OffsetDateTime.now(ZoneOffset.UTC));
            card.setPhysicalRetiredReason(PhysicalRetiredReasonV2.LOST_STOLEN);
            if (card.getPhysicalFulfilmentStatus() == PhysicalFulfilmentStatusV2.REQUESTED
                    || card.getPhysicalFulfilmentStatus() == PhysicalFulfilmentStatusV2.PERSONALIZATION_IN_PROGRESS
                    || card.getPhysicalFulfilmentStatus() == PhysicalFulfilmentStatusV2.PERSONALIZED
                    || card.getPhysicalFulfilmentStatus() == PhysicalFulfilmentStatusV2.IN_EMBOSSING_FILE) {
                card.setPhysicalFulfilmentStatus(PhysicalFulfilmentStatusV2.CANCELLED);
                entityManager.createNativeQuery("update cms_v2.card_production_request set request_status='CANCELLED', updated_at=now() where institution_id=? and card_id=? and request_status in ('PENDING','ELIGIBLE','RESERVED_FOR_BATCH')")
                        .setParameter(1, card.getInstitutionId()).setParameter(2, card.getCardId()).executeUpdate();
            }
        }
        return save(card, eventType, reason, request, idempotencyKey, actorId, requestId);
    }

    private CardV2 load(UUID institutionId, UUID cardId) {
        return cards.findForUpdate(institutionId, cardId)
                .orElseThrow(() -> new ResourceNotFoundException("Card not found"));
    }

    private CardLifecycleActionV2Response save(CardV2 card, String type, CardHoldReasonV2 reason,
            CardHoldV2Request request, String idempotencyKey, String actorId, String requestId) {
        CardStatusV2 previous = card.getCardStatus();
        CardV2 saved = cards.saveFactsAndRefresh(card);
        OffsetDateTime now = OffsetDateTime.now(ZoneOffset.UTC);
        UUID eventId = UUID.randomUUID();
        Map<String,Object> payload = new LinkedHashMap<>();
        payload.put("reason", reason.name());
        payload.put("reasonDetail", request.getReasonDetail());
        payload.put("lifecycleVersion", saved.getLifecycleVersion());
        UUID operationId = UUID.randomUUID();
        entityManager.persist(CardLifecycleOperationV2.builder()
                .operationId(operationId).institutionId(saved.getInstitutionId())
                .operationKind(type)
                .idempotencyKeyHash(sha256(idempotencyKey == null ? operationId.toString() : idempotencyKey))
                .requestFingerprint(sha256(saved.getCardId() + ":" + type + ":" + reason.name()))
                .targetCardId(saved.getCardId()).expectedVersion(saved.getLifecycleVersion())
                .actorReference(actorId == null ? "UNKNOWN" : actorId)
                .actorType(LifecycleActorTypeV2.OPS)
                .operationStatus(LifecycleOperationStatusV2.CONFIRMED)
                .resultPayload(new LinkedHashMap<>(payload)).createdAt(now).completedAt(now).build());
        CardEventV2 event = CardEventV2.builder().cardEventId(eventId)
                .institutionId(saved.getInstitutionId()).cardId(saved.getCardId()).eventType(type)
                .previousStatus(previous == null ? null : previous.name())
                .resultingStatus(saved.getCardStatus().name()).reasonCode(reason.name())
                .eventPayload(payload).occurredAt(now).actorReference(actorId)
                .correlationId(correlationId(requestId)).eventSequence(nextSequence(saved))
                .operationId(operationId).schemaVersion(1).source("CMS_V2")
                .transitionId(type).build();
        events.save(event);
        return CardLifecycleActionV2Response.builder().institutionId(saved.getInstitutionId())
                .cardId(saved.getCardId()).eventId(eventId).previousStatus(previous)
                .cardStatus(saved.getCardStatus()).lifecycleVersion(saved.getLifecycleVersion())
                .reason(reason.name()).occurredAt(now).build();
    }

    private long nextSequence(CardV2 card) {
        return events.findFirstByInstitutionIdAndCardIdOrderByEventSequenceDesc(card.getInstitutionId(), card.getCardId())
                .map(e -> e.getEventSequence() + 1).orElse(1L);
    }
    private static UUID correlationId(String id) { try { return UUID.fromString(id); } catch (Exception e) { return UUID.nameUUIDFromBytes((id == null ? "" : id).getBytes(StandardCharsets.UTF_8)); } }
    private static String sha256(String value) {
        try {
            byte[] bytes = MessageDigest.getInstance("SHA-256").digest(value.getBytes(StandardCharsets.UTF_8));
            StringBuilder result = new StringBuilder(64);
            for (byte b : bytes) result.append(String.format("%02x", b));
            return result.toString();
        } catch (Exception e) { throw new IllegalStateException("Unable to hash lifecycle operation", e); }
    }
    private static boolean permanent(CardHoldReasonV2 r) { return r == CardHoldReasonV2.LOST || r == CardHoldReasonV2.STOLEN || r == CardHoldReasonV2.COMPROMISED || r == CardHoldReasonV2.FRAUD_SUSPECTED; }
    private static int priority(CardHoldReasonV2 r) { return switch (r) { case RISK_BLOCK -> 1; case LOST, FRAUD_SUSPECTED -> 2; case STOLEN -> 3; case COMPROMISED -> 4; case KYC -> 1; case REGULATORY -> 2; default -> 1; }; }
    private static CardHoldReasonV2 current(CardV2 c, CardHoldReasonV2 r) { return switch (r) { case CUSTOMER_FREEZE -> c.getCustomerHoldReason() == null ? null : CardHoldReasonV2.CUSTOMER_FREEZE; case OPS_FREEZE -> c.getOpsHoldReason() == null ? null : CardHoldReasonV2.OPS_FREEZE; case RISK_BLOCK, LOST, STOLEN, COMPROMISED, FRAUD_SUSPECTED -> c.getSecurityHoldReason() == null ? null : CardHoldReasonV2.valueOf(c.getSecurityHoldReason().name()); case DELINQUENCY -> c.getCollectionsHoldReason() == null ? null : CardHoldReasonV2.DELINQUENCY; case KYC, REGULATORY -> c.getComplianceHoldReason() == null ? null : CardHoldReasonV2.valueOf(c.getComplianceHoldReason().name()); }; }
    private static boolean has(CardV2 c, CardHoldReasonV2 r) { return current(c,r) == r; }
    private static void set(CardV2 c, CardHoldReasonV2 r) { switch (r) { case CUSTOMER_FREEZE -> c.setCustomerHoldReason(com.xenomi.cms.core.v2.card.model.CustomerHoldReasonV2.CUSTOMER_FREEZE); case OPS_FREEZE -> c.setOpsHoldReason(com.xenomi.cms.core.v2.card.model.OpsHoldReasonV2.OPS_FREEZE); case RISK_BLOCK, LOST, STOLEN, COMPROMISED, FRAUD_SUSPECTED -> c.setSecurityHoldReason(com.xenomi.cms.core.v2.card.model.SecurityHoldReasonV2.valueOf(r.name())); case DELINQUENCY -> c.setCollectionsHoldReason(com.xenomi.cms.core.v2.card.model.CollectionsHoldReasonV2.DELINQUENCY); case KYC, REGULATORY -> c.setComplianceHoldReason(com.xenomi.cms.core.v2.card.model.ComplianceHoldReasonV2.valueOf(r.name())); } }
    private static void clear(CardV2 c, CardHoldReasonV2 r) { switch (r) { case CUSTOMER_FREEZE -> c.setCustomerHoldReason(null); case OPS_FREEZE -> c.setOpsHoldReason(null); case RISK_BLOCK -> c.setSecurityHoldReason(null); case DELINQUENCY -> c.setCollectionsHoldReason(null); case KYC, REGULATORY -> c.setComplianceHoldReason(null); default -> throw new BusinessRuleViolationException("Permanent security holds cannot be released"); } }
}
