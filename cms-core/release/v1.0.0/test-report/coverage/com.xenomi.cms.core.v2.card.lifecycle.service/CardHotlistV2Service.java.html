package com.xenomi.cms.core.v2.card.lifecycle.service;

import java.nio.charset.StandardCharsets;
import java.security.MessageDigest;
import java.time.OffsetDateTime;
import java.time.ZoneOffset;
import java.util.LinkedHashMap;
import java.util.Map;
import java.util.UUID;

import com.xenomi.cms.core.exception.BusinessRuleViolationException;
import com.xenomi.cms.core.exception.ResourceNotFoundException;
import com.xenomi.cms.core.v2.card.entity.CardV2;
import com.xenomi.cms.core.v2.card.lifecycle.client.CardLifecycleSwitchClient;
import com.xenomi.cms.core.v2.card.lifecycle.dto.HotlistCardV2Request;
import com.xenomi.cms.core.v2.card.lifecycle.dto.HotlistCardV2Response;

import com.xenomi.cms.core.v2.card.model.LifecycleActorTypeV2;

import com.xenomi.cms.core.v2.event.correlation.CorrelationIdResolver;
import jakarta.persistence.EntityManager;
import jakarta.persistence.PersistenceContext;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

@Service
public class CardHotlistV2Service {
    static final String HOTLIST_EVENT = "CARD_HOTLISTED";

    private final CardV2Repository cardRepository;
    private final CardEventV2Repository eventRepository;
    private final CardLifecycleSwitchClient switchClient;
    private final DatabaseUuidV7Generator uuidGenerator;
    private final CardEventSink eventSink;
    private final CorrelationIdResolver correlationIdResolver;
    @PersistenceContext
    private EntityManager entityManager;

    public CardHotlistV2Service(CardV2Repository cardRepository,
            CardEventV2Repository eventRepository,
            CardLifecycleSwitchClient switchClient,
            DatabaseUuidV7Generator uuidGenerator,
            CardEventSink eventSink,
            CorrelationIdResolver correlationIdResolver) {
        this.cardRepository = cardRepository;
        this.eventRepository = eventRepository;
        this.switchClient = switchClient;
        this.uuidGenerator = uuidGenerator;
        this.eventSink = eventSink;
        this.correlationIdResolver = correlationIdResolver;
    }

    @Transactional
    public HotlistCardV2Response hotlist(UUID institutionId, UUID cardId,
            HotlistCardV2Request request, CardHotlistCallerType callerType,
            String idempotencyKey, String requestId, String actorId) {
        CardV2 card = cardRepository.findForUpdate(institutionId, cardId)
                .orElseThrow(() -> new ResourceNotFoundException("Card", cardId.toString()));
        if (card.getClosedAt() != null || card.getReplacedAt() != null || card.getExpiredAt() != null) {
            throw new BusinessRuleViolationException("Terminal cards cannot be hot-listed");
        }
        if (request.getReason() == CardHotlistReason.CUSTOMER_REQUEST
                || request.getReason() == CardHotlistReason.OTHER) {
            throw new BusinessRuleViolationException("Use the hold API for reversible or unspecified restrictions");
        }
        var securityReason = request.getReason() == CardHotlistReason.SUSPECTED_FRAUD
                ? com.xenomi.cms.core.v2.card.model.SecurityHoldReasonV2.FRAUD_SUSPECTED
                : com.xenomi.cms.core.v2.card.model.SecurityHoldReasonV2.valueOf(request.getReason().name());
        if (card.getSecurityHoldReason() == securityReason) {
            throw new BusinessRuleViolationException("The requested security hold is already set");
        }
        if (card.getSecurityHoldReason() != null) {
            throw new BusinessRuleViolationException("A security hold is already set; use escalation");
        }
        String correlation = correlationIdResolver.resolve(requestId);
        long nextVersion = (card.getLifecycleVersion() == null ? 1L : card.getLifecycleVersion()) + 1L;
        switchClient.block(institutionId, cardId, nextVersion, request.getReason(), callerType,
                idempotencyKey, correlation);
        CardStatusV2 previous = card.getCardStatus() == null ? CardStatusV2.PENDING : card.getCardStatus();
        OffsetDateTime now = OffsetDateTime.now(ZoneOffset.UTC);
        card.setSecurityHoldReason(securityReason);
        if (card.getCardForm() != CardFormV2.VIRTUAL && card.getRetiredAt() == null) {
            card.setRetiredAt(now);
            card.setPhysicalRetiredReason(PhysicalRetiredReasonV2.LOST_STOLEN);
            if (card.getPhysicalFulfilmentStatus() == PhysicalFulfilmentStatusV2.REQUESTED
                    || card.getPhysicalFulfilmentStatus() == PhysicalFulfilmentStatusV2.PERSONALIZATION_IN_PROGRESS
                    || card.getPhysicalFulfilmentStatus() == PhysicalFulfilmentStatusV2.PERSONALIZED
                    || card.getPhysicalFulfilmentStatus() == PhysicalFulfilmentStatusV2.IN_EMBOSSING_FILE) {
                card.setPhysicalFulfilmentStatus(PhysicalFulfilmentStatusV2.CANCELLED);
            }
        }
        CardV2 saved = cardRepository.saveFactsAndRefresh(card);
        UUID eventId = uuidGenerator.generate();
        UUID operationId = uuidGenerator.generate();
        Map<String,Object> payload = new LinkedHashMap<>();
        payload.put("reason", request.getReason().name());
        payload.put("reasonDetail", request.getReasonDetail());
        payload.put("callerType", callerType.name());
        payload.put("lifecycleVersion", saved.getLifecycleVersion());
        OffsetDateTime occurred = OffsetDateTime.now(ZoneOffset.UTC);
        entityManager.persist(CardLifecycleOperationV2.builder().operationId(operationId)
                .institutionId(saved.getInstitutionId()).operationKind("CARD_HOTLISTED")
                .idempotencyKeyHash(sha256(idempotencyKey == null ? operationId.toString() : idempotencyKey))
                .requestFingerprint(sha256(saved.getCardId() + ":CARD_HOTLISTED:" + request.getReason().name()))
                .targetCardId(saved.getCardId()).expectedVersion(saved.getLifecycleVersion())
                .actorReference(actorId == null ? "UNKNOWN" : actorId)
                .actorType(callerType == CardHotlistCallerType.CUSTOMER ? LifecycleActorTypeV2.CUSTOMER : LifecycleActorTypeV2.OPS)
                .operationStatus(LifecycleOperationStatusV2.CONFIRMED).resultPayload(payload)
                .createdAt(occurred).completedAt(occurred).build());
        long sequence = eventRepository.findFirstByInstitutionIdAndCardIdOrderByEventSequenceDesc(
                saved.getInstitutionId(), saved.getCardId()).map(e -> e.getEventSequence() + 1).orElse(1L);
        eventRepository.save(CardEventV2.builder().cardEventId(eventId).institutionId(saved.getInstitutionId())
                .cardId(saved.getCardId()).eventType("CARD_HOTLISTED").previousStatus(previous.name())
                .resultingStatus(CardStatusV2.BLOCKED.name()).reasonCode(request.getReason().name())
                .eventPayload(payload).occurredAt(occurred).actorReference(actorId)
                .correlationId(correlationIdResolver.toUuid(correlation)).eventSequence(sequence).operationId(operationId)
                .schemaVersion(1).source("CMS_V2").transitionId("HOTLIST").build());
        HotlistCardV2Response response = HotlistCardV2Response.builder()
                .cardId(saved.getCardId()).institutionId(saved.getInstitutionId())
                .eventId(eventId).previousStatus(previous).cardStatus(CardStatusV2.BLOCKED)
                .reason(request.getReason()).callerType(callerType).lifecycleVersion(saved.getLifecycleVersion())
                .hotlistedAt(occurred).build();
        eventSink.onHotlisted(response, correlation);
        return response;
    }

    private java.util.Optional<HotlistCardV2Response> existingHotlist(
            UUID institutionId, UUID cardId) {
        return eventRepository
                .findFirstByInstitutionIdAndCardIdAndEventTypeOrderByOccurredAtDesc(
                        institutionId, cardId, HOTLIST_EVENT)
                .map(event -> {
                    Object caller = event.getEventPayload().get("callerType");
                    Object version = event.getEventPayload().get("lifecycleVersion");
                    return response(event,
                            version instanceof Number number ? number.longValue() : 1L,
                            caller == null ? CardHotlistCallerType.OPERATIONS
                                    : CardHotlistCallerType.valueOf(caller.toString()));
                });
    }

    private static HotlistCardV2Response response(CardEventV2 event, long version,
            CardHotlistCallerType callerType) {
        return HotlistCardV2Response.builder()
                .cardId(event.getCardId())
                .institutionId(event.getInstitutionId())
                .eventId(event.getCardEventId())
                .previousStatus(CardStatusV2.valueOf(event.getPreviousStatus()))
                .cardStatus(CardStatusV2.BLOCKED)
                .reason(CardHotlistReason.valueOf(event.getReasonCode()))
                .callerType(callerType)
                .lifecycleVersion(version)
                .hotlistedAt(event.getOccurredAt())
                .build();
    }

    private static UUID correlationId(String requestId) {
        try {
            return UUID.fromString(requestId);
        } catch (IllegalArgumentException ignored) {
            return UUID.nameUUIDFromBytes(requestId.getBytes(StandardCharsets.UTF_8));
        }
    }

    private static String trim(String value) {
        return value == null || value.isBlank() ? null : value.trim();
    }

    private static String sha256(String value) {
        try {
            byte[] bytes = MessageDigest.getInstance("SHA-256").digest(value.getBytes(StandardCharsets.UTF_8));
            StringBuilder result = new StringBuilder(64);
            for (byte b : bytes) result.append(String.format("%02x", b));
            return result.toString();
        } catch (Exception e) { throw new IllegalStateException("Unable to hash lifecycle operation", e); }
    }
}
