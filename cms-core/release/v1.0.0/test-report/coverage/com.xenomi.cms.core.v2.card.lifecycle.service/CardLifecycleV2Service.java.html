package com.xenomi.cms.core.v2.card.lifecycle.service;

import java.nio.charset.StandardCharsets;
import java.time.OffsetDateTime;
import java.time.ZoneOffset;
import java.time.LocalDate;
import java.security.MessageDigest;
import java.util.LinkedHashMap;
import java.util.Map;
import java.util.UUID;

import com.xenomi.cms.core.exception.BusinessRuleViolationException;
import com.xenomi.cms.core.exception.ResourceNotFoundException;
import com.xenomi.cms.core.v2.card.entity.CardV2;
import com.xenomi.cms.core.v2.card.lifecycle.client.CardLifecycleSwitchClient;
import com.xenomi.cms.core.v2.card.lifecycle.dto.CardLifecycleActionV2Request;
import com.xenomi.cms.core.v2.card.lifecycle.dto.CardLifecycleActionV2Response;
import com.xenomi.cms.core.v2.card.lifecycle.dto.CardHoldV2Request;
import com.xenomi.cms.core.v2.card.lifecycle.entity.CardEventV2;
import com.xenomi.cms.core.v2.card.lifecycle.entity.CardLifecycleOperationV2;
import com.xenomi.cms.core.v2.card.lifecycle.model.CardHoldReasonV2;
import com.xenomi.cms.core.v2.card.model.CardFormV2;
import com.xenomi.cms.core.v2.card.model.ClosureReasonV2;
import com.xenomi.cms.core.v2.card.model.LifecycleActorTypeV2;
import com.xenomi.cms.core.v2.card.model.LifecycleOperationStatusV2;
import com.xenomi.cms.core.v2.card.model.PhysicalFulfilmentStatusV2;
import com.xenomi.cms.core.v2.card.model.PhysicalRetiredReasonV2;
import com.xenomi.cms.core.v2.card.lifecycle.repository.CardEventV2Repository;
import com.xenomi.cms.core.v2.card.model.CardStatusV2;
import com.xenomi.cms.core.v2.card.repository.CardV2Repository;
import jakarta.persistence.EntityManager;
import jakarta.persistence.PersistenceContext;
import com.xenomi.cms.core.v2.common.DatabaseUuidV7Generator;
import com.xenomi.cms.core.v2.event.catalog.CardEventSink;
import com.xenomi.cms.core.v2.event.correlation.CorrelationIdResolver;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

@Service
public class CardLifecycleV2Service {
    private final CardV2Repository cardRepository;
    private final CardEventV2Repository eventRepository;
    private final CardLifecycleSwitchClient switchClient;
    private final DatabaseUuidV7Generator uuidGenerator;
    private final CardHoldV2Service holdService;
    private final CardEventSink eventSink;
    private final CorrelationIdResolver correlationIdResolver;
    @PersistenceContext
    private EntityManager entityManager;

    public CardLifecycleV2Service(CardV2Repository cardRepository,
            CardEventV2Repository eventRepository, CardLifecycleSwitchClient switchClient,
            DatabaseUuidV7Generator uuidGenerator, CardHoldV2Service holdService,
            CardEventSink eventSink, CorrelationIdResolver correlationIdResolver) {
        this.cardRepository = cardRepository;
        this.eventRepository = eventRepository;
        this.switchClient = switchClient;
        this.uuidGenerator = uuidGenerator;
        this.holdService = holdService;
        this.eventSink = eventSink;
        this.correlationIdResolver = correlationIdResolver;
    }

    @Transactional
    public CardLifecycleActionV2Response activate(UUID institutionId, UUID cardId,
            CardLifecycleActionV2Request request, String idempotencyKey,
            String requestId, String actorId) {
        CardV2 card = load(institutionId, cardId);
        if (terminal(card)) {
            throw new BusinessRuleViolationException("Terminal cards cannot be activated");
        }
        if (card.getUseReadyAt() != null) {
            throw new BusinessRuleViolationException("The card is already active");
        }
        if (card.getExpiryDate() != null && LocalDate.now(ZoneOffset.UTC).isAfter(card.getExpiryDate())) {
            throw new BusinessRuleViolationException("Expired cards cannot be activated");
        }
        if (card.getSecurityHoldReason() != null
                && card.getSecurityHoldReason() != com.xenomi.cms.core.v2.card.model.SecurityHoldReasonV2.RISK_BLOCK) {
            throw new BusinessRuleViolationException("Permanent security holds cannot be activated");
        }
        if (card.getCardForm() != null && card.getCardForm() != CardFormV2.VIRTUAL
                && card.getPhysicalFulfilmentStatus() != PhysicalFulfilmentStatusV2.DELIVERED) {
            throw new BusinessRuleViolationException("Physical card must be delivered before activation");
        }

        CardStatusV2 previous = status(card);
        long nextVersion = (card.getLifecycleVersion() == null ? 1L : card.getLifecycleVersion()) + 1L;
        switchClient.changeStatus(institutionId, cardId, projectedStatusAfterReady(card), nextVersion,
                request.getReason(), idempotencyKey, requestId);
        OffsetDateTime now = OffsetDateTime.now(ZoneOffset.UTC);
        card.setUseReadyAt(now);
        if (card.getCardForm() != null && card.getCardForm() != CardFormV2.VIRTUAL) {
            card.setActivatedAt(now);
        }
        // The V2 schema/state machine names readiness as CARD_READY; CARD_ACTIVATED is not an allowed card_event type.
        return save(card, "CARD_READY", request.getReason(), idempotencyKey, actorId, requestId, previous);
    }

    @Transactional
    public CardLifecycleActionV2Response suspend(UUID institutionId, UUID cardId,
            CardLifecycleActionV2Request request, String idempotencyKey,
            String requestId, String actorId) {
        return holdService.place(institutionId, cardId, holdRequest(request),
                idempotencyKey, requestId, actorId);
    }

    @Transactional
    public CardLifecycleActionV2Response unblock(UUID institutionId, UUID cardId,
            CardLifecycleActionV2Request request, String idempotencyKey,
            String requestId, String actorId) {
        return holdService.release(institutionId, cardId, holdRequest(request),
                idempotencyKey, requestId, actorId);
    }

    @Transactional
    public CardLifecycleActionV2Response close(UUID institutionId, UUID cardId,
            CardLifecycleActionV2Request request, String idempotencyKey,
            String requestId, String actorId) {
        final ClosureReasonV2 reason;
        try {
            reason = ClosureReasonV2.valueOf(request.getReason().trim().toUpperCase());
        } catch (IllegalArgumentException exception) {
            throw new BusinessRuleViolationException("Closure reason must be CUSTOMER_CLOSURE, BANK_CLOSURE or NON_ACTIVATION");
        }
        CardV2 card = load(institutionId, cardId);
        if (card.getClosedAt() != null) {
            throw new BusinessRuleViolationException("The card is already closed");
        }
        CardStatusV2 previous = status(card);
        long nextVersion = (card.getLifecycleVersion() == null ? 1L : card.getLifecycleVersion()) + 1L;
        switchClient.changeStatus(institutionId, cardId, CardStatusV2.CLOSED, nextVersion,
                reason.name(), idempotencyKey, requestId);
        OffsetDateTime now = OffsetDateTime.now(ZoneOffset.UTC);
        card.setClosedAt(now);
        card.setClosureReason(reason);
        retirePhysical(card, now);
        return save(card, "CARD_CLOSED", reason.name(), idempotencyKey, actorId, requestId, previous);
    }

    private CardLifecycleActionV2Response save(CardV2 card, String eventType, String reason,
            String idempotencyKey, String actorId, String requestId, CardStatusV2 previous) {
        String correlation = correlationIdResolver.resolve(requestId);
        UUID eventId = uuidGenerator.generate();
        // card.use_ready_evidence_id is a composite foreign key to this exact event.
        if ("CARD_READY".equals(eventType) && card.getUseReadyAt() != null) {
            card.setUseReadyEvidenceId(eventId);
        }
        CardV2 saved = cardRepository.saveFactsAndRefresh(card);
        OffsetDateTime now = OffsetDateTime.now(ZoneOffset.UTC);
        UUID operationId = uuidGenerator.generate();
        Map<String, Object> payload = new LinkedHashMap<>();
        payload.put("reason", reason);
        payload.put("lifecycleVersion", saved.getLifecycleVersion());
        entityManager.persist(CardLifecycleOperationV2.builder().operationId(operationId)
                .institutionId(saved.getInstitutionId()).operationKind(eventType)
                .idempotencyKeyHash(sha256(idempotencyKey == null ? operationId.toString() : idempotencyKey))
                .requestFingerprint(sha256(saved.getCardId() + ":" + eventType + ":" + reason))
                .targetCardId(saved.getCardId()).expectedVersion(saved.getLifecycleVersion())
                .actorReference(actorId == null ? "UNKNOWN" : actorId).actorType(LifecycleActorTypeV2.OPS)
                .operationStatus(LifecycleOperationStatusV2.CONFIRMED).resultPayload(payload)
                .createdAt(now).completedAt(now).build());
        long sequence = eventRepository.findFirstByInstitutionIdAndCardIdOrderByEventSequenceDesc(
                saved.getInstitutionId(), saved.getCardId()).map(e -> e.getEventSequence() + 1).orElse(1L);
        eventRepository.save(CardEventV2.builder().cardEventId(eventId).institutionId(saved.getInstitutionId())
                .cardId(saved.getCardId()).eventType(eventType).previousStatus(previous == null ? null : previous.name())
                .resultingStatus(status(saved).name()).reasonCode(reason).eventPayload(payload).occurredAt(now)
                .actorReference(actorId).correlationId(correlationIdResolver.toUuid(correlation)).eventSequence(sequence)
                .operationId(operationId).schemaVersion(1).source("CMS_V2").transitionId(eventType).build());
        CardLifecycleActionV2Response response = CardLifecycleActionV2Response.builder()
                .institutionId(saved.getInstitutionId()).cardId(saved.getCardId())
                .eventId(eventId).previousStatus(previous).cardStatus(status(saved))
                .lifecycleVersion(saved.getLifecycleVersion()).reason(reason).occurredAt(now).build();
        if ("CARD_READY".equals(eventType)) {
            eventSink.onActivated(response, correlation);
        } else {
            eventSink.onStatusChanged(response, correlation);
        }
        return response;
    }

    private CardV2 load(UUID institutionId, UUID cardId) {
        return cardRepository.findForUpdate(institutionId, cardId)
                .orElseThrow(() -> new ResourceNotFoundException("Card not found"));
    }

    private static boolean terminal(CardV2 card) {
        return card.getExpiredAt() != null || card.getClosedAt() != null || card.getReplacedAt() != null;
    }

    private static CardStatusV2 status(CardV2 card) {
        return card.getCardStatus() == null ? CardStatusV2.PENDING : card.getCardStatus();
    }

    private static CardStatusV2 projectedStatusAfterReady(CardV2 card) {
        if (card.getCustomerHoldReason() != null || card.getOpsHoldReason() != null) return CardStatusV2.FROZEN;
        if (card.getSecurityHoldReason() != null || card.getCollectionsHoldReason() != null
                || card.getComplianceHoldReason() != null) return CardStatusV2.BLOCKED;
        return CardStatusV2.ACTIVE;
    }

    private static void retirePhysical(CardV2 card, OffsetDateTime now) {
        if (card.getCardForm() == null || card.getCardForm() == CardFormV2.VIRTUAL || card.getRetiredAt() != null) return;
        card.setRetiredAt(now);
        card.setPhysicalRetiredReason(PhysicalRetiredReasonV2.ISSUER);
        if (card.getPhysicalFulfilmentStatus() == PhysicalFulfilmentStatusV2.REQUESTED
                || card.getPhysicalFulfilmentStatus() == PhysicalFulfilmentStatusV2.PERSONALIZATION_IN_PROGRESS
                || card.getPhysicalFulfilmentStatus() == PhysicalFulfilmentStatusV2.PERSONALIZED
                || card.getPhysicalFulfilmentStatus() == PhysicalFulfilmentStatusV2.IN_EMBOSSING_FILE) {
            card.setPhysicalFulfilmentStatus(PhysicalFulfilmentStatusV2.CANCELLED);
        }
    }

    private static UUID correlationId(String requestId) {
        try { return UUID.fromString(requestId); }
        catch (Exception e) { return UUID.nameUUIDFromBytes((requestId == null ? "" : requestId).getBytes(StandardCharsets.UTF_8)); }
    }

    private static String sha256(String value) {
        try {
            byte[] bytes = MessageDigest.getInstance("SHA-256").digest(value.getBytes(StandardCharsets.UTF_8));
            StringBuilder result = new StringBuilder(64);
            for (byte b : bytes) result.append(String.format("%02x", b));
            return result.toString();
        } catch (Exception e) { throw new IllegalStateException("Unable to hash lifecycle operation", e); }
    }

    private static CardHoldV2Request holdRequest(CardLifecycleActionV2Request request) {
        try {
            return CardHoldV2Request.builder()
                    .reason(CardHoldReasonV2.valueOf(request.getReason().trim().toUpperCase()))
                    .reasonDetail(request.getReason()).build();
        } catch (IllegalArgumentException exception) {
            throw new BusinessRuleViolationException(
                    "suspend/unblock compatibility accepts only a supported hold reason");
        }
    }

}
