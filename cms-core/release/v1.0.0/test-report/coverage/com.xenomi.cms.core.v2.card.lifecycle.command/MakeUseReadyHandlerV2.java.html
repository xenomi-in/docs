package com.xenomi.cms.core.v2.card.lifecycle.command;

import com.xenomi.cms.core.exception.BusinessRuleViolationException;
import com.xenomi.cms.core.v2.card.entity.CardV2;
import com.xenomi.cms.core.v2.card.model.*;
import jakarta.persistence.EntityManager;
import jakarta.validation.constraints.NotNull;
import org.springframework.stereotype.Component;
import java.time.LocalDate;
import java.time.ZoneOffset;
import java.util.*;

@Component
public class MakeUseReadyHandlerV2 implements LifecycleCommandHandlerV2<MakeUseReadyHandlerV2.Payload> {
    public record Payload(@NotNull UUID evidenceId, @NotNull ActivationTriggerV2 trigger) {}
    private final EntityManager em;
    public MakeUseReadyHandlerV2(EntityManager em) { this.em = em; }
    public String kind() { return "MAKE_USE_READY"; }
    public String permission() { return "cms:card:manage"; }
    public Set<LifecycleActorTypeV2> actors() { return Set.of(LifecycleActorTypeV2.CUSTOMER, LifecycleActorTypeV2.OPS, LifecycleActorTypeV2.SYSTEM); }
    public Class<Payload> payloadType() { return Payload.class; }
    public VersionRequirement versionRequirement() { return VersionRequirement.CARD; }
    public LockPlan plan(LifecycleCommandRequestV2 request, Payload payload) {
        if (request.targetCardId() == null) throw new IllegalArgumentException("Target card is required");
        UUID family = em.createQuery("select c.panAllocationId from CardV2 c where c.institutionId=:t and c.cardId=:id", UUID.class)
                .setParameter("t", request.institutionId()).setParameter("id", request.targetCardId()).getResultStream().findFirst()
                .orElseThrow(() -> new IllegalArgumentException("Card not found"));
        return new LockPlan(Set.of(family), Set.of(request.targetCardId()));
    }
    public void apply(LifecycleCommandContextV2 c, Payload p) {
        CardV2 card = c.card(c.request().targetCardId());
        if (card.getUseReadyAt() != null) throw new BusinessRuleViolationException("Card is already ready");
        if (card.getExpiredAt() != null || card.getClosedAt() != null || card.getReplacedAt() != null)
            throw new BusinessRuleViolationException("Terminal cards cannot be made ready");
        if (card.getExpiryDate() != null && LocalDate.now(ZoneOffset.UTC).isAfter(card.getExpiryDate()))
            throw new BusinessRuleViolationException("Expired cards cannot be made ready");
        if (card.getSecurityHoldReason() != null && card.getSecurityHoldReason() != SecurityHoldReasonV2.RISK_BLOCK)
            throw new BusinessRuleViolationException("Permanent security holds forbid readiness");
        if (card.getCardForm() != CardFormV2.VIRTUAL && card.getPhysicalActivationStatus() != PhysicalActivationStatusV2.ACTIVATED)
            throw new BusinessRuleViolationException("Physical card must be activated before readiness");
        c.changeCard(card.getCardId(), "MAKE_USE_READY", LifecycleEventTypeV2.CARD_READY, null, null, value -> {
            value.setUseReadyAt(c.now()); value.setUseReadyEvidenceId(p.evidenceId()); value.setActivationTrigger(p.trigger());
        });
    }
}
