package com.xenomi.cms.core.v2.card.lifecycle.command;

import java.util.UUID;
import java.util.concurrent.Executors;
import java.util.concurrent.ScheduledExecutorService;
import java.util.concurrent.TimeUnit;
import org.springframework.beans.factory.DisposableBean;
import org.springframework.beans.factory.SmartInitializingSingleton;
import org.springframework.stereotype.Component;

@Component
public class SecurityReplacementWorkerV2 implements SmartInitializingSingleton, DisposableBean {
    private static final org.slf4j.Logger log =
            org.slf4j.LoggerFactory.getLogger(SecurityReplacementWorkerV2.class);
    private final SecurityReplacementWorkV2 work;
    private final ScheduledExecutorService executor = Executors.newSingleThreadScheduledExecutor(r -> {
        Thread thread = new Thread(r, "cms-v2-security-replacement");
        thread.setDaemon(true);
        return thread;
    });

    public SecurityReplacementWorkerV2(SecurityReplacementWorkV2 work) {
        this.work = work;
    }

    @Override
    public void afterSingletonsInstantiated() {
        executor.scheduleWithFixedDelay(this::poll, 5, 5, TimeUnit.SECONDS);
    }

    private void poll() {
        try {
            for (UUID id : work.pending()) {
                try {
                    work.process(id);
                } catch (RuntimeException failure) {
                    log.warn("Security replacement operation {} remains unresolved ({})",
                            id, failure.getClass().getSimpleName());
                }
            }
        } catch (RuntimeException failure) {
            log.warn("Security replacement poll unavailable ({})",
                    failure.getClass().getSimpleName());
        }
    }

    @Override
    public void destroy() {
        executor.shutdownNow();
    }
}
