package com.xenomi.cms.core.v2.card.lifecycle.command;

import com.xenomi.cms.core.v2.card.model.LifecycleActorTypeV2;
import com.xenomi.cms.core.v2.card.replacement.dto.CreateCardReplacementV2Request;
import com.xenomi.cms.core.v2.product.model.ReplacementRequestChannel;
import jakarta.validation.Valid;
import jakarta.validation.constraints.NotNull;
import java.util.Set;
import java.util.UUID;
import org.springframework.stereotype.Component;

@Component
public class SecurityReplacementHandlerV2
        implements LifecycleCommandHandlerV2<SecurityReplacementHandlerV2.Payload> {
    public record Payload(@NotNull UUID sourceCardId,
            @Valid @NotNull CreateCardReplacementV2Request replacement) {}

    private final SecurityReplacementWorkV2 work;

    public SecurityReplacementHandlerV2(SecurityReplacementWorkV2 work) {
        this.work = work;
    }

    @Override public String kind() { return "REPLACE_SECURITY_NEW_PAN"; }
    @Override public String permission() { return "cms:card-replace:operations"; }
    @Override public Set<LifecycleActorTypeV2> actors() {
        return Set.of(LifecycleActorTypeV2.OPS, LifecycleActorTypeV2.SYSTEM);
    }
    @Override public Class<Payload> payloadType() { return Payload.class; }
    @Override public VersionRequirement versionRequirement() { return VersionRequirement.CREATE; }
    @Override public LockPlan plan(LifecycleCommandRequestV2 request, Payload payload) {
        return work.plan(request, payload);
    }
    @Override public void apply(LifecycleCommandContextV2 context, Payload payload) {
        work.stage(context, payload, ReplacementRequestChannel.OPERATIONS);
    }
}
