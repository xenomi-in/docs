package com.xenomi.cms.core.v2.card.lifecycle.command;

import com.fasterxml.jackson.databind.ObjectMapper;
import com.xenomi.cms.core.v2.card.dto.CreateCardV2Request;
import com.xenomi.cms.core.v2.card.lifecycle.entity.CardLifecycleOperationV2;
import com.xenomi.cms.core.exception.ResourceNotFoundException;
import jakarta.persistence.EntityManager;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import java.util.*;

@Service
public class OriginalCardIssuanceApiV2 {
    private final LifecycleCommandCoordinatorV2 coordinator;
    private final LifecycleCommandAuthorizerV2 authorizer;
    private final OriginalCardIssuanceHandlerV2 handler;
    private final EntityManager em;
    private final ObjectMapper mapper;
    public OriginalCardIssuanceApiV2(LifecycleCommandCoordinatorV2 coordinator, LifecycleCommandAuthorizerV2 authorizer,
            OriginalCardIssuanceHandlerV2 handler, EntityManager em, ObjectMapper mapper) {
        this.coordinator=coordinator; this.authorizer=authorizer; this.handler=handler; this.em=em; this.mapper=mapper;
    }
    public LifecycleCommandReceiptV2 submit(UUID tenant, CreateCardV2Request request, String key) {
        return coordinator.execute(handler.kind(),new LifecycleCommandRequestV2(tenant,request.getAccountReferenceId(),
                null,null,Map.of(),key,null,mapper.valueToTree(new OriginalCardIssuanceHandlerV2.Payload(request))));
    }
    @Transactional(readOnly=true)
    public LifecycleCommandReceiptV2 get(UUID tenant, UUID id) {
        var actor=authorizer.require(tenant,handler);
        var rows=em.createQuery("select o from CardLifecycleOperationV2 o where o.institutionId=:t and o.operationId=:id and o.operationKind=:k and o.actorReference=:a",CardLifecycleOperationV2.class)
                .setParameter("t",tenant).setParameter("id",id).setParameter("k",handler.kind()).setParameter("a",actor.reference()).getResultList();
        if(rows.isEmpty()) throw new ResourceNotFoundException("Issuance operation",id.toString());
        var op=rows.getFirst();
        return new LifecycleCommandReceiptV2(id,op.getOperationStatus(),false,mapper.valueToTree(op.getResultPayload()));
    }
}
