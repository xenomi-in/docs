package com.xenomi.cms.core.v2.card.lifecycle.command;

import org.springframework.beans.factory.DisposableBean;
import org.springframework.beans.factory.SmartInitializingSingleton;
import org.springframework.stereotype.Component;
import java.util.concurrent.*;

/** Dedicated poller; does not enable unrelated application schedules. */
@Component
public class OriginalCardIssuanceWorkerV2 implements SmartInitializingSingleton, DisposableBean {
    private final OriginalCardIssuanceWorkV2 work;
    private final ScheduledExecutorService executor=Executors.newSingleThreadScheduledExecutor(r -> {
        var t=new Thread(r,"cms-v2-original-issuance"); t.setDaemon(true); return t;
    });
    private static final org.slf4j.Logger log=org.slf4j.LoggerFactory.getLogger(OriginalCardIssuanceWorkerV2.class);
    public OriginalCardIssuanceWorkerV2(OriginalCardIssuanceWorkV2 work) { this.work=work; }
    public void afterSingletonsInstantiated() {
        executor.scheduleWithFixedDelay(this::poll,5,5,TimeUnit.SECONDS);
    }
    private void poll() {
        try {
            for(var id:work.pending()) {
                try { work.process(id); }
                catch(RuntimeException failure) { log.warn("Issuance operation {} remains unresolved ({})",id,failure.getClass().getSimpleName()); }
            }
        } catch(RuntimeException failure) { log.warn("Issuance poll unavailable ({})",failure.getClass().getSimpleName()); }
    }
    public void destroy() { executor.shutdownNow(); }
}
