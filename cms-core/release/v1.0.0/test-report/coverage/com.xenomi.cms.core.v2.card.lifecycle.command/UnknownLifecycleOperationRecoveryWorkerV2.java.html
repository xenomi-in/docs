package com.xenomi.cms.core.v2.card.lifecycle.command;

import java.time.Duration;
import java.util.concurrent.Executors;
import java.util.concurrent.ScheduledExecutorService;
import java.util.concurrent.TimeUnit;
import org.springframework.beans.factory.DisposableBean;
import org.springframework.beans.factory.SmartInitializingSingleton;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Component;

/** Periodically terminalizes stale UNKNOWN lifecycle commands without provider replay. */
@Component
public class UnknownLifecycleOperationRecoveryWorkerV2
        implements SmartInitializingSingleton, DisposableBean {
    private static final org.slf4j.Logger log =
            org.slf4j.LoggerFactory.getLogger(
                    UnknownLifecycleOperationRecoveryWorkerV2.class);

    private final UnknownLifecycleOperationReconcilerV2 reconciler;
    private final long pollDelayMillis;
    private final ScheduledExecutorService executor =
            Executors.newSingleThreadScheduledExecutor(runnable -> {
                Thread thread = new Thread(
                        runnable, "cms-v2-unknown-operation-recovery");
                thread.setDaemon(true);
                return thread;
            });

    public UnknownLifecycleOperationRecoveryWorkerV2(
            UnknownLifecycleOperationReconcilerV2 reconciler,
            @Value("${cms.lifecycle.external.unknown-recovery-poll-delay:PT30S}")
            Duration pollDelay) {
        if (pollDelay == null || pollDelay.isZero() || pollDelay.isNegative()) {
            throw new IllegalArgumentException(
                    "cms.lifecycle.external.unknown-recovery-poll-delay must be positive");
        }
        this.reconciler = reconciler;
        this.pollDelayMillis = Math.max(1_000L, pollDelay.toMillis());
    }

    @Override
    public void afterSingletonsInstantiated() {
        executor.scheduleWithFixedDelay(
                this::poll, pollDelayMillis, pollDelayMillis, TimeUnit.MILLISECONDS);
    }

    private void poll() {
        try {
            int reconciled = reconciler.reconcileStale();
            if (reconciled > 0) {
                log.warn(
                        "Terminalized {} lifecycle operation(s) with unresolved provider outcomes",
                        reconciled);
            }
        } catch (RuntimeException failure) {
            log.warn("UNKNOWN lifecycle reconciliation unavailable ({})",
                    failure.getClass().getSimpleName());
        }
    }

    @Override
    public void destroy() {
        executor.shutdownNow();
    }
}
