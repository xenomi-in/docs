package com.xenomi.cms.core.v2.card.lifecycle.command;

import com.fasterxml.jackson.databind.DeserializationFeature;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.xenomi.cms.core.exception.ResourceNotFoundException;
import com.xenomi.cms.core.v2.card.entity.CardV2;
import com.xenomi.cms.core.v2.card.entity.PanAllocationV2;
import com.xenomi.cms.core.v2.card.lifecycle.entity.CardLifecycleOperationV2;
import com.xenomi.cms.core.v2.card.model.LifecycleOperationStatusV2;
import jakarta.persistence.EntityManager;
import jakarta.persistence.LockModeType;
import jakarta.validation.ConstraintViolationException;
import jakarta.validation.Validator;
import org.springframework.stereotype.Service;
import org.springframework.transaction.PlatformTransactionManager;
import org.springframework.transaction.support.TransactionSynchronizationManager;
import org.springframework.transaction.support.TransactionTemplate;
import java.time.OffsetDateTime;
import java.time.ZoneOffset;
import java.util.*;

import static com.xenomi.cms.core.v2.card.lifecycle.command.LifecycleCommandConflictV2.Code.*;

@Service
public class LifecycleCommandCoordinatorV2 {
    private final EntityManager em;
    private final ObjectMapper mapper;
    private final Validator validator;
    private final LifecycleCommandAuthorizerV2 authorizer;
    private final Map<String, LifecycleCommandHandlerV2<?>> handlers;
    private final TransactionTemplate transaction;

    public LifecycleCommandCoordinatorV2(EntityManager em, ObjectMapper mapper, Validator validator,
            LifecycleCommandAuthorizerV2 authorizer, PlatformTransactionManager transactionManager,
            List<LifecycleCommandHandlerV2<?>> handlers) {
        this.em = em; this.mapper = mapper.copy().enable(DeserializationFeature.FAIL_ON_UNKNOWN_PROPERTIES);
        this.validator = validator; this.authorizer = authorizer;
        this.handlers = new HashMap<>();
        for (var handler : handlers) {
            if (!handler.kind().matches("[A-Z][A-Z0-9_]{1,59}") || handler.permission().isBlank()
                    || handler.actors().isEmpty() || !handler.payloadType().isRecord()) {
                throw new IllegalArgumentException("Invalid lifecycle handler definition");
            }
            if (this.handlers.putIfAbsent(handler.kind(), handler) != null) throw new IllegalArgumentException("Duplicate lifecycle command kind");
        }
        transaction = new TransactionTemplate(transactionManager);
        transaction.setTimeout(30);
    }

    public LifecycleCommandReceiptV2 execute(String kind, LifecycleCommandRequestV2 request) {
        if (request == null || request.institutionId() == null || request.accountReferenceId() == null) {
            throw new IllegalArgumentException("Institution and canonical account are required");
        }
        if (TransactionSynchronizationManager.isActualTransactionActive()) {
            throw new IllegalStateException("Lifecycle commands must own their transaction; invoke outside an existing transaction");
        }
        var handler = handlers.get(kind);
        if (handler == null) throw new IllegalArgumentException("Unsupported lifecycle command");
        var actor = authorizer.require(request.institutionId(), handler);
        validateEnvelope(request, handler.versionRequirement());
        return executeTyped(handler, request, actor);
    }

    private <P> LifecycleCommandReceiptV2 executeTyped(LifecycleCommandHandlerV2<P> handler,
            LifecycleCommandRequestV2 request, LifecycleCommandAuthorizerV2.Actor actor) {
        final P payload;
        try { payload = mapper.treeToValue(request.payload(), handler.payloadType()); }
        catch (com.fasterxml.jackson.core.JsonProcessingException invalid) {
            // Do not echo raw request values (or Jackson source snippets) into logs/responses.
            throw new IllegalArgumentException("Invalid lifecycle payload");
        }
        var violations = validator.validate(payload);
        if (!violations.isEmpty()) throw new ConstraintViolationException(violations);
        String keyHash = LifecycleCommandFingerprintV2.hash(request.idempotencyKey());
        String fingerprint = LifecycleCommandFingerprintV2.request(mapper, handler.kind(), request, actor, payload);
        return transaction.execute(status -> {
            // Explicit PostgreSQL transaction setting also works with plain JPA transaction managers.
            em.createNativeQuery("set transaction isolation level read committed").executeUpdate();
            // Serializes equal idempotency keys even when no operation row exists yet.
            // Hash collisions only serialize unrelated work; uniqueness is still enforced by the full DB key.
            long lock = LifecycleCommandFingerprintV2.lockKey(request.institutionId() + ":" + handler.kind() + ":" + keyHash);
            em.createNativeQuery("select 1 from pg_advisory_xact_lock(:key)").setParameter("key", lock).getSingleResult();
            var existing = em.createQuery("select o from CardLifecycleOperationV2 o where o.institutionId = :tenant and o.operationKind = :kind and o.idempotencyKeyHash = :hash", CardLifecycleOperationV2.class)
                    .setParameter("tenant", request.institutionId()).setParameter("kind", handler.kind()).setParameter("hash", keyHash).getResultList();
            if (!existing.isEmpty()) {
                var saved = existing.get(0);
                if (!saved.getRequestFingerprint().equals(fingerprint)) {
                    throw new LifecycleCommandConflictV2(IDEMPOTENCY_CONFLICT, "Idempotency key was already used for another logical command");
                }
                return receipt(saved, true); // Includes PENDING/UNKNOWN; never dispatches external work on replay.
            }
            var accounts = em.createQuery("select a from ExternalAccountReferenceV2 a where a.institutionId = :tenant and a.externalAccountReferenceId = :account")
                    .setParameter("tenant", request.institutionId()).setParameter("account", request.accountReferenceId())
                    .setLockMode(LockModeType.PESSIMISTIC_WRITE).getResultList();
            if (accounts.isEmpty()) throw new ResourceNotFoundException("Account", request.accountReferenceId().toString());
            var plan = Objects.requireNonNull(handler.plan(request, payload));
            var families = new LinkedHashMap<UUID, PanAllocationV2>();
            for (UUID id : sorted(plan.familyIds())) {
                var rows = em.createQuery("select f from PanAllocationV2 f where f.institutionId = :tenant and f.panAllocationId = :id and f.accountReferenceId = :account", PanAllocationV2.class)
                        .setParameter("tenant", request.institutionId()).setParameter("id", id).setParameter("account", request.accountReferenceId())
                        .setLockMode(LockModeType.PESSIMISTIC_WRITE).getResultList();
                if (rows.isEmpty()) throw new ResourceNotFoundException("PAN family", id.toString());
                em.refresh(rows.get(0)); families.put(id, rows.get(0));
            }
            var cards = new LinkedHashMap<UUID, CardV2>();
            for (UUID id : sorted(plan.cardIds())) {
                var rows = em.createQuery("select c from CardV2 c where c.institutionId = :tenant and c.cardId = :id", CardV2.class)
                        .setParameter("tenant", request.institutionId()).setParameter("id", id)
                        .setLockMode(LockModeType.PESSIMISTIC_WRITE).getResultList();
                if (rows.isEmpty()) throw new ResourceNotFoundException("Card", id.toString());
                CardV2 card = rows.get(0); em.refresh(card);
                if (!families.containsKey(card.getPanAllocationId())) throw new IllegalArgumentException("Card family is missing from command lock plan");
                cards.put(id, card);
            }
            checkVersions(handler.versionRequirement(), request, families, cards);
            var operation = CardLifecycleOperationV2.builder()
                    .operationId((UUID) em.createNativeQuery("select uuidv7()", UUID.class).getSingleResult())
                    .institutionId(request.institutionId()).operationKind(handler.kind()).idempotencyKeyHash(keyHash)
                    .requestFingerprint(fingerprint).targetCardId(request.targetCardId()).expectedVersion(request.expectedVersion())
                    .actorReference(actor.reference()).actorType(actor.type()).build();
            em.persist(operation);
            var context = new LifecycleCommandContextV2(em, mapper, request, operation, actor, families, cards);
            handler.apply(context, payload);
            operation.setResultPayload(context.result()); operation.setOperationStatus(context.status());
            if (operation.getOperationStatus() == LifecycleOperationStatusV2.CONFIRMED) operation.setCompletedAt(OffsetDateTime.now(ZoneOffset.UTC));
            em.flush();
            return receipt(operation, false);
        });
    }

    private LifecycleCommandReceiptV2 receipt(CardLifecycleOperationV2 operation, boolean replay) {
        return new LifecycleCommandReceiptV2(operation.getOperationId(), operation.getOperationStatus(), replay,
                mapper.valueToTree(operation.getResultPayload()));
    }

    private static List<UUID> sorted(Set<UUID> ids) {
        // Lexical UUID order matches PostgreSQL's unsigned byte ordering for all writers.
        return ids.stream().sorted(Comparator.comparing(UUID::toString)).toList();
    }

    private static void validateEnvelope(LifecycleCommandRequestV2 r, LifecycleCommandHandlerV2.VersionRequirement version) {
        if (r.idempotencyKey() == null || r.idempotencyKey().isBlank() || r.idempotencyKey().length() > 128
                || r.payload() == null || !r.payload().isObject()) throw new IllegalArgumentException("Invalid lifecycle command envelope");
        if (r.expectedVersion() != null && (r.expectedVersion() <= 0 || r.targetCardId() == null)) throw new IllegalArgumentException("Invalid expected card version");
        if (r.expectedFamilyRevisions().values().stream().anyMatch(v -> v <= 0)) throw new IllegalArgumentException("Invalid expected family revision");
        switch (version) {
            case CARD -> { if (r.targetCardId() == null || r.expectedVersion() == null) throw new IllegalArgumentException("Target card and expected version are required"); }
            case FAMILY -> { if (r.expectedFamilyRevisions().isEmpty()) throw new IllegalArgumentException("Expected family revisions are required"); }
            case CREATE -> { if (r.targetCardId() != null || r.expectedVersion() != null) throw new IllegalArgumentException("Creation has no existing target card version"); }
        }
    }

    private static void checkVersions(LifecycleCommandHandlerV2.VersionRequirement requirement,
            LifecycleCommandRequestV2 r, Map<UUID, PanAllocationV2> families, Map<UUID, CardV2> cards) {
        if (r.targetCardId() != null) {
            var target = cards.get(r.targetCardId());
            if (target == null) throw new IllegalArgumentException("Target card is missing from command lock plan");
            if (r.expectedVersion() != null && !r.expectedVersion().equals(target.getLifecycleVersion())) {
                throw new LifecycleCommandConflictV2(VERSION_CONFLICT, "Card version changed; reload before issuing a new command");
            }
        }
        if (requirement == LifecycleCommandHandlerV2.VersionRequirement.FAMILY
                && !r.expectedFamilyRevisions().keySet().equals(families.keySet())) throw new IllegalArgumentException("Expected revisions must cover the entire family lock plan");
        r.expectedFamilyRevisions().forEach((id, revision) -> {
            if (!families.containsKey(id)) throw new IllegalArgumentException("Unexpected family version");
            if (families.get(id).getFamilyRevision() != revision) throw new LifecycleCommandConflictV2(VERSION_CONFLICT, "Family revision changed; reload before issuing a new command");
        });
    }
}
