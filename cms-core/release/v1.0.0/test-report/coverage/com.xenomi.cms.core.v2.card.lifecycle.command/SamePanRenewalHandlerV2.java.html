package com.xenomi.cms.core.v2.card.lifecycle.command;

import com.xenomi.cms.core.v2.card.model.LifecycleActorTypeV2;
import com.xenomi.cms.core.v2.card.replacement.dto.CreateCardReplacementV2Request;
import jakarta.validation.Valid;
import jakarta.validation.constraints.NotNull;
import org.springframework.stereotype.Component;
import java.util.*;

@Component
public class SamePanRenewalHandlerV2 implements LifecycleCommandHandlerV2<SamePanRenewalHandlerV2.Payload> {
    public record Payload(@NotNull UUID sourceCardId, @Valid @NotNull CreateCardReplacementV2Request replacement) {}
    private final SamePanRenewalWorkV2 work;
    public SamePanRenewalHandlerV2(SamePanRenewalWorkV2 work) { this.work=work; }
    public String kind() { return "RENEW_SAME_PAN"; }
    public String permission() { return "cms:card-replace:operations"; }
    public Set<LifecycleActorTypeV2> actors() { return Set.of(LifecycleActorTypeV2.OPS,LifecycleActorTypeV2.SYSTEM); }
    public Class<Payload> payloadType() { return Payload.class; }
    public VersionRequirement versionRequirement() { return VersionRequirement.CREATE; }
    public LockPlan plan(LifecycleCommandRequestV2 request, Payload payload) { return work.plan(request,payload); }
    public void apply(LifecycleCommandContextV2 context, Payload payload) { work.stage(context,payload); }
}
