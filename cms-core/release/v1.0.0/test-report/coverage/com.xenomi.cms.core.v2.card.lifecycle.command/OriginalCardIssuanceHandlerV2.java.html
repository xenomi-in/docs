package com.xenomi.cms.core.v2.card.lifecycle.command;

import com.xenomi.cms.core.v2.card.dto.CreateCardV2Request;
import com.xenomi.cms.core.v2.card.model.LifecycleActorTypeV2;
import jakarta.validation.Valid;
import jakarta.validation.constraints.NotNull;
import org.springframework.stereotype.Component;
import java.util.*;

@Component
public class OriginalCardIssuanceHandlerV2 implements LifecycleCommandHandlerV2<OriginalCardIssuanceHandlerV2.Payload> {
    public record Payload(@Valid @NotNull CreateCardV2Request card) {}
    private final OriginalCardIssuanceWorkV2 work;
    public OriginalCardIssuanceHandlerV2(OriginalCardIssuanceWorkV2 work) { this.work = work; }
    public String kind() { return "ISSUE_ORIGINAL"; }
    public String permission() { return "cms:card:issue"; }
    public Set<LifecycleActorTypeV2> actors() { return Set.of(LifecycleActorTypeV2.OPS, LifecycleActorTypeV2.SYSTEM); }
    public Class<Payload> payloadType() { return Payload.class; }
    public VersionRequirement versionRequirement() { return VersionRequirement.CREATE; }
    public LockPlan plan(LifecycleCommandRequestV2 request, Payload payload) {
        if (!request.accountReferenceId().equals(payload.card().getAccountReferenceId())) throw new IllegalArgumentException("Account mismatch");
        return new LockPlan(Set.of(), Set.of());
    }
    public void apply(LifecycleCommandContextV2 context, Payload payload) { work.stage(context, payload.card()); }
}
