package com.xenomi.cms.core.v2.card.lifecycle.command;

import com.xenomi.cms.core.exception.BusinessRuleViolationException;
import com.xenomi.cms.core.v2.card.entity.CardV2;
import com.xenomi.cms.core.v2.card.model.*;
import jakarta.persistence.EntityManager;
import jakarta.validation.constraints.NotNull;
import org.springframework.stereotype.Component;
import java.util.*;

@Component
public class CloseCardHandlerV2 implements LifecycleCommandHandlerV2<CloseCardHandlerV2.Payload> {
    public record Payload(@NotNull ClosureReasonV2 closureReason) {}
    private final EntityManager em;
    public CloseCardHandlerV2(EntityManager em) { this.em = em; }
    public String kind() { return "CLOSE_CARD"; }
    public String permission() { return "cms:card:manage"; }
    public Set<LifecycleActorTypeV2> actors() { return Set.of(LifecycleActorTypeV2.CUSTOMER, LifecycleActorTypeV2.OPS, LifecycleActorTypeV2.COMPLIANCE, LifecycleActorTypeV2.COLLECTIONS, LifecycleActorTypeV2.SYSTEM); }
    public Class<Payload> payloadType() { return Payload.class; }
    public VersionRequirement versionRequirement() { return VersionRequirement.CARD; }
    public LockPlan plan(LifecycleCommandRequestV2 request, Payload payload) {
        if (request.targetCardId() == null) throw new IllegalArgumentException("Target card is required");
        UUID family = em.createQuery("select c.panAllocationId from CardV2 c where c.institutionId=:t and c.cardId=:id", UUID.class)
                .setParameter("t", request.institutionId()).setParameter("id", request.targetCardId()).getResultStream().findFirst()
                .orElseThrow(() -> new IllegalArgumentException("Card not found"));
        return new LockPlan(Set.of(family), Set.of(request.targetCardId()));
    }
    public void apply(LifecycleCommandContextV2 c, Payload p) {
        CardV2 card = c.card(c.request().targetCardId());
        if (card.getClosedAt() != null) throw new BusinessRuleViolationException("Card is already closed");
        if (p.closureReason() == ClosureReasonV2.NON_ACTIVATION && c.actor().type() != LifecycleActorTypeV2.SYSTEM)
            throw new BusinessRuleViolationException("NON_ACTIVATION closure is restricted to SYSTEM");
        c.changeCard(card.getCardId(), "CLOSE", LifecycleEventTypeV2.CARD_CLOSED, p.closureReason().name(), null, value -> {
            value.setClosedAt(c.now()); value.setClosureReason(p.closureReason());
            if (value.getCardForm() != CardFormV2.VIRTUAL && value.getRetiredAt() == null) {
                value.setRetiredAt(c.now()); value.setPhysicalRetiredReason(PhysicalRetiredReasonV2.ISSUER);
                if (value.getPhysicalFulfilmentStatus() == PhysicalFulfilmentStatusV2.REQUESTED
                        || value.getPhysicalFulfilmentStatus() == PhysicalFulfilmentStatusV2.PERSONALIZATION_IN_PROGRESS
                        || value.getPhysicalFulfilmentStatus() == PhysicalFulfilmentStatusV2.PERSONALIZED
                        || value.getPhysicalFulfilmentStatus() == PhysicalFulfilmentStatusV2.IN_EMBOSSING_FILE)
                    value.setPhysicalFulfilmentStatus(PhysicalFulfilmentStatusV2.CANCELLED);
            }
        });
    }
}
