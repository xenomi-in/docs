package com.xenomi.cms.core.v2.card.lifecycle.command;

import com.fasterxml.jackson.databind.ObjectMapper;
import com.xenomi.cms.core.v2.card.lifecycle.entity.CardLifecycleOperationV2;
import com.xenomi.cms.core.v2.card.lifecycle.model.CardHotlistCallerType;
import com.xenomi.cms.core.v2.card.replacement.dto.CardReplacementV2Response;
import com.xenomi.cms.core.v2.card.replacement.dto.CreateCardReplacementV2Request;
import com.xenomi.cms.core.v2.card.replacement.service.CardReplacementV2Service;
import jakarta.persistence.EntityManager;
import java.util.Map;
import java.util.UUID;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

@Service
public class SecurityReplacementApiV2 {
    private final LifecycleCommandCoordinatorV2 coordinator;
    private final SecurityReplacementWorkV2 work;
    private final CardReplacementV2Service reads;
    private final ObjectMapper mapper;
    private final EntityManager entityManager;

    public SecurityReplacementApiV2(LifecycleCommandCoordinatorV2 coordinator,
            SecurityReplacementWorkV2 work, CardReplacementV2Service reads,
            ObjectMapper mapper, EntityManager entityManager) {
        this.coordinator = coordinator;
        this.work = work;
        this.reads = reads;
        this.mapper = mapper;
        this.entityManager = entityManager;
    }

    public CardReplacementV2Response submit(UUID institutionId, UUID sourceCardId,
            CreateCardReplacementV2Request input, String idempotencyKey,
            CardHotlistCallerType callerType) {
        String kind = callerType == CardHotlistCallerType.CUSTOMER
                ? "REPLACE_SECURITY_NEW_PAN_CUSTOMER" : "REPLACE_SECURITY_NEW_PAN";
        LifecycleCommandReceiptV2 receipt = coordinator.execute(kind,
                new LifecycleCommandRequestV2(institutionId,
                        work.account(institutionId, sourceCardId), null, null, Map.of(),
                        idempotencyKey, null,
                        mapper.valueToTree(new SecurityReplacementHandlerV2.Payload(
                                sourceCardId, input))));
        CardReplacementV2Response response = reads.get(
                institutionId, sourceCardId, receipt.operationId());
        response.setOperationId(receipt.operationId());
        response.setOperationStatus(receipt.status());
        response.setFailureCode(failureCode(receipt.result()));
        response.setReplay(receipt.replay());
        return response;
    }

    @Transactional(readOnly = true)
    public CardReplacementV2Response get(UUID institutionId, UUID sourceCardId,
            UUID replacementRequestId) {
        CardReplacementV2Response response = reads.get(
                institutionId, sourceCardId, replacementRequestId);
        CardLifecycleOperationV2 operation = entityManager.find(
                CardLifecycleOperationV2.class, replacementRequestId);
        if (operation != null && operation.getInstitutionId().equals(institutionId)) {
            response.setOperationId(operation.getOperationId());
            response.setOperationStatus(operation.getOperationStatus());
            response.setFailureCode(failureCode(
                    mapper.valueToTree(operation.getResultPayload())));
        }
        return response;
    }

    private static String failureCode(com.fasterxml.jackson.databind.JsonNode result) {
        com.fasterxml.jackson.databind.JsonNode value =
                result.path("failure").path("code");
        return value.isTextual() ? value.asText() : null;
    }
}
