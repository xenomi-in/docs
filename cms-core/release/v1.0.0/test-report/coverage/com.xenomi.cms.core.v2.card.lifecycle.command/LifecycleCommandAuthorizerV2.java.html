package com.xenomi.cms.core.v2.card.lifecycle.command;

import com.xenomi.cms.core.v2.card.model.LifecycleActorTypeV2;
import org.springframework.security.access.AccessDeniedException;
import org.springframework.security.core.context.SecurityContextHolder;
import org.springframework.security.oauth2.server.resource.authentication.JwtAuthenticationToken;
import org.springframework.stereotype.Component;
import java.util.UUID;

@Component
public class LifecycleCommandAuthorizerV2 {
    public record Actor(String reference, LifecycleActorTypeV2 type) {}

    public Actor require(UUID institutionId, LifecycleCommandHandlerV2<?> handler) {
        var authentication = SecurityContextHolder.getContext().getAuthentication();
        if (!(authentication instanceof JwtAuthenticationToken jwt) || !jwt.isAuthenticated()) {
            throw new AccessDeniedException("Authenticated lifecycle context is required");
        }
        String subject = jwt.getToken().getSubject();
        if (subject == null || subject.isBlank() || subject.length() > 120) {
            throw new AccessDeniedException("Invalid lifecycle actor subject");
        }
        boolean platform = Boolean.TRUE.equals(jwt.getToken().getClaims().get("platform_access"));
        if (!platform && !institutionId.toString().equals(jwt.getToken().getClaimAsString("institution_id"))) {
            throw new AccessDeniedException("Lifecycle tenant access denied");
        }
        if (jwt.getAuthorities().stream().noneMatch(a -> a.getAuthority().equals("PERM_" + handler.permission()))) {
            throw new AccessDeniedException("Lifecycle permission denied");
        }
        LifecycleActorTypeV2 actor;
        try { actor = LifecycleActorTypeV2.valueOf(jwt.getToken().getClaimAsString("lifecycle_actor_type")); }
        catch (IllegalArgumentException | NullPointerException invalid) {
            throw new AccessDeniedException("A signed lifecycle_actor_type claim is required");
        }
        if (!handler.actors().contains(actor)) throw new AccessDeniedException("Actor cannot perform this lifecycle command");
        return new Actor(subject, actor);
    }
}
