package com.xenomi.cms.core.v2.card.lifecycle.command;

import com.xenomi.cms.core.v2.card.model.LifecycleActorTypeV2;
import com.xenomi.cms.core.v2.product.model.ReplacementRequestChannel;
import java.util.Set;
import org.springframework.stereotype.Component;

@Component
public class CustomerSecurityReplacementHandlerV2
        implements LifecycleCommandHandlerV2<SecurityReplacementHandlerV2.Payload> {
    private final SecurityReplacementWorkV2 work;

    public CustomerSecurityReplacementHandlerV2(SecurityReplacementWorkV2 work) {
        this.work = work;
    }

    @Override public String kind() { return "REPLACE_SECURITY_NEW_PAN_CUSTOMER"; }
    @Override public String permission() { return "cms:card-replace:customer"; }
    @Override public Set<LifecycleActorTypeV2> actors() {
        return Set.of(LifecycleActorTypeV2.CUSTOMER);
    }
    @Override public Class<SecurityReplacementHandlerV2.Payload> payloadType() {
        return SecurityReplacementHandlerV2.Payload.class;
    }
    @Override public VersionRequirement versionRequirement() { return VersionRequirement.CREATE; }
    @Override public LockPlan plan(LifecycleCommandRequestV2 request,
            SecurityReplacementHandlerV2.Payload payload) {
        return work.plan(request, payload);
    }
    @Override public void apply(LifecycleCommandContextV2 context,
            SecurityReplacementHandlerV2.Payload payload) {
        work.stage(context, payload, ReplacementRequestChannel.CUSTOMER);
    }
}
