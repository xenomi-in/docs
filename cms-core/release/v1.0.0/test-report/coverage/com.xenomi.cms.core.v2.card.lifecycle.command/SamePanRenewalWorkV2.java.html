/** Same-PAN renewal: account/family locks, durable provider work and atomic successor creation. */
@Service
public class SamePanRenewalWorkV2 {
    private final EntityManager em;

    private final CardIssuanceV2Service issuance;
    private final CmsCardSecurityService security;
    private final TransactionTemplate tx;
    public SamePanRenewalWorkV2(EntityManager em, ObjectMapper mapper, CardIssuanceV2Service issuance,
            CmsCardSecurityService security, PlatformTransactionManager manager) {
        this.em=em; this.mapper=mapper; this.issuance=issuance; this.security=security;
        tx=new TransactionTemplate(manager); tx.setTimeout(30);
    }
    public UUID account(UUID tenant, UUID sourceId) {
        return tx.execute(s -> family(source(tenant,sourceId)).getAccountReferenceId());
    }
    public LifecycleCommandHandlerV2.LockPlan plan(LifecycleCommandRequestV2 request, SamePanRenewalHandlerV2.Payload payload) {
        var source=source(request.institutionId(),payload.sourceCardId());
        var family=family(source);
        require(family.getAccountReferenceId().equals(request.accountReferenceId()),"Source account mismatch");
        return new LifecycleCommandHandlerV2.LockPlan(Set.of(family.getPanAllocationId()),
                new HashSet<>(em.createQuery("select c.cardId from CardV2 c where c.institutionId=:t and c.panAllocationId=:f",UUID.class)
                        .setParameter("t",request.institutionId()).setParameter("f",family.getPanAllocationId()).getResultList()));
    }
    public void stage(LifecycleCommandContextV2 context, SamePanRenewalHandlerV2.Payload payload) {
        var source=context.card(payload.sourceCardId());
        validateSource(source); restrictions(source.getInstitutionId(),context.request().accountReferenceId());
        var input=payload.replacement();
        require(input.getReplacementReason()==ReplacementReason.RENEWAL,"This lifecycle command supports RENEWAL only");
        require(input.getRequestedPanStrategy()==null || input.getRequestedPanStrategy()==CardReplacementPanStrategy.PRESERVE_EXISTING_PAN,
                "Same-PAN renewal requires PRESERVE_EXISTING_PAN");
        require(context.actor().reference().length()<=100,"Actor reference exceeds storage limit");
        require(em.createQuery("select count(r) from CardReplacementRequestV2 r where r.institutionId=:t and r.sourceCardId=:s",Long.class)
                .setParameter("t",source.getInstitutionId()).setParameter("s",source.getCardId()).getSingleResult()==0,
                "A replacement request already exists for this source");
        var policy=policy(source);
        validatePolicy(policy);
        var create=CreateCardV2Request.builder().customerReferenceId(source.getCustomerReferenceId())
                .accountReferenceId(context.request().accountReferenceId()).productId(source.getProductId())
                .productVersionId(source.getProductVersionId()).binRangeId(source.getBinRangeId())
                .activationPolicyVersionId(source.getActivationPolicyVersion())
                .accountLinkType(CardAccountLinkType.PRIMARY_FUNDING)
                .cardDesignId(source.getCardDesignId()).technologyProfileId(source.getTechnologyProfileId()).cvmProfileId(source.getCvmProfileId())
                .shippingAddressReferenceId(input.getShippingAddressReferenceId()==null ? source.getShippingAddressReferenceId() : input.getShippingAddressReferenceId())
                .productionBatchConfigurationId(input.getProductionBatchConfigurationId())
                .embossedNameLine1(source.getEmbossedNameLine1()).embossedNameLine2(source.getEmbossedNameLine2())
                .embossedNameLine3(source.getEmbossedNameLine3()).embossedNameLine4(source.getEmbossedNameLine4()).build();
        var d=issuance.prepareDraft(source.getInstitutionId(),create,context.actor().reference(),context.operationId(),false);
        require(d.cardForm()==source.getCardForm(),"Renewal cannot change card form; use an explicit migration command");
        var renewalExpiry=d.expiryDate();
        if (!renewalExpiry.isAfter(source.getExpiryDate())) {
            var version=em.find(com.xenomi.cms.core.v2.product.entity.ProductVersionV2.class,source.getProductVersionId());
            renewalExpiry=source.getExpiryDate().plusMonths(version.getValidityMonths()).with(java.time.temporal.TemporalAdjusters.lastDayOfMonth());
        }
        Draft draft=new Draft(d.institutionId(),d.request(),d.actor(),d.operationId(),d.cardId(),source.getPanAllocationId(),source.getPinDataRef(),
                d.issuedAt(),d.issueDate(),renewalExpiry,d.serviceCode(),d.cardForm(),d.formPolicy(),d.profiles(),d.production(),d.range(),d.pinPolicy());
        UUID external=context.stageExternalIntent("renew-credentials","HSM","REISSUE_CREDENTIALS",
                Map.of("cardId",source.getCardId(),"credentialIssuanceRef",context.operationId()));
        em.persist(CardReplacementRequestV2.builder().replacementRequestId(context.operationId()).institutionId(source.getInstitutionId())
                .sourceCardId(source.getCardId()).replacementReason(ReplacementReason.RENEWAL).policyPanStrategy(policy.getPanStrategy())
                .requestedPanStrategy(input.getRequestedPanStrategy()).resolvedPanStrategy(CardReplacementPanStrategy.PRESERVE_EXISTING_PAN)
                .overrideReason(input.getOverrideReason()).requestStatus(CardReplacementRequestStatus.PROCESSING)
                .policySnapshot(map(policy)).requestedAt(context.now()).idempotencyKey(context.operationId())
                .createdBy(context.actor().reference()).operationId(context.operationId()).build());
        var p=new CardRenewalPreparationV2(); p.setOperationId(context.operationId()); p.setInstitutionId(draft.institutionId());
        p.setAccountReferenceId(context.request().accountReferenceId()); p.setSourceCardId(source.getCardId());
        p.setExternalOperationId(external); p.setDraft(map(draft)); em.persist(p); context.awaitExternalCompletion();
    }
    public List<UUID> pending() {
        return tx.execute(s -> em.createQuery("select p.operationId from CardRenewalPreparationV2 p, CardExternalOperationV2 e, CardLifecycleOperationV2 o where p.externalOperationId=e.externalOperationId and p.operationId=o.operationId and o.operationStatus=:pending and (e.nextAttemptAt is null or e.nextAttemptAt<=:now) order by o.createdAt",UUID.class)
                .setParameter("pending",LifecycleOperationStatusV2.PENDING).setParameter("now",now()).setMaxResults(10).getResultList());
    }
    private record Claimed(CardRenewalPreparationV2 preparation, CardV2 source, PanAllocationV2 family) {}
    public void process(UUID id) {
        if(TransactionSynchronizationManager.isActualTransactionActive()) throw new IllegalStateException("Renewal worker must own its transactions");
        Claimed claim=tx.execute(s -> {
            var p=lock(id); var op=em.find(CardLifecycleOperationV2.class,id);
            if(op.getOperationStatus()!=LifecycleOperationStatusV2.PENDING) return null;
            var source=source(p.getInstitutionId(),p.getSourceCardId()); validateSource(source);
            restrictions(p.getInstitutionId(),p.getAccountReferenceId());
            if(p.getProtectedResult()==null) {
                var ext=em.find(CardExternalOperationV2.class,p.getExternalOperationId(),LockModeType.PESSIMISTIC_WRITE);
                if(ext.getOperationStatus()!=LifecycleOperationStatusV2.PENDING) return null;
                ext.setOperationStatus(LifecycleOperationStatusV2.UNKNOWN); ext.setDispatchedAt(now()); ext.setAttemptCount(ext.getAttemptCount()+1);
                op.setOperationStatus(LifecycleOperationStatusV2.UNKNOWN);
            }
            return new Claimed(p,source,family(source));
        });
        if(claim==null) return;
        if(claim.preparation().getProtectedResult()==null) {
            Draft d=mapper.convertValue(claim.preparation().getDraft(),Draft.class);
            var material=security.prepareSamePan(d.institutionId(),claim.source(),claim.family(),d.expiryDate(),d.serviceCode(),"D");
            tx.executeWithoutResult(s -> {
                var p=lock(id); p.setProtectedResult(map(material));
                var ext=em.find(CardExternalOperationV2.class,p.getExternalOperationId()); ext.setOperationStatus(LifecycleOperationStatusV2.CONFIRMED);
                ext.setResolvedAt(now()); ext.setEvidenceReference("cms-renewal-preparation:"+id);
                em.find(CardLifecycleOperationV2.class,id).setOperationStatus(LifecycleOperationStatusV2.PENDING);
            });
        }
        try { tx.executeWithoutResult(s -> finish(id)); }
        catch(RuntimeException failure) {
            tx.executeWithoutResult(s -> em.find(CardExternalOperationV2.class,lock(id).getExternalOperationId()).setNextAttemptAt(now().plusSeconds(30)));
            throw failure;
        }
    }
    private void finish(UUID id) {
        var p=lock(id); var op=em.find(CardLifecycleOperationV2.class,id);
        if(op.getOperationStatus()==LifecycleOperationStatusV2.CONFIRMED) return;
        require(p.getProtectedResult()!=null,"Renewal credentials are unresolved");
        var source=source(p.getInstitutionId(),p.getSourceCardId()); validateSource(source);
        var holds=restrictions(p.getInstitutionId(),p.getAccountReferenceId());
        var request=em.find(CardReplacementRequestV2.class,id);
        var activePolicy=policy(source); validatePolicy(activePolicy);
        require(map(activePolicy).equals(request.getPolicySnapshot()),"Replacement policy changed after acceptance; reconciliation required");
        var draft=mapper.convertValue(p.getDraft(),Draft.class);
        require(draft.expiryDate().isAfter(now().toLocalDate()),"Prepared renewal credentials expired");
        var material=mapper.convertValue(p.getProtectedResult(),CardSecurityMaterial.class);
        var family=family(source);
        var pin=em.find(PinDataV2.class,source.getPinDataRef(),LockModeType.PESSIMISTIC_WRITE);
        require(pin!=null && pin.getInstitutionId().equals(p.getInstitutionId()) && pin.getPanAllocationId().equals(family.getPanAllocationId()),"Shared PIN identity mismatch");
        var successor=issuance.finalizeDraft(draft,material,pin,holds,em,source);
        family.setFamilyRevision(Math.addExact(family.getFamilyRevision(),1));
        // Apply the reissue-off defaults to every live family member, including the successor.
        List<String> eventIds=new ArrayList<>();
        for(var member:members(source)) {
            if(member.getClosedAt()!=null || member.getReplacedAt()!=null || member.getExpiredAt()!=null) continue;
            int changed=em.createNativeQuery("update cms_v2.card_transaction_control set enabled=false,effective_at=now() where institution_id=:t and card_control_profile_id in (select card_control_profile_id from cms_v2.card_control_profile where institution_id=:t and card_id=:c) and enabled and (geography='INTERNATIONAL' or channel in ('ECOM','MOTO','CONTACTLESS_POS'))")
                    .setParameter("t",p.getInstitutionId()).setParameter("c",member.getCardId()).executeUpdate();
            if(changed>0) {
                em.createNativeQuery("update cms_v2.card_control_profile set international_enabled=false,control_version=control_version+1,profile_status='PENDING_SWITCH',updated_at=now(),updated_by=:actor where institution_id=:t and card_id=:c")
                        .setParameter("actor",op.getActorReference()).setParameter("t",p.getInstitutionId()).setParameter("c",member.getCardId()).executeUpdate();
                eventIds.add(event(op,member,successor.getCardId(),"CONTROLS_CHANGED","REISSUE_CREDENTIALS",family.getFamilyRevision()));
            }
        }
        eventIds.add(event(op,successor,source.getCardId(),"CARD_CREATED",successor.getCardStatus()==CardStatusV2.BLOCKED?"CS-03":successor.getCardStatus()==CardStatusV2.FROZEN?"CS-02":"CS-01",family.getFamilyRevision()));
        if(successor.getCardForm()!=CardFormV2.VIRTUAL) eventIds.add(event(op,successor,source.getCardId(),"PHYSICAL_REQUESTED","PF-01",family.getFamilyRevision()));
        eventIds.add(event(op,successor,source.getCardId(),"CREDENTIALS_REISSUED","REISSUE_CREDENTIALS",family.getFamilyRevision()));
        request.setReplacementCardId(successor.getCardId()); request.setRequestStatus(CardReplacementRequestStatus.COMPLETED); request.setCompletedAt(now());
        op.setResultPayload(Map.of("replacementRequestId",id.toString(),"cards",List.of(Map.of("cardId",successor.getCardId().toString(),"cardStatus",successor.getCardStatus().name())),"eventIds",eventIds));
        op.setOperationStatus(LifecycleOperationStatusV2.CONFIRMED); op.setCompletedAt(now()); em.flush();
    }
    private String event(CardLifecycleOperationV2 op, CardV2 card, UUID related, String type, String transition, long revision) {
        long sequence=em.createQuery("select coalesce(max(e.eventSequence),0) from CardEventV2 e where e.institutionId=:t and e.cardId=:c",Long.class)
                .setParameter("t",op.getInstitutionId()).setParameter("c",card.getCardId()).getSingleResult()+1;
        UUID id=id(); var at=now();
        var payload=Map.<String,Object>of("eventId",id.toString(),"operationId",op.getOperationId().toString(),"institutionId",op.getInstitutionId().toString(),
                "cardId",card.getCardId().toString(),"relatedCardId",related.toString(),"eventSequence",sequence,"transitionId",transition,
                "schemaVersion",1,"cardStatus",card.getCardStatus().name(),"actorType",op.getActorType().name());
        em.persist(CardEventV2.builder().cardEventId(id).institutionId(op.getInstitutionId()).cardId(card.getCardId()).relatedCardId(related)
                .operationId(op.getOperationId()).eventType(type).eventSequence(sequence).resultingStatus(card.getCardStatus().name())
                .reasonCode(card.getCardStatusReason()==null?null:card.getCardStatusReason().name()).source("CMS_V2").transitionId(transition)
                .actorReference(op.getActorReference()).correlationId(op.getOperationId()).occurredAt(at).eventPayload(payload).build());
        em.persist(OutboxEventV2.builder().outboxEventId(id()).institutionId(op.getInstitutionId()).operationId(op.getOperationId())
                .aggregateType("CARD").aggregateId(card.getCardId().toString()).eventType(type).eventPayload(payload).correlationId(op.getOperationId())
                .idempotencyKey(id).familyRevision(revision).occurredAt(at).build()); return id.toString();
    }
    private CardRenewalPreparationV2 lock(UUID id) {
        var p=em.find(CardRenewalPreparationV2.class,id);
        if(p==null) throw new ResourceNotFoundException("Renewal preparation",id.toString());
        em.createQuery("select a from ExternalAccountReferenceV2 a where a.institutionId=:t and a.externalAccountReferenceId=:a")
                .setParameter("t",p.getInstitutionId()).setParameter("a",p.getAccountReferenceId()).setLockMode(LockModeType.PESSIMISTIC_WRITE).getSingleResult();
        var source=source(p.getInstitutionId(),p.getSourceCardId());
        em.refresh(family(source),LockModeType.PESSIMISTIC_WRITE);
        for(var c:members(source)) em.refresh(c,LockModeType.PESSIMISTIC_WRITE);
        em.refresh(p,LockModeType.PESSIMISTIC_WRITE); em.refresh(em.find(CardLifecycleOperationV2.class,id),LockModeType.PESSIMISTIC_WRITE); return p;
    }
    private CardV2 source(UUID tenant, UUID id) {
        return em.createQuery("select c from CardV2 c where c.institutionId=:t and c.cardId=:c",CardV2.class)
                .setParameter("t",tenant).setParameter("c",id).getResultStream().findFirst().orElseThrow(()->new ResourceNotFoundException("Card",id.toString()));
    }
    private PanAllocationV2 family(CardV2 source) { return em.find(PanAllocationV2.class,source.getPanAllocationId()); }
    private List<CardV2> members(CardV2 source) {
        return em.createQuery("select c from CardV2 c where c.institutionId=:t and c.panAllocationId=:f order by c.cardId",CardV2.class)
                .setParameter("t",source.getInstitutionId()).setParameter("f",source.getPanAllocationId()).getResultList();
    }
    private void validateSource(CardV2 source) {
        require(source.getClosedAt()==null && source.getReplacedAt()==null,"Closed or replaced sources cannot be renewed");
        for(var member:members(source)) {
            require(member.getSecurityHoldReason()==null || member.getSecurityHoldReason()==SecurityHoldReasonV2.RISK_BLOCK,
                    "A permanently security-restricted PAN family requires a new PAN");
            if(member.getClosedAt()==null && member.getReplacedAt()==null && member.getExpiredAt()==null)
                require(Objects.equals(member.getCustomerHoldReason(),source.getCustomerHoldReason())
                        && Objects.equals(member.getOpsHoldReason(),source.getOpsHoldReason())
                        && Objects.equals(member.getSecurityHoldReason(),source.getSecurityHoldReason()),
                        "PAN-family holds must be synchronized before same-PAN renewal");
        }
    }
    private ProductReplacementPolicyV2 policy(CardV2 source) {
        return em.createQuery("select p from ProductReplacementPolicyV2 p where p.institutionId=:t and p.productVersionId=:v and p.replacementReason=:r and p.policyStatus=:s",ProductReplacementPolicyV2.class)
                .setParameter("t",source.getInstitutionId()).setParameter("v",source.getProductVersionId()).setParameter("r",ReplacementReason.RENEWAL)
                .setParameter("s",ProductPolicyStatus.ACTIVE).getResultStream().findFirst().orElseThrow(()->new BusinessRuleViolationException("An ACTIVE renewal policy is required"));
    }
    private void validatePolicy(ProductReplacementPolicyV2 p) {
        require(p.isReplacementAllowed() && p.getPermittedChannels().contains(ReplacementRequestChannel.OPERATIONS),"Renewal is not allowed through OPERATIONS");
        require(!p.isApprovalRequired(),"Approval-required renewal needs the approval lifecycle implementation");
        require(p.getPanStrategy()==ReplacementPanStrategy.PRESERVE_EXISTING_PAN,"Renewal policy must preserve PAN");
        require(p.getExpiryStrategy()==ReplacementExpiryStrategy.GENERATE_NEW_EXPIRY,"Renewal must generate a new expiry");
        require(p.getOldCardUsagePolicy()==OldCardUsagePolicy.BLOCK_ON_EXPIRY,"Renewal preserves the source until expiry; configure BLOCK_ON_EXPIRY");
        require(p.getAccountLinkStrategy()==ReplacementAccountLinkStrategy.PRESERVE_ACCOUNT_LINK,"Same-PAN renewal must preserve the account");
    }
    private AccountRestrictionSnapshotV2 restrictions(UUID tenant,UUID account) {
        var h=em.find(AccountRestrictionSnapshotV2.class,new AccountRestrictionSnapshotV2.Key(tenant,account),LockModeType.PESSIMISTIC_READ);
        require(h!=null && h.getValidUntil().isAfter(now()) && !h.getObservedAt().isAfter(now()),"Fresh authoritative account restrictions required"); return h;
    }
    private Map<String,Object> map(Object value) { return mapper.convertValue(value,new TypeReference<LinkedHashMap<String,Object>>(){}); }
    private UUID id() { return (UUID)em.createNativeQuery("select uuidv7()",UUID.class).getSingleResult(); }
    private static OffsetDateTime now() { return OffsetDateTime.now(ZoneOffset.UTC); }
    private static void require(boolean ok,String message) { if(!ok) throw new BusinessRuleViolationException(message); }
