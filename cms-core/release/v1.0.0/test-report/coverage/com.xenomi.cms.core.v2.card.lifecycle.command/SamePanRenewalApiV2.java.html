package com.xenomi.cms.core.v2.card.lifecycle.command;

import com.fasterxml.jackson.databind.ObjectMapper;
import com.xenomi.cms.core.v2.card.lifecycle.entity.CardLifecycleOperationV2;
import com.xenomi.cms.core.v2.card.replacement.dto.*;
import com.xenomi.cms.core.v2.card.replacement.service.CardReplacementV2Service;
import jakarta.persistence.EntityManager;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import java.util.*;

@Service
public class SamePanRenewalApiV2 {
    private final LifecycleCommandCoordinatorV2 coordinator;
    private final SamePanRenewalWorkV2 work;
    private final CardReplacementV2Service reads;
    private final ObjectMapper mapper;
    private final EntityManager em;
    public SamePanRenewalApiV2(LifecycleCommandCoordinatorV2 coordinator,SamePanRenewalWorkV2 work,
            CardReplacementV2Service reads,ObjectMapper mapper,EntityManager em) {
        this.coordinator=coordinator; this.work=work; this.reads=reads; this.mapper=mapper; this.em=em;
    }
    public CardReplacementV2Response submit(UUID tenant,UUID source,CreateCardReplacementV2Request input,String key) {
        var receipt=coordinator.execute("RENEW_SAME_PAN",new LifecycleCommandRequestV2(tenant,work.account(tenant,source),null,null,
                Map.of(),key,null,mapper.valueToTree(new SamePanRenewalHandlerV2.Payload(source,input))));
        var result=reads.get(tenant,source,receipt.operationId());
        result.setOperationId(receipt.operationId()); result.setOperationStatus(receipt.status());
        result.setFailureCode(failureCode(receipt.result()));
        result.setReplay(receipt.replay()); return result;
    }
    @Transactional(readOnly=true)
    public CardReplacementV2Response get(UUID tenant,UUID source,UUID request) {
        var result=reads.get(tenant,source,request);
        var op=em.find(CardLifecycleOperationV2.class,request);
        if(op!=null && op.getInstitutionId().equals(tenant)) {
            result.setOperationId(op.getOperationId());
            result.setOperationStatus(op.getOperationStatus());
            result.setFailureCode(failureCode(mapper.valueToTree(op.getResultPayload())));
        }
        return result;
    }
    private static String failureCode(com.fasterxml.jackson.databind.JsonNode result) {
        var value=result.path("failure").path("code");
        return value.isTextual()?value.asText():null;
    }
}
