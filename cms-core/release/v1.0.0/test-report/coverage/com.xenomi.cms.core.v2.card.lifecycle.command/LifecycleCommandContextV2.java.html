package com.xenomi.cms.core.v2.card.lifecycle.command;

import com.fasterxml.jackson.databind.ObjectMapper;
import com.xenomi.cms.core.v2.card.entity.CardV2;
import com.xenomi.cms.core.v2.card.entity.PanAllocationV2;
import com.xenomi.cms.core.v2.card.lifecycle.entity.*;
import com.xenomi.cms.core.v2.card.model.LifecycleOperationStatusV2;
import com.xenomi.cms.core.v2.card.repository.CardFactsV2RepositoryImpl;
import jakarta.persistence.EntityManager;
import java.time.OffsetDateTime;
import java.time.ZoneOffset;
import java.util.*;
import java.util.function.Consumer;

/** Transaction-local helpers. A handler must perform its transition/ownership guards before mutation. */
public final class LifecycleCommandContextV2 {
    private final EntityManager em;
    private final ObjectMapper mapper;
    private final LifecycleCommandRequestV2 request;
    private final CardLifecycleOperationV2 operation;
    private final LifecycleCommandAuthorizerV2.Actor actor;
    private final Map<UUID, PanAllocationV2> families;
    private final Map<UUID, CardV2> cards;
    private final Set<UUID> changedFamilies = new LinkedHashSet<>();
    private final List<UUID> events = new ArrayList<>();
    private final Map<String, CardExternalOperationV2> intents = new LinkedHashMap<>();
    private final Map<UUID, Long> sequences = new HashMap<>();
    private boolean awaitingExternalCompletion;

    LifecycleCommandContextV2(EntityManager em, ObjectMapper mapper, LifecycleCommandRequestV2 request,
            CardLifecycleOperationV2 operation, LifecycleCommandAuthorizerV2.Actor actor,
            Map<UUID, PanAllocationV2> families, Map<UUID, CardV2> cards) {
        this.em = em; this.mapper = mapper; this.request = request; this.operation = operation;
        this.actor = actor; this.families = families; this.cards = cards;
    }

    public UUID operationId() { return operation.getOperationId(); }
    public LifecycleCommandAuthorizerV2.Actor actor() { return actor; }
    public LifecycleCommandRequestV2 request() { return request; }
    public UUID newId() { return (UUID) em.createNativeQuery("select uuidv7()", UUID.class).getSingleResult(); }
    public OffsetDateTime now() { return OffsetDateTime.now(ZoneOffset.UTC); }
    public CardV2 card(UUID id) {
        CardV2 card = cards.get(id);
        if (card == null) throw new IllegalArgumentException("Card is outside the locked command plan");
        return card;
    }

    /** All related rows must be in the locked plan. Returns false without an event for unchanged facts. */
    public boolean changeCard(UUID id, String transition, LifecycleEventTypeV2 eventType,
            String reason, UUID relatedCardId, Consumer<CardV2> change) {
        Objects.requireNonNull(eventType);
        if (transition == null || !transition.matches("[A-Z][A-Z0-9_-]{1,59}")) throw new IllegalArgumentException("Invalid transition ID");
        if (reason != null && !reason.matches("[A-Z_]{1,50}")) throw new IllegalArgumentException("Invalid reason code");
        if (relatedCardId != null) card(relatedCardId);
        CardV2 card = card(id);
        Map<String, Object> before = snapshot(card);
        change.accept(card);
        if (before.equals(snapshot(card))) return false;
        card.setUpdatedBy(actor.reference());
        new CardFactsV2RepositoryImpl(em).saveFactsAndRefresh(card);
        Map<String, Object> after = snapshot(card);
        PanAllocationV2 family = families.get(card.getPanAllocationId());
        if (changedFamilies.add(family.getPanAllocationId())) {
            family.setFamilyRevision(Math.addExact(family.getFamilyRevision(), 1L));
        }
        persistEvent(card, transition, eventType, reason, relatedCardId, before, after);
        return true;
    }

    /** Records a second event for an already-committed compound fact change. */
    public void recordCardEvent(UUID id, String transition, LifecycleEventTypeV2 eventType,
            String reason, UUID relatedCardId) {
        Objects.requireNonNull(eventType);
        if (transition == null || !transition.matches("[A-Z][A-Z0-9_-]{1,59}")) throw new IllegalArgumentException("Invalid transition ID");
        if (reason != null && !reason.matches("[A-Z_]{1,50}")) throw new IllegalArgumentException("Invalid reason code");
        if (relatedCardId != null) card(relatedCardId);
        CardV2 card = card(id);
        Map<String, Object> current = snapshot(card);
        persistEvent(card, transition, eventType, reason, relatedCardId, current, current);
    }

    private void persistEvent(CardV2 card, String transition, LifecycleEventTypeV2 eventType,
            String reason, UUID relatedCardId, Map<String, Object> before,
            Map<String, Object> after) {
        UUID id = card.getCardId();
        PanAllocationV2 family = families.get(card.getPanAllocationId());
        long sequence = sequences.compute(id, (key, previous) -> previous == null
                ? Math.addExact(em.createQuery("select coalesce(max(e.eventSequence), 0) from CardEventV2 e where e.institutionId = :tenant and e.cardId = :card", Long.class)
                    .setParameter("tenant", request.institutionId()).setParameter("card", id).getSingleResult(), 1L)
                : Math.addExact(previous, 1L));
        UUID eventId = newId();
        UUID correlation = request.correlationId() == null ? operationId() : request.correlationId();
        var payload = new LinkedHashMap<String, Object>();
        payload.put("eventId", eventId.toString()); payload.put("operationId", operationId().toString());
        payload.put("eventSequence", sequence); payload.put("transitionId", transition);
        payload.put("source", "CMS_V2"); payload.put("schemaVersion", 1);
        payload.put("institutionId", request.institutionId().toString());
        payload.put("cardId", id.toString()); payload.put("relatedCardId", relatedCardId == null ? null : relatedCardId.toString());
        payload.put("actorReference", actor.reference()); payload.put("actorType", actor.type().name());
        payload.put("before", before); payload.put("after", after);
        var occurredAt = now();
        em.persist(CardEventV2.builder().cardEventId(eventId).institutionId(request.institutionId()).cardId(id)
                .operationId(operationId()).eventType(eventType.name()).eventSequence(sequence)
                .previousStatus((String) before.get("cardStatus")).resultingStatus(card.getCardStatus().name())
                .reasonCode(reason).source("CMS_V2").transitionId(transition).relatedCardId(relatedCardId)
                .actorReference(actor.reference()).correlationId(correlation).occurredAt(occurredAt)
                .eventPayload(payload).build());
        // Domain event publication only. This is not a switch installation/acknowledgement.
        em.persist(OutboxEventV2.builder().outboxEventId(newId()).institutionId(request.institutionId())
                .operationId(operationId()).aggregateType("CARD").aggregateId(id.toString())
                .eventType(eventType.name()).eventPayload(payload).correlationId(correlation)
                .idempotencyKey(eventId).familyRevision(family.getFamilyRevision()).occurredAt(occurredAt).build());
        events.add(eventId);
    }

    /** Store references only; no clear PAN, PIN, offsets or key material in durable intent JSON. */
    public UUID stageExternalIntent(String effectKey, String provider, String kind, Map<String, UUID> references) {
        requireText(effectKey, 200); requireText(provider, 100); requireText(kind, 60);
        var payload = new TreeMap<String, Object>();
        references.forEach((key, value) -> {
            if (!Set.of("cardId", "panAllocationId", "credentialIssuanceRef", "keyAssignmentId",
                    "productionRequestId", "tokenReferenceId", "evidenceId").contains(key)) {
                throw new IllegalArgumentException("External intent reference is not allowlisted");
            }
            payload.put(key, Objects.requireNonNull(value).toString());
        });
        var existing = intents.get(effectKey);
        if (existing != null) {
            if (!existing.getProvider().equals(provider) || !existing.getOperationKind().equals(kind)
                    || !existing.getIntentPayload().equals(payload)) throw new IllegalArgumentException("Conflicting external effect key");
            return existing.getExternalOperationId();
        }
        var intent = CardExternalOperationV2.builder().externalOperationId(newId()).institutionId(request.institutionId())
                .operationId(operationId()).effectKey(effectKey).provider(provider).operationKind(kind)
                .intentPayload(payload).build();
        em.persist(intent); intents.put(effectKey, intent); return intent.getExternalOperationId();
    }

    public void awaitExternalCompletion() { awaitingExternalCompletion = true; }

    LifecycleOperationStatusV2 status() {
        if (awaitingExternalCompletion && intents.isEmpty()) throw new IllegalStateException("A pending operation needs durable external work");
        return awaitingExternalCompletion ? LifecycleOperationStatusV2.PENDING : LifecycleOperationStatusV2.CONFIRMED;
    }

    Map<String, Object> result() {
        var result = new LinkedHashMap<String, Object>();
        result.put("cards", cards.values().stream().map(this::snapshot).toList());
        result.put("eventIds", events.stream().map(UUID::toString).toList());
        result.put("externalOperationIds", intents.values().stream().map(i -> i.getExternalOperationId().toString()).toList());
        var revisions = new TreeMap<String, Long>();
        changedFamilies.forEach(id -> revisions.put(id.toString(), families.get(id).getFamilyRevision()));
        result.put("familyRevisions", revisions);
        return result;
    }

    private static void requireText(String value, int max) {
        if (value == null || value.isBlank() || value.length() > max) throw new IllegalArgumentException("Invalid external intent metadata");
    }

    /** Explicit allowlist: never serialize the credential-bearing CardV2 entity. */
    private Map<String, Object> snapshot(CardV2 c) {
        var s = new LinkedHashMap<String, Object>();
        s.put("cardId", c.getCardId()); s.put("panAllocationId", c.getPanAllocationId());
        s.put("lifecycleVersion", c.getLifecycleVersion()); s.put("cardStatus", c.getCardStatus());
        s.put("cardStatusReason", c.getCardStatusReason()); s.put("cardForm", c.getCardForm());
        s.put("digitalAvailable", c.isDigitalAvailable()); s.put("cardSequenceNumber", c.getCardSequenceNumber());
        s.put("useReadyAt", c.getUseReadyAt()); s.put("useReadyEvidenceId", c.getUseReadyEvidenceId());
        s.put("expiredAt", c.getExpiredAt()); s.put("closedAt", c.getClosedAt()); s.put("closureReason", c.getClosureReason());
        s.put("replacedAt", c.getReplacedAt()); s.put("customerHoldReason", c.getCustomerHoldReason());
        s.put("opsHoldReason", c.getOpsHoldReason()); s.put("securityHoldReason", c.getSecurityHoldReason());
        s.put("collectionsHoldReason", c.getCollectionsHoldReason()); s.put("complianceHoldReason", c.getComplianceHoldReason());
        s.put("customerLockReason", c.getCustomerLockReason()); s.put("issuerLockReason", c.getIssuerLockReason());
        s.put("physicalFulfilmentStatus", c.getPhysicalFulfilmentStatus()); s.put("physicalActivationStatus", c.getPhysicalActivationStatus());
        s.put("activatedAt", c.getActivatedAt()); s.put("retiredAt", c.getRetiredAt()); s.put("physicalRetiredReason", c.getPhysicalRetiredReason());
        s.put("persoFileRef", c.getPersoFileRef()); s.put("courierProviderId", c.getCourierProviderId()); s.put("awbNumber", c.getAwbNumber());
        s.put("shippingAddressReferenceId", c.getShippingAddressReferenceId());
        s.put("firstIntentAt", c.getFirstIntentAt()); s.put("activationTrigger", c.getActivationTrigger());
        s.put("consentRequestedAt", c.getConsentRequestedAt()); s.put("closureDueAt", c.getClosureDueAt());
        return mapper.convertValue(s, new com.fasterxml.jackson.core.type.TypeReference<LinkedHashMap<String, Object>>() {});
    }
}
