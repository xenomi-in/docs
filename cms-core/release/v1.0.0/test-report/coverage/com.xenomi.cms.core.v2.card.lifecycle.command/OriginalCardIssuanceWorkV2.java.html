package com.xenomi.cms.core.v2.card.lifecycle.command;

import com.fasterxml.jackson.core.type.TypeReference;

import com.xenomi.cms.core.v2.bin.model.*;

/** Durable credentials boundary: commit UNKNOWN before dispatch, persist protected results before finalization. */
@Service
public class OriginalCardIssuanceWorkV2 {
    private final EntityManager em;
    private final ObjectMapper mapper;
    private final CardIssuanceV2Service issuance;
    private final CmsCardSecurityService security;
    private final PinLifecycleV2Service pins;
    private final TransactionTemplate tx;
    private final SecureRandom random = new SecureRandom();
    public OriginalCardIssuanceWorkV2(EntityManager em, ObjectMapper mapper, CardIssuanceV2Service issuance,
            CmsCardSecurityService security, PinLifecycleV2Service pins, PlatformTransactionManager tm) {
        this.em=em; this.mapper=mapper; this.issuance=issuance; this.security=security; this.pins=pins;
        this.tx=new TransactionTemplate(tm); this.tx.setTimeout(30);
    }
    public void stage(LifecycleCommandContextV2 context, CreateCardV2Request request) {
        if(context.actor().reference().length()>100) throw new IllegalArgumentException("Issuance actor reference exceeds 100 characters");
        restrictions(context.request().institutionId(), request.getAccountReferenceId());
        Draft draft=issuance.prepareDraft(context.request().institutionId(), request, context.actor().reference(), context.operationId());
        var range=draft.range(); // Already locked by prepareDraft; no HSM call has occurred.
        if (range.getIssuedPanCount() >= range.getUsableCapacity()) throw new IllegalStateException("PAN range exhausted");
        BigInteger start=new BigInteger(range.getRangeStart()), end=new BigInteger(range.getRangeEnd());
        long width=end.subtract(start).add(BigInteger.ONE).longValueExact();
        long candidate=range.getNextSequenceValue();
        boolean found=false;
        for (int attempts=0; attempts<10000; attempts++) {
            if (range.getAllocationStrategy()==AllocationStrategy.RANDOM) candidate=random.nextLong(width);
            if (candidate>=width) break;
            String pan=start.add(BigInteger.valueOf(candidate)).toString();
            pan="0".repeat(Math.max(0, range.getRangeStart().length()-pan.length()))+pan;
            if (LuhnAlgorithm.validate(pan) && em.createQuery("select count(p) from CardIssuancePreparationV2 p where p.institutionId=:t and p.binRangeId=:r and p.candidateOffset=:o", Long.class)
                    .setParameter("t",draft.institutionId()).setParameter("r",range.getBinRangeId()).setParameter("o",candidate).getSingleResult()==0) { found=true; break; }
            if (range.getAllocationStrategy()==AllocationStrategy.SEQUENTIAL) candidate++;
        }
        if (!found) throw new IllegalStateException("No available PAN candidate");
        range.setIssuedPanCount(Math.addExact(range.getIssuedPanCount(),1));
        if (range.getAllocationStrategy()==AllocationStrategy.SEQUENTIAL) range.setNextSequenceValue(Math.addExact(candidate,1));
        if (range.getIssuedPanCount()>=range.getUsableCapacity()) range.setRangeStatus(BinRangeStatus.EXHAUSTED);
        UUID external=context.stageExternalIntent("prepare-credentials","HSM","PREPARE_CREDENTIALS",
                Map.of("credentialIssuanceRef",context.operationId()));
        var preparation=new CardIssuancePreparationV2();
        preparation.setOperationId(context.operationId()); preparation.setInstitutionId(draft.institutionId());
        preparation.setAccountReferenceId(request.getAccountReferenceId()); preparation.setExternalOperationId(external);
        preparation.setBinRangeId(range.getBinRangeId()); preparation.setCandidateOffset(candidate); preparation.setDraft(map(draft));
        em.persist(preparation); context.awaitExternalCompletion();
    }
    public List<UUID> pending() {
        return tx.execute(s -> em.createQuery("select p.operationId from CardIssuancePreparationV2 p, CardExternalOperationV2 e, CardLifecycleOperationV2 o where p.externalOperationId=e.externalOperationId and p.operationId=o.operationId and o.operationStatus=:pending and (e.nextAttemptAt is null or e.nextAttemptAt<=:now) order by o.createdAt", UUID.class)
                .setParameter("pending",LifecycleOperationStatusV2.PENDING).setParameter("now",now()).setMaxResults(10).getResultList());
    }
    /** Invoked outside a transaction; UNKNOWN work is deliberately never automatically dispatched again. */
    public void process(UUID operationId) {
        if(org.springframework.transaction.support.TransactionSynchronizationManager.isActualTransactionActive())
            throw new IllegalStateException("Issuance worker must own its database transactions");
        CardIssuancePreparationV2 claimed=tx.execute(s -> {
            var p=lock(operationId); var op=em.find(CardLifecycleOperationV2.class, operationId);
            if (op.getOperationStatus()!=LifecycleOperationStatusV2.PENDING) return null;
            if (p.getProtectedResult()!=null) return p;
            var ext=em.find(CardExternalOperationV2.class,p.getExternalOperationId(),LockModeType.PESSIMISTIC_WRITE);
            if (ext.getOperationStatus()!=LifecycleOperationStatusV2.PENDING) return null;
            ext.setOperationStatus(LifecycleOperationStatusV2.UNKNOWN); ext.setDispatchedAt(now());
            ext.setAttemptCount(Math.addExact(ext.getAttemptCount(),1)); op.setOperationStatus(LifecycleOperationStatusV2.UNKNOWN);
            return p;
        });
        if (claimed==null) return;
        if (claimed.getProtectedResult()==null) {
            Draft draft=mapper.convertValue(claimed.getDraft(),Draft.class);
            // No DB transaction across any provider command.
            CardSecurityMaterial material=security.generateReserved(draft.institutionId(),draft.range(),claimed.getCandidateOffset(),draft.expiryDate(),draft.serviceCode());
            CardV2 transientCard=CardV2.builder().cardId(draft.cardId()).institutionId(draft.institutionId())
                    .binId(draft.range().getBinId()).productVersionId(draft.request().getProductVersionId())
                    .panAllocationId(draft.panAllocationId()).pinDataRef(draft.pinDataId())
                    .panEncrypted(material.encryptedPan()).dataEncryptionKeyId(material.encryptionKeyId()).build();
            PinDataV2 pin=pins.prepareOriginal(transientCard,draft.actor(),draft.pinPolicy());
            tx.executeWithoutResult(s -> {
                var p=lock(operationId);
                p.setProtectedResult(Map.of("credentials",map(material),"pin",map(pin)));
                var ext=em.find(CardExternalOperationV2.class,p.getExternalOperationId(),LockModeType.PESSIMISTIC_WRITE);
                ext.setOperationStatus(LifecycleOperationStatusV2.CONFIRMED); ext.setResolvedAt(now());
                ext.setEvidenceReference("cms-issuance-preparation:"+operationId);
                em.find(CardLifecycleOperationV2.class,operationId).setOperationStatus(LifecycleOperationStatusV2.PENDING);
            });
        }
        try { tx.executeWithoutResult(s -> finalizePrepared(operationId)); }
        catch (RuntimeException failure) {
            tx.executeWithoutResult(s -> {
                var p=lock(operationId);
                em.find(CardExternalOperationV2.class,p.getExternalOperationId()).setNextAttemptAt(now().plusSeconds(30));
            });
            throw failure;
        }
    }
    private void finalizePrepared(UUID operationId) {
        var p=lock(operationId); var op=em.find(CardLifecycleOperationV2.class,operationId);
        if (op.getOperationStatus()==LifecycleOperationStatusV2.CONFIRMED) return;
        if (p.getProtectedResult()==null) throw new IllegalStateException("Protected credentials missing");
        Draft draft=mapper.convertValue(p.getDraft(),Draft.class);
        if (!draft.expiryDate().isAfter(now().toLocalDate())) throw new IllegalStateException("Prepared credentials expired");
        var hold=restrictions(p.getInstitutionId(),p.getAccountReferenceId());
        CardV2 card=issuance.finalizeDraft(draft,mapper.convertValue(p.getProtectedResult().get("credentials"),CardSecurityMaterial.class),
                mapper.convertValue(p.getProtectedResult().get("pin"),PinDataV2.class),hold,em);
        List<String> events=new ArrayList<>();
        events.add(event(op,card,"CARD_CREATED",card.getCardStatus()==CardStatusV2.BLOCKED ? "CS-03" : "CS-01",1));
        if (card.getCardForm()!=CardFormV2.VIRTUAL) events.add(event(op,card,"PHYSICAL_REQUESTED","PF-01",2));
        op.setResultPayload(Map.of("cards",List.of(Map.of("cardId",card.getCardId().toString(),
                "cardStatus",card.getCardStatus().name(),"cardForm",card.getCardForm().name(),"lifecycleVersion",card.getLifecycleVersion())),
                "eventIds",events,"familyRevisions",Map.of(card.getPanAllocationId().toString(),1),
                "externalOperationIds",List.of(p.getExternalOperationId().toString())));
        op.setOperationStatus(LifecycleOperationStatusV2.CONFIRMED); op.setCompletedAt(now());
    }
    private String event(CardLifecycleOperationV2 op, CardV2 card, String type, String transition, long sequence) {
        UUID id=id(); var at=now();
        Map<String,Object> payload=new LinkedHashMap<>(Map.of("eventId",id.toString(),"operationId",op.getOperationId().toString(),
                "cardId",card.getCardId().toString(),"institutionId",card.getInstitutionId().toString(),
                "eventSequence",sequence,"transitionId",transition,"schemaVersion",1,
                "cardStatus",card.getCardStatus().name(),"actorType",op.getActorType().name()));
        payload.put("source","CMS_V2"); payload.put("actorReference",op.getActorReference());
        payload.put("occurredAt",at.toString()); payload.put("correlationId",op.getOperationId().toString());
        var facts=new LinkedHashMap<String,Object>();
        facts.put("cardId",card.getCardId().toString()); facts.put("panAllocationId",card.getPanAllocationId().toString());
        facts.put("cardStatus",card.getCardStatus().name()); facts.put("cardStatusReason",card.getCardStatusReason()==null?null:card.getCardStatusReason().name());
        facts.put("cardForm",card.getCardForm().name()); facts.put("lifecycleVersion",card.getLifecycleVersion());
        facts.put("useReadyAt",null); facts.put("activatedAt",null); facts.put("cardSequenceNumber",null);
        facts.put("physicalFulfilmentStatus",card.getPhysicalFulfilmentStatus().name());
        facts.put("physicalActivationStatus",card.getPhysicalActivationStatus().name());
        payload.put("before",type.equals("CARD_CREATED")?Map.of():facts); payload.put("after",facts);
        em.persist(CardEventV2.builder().cardEventId(id).institutionId(card.getInstitutionId()).cardId(card.getCardId())
                .operationId(op.getOperationId()).eventType(type).eventSequence(sequence).resultingStatus(card.getCardStatus().name())
                .reasonCode(card.getCardStatusReason()==null?null:card.getCardStatusReason().name()).source("CMS_V2")
                .transitionId(transition).actorReference(op.getActorReference()).correlationId(op.getOperationId())
                .occurredAt(at).eventPayload(payload).build());
        em.persist(OutboxEventV2.builder().outboxEventId(id()).institutionId(card.getInstitutionId()).operationId(op.getOperationId())
                .aggregateType("CARD").aggregateId(card.getCardId().toString()).eventType(type).eventPayload(payload)
                .correlationId(op.getOperationId()).idempotencyKey(id).familyRevision(1L).occurredAt(at).build());
        return id.toString();
    }
    private CardIssuancePreparationV2 lock(UUID id) {
        var p=em.find(CardIssuancePreparationV2.class,id);
        if(p==null) throw new IllegalArgumentException("Issuance preparation missing");
        em.createQuery("select a from ExternalAccountReferenceV2 a where a.institutionId=:t and a.externalAccountReferenceId=:a")
                .setParameter("t",p.getInstitutionId()).setParameter("a",p.getAccountReferenceId()).setLockMode(LockModeType.PESSIMISTIC_WRITE).getSingleResult();
        em.refresh(p,LockModeType.PESSIMISTIC_WRITE);
        var operation=em.find(CardLifecycleOperationV2.class,id,LockModeType.PESSIMISTIC_WRITE); em.refresh(operation);
        return p;
    }
    private AccountRestrictionSnapshotV2 restrictions(UUID tenant, UUID account) {
        var hold=em.find(AccountRestrictionSnapshotV2.class,new AccountRestrictionSnapshotV2.Key(tenant,account),LockModeType.PESSIMISTIC_READ);
        if(hold==null || !hold.getValidUntil().isAfter(now()) || hold.getObservedAt().isAfter(now()))
            throw new IllegalStateException("Fresh authoritative account restrictions required");
        return hold;
    }
    private Map<String,Object> map(Object value) { return mapper.convertValue(value,new TypeReference<LinkedHashMap<String,Object>>(){}); }
    private UUID id() { return (UUID)em.createNativeQuery("select uuidv7()",UUID.class).getSingleResult(); }
    private static OffsetDateTime now() { return OffsetDateTime.now(ZoneOffset.UTC); }
}
