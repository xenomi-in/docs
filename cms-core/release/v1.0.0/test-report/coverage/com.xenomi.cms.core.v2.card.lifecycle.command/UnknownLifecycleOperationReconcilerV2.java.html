package com.xenomi.cms.core.v2.card.lifecycle.command;

import com.xenomi.cms.core.v2.card.lifecycle.entity.CardExternalOperationV2;
import com.xenomi.cms.core.v2.card.lifecycle.entity.CardLifecycleOperationV2;
import com.xenomi.cms.core.v2.card.model.LifecycleOperationStatusV2;
import com.xenomi.cms.core.v2.card.replacement.entity.CardReplacementRequestV2;
import com.xenomi.cms.core.v2.card.replacement.model.CardReplacementRequestStatus;
import jakarta.persistence.EntityManager;
import jakarta.persistence.LockModeType;
import java.time.Duration;
import java.time.OffsetDateTime;
import java.time.ZoneOffset;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;
import java.util.UUID;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Service;
import org.springframework.transaction.PlatformTransactionManager;
import org.springframework.transaction.support.TransactionTemplate;

/**
 * Terminates lifecycle commands whose external provider outcome stayed uncertain
 * beyond the configured reconciliation window.
 *
 * <p>The external operation deliberately remains UNKNOWN: FAILED would incorrectly
 * claim that the provider did not perform the operation. The parent lifecycle
 * command becomes FAILED so clients do not poll forever. Its reserved credential
 * candidate remains quarantined and is never returned to the available pool.</p>
 */
@Service
public class UnknownLifecycleOperationReconcilerV2 {
    public static final String FAILURE_CODE = "PROVIDER_OUTCOME_UNRESOLVED";
    private static final int BATCH_SIZE = 100;

    private final EntityManager entityManager;
    private final TransactionTemplate transaction;
    private final Duration uncertaintyTimeout;

    public UnknownLifecycleOperationReconcilerV2(
            EntityManager entityManager,
            PlatformTransactionManager transactionManager,
            @Value("${cms.lifecycle.external.unknown-timeout:PT15M}")
            Duration uncertaintyTimeout) {
        if (uncertaintyTimeout == null || uncertaintyTimeout.isZero()
                || uncertaintyTimeout.isNegative()) {
            throw new IllegalArgumentException(
                    "cms.lifecycle.external.unknown-timeout must be positive");
        }
        this.entityManager = entityManager;
        this.transaction = new TransactionTemplate(transactionManager);
        this.transaction.setTimeout(30);
        this.uncertaintyTimeout = uncertaintyTimeout;
    }

    public int reconcileStale() {
        OffsetDateTime now = OffsetDateTime.now(ZoneOffset.UTC);
        OffsetDateTime cutoff = now.minus(uncertaintyTimeout);
        List<UUID> candidates = transaction.execute(status -> entityManager.createQuery(
                        "select e.externalOperationId "
                                + "from CardExternalOperationV2 e, CardLifecycleOperationV2 o "
                                + "where e.institutionId=o.institutionId "
                                + "and e.operationId=o.operationId "
                                + "and e.operationStatus=:unknown "
                                + "and o.operationStatus=:unknown "
                                + "and e.dispatchedAt is not null "
                                + "and e.dispatchedAt<=:cutoff "
                                + "order by e.dispatchedAt,e.externalOperationId",
                        UUID.class)
                .setParameter("unknown", LifecycleOperationStatusV2.UNKNOWN)
                .setParameter("cutoff", cutoff)
                .setMaxResults(BATCH_SIZE)
                .getResultList());
        if (candidates == null || candidates.isEmpty()) return 0;

        int reconciled = 0;
        for (UUID externalOperationId : candidates) {
            Boolean changed = transaction.execute(status ->
                    failParentIfStillUnknown(externalOperationId, now));
            if (Boolean.TRUE.equals(changed)) reconciled++;
        }
        return reconciled;
    }

    private boolean failParentIfStillUnknown(
            UUID externalOperationId, OffsetDateTime now) {
        CardExternalOperationV2 external = entityManager.find(
                CardExternalOperationV2.class, externalOperationId,
                LockModeType.PESSIMISTIC_WRITE);
        if (external == null
                || external.getOperationStatus() != LifecycleOperationStatusV2.UNKNOWN
                || external.getDispatchedAt() == null
                || external.getDispatchedAt().isAfter(now.minus(uncertaintyTimeout))) {
            return false;
        }

        CardLifecycleOperationV2 operation = entityManager.find(
                CardLifecycleOperationV2.class, external.getOperationId(),
                LockModeType.PESSIMISTIC_WRITE);
        if (operation == null
                || operation.getOperationStatus() != LifecycleOperationStatusV2.UNKNOWN
                || !operation.getInstitutionId().equals(external.getInstitutionId())) {
            return false;
        }

        Map<String, Object> result = new LinkedHashMap<>(
                operation.getResultPayload() == null
                        ? Map.of() : operation.getResultPayload());
        result.put("failure", Map.of(
                "code", FAILURE_CODE,
                "stage", "EXTERNAL_PROVIDER",
                "externalOperationId", externalOperationId.toString(),
                "failedAt", now.toString()));
        operation.setResultPayload(result);
        operation.setOperationStatus(LifecycleOperationStatusV2.FAILED);
        operation.setCompletedAt(now);

        // Replacement requests share the lifecycle operation ID. If present,
        // expose the same terminal business state to replacement polling.
        List<CardReplacementRequestV2> replacements = entityManager.createQuery(
                        "select r from CardReplacementRequestV2 r "
                                + "where r.institutionId=:tenant "
                                + "and r.operationId=:operation",
                        CardReplacementRequestV2.class)
                .setParameter("tenant", external.getInstitutionId())
                .setParameter("operation", operation.getOperationId())
                .setLockMode(LockModeType.PESSIMISTIC_WRITE)
                .getResultList();
        for (CardReplacementRequestV2 replacement : replacements) {
            if (replacement.getRequestStatus()
                    == CardReplacementRequestStatus.PROCESSING) {
                replacement.setRequestStatus(CardReplacementRequestStatus.FAILED);
                replacement.setCompletedAt(now);
            }
        }
        entityManager.flush();
        return true;
    }
}
