    public UUID account(UUID institutionId, UUID sourceCardId) {

    public LifecycleCommandHandlerV2.LockPlan plan(LifecycleCommandRequestV2 request,
            SecurityReplacementHandlerV2.Payload payload) {
        CardV2 source = source(request.institutionId(), payload.sourceCardId());
        PanAllocationV2 family = family(source);
        require(family.getAccountReferenceId().equals(request.accountReferenceId()),
                "Source account mismatch");
        Set<UUID> cardIds = new HashSet<>(entityManager.createQuery(
                        "select c.cardId from CardV2 c where c.institutionId=:tenant and c.panAllocationId=:family",
                        UUID.class)
                .setParameter("tenant", request.institutionId())
                .setParameter("family", family.getPanAllocationId())
                .getResultList());
        return new LifecycleCommandHandlerV2.LockPlan(
                Set.of(family.getPanAllocationId()), cardIds);
    }

    public void stage(LifecycleCommandContextV2 context,
            SecurityReplacementHandlerV2.Payload payload,
            ReplacementRequestChannel requestChannel) {
        CardV2 source = context.card(payload.sourceCardId());
        var input = payload.replacement();
        require(SUPPORTED.contains(input.getReplacementReason()),
                "This lifecycle command supports LOST, STOLEN, COMPROMISED and FRAUD only");
        require(input.getRequestedPanStrategy() == null
                        || input.getRequestedPanStrategy()
                                == CardReplacementPanStrategy.GENERATE_NEW_PAN,
                "Permanent-security replacement requires GENERATE_NEW_PAN");
        require(source.getClosedAt() == null && source.getReplacedAt() == null
                        && source.getExpiredAt() == null,
                "A terminal source card cannot be security-replaced");
        require(context.actor().reference().length() <= 100,
                "Actor reference exceeds storage limit");
        require(entityManager.createQuery(
                        "select count(r) from CardReplacementRequestV2 r where r.institutionId=:tenant and r.sourceCardId=:source",
                        Long.class)
                .setParameter("tenant", source.getInstitutionId())
                .setParameter("source", source.getCardId()).getSingleResult() == 0,
                "A replacement request already exists for this source");
        AccountRestrictionSnapshotV2 restrictions = restrictions(
                source.getInstitutionId(), context.request().accountReferenceId());
        ProductReplacementPolicyV2 policy = policy(source, input.getReplacementReason());
        validatePolicy(policy, requestChannel);

        SecurityHoldReasonV2 securityReason = securityReason(input.getReplacementReason());
        for (CardV2 member : members(source)) {
            SecurityHoldReasonV2 existing = member.getSecurityHoldReason();
            require(existing == null || existing == SecurityHoldReasonV2.RISK_BLOCK
                            || existing == securityReason,
                    "The PAN family already has a different permanent security restriction");
        }

        CreateCardV2Request create = CreateCardV2Request.builder()
                .customerReferenceId(source.getCustomerReferenceId())
                .accountReferenceId(context.request().accountReferenceId())
                .productId(source.getProductId())
                .productVersionId(source.getProductVersionId())
                .binRangeId(source.getBinRangeId())
                .activationPolicyVersionId(source.getActivationPolicyVersion())
                .accountLinkType(CardAccountLinkType.PRIMARY_FUNDING)
                .cardDesignId(source.getCardDesignId())
                .technologyProfileId(source.getTechnologyProfileId())
                .cvmProfileId(source.getCvmProfileId())
                .shippingAddressReferenceId(input.getShippingAddressReferenceId() == null
                        ? source.getShippingAddressReferenceId()
                        : input.getShippingAddressReferenceId())
                .productionBatchConfigurationId(input.getProductionBatchConfigurationId())
                .embossedNameLine1(source.getEmbossedNameLine1())
                .embossedNameLine2(source.getEmbossedNameLine2())
                .embossedNameLine3(source.getEmbossedNameLine3())
                .embossedNameLine4(source.getEmbossedNameLine4())
                .build();
        Draft draft = issuance.prepareDraft(source.getInstitutionId(), create,
                context.actor().reference(), context.operationId(), true);
        require(draft.cardForm() == source.getCardForm(),
                "Security replacement cannot change card form");
        long candidate = reserveCandidate(draft);

        CardReplacementRequestV2 replacement = CardReplacementRequestV2.builder()
                .replacementRequestId(context.operationId())
                .institutionId(source.getInstitutionId())
                .sourceCardId(source.getCardId())
                .replacementReason(input.getReplacementReason())
                .policyPanStrategy(policy.getPanStrategy())
                .requestedPanStrategy(input.getRequestedPanStrategy())
                .resolvedPanStrategy(CardReplacementPanStrategy.GENERATE_NEW_PAN)
                .overrideReason(input.getOverrideReason())
                .requestStatus(CardReplacementRequestStatus.PROCESSING)
                .policySnapshot(map(policy))
                .requestedAt(context.now())
                .idempotencyKey(context.operationId())
                .createdBy(context.actor().reference())
                .operationId(context.operationId())
                .build();
        entityManager.persist(replacement);

        for (CardV2 member : members(source)) {
            if (member.getClosedAt() != null || member.getReplacedAt() != null) continue;
            if (member.getSecurityHoldReason() == securityReason) continue;
            boolean physicalWasRetired = member.getRetiredAt() != null;
            boolean changed = context.changeCard(member.getCardId(), "SECURITY_REPLACEMENT_BLOCK",
                    LifecycleEventTypeV2.CARD_BLOCKED, securityReason.name(), null, card -> {
                        card.setSecurityHoldReason(securityReason);
                        retirePhysical(card, input.getReplacementReason(), context.now());
                    });
            if (changed) {
                context.recordCardEvent(member.getCardId(), "HD-01",
                        LifecycleEventTypeV2.HOLD_PLACED, securityReason.name(), null);
                if (member.getCardForm() != CardFormV2.VIRTUAL
                        && !physicalWasRetired && member.getRetiredAt() != null) {
                    context.recordCardEvent(member.getCardId(), "PA-04",
                            LifecycleEventTypeV2.PHYSICAL_RETIRED,
                            member.getPhysicalRetiredReason().name(), null);
                }
            }
            cancelPendingProduction(member);
        }

        UUID externalOperationId = context.stageExternalIntent(
                "prepare-replacement-credentials", "HSM", "PREPARE_CREDENTIALS",
                Map.of("credentialIssuanceRef", context.operationId()));
        CardSecurityReplacementPreparationV2 preparation =
                new CardSecurityReplacementPreparationV2();
        preparation.setOperationId(context.operationId());
        preparation.setInstitutionId(source.getInstitutionId());
        preparation.setAccountReferenceId(context.request().accountReferenceId());
        preparation.setSourceCardId(source.getCardId());
        preparation.setExternalOperationId(externalOperationId);
        preparation.setBinRangeId(draft.range().getBinRangeId());
        preparation.setCandidateOffset(candidate);
        preparation.setDraft(map(draft));
        entityManager.persist(preparation);
        context.awaitExternalCompletion();
    }

    public List<UUID> pending() {
        return transaction.execute(status -> entityManager.createQuery(
                        "select p.operationId from CardSecurityReplacementPreparationV2 p, CardExternalOperationV2 e, CardLifecycleOperationV2 o "
                                + "where p.externalOperationId=e.externalOperationId and p.operationId=o.operationId "
                                + "and o.operationStatus=:pending and (e.nextAttemptAt is null or e.nextAttemptAt<=:now) order by o.createdAt",
                        UUID.class)
                .setParameter("pending", LifecycleOperationStatusV2.PENDING)
                .setParameter("now", now()).setMaxResults(10).getResultList());
    }

    public void process(UUID operationId) {
        if (TransactionSynchronizationManager.isActualTransactionActive()) {
            throw new IllegalStateException("Security replacement worker must own its transactions");
        }
        CardSecurityReplacementPreparationV2 claimed = transaction.execute(status -> {
            CardSecurityReplacementPreparationV2 preparation = lock(operationId);
            CardLifecycleOperationV2 operation = entityManager.find(
                    CardLifecycleOperationV2.class, operationId);
            if (operation.getOperationStatus() != LifecycleOperationStatusV2.PENDING) return null;
            if (preparation.getProtectedResult() != null) return preparation;
            CardExternalOperationV2 external = entityManager.find(
                    CardExternalOperationV2.class, preparation.getExternalOperationId(),
                    LockModeType.PESSIMISTIC_WRITE);
            if (external.getOperationStatus() != LifecycleOperationStatusV2.PENDING) return null;
            external.setOperationStatus(LifecycleOperationStatusV2.UNKNOWN);
            external.setDispatchedAt(now());
            external.setAttemptCount(Math.addExact(external.getAttemptCount(), 1));
            operation.setOperationStatus(LifecycleOperationStatusV2.UNKNOWN);
            return preparation;
        });
        if (claimed == null) return;

        if (claimed.getProtectedResult() == null) {
            Draft draft = mapper.convertValue(claimed.getDraft(), Draft.class);
            CardReplacementRequestV2 request = transaction.execute(status -> entityManager.find(
                    CardReplacementRequestV2.class, operationId));
            CardSecurityMaterial material = security.generateReserved(draft.institutionId(),
                    draft.range(), claimed.getCandidateOffset(), draft.expiryDate(),
                    draft.serviceCode());
            CardV2 transientCard = CardV2.builder().cardId(draft.cardId())
                    .institutionId(draft.institutionId()).binId(draft.range().getBinId())
                    .productVersionId(draft.request().getProductVersionId())
                    .panAllocationId(draft.panAllocationId()).pinDataRef(draft.pinDataId())
                    .panEncrypted(material.encryptedPan())
                    .dataEncryptionKeyId(material.encryptionKeyId()).build();
            ProductReplacementPolicyV2 activePolicy = transaction.execute(status -> policy(
                    source(claimed.getInstitutionId(), claimed.getSourceCardId()),
                    request.getReplacementReason()));
            PinDataV2 pin = pins.prepareReplacement(transientCard, draft.actor(),
                    draft.pinPolicy(), activePolicy.getPinStrategy());
            transaction.executeWithoutResult(status -> {
                CardSecurityReplacementPreparationV2 preparation = lock(operationId);
                preparation.setProtectedResult(Map.of(
                        "credentials", map(material), "pin", map(pin)));
                CardExternalOperationV2 external = entityManager.find(
                        CardExternalOperationV2.class, preparation.getExternalOperationId(),
                        LockModeType.PESSIMISTIC_WRITE);
                external.setOperationStatus(LifecycleOperationStatusV2.CONFIRMED);
                external.setResolvedAt(now());
                external.setEvidenceReference("cms-security-replacement-preparation:" + operationId);
                entityManager.find(CardLifecycleOperationV2.class, operationId)
                        .setOperationStatus(LifecycleOperationStatusV2.PENDING);
            });
        }

        try {
            transaction.executeWithoutResult(status -> finalizePrepared(operationId));
        } catch (RuntimeException failure) {
            transaction.executeWithoutResult(status -> {
                CardSecurityReplacementPreparationV2 preparation = lock(operationId);
                entityManager.find(CardExternalOperationV2.class,
                        preparation.getExternalOperationId()).setNextAttemptAt(now().plusSeconds(30));
            });
            throw failure;
        }
    }

    private void finalizePrepared(UUID operationId) {
        CardSecurityReplacementPreparationV2 preparation = lock(operationId);
        CardLifecycleOperationV2 operation = entityManager.find(
                CardLifecycleOperationV2.class, operationId);
        if (operation.getOperationStatus() == LifecycleOperationStatusV2.CONFIRMED) return;
        require(preparation.getProtectedResult() != null,
                "Replacement credentials are unresolved");
        CardV2 source = source(preparation.getInstitutionId(), preparation.getSourceCardId());
        CardReplacementRequestV2 request = entityManager.find(
                CardReplacementRequestV2.class, operationId, LockModeType.PESSIMISTIC_WRITE);
        SecurityHoldReasonV2 expectedReason = securityReason(request.getReplacementReason());
        require(source.getSecurityHoldReason() == expectedReason,
                "Source security restriction changed after replacement acceptance");
        ProductReplacementPolicyV2 activePolicy = policy(source, request.getReplacementReason());
        validatePolicy(activePolicy, operation.getActorType()
                == com.xenomi.cms.core.v2.card.model.LifecycleActorTypeV2.CUSTOMER
                        ? ReplacementRequestChannel.CUSTOMER
                        : ReplacementRequestChannel.OPERATIONS);
        require(map(activePolicy).equals(request.getPolicySnapshot()),
                "Replacement policy changed after acceptance; reconciliation required");
        Draft draft = mapper.convertValue(preparation.getDraft(), Draft.class);
        require(draft.expiryDate().isAfter(now().toLocalDate()),
                "Prepared replacement credentials expired");
        AccountRestrictionSnapshotV2 holds = restrictions(
                preparation.getInstitutionId(), preparation.getAccountReferenceId());
        CardV2 successor = issuance.finalizeNewPanReplacementDraft(draft,
                mapper.convertValue(preparation.getProtectedResult().get("credentials"),
                        CardSecurityMaterial.class),
                mapper.convertValue(preparation.getProtectedResult().get("pin"), PinDataV2.class),
                holds, entityManager, source, request.getReplacementReason());

        List<String> eventIds = new ArrayList<>(entityManager.createQuery(
                        "select cast(e.cardEventId as string) from CardEventV2 e where e.institutionId=:tenant and e.operationId=:operation order by e.occurredAt,e.cardEventId",
                        String.class)
                .setParameter("tenant", preparation.getInstitutionId())
                .setParameter("operation", operationId).getResultList());
        long sequence = 1;
        eventIds.add(event(operation, successor, source.getCardId(), "CARD_CREATED",
                successor.getCardStatus() == CardStatusV2.BLOCKED ? "CS-03" : "CS-01",
                sequence++));
        if (successor.getCardForm() != CardFormV2.VIRTUAL) {
            eventIds.add(event(operation, successor, source.getCardId(),
                    "PHYSICAL_REQUESTED", "PF-01", sequence++));
        }
        eventIds.add(event(operation, successor, source.getCardId(),
                "CREDENTIALS_REISSUED", "REISSUE_CREDENTIALS", sequence));
        request.setReplacementCardId(successor.getCardId());
        request.setRequestStatus(CardReplacementRequestStatus.COMPLETED);
        request.setCompletedAt(now());
        operation.setResultPayload(Map.of(
                "replacementRequestId", operationId.toString(),
                "cards", List.of(Map.of(
                        "cardId", successor.getCardId().toString(),
                        "cardStatus", successor.getCardStatus().name(),
                        "cardForm", successor.getCardForm().name(),
                        "lifecycleVersion", successor.getLifecycleVersion())),
                "eventIds", eventIds,
                "externalOperationIds", List.of(
                        preparation.getExternalOperationId().toString())));
        operation.setOperationStatus(LifecycleOperationStatusV2.CONFIRMED);
        operation.setCompletedAt(now());
        entityManager.flush();
    }

    private long reserveCandidate(Draft draft) {
        var range = draft.range();
        if (range.getIssuedPanCount() >= range.getUsableCapacity()) {
            throw new BusinessRuleViolationException("PAN range exhausted");
        }
        BigInteger start = new BigInteger(range.getRangeStart());
        BigInteger end = new BigInteger(range.getRangeEnd());
        long width = end.subtract(start).add(BigInteger.ONE).longValueExact();
        long candidate = range.getNextSequenceValue();
        boolean found = false;
        for (int attempts = 0; attempts < 10_000; attempts++) {
            if (range.getAllocationStrategy() == AllocationStrategy.RANDOM) {
                candidate = random.nextLong(width);
            }
            if (candidate >= width) break;
            String pan = start.add(BigInteger.valueOf(candidate)).toString();
            pan = "0".repeat(Math.max(0, range.getRangeStart().length() - pan.length())) + pan;
            if (LuhnAlgorithm.validate(pan) && candidateAvailable(
                    draft.institutionId(), range.getBinRangeId(), candidate)) {
                found = true;
                break;
            }
            if (range.getAllocationStrategy() == AllocationStrategy.SEQUENTIAL) candidate++;
        }
        if (!found) throw new BusinessRuleViolationException("No available PAN candidate");
        range.setIssuedPanCount(Math.addExact(range.getIssuedPanCount(), 1));
        if (range.getAllocationStrategy() == AllocationStrategy.SEQUENTIAL) {
            range.setNextSequenceValue(Math.addExact(candidate, 1));
        }
        if (range.getIssuedPanCount() >= range.getUsableCapacity()) {
            range.setRangeStatus(BinRangeStatus.EXHAUSTED);
        }
        return candidate;
    }

    private boolean candidateAvailable(UUID institutionId, UUID rangeId, long candidate) {
        long issuanceCount = entityManager.createQuery(
                        "select count(p) from CardIssuancePreparationV2 p where p.institutionId=:tenant and p.binRangeId=:range and p.candidateOffset=:candidate",
                        Long.class)
                .setParameter("tenant", institutionId).setParameter("range", rangeId)
                .setParameter("candidate", candidate).getSingleResult();
        long replacementCount = entityManager.createQuery(
                        "select count(p) from CardSecurityReplacementPreparationV2 p where p.institutionId=:tenant and p.binRangeId=:range and p.candidateOffset=:candidate",
                        Long.class)
                .setParameter("tenant", institutionId).setParameter("range", rangeId)
                .setParameter("candidate", candidate).getSingleResult();
        return issuanceCount == 0 && replacementCount == 0;
    }

    private CardSecurityReplacementPreparationV2 lock(UUID operationId) {
        CardSecurityReplacementPreparationV2 preparation = entityManager.find(
                CardSecurityReplacementPreparationV2.class, operationId);
        if (preparation == null) {
            throw new ResourceNotFoundException(
                    "Security replacement preparation", operationId.toString());
        }
        entityManager.createQuery(
                        "select a from ExternalAccountReferenceV2 a where a.institutionId=:tenant and a.externalAccountReferenceId=:account")
                .setParameter("tenant", preparation.getInstitutionId())
                .setParameter("account", preparation.getAccountReferenceId())
                .setLockMode(LockModeType.PESSIMISTIC_WRITE).getSingleResult();
        CardV2 source = source(preparation.getInstitutionId(), preparation.getSourceCardId());
        entityManager.refresh(family(source), LockModeType.PESSIMISTIC_WRITE);
        for (CardV2 member : members(source)) {
            entityManager.refresh(member, LockModeType.PESSIMISTIC_WRITE);
        }
        entityManager.createQuery(
                        "select r from BinRangeV2 r where r.institutionId=:tenant and r.binRangeId=:range")
                .setParameter("tenant", preparation.getInstitutionId())
                .setParameter("range", preparation.getBinRangeId())
                .setLockMode(LockModeType.PESSIMISTIC_WRITE).getSingleResult();
        entityManager.refresh(preparation, LockModeType.PESSIMISTIC_WRITE);
        CardLifecycleOperationV2 operation = entityManager.find(
                CardLifecycleOperationV2.class, operationId, LockModeType.PESSIMISTIC_WRITE);
        entityManager.refresh(operation);
        return preparation;
    }

    private ProductReplacementPolicyV2 policy(CardV2 source, ReplacementReason reason) {
        return entityManager.createQuery(
                        "select p from ProductReplacementPolicyV2 p where p.institutionId=:tenant and p.productVersionId=:version and p.replacementReason=:reason and p.policyStatus=:status",
                        ProductReplacementPolicyV2.class)
                .setParameter("tenant", source.getInstitutionId())
                .setParameter("version", source.getProductVersionId())
                .setParameter("reason", reason)
                .setParameter("status", ProductPolicyStatus.ACTIVE)
                .getResultStream().findFirst()
                .orElseThrow(() -> new BusinessRuleViolationException(
                        "An ACTIVE security replacement policy is required"));
    }

    private static void validatePolicy(ProductReplacementPolicyV2 policy,
            ReplacementRequestChannel requestChannel) {
        require(policy.isReplacementAllowed()
                        && policy.getPermittedChannels().contains(requestChannel),
                "Security replacement is not allowed through " + requestChannel);
        require(!policy.isApprovalRequired(),
                "Approval-required security replacement needs the approval lifecycle implementation");
        require(policy.getPanStrategy() == ReplacementPanStrategy.GENERATE_NEW_PAN,
                "Permanent-security replacement policy must generate a new PAN");
        require(policy.getOldCardUsagePolicy() == OldCardUsagePolicy.BLOCK_IMMEDIATELY,
                "Permanent-security replacement must block the source immediately");
        require(policy.getExpiryStrategy() == ReplacementExpiryStrategy.GENERATE_NEW_EXPIRY,
                "New-PAN security replacement must generate a new expiry");
        require(policy.getAccountLinkStrategy()
                        == ReplacementAccountLinkStrategy.PRESERVE_ACCOUNT_LINK,
                "Security replacement must preserve the account link");
        require(policy.getControlStrategy() == ReplacementControlStrategy.APPLY_PRODUCT_DEFAULTS,
                "New-PAN security replacement must apply product-default controls");
    }

    private AccountRestrictionSnapshotV2 restrictions(UUID institutionId, UUID accountId) {
        AccountRestrictionSnapshotV2 snapshot = entityManager.find(
                AccountRestrictionSnapshotV2.class,
                new AccountRestrictionSnapshotV2.Key(institutionId, accountId),
                LockModeType.PESSIMISTIC_READ);
        require(snapshot != null && snapshot.getValidUntil().isAfter(now())
                        && !snapshot.getObservedAt().isAfter(now()),
                "Fresh authoritative account restrictions required");
        return snapshot;
    }

    private CardV2 source(UUID institutionId, UUID cardId) {
        return entityManager.createQuery(
                        "select c from CardV2 c where c.institutionId=:tenant and c.cardId=:card",
                        CardV2.class)
                .setParameter("tenant", institutionId).setParameter("card", cardId)
                .getResultStream().findFirst()
                .orElseThrow(() -> new ResourceNotFoundException("Card", cardId.toString()));
    }

    private PanAllocationV2 family(CardV2 source) {
        return entityManager.find(PanAllocationV2.class, source.getPanAllocationId());
    }

    private List<CardV2> members(CardV2 source) {
        return entityManager.createQuery(
                        "select c from CardV2 c where c.institutionId=:tenant and c.panAllocationId=:family order by c.cardId",
                        CardV2.class)
                .setParameter("tenant", source.getInstitutionId())
                .setParameter("family", source.getPanAllocationId()).getResultList();
    }

    private void cancelPendingProduction(CardV2 card) {
        if (card.getPhysicalFulfilmentStatus() == PhysicalFulfilmentStatusV2.CANCELLED) {
            entityManager.createNativeQuery(
                            "update cms_v2.card_production_request set request_status='CANCELLED', updated_at=now() "
                                    + "where institution_id=:tenant and card_id=:card and request_status in ('PENDING','ELIGIBLE','RESERVED_FOR_BATCH')")
                    .setParameter("tenant", card.getInstitutionId())
                    .setParameter("card", card.getCardId()).executeUpdate();
        }
    }

    private static void retirePhysical(CardV2 card, ReplacementReason reason,
            OffsetDateTime occurredAt) {
        if (card.getCardForm() == CardFormV2.VIRTUAL || card.getRetiredAt() != null) return;
        card.setRetiredAt(occurredAt);
        card.setPhysicalRetiredReason(reason == ReplacementReason.LOST
                || reason == ReplacementReason.STOLEN
                ? PhysicalRetiredReasonV2.LOST_STOLEN : PhysicalRetiredReasonV2.FRAUD);
        if (card.getPhysicalFulfilmentStatus() == PhysicalFulfilmentStatusV2.REQUESTED
                || card.getPhysicalFulfilmentStatus()
                        == PhysicalFulfilmentStatusV2.PERSONALIZATION_IN_PROGRESS
                || card.getPhysicalFulfilmentStatus() == PhysicalFulfilmentStatusV2.PERSONALIZED
                || card.getPhysicalFulfilmentStatus()
                        == PhysicalFulfilmentStatusV2.IN_EMBOSSING_FILE) {
            card.setPhysicalFulfilmentStatus(PhysicalFulfilmentStatusV2.CANCELLED);
        }
    }

    private String event(CardLifecycleOperationV2 operation, CardV2 card, UUID relatedCardId,
            String eventType, String transition, long sequence) {
        UUID eventId = id();
        OffsetDateTime occurredAt = now();
        Map<String, Object> payload = new LinkedHashMap<>();
        payload.put("eventId", eventId.toString());
        payload.put("operationId", operation.getOperationId().toString());
        payload.put("institutionId", operation.getInstitutionId().toString());
        payload.put("cardId", card.getCardId().toString());
        payload.put("relatedCardId", relatedCardId.toString());
        payload.put("eventSequence", sequence);
        payload.put("transitionId", transition);
        payload.put("schemaVersion", 1);
        payload.put("source", "CMS_V2");
        payload.put("cardStatus", card.getCardStatus().name());
        payload.put("actorType", operation.getActorType().name());
        entityManager.persist(CardEventV2.builder().cardEventId(eventId)
                .institutionId(operation.getInstitutionId()).cardId(card.getCardId())
                .relatedCardId(relatedCardId).operationId(operation.getOperationId())
                .eventType(eventType).eventSequence(sequence)
                .resultingStatus(card.getCardStatus().name())
                .reasonCode(card.getReplacementReason() == null ? null
                        : card.getReplacementReason().name())
                .source("CMS_V2").transitionId(transition)
                .actorReference(operation.getActorReference())
                .correlationId(operation.getOperationId()).occurredAt(occurredAt)
                .eventPayload(payload).build());
        PanAllocationV2 allocation = entityManager.find(
                PanAllocationV2.class, card.getPanAllocationId());
        entityManager.persist(OutboxEventV2.builder().outboxEventId(id())
                .institutionId(operation.getInstitutionId())
                .operationId(operation.getOperationId()).aggregateType("CARD")
                .aggregateId(card.getCardId().toString()).eventType(eventType)
                .eventPayload(payload).correlationId(operation.getOperationId())
                .idempotencyKey(eventId).familyRevision(allocation.getFamilyRevision())
                .occurredAt(occurredAt).build());
        return eventId.toString();
    }

    private static SecurityHoldReasonV2 securityReason(ReplacementReason reason) {
        return switch (reason) {
            case LOST -> SecurityHoldReasonV2.LOST;
            case STOLEN -> SecurityHoldReasonV2.STOLEN;
            case COMPROMISED -> SecurityHoldReasonV2.COMPROMISED;
            case FRAUD -> SecurityHoldReasonV2.FRAUD_SUSPECTED;
            default -> throw new BusinessRuleViolationException(
                    "Unsupported permanent-security replacement reason");
        };
    }

    private Map<String, Object> map(Object value) {
        return mapper.convertValue(value,
                new TypeReference<LinkedHashMap<String, Object>>() {});
    }

    private UUID id() {
        return (UUID) entityManager.createNativeQuery("select uuidv7()", UUID.class)
                .getSingleResult();
    }

    private static OffsetDateTime now() {
        return OffsetDateTime.now(ZoneOffset.UTC);
    }

    private static void require(boolean condition, String message) {
        if (!condition) throw new BusinessRuleViolationException(message);
    }
