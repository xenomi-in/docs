package com.xenomi.cms.core.v2.card.pin.entity;

import java.time.OffsetDateTime;
import java.util.UUID;

import com.xenomi.cms.core.v2.card.pin.model.PinOperationTypeV2;
import com.xenomi.cms.core.v2.card.pin.model.PinStatusV2;
import jakarta.persistence.Column;
import jakarta.persistence.Entity;
import jakarta.persistence.EnumType;
import jakarta.persistence.Enumerated;
import jakarta.persistence.Id;
import jakarta.persistence.PrePersist;
import jakarta.persistence.Table;
import lombok.AllArgsConstructor;
import lombok.Builder;
import lombok.Data;
import lombok.NoArgsConstructor;

@org.hibernate.annotations.Immutable
@Entity(name = "PinHistoryV2")
@Table(name = "pin_history", schema = "cms_v2")
@Data @Builder @NoArgsConstructor @AllArgsConstructor
public class PinHistoryV2 {
    @Id @Column(name = "pin_history_id", nullable = false, updatable = false)
    private UUID pinHistoryId;
    @Column(name = "institution_id", nullable = false, updatable = false)
    private UUID institutionId;
    @Column(name = "card_id", nullable = false, updatable = false)
    private UUID cardId;
    @Enumerated(EnumType.STRING)
    @Column(name = "operation_type", nullable = false, length = 30, updatable = false)
    private PinOperationTypeV2 operationType;
    @Enumerated(EnumType.STRING)
    @Column(name = "previous_status", length = 24, updatable = false)
    private PinStatusV2 previousStatus;
    @Enumerated(EnumType.STRING)
    @Column(name = "resulting_status", nullable = false, length = 24, updatable = false)
    private PinStatusV2 resultingStatus;
    @Column(name = "hsm_key_id", updatable = false)
    private UUID hsmKeyId;
    @Column(name = "occurred_at", nullable = false, updatable = false)
    private OffsetDateTime occurredAt;
    @Column(name = "actor_reference", length = 120, updatable = false)
    private String actorReference;
    @Column(name = "operation_reason", length = 200, updatable = false)
    private String operationReason;
    @Column(name = "correlation_id", nullable = false, updatable = false)
    private UUID correlationId;

    @Column(name = "pin_data_id", nullable = false, updatable = false)
    private UUID pinDataId;

    @PrePersist
    void beforeInsert() {
        if (occurredAt == null) occurredAt = OffsetDateTime.now(java.time.ZoneOffset.UTC);
    }
}
