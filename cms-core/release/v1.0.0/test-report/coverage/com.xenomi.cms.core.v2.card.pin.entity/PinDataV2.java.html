package com.xenomi.cms.core.v2.card.pin.entity;

import java.time.OffsetDateTime;
import java.util.UUID;

import com.xenomi.cms.core.v2.bin.model.PinMethod;
import com.xenomi.cms.core.v2.card.pin.model.PinStatusV2;
import com.xenomi.cms.core.v2.product.model.PinDeliveryMethod;
import com.xenomi.cms.core.v2.product.model.PinGenerationMethod;
import jakarta.persistence.Column;
import jakarta.persistence.Entity;
import jakarta.persistence.EnumType;
import jakarta.persistence.Enumerated;
import jakarta.persistence.Id;
import jakarta.persistence.PrePersist;
import jakarta.persistence.PreUpdate;
import jakarta.persistence.Table;
import lombok.AllArgsConstructor;
import lombok.Builder;
import lombok.Data;
import lombok.NoArgsConstructor;

@Entity(name = "PinDataV2")
@Table(name = "pin_data", schema = "cms_v2")
@Data @Builder @NoArgsConstructor @AllArgsConstructor
public class PinDataV2 {
    @Id @Column(name = "pin_data_id", nullable = false, updatable = false)
    private UUID pinDataId;
    @Column(name = "institution_id", nullable = false, updatable = false)
    private UUID institutionId;
    @Enumerated(EnumType.STRING)
    @Column(name = "pin_status", nullable = false, length = 24)
    private PinStatusV2 pinStatus;
    @Column(name = "failed_attempt_count", nullable = false)
    private int failedAttemptCount;
    @Column(name = "pin_offset", length = 32)
    @lombok.ToString.Exclude
    private String pinOffset;
    @Enumerated(EnumType.STRING)
    @Column(name = "pin_method", length = 20)
    private PinMethod pinMethod;
    @Column(name = "pin_length")
    private Integer pinLength;
    @Column(name = "pvki")
    private Integer pvki;
    @Column(name = "maximum_attempts")
    private Integer maximumAttempts;
    @Enumerated(EnumType.STRING)
    @Column(name = "generation_method", length = 32)
    private PinGenerationMethod generationMethod;
    @Enumerated(EnumType.STRING)
    @Column(name = "delivery_method", length = 32)
    private PinDeliveryMethod deliveryMethod;
    @Column(name = "last_changed_at")
    private OffsetDateTime lastChangedAt;
    @Column(name = "hsm_key_id")
    private UUID hsmKeyId;
    @Column(name = "pin_block_key_id")
    private UUID pinBlockKeyId;
    @Column(name = "updated_at", nullable = false)
    private OffsetDateTime updatedAt;
    @Column(name = "updated_by", nullable = false, length = 100)
    private String updatedBy;

    @Column(name = "pan_allocation_id", nullable = false, updatable = false)
    private UUID panAllocationId;
    @Builder.Default
    @Column(name = "pin_authority", nullable = false)
    private String pinAuthority = "CMS";
    /** Immutable initial Switch profile, captured from the issuance policy; contains no PIN material. */
    @org.hibernate.annotations.JdbcTypeCode(org.hibernate.type.SqlTypes.JSON)
    @Column(name = "switch_pin_profile", columnDefinition = "jsonb")
    private java.util.Map<String, Object> switchPinProfile;
    @jakarta.persistence.Version
    @lombok.Setter(lombok.AccessLevel.NONE)
    @Column(name = "version", nullable = false)
    private Long version;

    @PrePersist @PreUpdate
    void touch() {
        if (version == null) version = 1L;
        updatedAt = OffsetDateTime.now(java.time.ZoneOffset.UTC);
    }
}
