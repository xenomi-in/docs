package com.xenomi.cms.core.v2.product.entity;

import java.time.OffsetDateTime;
import java.util.LinkedHashMap;
import java.util.Map;
import java.util.UUID;

import org.hibernate.annotations.JdbcTypeCode;
import org.hibernate.type.SqlTypes;
import jakarta.persistence.Column;
import jakarta.persistence.Entity;
import jakarta.persistence.Id;
import jakarta.persistence.PrePersist;
import jakarta.persistence.PreUpdate;
import jakarta.persistence.Table;
import lombok.AllArgsConstructor;
import lombok.Builder;
import lombok.Data;
import lombok.NoArgsConstructor;

@Entity(name = "ProductDefaultControlsV2")
@Table(name = "product_default_controls", schema = "cms_v2")
@Data
@Builder
@NoArgsConstructor
@AllArgsConstructor
public class ProductDefaultControlsV2 {
    @Id
    @Column(name = "product_version_id", nullable = false, updatable = false)
    private UUID productVersionId;
    @Column(name = "institution_id", nullable = false, updatable = false)
    private UUID institutionId;
    private boolean domesticAtmEnabled;
    @Builder.Default private boolean domesticAtmAllowed = true;
    private boolean domesticPosEnabled;
    @Builder.Default private boolean domesticPosAllowed = true;
    private boolean domesticEcommerceEnabled;
    @Builder.Default private boolean domesticEcommerceAllowed = true;
    private boolean domesticMotoEnabled;
    @Builder.Default private boolean domesticMotoAllowed = true;
    private boolean domesticContactlessEnabled;
    @Builder.Default private boolean domesticContactlessAllowed = true;
    private boolean domesticCashWithdrawalEnabled;
    @Builder.Default private boolean internationalEnabled = true;
    private boolean internationalAtmEnabled;
    @Builder.Default private boolean internationalAtmAllowed = true;
    private boolean internationalPosEnabled;
    @Builder.Default private boolean internationalPosAllowed = true;
    private boolean internationalEcommerceEnabled;
    @Builder.Default private boolean internationalEcommerceAllowed = true;
    private boolean internationalMotoEnabled;
    @Builder.Default private boolean internationalMotoAllowed = true;
    private boolean internationalContactlessEnabled;
    @Builder.Default private boolean internationalContactlessAllowed = true;
    private boolean internationalCashWithdrawalEnabled;
    @JdbcTypeCode(SqlTypes.JSON)
    @Column(name = "domestic_merchant_restrictions", nullable = false, columnDefinition = "jsonb")
    @Builder.Default
    private Map<String, Object> domesticMerchantRestrictions = new LinkedHashMap<>();
    @JdbcTypeCode(SqlTypes.JSON)
    @Column(name = "international_merchant_restrictions", nullable = false, columnDefinition = "jsonb")
    @Builder.Default
    private Map<String, Object> internationalMerchantRestrictions = new LinkedHashMap<>();
    @Column(name = "updated_at", nullable = false)
    private OffsetDateTime updatedAt;
    @Column(name = "updated_by", nullable = false, length = 100)
    private String updatedBy;

    @PrePersist
    @PreUpdate
    void touch() {
        updatedAt = OffsetDateTime.now(java.time.ZoneOffset.UTC);
    }
}
