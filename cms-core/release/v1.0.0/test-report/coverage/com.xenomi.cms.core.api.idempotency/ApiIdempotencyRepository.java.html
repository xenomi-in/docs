package com.xenomi.cms.core.api.idempotency;

import org.springframework.beans.factory.annotation.Value;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.stereotype.Repository;

import java.sql.ResultSet;
import java.sql.SQLException;
import java.time.OffsetDateTime;
import java.util.List;
import java.util.concurrent.TimeUnit;
import java.util.concurrent.atomic.AtomicLong;

@Repository
public class ApiIdempotencyRepository {

    private final JdbcTemplate jdbcTemplate;
    private final long retentionHours;
    private final AtomicLong nextCleanupAt = new AtomicLong();

    public ApiIdempotencyRepository(
            JdbcTemplate jdbcTemplate,
            @Value("${cms.idempotency.retention-hours:24}") long retentionHours) {
        this.jdbcTemplate = jdbcTemplate;
        this.retentionHours = retentionHours;
    }

    public ClaimResult claim(RequestIdentity identity, String requestHash) {
        cleanupExpiredPeriodically();
        int inserted = jdbcTemplate.update("""
                INSERT INTO cms_v2.api_idempotency_record
                    (tenant_scope, actor_id, http_method, request_path,
                     idempotency_key_hash, request_hash, expires_at)
                VALUES (?, ?, ?, ?, ?, ?, ?)
                ON CONFLICT (tenant_scope, actor_id, http_method, request_path, idempotency_key_hash)
                DO NOTHING
                """,
                identity.tenantScope(), identity.actorId(), identity.httpMethod(),
                identity.requestPath(), identity.idempotencyKeyHash(), requestHash,
                OffsetDateTime.now().plusHours(retentionHours));
        if (inserted == 1) {
            return ClaimResult.started();
        }

        List<StoredResponse> records = jdbcTemplate.query("""
                SELECT request_hash, processing_status, response_status,
                       response_content_type, response_location, response_body
                  FROM cms_v2.api_idempotency_record
                 WHERE tenant_scope = ? AND actor_id = ? AND http_method = ?
                   AND request_path = ? AND idempotency_key_hash = ?
                   AND expires_at > CURRENT_TIMESTAMP
                """,
                (rs, rowNum) -> storedResponse(rs),
                identity.tenantScope(), identity.actorId(), identity.httpMethod(),
                identity.requestPath(), identity.idempotencyKeyHash());
        if (records.isEmpty()) {
            delete(identity);
            return claim(identity, requestHash);
        }

        StoredResponse record = records.get(0);
        if (!record.requestHash().equals(requestHash)) {
            return ClaimResult.conflict();
        }
        if ("COMPLETED".equals(record.processingStatus())) {
            return ClaimResult.replay(record);
        }
        return ClaimResult.inProgress();
    }

    public void complete(RequestIdentity identity, int responseStatus,
                         String contentType, String location, byte[] responseBody) {
        jdbcTemplate.update("""
                UPDATE cms_v2.api_idempotency_record
                   SET processing_status = 'COMPLETED', response_status = ?,
                       response_content_type = ?, response_location = ?, response_body = ?,
                       completed_at = CURRENT_TIMESTAMP
                 WHERE tenant_scope = ? AND actor_id = ? AND http_method = ?
                   AND request_path = ? AND idempotency_key_hash = ?
                """,
                responseStatus, contentType, location, responseBody,
                identity.tenantScope(), identity.actorId(), identity.httpMethod(),
                identity.requestPath(), identity.idempotencyKeyHash());
    }

    public void delete(RequestIdentity identity) {
        jdbcTemplate.update("""
                DELETE FROM cms_v2.api_idempotency_record
                 WHERE tenant_scope = ? AND actor_id = ? AND http_method = ?
                   AND request_path = ? AND idempotency_key_hash = ?
                """,
                identity.tenantScope(), identity.actorId(), identity.httpMethod(),
                identity.requestPath(), identity.idempotencyKeyHash());
    }

    public void deleteExpired() {
        jdbcTemplate.update("DELETE FROM cms_v2.api_idempotency_record WHERE expires_at <= CURRENT_TIMESTAMP");
    }

    private void cleanupExpiredPeriodically() {
        long now = System.currentTimeMillis();
        long next = nextCleanupAt.get();
        if (now >= next && nextCleanupAt.compareAndSet(next, now + TimeUnit.HOURS.toMillis(1))) {
            deleteExpired();
        }
    }

    private static StoredResponse storedResponse(ResultSet rs) throws SQLException {
        Integer status = (Integer) rs.getObject("response_status");
        return new StoredResponse(
                rs.getString("request_hash"),
                rs.getString("processing_status"),
                status,
                rs.getString("response_content_type"),
                rs.getString("response_location"),
                rs.getBytes("response_body"));
    }

    public record RequestIdentity(
            String tenantScope,
            String actorId,
            String httpMethod,
            String requestPath,
            String idempotencyKeyHash) {
    }

    public record StoredResponse(
            String requestHash,
            String processingStatus,
            Integer responseStatus,
            String contentType,
            String location,
            byte[] body) {
    }

    public record ClaimResult(ClaimState state, StoredResponse response) {
        static ClaimResult started() { return new ClaimResult(ClaimState.STARTED, null); }
        static ClaimResult conflict() { return new ClaimResult(ClaimState.CONFLICT, null); }
        static ClaimResult inProgress() { return new ClaimResult(ClaimState.IN_PROGRESS, null); }
        static ClaimResult replay(StoredResponse response) { return new ClaimResult(ClaimState.REPLAY, response); }
    }

    public enum ClaimState {
        STARTED,
        REPLAY,
        CONFLICT,
        IN_PROGRESS
    }
}
