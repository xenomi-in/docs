package com.xenomi.cms.core.v2.card.lifecycle.controller;

import java.util.UUID;
import com.xenomi.cms.core.api.ApiResponseEnvelope;
import com.xenomi.cms.core.api.ApiResponseHelper;
import com.xenomi.cms.core.v2.card.lifecycle.dto.CardHoldV2Request;
import com.xenomi.cms.core.v2.card.lifecycle.dto.CardLifecycleActionV2Response;
import com.xenomi.cms.core.v2.card.lifecycle.service.CardHoldV2Service;
import io.swagger.v3.oas.annotations.tags.Tag;
import jakarta.validation.Valid;
import jakarta.validation.constraints.NotBlank;
import jakarta.validation.constraints.Size;
import org.springframework.http.ResponseEntity;
import org.springframework.validation.annotation.Validated;
import org.springframework.web.bind.annotation.*;

@Validated
@RestController
@RequestMapping("/cms/api/v2/institutions/{institutionId}/cards/{cardId}/holds")
@Tag(name = "V2 - Cards", description = "Tenant-scoped card issuance and lifecycle")
public class CardHoldV2Controller {
    private final CardHoldV2Service service;
    public CardHoldV2Controller(CardHoldV2Service service) { this.service = service; }
    @PostMapping("/place") public ResponseEntity<ApiResponseEnvelope<CardLifecycleActionV2Response>> place(@PathVariable UUID institutionId,@PathVariable UUID cardId,@Valid @RequestBody CardHoldV2Request request,@RequestHeader("Idempotency-Key") @NotBlank @Size(max=128) String key,@RequestHeader(value="X-Request-Id",required=false) String requestId,@RequestHeader("X-Actor-Id") @NotBlank String actor) { return ApiResponseHelper.ok(service.place(institutionId,cardId,request,key,requestId,actor)); }
    @PostMapping("/escalate") public ResponseEntity<ApiResponseEnvelope<CardLifecycleActionV2Response>> escalate(@PathVariable UUID institutionId,@PathVariable UUID cardId,@Valid @RequestBody CardHoldV2Request request,@RequestHeader("Idempotency-Key") @NotBlank @Size(max=128) String key,@RequestHeader(value="X-Request-Id",required=false) String requestId,@RequestHeader("X-Actor-Id") @NotBlank String actor) { return ApiResponseHelper.ok(service.escalate(institutionId,cardId,request,key,requestId,actor)); }
    @PostMapping("/release") public ResponseEntity<ApiResponseEnvelope<CardLifecycleActionV2Response>> release(@PathVariable UUID institutionId,@PathVariable UUID cardId,@Valid @RequestBody CardHoldV2Request request,@RequestHeader("Idempotency-Key") @NotBlank @Size(max=128) String key,@RequestHeader(value="X-Request-Id",required=false) String requestId,@RequestHeader("X-Actor-Id") @NotBlank String actor) { return ApiResponseHelper.ok(service.release(institutionId,cardId,request,key,requestId,actor)); }
}
