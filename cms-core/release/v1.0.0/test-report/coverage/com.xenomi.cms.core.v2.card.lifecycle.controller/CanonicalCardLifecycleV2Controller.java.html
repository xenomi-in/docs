package com.xenomi.cms.core.v2.card.lifecycle.controller;

import com.fasterxml.jackson.databind.ObjectMapper;
import com.xenomi.cms.core.api.ApiResponseEnvelope;
import com.xenomi.cms.core.api.ApiResponseHelper;
import com.xenomi.cms.core.v2.card.lifecycle.command.*;
import com.xenomi.cms.core.v2.card.lifecycle.dto.CanonicalCardLifecycleRequestV2;
import io.swagger.v3.oas.annotations.tags.Tag;
import jakarta.validation.Valid;
import jakarta.validation.constraints.NotBlank;
import jakarta.validation.constraints.Size;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.*;
import java.util.Map;
import java.util.UUID;

@RestController
@RequestMapping("/cms/api/v2/institutions/{institutionId}/cards/{cardId}/commands")
@Tag(name = "V2 - Cards", description = "Tenant-scoped card issuance and lifecycle")
public class CanonicalCardLifecycleV2Controller {
    private final LifecycleCommandCoordinatorV2 coordinator;
    private final ObjectMapper mapper;
    public CanonicalCardLifecycleV2Controller(LifecycleCommandCoordinatorV2 coordinator, ObjectMapper mapper) {
        this.coordinator = coordinator; this.mapper = mapper;
    }
    @PostMapping("/make-use-ready")
    public ResponseEntity<ApiResponseEnvelope<LifecycleCommandReceiptV2>> makeUseReady(
            @PathVariable UUID institutionId, @PathVariable UUID cardId,
            @Valid @RequestBody CanonicalCardLifecycleRequestV2 request,
            @RequestHeader("Idempotency-Key") @NotBlank @Size(max = 128) String key) {
        if (request.evidenceId() == null || request.trigger() == null)
            throw new IllegalArgumentException("evidenceId and trigger are required");
        var payload = mapper.createObjectNode().put("evidenceId", request.evidenceId().toString()).put("trigger", request.trigger().name());
        return ApiResponseHelper.ok(coordinator.execute("MAKE_USE_READY", envelope(institutionId, cardId, request, key, payload)));
    }

    @PostMapping("/activate-physical")
    public ResponseEntity<ApiResponseEnvelope<LifecycleCommandReceiptV2>> activatePhysical(
            @PathVariable UUID institutionId, @PathVariable UUID cardId,
            @Valid @RequestBody CanonicalCardLifecycleRequestV2 request,
            @RequestHeader("Idempotency-Key") @NotBlank @Size(max = 128) String key) {
        if (request.evidenceId() == null || request.trigger() == null)
            throw new IllegalArgumentException("evidenceId and trigger are required");
        var payload = mapper.createObjectNode().put("evidenceId", request.evidenceId().toString()).put("trigger", request.trigger().name());
        return ApiResponseHelper.ok(coordinator.execute("ACTIVATE_PHYSICAL", envelope(institutionId, cardId, request, key, payload)));
    }
    @PostMapping("/close")
    public ResponseEntity<ApiResponseEnvelope<LifecycleCommandReceiptV2>> close(
            @PathVariable UUID institutionId, @PathVariable UUID cardId,
            @Valid @RequestBody CanonicalCardLifecycleRequestV2 request,
            @RequestHeader("Idempotency-Key") @NotBlank @Size(max = 128) String key) {
        if (request.closureReason() == null) throw new IllegalArgumentException("closureReason is required");
        var payload = mapper.createObjectNode().put("closureReason", request.closureReason().name());
        return ApiResponseHelper.ok(coordinator.execute("CLOSE_CARD", envelope(institutionId, cardId, request, key, payload)));
    }
    private static LifecycleCommandRequestV2 envelope(UUID institutionId, UUID cardId,
            CanonicalCardLifecycleRequestV2 r, String key, com.fasterxml.jackson.databind.JsonNode payload) {
        return new LifecycleCommandRequestV2(institutionId, r.accountReferenceId(), cardId, r.expectedVersion(), Map.of(), key, null, payload);
    }
}
