package com.xenomi.cms.core.v2.card.lifecycle.controller;

import java.util.UUID;

import com.xenomi.cms.core.api.ApiResponseEnvelope;
import com.xenomi.cms.core.api.ApiResponseHelper;
import com.xenomi.cms.core.v2.card.lifecycle.dto.CardLifecycleActionV2Request;
import com.xenomi.cms.core.v2.card.lifecycle.dto.CardLifecycleActionV2Response;
import com.xenomi.cms.core.v2.card.lifecycle.service.CardLifecycleV2Service;
import io.swagger.v3.oas.annotations.Operation;
import io.swagger.v3.oas.annotations.tags.Tag;
import jakarta.validation.Valid;
import jakarta.validation.constraints.NotBlank;
import jakarta.validation.constraints.Size;
import org.springframework.http.ResponseEntity;
import org.springframework.validation.annotation.Validated;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RequestHeader;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RestController;

@Validated
@RestController
@RequestMapping("/cms/api/v2/institutions/{institutionId}/cards/{cardId}/actions")
@Tag(name = "V2 - Cards", description = "Tenant-scoped card issuance and lifecycle")
public class CardLifecycleV2Controller {
    private final CardLifecycleV2Service service;

    public CardLifecycleV2Controller(CardLifecycleV2Service service) {
        this.service = service;
    }

    @PostMapping("/activate")
    @Operation(summary = "Activate a newly issued card",
            description = "Transitions READY to ACTIVE and synchronously updates the authorization switch before committing the CMS status.")
    public ResponseEntity<ApiResponseEnvelope<CardLifecycleActionV2Response>> activate(
            @PathVariable UUID institutionId, @PathVariable UUID cardId,
            @Valid @RequestBody CardLifecycleActionV2Request request,
            @RequestHeader("Idempotency-Key") @NotBlank @Size(max = 128) String idempotencyKey,
            @RequestHeader(value = "X-Request-Id", required = false) @Size(max = 128) String requestId,
            @RequestHeader("X-Actor-Id") @NotBlank @Size(max = 100) String actorId) {
        return ApiResponseHelper.ok(service.activate(
                institutionId, cardId, request, idempotencyKey, requestId, actorId));
    }

    @PostMapping("/suspend")
    @Operation(summary = "Suspend an active card",
            description = "Deprecated compatibility wrapper. Delegates to the canonical hold placement command; use /holds/place.")
    public ResponseEntity<ApiResponseEnvelope<CardLifecycleActionV2Response>> suspend(
            @PathVariable UUID institutionId, @PathVariable UUID cardId,
            @Valid @RequestBody CardLifecycleActionV2Request request,
            @RequestHeader("Idempotency-Key") @NotBlank @Size(max = 128) String idempotencyKey,
            @RequestHeader(value = "X-Request-Id", required = false) @Size(max = 128) String requestId,
            @RequestHeader("X-Actor-Id") @NotBlank @Size(max = 100) String actorId) {
        return ApiResponseHelper.ok(service.suspend(
                institutionId, cardId, request, idempotencyKey, requestId, actorId));
    }

    @PostMapping("/unblock")
    @Operation(summary = "Unblock or resume a card",
            description = "Deprecated compatibility wrapper. Delegates to the canonical hold release command; use /holds/release.")
    public ResponseEntity<ApiResponseEnvelope<CardLifecycleActionV2Response>> unblock(
            @PathVariable UUID institutionId, @PathVariable UUID cardId,
            @Valid @RequestBody CardLifecycleActionV2Request request,
            @RequestHeader("Idempotency-Key") @NotBlank @Size(max = 128) String idempotencyKey,
            @RequestHeader(value = "X-Request-Id", required = false) @Size(max = 128) String requestId,
            @RequestHeader("X-Actor-Id") @NotBlank @Size(max = 100) String actorId) {
        return ApiResponseHelper.ok(service.unblock(
                institutionId, cardId, request, idempotencyKey, requestId, actorId));
    }

    @PostMapping("/close")
    @Operation(summary = "Permanently close a card",
            description = "Transitions any non-closed card to the terminal CLOSED state and synchronously updates the authorization switch. Closed cards cannot be reactivated.")
    public ResponseEntity<ApiResponseEnvelope<CardLifecycleActionV2Response>> close(
            @PathVariable UUID institutionId, @PathVariable UUID cardId,
            @Valid @RequestBody CardLifecycleActionV2Request request,
            @RequestHeader("Idempotency-Key") @NotBlank @Size(max = 128) String idempotencyKey,
            @RequestHeader(value = "X-Request-Id", required = false) @Size(max = 128) String requestId,
            @RequestHeader("X-Actor-Id") @NotBlank @Size(max = 100) String actorId) {
        return ApiResponseHelper.ok(service.close(
                institutionId, cardId, request, idempotencyKey, requestId, actorId));
    }
}
