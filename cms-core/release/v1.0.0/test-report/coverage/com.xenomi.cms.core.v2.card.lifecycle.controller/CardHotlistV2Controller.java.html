package com.xenomi.cms.core.v2.card.lifecycle.controller;

import java.util.UUID;

import com.xenomi.cms.core.api.ApiResponseEnvelope;
import com.xenomi.cms.core.api.ApiResponseHelper;
import com.xenomi.cms.core.config.CmsPermissions;
import com.xenomi.cms.core.v2.card.lifecycle.dto.HotlistCardV2Request;
import com.xenomi.cms.core.v2.card.lifecycle.dto.HotlistCardV2Response;
import com.xenomi.cms.core.v2.card.lifecycle.model.CardHotlistCallerType;
import com.xenomi.cms.core.v2.card.lifecycle.service.CardHotlistV2Service;
import io.swagger.v3.oas.annotations.Operation;
import io.swagger.v3.oas.annotations.tags.Tag;
import jakarta.validation.Valid;
import jakarta.validation.constraints.NotBlank;
import jakarta.validation.constraints.Size;
import org.springframework.security.access.AccessDeniedException;
import org.springframework.security.core.Authentication;
import org.springframework.http.ResponseEntity;
import org.springframework.validation.annotation.Validated;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RequestHeader;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RestController;

@Validated
@RestController
@RequestMapping("/cms/api/v2/institutions/{institutionId}/cards/{cardId}/hotlist")
@Tag(name = "V2 - Cards", description = "Tenant-scoped card issuance and lifecycle")
public class CardHotlistV2Controller {
    private static final String ACTOR_HEADER = "X-Actor-Id";
    private final CardHotlistV2Service service;

    public CardHotlistV2Controller(CardHotlistV2Service service) {
        this.service = service;
    }

    @PostMapping
    @Operation(
            operationId = "hotlistCardV2",
            summary = "Immediately hot-list a card",
            description = "Available to separately authorized customer-channel and operations clients. The caller type is derived from the signed access-token permission, not accepted from the payload. The authorization switch must acknowledge BLOCKED before CMS commits the lifecycle status and immutable audit event.")
    public ResponseEntity<ApiResponseEnvelope<HotlistCardV2Response>> hotlist(
            @PathVariable UUID institutionId,
            @PathVariable UUID cardId,
            @Valid @RequestBody HotlistCardV2Request request,
            @RequestHeader("Idempotency-Key") @NotBlank @Size(max = 128) String idempotencyKey,
            @RequestHeader(value = "X-Request-Id", required = false) @Size(max = 128) String requestId,
            @RequestHeader(ACTOR_HEADER) @NotBlank @Size(max = 100) String actorId,
            Authentication authentication) {
        return ApiResponseHelper.ok(service.hotlist(institutionId, cardId, request,
                callerType(authentication), idempotencyKey, requestId, actorId));
    }

    private static CardHotlistCallerType callerType(Authentication authentication) {
        if (has(authentication, CmsPermissions.CARD_HOTLIST_OPERATIONS)) {
            return CardHotlistCallerType.OPERATIONS;
        }
        if (has(authentication, CmsPermissions.CARD_HOTLIST_CUSTOMER)) {
            return CardHotlistCallerType.CUSTOMER;
        }
        throw new AccessDeniedException("Card hot-listing permission is required");
    }

    private static boolean has(Authentication authentication, String authority) {
        return authentication != null && authentication.getAuthorities().stream()
                .anyMatch(value -> authority.equals(value.getAuthority()));
    }
}
