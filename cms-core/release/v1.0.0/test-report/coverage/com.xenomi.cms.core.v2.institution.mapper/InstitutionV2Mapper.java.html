package com.xenomi.cms.core.v2.institution.mapper;

import java.util.Locale;
import java.util.UUID;

import com.xenomi.cms.core.v2.institution.dto.CreateInstitutionV2Request;
import com.xenomi.cms.core.v2.institution.dto.InstitutionAddressV2;
import com.xenomi.cms.core.v2.institution.dto.InstitutionV2Response;
import com.xenomi.cms.core.v2.institution.entity.InstitutionV2;
import com.xenomi.cms.core.v2.institution.model.InstitutionStatus;

import org.springframework.stereotype.Component;

@Component
public class InstitutionV2Mapper {

    public InstitutionV2 toEntity(CreateInstitutionV2Request request, UUID institutionId, String actorId) {
        InstitutionAddressV2 address = request.getRegisteredAddress();
        return InstitutionV2.builder()
                .institutionId(institutionId)
                .institutionName(request.getInstitutionName().trim())
                .institutionCode(upper(request.getInstitutionCode()))
                .institutionShortName(trimToNull(request.getInstitutionShortName()))
                .institutionType(request.getInstitutionType())
                .visaMemberFlag(Boolean.TRUE.equals(request.getVisaMemberFlag()))
                .visaMemberId(trimToNull(request.getVisaMemberId()))
                .rupayMemberFlag(Boolean.TRUE.equals(request.getRupayMemberFlag()))
                .npciMemberId(trimToNull(request.getNpciMemberId()))
                .registeredAddressLine1(address.getAddressLine1().trim())
                .registeredAddressLine2(trimToNull(address.getAddressLine2()))
                .city(address.getCity().trim())
                .stateCode(upper(address.getStateCode()))
                .countryCode(upperOrDefault(address.getCountryCode(), "IND"))
                .postalCode(address.getPostalCode().trim())
                .contactPhonePrimary(request.getContactPhonePrimary().trim())
                .contactPhoneSecondary(trimToNull(request.getContactPhoneSecondary()))
                .contactEmailPrimary(request.getContactEmailPrimary().trim().toLowerCase(Locale.ROOT))
                .contactEmailSecondary(lowerOrNull(request.getContactEmailSecondary()))
                .websiteUrl(trimToNull(request.getWebsiteUrl()))
                .timezone(valueOrDefault(request.getTimezone(), "Asia/Kolkata"))
                .reportDeliverySftpPath(trimToNull(request.getReportDeliverySftpPath()))
                .reportDeliveryEmail(lowerOrNull(request.getReportDeliveryEmail()))
               // .prepaidCardIssuingEnabled(Boolean.TRUE.equals(request.getPrepaidCardIssuingEnabled()))
                .creditCardIssuingEnabled(Boolean.TRUE.equals(request.getCreditCardIssuingEnabled()))
                .defaultCardValidityMonths(request.getDefaultCardValidityMonths() == null
                        ? 60 : request.getDefaultCardValidityMonths())
                .baseCurrency(upper(request.getBaseCurrency()))
                .currencyExponent(request.getCurrencyExponent() == null
                        ? (short) 2 : request.getCurrencyExponent())
                .businessDate(request.getBusinessDate())
                .status(request.getStatus() == null ? InstitutionStatus.ACTIVE : request.getStatus())
                .createdBy(actorId)
                .updatedBy(actorId)
                .build();
    }

    public InstitutionV2Response toResponse(InstitutionV2 institution) {
        return InstitutionV2Response.builder()
                .institutionId(institution.getInstitutionId())
                .institutionName(institution.getInstitutionName())
                .institutionCode(institution.getInstitutionCode())
                .institutionShortName(institution.getInstitutionShortName())
                .institutionType(institution.getInstitutionType())
                .visaMemberFlag(institution.isVisaMemberFlag())
                .visaMemberId(institution.getVisaMemberId())
                .rupayMemberFlag(institution.isRupayMemberFlag())
                .npciMemberId(institution.getNpciMemberId())
                .registeredAddress(InstitutionAddressV2.builder()
                        .addressLine1(institution.getRegisteredAddressLine1())
                        .addressLine2(institution.getRegisteredAddressLine2())
                        .city(institution.getCity())
                        .stateCode(institution.getStateCode())
                        .countryCode(institution.getCountryCode())
                        .postalCode(institution.getPostalCode())
                        .build())
                .contactPhonePrimary(institution.getContactPhonePrimary())
                .contactPhoneSecondary(institution.getContactPhoneSecondary())
                .contactEmailPrimary(institution.getContactEmailPrimary())
                .contactEmailSecondary(institution.getContactEmailSecondary())
                .websiteUrl(institution.getWebsiteUrl())
                .timezone(institution.getTimezone())
                .reportDeliverySftpPath(institution.getReportDeliverySftpPath())
                .reportDeliveryEmail(institution.getReportDeliveryEmail())
                .prepaidCardIssuingEnabled(institution.isPrepaidCardIssuingEnabled())
                .creditCardIssuingEnabled(institution.isCreditCardIssuingEnabled())
                .defaultCardValidityMonths(institution.getDefaultCardValidityMonths())
                .baseCurrency(institution.getBaseCurrency())
                .currencyExponent(institution.getCurrencyExponent())
                .businessDate(institution.getBusinessDate())
                .status(institution.getStatus())
                .createdAt(institution.getCreatedAt())
                .updatedAt(institution.getUpdatedAt())
                .createdBy(institution.getCreatedBy())
                .updatedBy(institution.getUpdatedBy())
                .build();
    }

    public static String upper(String value) {
        return value.trim().toUpperCase(Locale.ROOT);
    }

    public static String trimToNull(String value) {
        if (value == null || value.isBlank()) {
            return null;
        }
        return value.trim();
    }

    private static String upperOrDefault(String value, String defaultValue) {
        return value == null || value.isBlank() ? defaultValue : upper(value);
    }

    private static String valueOrDefault(String value, String defaultValue) {
        return value == null || value.isBlank() ? defaultValue : value.trim();
    }

    private static String lowerOrNull(String value) {
        String normalized = trimToNull(value);
        return normalized == null ? null : normalized.toLowerCase(Locale.ROOT);
    }
}
