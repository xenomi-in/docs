package com.xenomi.cms.core.v2.card.lifecycle.entity;

import jakarta.persistence.*;
import lombok.*;
import java.util.UUID;
import java.util.Map;
import java.time.OffsetDateTime;
import com.xenomi.cms.core.v2.card.model.*;

@Entity(name = "CardExternalOperationV2")
@Table(name = "card_external_operation", schema = "cms_v2")
@Data @Builder @NoArgsConstructor @AllArgsConstructor
public class CardExternalOperationV2 {
    @Id
    @Column(name = "external_operation_id", nullable = false, updatable = false)
    private UUID externalOperationId;
    @Column(name = "institution_id", nullable = false)
    private UUID institutionId;
    @Column(name = "operation_id", nullable = false)
    private UUID operationId;
    @Column(name = "effect_key", length = 200, nullable = false)
    private String effectKey;
    @Column(name = "provider", length = 100, nullable = false)
    private String provider;
    @Column(name = "operation_kind", length = 60, nullable = false)
    private String operationKind;
    @Builder.Default
    @Enumerated(EnumType.STRING)
    @Column(name = "operation_status", length = 24, nullable = false)
    private LifecycleOperationStatusV2 operationStatus = LifecycleOperationStatusV2.PENDING;
    @Column(name = "provider_reference", length = 300)
    private String providerReference;
    @Column(name = "evidence_reference", length = 500)
    private String evidenceReference;
    @org.hibernate.annotations.JdbcTypeCode(org.hibernate.type.SqlTypes.JSON)
    @Builder.Default
    @lombok.ToString.Exclude
    @Column(name = "intent_payload", columnDefinition = "jsonb", nullable = false)
    private Map<String, Object> intentPayload = new java.util.LinkedHashMap<>();
    @Builder.Default
    @Column(name = "attempt_count", nullable = false)
    private Integer attemptCount = 0;
    @Version
    @lombok.Setter(lombok.AccessLevel.NONE)
    @Column(name = "version", nullable = false)
    private Long version;
    @Column(name = "dispatched_at")
    private OffsetDateTime dispatchedAt;
    @Column(name = "resolved_at")
    private OffsetDateTime resolvedAt;
    @Column(name = "next_attempt_at")
    private OffsetDateTime nextAttemptAt;
    @PrePersist
    void beforeInsert() {
        if (version == null) version = 1L;
    }
}
