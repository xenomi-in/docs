package com.xenomi.cms.core.v2.card.lifecycle.entity;

import jakarta.persistence.*;
import lombok.*;
import java.util.UUID;
import java.util.Map;
import java.time.OffsetDateTime;
import com.xenomi.cms.core.v2.card.model.*;

@Entity(name = "LifecycleOutboxEventV2")
@Table(name = "outbox_event", schema = "cms_v2")
@Data @Builder @NoArgsConstructor @AllArgsConstructor
public class OutboxEventV2 {
    @Id
    @Column(name = "outbox_event_id", nullable = false, updatable = false)
    private UUID outboxEventId;
    @Column(name = "institution_id", nullable = false)
    private UUID institutionId;
    @Column(name = "aggregate_type", length = 60, nullable = false)
    private String aggregateType;
    @Column(name = "aggregate_id", length = 120, nullable = false)
    private String aggregateId;
    @Column(name = "event_type", length = 100, nullable = false)
    private String eventType;
    @Builder.Default
    @Column(name = "event_version", nullable = false)
    private Integer eventVersion = 1;
    @org.hibernate.annotations.JdbcTypeCode(org.hibernate.type.SqlTypes.JSON)
    @lombok.ToString.Exclude
    @Column(name = "event_payload", columnDefinition = "jsonb", nullable = false)
    private Map<String, Object> eventPayload;
    @Column(name = "correlation_id", nullable = false)
    private UUID correlationId;
    @Column(name = "idempotency_key", nullable = false)
    private UUID idempotencyKey;
    @Builder.Default
    @Enumerated(EnumType.STRING)
    @Column(name = "publication_status", length = 24, nullable = false)
    private OutboxPublicationStatusV2 publicationStatus = OutboxPublicationStatusV2.PENDING;
    @Column(name = "occurred_at", nullable = false)
    private OffsetDateTime occurredAt;
    @Column(name = "published_at")
    private OffsetDateTime publishedAt;
    @Builder.Default
    @Column(name = "retry_count", nullable = false)
    private Integer retryCount = 0;
    @Column(name = "next_retry_at")
    private OffsetDateTime nextRetryAt;
    @Column(name = "last_error", length = 1000)
    private String lastError;
    @Column(name = "operation_id", nullable = false)
    private UUID operationId;
    @Column(name = "family_revision")
    private Long familyRevision;
    @PrePersist
    void beforeInsert() {
        if (occurredAt == null) occurredAt = OffsetDateTime.now(java.time.ZoneOffset.UTC);
    }

}
