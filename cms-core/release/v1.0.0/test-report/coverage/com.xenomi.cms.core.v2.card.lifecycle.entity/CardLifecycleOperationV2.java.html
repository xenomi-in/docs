package com.xenomi.cms.core.v2.card.lifecycle.entity;

import jakarta.persistence.*;
import lombok.*;
import java.util.UUID;
import java.util.Map;
import java.time.OffsetDateTime;
import com.xenomi.cms.core.v2.card.model.*;

@Entity(name = "CardLifecycleOperationV2")
@Table(name = "card_lifecycle_operation", schema = "cms_v2")
@Data @Builder @NoArgsConstructor @AllArgsConstructor
public class CardLifecycleOperationV2 {
    @Id
    @Column(name = "operation_id", nullable = false, updatable = false)
    private UUID operationId;
    @Column(name = "institution_id", nullable = false)
    private UUID institutionId;
    @Column(name = "operation_kind", length = 60, nullable = false)
    private String operationKind;
    @lombok.ToString.Exclude
    @org.hibernate.annotations.JdbcTypeCode(org.hibernate.type.SqlTypes.CHAR)
    @Column(name = "idempotency_key_hash", length = 64, columnDefinition = "char(64)", nullable = false)
    private String idempotencyKeyHash;
    @lombok.ToString.Exclude
    @org.hibernate.annotations.JdbcTypeCode(org.hibernate.type.SqlTypes.CHAR)
    @Column(name = "request_fingerprint", length = 64, columnDefinition = "char(64)", nullable = false)
    private String requestFingerprint;
    @Column(name = "target_card_id")
    private UUID targetCardId;
    @Column(name = "expected_version")
    private Long expectedVersion;
    @Column(name = "actor_reference", length = 120, nullable = false)
    private String actorReference;
    @Enumerated(EnumType.STRING)
    @Column(name = "actor_type", length = 24, nullable = false)
    private LifecycleActorTypeV2 actorType;
    @Builder.Default
    @Enumerated(EnumType.STRING)
    @Column(name = "operation_status", length = 24, nullable = false)
    private LifecycleOperationStatusV2 operationStatus = LifecycleOperationStatusV2.PENDING;
    @org.hibernate.annotations.JdbcTypeCode(org.hibernate.type.SqlTypes.JSON)
    @Builder.Default
    @lombok.ToString.Exclude
    @Column(name = "result_payload", columnDefinition = "jsonb", nullable = false)
    private Map<String, Object> resultPayload = new java.util.LinkedHashMap<>();
    @Column(name = "created_at", nullable = false)
    private OffsetDateTime createdAt;
    @Column(name = "completed_at")
    private OffsetDateTime completedAt;
    @PrePersist
    void beforeInsert() {
        if (createdAt == null) createdAt = OffsetDateTime.now(java.time.ZoneOffset.UTC);
    }

}
