package com.xenomi.cms.core.v2.card.lifecycle.entity;

import jakarta.persistence.*;
import lombok.*;
import java.util.UUID;
import java.util.Map;
import java.time.OffsetDateTime;
import com.xenomi.cms.core.v2.card.model.*;

@Entity(name = "CourierShipmentV2")
@Table(name = "courier_shipment", schema = "cms_v2")
@Data @Builder @NoArgsConstructor @AllArgsConstructor
public class CourierShipmentV2 {
    @Id
    @Column(name = "shipment_id", nullable = false, updatable = false)
    private UUID shipmentId;
    @Column(name = "institution_id", nullable = false)
    private UUID institutionId;
    @Column(name = "card_id", nullable = false)
    private UUID cardId;
    @Column(name = "production_request_id", nullable = false)
    private UUID productionRequestId;
    @Column(name = "courier_provider_id", nullable = false)
    private UUID courierProviderId;
    @Column(name = "courier_waybill_id", nullable = false)
    private UUID courierWaybillId;
    @Column(name = "shipping_address_reference_id", nullable = false)
    private UUID shippingAddressReferenceId;
    @Column(name = "shipment_status", length = 32, nullable = false)
    private String shipmentStatus;
    @Column(name = "dispatched_at")
    private OffsetDateTime dispatchedAt;
    @Column(name = "delivered_at")
    private OffsetDateTime deliveredAt;
    @Column(name = "created_at", nullable = false)
    private OffsetDateTime createdAt;
    @Builder.Default
    @Column(name = "shipment_attempt", nullable = false)
    private Integer shipmentAttempt = 1;
    @PrePersist
    void beforeInsert() {
        if (createdAt == null) createdAt = OffsetDateTime.now(java.time.ZoneOffset.UTC);
        if (shipmentStatus == null) shipmentStatus = "CREATED";
    }

}
