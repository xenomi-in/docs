package com.xenomi.cms.core.v2.card.lifecycle.entity;

import jakarta.persistence.*;
import lombok.*;
import java.util.UUID;
import java.util.Map;
import java.time.OffsetDateTime;
import com.xenomi.cms.core.v2.card.model.*;

@Entity(name = "PhysicalSequenceReservationV2")
@Table(name = "physical_sequence_reservation", schema = "cms_v2")
@Data @Builder @NoArgsConstructor @AllArgsConstructor
public class PhysicalSequenceReservationV2 {
    @Id
    @Column(name = "reservation_id", nullable = false, updatable = false)
    private UUID reservationId;
    @Column(name = "institution_id", nullable = false)
    private UUID institutionId;
    @Column(name = "pan_allocation_id", nullable = false)
    private UUID panAllocationId;
    @Column(name = "card_id", nullable = false)
    private UUID cardId;
    @Column(name = "operation_id", nullable = false)
    private UUID operationId;
    @org.hibernate.annotations.JdbcTypeCode(org.hibernate.type.SqlTypes.CHAR)
    @Column(name = "reserved_sequence", length = 3, columnDefinition = "char(3)")
    private String reservedSequence;
    @Column(name = "bureau_reference", length = 300)
    private String bureauReference;
    @Column(name = "reserved_at", nullable = false)
    private OffsetDateTime reservedAt;
    @Column(name = "verified_at")
    private OffsetDateTime verifiedAt;
    @PrePersist
    void beforeInsert() {
        if (reservedAt == null) reservedAt = OffsetDateTime.now(java.time.ZoneOffset.UTC);
    }

}
