package com.xenomi.cms.core.v2.card.lifecycle.entity;

import jakarta.persistence.*;
import lombok.*;
import java.util.UUID;
import java.util.Map;
import java.time.OffsetDateTime;
import com.xenomi.cms.core.v2.card.model.*;

@org.hibernate.annotations.Immutable
@Entity(name = "ActivationPolicyVersionV2")
@Table(name = "activation_policy_version", schema = "cms_v2")
@Data @Builder @NoArgsConstructor @AllArgsConstructor
public class ActivationPolicyVersionV2 {
    @Id
    @Column(name = "activation_policy_version_id", nullable = false, updatable = false)
    private UUID activationPolicyVersionId;
    @Column(name = "institution_id", nullable = false, updatable = false)
    private UUID institutionId;
    @Column(name = "policy_reference", length = 200, nullable = false, updatable = false)
    private String policyReference;
    @Column(name = "version", nullable = false, updatable = false)
    private Integer version;
    @Column(name = "calendar_reference", length = 200, nullable = false, updatable = false)
    private String calendarReference;
    @Column(name = "expiry_timezone", length = 60, nullable = false, updatable = false)
    private String expiryTimezone;
    @Column(name = "policy_document_reference", length = 500, nullable = false, updatable = false)
    private String policyDocumentReference;
    @Column(name = "effective_from", nullable = false, updatable = false)
    private OffsetDateTime effectiveFrom;
    @PrePersist
    void beforeInsert() {
        if (expiryTimezone == null) expiryTimezone = "Asia/Kolkata";
    }

}
