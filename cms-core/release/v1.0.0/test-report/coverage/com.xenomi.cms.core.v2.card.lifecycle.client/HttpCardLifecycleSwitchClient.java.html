package com.xenomi.cms.core.v2.card.lifecycle.client;

import java.util.UUID;

import com.xenomi.cms.core.v2.card.lifecycle.model.CardHotlistCallerType;
import com.xenomi.cms.core.v2.card.lifecycle.model.CardHotlistReason;
import com.xenomi.cms.core.v2.card.model.CardStatusV2;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.http.MediaType;
import org.springframework.http.client.SimpleClientHttpRequestFactory;
import org.springframework.stereotype.Component;
import org.springframework.web.client.RestClient;
import org.springframework.web.client.RestClientException;

@Component
public class HttpCardLifecycleSwitchClient implements CardLifecycleSwitchClient {
    @Value("${cms.switch.sync.enabled:false}")
    private boolean durableSyncEnabled;
    private static final String BLOCKED = "BLOCKED";
    private final RestClient restClient;
    private final String baseUrl;

    public HttpCardLifecycleSwitchClient(RestClient.Builder builder,
            @Value("${cms.switch.base-url:}") String baseUrl,
            @Value("${cms.switch.timeout-ms:3000}") int timeoutMs) {
        SimpleClientHttpRequestFactory requestFactory = new SimpleClientHttpRequestFactory();
        requestFactory.setConnectTimeout(timeoutMs);
        requestFactory.setReadTimeout(timeoutMs);
        this.restClient = builder.requestFactory(requestFactory).build();
        this.baseUrl = baseUrl == null ? "" : baseUrl.trim();
    }

    @Override
    public void block(UUID institutionId, UUID cardId, long lifecycleVersion,
            CardHotlistReason reason, CardHotlistCallerType callerType,
            String idempotencyKey, String requestId) {
        // The deferred database trigger captures the committed final lifecycle facts.
        if (durableSyncEnabled) return;
        if (baseUrl.isBlank()) {
            throw new CardLifecycleSwitchException("Authorization switch is not configured");
        }
        try {
            SwitchStatusResponse response = restClient.put()
                    .uri(baseUrl + "/internal/v1/institutions/{institutionId}/cards/{cardId}/status",
                            institutionId, cardId)
                    .contentType(MediaType.APPLICATION_JSON)
                    .header("Idempotency-Key", idempotencyKey)
                    .header("X-Request-Id", requestId)
                    .body(new SwitchStatusRequest(BLOCKED, lifecycleVersion,
                            reason.name(), callerType.name()))
                    .retrieve()
                    .body(SwitchStatusResponse.class);
            if (response == null || !response.applied()
                    || !BLOCKED.equals(response.appliedStatus())
                    || response.appliedVersion() != lifecycleVersion) {
                throw new CardLifecycleSwitchException(
                        "Authorization switch did not confirm the hot-listing version");
            }
        } catch (RestClientException exception) {
            throw new CardLifecycleSwitchException(
                    "Authorization switch hot-listing failed", exception);
        }
    }

    @Override
    public void changeStatus(UUID institutionId, UUID cardId, CardStatusV2 targetStatus,
            long lifecycleVersion, String reason, String idempotencyKey, String requestId) {
        if (durableSyncEnabled) return;
        if (baseUrl.isBlank()) {
            throw new CardLifecycleSwitchException("Authorization switch is not configured");
        }
        try {
            SwitchStatusResponse response = restClient.put()
                    .uri(baseUrl + "/internal/v1/institutions/{institutionId}/cards/{cardId}/status",
                            institutionId, cardId)
                    .contentType(MediaType.APPLICATION_JSON)
                    .header("Idempotency-Key", idempotencyKey)
                    .header("X-Request-Id", requestId)
                    .body(new SwitchStatusRequest(targetStatus.name(), lifecycleVersion,
                            reason, "OPERATIONS"))
                    .retrieve()
                    .body(SwitchStatusResponse.class);
            if (response == null || !response.applied()
                    || !targetStatus.name().equals(response.appliedStatus())
                    || response.appliedVersion() != lifecycleVersion) {
                throw new CardLifecycleSwitchException(
                        "Authorization switch did not confirm the card lifecycle version");
            }
        } catch (RestClientException exception) {
            throw new CardLifecycleSwitchException(
                    "Authorization switch card lifecycle update failed", exception);
        }
    }

    private record SwitchStatusRequest(String status, long lifecycleVersion,
            String reason, String callerType) {
    }

    private record SwitchStatusResponse(boolean applied, String appliedStatus,
            long appliedVersion) {
    }
}
