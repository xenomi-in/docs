package com.xenomi.cms.core.v2.event.catalog;

import java.time.OffsetDateTime;
import java.util.Map;
import java.util.Objects;

import com.xenomi.cms.core.v2.event.mapper.PayloadValues;
import com.xenomi.cms.core.v2.event.model.EventTypes;
import com.xenomi.cms.core.v2.event.model.IdempotencyKeys;
import com.xenomi.cms.core.v2.event.model.NotificationPriority;
import com.xenomi.cms.core.v2.event.model.OutboxEventCommand;
import com.xenomi.cms.core.v2.event.writer.OutboxEventWriter;
import com.xenomi.cms.core.v2.product.dto.ProductVersionV2Response;
import com.xenomi.cms.core.v2.product.model.ExpiryCalculationRule;
import com.xenomi.cms.core.v2.product.model.ProductVersionStatus;
import com.xenomi.cms.core.v2.product.model.ValidityStartRule;
import org.springframework.stereotype.Component;

@Component
public class ProductVersionEventSink {

    private final OutboxEventWriter writer;

    public ProductVersionEventSink(OutboxEventWriter writer) {
        this.writer = writer;
    }

    public void onCreated(ProductVersionV2Response version, String correlationId) {
        if (version == null) {
            return;
        }
        writer.enqueue(command(version, EventTypes.PRODUCT_VERSION_CREATED, correlationId,
                IdempotencyKeys.of(version.getInstitutionId(), EventTypes.PRODUCT_VERSION_CREATED,
                        version.getProductVersionId()),
                payload(version, null, version.getVersionStatus(), null)));
    }

    /**
     * @param before snapshot taken before the update ran; {@code null} when unknown
     */
    public void onUpdated(ProductVersionV2Response version, VersionSnapshot before, String correlationId) {
        if (version == null) {
            return;
        }
        ProductVersionStatus previousStatus = before == null ? null : before.status();
        boolean statusChanged = previousStatus != null && previousStatus != version.getVersionStatus();
        boolean changed = before == null || statusChanged || before.differsFrom(version);
        if (!changed) {
            return;
        }
        Object stamp = PayloadValues.instant(version.getUpdatedAt());
        writer.enqueue(command(version, EventTypes.PRODUCT_VERSION_UPDATED, correlationId,
                IdempotencyKeys.of(version.getInstitutionId(), EventTypes.PRODUCT_VERSION_UPDATED,
                        version.getProductVersionId(), stamp),
                payload(version, previousStatus, version.getVersionStatus(), "PRODUCT_VERSION_UPDATED")));
        if (!statusChanged) {
            return;
        }
        writer.enqueue(command(version, EventTypes.PRODUCT_VERSION_STATUS_CHANGED, correlationId,
                IdempotencyKeys.of(version.getInstitutionId(), EventTypes.PRODUCT_VERSION_STATUS_CHANGED,
                        version.getProductVersionId(), previousStatus, version.getVersionStatus(), stamp),
                payload(version, previousStatus, version.getVersionStatus(), "STATUS_CHANGED")));
    }

    private static OutboxEventCommand command(ProductVersionV2Response version, String eventType,
            String correlationId, java.util.UUID idempotencyKey, Map<String, Object> payload) {
        return OutboxEventCommand.builder()
                .institutionId(version.getInstitutionId())
                .eventType(eventType)
                .aggregateType(EventTypes.AGGREGATE_PRODUCT_VERSION)
                .aggregateId(version.getProductVersionId())
                .customerId(null)
                .correlationId(correlationId)
                .priority(NotificationPriority.MEDIUM)
                .idempotencyKey(idempotencyKey)
                .payload(payload)
                .build();
    }

    private static Map<String, Object> payload(ProductVersionV2Response version,
            ProductVersionStatus oldStatus, ProductVersionStatus newStatus, String reason) {
        return PayloadValues.entries(
                "productVersionId", PayloadValues.uuid(version.getProductVersionId()),
                "productId", PayloadValues.uuid(version.getProductId()),
                "institutionId", PayloadValues.uuid(version.getInstitutionId()),
                "versionNumber", version.getVersionNumber(),
                "oldStatus", PayloadValues.name(oldStatus),
                "newStatus", PayloadValues.name(newStatus),
                "reason", reason,
                "effectiveFrom", PayloadValues.instant(version.getEffectiveFrom()),
                "effectiveTo", PayloadValues.instant(version.getEffectiveTo()),
                "activatedAt", PayloadValues.instant(version.getActivatedAt()),
                "changedBy", version.getUpdatedBy(),
                "changedAt", PayloadValues.instant(version.getUpdatedAt()));
    }

    /** Mutable version fields captured before an update, used to suppress no-op events. */
    public record VersionSnapshot(
            ProductVersionStatus status,
            int validityMonths,
            ValidityStartRule validityStartRule,
            ExpiryCalculationRule expiryCalculationRule,
            Integer renewalNoticeDays,
            Integer renewalProductionDays,
            boolean automaticRenewalEnabled,
            String defaultServiceCode,
            String icvvServiceCode,
            OffsetDateTime effectiveFrom,
            OffsetDateTime effectiveTo) {

        boolean differsFrom(ProductVersionV2Response after) {
            return status != after.getVersionStatus()
                    || validityMonths != after.getValidityMonths()
                    || validityStartRule != after.getValidityStartRule()
                    || expiryCalculationRule != after.getExpiryCalculationRule()
                    || !Objects.equals(renewalNoticeDays, after.getRenewalNoticeDays())
                    || !Objects.equals(renewalProductionDays, after.getRenewalProductionDays())
                    || automaticRenewalEnabled != after.isAutomaticRenewalEnabled()
                    || !Objects.equals(defaultServiceCode, after.getDefaultServiceCode())
                    || !Objects.equals(icvvServiceCode, after.getIcvvServiceCode())
                    || !Objects.equals(effectiveFrom, after.getEffectiveFrom())
                    || !Objects.equals(effectiveTo, after.getEffectiveTo());
        }
    }
}
