package com.xenomi.cms.core.v2.event.catalog;

import java.util.List;
import java.util.Map;

import com.xenomi.cms.core.v2.cardcontrol.dto.CardControlDimensionV2Response;
import com.xenomi.cms.core.v2.cardcontrol.dto.CardControlsV2Response;
import com.xenomi.cms.core.v2.event.mapper.PayloadValues;
import com.xenomi.cms.core.v2.event.model.EventTypes;
import com.xenomi.cms.core.v2.event.model.IdempotencyKeys;
import com.xenomi.cms.core.v2.event.model.NotificationPriority;
import com.xenomi.cms.core.v2.event.model.OutboxEventCommand;
import com.xenomi.cms.core.v2.event.writer.OutboxEventWriter;
import org.springframework.stereotype.Component;

/**
 * Card-control events, built solely from the response the service already returns.
 * One V2 update can change enablement and limits together, so a single
 * {@code CARD_CONTROLS_UPDATED} covers both.
 */
@Component
public class CardControlEventSink {

    private final OutboxEventWriter writer;

    public CardControlEventSink(OutboxEventWriter writer) {
        this.writer = writer;
    }

    public void onUpdated(CardControlsV2Response controls, String correlationId) {
        if (controls == null || controls.getCardId() == null || controls.getInstitutionId() == null) {
            return;
        }
        writer.enqueue(OutboxEventCommand.builder()
                .institutionId(controls.getInstitutionId())
                .eventType(EventTypes.CARD_CONTROLS_UPDATED)
                .aggregateType(EventTypes.AGGREGATE_CARD)
                .aggregateId(controls.getCardId())
                .correlationId(correlationId)
                .priority(NotificationPriority.MEDIUM)
                .idempotencyKey(IdempotencyKeys.of(
                        controls.getInstitutionId(), EventTypes.CARD_CONTROLS_UPDATED, key(controls)))
                .payload(PayloadValues.entries(
                        "cardControlEventId", PayloadValues.uuid(controls.getCardControlEventId()),
                        "cardControlProfileId", PayloadValues.uuid(controls.getCardControlProfileId()),
                        "institutionId", PayloadValues.uuid(controls.getInstitutionId()),
                        "cardId", PayloadValues.uuid(controls.getCardId()),
                        "controlVersion", controls.getControlVersion(),
                        "profileStatus", PayloadValues.name(controls.getProfileStatus()),
                        "switchAppliedVersion", controls.getSwitchAppliedVersion(),
                        "reasonCode", controls.getReasonCode(),
                        "updatedBy", controls.getUpdatedBy(),
                        "updatedAt", PayloadValues.instant(controls.getUpdatedAt()),
                        "switchAppliedAt", PayloadValues.instant(controls.getSwitchAppliedAt()),
                        "accountingAssignedLimitMinor", controls.getAccountingAssignedLimitMinor(),
                        "accountingAvailableAmountMinor", controls.getAccountingAvailableAmountMinor(),
                        "accountingCurrencyCode", controls.getAccountingCurrencyCode(),
                        "accountingLimitRetrievedAt",
                                PayloadValues.instant(controls.getAccountingLimitRetrievedAt()),
                        "controls", dimensions(controls.getControls())))
                .build());
    }

    /**
     * Prefer the immutable audit-event id. Fall back to card + version so a response that
     * omits the event id still produces a stable, per-apply key.
     */
    private static Object key(CardControlsV2Response controls) {
        return controls.getCardControlEventId() != null
                ? controls.getCardControlEventId()
                : controls.getCardId() + "|" + controls.getControlVersion();
    }

    private static List<Map<String, Object>> dimensions(List<CardControlDimensionV2Response> controls) {
        if (controls == null) {
            return List.of();
        }
        return controls.stream()
                .map(dimension -> PayloadValues.entries(
                        "geography", PayloadValues.name(dimension.getGeography()),
                        "channel", PayloadValues.name(dimension.getChannel()),
                        "enabled", dimension.isEnabled(),
                        "perTransactionLimitMinor", dimension.getPerTransactionLimitMinor(),
                        "contactlessEnabled", dimension.getContactlessEnabled(),
                        "contactlessLimitMinor", dimension.getContactlessLimitMinor(),
                        "currencyCode", dimension.getCurrencyCode(),
                        "effectiveAt", PayloadValues.instant(dimension.getEffectiveAt())))
                .toList();
    }
}
