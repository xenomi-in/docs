package com.xenomi.cms.core.v2.event.catalog;

import com.xenomi.cms.core.v2.bin.dto.BinConfigurationV2Response;
import com.xenomi.cms.core.v2.bin.dto.BinRangeV2Response;
import com.xenomi.cms.core.v2.event.mapper.PayloadValues;
import com.xenomi.cms.core.v2.event.model.EventTypes;
import com.xenomi.cms.core.v2.event.model.IdempotencyKeys;
import com.xenomi.cms.core.v2.event.model.NotificationPriority;
import com.xenomi.cms.core.v2.event.model.OutboxEventCommand;
import com.xenomi.cms.core.v2.event.writer.OutboxEventWriter;
import org.springframework.stereotype.Component;

@Component
public class BinEventSink {

    private final OutboxEventWriter writer;

    public BinEventSink(OutboxEventWriter writer) {
        this.writer = writer;
    }

    public void onBinCreated(BinConfigurationV2Response bin, String correlationId) {
        if (bin == null) {
            return;
        }
        enqueueBin(bin, EventTypes.BIN_CREATED, correlationId,
                IdempotencyKeys.of(bin.getInstitutionId(), EventTypes.BIN_CREATED, bin.getBinId()));
    }

    public void onBinUpdated(BinConfigurationV2Response bin, String correlationId) {
        if (bin == null) {
            return;
        }
        // The BIN PUT contract bumps the optimistic-lock version on every applied update.
        enqueueBin(bin, EventTypes.BIN_UPDATED, correlationId,
                IdempotencyKeys.of(bin.getInstitutionId(), EventTypes.BIN_UPDATED, bin.getBinId(), bin.getVersion()));
    }

    public void onRangeCreated(BinRangeV2Response range, String correlationId) {
        if (range == null) {
            return;
        }
        enqueueRange(range, EventTypes.BIN_RANGE_CREATED, correlationId,
                IdempotencyKeys.of(range.getInstitutionId(), EventTypes.BIN_RANGE_CREATED, range.getBinRangeId()));
    }

    public void onRangeUpdated(BinRangeV2Response range, String correlationId) {
        if (range == null) {
            return;
        }
        enqueueRange(range, EventTypes.BIN_RANGE_UPDATED, correlationId,
                IdempotencyKeys.of(range.getInstitutionId(), EventTypes.BIN_RANGE_UPDATED,
                        range.getBinRangeId(), PayloadValues.instant(range.getUpdatedAt())));
    }

    private void enqueueBin(BinConfigurationV2Response bin, String eventType, String correlationId,
            java.util.UUID idempotencyKey) {
        writer.enqueue(OutboxEventCommand.builder()
                .institutionId(bin.getInstitutionId())
                .eventType(eventType)
                .aggregateType(EventTypes.AGGREGATE_BIN)
                .aggregateId(bin.getBinId())
                .correlationId(correlationId)
                .priority(NotificationPriority.MEDIUM)
                .idempotencyKey(idempotencyKey)
                .payload(PayloadValues.entries(
                        "binId", PayloadValues.uuid(bin.getBinId()),
                        "institutionId", PayloadValues.uuid(bin.getInstitutionId()),
                        "binNumber", bin.getBinNumber(),
                        "binLength", bin.getBinLength(),
                        "cardNetwork", PayloadValues.name(bin.getCardNetwork()),
                        "binType", PayloadValues.name(bin.getBinType()),
                        "binStatus", PayloadValues.name(bin.getBinStatus()),
                        "effectiveFrom", PayloadValues.date(bin.getEffectiveFrom()),
                        "effectiveTo", PayloadValues.date(bin.getEffectiveTo()),
                        "version", bin.getVersion(),
                        "updatedBy", bin.getUpdatedBy(),
                        "updatedAt", PayloadValues.instant(bin.getUpdatedAt())))
                .build());
    }

    private void enqueueRange(BinRangeV2Response range, String eventType, String correlationId,
            java.util.UUID idempotencyKey) {
        writer.enqueue(OutboxEventCommand.builder()
                .institutionId(range.getInstitutionId())
                .eventType(eventType)
                .aggregateType(EventTypes.AGGREGATE_BIN_RANGE)
                .aggregateId(range.getBinRangeId())
                .correlationId(correlationId)
                .priority(NotificationPriority.MEDIUM)
                .idempotencyKey(idempotencyKey)
                .payload(PayloadValues.entries(
                        "binRangeId", PayloadValues.uuid(range.getBinRangeId()),
                        "binId", PayloadValues.uuid(range.getBinId()),
                        "institutionId", PayloadValues.uuid(range.getInstitutionId()),
                        "rangeStart", range.getRangeStart(),
                        "rangeEnd", range.getRangeEnd(),
                        "allocationStrategy", PayloadValues.name(range.getAllocationStrategy()),
                        "usableCapacity", range.getUsableCapacity(),
                        "exhaustionThreshold", range.getExhaustionThreshold(),
                        "issuedPanCount", range.getIssuedPanCount(),
                        "remainingPanCapacity", range.getRemainingPanCapacity(),
                        "rangeStatus", PayloadValues.name(range.getRangeStatus()),
                        "effectiveFrom", PayloadValues.instant(range.getEffectiveFrom()),
                        "effectiveTo", PayloadValues.instant(range.getEffectiveTo()),
                        "updatedBy", range.getUpdatedBy(),
                        "updatedAt", PayloadValues.instant(range.getUpdatedAt())))
                .build());
    }
}
