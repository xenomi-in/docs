package com.xenomi.cms.core.v2.event.catalog;

import java.util.Map;
import java.util.UUID;

import com.xenomi.cms.core.v2.card.dto.CardV2Response;
import com.xenomi.cms.core.v2.card.lifecycle.dto.CardLifecycleActionV2Response;
import com.xenomi.cms.core.v2.card.lifecycle.dto.HotlistCardV2Response;
import com.xenomi.cms.core.v2.card.replacement.dto.CardReplacementV2Response;
import com.xenomi.cms.core.v2.event.mapper.PayloadValues;
import com.xenomi.cms.core.v2.event.model.EventTypes;
import com.xenomi.cms.core.v2.event.model.IdempotencyKeys;
import com.xenomi.cms.core.v2.event.model.NotificationPriority;
import com.xenomi.cms.core.v2.event.model.OutboxEventCommand;
import com.xenomi.cms.core.v2.event.writer.OutboxEventWriter;
import org.springframework.stereotype.Component;

/**
 * Card lifecycle events, built solely from the response the service already returns.
 * The sink performs no database reads: every payload field comes from the DTO.
 *
 * <p>Idempotency keys are the immutable {@code card_event} ids (or the card /
 * replacement-request id for creation), so a service that returns an existing result on
 * replay produces the same key and the duplicate is discarded at the database.
 */
@Component
public class CardEventSink {

    private final OutboxEventWriter writer;

    public CardEventSink(OutboxEventWriter writer) {
        this.writer = writer;
    }

    public void onCreated(CardV2Response card, String correlationId) {
        if (card == null) {
            return;
        }
        String customerId = PayloadValues.uuid(card.getCustomerReferenceId());
        writer.enqueue(cardCommand(card.getInstitutionId(), card.getCardId(), EventTypes.CARD_CREATED,
                customerId, correlationId, NotificationPriority.MEDIUM,
                IdempotencyKeys.of(card.getInstitutionId(), EventTypes.CARD_CREATED, card.getCardId()),
                PayloadValues.entries(
                        "cardId", PayloadValues.uuid(card.getCardId()),
                        "institutionId", PayloadValues.uuid(card.getInstitutionId()),
                        "customerId", customerId,
                        "accountReferenceId", PayloadValues.uuid(card.getAccountReferenceId()),
                        "productId", PayloadValues.uuid(card.getProductId()),
                        "productVersionId", PayloadValues.uuid(card.getProductVersionId()),
                        "binId", PayloadValues.uuid(card.getBinId()),
                        "binRangeId", PayloadValues.uuid(card.getBinRangeId()),
                        "cardNumberLastFour", card.getPanLastFour(),
                        "cardForm", PayloadValues.name(card.getCardForm()),
                        "issuanceMode", PayloadValues.name(card.getIssuanceMode()),
                        "status", PayloadValues.name(card.getCardStatus()),
                        "digitalAvailable", card.isDigitalAvailable(),
                        "lifecycleVersion", card.getLifecycleVersion(),
                        "issueDate", PayloadValues.date(card.getIssueDate()),
                        "expiryDate", PayloadValues.date(card.getExpiryDate()),
                        "productionRequestId", PayloadValues.uuid(card.getProductionRequestId()),
                        "createdAt", PayloadValues.instant(card.getCreatedAt()))));
    }

    public void onActivated(CardLifecycleActionV2Response action, String correlationId) {
        lifecycle(action, EventTypes.CARD_ACTIVATED, "activatedAt", correlationId);
    }

    public void onStatusChanged(CardLifecycleActionV2Response action, String correlationId) {
        lifecycle(action, EventTypes.CARD_STATUS_CHANGED, "changedAt", correlationId);
    }

    private void lifecycle(CardLifecycleActionV2Response action, String eventType,
            String timestampKey, String correlationId) {
        if (action == null) {
            return;
        }
        writer.enqueue(cardCommand(action.getInstitutionId(), action.getCardId(), eventType,
                null, correlationId, NotificationPriority.MEDIUM,
                IdempotencyKeys.of(action.getInstitutionId(), eventType, action.getEventId()),
                PayloadValues.entries(
                        "cardId", PayloadValues.uuid(action.getCardId()),
                        "institutionId", PayloadValues.uuid(action.getInstitutionId()),
                        "previousStatus", PayloadValues.name(action.getPreviousStatus()),
                        "newStatus", PayloadValues.name(action.getCardStatus()),
                        "lifecycleVersion", action.getLifecycleVersion(),
                        "reason", action.getReason(),
                        "cardEventId", PayloadValues.uuid(action.getEventId()),
                        timestampKey, PayloadValues.instant(action.getOccurredAt()))));
    }

    public void onHotlisted(HotlistCardV2Response hotlist, String correlationId) {
        if (hotlist == null) {
            return;
        }
        writer.enqueue(cardCommand(hotlist.getInstitutionId(), hotlist.getCardId(), EventTypes.CARD_HOTLISTED,
                null, correlationId, NotificationPriority.HIGH,
                IdempotencyKeys.of(hotlist.getInstitutionId(), EventTypes.CARD_HOTLISTED, hotlist.getEventId()),
                PayloadValues.entries(
                        "cardId", PayloadValues.uuid(hotlist.getCardId()),
                        "institutionId", PayloadValues.uuid(hotlist.getInstitutionId()),
                        "previousStatus", PayloadValues.name(hotlist.getPreviousStatus()),
                        "newStatus", PayloadValues.name(hotlist.getCardStatus()),
                        "lifecycleVersion", hotlist.getLifecycleVersion(),
                        "reason", PayloadValues.name(hotlist.getReason()),
                        "callerType", PayloadValues.name(hotlist.getCallerType()),
                        "cardEventId", PayloadValues.uuid(hotlist.getEventId()),
                        "hotlistedAt", PayloadValues.instant(hotlist.getHotlistedAt()))));
    }

    public void onReplacementRequested(CardReplacementV2Response replacement, String correlationId) {
        if (replacement == null) {
            return;
        }
        writer.enqueue(OutboxEventCommand.builder()
                .institutionId(replacement.getInstitutionId())
                .eventType(EventTypes.CARD_REPLACEMENT_REQUESTED)
                .aggregateType(EventTypes.AGGREGATE_REPLACEMENT)
                .aggregateId(replacement.getReplacementRequestId())
                .correlationId(correlationId)
                .priority(NotificationPriority.MEDIUM)
                .idempotencyKey(IdempotencyKeys.of(replacement.getInstitutionId(),
                        EventTypes.CARD_REPLACEMENT_REQUESTED, replacement.getReplacementRequestId()))
                .payload(PayloadValues.entries(
                        "replacementRequestId", PayloadValues.uuid(replacement.getReplacementRequestId()),
                        "institutionId", PayloadValues.uuid(replacement.getInstitutionId()),
                        "sourceCardId", PayloadValues.uuid(replacement.getSourceCardId()),
                        "replacementCardId", PayloadValues.uuid(replacement.getReplacementCardId()),
                        "requestType", PayloadValues.name(replacement.getReplacementReason()),
                        "policyPanStrategy", PayloadValues.name(replacement.getPolicyPanStrategy()),
                        "resolvedPanStrategy", PayloadValues.name(replacement.getResolvedPanStrategy()),
                        "requestStatus", PayloadValues.name(replacement.getRequestStatus()),
                        "sourceCardStatus", PayloadValues.name(replacement.getSourceCardStatus()),
                        "replacementCardStatus", PayloadValues.name(replacement.getReplacementCardStatus()),
                        "productionRequestId", PayloadValues.uuid(replacement.getProductionRequestId()),
                        "requestedAt", PayloadValues.instant(replacement.getRequestedAt()),
                        "completedAt", PayloadValues.instant(replacement.getCompletedAt())))
                .build());
    }

    private static OutboxEventCommand cardCommand(UUID institutionId, UUID cardId, String eventType,
            String customerId, String correlationId, NotificationPriority priority,
            UUID idempotencyKey, Map<String, Object> payload) {
        return OutboxEventCommand.builder()
                .institutionId(institutionId)
                .eventType(eventType)
                .aggregateType(EventTypes.AGGREGATE_CARD)
                .aggregateId(cardId)
                .customerId(customerId)
                .correlationId(correlationId)
                .priority(priority)
                .idempotencyKey(idempotencyKey)
                .payload(payload)
                .build();
    }
}
