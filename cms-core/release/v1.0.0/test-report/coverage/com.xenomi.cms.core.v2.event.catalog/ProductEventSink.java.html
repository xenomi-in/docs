package com.xenomi.cms.core.v2.event.catalog;

import java.util.Map;

import com.xenomi.cms.core.v2.event.mapper.PayloadValues;
import com.xenomi.cms.core.v2.event.model.EventTypes;
import com.xenomi.cms.core.v2.event.model.IdempotencyKeys;
import com.xenomi.cms.core.v2.event.model.NotificationPriority;
import com.xenomi.cms.core.v2.event.model.OutboxEventCommand;
import com.xenomi.cms.core.v2.event.writer.OutboxEventWriter;
import com.xenomi.cms.core.v2.product.dto.CardProductV2Response;
import com.xenomi.cms.core.v2.product.model.ProductStatus;
import org.springframework.stereotype.Component;

@Component
public class ProductEventSink {

    private final OutboxEventWriter writer;

    public ProductEventSink(OutboxEventWriter writer) {
        this.writer = writer;
    }

    public void onCreated(CardProductV2Response product, String correlationId) {
        if (product == null) {
            return;
        }
        writer.enqueue(command(product, EventTypes.PRODUCT_CREATED, correlationId,
                IdempotencyKeys.of(product.getInstitutionId(), EventTypes.PRODUCT_CREATED, product.getProductId()),
                payload(product, null, product.getProductStatus(), null)));
    }

    /**
     * @param before snapshot taken before the update ran; {@code null} when unknown
     */
    public void onUpdated(CardProductV2Response product, ProductSnapshot before, String correlationId) {
        if (product == null) {
            return;
        }
        ProductStatus previousStatus = before == null ? null : before.status();
        boolean statusChanged = previousStatus != null && previousStatus != product.getProductStatus();
        boolean changed = before == null || statusChanged || before.differsFrom(product);
        if (!changed) {
            return;
        }
        Object version = PayloadValues.instant(product.getUpdatedAt());
        writer.enqueue(command(product, EventTypes.PRODUCT_UPDATED, correlationId,
                IdempotencyKeys.of(product.getInstitutionId(), EventTypes.PRODUCT_UPDATED,
                        product.getProductId(), version),
                payload(product, previousStatus, product.getProductStatus(), "PRODUCT_UPDATED")));
        if (!statusChanged) {
            return;
        }
        writer.enqueue(command(product, EventTypes.PRODUCT_STATUS_CHANGED, correlationId,
                IdempotencyKeys.of(product.getInstitutionId(), EventTypes.PRODUCT_STATUS_CHANGED,
                        product.getProductId(), previousStatus, product.getProductStatus(), version),
                payload(product, previousStatus, product.getProductStatus(), "STATUS_CHANGED")));
        if (product.getProductStatus() == ProductStatus.RETIRED) {
            writer.enqueue(command(product, EventTypes.PRODUCT_DEACTIVATED, correlationId,
                    IdempotencyKeys.of(product.getInstitutionId(), EventTypes.PRODUCT_DEACTIVATED,
                            product.getProductId(), version),
                    payload(product, previousStatus, product.getProductStatus(), "RETIRED")));
        }
    }

    private static OutboxEventCommand command(CardProductV2Response product, String eventType,
            String correlationId, java.util.UUID idempotencyKey, Map<String, Object> payload) {
        return OutboxEventCommand.builder()
                .institutionId(product.getInstitutionId())
                .eventType(eventType)
                .aggregateType(EventTypes.AGGREGATE_PRODUCT)
                .aggregateId(product.getProductId())
                .customerId(null)
                .correlationId(correlationId)
                .priority(NotificationPriority.MEDIUM)
                .idempotencyKey(idempotencyKey)
                .payload(payload)
                .build();
    }

    private static Map<String, Object> payload(CardProductV2Response product,
            ProductStatus oldStatus, ProductStatus newStatus, String reason) {
        return PayloadValues.entries(
                "productId", PayloadValues.uuid(product.getProductId()),
                "institutionId", PayloadValues.uuid(product.getInstitutionId()),
                "productCode", product.getProductCode(),
                "productName", product.getProductName(),
                "cardNetwork", PayloadValues.name(product.getCardNetwork()),
                "fundingType", PayloadValues.name(product.getFundingType()),
                "cardVariant", PayloadValues.name(product.getCardVariant()),
                "oldStatus", PayloadValues.name(oldStatus),
                "newStatus", PayloadValues.name(newStatus),
                "reason", reason,
                "changedBy", product.getUpdatedBy(),
                "changedAt", PayloadValues.instant(product.getUpdatedAt()));
    }

    /** Mutable product fields captured before an update, used to suppress no-op events. */
    public record ProductSnapshot(String name, String description, ProductStatus status) {

        boolean differsFrom(CardProductV2Response after) {
            return !java.util.Objects.equals(name, after.getProductName())
                    || !java.util.Objects.equals(description, after.getDescription())
                    || status != after.getProductStatus();
        }
    }
}
