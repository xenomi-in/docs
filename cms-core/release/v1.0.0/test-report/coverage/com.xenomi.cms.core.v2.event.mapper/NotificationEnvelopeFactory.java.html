package com.xenomi.cms.core.v2.event.mapper;

import java.util.List;
import java.util.UUID;

import com.xenomi.cms.core.v2.event.model.EventTypes;
import com.xenomi.cms.core.v2.event.model.NotificationEnvelope;
import com.xenomi.cms.core.v2.event.model.OutboxEventCommand;
import org.springframework.stereotype.Component;

@Component
public class NotificationEnvelopeFactory {

    public NotificationEnvelope create(UUID eventId, OutboxEventCommand command, String correlationId) {
        return NotificationEnvelope.builder()
                .eventId(eventId.toString())
                .eventType(command.eventType())
                .tenantId(command.institutionId().toString())
                .aggregateType(command.aggregateType())
                .aggregateId(command.aggregateId())
                .customerId(command.customerId())
                .eventSource(EventTypes.SOURCE_CMS)
                .correlationId(correlationId)
                .channels(List.of())
                .priority(command.priority().name())
                .payload(command.payload())
                .build();
    }
}
