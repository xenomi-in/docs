package com.xenomi.cms.core.v2.event.mapper;

import java.time.Instant;
import java.time.LocalDate;
import java.time.OffsetDateTime;
import java.time.ZoneOffset;
import java.time.format.DateTimeFormatter;
import java.util.LinkedHashMap;
import java.util.Map;
import java.util.UUID;

/** Formats payload values per the Falcon contract: ISO-8601 UTC instants, ISO dates, UUID strings, enum names. */
public final class PayloadValues {

    private PayloadValues() {
    }

    /**
     * Builds an insertion-ordered payload from key/value pairs. Unlike {@link Map#of}, null
     * values are kept so the envelope serialises absent fields explicitly as {@code null}.
     */
    public static Map<String, Object> entries(Object... keyValues) {
        if (keyValues.length % 2 != 0) {
            throw new IllegalArgumentException("Payload entries must be key/value pairs");
        }
        Map<String, Object> payload = new LinkedHashMap<>();
        for (int i = 0; i < keyValues.length; i += 2) {
            payload.put((String) keyValues[i], keyValues[i + 1]);
        }
        return payload;
    }

    public static String instant(OffsetDateTime value) {
        if (value == null) {
            return null;
        }
        Instant utc = value.withOffsetSameInstant(ZoneOffset.UTC).toInstant();
        return DateTimeFormatter.ISO_INSTANT.format(utc);
    }

    public static String date(LocalDate value) {
        return value == null ? null : value.toString();
    }

    public static String uuid(UUID value) {
        return value == null ? null : value.toString();
    }

    public static String name(Enum<?> value) {
        return value == null ? null : value.name();
    }
}
