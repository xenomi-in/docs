/**
 * Replacement read model and compatibility boundary.
 *
 * <p>Supported same-PAN renewal and permanent-security replacement commands are owned by
 * {@code SamePanRenewalApiV2} and {@code SecurityReplacementApiV2}. This service retains the
 * existing public signatures for request lookup and for explicit rejection of replacement
 * reasons that have not yet been migrated to a lifecycle command.</p>
 */
@Service
public class CardReplacementV2Service {
    private static final EnumSet<ReplacementReason> PHYSICAL_ONLY_REASONS = EnumSet.of(
            ReplacementReason.DAMAGED, ReplacementReason.CHIP_FAILURE,
            ReplacementReason.MAGSTRIPE_FAILURE);
    private static final EnumSet<ReplacementReason> SECURITY_REPLACEMENT_REASONS = EnumSet.of(
            ReplacementReason.LOST, ReplacementReason.STOLEN, ReplacementReason.COMPROMISED);

    private final CardV2Repository cardRepository;
    private final CardReplacementRequestV2Repository requestRepository;
    private final ProductReplacementPolicyV2Repository policyRepository;
    private final InstitutionV2Repository institutionRepository;
    private final CardProductionRequestV2Repository productionRepository;
    private final CardEventSink eventSink;
    private final CorrelationIdResolver correlationIdResolver;

    /** Keeps the wiring signature stable while legacy collaborators are removed from callers. */
    public CardReplacementV2Service(CardV2Repository cardRepository,
            CardReplacementRequestV2Repository requestRepository,
            ProductReplacementPolicyV2Repository policyRepository,
            ProductVersionV2Repository versionRepository,
            InstitutionV2Repository institutionRepository,
            PanAllocationV2Repository panAllocationRepository,
            BinRangeV2Repository binRangeRepository,
            BinConfigurationV2Repository binRepository,
            AddressReferenceV2Repository addressRepository,
            CardAccountLinkV2Repository accountLinkRepository,
            ProductPersonalisationPolicyV2Repository personalisationPolicyRepository,
            CardVendorV2Repository vendorRepository,
            PersonalisationBatchConfigurationV2Repository batchRepository,
            CardProductionRequestV2Repository productionRepository,
            CardEventV2Repository eventRepository,
            CardControlProfileInitializerV2 controlInitializer,
            AccountingCardLimitClient accountingClient,
            CardHotlistV2Service hotlistService,
            CmsCardSecurityService cardSecurityService,
            PinLifecycleV2Service pinLifecycleService,
            DatabaseUuidV7Generator uuidGenerator,
            CardEventSink eventSink,
            CorrelationIdResolver correlationIdResolver) {
        this.cardRepository = cardRepository;
        this.requestRepository = requestRepository;
        this.policyRepository = policyRepository;
        this.institutionRepository = institutionRepository;
        this.productionRepository = productionRepository;
        this.eventSink = eventSink;
        this.correlationIdResolver = correlationIdResolver;
    }

    @Transactional
    public CardReplacementV2Response create(UUID institutionId, UUID sourceCardId,
            CreateCardReplacementV2Request input, CardHotlistCallerType callerType,
            String idempotencyKey, String requestId, String actorId) {
        UUID domainIdempotency = deterministicId(institutionId, "REPLACEMENT:" + idempotencyKey);
        CardReplacementRequestV2 existing = requestRepository
                .findByInstitutionIdAndIdempotencyKey(institutionId, domainIdempotency)
                .orElse(null);
        if (existing != null) {
            CardReplacementV2Response response = response(existing);
            eventSink.onReplacementRequested(response, correlationIdResolver.resolve(requestId));
            return response;
        }

        actorId.trim();
        correlationIdResolver.resolve(requestId);
        CardV2 source = cardRepository.findForUpdate(institutionId, sourceCardId)
                .orElseThrow(() -> new ResourceNotFoundException("Card", sourceCardId.toString()));
        if (source.getCardStatus() == CardStatusV2.CLOSED) {
            throw new BusinessRuleViolationException("A closed card cannot be replaced");
        }
        if (requestRepository.existsByInstitutionIdAndSourceCardId(institutionId, sourceCardId)) {
            throw new BusinessRuleViolationException(
                    "A replacement request already exists for the source card");
        }
        validateReasonForCard(source, input.getReplacementReason());
        institutionRepository.findById(institutionId)
                .orElseThrow(() -> new ResourceNotFoundException(
                        "Institution", institutionId.toString()));
        ProductReplacementPolicyV2 policy = policyRepository
                .findByInstitutionIdAndProductVersionIdAndReplacementReasonAndPolicyStatus(
                        institutionId, source.getProductVersionId(), input.getReplacementReason(),
                        ProductPolicyStatus.ACTIVE)
                .orElseThrow(() -> new BusinessRuleViolationException(
                        "No ACTIVE replacement policy exists for the requested reason"));
        if (!policy.isReplacementAllowed()) {
            throw new BusinessRuleViolationException(
                    "Replacement is disabled by the active product policy");
        }
        ReplacementRequestChannel requestChannel =
                ReplacementRequestChannel.valueOf(callerType.name());
        if (!policy.getPermittedChannels().contains(requestChannel)) {
            throw new BusinessRuleViolationException(
                    "Replacement is not permitted through the authenticated "
                            + requestChannel + " channel");
        }
        CardReplacementPanStrategy resolved = resolvePanStrategy(policy, input);
        if (SECURITY_REPLACEMENT_REASONS.contains(input.getReplacementReason())
                && resolved != CardReplacementPanStrategy.GENERATE_NEW_PAN) {
            throw new BusinessRuleViolationException(
                    input.getReplacementReason() + " replacement must generate a new PAN");
        }
        throw lifecycleCommandPending("create");
    }

    @Transactional(readOnly = true)
    public CardReplacementV2Response get(UUID institutionId, UUID sourceCardId,
            UUID replacementRequestId) {
        CardReplacementRequestV2 request = requestRepository
                .findByReplacementRequestIdAndInstitutionId(replacementRequestId, institutionId)
                .orElseThrow(() -> new ResourceNotFoundException(
                        "Card replacement request", replacementRequestId.toString()));
        requireSourceCard(request, sourceCardId);
        return response(request);
    }

    @Transactional
    public CardReplacementV2Response approve(UUID institutionId, UUID sourceCardId,
            UUID replacementRequestId, ReplacementDecisionV2Request input,
            String requestId, String actorId) {
        throw lifecycleCommandPending("approve");
    }

    @Transactional
    public CardReplacementV2Response reject(UUID institutionId, UUID sourceCardId,
            UUID replacementRequestId, ReplacementDecisionV2Request input,
            String requestId, String actorId) {
        throw lifecycleCommandPending("reject");
    }

    @Transactional
    public CardReplacementV2Response complete(UUID institutionId, UUID sourceCardId,
            UUID replacementRequestId, String idempotencyKey,
            String requestId, String actorId) {
        throw lifecycleCommandPending("complete");
    }

    private static void requireSourceCard(CardReplacementRequestV2 request, UUID sourceCardId) {
        if (!sourceCardId.equals(request.getSourceCardId())) {
            throw new ResourceNotFoundException(
                    "Card replacement request", request.getReplacementRequestId().toString());
        }
    }

    private CardReplacementV2Response response(CardReplacementRequestV2 request) {
        CardStatusV2 sourceStatus = cardRepository
                .findByCardIdAndInstitutionId(request.getSourceCardId(), request.getInstitutionId())
                .map(CardV2::getCardStatus).orElse(null);
        CardStatusV2 replacementStatus = request.getReplacementCardId() == null ? null
                : cardRepository.findByCardIdAndInstitutionId(
                        request.getReplacementCardId(), request.getInstitutionId())
                        .map(CardV2::getCardStatus).orElse(null);
        UUID productionId = request.getReplacementCardId() == null ? null
                : productionRepository.findByInstitutionIdAndCardId(
                        request.getInstitutionId(), request.getReplacementCardId())
                        .map(CardProductionRequestV2::getProductionRequestId).orElse(null);
        return CardReplacementV2Response.builder()
                .replacementRequestId(request.getReplacementRequestId())
                .institutionId(request.getInstitutionId())
                .sourceCardId(request.getSourceCardId())
                .replacementCardId(request.getReplacementCardId())
                .replacementReason(request.getReplacementReason())
                .policyPanStrategy(request.getPolicyPanStrategy())
                .requestedPanStrategy(request.getRequestedPanStrategy())
                .resolvedPanStrategy(request.getResolvedPanStrategy())
                .requestStatus(request.getRequestStatus())
                .sourceCardStatus(sourceStatus)
                .replacementCardStatus(replacementStatus)
                .productionRequestId(productionId)
                .requestedAt(request.getRequestedAt())
                .completedAt(request.getCompletedAt())
                .build();
    }

    private static CardReplacementPanStrategy resolvePanStrategy(
            ProductReplacementPolicyV2 policy, CreateCardReplacementV2Request input) {
        CardReplacementPanStrategy requested = input.getRequestedPanStrategy();
        CardReplacementPanStrategy deterministic = switch (policy.getPanStrategy()) {
            case PRESERVE_EXISTING_PAN -> CardReplacementPanStrategy.PRESERVE_EXISTING_PAN;
            case GENERATE_NEW_PAN -> CardReplacementPanStrategy.GENERATE_NEW_PAN;
            case RISK_BASED, MANUAL_DECISION -> null;
        };
        if (deterministic != null) {
            if (requested != null && requested != deterministic) {
                throw new BusinessRuleViolationException(
                        "requestedPanStrategy conflicts with the active replacement policy");
            }
            return deterministic;
        }
        if (requested == null || input.getOverrideReason() == null
                || input.getOverrideReason().isBlank()) {
            throw new BusinessRuleViolationException(
                    "Risk/manual replacement requires requestedPanStrategy and overrideReason");
        }
        return requested;
    }

    private static void validateReasonForCard(CardV2 source, ReplacementReason reason) {
        if (source.getCardForm() == CardFormV2.VIRTUAL
                && PHYSICAL_ONLY_REASONS.contains(reason)) {
            throw new BusinessRuleViolationException(
                    reason + " is not applicable to a virtual-only card");
        }
    }

    private static UUID deterministicId(UUID institutionId, String value) {
        return UUID.nameUUIDFromBytes((institutionId + ":" + value)
                .getBytes(StandardCharsets.UTF_8));
    }

    private static BusinessRuleViolationException lifecycleCommandPending(String operation) {
        return new BusinessRuleViolationException(
                "V2 lifecycle command implementation pending: " + operation);
    }
}
