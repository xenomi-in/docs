package com.xenomi.cms.core.v2.bin.dto;

import java.time.LocalDate;
import java.util.LinkedHashMap;
import java.util.Map;

import com.fasterxml.jackson.annotation.JsonAnySetter;
import com.fasterxml.jackson.annotation.JsonIgnore;
import com.xenomi.cms.core.v2.bin.model.BinStatus;
import com.xenomi.cms.core.v2.bin.model.CardTechnology;
import com.xenomi.cms.core.v2.bin.model.PinMethod;

import io.swagger.v3.oas.annotations.media.Schema;
import jakarta.validation.constraints.Max;
import jakarta.validation.constraints.Min;
import jakarta.validation.constraints.NotBlank;
import jakarta.validation.constraints.NotNull;
import jakarta.validation.constraints.Pattern;
import jakarta.validation.constraints.PositiveOrZero;
import jakarta.validation.constraints.Size;
import lombok.AllArgsConstructor;
import lombok.Builder;
import lombok.Data;
import lombok.NoArgsConstructor;

@Data
@Builder
@NoArgsConstructor
@AllArgsConstructor
@Schema(description = "Complete mutable representation used to replace a BIN configuration. "
        + "Identity, network classification and audit fields are immutable and must not be supplied.")
public class UpdateBinConfigurationV2Request {

    @NotNull
    @Schema(description = "Card technology applied to cards issued under the BIN", requiredMode = Schema.RequiredMode.REQUIRED)
    private CardTechnology cardTechnology;

    @NotNull
    @Schema(description = "PIN verification method configured for the BIN", requiredMode = Schema.RequiredMode.REQUIRED)
    private PinMethod pinMethod;

    @NotNull
    @Min(4)
    @Max(12)
    @Schema(description = "PIN length in digits", minimum = "4", maximum = "12", requiredMode = Schema.RequiredMode.REQUIRED)
    private Integer pinLength;

    @NotBlank
    @Pattern(regexp = "[A-Za-z]{3}")
    @Schema(description = "ISO 3166-1 alpha-3 issuing country code", example = "IND", requiredMode = Schema.RequiredMode.REQUIRED)
    private String issuingCountryCode;

    @NotBlank
    @Pattern(regexp = "[A-Za-z]{3}")
    @Schema(description = "ISO 4217 three-letter default currency code", example = "INR", requiredMode = Schema.RequiredMode.REQUIRED)
    private String defaultCurrencyCode;

    @NotNull
    private Boolean defaultChipEnabled;
    @NotNull
    private Boolean defaultDomesticContactEnabled;
    @NotNull
    private Boolean defaultDomesticContactlessEnabled;
    @NotNull
    private Boolean defaultDomesticEcommerceEnabled;
    @NotNull
    private Boolean defaultInternationalEnabled;

    @NotNull
    @Schema(description = "Current BIN lifecycle status", requiredMode = Schema.RequiredMode.REQUIRED)
    private BinStatus binStatus;

    @NotNull
    @Schema(description = "Date from which this configuration is effective", requiredMode = Schema.RequiredMode.REQUIRED)
    private LocalDate effectiveFrom;

    @Schema(description = "Optional end date; null keeps the configuration open-ended")
    private LocalDate effectiveTo;

    @Size(max = 50)
    private String networkProgramIdentifier;

    @Size(max = 100)
    private String networkCertificationReference;

    @Pattern(regexp = "[D=]")
    @Schema(description = "Optional Track 2 field separator; null clears the value", allowableValues = {"D", "="})
    private String track2Separator;

    @Size(max = 20)
    @Schema(description = "Optional PIN decimalization table; null clears the value", maxLength = 20)
    private String decimalizationTable;

    @Min(0)
    @Max(9)
    @Schema(description = "Optional PIN verification key index; null clears the value", minimum = "0", maximum = "9")
    private Integer pvki;

    @NotNull
    @PositiveOrZero
    @Schema(description = "Optimistic-lock version returned by the latest BIN GET response", requiredMode = Schema.RequiredMode.REQUIRED)
    private Long version;

    @JsonIgnore
    @Schema(hidden = true)
    @Builder.Default
    private Map<String, Object> unsupportedFields = new LinkedHashMap<>();

    @JsonAnySetter
    public void captureUnsupportedField(String fieldName, Object value) {
        if (unsupportedFields == null) {
            unsupportedFields = new LinkedHashMap<>();
        }
        unsupportedFields.put(fieldName, value);
    }

    @JsonIgnore
    public Map<String, Object> getUnsupportedFields() {
        return unsupportedFields;
    }
}
