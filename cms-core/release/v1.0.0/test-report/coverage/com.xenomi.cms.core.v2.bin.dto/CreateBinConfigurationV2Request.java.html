package com.xenomi.cms.core.v2.bin.dto;

import java.time.LocalDate;

import com.xenomi.cms.core.v2.bin.model.BinStatus;
import com.xenomi.cms.core.v2.bin.model.BinType;
import com.xenomi.cms.core.v2.bin.model.CardNetwork;
import com.xenomi.cms.core.v2.bin.model.CardTechnology;
import com.xenomi.cms.core.v2.bin.model.PinMethod;

import io.swagger.v3.oas.annotations.media.Schema;
import jakarta.validation.constraints.Max;
import jakarta.validation.constraints.Min;
import jakarta.validation.constraints.NotBlank;
import jakarta.validation.constraints.NotNull;
import jakarta.validation.constraints.Pattern;
import jakarta.validation.constraints.Size;
import jakarta.validation.constraints.AssertTrue;
import lombok.AllArgsConstructor;
import lombok.Builder;
import lombok.Data;
import lombok.NoArgsConstructor;

@Data
@Builder
@NoArgsConstructor
@AllArgsConstructor
@Schema(description = "CMS V2 BIN configuration creation request")
public class CreateBinConfigurationV2Request {

    @NotBlank
    @Pattern(regexp = "(?:[0-9]{6}|[0-9]{8})", message = "binNumber must contain exactly 6 or 8 digits")
    private String binNumber;

    @Schema(description = "BIN length; only ISO/IIN lengths 6 and 8 are supported",
            example = "6", minimum = "6", maximum = "8")
    private Integer binLength;

    @NotNull
    @Schema(description = "Total numeric PAN length, including BIN and check digit; only 16 or 19 is supported",
            example = "16", allowableValues = {"16", "19"}, requiredMode = Schema.RequiredMode.REQUIRED)
    private Integer panLength;

    @AssertTrue(message = "panLength must be 16 or 19")
    @Schema(hidden = true)
    public boolean isPanLengthValid() {
        return panLength != null && (panLength == 16 || panLength == 19);
    }

    @NotNull
    private CardNetwork cardNetwork;

    @NotNull
    private BinType binType;

    @NotBlank
    @Size(max = 20)
    @Schema(description = "Scheme/issuer-defined product class", example = "CLASSIC")
    private String cardProductClass;

    private CardTechnology cardTechnology;

    private PinMethod pinMethod;

    @Min(4)
    @Max(12)
    private Integer pinLength;

    @Pattern(regexp = "[A-Za-z]{3}")
    private String issuingCountryCode;

    @Pattern(regexp = "[A-Za-z]{3}")
    private String defaultCurrencyCode;

    private Boolean defaultChipEnabled;
    private Boolean defaultDomesticContactEnabled;
    private Boolean defaultDomesticContactlessEnabled;
    private Boolean defaultDomesticEcommerceEnabled;
    private Boolean defaultInternationalEnabled;
    private BinStatus binStatus;

    @NotNull
    private LocalDate effectiveFrom;

    private LocalDate effectiveTo;

    @Size(max = 50)
    @Schema(description = "Network-assigned program identifier, for example the NPCI program record reference")
    private String networkProgramIdentifier;

    @Size(max = 100)
    @Schema(description = "Network certification or onboarding record reference")
    private String networkCertificationReference;

    @Pattern(regexp = "[D=]")
    private String track2Separator;

    @Size(max = 20)
    private String decimalizationTable;

    @Min(0)
    @Max(9)
    private Integer pvki;
}
