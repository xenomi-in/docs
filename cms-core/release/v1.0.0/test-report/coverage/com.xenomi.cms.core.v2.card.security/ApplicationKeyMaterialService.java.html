package com.xenomi.cms.core.v2.card.security;

import java.security.MessageDigest;
import java.security.NoSuchAlgorithmException;
import java.security.SecureRandom;
import java.nio.charset.StandardCharsets;
import java.time.Duration;
import java.time.OffsetDateTime;
import java.time.ZoneOffset;
import java.util.Arrays;
import java.util.HexFormat;
import java.util.Map;
import java.util.UUID;
import java.util.concurrent.ConcurrentHashMap;

import com.xenomi.cms.core.v2.hsm.model.HsmKeyType;
import com.xenomi.cms.core.v2.institution.entity.HsmKeyReferenceV2;
import jakarta.annotation.PreDestroy;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.stereotype.Service;

/** Resolves short-lived clear application keys from HSM-wrapped database material. */
@Service
class ApplicationKeyMaterialService {
    private static final int KEY_BYTES = 32;

    private final ApplicationKeyMaterialV2Repository repository;
    private final HsmApplicationKeyWrapper wrapper;
    private final JdbcTemplate jdbcTemplate;
    private final SecureRandom random = new SecureRandom();
    private final Map<UUID, CacheEntry> cache = new ConcurrentHashMap<>();
    private final long cacheTtlNanos;

    ApplicationKeyMaterialService(ApplicationKeyMaterialV2Repository repository,
            HsmApplicationKeyWrapper wrapper,
            JdbcTemplate jdbcTemplate,
            @Value("${cms.card-security.application-key-cache-ttl:PT5M}") Duration cacheTtl) {
        this.repository = repository;
        this.wrapper = wrapper;
        this.jdbcTemplate = jdbcTemplate;
        if (cacheTtl.isZero() || cacheTtl.isNegative()) {
            throw new IllegalArgumentException("Application key cache TTL must be positive");
        }
        this.cacheTtlNanos = cacheTtl.toNanos();
    }

    byte[] resolve(HsmKeyReferenceV2 hsmKey, ApplicationKeyUse use) {
        validate(hsmKey, use);
        long now = System.nanoTime();
        CacheEntry cached = cache.get(hsmKey.getKeyId());
        if (cached != null && cached.expiresAtNanos > now) return cached.copy();
        synchronized (cache) {
            cached = cache.get(hsmKey.getKeyId());
            if (cached != null && cached.expiresAtNanos > System.nanoTime()) return cached.copy();
            if (cached != null) cached.destroy();
            byte[] clearKey = loadOrCreate(hsmKey, use);
            cache.put(hsmKey.getKeyId(), new CacheEntry(clearKey,
                    System.nanoTime() + cacheTtlNanos));
            byte[] result = clearKey.clone();
            Arrays.fill(clearKey, (byte) 0);
            return result;
        }
    }

    private byte[] loadOrCreate(HsmKeyReferenceV2 hsmKey, ApplicationKeyUse use) {
        return repository
                .findByHsmKeyIdAndInstitutionIdAndKeyUse(
                        hsmKey.getKeyId(), hsmKey.getInstitutionId(), use)
                .map(material -> unwrap(material, hsmKey, use))
                .orElseGet(() -> create(hsmKey, use));
    }

    private byte[] unwrap(ApplicationKeyMaterialV2 material,
            HsmKeyReferenceV2 hsmKey, ApplicationKeyUse use) {
        if (!use.algorithm().equals(material.getAlgorithm())) {
            throw new CardSecurityException("Unsupported application key algorithm");
        }
        String clearHex = wrapper.unwrap(material.getWrappedKeyMaterial(), hsmKey);
        byte[] clearKey;
        try {
            clearKey = HexFormat.of().parseHex(clearHex);
        } catch (IllegalArgumentException exception) {
            throw new CardSecurityException("HSM returned invalid wrapped key material", exception);
        }
        if (clearKey.length != KEY_BYTES
                || !MessageDigest.isEqual(
                        checksum(clearKey).getBytes(StandardCharsets.US_ASCII),
                        material.getKeyChecksum().getBytes(StandardCharsets.US_ASCII))) {
            Arrays.fill(clearKey, (byte) 0);
            throw new CardSecurityException("Wrapped application key integrity check failed");
        }
        return clearKey;
    }

    private byte[] create(HsmKeyReferenceV2 hsmKey, ApplicationKeyUse use) {
        byte[] clearKey = new byte[KEY_BYTES];
        random.nextBytes(clearKey);
        try {
            String clearHex = HexFormat.of().withUpperCase().formatHex(clearKey);
            String wrapped = wrapper.wrap(clearHex, hsmKey);
            int inserted = jdbcTemplate.update("""
                    INSERT INTO cms_v2.application_key_material
                        (hsm_key_id, institution_id, key_use, algorithm,
                         wrapped_key_material, key_checksum, created_at)
                    VALUES (?, ?, ?, ?, ?, ?, ?)
                    ON CONFLICT (hsm_key_id) DO NOTHING
                    """, hsmKey.getKeyId(), hsmKey.getInstitutionId(), use.name(),
                    use.algorithm(), wrapped, checksum(clearKey),
                    OffsetDateTime.now(ZoneOffset.UTC));
            if (inserted == 1) {
                byte[] result = clearKey;
                clearKey = null;
                return result;
            }
            ApplicationKeyMaterialV2 winner = repository
                    .findByHsmKeyIdAndInstitutionIdAndKeyUse(
                            hsmKey.getKeyId(), hsmKey.getInstitutionId(), use)
                    .orElseThrow(() -> new CardSecurityException(
                            "Application key material could not be provisioned"));
            return unwrap(winner, hsmKey, use);
        } finally {
            if (clearKey != null) Arrays.fill(clearKey, (byte) 0);
        }
    }

    private static void validate(HsmKeyReferenceV2 key, ApplicationKeyUse use) {
        HsmKeyType expected = use == ApplicationKeyUse.CARD_DATA_ENCRYPTION
                ? HsmKeyType.DEK : HsmKeyType.HMAC;
        if (key.getKeyType() != expected) {
            throw new CardSecurityException(use + " requires an institution " + expected + " key");
        }
    }

    private static String checksum(byte[] value) {
        try {
            return HexFormat.of().withUpperCase().formatHex(
                    MessageDigest.getInstance("SHA-256").digest(value));
        } catch (NoSuchAlgorithmException impossible) {
            throw new IllegalStateException(impossible);
        }
    }

    @PreDestroy
    void destroy() {
        cache.values().forEach(CacheEntry::destroy);
        cache.clear();
    }

    private static final class CacheEntry {
        private final byte[] key;
        private final long expiresAtNanos;

        private CacheEntry(byte[] key, long expiresAtNanos) {
            this.key = key.clone();
            this.expiresAtNanos = expiresAtNanos;
        }

        private byte[] copy() { return key.clone(); }
        private void destroy() { Arrays.fill(key, (byte) 0); }
    }
}
