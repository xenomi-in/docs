/** Keeps clear PAN, CVV and Track 2 transient and resolves all keys from approved assignments. */
@Service
public class CmsCardSecurityService {
    private static final DateTimeFormatter EXPIRY_YYMM = DateTimeFormatter.ofPattern("yyMM");
    private static final int MAX_RANDOM_ATTEMPTS = 200;

    private final InstitutionDataKeyAssignmentV2Repository institutionAssignmentRepository;
    private final BinHsmKeyAssignmentV2Repository binAssignmentRepository;
    private final HsmKeyReferenceV2Repository keyRepository;
    private final PanAllocationV2Repository panAllocationRepository;
    private final CardHsmOperations hsm;
    private final SecureRandom random = new SecureRandom();

    public CmsCardSecurityService(
            InstitutionDataKeyAssignmentV2Repository institutionAssignmentRepository,
            BinHsmKeyAssignmentV2Repository binAssignmentRepository,
            HsmKeyReferenceV2Repository keyRepository,
            PanAllocationV2Repository panAllocationRepository,
            CardHsmOperations hsm) {
        this.institutionAssignmentRepository = institutionAssignmentRepository;
        this.binAssignmentRepository = binAssignmentRepository;
        this.keyRepository = keyRepository;
        this.panAllocationRepository = panAllocationRepository;
        this.hsm = hsm;
    }

    public CardSecurityMaterial generate(UUID institutionId, BinRangeV2 range,
            LocalDate expiryDate, String serviceCode, String track2Separator) {
        KeySet keys = activeKeys(institutionId, range.getBinId());
        PanCandidate candidate = allocatePan(institutionId, range, keys.hmacKey());
        recordAllocation(range, candidate);
        return protect(candidate.pan(), candidate.sequenceValue(), expiryDate, serviceCode,
                track2Separator, keys);
    }

    /** Metadata validation only; no HSM command. */
    public void validateIssuanceKeys(UUID institutionId, UUID binId) {
        activeKeys(institutionId, binId);
    }

    /** Candidate was durably reserved before this call. Never mutate allocation counters here. */
    public CardSecurityMaterial generateReserved(UUID institutionId, BinRangeV2 range, long offset,
            LocalDate expiryDate, String serviceCode) {
        if (org.springframework.transaction.support.TransactionSynchronizationManager.isActualTransactionActive()) {
            throw new IllegalStateException("Credential preparation must run outside a database transaction");
        }
        var value = new BigInteger(range.getRangeStart()).add(BigInteger.valueOf(offset));
        String pan = leftPad(value.toString(), range.getRangeStart().length());
        if (offset < 0 || value.compareTo(new BigInteger(range.getRangeEnd())) > 0 || !LuhnAlgorithm.validate(pan)) {
            throw new CardSecurityException("Invalid reserved PAN candidate");
        }
        Long sequenceValue = range.getAllocationStrategy() == AllocationStrategy.SEQUENTIAL ? offset : null;
        return protect(pan, sequenceValue, expiryDate, serviceCode, "D", activeKeys(institutionId, range.getBinId()));
    }

    public CardSecurityMaterial reissueSamePan(UUID institutionId, CardV2 source,
            PanAllocationV2 allocation, LocalDate expiryDate, String serviceCode,
            String track2Separator) {
        CardSecurityMaterial material = prepareSamePan(institutionId, source, allocation, expiryDate, serviceCode, track2Separator);
        allocation.setPanHmac(material.panHmac());
        allocation.setPanFingerprint(material.panFingerprint());
        allocation.setSecurePanReference(material.secureDataReference());
        panAllocationRepository.save(allocation);
        return material;
    }

    /** Prepares successor credentials without changing the PAN family; caller owns durable finalization. */
    public CardSecurityMaterial prepareSamePan(UUID institutionId, CardV2 source,
            PanAllocationV2 allocation, LocalDate expiryDate, String serviceCode, String track2Separator) {
        HsmKeyReferenceV2 sourceEncryptionKey = historicalKey(institutionId,
                source.getDataEncryptionKeyId(), HsmKeyType.DEK);
        String pan = hsm.decrypt(source.getPanEncrypted(), sourceEncryptionKey);
        if (pan == null || !pan.matches("[0-9]{12,19}") || !LuhnAlgorithm.validate(pan)) {
            throw new CardSecurityException("The source PAN could not be recovered securely");
        }
        HsmKeyReferenceV2 sourceHmacKey = historicalKey(institutionId,
                source.getPanHmacKeyId(), HsmKeyType.HMAC);
        String recoveredHmac = source.getPanEncrypted().startsWith("HSM1:")
                ? hsm.generateLegacyPanHmac(pan, sourceHmacKey)
                : hsm.generatePanHmac(pan, sourceHmacKey);
        if (!MessageDigest.isEqual(recoveredHmac.getBytes(StandardCharsets.UTF_8),
                allocation.getPanHmac().getBytes(StandardCharsets.UTF_8))) {
            throw new CardSecurityException("The source PAN identity check failed");
        }
        KeySet keys = activeKeys(institutionId, source.getBinId());
        CardSecurityMaterial material = protect(pan, allocation.getSequenceValue(),
                expiryDate, serviceCode, track2Separator, keys);
        return material;
    }

    /**
     * Exposes a recovered PAN only to a synchronous secure operation. The callback must not
     * persist, return, or log the clear PAN.
     */
    public <T> T withDecryptedPan(UUID institutionId, CardV2 card, Function<String, T> operation) {
        HsmKeyReferenceV2 encryptionKey = historicalKey(
                institutionId, card.getDataEncryptionKeyId(), HsmKeyType.DEK);
        String pan = hsm.decrypt(card.getPanEncrypted(), encryptionKey);
        if (pan == null || !pan.matches("[0-9]{12,19}") || !LuhnAlgorithm.validate(pan)) {
            throw new CardSecurityException("The card PAN could not be recovered securely");
        }
        return operation.apply(pan);
    }

    private CardSecurityMaterial protect(String pan, Long sequenceValue, LocalDate expiryDate,
            String serviceCode, String separator, KeySet keys) {
        String expiry = expiryDate.format(EXPIRY_YYMM);
        CardHsmOperations.CvvPair cvv = hsm.generateCvv(
                pan, expiry, requireServiceCode(serviceCode), keys.cvvKey());
        String track2 = pan + normalizeSeparator(separator) + expiry
                + requireServiceCode(serviceCode) + "0000" + cvv.cvv1() + "0";
        String hmacValue = hsm.generatePanHmac(pan, keys.hmacKey());
        return new CardSecurityMaterial(mask(pan), pan.substring(pan.length() - 4),
                fingerprint(hmacValue), hmacValue, sequenceValue,
                hsm.encrypt(pan, keys.encryptionKey()),
                hsm.encrypt(cvv.cvv1(), keys.encryptionKey()),
                hsm.encrypt(cvv.cvv2(), keys.encryptionKey()),
                hsm.encrypt(track2, keys.encryptionKey()),
                keys.encryptionKey().getKeyId(),
                keys.hmacKey().getKeyId(),
                keys.cvvKey().getKeyId(), "cms-secure-card-data:" + UUID.randomUUID());
    }

    private PanCandidate allocatePan(UUID institutionId, BinRangeV2 range,
            HsmKeyReferenceV2 hmacKey) {
        if (range.getIssuedPanCount() == null) range.setIssuedPanCount(0L);
        if (range.getNextSequenceValue() == null) range.setNextSequenceValue(0L);
        if (range.getUsableCapacity() == null
                || range.getIssuedPanCount() >= range.getUsableCapacity()) {
            throw new CardSecurityException("BIN range has no remaining PAN capacity");
        }
        BigInteger start = new BigInteger(range.getRangeStart());
        BigInteger end = new BigInteger(range.getRangeEnd());
        if (start.compareTo(end) > 0) throw new CardSecurityException("Invalid BIN range bounds");
        if (range.getAllocationStrategy() == AllocationStrategy.SEQUENTIAL) {
            long sequence = range.getNextSequenceValue();
            for (long offset = sequence; ; offset++) {
                BigInteger value = start.add(BigInteger.valueOf(offset));
                if (value.compareTo(end) > 0) throw new CardSecurityException("BIN range is exhausted");
                String pan = leftPad(value.toString(), range.getRangeStart().length());
                if (LuhnAlgorithm.validate(pan)
                        && unique(institutionId, pan, hmacKey)) return new PanCandidate(pan, offset);
            }
        }
        BigInteger width = end.subtract(start).add(BigInteger.ONE);
        for (int attempt = 0; attempt < MAX_RANDOM_ATTEMPTS; attempt++) {
            BigInteger offset;
            do offset = new BigInteger(width.bitLength(), random); while (offset.compareTo(width) >= 0);
            String pan = leftPad(start.add(offset).toString(), range.getRangeStart().length());
            if (LuhnAlgorithm.validate(pan) && unique(institutionId, pan, hmacKey)) {
                return new PanCandidate(pan, null);
            }
        }
        throw new CardSecurityException("No available PAN could be allocated from the BIN range");
    }

    private static void recordAllocation(BinRangeV2 range, PanCandidate candidate) {
        range.setIssuedPanCount(Math.addExact(range.getIssuedPanCount(), 1L));
        if (candidate.sequenceValue() != null) {
            range.setNextSequenceValue(Math.addExact(candidate.sequenceValue(), 1L));
        }
        if (range.getIssuedPanCount() >= range.getUsableCapacity()) {
            range.setRangeStatus(BinRangeStatus.EXHAUSTED);
        }
    }

    private boolean unique(UUID institutionId, String pan, HsmKeyReferenceV2 hmacKey) {
        String value = hsm.generatePanHmac(pan, hmacKey);
        return !panAllocationRepository.existsByInstitutionIdAndPanHmac(institutionId, value);
    }

    private static String leftPad(String value, int length) {
        return "0".repeat(Math.max(0, length - value.length())) + value;
    }

    private KeySet activeKeys(UUID institutionId, UUID binId) {
        OffsetDateTime now = OffsetDateTime.now(ZoneOffset.UTC);
        HsmKeyReferenceV2 encryption = institutionKey(institutionId,
                KeyPurpose.PAN_DATA_ENCRYPTION, HsmKeyType.DEK, now);
        HsmKeyReferenceV2 hmacKey = institutionKey(institutionId,
                KeyPurpose.PAN_HMAC, HsmKeyType.HMAC, now);
        if (encryption.getKeyId().equals(hmacKey.getKeyId())) {
            throw new CardSecurityException("PAN encryption and PAN HMAC must use different keys");
        }
        List<BinHsmKeyAssignmentV2> assignments = binAssignmentRepository.findEffective(
                institutionId, binId, KeyPurpose.CVV_GENERATION,
                BinHsmKeyAssignmentStatus.ACTIVE, now);
        if (assignments.size() != 1) {
            throw new CardSecurityException("Exactly one active BIN CVV key assignment is required");
        }
        HsmKeyReferenceV2 cvv = requireUsableKey(
                institutionId, assignments.get(0).getHsmKeyId(), HsmKeyScope.BIN, HsmKeyType.CVK);
        if (!binId.equals(cvv.getBinId())) {
            throw new CardSecurityException("The CVV key is not scoped to the selected BIN");
        }
        return new KeySet(encryption, hmacKey, cvv);
    }

    private HsmKeyReferenceV2 institutionKey(UUID institutionId, KeyPurpose purpose,
            HsmKeyType type, OffsetDateTime now) {
        List<InstitutionDataKeyAssignmentV2> assignments = institutionAssignmentRepository
                .findEffective(institutionId, purpose, DataKeyAssignmentStatus.ACTIVE, now);
        if (assignments.size() != 1) {
            throw new CardSecurityException("Exactly one active " + purpose + " assignment is required");
        }
        return requireUsableKey(institutionId, assignments.get(0).getHsmKeyId(),
                HsmKeyScope.INSTITUTION, type);
    }

    private HsmKeyReferenceV2 requireUsableKey(UUID institutionId, UUID keyId,
            HsmKeyScope scope, HsmKeyType type) {
        LocalDate today = LocalDate.now(ZoneOffset.UTC);
        return keyRepository.findByKeyIdAndInstitutionId(keyId, institutionId)
                .filter(key -> key.getStatus() == HsmKeyStatus.ACTIVE)
                .filter(key -> key.getKeyScope() == scope && key.getKeyType() == type)
                .filter(key -> !today.isBefore(key.getEffectiveFrom()) && today.isBefore(key.getExpiryDate()))
                .filter(key -> key.getEncryptedKeyBlock() != null && !key.getEncryptedKeyBlock().isBlank())
                .orElseThrow(() -> new CardSecurityException(
                        "Assigned " + type + " key is not active, effective, or HSM-protected"));
    }

    public String hashForLookup(UUID institutionId, UUID keyId, String pan, boolean legacy) {
        HsmKeyReferenceV2 key = historicalKey(institutionId, keyId, HsmKeyType.HMAC);
        return legacy ? hsm.generateLegacyPanHmac(pan, key) : hsm.generatePanHmac(pan, key);
    }

    private HsmKeyReferenceV2 historicalKey(UUID institutionId, UUID keyId, HsmKeyType type) {
        return keyRepository.findByKeyIdAndInstitutionId(keyId, institutionId)
                .filter(key -> Set.of(HsmKeyStatus.ACTIVE, HsmKeyStatus.DECRYPT_ONLY)
                        .contains(key.getStatus()))
                .filter(key -> key.getKeyScope() == HsmKeyScope.INSTITUTION
                        && key.getKeyType() == type)
                .filter(key -> key.getEncryptedKeyBlock() != null
                        && !key.getEncryptedKeyBlock().isBlank())
                .orElseThrow(() -> new CardSecurityException(
                        "The historical " + type + " key is unavailable"));
    }

    private static String requireServiceCode(String value) {
        return value != null && value.matches("[0-9]{3}") ? value : "201";
    }

    private static String normalizeSeparator(String value) {
        return value != null && value.length() == 1 ? value : "D";
    }

    private static String mask(String pan) {
        int visiblePrefix = Math.min(6, pan.length() - 4);
        return pan.substring(0, visiblePrefix)
                + "*".repeat(pan.length() - visiblePrefix - 4)
                + pan.substring(pan.length() - 4);
    }

    private static String fingerprint(String keyedHmac) {
        try {
            return HexFormat.of().formatHex(MessageDigest.getInstance("SHA-256")
                    .digest(keyedHmac.getBytes(StandardCharsets.UTF_8)));
        } catch (NoSuchAlgorithmException impossible) {
            throw new IllegalStateException(impossible);
        }
    }

    private record PanCandidate(String pan, Long sequenceValue) { }
    private record KeySet(HsmKeyReferenceV2 encryptionKey,
            HsmKeyReferenceV2 hmacKey, HsmKeyReferenceV2 cvvKey) { }
}
