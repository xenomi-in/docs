package com.xenomi.cms.core.v2.card.security;

import com.xenomi.cms.core.hms.service.NettyHsmService;
import com.xenomi.cms.core.v2.hsm.model.HsmKeyScheme;
import com.xenomi.cms.core.v2.institution.entity.HsmKeyReferenceV2;
import org.springframework.stereotype.Component;

@Component
class FuturexHsmApplicationKeyWrapper implements HsmApplicationKeyWrapper {
    private final NettyHsmService hsm;

    FuturexHsmApplicationKeyWrapper(NettyHsmService hsm) {
        this.hsm = hsm;
    }

    @Override
    public String wrap(String clearKeyHex, HsmKeyReferenceV2 wrappingKey) {
        return require(hsm.encryptDataWithKeyReference(clearKeyHex,
                wrappingKey.getEncryptedKeyBlock(), algorithm(wrappingKey)), "key wrapping");
    }

    @Override
    public String unwrap(String wrappedKey, HsmKeyReferenceV2 wrappingKey) {
        return require(hsm.decryptDataWithKeyReference(wrappedKey,
                wrappingKey.getEncryptedKeyBlock(), algorithm(wrappingKey)), "key unwrapping");
    }

    private static String algorithm(HsmKeyReferenceV2 key) {
        if (key.getScheme() == HsmKeyScheme.AES) return "AES";
        if (key.getScheme() == HsmKeyScheme.TR31 && key.getTr31Algorithm() != null
                && key.getTr31Algorithm().toUpperCase().startsWith("A")) return "AES";
        return "TDES";
    }

    private static String require(String value, String operation) {
        if (value == null || value.isBlank()) {
            throw new CardSecurityException("HSM " + operation + " failed");
        }
        return value;
    }
}
