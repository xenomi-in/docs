package com.xenomi.cms.core.v2.card.security;

import java.nio.charset.StandardCharsets;
import java.security.GeneralSecurityException;
import java.security.SecureRandom;
import java.util.Arrays;
import java.util.Base64;

import javax.crypto.Cipher;
import javax.crypto.Mac;
import javax.crypto.spec.GCMParameterSpec;
import javax.crypto.spec.SecretKeySpec;

import com.xenomi.cms.core.hms.service.NettyHsmService;
import com.xenomi.cms.core.v2.institution.entity.HsmKeyReferenceV2;
import org.springframework.stereotype.Component;

@Component
public class FuturexCardHsmOperations implements CardHsmOperations {
    private static final String ENVELOPE_VERSION = "CMS2";
    private static final String ENCRYPTION_ALGORITHM = "A256GCM";
    private static final int GCM_NONCE_BYTES = 12;
    private static final int GCM_TAG_BITS = 128;

    private final NettyHsmService hsm;
    private final ApplicationKeyMaterialService applicationKeys;
    private final SecureRandom random = new SecureRandom();

    public FuturexCardHsmOperations(NettyHsmService hsm,
            ApplicationKeyMaterialService applicationKeys) {
        this.hsm = hsm;
        this.applicationKeys = applicationKeys;
    }

    @Override
    public String encrypt(String clearText, HsmKeyReferenceV2 key) {
        byte[] clearKey = applicationKeys.resolve(key, ApplicationKeyUse.CARD_DATA_ENCRYPTION);
        try {
            byte[] nonce = new byte[GCM_NONCE_BYTES];
            random.nextBytes(nonce);
            Cipher cipher = Cipher.getInstance("AES/GCM/NoPadding");
            cipher.init(Cipher.ENCRYPT_MODE, new SecretKeySpec(clearKey, "AES"),
                    new GCMParameterSpec(GCM_TAG_BITS, nonce));
            cipher.updateAAD(aad(key));
            byte[] encrypted = cipher.doFinal(clearText.getBytes(StandardCharsets.UTF_8));
            return ENVELOPE_VERSION + ":" + ENCRYPTION_ALGORITHM + ":" + key.getKeyId()
                    + ":" + Base64.getUrlEncoder().withoutPadding().encodeToString(nonce)
                    + ":" + Base64.getUrlEncoder().withoutPadding().encodeToString(encrypted);
        } catch (GeneralSecurityException exception) {
            throw new CardSecurityException("Card data encryption failed", exception);
        } finally {
            Arrays.fill(clearKey, (byte) 0);
        }
    }

    @Override
    public String decrypt(String cipherText, HsmKeyReferenceV2 key) {
        if (cipherText != null && cipherText.startsWith("HSM1:")) {
            return require(hsm.decryptDataWithKeyReference(cipherText,
                    key.getEncryptedKeyBlock(), algorithm(key)), "legacy decryption");
        }
        String[] parts = cipherText == null ? new String[0] : cipherText.split(":", 5);
        if (parts.length != 5 || !ENVELOPE_VERSION.equals(parts[0])
                || !ENCRYPTION_ALGORITHM.equals(parts[1])
                || !key.getKeyId().toString().equals(parts[2])) {
            throw new CardSecurityException("Invalid or mismatched card data envelope");
        }
        byte[] clearKey = applicationKeys.resolve(key, ApplicationKeyUse.CARD_DATA_ENCRYPTION);
        try {
            byte[] nonce = Base64.getUrlDecoder().decode(parts[3]);
            byte[] encrypted = Base64.getUrlDecoder().decode(parts[4]);
            if (nonce.length != GCM_NONCE_BYTES) {
                throw new CardSecurityException("Invalid card data envelope nonce");
            }
            Cipher cipher = Cipher.getInstance("AES/GCM/NoPadding");
            cipher.init(Cipher.DECRYPT_MODE, new SecretKeySpec(clearKey, "AES"),
                    new GCMParameterSpec(GCM_TAG_BITS, nonce));
            cipher.updateAAD(aad(key));
            return new String(cipher.doFinal(encrypted), StandardCharsets.UTF_8);
        } catch (IllegalArgumentException | GeneralSecurityException exception) {
            throw new CardSecurityException("Card data decryption or integrity check failed", exception);
        } finally {
            Arrays.fill(clearKey, (byte) 0);
        }
    }

    @Override
    public String generatePanHmac(String pan, HsmKeyReferenceV2 key) {
        byte[] clearKey = applicationKeys.resolve(key, ApplicationKeyUse.PAN_HMAC);
        try {
            Mac mac = Mac.getInstance("HmacSHA256");
            mac.init(new SecretKeySpec(clearKey, "HmacSHA256"));
            return java.util.HexFormat.of().withUpperCase().formatHex(
                    mac.doFinal(pan.getBytes(StandardCharsets.US_ASCII)));
        } catch (GeneralSecurityException exception) {
            throw new CardSecurityException("PAN HMAC generation failed", exception);
        } finally {
            Arrays.fill(clearKey, (byte) 0);
        }
    }

    @Override
    public String generateLegacyPanHmac(String pan, HsmKeyReferenceV2 key) {
        return require(hsm.generateMacWithKeyReference(pan, key.getEncryptedKeyBlock()),
                "legacy PAN HMAC");
    }

    @Override
    public CvvPair generateCvv(String pan, String expiryYymm, String serviceCode,
            HsmKeyReferenceV2 key) {
        String result = require(hsm.generateCvv(pan, expiryYymm, serviceCode,
                key.getEncryptedKeyBlock()), "CVV generation");
        String[] values = result.split(";", -1);
        if (values.length != 2 || !values[0].matches("[0-9]{3,4}")
                || !values[1].matches("[0-9]{3,4}")) {
            throw new CardSecurityException("HSM returned invalid CVV generation metadata");
        }
        return new CvvPair(values[0], values[1]);
    }

    private static String require(String value, String operation) {
        if (value == null || value.isBlank()) {
            throw new CardSecurityException("HSM " + operation + " failed");
        }
        return value;
    }

    private static String algorithm(HsmKeyReferenceV2 key) {
        if (key.getScheme() == com.xenomi.cms.core.v2.hsm.model.HsmKeyScheme.AES) return "AES";
        if (key.getScheme() == com.xenomi.cms.core.v2.hsm.model.HsmKeyScheme.TR31
                && key.getTr31Algorithm() != null
                && key.getTr31Algorithm().toUpperCase().startsWith("A")) return "AES";
        return "TDES";
    }

    private static byte[] aad(HsmKeyReferenceV2 key) {
        return ("CMS2|CARD_DATA|" + key.getInstitutionId() + "|" + key.getKeyId())
                .getBytes(StandardCharsets.US_ASCII);
    }
}
